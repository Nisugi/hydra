1//! The Uberbar: one panel of what a hunter glances at, beside the injury
2//! doll. Labels for the day (experience gained today, the rate an hour,
3//! the last pulse, the room's number, today's silver and bounty points,
4//! the Voln favor) and bars under them (the vitals, the next level, the
5//! mind, the stance, the burden, the profession's resource, a sorcerer's
6//! shadow essence, how far to the experience cap, the next ascension
7//! point). Each row is on or off
8//! on the widget's page.
9//!
10//! The vitals, the mind, the stance, the burden, the next level, the
11//! resources, the favor and the room are the model's. The day's numbers
12//! are the ledger's: [`Facts`] read from it by whoever draws the panel, no
13//! oftener than [`FACTS_EVERY`], so a frame never waits on the database.
14

188impl UberbarLook {
189    /// The switch the page's `key` names.
190    fn switch(&mut self, key: &str) -> Option<&mut bool> {
191        Some(match key {
192            "today" => &mut self.today,
193            "per_hour" => &mut self.per_hour,
194            "pulse" => &mut self.pulse,
195            "room" => &mut self.room,
196            "silver" => &mut self.silver,
197            "bounty" => &mut self.bounty,
198            "favor" => &mut self.favor,
199            "vitals" => &mut self.vitals,
200            "ascension" => &mut self.ascension,
201            "next_level" => &mut self.next_level,
202            "mind" => &mut self.mind,
203            "field_experience" => &mut self.field_experience,
204            "stance" => &mut self.stance,
205            "encumbrance" => &mut self.encumbrance,
206            "resources" => &mut self.resources,
207            "shadow_essence" => &mut self.shadow_essence,
208            "percent_capped" => &mut self.percent_capped,
209            _ => return None,
210        })
211    }
212
213    /// Whether the row `key` names is on.
214    #[must_use]
215    pub fn shows(&self, key: &str) -> bool {
216        self.clone().switch(key).is_some_and(|on| *on)
217    }
218
219    /// The row `key` names set `on`; or back to its own when `on` is none.
220    ///
221    /// # Errors
222    ///
223    /// A key that names no row.
224    pub fn set(&mut self, key: &str, on: Option<bool>) -> Result<(), String> {
225        let shown = on.unwrap_or_else(|| Self::default().shows(key));
226        *self
227            .switch(key)
228            .ok_or_else(|| format!("The Uberbar has no row {key}."))? = shown;
229        Ok(())
230    }
231
232    /// Whether any of the day's numbers is shown, so the ledger is read.
233    #[must_use]
234    pub const fn wants_facts(&self) -> bool {
235        self.today || self.per_hour || self.pulse || self.silver || self.bounty
236    }
237}

239/// The day's numbers, read from the ledger.
240#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
241pub struct Facts {
242    /// Experience and ascension experience gained since midnight.
243    pub today: i64,
244    /// Experience an hour, over today's pulses, by the time spent absorbing.
245    pub per_hour: i64,
246    /// What the last pulse brought, experience and ascension together.
247    pub pulse: i64,
248    /// Today's silver: income less outgo, as the ledger has it.
249    pub silver: i64,
250    /// Today's bounty points.
251    pub bounty: i64,
252}
253
254impl Facts {
255    /// The day's numbers for `who` at `now`, read from `reader`.
256    ///
257    /// # Errors
258    ///
259    /// The ledger cannot be read.
260    pub fn read(reader: &Reader, who: &Who, now: i64) -> Result<Self, Error> {
261        let pulses = reader.pulses(who, hydra_exp::report::day_start(now), now + 1)?;
262        let today = hydra_exp::report::Totals::of(&pulses).map_or(0, |totals| totals.total());
263        let (per_hour, _, _) = hydra_exp::report::rate(&pulses);
264        let pulse = reader
265            .last_pulses(who, 1)?
266            .first()
267            .map_or(0, |last| last.exp_gained + last.axp_gained);
268        let day = hydra_net::eastern::format(hydra_net::eastern::date(now));
269        let sums = reader.transaction_sums(who, &day)?;
270        let total = |picks: fn(TransactionKind) -> bool| -> i64 {
271            sums.iter()
272                .filter(|(kind, _)| picks(*kind))
273                .map(|(_, amount)| amount)
274                .sum()
275        };
276        let bounty = sums
277            .iter()
278            .filter(|(kind, _)| *kind == TransactionKind::Bounty)
279            .map(|(_, amount)| amount)
280            .sum();
281        Ok(Self {
282            today,
283            per_hour,
284            pulse,
285            silver: total(TransactionKind::is_income) - total(TransactionKind::is_outgo),
286            bounty,
287        })
288    }
289}

291/// A number with its thousands marked, as the labels say them.
292#[must_use]
293pub fn grouped(n: i64) -> String {
294    let digits = n.abs().to_string();
295    let mut out = String::new();
296    for (i, c) in digits.chars().enumerate() {
297        if i > 0 && (digits.len() - i).is_multiple_of(3) {
298            out.push(',');
299        }
300        out.push(c);
301    }
302    if n < 0 {
303        out.insert(0, '-');
304    }
305    out
306}

308#[cfg(test)]
309mod tests {
310    use super::*;
311
312    #[test]
313    fn every_row_switches_and_the_default_writes_nothing() {
314        let mut look = UberbarLook::default();
315        for (key, _, _) in ROWS {
316            let was = look.shows(key);
317            look.set(key, Some(!was)).unwrap_or_else(|e| panic!("{e}"));
318            assert_eq!(look.shows(key), !was, "{key}");
319            look.set(key, None).unwrap_or_else(|e| panic!("{e}"));
320            assert_eq!(look.shows(key), was, "{key}");
321        }
322        assert!(look.set("doll", Some(true)).is_err());
323        let written = serde_json::to_string(&look).unwrap_or_else(|e| panic!("{e}"));
324        assert_eq!(written, "{}", "every default row is left out of the file");
325    }
326
327    #[test]
328    fn numbers_are_grouped() {
329        assert_eq!(grouped(0), "0");
330        assert_eq!(grouped(999), "999");
331        assert_eq!(grouped(1234), "1,234");
332        assert_eq!(grouped(-1_234_567), "-1,234,567");
333    }

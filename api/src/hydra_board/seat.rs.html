1//! One character as its windows see it: what its feed heard and the last
2//! snapshot, and the way to send it a line.
3//!
4//! The window has the main thread; the feed runs on the binary's runtime,
5//! beside the session. It hears each event into the story, asks for a fresh
6//! snapshot at most every 100 ms while events come, and wakes the hub and the
7//! character's play window: each is its own native window, and a repaint of
8//! one does not repaint the other.
9//! Lines typed in the window go to the session in order, each written at
10//! once (`SessionHandle::send_typed`), as the terminal client sends them.
11

25/// How often, at most, the feed asks for a fresh snapshot.
26const REFRESH: Duration = Duration::from_millis(100);

28/// What draws the board while it is open, as a window toolkit fills it
29/// in: asked to draw again when a feed has something new, and to open a
30/// link.
31pub trait Windows: Send + Sync {
32    /// Draw the hub again: the board changed.
33    fn repaint(&self);
34    /// Draw the hub and the play window of the seat `id` again.
35    fn repaint_seat(&self, id: u32);
36    /// Open `url` in the browser.
37    fn open_url(&self, url: &str);
38}

40/// The windows to wake when a feed has something new: what draws them
41/// while they are open, nothing while Hydra rests. They open and close
42/// again and again in one run, each time drawn anew.
43///
44/// The terminals attached (`hydra-tui`) are woken too, each its own screen,
45/// as many as are attached; raising and the windows' being open are the
46/// windows' alone.
47#[derive(Clone, Default)]
48pub struct Wake(Arc<Woken>);
49
50#[derive(Default)]
51struct Woken {
52    /// What draws the windows, while they are open.
53    windows: Mutex<Option<Arc<dyn Windows>>>,
54    /// What draws each terminal attached, by the number it was given.
55    terminals: Mutex<Vec<(u64, Arc<dyn Windows>)>>,
56    /// The number the next terminal attached is given.
57    next_terminal: std::sync::atomic::AtomicU64,
58    /// The hub is wanted in front, shown again if it was closed: done by
59    /// its own next frame, which alone can move it.
60    raise: AtomicBool,
61    /// The seats whose play windows are wanted in front, opened again if
62    /// they were closed: done by the windows' next frame.
63    raise_seats: Mutex<Vec<u32>>,
64}
65
66impl std::fmt::Debug for Wake {
67    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
68        f.debug_struct("Wake")
69            .field("open", &self.is_open())
70            .finish_non_exhaustive()
71    }
72}

74impl Wake {
75    /// The windows are open, drawn by `windows`.
76    pub fn opened(&self, windows: Arc<dyn Windows>) {
77        *self.lock() = Some(windows);
78    }
79
80    /// A terminal attached, drawn by `screen`: the number to say it left by.
81    pub fn attached(&self, screen: Arc<dyn Windows>) -> u64 {
82        let number = self.0.next_terminal.fetch_add(1, Ordering::Relaxed);
83        self.screens().push((number, screen));
84        number
85    }
86
87    /// The terminal numbered `number` has left.
88    pub fn detached(&self, number: u64) {
89        self.screens().retain(|(each, _)| *each != number);
90    }
91
92    /// How many terminals are attached.
93    #[must_use]
94    pub fn terminals(&self) -> usize {
95        self.screens().len()
96    }
97
98    fn screens(&self) -> MutexGuard<'_, Vec<(u64, Arc<dyn Windows>)>> {
99        self.0
100            .terminals
101            .lock()
102            .unwrap_or_else(PoisonError::into_inner)
103    }
104
105    /// Every terminal's screen attached now.
106    fn attached_now(&self) -> Vec<Arc<dyn Windows>> {
107        self.screens()
108            .iter()
109            .map(|(_, screen)| Arc::clone(screen))
110            .collect()
111    }
112
113    /// The windows have closed: nothing to wake until they open again.
114    pub fn closed(&self) {
115        *self.lock() = None;
116        self.0.raise.store(false, Ordering::Relaxed);
117        self.raising().clear();
118    }
119
120    /// Bring the hub to the front, shown again if it was closed, when the
121    /// windows are open: whether they were. The system may only flash it,
122    /// when another program has the focus.
123    #[must_use]
124    pub fn raise(&self) -> bool {
125        let Some(windows) = self.windows() else {
126            return false;
127        };
128        self.0.raise.store(true, Ordering::Relaxed);
129        windows.repaint();
130        true
131    }
132
133    /// Whether the hub was asked to the front since this was last asked.
134    #[must_use]
135    pub fn raised(&self) -> bool {
136        self.0.raise.swap(false, Ordering::Relaxed)
137    }
138
139    /// Bring the play window of the seat `id` to the front, opened again if
140    /// it was closed, when the windows are open: whether they were. The hub
141    /// is left as it is.
142    #[must_use]
143    pub fn raise_seat(&self, id: u32) -> bool {
144        let Some(windows) = self.windows() else {
145            return false;
146        };
147        self.raising().push(id);
148        windows.repaint();
149        true
150    }
151
152    /// The seats whose play windows were asked to the front since this was
153    /// last asked.
154    #[must_use]
155    pub fn raised_seats(&self) -> Vec<u32> {
156        std::mem::take(&mut *self.raising())
157    }
158
159    fn raising(&self) -> MutexGuard<'_, Vec<u32>> {
160        self.0
161            .raise_seats
162            .lock()
163            .unwrap_or_else(PoisonError::into_inner)
164    }
165
166    /// Whether the windows are open.
167    #[must_use]
168    pub fn is_open(&self) -> bool {
169        self.lock().is_some()
170    }
171
172    fn lock(&self) -> MutexGuard<'_, Option<Arc<dyn Windows>>> {
173        self.0
174            .windows
175            .lock()
176            .unwrap_or_else(PoisonError::into_inner)
177    }
178
179    fn windows(&self) -> Option<Arc<dyn Windows>> {
180        self.lock().clone()
181    }
182
183    /// Repaint the hub, and every terminal: the board changed.
184    pub fn hub(&self) {
185        if let Some(windows) = self.windows() {
186            windows.repaint();
187        }
188        for screen in self.attached_now() {
189            screen.repaint();
190        }
191    }
192
193    /// Open `url` in the browser: the windows', while they are open; else
194    /// the terminal attached last, on its own computer.
195    fn open(&self, url: &str) {
196        if let Some(windows) = self.windows() {
197            windows.open_url(url);
198        } else if let Some(screen) = self.attached_now().pop() {
199            screen.open_url(url);
200        }
201    }
202
203    /// Repaint the hub and the play window of the seat `id`, and every
204    /// terminal.
205    pub(crate) fn wake(&self, id: u32) {
206        if let Some(windows) = self.windows() {
207            windows.repaint_seat(id);
208        }
209        for screen in self.attached_now() {
210            screen.repaint_seat(id);
211        }
212    }
213}

263/// The most presses or rumbles kept for the window: a script that puts more
264/// before a frame takes them loses the oldest, rather than banking them.
265const KEPT: usize = 32;
266
267/// Keep `item` among `kept`, the oldest let go past [`KEPT`].
268fn keep(kept: &Mutex<Vec<String>>, item: String) {
269    let mut kept = lock(kept);
270    kept.push(item);
271    if kept.len() > KEPT {
272        let over = kept.len() - KEPT;
273        kept.drain(..over);
274    }
275}

277/// Whether `line`, claimed by the command line, is `.hotbar`'s, whatever
278/// the command symbol before it.
279fn is_hotbar_press(line: &str) -> bool {
280    let line = line.trim_start_matches(|c: char| !c.is_alphanumeric());
281    line.split_whitespace().next().is_some_and(|word| {
282        word.eq_ignore_ascii_case("hotbar") || word.eq_ignore_ascii_case("macro")
283    })
284}

436    /// Its command symbol and Lich's: as its session holds them, which the
437    /// settings menu keeps up to date; else as its settings file in the
438    /// data folder `data` says, or the defaults.
439    #[must_use]
440    pub fn symbols(&self, data: &std::path::Path) -> (char, char) {
441        use hydra_session::claimant;
442        if let (Some(symbol), Some(lich)) = (self.symbol(), self.lich_symbol()) {
443            return (symbol, lich);
444        }
445        let settings = self
446            .instance
447            .as_deref()
448            .and_then(|instance| {
449                hydra_session::settings_store::load(data, instance, &self.name).ok()
450            })
451            .and_then(|file| file.section::<claimant::Settings>(claimant::SECTION).ok());
452        (
453            self.symbol().unwrap_or_else(|| {
454                settings
455                    .as_ref()
456                    .map_or(claimant::DEFAULT_SYMBOL, claimant::Settings::symbol)
457            }),
458            self.lich_symbol().unwrap_or_else(|| {
459                settings.as_ref().map_or(
460                    claimant::DEFAULT_LICH_SYMBOL,

485    /// A chain run on the session, each command through the aliases as a
486    /// typed line is; an idle seat, with no session, sends each command.
487    pub fn run_steps(&self, steps: Vec<hydra_session::steps::Step>) {
488        use hydra_session::steps::Step;
489        self.story().typing(std::time::Instant::now());
490        if let (Some(handle), Some(runtime)) = (&self.handle, &self.runtime) {
491            let handle = handle.clone();
492            runtime.spawn(async move {
493                let _ = handle.run_steps(steps, true).await;
494            });
495            return;
496        }
497        for step in steps {
498            let line = match step {
499                Step::Line(line) => line,
500                Step::Behaviour { word, rest } if rest.is_empty() => word,
501                Step::Behaviour { word, rest } => format!("{word} {rest}"),
502                Step::Wait(_) | Step::Roundtime | Step::Casttime => continue,
503            };
504            let _ = self.typed.send(line);
505        }
506    }

611    /// Apply its trainer's goals on the website, on `runtime`, waking
612    /// `wake` with what came of it.
613    pub fn apply_goals(self: &Arc<Self>, runtime: &tokio::runtime::Handle, wake: Wake) {
614        let (session, goals) = {
615            let mut trainer = lock(&self.trainer);
616            let (Some(session), Some(goals)) =
617                (trainer.session.clone(), trainer.view.goals.clone())
618            else {
619                return;
620            };
621            trainer.view.status = crate::trainer::Status::Applying;
622            trainer.view.saved = false;
623            (session, goals)
624        };
625        let seat = Arc::clone(self);
626        runtime.spawn(async move {
627            let applied = session.apply(&goals).await;
628            lock(&seat.trainer).applied(applied);
629            wake.wake(seat.id);
630        });
631    }

633    /// The game sent the link `url`: read as the trainer's, opened in the
634    /// browser, or neither, as `goals` asked.
635    fn launched(self: &Arc<Self>, url: &str, wake: &Wake) {
636        let link = lock(&self.trainer).launched(url);
637        match link {
638            crate::trainer::Link::Open(url) => wake.open(&url),
639            crate::trainer::Link::Page(url) => lock(&self.trainer).offer(url),
640            crate::trainer::Link::Refused => {}
641            crate::trainer::Link::Read(url) => {
642                let seat = Arc::clone(self);
643                let wake = wake.clone();
644                tokio::spawn(async move {
645                    let read = match hydra_net::trainer::Trainer::new() {
646                        Ok(session) => session.fetch(&url).await.map(|goals| (session, goals)),
647                        Err(why) => Err(why),
648                    };
649                    lock(&seat.trainer).read(read);
650                    wake.wake(seat.id);
651                });
652            }
653        }
654    }
655}

687/// What `seat`'s story has called for since last asked, handed on: its
688/// commands to the game, its sounds and notices to the highlights' desk, its
689/// rumbles to the pad, and its timers to the character. While the
690/// highlights are hushed (`Highlights::hush_for`), the commands, sounds and
691/// rumbles are let go of.
692async fn delivered(seat: &Seat, handle: &SessionHandle, highlights: &Highlights, wake: &Wake) {
693    let (calls, timers, sends) = {
694        let mut story = seat.story();
695        (story.calls(), story.timers(), story.sends())
696    };
697    let hushed = highlights.hushed();
698    if !hushed {
699        triggered(handle, sends).await;
700    }
701    for call in calls.into_iter().filter(|_| !hushed) {
702        if let Some(rumble) = &call.rumble {
703            seat.rumble(rumble, wake);
704        }
705        if let (Some(body), Some(data)) = (&call.push, highlights.data()) {
706            pushed(seat, body, data);
707        }
708        highlights.call(&seat.name, call);
709    }
710    // A highlight's timer of the player's own, kept with the character as
711    // `.timer`'s is.
712    let snapshot = lock(&seat.snapshot).clone();
713    let frames = timers
714        .into_iter()
715        .map(|timing| Frame::Timer {
716            value: timing.seconds.map_or(0, |seconds| {
717                snapshot.as_ref().map_or_else(
718                    || GameState::default().timer_ends_in(seconds),
719                    |snapshot| snapshot.state.timer_ends_in(seconds),
720                )
721            }),
722            id: timing.name,
723            category: Some(timing.list.name().to_owned()),
724        })
725        .collect();
726    let _ = handle.fold_in(frames);
727}

729/// Hear the session's events into `seat` until it ends, refreshing its
730/// snapshot while they come; its story looks through `highlights`, built
731/// again when they change, and what they call for goes to their desk.
732async fn feed(
733    seat: Arc<Seat>,
734    handle: SessionHandle,
735    mut events: broadcast::Receiver<Event>,
736    (wake, highlights, merged): (Wake, Highlights, crate::sessions::Merged),
737) {
738    let character = format!("{}:{}", seat.game, seat.name);
739    let mut built = 0;
740    let mut dirty = true;
741    let mut ended = false;
742    let mut tick = tokio::time::interval(REFRESH);
743    tick.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
744    loop {
745        tokio::select! {
746            event = events.recv(), if !ended => {
747                match event {
748                    Ok(event) => {
749                        if let Event::StateChanged(state) = &event {
750                            *lock(&seat.lifecycle) = *state;
751                        }
752                        if let Event::Claimed(line) = &event
753                            && is_hotbar_press(line)
754                        {
755                            seat.press_hotbar(line, &wake);
756                        }
757                        if highlights.generation() != built {
758                            let (generation, matcher) = highlights.for_character(&character);
759                            built = generation;
760                            seat.story().highlight_with(Some(matcher));
761                        }
762                        if let Event::Notice(notice) = &event
763                            && notice.closes_a_hunt()
764                        {
765                            seat.hunts_closed.fetch_add(1, Ordering::Relaxed);
766                        }
767                        if let Event::Frame(frame) = &event {
768                            match frame.as_ref() {
769                                hydra_protocol::Frame::LaunchUrl { url } => {
770                                    seat.launched(url, &wake);
771                                }
772                                hydra_protocol::Frame::EbonGateMerchant(said) => {
773                                    crate::merchants::heard(&seat, said, &highlights);
774                                }
775                                _ => {}
776                            }
777                        }
778                        hydra_mem::part(hydra_mem::Part::Story, || seat.story().hear(&event));
779                        let heard = seat.story().take_merging();
780                        if !heard.is_empty()
781                            && lock(&merged).offer(
782                                Instant::now(),
783                                (seat.id, &character, &seat.name),
784                                &heard,
785                            )
786                        {
787                            wake.hub();
788                        }
789                        delivered(&seat, &handle, &highlights, &wake).await;
790                    }
791                    Err(RecvError::Lagged(_)) => seat.story().missed(),
792                    Err(RecvError::Closed) => ended = true,
793                }
794                dirty = true;
795                wake.wake(seat.id);
796            }
797            _ = tick.tick(), if dirty => {
798                dirty = false;
799                if let Some(snapshot) = handle.snapshot().await {
800                    // The live prompt loses its `R` when roundtime ends,
801                    // which no event says: so a roundtime keeps it asking.
802                    let snapshot = Arc::new(snapshot);
803                    // A chunk held past its time is shown, and one still
804                    // held keeps the tick asking.
805                    let holding = hydra_mem::part(hydra_mem::Part::Story, || {
806                        let mut story = seat.story();
807                        story.release_due(Instant::now());
808                        story.settle(&snapshot.state);
809                        story.see(Arc::clone(&snapshot));
810                        story.holding()
811                    });
812                    dirty = snapshot.state.in_roundtime() == Some(true) || holding;
813                    *lock(&seat.lifecycle) = snapshot.lifecycle;
814                    *lock(&seat.snapshot) = Some(snapshot);
815                    // A trigger on a condition alone fires as the story
816                    // sees the character, and is handed on as one on a line.
817                    delivered(&seat, &handle, &highlights, &wake).await;
818                } else {
819                    // The session has ended; its last snapshot stays.
820                    *lock(&seat.lifecycle) = State::Closed;
821                    ended = true;
822                }
823                wake.wake(seat.id);
824            }
825            else => return,
826        }
827    }
828}

845/// The next number `asked` gives an object's menu: one past the last, never
846/// 0.
847pub fn next_menu(asked: &AtomicU32) -> u32 {
848    let mut was = asked.load(Ordering::Relaxed);
849    loop {
850        let next = was.wrapping_add(1).max(1);
851        match asked.compare_exchange_weak(was, next, Ordering::Relaxed, Ordering::Relaxed) {
852            Ok(_) => return next,
853            Err(now) => was = now,
854        }
855    }
856}

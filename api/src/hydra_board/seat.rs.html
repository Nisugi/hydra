1//! One character as its windows see it: what its feed heard and the last
2//! snapshot, and the way to send it a line.
3//!
4//! The window has the main thread; the feed runs on the binary's runtime,
5//! beside the session. It hears each event into the story, asks for a fresh
6//! snapshot at most every 100 ms while events come, and wakes the hub and the
7//! character's play window: each is its own native window, and a repaint of
8//! one does not repaint the other.
9//! Lines typed in the window go to the session in order, each written at
10//! once (`SessionHandle::send_typed`), as the terminal client sends them.
11

25/// How often, at most, the feed asks for a fresh snapshot.
26const REFRESH: Duration = Duration::from_millis(100);

28/// What draws the board while it is open, as a window toolkit fills it
29/// in: asked to draw again when a feed has something new, and to open a
30/// link.
31pub trait Windows: Send + Sync {
32    /// Draw the hub again: the board changed.
33    fn repaint(&self);
34    /// Draw the hub and the play window of the seat `id` again.
35    fn repaint_seat(&self, id: u32);
36    /// Open `url` in the browser.
37    fn open_url(&self, url: &str);
38}

40/// The windows to wake when a feed has something new: what draws them
41/// while they are open, nothing while Hydra rests. They open and close
42/// again and again in one run, each time drawn anew.
43#[derive(Clone, Default)]
44pub struct Wake(Arc<Woken>);
45
46#[derive(Default)]
47struct Woken {
48    /// What draws the windows, while they are open.
49    windows: Mutex<Option<Arc<dyn Windows>>>,
50    /// The hub is wanted in front, shown again if it was closed: done by
51    /// its own next frame, which alone can move it.
52    raise: AtomicBool,
53    /// The seats whose play windows are wanted in front, opened again if
54    /// they were closed: done by the windows' next frame.
55    raise_seats: Mutex<Vec<u32>>,
56}
57
58impl std::fmt::Debug for Wake {
59    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
60        f.debug_struct("Wake")
61            .field("open", &self.is_open())
62            .finish_non_exhaustive()
63    }
64}

66impl Wake {
67    /// The windows are open, drawn by `windows`.
68    pub fn opened(&self, windows: Arc<dyn Windows>) {
69        *self.lock() = Some(windows);
70    }
71
72    /// The windows have closed: nothing to wake until they open again.
73    pub fn closed(&self) {
74        *self.lock() = None;
75        self.0.raise.store(false, Ordering::Relaxed);
76        self.raising().clear();
77    }
78
79    /// Bring the hub to the front, shown again if it was closed, when the
80    /// windows are open: whether they were. The system may only flash it,
81    /// when another program has the focus.
82    #[must_use]
83    pub fn raise(&self) -> bool {
84        let Some(windows) = self.windows() else {
85            return false;
86        };
87        self.0.raise.store(true, Ordering::Relaxed);
88        windows.repaint();
89        true
90    }
91
92    /// Whether the hub was asked to the front since this was last asked.
93    #[must_use]
94    pub fn raised(&self) -> bool {
95        self.0.raise.swap(false, Ordering::Relaxed)
96    }
97
98    /// Bring the play window of the seat `id` to the front, opened again if
99    /// it was closed, when the windows are open: whether they were. The hub
100    /// is left as it is.
101    #[must_use]
102    pub fn raise_seat(&self, id: u32) -> bool {
103        let Some(windows) = self.windows() else {
104            return false;
105        };
106        self.raising().push(id);
107        windows.repaint();
108        true
109    }
110
111    /// The seats whose play windows were asked to the front since this was
112    /// last asked.
113    #[must_use]
114    pub fn raised_seats(&self) -> Vec<u32> {
115        std::mem::take(&mut *self.raising())
116    }
117
118    fn raising(&self) -> MutexGuard<'_, Vec<u32>> {
119        self.0
120            .raise_seats
121            .lock()
122            .unwrap_or_else(PoisonError::into_inner)
123    }
124
125    /// Whether the windows are open.
126    #[must_use]
127    pub fn is_open(&self) -> bool {
128        self.lock().is_some()
129    }
130
131    fn lock(&self) -> MutexGuard<'_, Option<Arc<dyn Windows>>> {
132        self.0
133            .windows
134            .lock()
135            .unwrap_or_else(PoisonError::into_inner)
136    }
137
138    fn windows(&self) -> Option<Arc<dyn Windows>> {
139        self.lock().clone()
140    }
141
142    /// Repaint the hub: the board changed.
143    pub fn hub(&self) {
144        if let Some(windows) = self.windows() {
145            windows.repaint();
146        }
147    }
148
149    /// Open `url` in the browser.
150    fn open(&self, url: &str) {
151        if let Some(windows) = self.windows() {
152            windows.open_url(url);
153        }
154    }
155
156    /// Repaint the hub and the play window of the seat `id`.
157    fn wake(&self, id: u32) {
158        if let Some(windows) = self.windows() {
159            windows.repaint_seat(id);
160        }
161    }
162}

202/// Whether `line`, claimed by the command line, is `.hotbar`'s, whatever
203/// the command symbol before it.
204fn is_hotbar_press(line: &str) -> bool {
205    let line = line.trim_start_matches(|c: char| !c.is_alphanumeric());
206    line.split_whitespace()
207        .next()
208        .is_some_and(|word| word.eq_ignore_ascii_case("hotbar"))
209}

212    /// Follow session `id`, the character `name` on the game `game` (its
213    /// code), from `events` (taken from `handle` before the session ran, so
214    /// none is missed), on `runtime`, waking `wake` when something changed;
215    /// its story looks through `highlights`, and what it hears on the
216    /// shared streams goes to `merged`.
217    #[must_use]
218    pub fn start(
219        (id, name, game): (u32, &str, &str),
220        handle: &SessionHandle,
221        events: broadcast::Receiver<Event>,
222        runtime: &tokio::runtime::Handle,
223        (wake, highlights, merged): (Wake, Highlights, crate::sessions::Merged),
224    ) -> Arc<Self> {
225        let (typed, lines) = mpsc::unbounded_channel();
226        let seat = Arc::new(Self {
227            id,
228            name: name.to_owned(),
229            game: game.to_owned(),
230            instance: hydra_net::instance(game).map(str::to_owned),
231            story: Mutex::new(Story::default()),
232            snapshot: Mutex::new(None),
233            lifecycle: Mutex::new(State::Connecting),
234            typed,
235            log: std::sync::OnceLock::new(),
236            team: std::sync::OnceLock::new(),
237            keeps_window: AtomicBool::new(true),
238            handle: Some(handle.clone()),
239            trainer: Mutex::default(),
240            rumbles: Mutex::default(),
241            hotbar_presses: Mutex::default(),
242        });
243        runtime.spawn(hydra_mem::task(
244            hydra_mem::Part::Feeds,
245            feed(
246                Arc::clone(&seat),
247                handle.clone(),
248                events,
249                (wake, highlights, merged),
250            ),
251        ));
252        runtime.spawn(hydra_mem::task(
253            hydra_mem::Part::Feeds,
254            send(handle.clone(), lines),
255        ));
256        seat
257    }

404    /// Apply its trainer's goals on the website, on `runtime`, waking
405    /// `wake` with what came of it.
406    pub fn apply_goals(self: &Arc<Self>, runtime: &tokio::runtime::Handle, wake: Wake) {
407        let (session, goals) = {
408            let mut trainer = lock(&self.trainer);
409            let (Some(session), Some(goals)) =
410                (trainer.session.clone(), trainer.view.goals.clone())
411            else {
412                return;
413            };
414            trainer.view.status = crate::trainer::Status::Applying;
415            (session, goals)
416        };
417        let seat = Arc::clone(self);
418        runtime.spawn(async move {
419            let applied = session.apply(&goals).await;
420            lock(&seat.trainer).applied(applied);
421            wake.wake(seat.id);
422        });
423    }

425    /// The game sent the link `url`: read as the trainer's, opened in the
426    /// browser, or neither, as `goals` asked.
427    fn launched(self: &Arc<Self>, url: &str, wake: &Wake) {
428        let link = lock(&self.trainer).launched(url);
429        match link {
430            crate::trainer::Link::Open(url) => wake.open(&url),
431            crate::trainer::Link::Refused => {}
432            crate::trainer::Link::Read(url) => {
433                let seat = Arc::clone(self);
434                let wake = wake.clone();
435                tokio::spawn(async move {
436                    let read = match hydra_net::trainer::Trainer::new() {
437                        Ok(session) => session.fetch(&url).await.map(|goals| (session, goals)),
438                        Err(why) => Err(why),
439                    };
440                    lock(&seat.trainer).read(read);
441                    wake.wake(seat.id);
442                });
443            }
444        }
445    }

460/// What `seat`'s story has called for since last asked, handed on: its
461/// commands to the game, its sounds and notices to the highlights' desk, its
462/// rumbles to the pad, and its timers to the character. While the
463/// highlights are hushed (`Highlights::hush_for`), the commands, sounds and
464/// rumbles are let go of.
465async fn delivered(seat: &Seat, handle: &SessionHandle, highlights: &Highlights) {
466    let (calls, timers, sends) = {
467        let mut story = seat.story();
468        (story.calls(), story.timers(), story.sends())
469    };
470    let hushed = highlights.hushed();
471    if !hushed {
472        triggered(handle, sends).await;
473    }
474    for call in calls.into_iter().filter(|_| !hushed) {
475        if let Some(rumble) = &call.rumble {
476            lock(&seat.rumbles).push(rumble.clone());
477        }
478        highlights.call(&seat.name, call);
479    }
480    // A highlight's timer of the player's own, kept with the character as
481    // `.timer`'s is.
482    let snapshot = lock(&seat.snapshot).clone();
483    let frames = timers
484        .into_iter()
485        .map(|timing| Frame::Timer {
486            value: timing.seconds.map_or(0, |seconds| {
487                snapshot.as_ref().map_or_else(
488                    || GameState::default().timer_ends_in(seconds),
489                    |snapshot| snapshot.state.timer_ends_in(seconds),
490                )
491            }),
492            id: timing.name,
493            category: Some(timing.list.name().to_owned()),
494        })
495        .collect();
496    let _ = handle.fold_in(frames);
497}

499/// Hear the session's events into `seat` until it ends, refreshing its
500/// snapshot while they come; its story looks through `highlights`, built
501/// again when they change, and what they call for goes to their desk.
502async fn feed(
503    seat: Arc<Seat>,
504    handle: SessionHandle,
505    mut events: broadcast::Receiver<Event>,
506    (wake, highlights, merged): (Wake, Highlights, crate::sessions::Merged),
507) {
508    let character = format!("{}:{}", seat.game, seat.name);
509    let mut built = 0;
510    let mut dirty = true;
511    let mut ended = false;
512    let mut tick = tokio::time::interval(REFRESH);
513    tick.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
514    loop {
515        tokio::select! {
516            event = events.recv(), if !ended => {
517                match event {
518                    Ok(event) => {
519                        if let Event::StateChanged(state) = &event {
520                            *lock(&seat.lifecycle) = *state;
521                        }
522                        if let Event::Claimed(line) = &event
523                            && is_hotbar_press(line)
524                        {
525                            lock(&seat.hotbar_presses).push(line.clone());
526                            wake.wake(seat.id);
527                        }
528                        if highlights.generation() != built {
529                            let (generation, matcher) = highlights.for_character(&character);
530                            built = generation;
531                            seat.story().highlight_with(Some(matcher));
532                        }
533                        if let Event::Frame(frame) = &event
534                            && let hydra_protocol::Frame::LaunchUrl { url } = frame.as_ref()
535                        {
536                            seat.launched(url, &wake);
537                        }
538                        hydra_mem::part(hydra_mem::Part::Story, || seat.story().hear(&event));
539                        let heard = seat.story().take_merging();
540                        if !heard.is_empty()
541                            && lock(&merged).offer(
542                                Instant::now(),
543                                (seat.id, &character, &seat.name),
544                                &heard,
545                            )
546                        {
547                            wake.hub();
548                        }
549                        delivered(&seat, &handle, &highlights).await;
550                    }
551                    Err(RecvError::Lagged(_)) => seat.story().missed(),
552                    Err(RecvError::Closed) => ended = true,
553                }
554                dirty = true;
555                wake.wake(seat.id);
556            }
557            _ = tick.tick(), if dirty => {
558                dirty = false;
559                if let Some(snapshot) = handle.snapshot().await {
560                    // The live prompt loses its `R` when roundtime ends,
561                    // which no event says: so a roundtime keeps it asking.
562                    let snapshot = Arc::new(snapshot);
563                    hydra_mem::part(hydra_mem::Part::Story, || {
564                        let mut story = seat.story();
565                        story.settle(&snapshot.state);
566                        story.see(Arc::clone(&snapshot));
567                    });
568                    dirty = snapshot.state.in_roundtime() == Some(true);
569                    *lock(&seat.lifecycle) = snapshot.lifecycle;
570                    *lock(&seat.snapshot) = Some(snapshot);
571                    // A trigger on a condition alone fires as the story
572                    // sees the character, and is handed on as one on a line.
573                    delivered(&seat, &handle, &highlights).await;
574                } else {
575                    // The session has ended; its last snapshot stays.
576                    *lock(&seat.lifecycle) = State::Closed;
577                    ended = true;
578                }
579                wake.wake(seat.id);
580            }
581            else => return,
582        }
583    }
584}

586/// Send each typed line in order, each written as it comes.
587async fn send(handle: SessionHandle, mut lines: mpsc::UnboundedReceiver<String>) {
588    while let Some(line) = lines.recv().await {
589        let _ = handle.send_typed(&line).await;
590    }

604        seat.story()
605            .lines
606            .iter()
607            .map(|(_, shown)| match shown {
608                Shown::Game(runs) | Shown::From(_, runs) => {
609                    runs.iter().map(|run| run.text.as_str()).collect()
610                }
611                Shown::Typed { prompt, line } => format!("{prompt}{line}"),
612                Shown::Prompt(prompt) => prompt.clone(),
613                Shown::Gap => "[gap]".to_owned(),
614                Shown::Hydra(text) => format!("[hydra] {text}"),
615                Shown::Said(notice) => notice.lines().join(
616                    "
617",
618                ),
619            })

621    }

1//! What a key does: a **macro** is a keybind that sends one or more
2//! commands to the game, fills the command input, or performs one of
3//! Hydra's actions.
4//!
5//! A send macro's text is cut into commands at each `\r`, each piece one
6//! command, empty pieces dropped, and a piece that is `s` and a number of
7//! seconds (`s1.5`) a wait before the rest. Each command is one line, as
8//! every key's must be (`hydra_ui::validate_line`). A trailing `\r` means
9//! nothing: a macro's kind, not its last character, says whether it is
10//! sent.
11
12/// One step of a send macro: a command, or a wait before the rest; read as
13/// an alias's are (`hydra_session::steps`).
14pub use hydra_session::steps::Step;
15
16/// What a key does.
17#[derive(Clone, Debug, PartialEq, Eq)]
18pub enum Macro {
19    /// Send this chain to the game, as if typed: a comma (or `\r`) between
20    /// two commands, `s1.5` for a wait, `rt` and `ct` for the clock, and the
21    /// behaviour words (`hydra_session::steps`).
22    Send(String),
23    /// Put this in the command input, not sent.
24    Fill(String),
25    /// Perform one of Hydra's actions.
26    Act(Action),
27    /// Press a hotbar's button, by the bar's name and the button's id or
28    /// nickname: a button's own hotkey, or a controller's bind.
29    Hotbar {
30        /// The bar's name.
31        bar: String,
32        /// The button's id or nickname.
33        button: String,
34    },
35    /// Run a macro, by its name (`hydra_ui::macros`): a macro's own key,
36    /// or a keybind that names one.
37    Run(String),
38}

40/// One of Hydra's actions a key performs. None is named before it does
41/// something.
42#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]

93    TargetClear,

95    DrawerTop,
96    /// The bottom drawer.
97    DrawerBottom,
98    /// The left drawer.
99    DrawerLeft,
100    /// The right drawer.
101    DrawerRight,
102    /// The top bar's Lock: every window kept where it is, or free again.
103    Lock,
104    /// The top bar's Arrange.
105    Arrange,
106    /// The play window of the hub's first to ninth character, 1 to 9: open,

314impl Macro {
315    /// A send macro's commands and waits, in order; any other kind, none.
316    ///
317    /// # Errors
318    ///
319    /// Why it cannot be sent, in words for the player: a command that is not
320    /// one line, a wait too long, or nothing to send.
321    pub fn steps(&self) -> Result<Vec<Step>, String> {
322        let Self::Send(text) = self else {
323            return Ok(Vec::new());
324        };
325        let steps = hydra_session::steps::steps(text)?;
326        for step in &steps {
327            if let Step::Line(line) = step {
328                hydra_ui::validate_line(line).map_err(|why| format!("`{line}`: {why}"))?;
329            }
330        }
331        Ok(steps)
332    }
333
334    /// Whether it can be done as written.
335    ///
336    /// # Errors
337    ///
338    /// Why not, in words for the player.
339    pub fn check(&self) -> Result<(), String> {
340        match self {
341            Self::Send(_) => self.steps().map(|_| ()),
342            Self::Fill(text) => hydra_ui::validate_line(text).map_err(|why| why.to_string()),
343            Self::Hotbar { bar, button } if bar.trim().is_empty() || button.trim().is_empty() => {
344                Err("a hotbar's button is named by its bar and its id".to_owned())
345            }
346            Self::Run(name) if name.trim().is_empty() => {
347                Err("a macro is named by its name".to_owned())
348            }
349            Self::Act(_) | Self::Hotbar { .. } | Self::Run(_) => Ok(()),
350        }
351    }
352
353    /// What it does, for a player: that it sends, fills or does what.
354    #[must_use]
355    pub fn said(&self) -> String {
356        match self {
357            Self::Send(text) => format!("sends `{}`", shown(text)),
358            Self::Fill(text) => format!("fills the input with `{text}`"),
359            Self::Act(action) => format!("does {}", action.label()),
360            Self::Hotbar { bar, button } => format!("presses {bar}'s {button}"),
361            Self::Run(name) => format!("runs the macro {name}"),
362        }
363    }
364
365    /// It as the keybinds file writes it, a TOML value: a send macro a
366    /// string, the others an inline table.
367    #[must_use]
368    pub fn written(&self) -> String {
369        let quoted = |text: &str| toml::Value::String(text.to_owned()).to_string();
370        match self {
371            Self::Send(text) => quoted(text),
372            Self::Fill(text) => format!("{{ fill = {} }}", quoted(text)),
373            Self::Act(action) => format!("{{ action = {} }}", quoted(action.name())),
374            Self::Hotbar { bar, button } => {
375                format!(
376                    "{{ hotbar = {}, button = {} }}",
377                    quoted(bar),
378                    quoted(button)
379                )
380            }
381            Self::Run(name) => format!("{{ macro = {} }}", quoted(name)),
382        }
383    }
384
385    /// The macro a keybinds file's value is: `Ok(None)` for `""`, which
386    /// unbinds a key Hydra binds.
387    ///
388    /// # Errors
389    ///
390    /// What is wrong with it, in words for the player.
391    pub fn read(value: &toml::Value) -> Result<Option<Self>, String> {
392        let made = match value {
393            toml::Value::String(text) if text.trim().is_empty() => return Ok(None),
394            toml::Value::String(text) => Self::Send(text.clone()),
395            toml::Value::Table(table) if table.len() == 1 => {
396                match table
397                    .iter()
398                    .next()
399                    .map(|(kind, value)| (kind.as_str(), value))
400                {
401                    Some(("fill", toml::Value::String(text))) => Self::Fill(text.clone()),
402                    Some(("action", toml::Value::String(name))) => Self::Act(
403                        Action::named(name)
404                            .ok_or_else(|| format!("`{name}` is not an action Hydra has"))?,
405                    ),
406                    Some(("macro", toml::Value::String(name))) => Self::Run(name.clone()),
407                    _ => return Err(KINDS.to_owned()),
408                }
409            }
410            _ => return Err(KINDS.to_owned()),
411        };
412        made.check()?;
413        Ok(Some(made))
414    }
415}

417/// What a keybinds file's value may be.
418const KINDS: &str = "a key is bound to \"commands\", { fill = \"text\" }, { action = \"name\" } or { macro = \"name\" }";

420/// A send macro's text as the Keys page shows it and a player types it:
421/// each `\r` written as the two characters `\r`.
422#[must_use]
423pub fn shown(text: &str) -> String {
424    text.replace('\r', "\\r")
425}
426
427/// Text a player typed on the Keys page, each `\r` they wrote as two
428/// characters made the one.
429#[must_use]
430pub fn typed(text: &str) -> String {
431    text.replace("\\r", "\r")
432}

438    fn lines(steps: &[Step]) -> Vec<String> {
439        steps
440            .iter()
441            .map(|step| match step {
442                Step::Line(line) => line.clone(),
443                Step::Wait(wait) => format!("wait {}ms", wait.as_millis()),
444                other => hydra_session::steps::said(std::slice::from_ref(other)),
445            })
446            .collect()
447    }

449    /// A send macro is cut at each `\r` (or line end) into commands and
450    /// waits, in order, empty pieces dropped.
451    #[test]
452    fn a_send_macro_is_cut_into_commands_and_waits() {
453        let steps = Macro::Send("stance off\r\rprep 118\r s1.5 \rcast\r".to_owned()).steps();
454        assert_eq!(
455            steps.as_deref().map(lines),
456            Ok(vec![
457                "stance offensive".to_owned(),
458                "prep 118".to_owned(),
459                "wait 1500ms".to_owned(),
460                "cast".to_owned()
461            ])
462        );
463        assert_eq!(
464            Macro::Send("look\nsearch".to_owned())
465                .steps()
466                .map(|steps| steps.len()),
467            Ok(2)
468        );
469        assert_eq!(
470            Macro::Send("s".to_owned()).steps().as_deref().map(lines),
471            Ok(vec!["s".to_owned()]),
472            "south is a command, not a wait"
473        );
474    }

495    /// Each kind is written as the file reads it back, and `""` unbinds.
496    #[test]
497    fn each_kind_is_written_as_it_is_read() {
498        for made in [
499            Macro::Send("stance off\rincant 610".to_owned()),
500            Macro::Fill("prep 111 ".to_owned()),
501            Macro::Act(Action::Lock),
502        ] {
503            let text = format!("key = {}", made.written());
504            let table: toml::Table = toml::from_str(&text).expect("TOML");
505            assert_eq!(Macro::read(&table["key"]), Ok(Some(made)), "{text}");
506        }
507        assert_eq!(Macro::read(&toml::Value::String(String::new())), Ok(None));
508        let set: toml::Table = toml::from_str("key = { action = \"macro_set_3\" }").expect("TOML");
509        assert_eq!(
510            Macro::read(&set["key"]),
511            Ok(Some(Macro::Act(Action::Set(3))))
512        );
513        let unknown: toml::Table = toml::from_str("key = { action = \"fly\" }").expect("TOML");
514        assert!(Macro::read(&unknown["key"]).is_err());
515        let two: toml::Table =
516            toml::from_str("key = { fill = \"a\", action = \"lock\" }").expect("TOML");
517        assert!(Macro::read(&two["key"]).is_err());
518    }

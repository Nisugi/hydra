10/// Where `character` on `instance`'s own macros are kept, beside its
11/// layout.
12#[must_use]
13pub fn character_path(layouts: &Path, instance: Option<&str>, name: &str) -> PathBuf {
14    crate::kept::file(layouts, instance, name).with_extension("macros.toml")
15}

17/// Everyone's macros file and `name`'s own, under `data`.
18#[must_use]
19pub fn macros_paths(data: &Path, instance: Option<&str>, name: &str) -> [PathBuf; 2] {
20    [
21        data.join(FILE),
22        character_path(&data.join("layouts"), instance, name),
23    ]
24}

26/// The macros `name` on `instance` has, everyone's with its own over them,
27/// and why a file did not read.
28#[must_use]
29pub fn macros_for(data: &Path, instance: Option<&str>, name: &str) -> (MacroFile, Vec<String>) {
30    let mut problems = Vec::new();
31    let mut read = |path: &Path| {
32        MacroFile::load(path).unwrap_or_else(|why| {
33            problems.push(why);
34            MacroFile::default()
35        })
36    };
37    let [every, own] = macros_paths(data, instance, name);
38    let every = read(&every);
39    let own = read(&own);
40    (every.merged(own), problems)
41}

43/// The chain the macro `name` runs in `world` at `now`, held against
44/// `runs`, which notes the run.
45///
46/// # Errors
47///
48/// No such macro, one its condition or state makes unavailable now, or one
49/// that ran too recently or in this room already.
50pub fn macro_chain(
51    file: &MacroFile,
52    name: &str,
53    world: &World<'_>,
54    (runs, now): (&mut Runs, std::time::Instant),
55) -> Result<String, String> {
56    let made = file
57        .named(name)
58        .ok_or_else(|| format!("There is no macro {}.", name.trim()))?;
59    let chain = made.chain_in(world)?;
60    if chain.trim().is_empty() {
61        return Err(format!("The macro {} runs nothing.", made.name));
62    }
63    let room = world.state.room.id.as_deref();
64    runs.allows(made, now, room)?;
65    runs.note(&made.name, now, room);
66    Ok(chain.to_owned())
67}

69/// `made` kept in the macros file at `path`, in place of the macro named
70/// `was` (its old name), the rest of the file kept; a new one is added.
71///
72/// # Errors
73///
74/// The file does not read or would not be written, the name is not one
75/// word, or another macro in the file has it.
76pub fn keep(path: &Path, made: &MacroDef, was: Option<&str>) -> Result<(), String> {
77    // The file read, changed and written with no other change to it made
78    // meanwhile: the Macros page and `.macro` both change it.
79    hydra_session::store::changing(path, || keep_held(path, made, was))
80}
81
82/// [`keep`], the file's lock held.
83fn keep_held(path: &Path, made: &MacroDef, was: Option<&str>) -> Result<(), String> {
84    let name = made.name.trim();
85    if name.is_empty() || name.contains(char::is_whitespace) {
86        return Err(format!("A macro's name is one word: `{name}` is not."));
87    }
88    let mut file = MacroFile::load(path)?;
89    let taken = file.macros.iter().position(|had| {
90        had.name.eq_ignore_ascii_case(name)
91            && !was.is_some_and(|was| was.eq_ignore_ascii_case(&had.name))
92    });
93    if taken.is_some() {
94        return Err(format!("There is a macro {name} already."));
95    }
96    let mut made = made.clone();
97    name.clone_into(&mut made.name);
98    let old = was.and_then(|was| {
99        file.macros
100            .iter()
101            .position(|had| had.name.eq_ignore_ascii_case(was))
102    });
103    match old {
104        Some(at) => file.macros[at] = made,
105        None => file.macros.push(made),
106    }
107    write(path, &file)
108}

110/// The macro named `name` taken out of the file at `path`.
111///
112/// # Errors
113///
114/// The file does not read or would not be written, or has no such macro.
115pub fn remove(path: &Path, name: &str) -> Result<(), String> {
116    // The file read, changed and written with no other change to it made
117    // meanwhile: the Macros page and `.macro` both change it.
118    hydra_session::store::changing(path, || remove_held(path, name))
119}
120
121/// [`remove`], the file's lock held.
122fn remove_held(path: &Path, name: &str) -> Result<(), String> {
123    let mut file = MacroFile::load(path)?;
124    let at = file
125        .macros
126        .iter()
127        .position(|had| had.name.eq_ignore_ascii_case(name.trim()))
128        .ok_or_else(|| format!("There is no macro {}.", name.trim()))?;
129    file.macros.remove(at);
130    write(path, &file)
131}
132
133fn write(path: &Path, file: &MacroFile) -> Result<(), String> {
134    let folder = path.parent().map(Path::to_path_buf).unwrap_or_default();
135    hydra_session::store::save_bytes(&folder, path, file.to_toml().as_bytes())
136        .map_err(|why| why.to_string())
137}

139#[cfg(test)]
140mod tests {
141    use super::*;
142
143    /// A macro kept is added, one kept under its old name replaces it, a
144    /// name another has is refused, and a removed one is gone.
145    #[test]
146    fn a_macro_is_kept_replaced_refused_and_removed() {
147        let dir = std::env::temp_dir().join(format!("hydra-macros-{}", std::process::id()));
148        let _ = std::fs::remove_dir_all(&dir);
149        std::fs::create_dir_all(&dir).expect("a test folder");
150        let path = dir.join(FILE);
151        let made = |name: &str, chain: &str| MacroDef {
152            name: name.to_owned(),
153            chain: chain.to_owned(),
154            ..MacroDef::default()
155        };
156        keep(&path, &made("barrage", "weapon barrage"), None).expect("kept");
157        keep(&path, &made("flee", "stance def"), None).expect("kept");
158        assert!(keep(&path, &made("Barrage", "x"), None).is_err(), "taken");
159        assert!(keep(&path, &made("two words", "x"), None).is_err());
160        keep(
161            &path,
162            &made("wb", "stance off, rt, weapon barrage"),
163            Some("barrage"),
164        )
165        .expect("renamed");
166        let file = MacroFile::load(&path).expect("reads");
167        assert_eq!(
168            file.macros
169                .iter()
170                .map(|m| m.name.as_str())
171                .collect::<Vec<_>>(),
172            ["wb", "flee"]
173        );
174        remove(&path, "FLEE").expect("removed");
175        assert!(remove(&path, "flee").is_err());
176        assert_eq!(MacroFile::load(&path).expect("reads").macros.len(), 1);
177        let _ = std::fs::remove_dir_all(&dir);
178    }

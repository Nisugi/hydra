1//! What the binary and the window share: every character seated, and what
2//! the hub shows of the roster, and the way the hub asks the binary for
3//! something.
4//!
5//! The binary owns the session table and the roster: it seats a
6//! character as it starts one, unseats one it takes off, and tells the board
7//! the roster, the characters it can start, and an account's listing. The
8//! hub asks it through [`HubControl`], and the answer is shown for a while.
9//!
10//! The web page reads the same board: the colours the window wears, and
11//! whether the page is wanted and on which port, which the window sets from
12//! its own settings, a team started for the web asks for besides, and the
13//! page's server follows ([`Sessions::web_wanted`]); the server says where
14//! it listens, or why it does not, for the hub.
15//!
16//! The binary also says here how the windows are opened
17//! ([`Sessions::opener`]): the session table asks for them so, with or
18//! without the hub, when a team starts with its play windows; and for a
19//! terminal of its own on a character shown in one.
20//!
21//! Hydra's own settings (`window.toml`, [`Own`]) are held here once, for
22//! every front end: a change one makes is the one every other sees, and
23//! none writes back what it held before. What Hydra has to tell the
24//! player with no character to tell it to -- a file that would not write,
25//! highlights that cannot be used -- is said here ([`Sessions::tell`]),
26//! and every hub shows what it has not shown yet.
27//!
28//! And where an update of Hydra is ([`Update`]): the binary looks for one
29//! and fetches it, the hub and the tray say where it is, and either asks
30//! for it to be fetched ([`Sessions::fetch_update`]).
31

48/// How long the hub shows the binary's answer to a request.
49const SAID_FOR: Duration = Duration::from_secs(8);

51/// How long a terminal's wait for a character it asked to start holds.
52const AWAIT_FOR: Duration = Duration::from_mins(5);

54/// How many of the things told a hub that has not looked can still find.
55const TOLD_KEPT: usize = 32;

57/// The board, shared: clones are the same board.

61    wake: Wake,
62    highlights: Highlights,
63    /// The stream rules every character's session sorts its lines by.

67    merged: Merged,
68    /// The port the web page is wanted on; `None`, not wanted.
69    web: watch::Sender<Option<u16>>,
70    /// The run is over: the window closes, if it is open, and the binary

72    closing: watch::Sender<bool>,
73    /// Hydra's own settings, one for every front end.
74    own: Arc<Mutex<Own>>,
75    /// The hub or the tray asked for the update out to be fetched.
76    update_asked: Arc<tokio::sync::Notify>,
77    /// *Performance* is on: each character's stages shown on the hubs and

88/// How the windows are asked to open.
89#[derive(Clone, Copy, Debug, PartialEq, Eq)]
90pub enum Open {
91    /// The hub, in front: shown again if it was closed.
92    Hub,
93    /// The play windows the characters keep, the hub left hidden: a team
94    /// started with its play windows while Hydra rests.

96    /// The play window of the seat with this number, in front, the hub

101    /// A terminal of its own on the seat with this number, the windows left
102    /// as they are: a character shown in a terminal, started.
103    Terminal(u32),
104}

118/// A newer release of Hydra, and how far it has come.
119#[derive(Clone, Debug, PartialEq, Eq)]
120pub struct Update {

123    /// Its page on the releases site, where its notes are.

126    pub fetch: Fetch,

129/// How far an update has come.
130#[derive(Clone, Debug, PartialEq, Eq)]
131pub enum Fetch {
132    /// Out, not fetched.
133    Out,
134    /// Being fetched: the bytes so far, and how many there are.
135    Fetching(u64, u64),
136    /// Fetched, checked and in place: Hydra is this release from its next
137    /// start.
138    Ready,
139    /// Fetching it failed, and why; it may be asked for again.
140    Failed(String),
141}

251    /// Tell the player `said`, with no character to tell it to: every hub
252    /// shows it once.
253    pub fn tell(&self, said: String) {
254        {
255            let mut board = self.board();
256            board.told_last += 1;
257            let number = board.told_last;
258            board.told.push_back((number, said));
259            while board.told.len() > TOLD_KEPT {
260                board.told.pop_front();
261            }
262        }
263        self.wake.hub();
264    }
265
266    /// What was told after the one numbered `seen` (0 for none), and the
267    /// number of the last told, for the next ask.
268    #[must_use]
269    pub fn told_since(&self, seen: u64) -> (u64, Vec<String>) {
270        let board = self.board();
271        let told = board
272            .told
273            .iter()
274            .filter(|(number, _)| *number > seen)
275            .map(|(_, said)| said.clone())
276            .collect();
277        (board.told_last, told)
278    }

320    /// A terminal asked to start or show the character `name` -- `GAME:Name`,
321    /// or its name -- and shows it itself once it is seated: nothing else is
322    /// opened for it, unless its own *Shown in* says so.
323    pub fn await_in_terminal(&self, name: &str) {
324        let mut board = self.board();
325        let now = Instant::now();
326        board
327            .awaited
328            .retain(|(_, at)| now.saturating_duration_since(*at) < AWAIT_FOR);
329        board.awaited.push((name.to_owned(), now));
330    }

332    /// Whether a terminal waits for `character` on `game`, as asked lately.
333    #[must_use]
334    pub fn awaited_in_terminal(&self, game: &str, character: &str) -> bool {
335        let now = Instant::now();
336        self.board().awaited.iter().any(|(name, at)| {
337            now.saturating_duration_since(*at) < AWAIT_FOR && names(name, game, character)
338        })
339    }

528    /// Ask the binary for `request` on `runtime`; its answer is shown once
529    /// it comes, in the hub and on the web page. Nothing answers before the
530    /// binary has said what does. The tray asks as the hub does.
531    pub fn ask(&self, request: HubRequest, runtime: &tokio::runtime::Handle) {
532        let Some(control) = self.board().control.clone() else {
533            return;
534        };
535        let sessions = self.clone();
536        runtime.spawn(async move {
537            let said = control(request).await;
538            // An answer of nothing is shown as nothing: the pages a menu
539            // asked for are their own answer.
540            if !said.is_empty() {
541                sessions.board().said = Some((said, Instant::now()));
542            }
543            sessions.wake.hub();
544        });
545    }

553            offered: board.offered.clone(),
554            roster: board.roster.clone(),

558                .said

560                .filter(|(_, at)| now.saturating_duration_since(*at) < SAID_FOR)

563            web_at: board.web_at.clone(),

569/// Whether `name`, as a character was asked for, names `character` on
570/// `game`: `GAME:Name` or the name alone, whatever its case.
571fn names(name: &str, game: &str, character: &str) -> bool {
572    match name.split_once(':') {
573        Some((asked_game, asked)) => {
574            asked_game.eq_ignore_ascii_case(game) && asked.eq_ignore_ascii_case(character)
575        }
576        None => name.eq_ignore_ascii_case(character),
577    }
578}

1//! The highlights every character's story looks through, shared between the
2//! window, which reads the file when it changes, and each character's feed,
3//! which builds its own from them; and the desk that plays their sounds and
4//! shows their notifications, and what Hydra calls attention to beside them:
5//! an Ebon Gate merchant opening, told once however many characters hear it.
6//!
7//! The feed runs whether the character's window is open or not, so a
8//! character nobody is watching still sounds.
9

18/// One call for attention, and whose.
19#[derive(Clone, Debug, PartialEq, Eq)]
20pub struct Called {
21    /// The character whose highlight called.
22    pub character: String,
23    /// What it called for.
24    pub call: Call,
25    /// The window's volume, a percent, when the call was made.
26    pub master: u8,
27}

29/// What plays a highlight's sounds and shows its notifications, given each
30/// call: the window's, which has the speakers and the desktop.
31pub type Desk = Arc<dyn Fn(Called) + Send + Sync>;
32
33/// The highlights, shared: clones are the same.
34#[derive(Clone, Default)]
35pub struct Highlights(Arc<Shared>);
36
37struct Shared {
38    /// How many times the highlights have been set; 0, never.
39    generation: AtomicU64,
40    file: Mutex<Arc<HighlightFile>>,
41    /// The highlights as last built, and the generation they are: held
42    /// by the characters' matchers alone, so with no character left they
43    /// go.
44    compiled: Mutex<Option<(u64, Weak<Compiled>)>>,
45    desk: OnceLock<Desk>,
46    /// The window's volume, a percent, which every sound is scaled by.
47    volume: AtomicU8,
48    /// Until when what the highlights call for is let go of rather than
49    /// done: the text a phone catches up on as it wakes.
50    hushed_until: Mutex<Option<Instant>>,
51    /// How many terminals are covered by work mode: above none, and with
52    /// no window open, a sound or a notification would give the game away
53    /// at a desk, so the desk is not called.
54    covered: AtomicUsize,
55    /// Whether an Ebon Gate merchant opening shows a banner in each
56    /// character's window: Hydra's General page's.
57    merchant_banners: AtomicBool,
58    /// Whether it is told by a notification from the computer, once.
59    merchant_notices: AtomicBool,
60    /// The calls made once for every character: each one's key, and until
61    /// when another by it is let go of.
62    once: Mutex<Vec<(Instant, String)>>,
63    /// The data folder, whose `push.json` says the phones a highlight's
64    /// push goes to: set as the highlights are first read.
65    data: OnceLock<PathBuf>,
66    /// Held across one change's read, edit and setting: two changes at once
67    /// each start from what the other left, and neither is lost.
68    changing: Mutex<()>,
69}
70
71impl Default for Shared {
72    fn default() -> Self {
73        Self {
74            generation: AtomicU64::default(),
75            file: Mutex::default(),
76            compiled: Mutex::default(),
77            desk: OnceLock::new(),
78            volume: AtomicU8::new(100),
79            hushed_until: Mutex::default(),
80            covered: AtomicUsize::new(0),
81            merchant_banners: AtomicBool::new(true),
82            merchant_notices: AtomicBool::new(true),
83            once: Mutex::default(),
84            data: OnceLock::new(),
85            changing: Mutex::default(),
86        }
87    }
88}

98impl Highlights {
99    /// The data folder, for a highlight's push to phone; none before the
100    /// highlights are first read.
101    #[must_use]
102    pub fn data(&self) -> Option<&Path> {
103        self.0.data.get().map(PathBuf::as_path)
104    }
105
106    /// Every character's feed builds its own from `file` at its next line.
107    pub fn set(&self, file: HighlightFile) {
108        *self.0.file.lock().unwrap_or_else(PoisonError::into_inner) = Arc::new(file);
109        self.0.generation.fetch_add(1, Ordering::Release);
110    }
111
112    /// `change` made at once to the highlights every story looks through,
113    /// and to the file in the data folder `dir` with the next batch of saves
114    /// (`crate::store`), holding it as `.highlight` does. What it said.
115    ///
116    /// # Errors
117    ///
118    /// Why `change` refused; nothing is changed.
119    pub fn change<F>(&self, dir: &Path, change: F) -> Result<String, String>
120    where
121        F: Fn(&mut HighlightFile) -> Result<String, String> + Send + 'static,
122    {
123        let said = {
124            let _one_at_a_time = self
125                .0
126                .changing
127                .lock()
128                .unwrap_or_else(PoisonError::into_inner);
129            let mut file = (*self.file()).clone();
130            let said = change(&mut file)?;
131            self.set(file);
132            said
133        };
134        let path = dir.join(hydra_ui::highlight::FILE);
135        crate::store::edit_locked(
136            dir,
137            &path,
138            Box::new(move |text: &str| {
139                let mut file = HighlightFile::parse(text)?;
140                change(&mut file)?;
141                Ok(file.to_toml())
142            }),
143        );
144        Ok(said)
145    }
146
147    /// How many times they have been set.
148    #[must_use]
149    pub fn generation(&self) -> u64 {
150        self.0.generation.load(Ordering::Acquire)
151    }
152
153    /// The highlights `character` (`GAME:Name`) has now, built, and the
154    /// generation they are. Every character's are of the one built set: a
155    /// highlight several characters have is held once.
156    pub fn for_character(&self, character: &str) -> (u64, Arc<Matcher>) {
157        let generation = self.generation();
158        let compiled = {
159            let mut kept = self
160                .0
161                .compiled
162                .lock()
163                .unwrap_or_else(PoisonError::into_inner);
164            let built = kept
165                .as_ref()
166                .filter(|(of, _)| *of == generation)
167                .and_then(|(_, compiled)| compiled.upgrade());
168            built.unwrap_or_else(|| {
169                let compiled = Arc::new(Compiled::new(&self.file()).0);
170                *kept = Some((generation, Arc::downgrade(&compiled)));
171                compiled
172            })
173        };
174        (generation, Arc::new(compiled.matcher(character)))
175    }
176
177    /// The highlights file as last set.
178    #[must_use]
179    pub fn file(&self) -> Arc<HighlightFile> {
180        Arc::clone(&self.0.file.lock().unwrap_or_else(PoisonError::into_inner))
181    }
182
183    /// The desk `start` makes, the first time: a second is not made.
184    pub fn start_desk(&self, start: impl FnOnce() -> Desk) {
185        if self.0.desk.get().is_none() {
186            let _ = self.0.desk.set(start());
187        }
188    }
189
190    /// Every sound from now at `volume` percent of its own loudness:
191    /// Hydra's General page's.
192    pub fn set_volume(&self, volume: u8) {
193        self.0.volume.store(volume.min(100), Ordering::Relaxed);
194    }
195
196    /// What the highlights call for, sends, sounds and rumbles alike, let
197    /// go of for `long` from now: the text that comes as a phone wakes is
198    /// shown, and nothing it says is done twice or late.
199    pub fn hush_for(&self, long: Duration) {
200        *self
201            .0
202            .hushed_until
203            .lock()
204            .unwrap_or_else(PoisonError::into_inner) = Some(Instant::now() + long);
205    }
206
207    /// A terminal is covered by work mode.
208    pub fn cover(&self) {
209        self.0.covered.fetch_add(1, Ordering::Relaxed);
210    }
211
212    /// A terminal covered by work mode is uncovered, or gone.
213    pub fn uncover(&self) {
214        let _ = self
215            .0
216            .covered
217            .fetch_update(Ordering::Relaxed, Ordering::Relaxed, |n| {
218                Some(n.saturating_sub(1))
219            });
220    }
221
222    /// How many terminals are covered by work mode.
223    #[must_use]
224    pub fn covered(&self) -> usize {
225        self.0.covered.load(Ordering::Relaxed)
226    }
227
228    /// Whether what the highlights call for is let go of now.
229    #[must_use]
230    pub fn hushed(&self) -> bool {
231        self.0
232            .hushed_until
233            .lock()
234            .unwrap_or_else(PoisonError::into_inner)
235            .is_some_and(|until| Instant::now() < until)
236    }

238    /// Whether an Ebon Gate merchant opening shows a banner in each
239    /// character's window, and a notification from the computer: the Window
240    /// page's. Both are on until set.
241    pub fn set_merchants(&self, banners: bool, notices: bool) {
242        self.0.merchant_banners.store(banners, Ordering::Relaxed);
243        self.0.merchant_notices.store(notices, Ordering::Relaxed);
244    }
245
246    /// Whether a merchant opening shows a banner in each character's window.
247    #[must_use]
248    pub fn merchant_banners(&self) -> bool {
249        self.0.merchant_banners.load(Ordering::Relaxed)
250    }
251
252    /// Whether a merchant opening is told by a notification from the
253    /// computer.
254    #[must_use]
255    pub fn merchant_notices(&self) -> bool {
256        self.0.merchant_notices.load(Ordering::Relaxed)
257    }
258
259    /// `character` called for `call`, unless a call by `key` was made in the
260    /// last `within`, for this character or another: one thing the game
261    /// tells every character, called for once. Whether it was made.
262    pub fn call_once(&self, key: &str, within: Duration, character: &str, call: Call) -> bool {
263        let now = Instant::now();
264        {
265            let mut once = self.0.once.lock().unwrap_or_else(PoisonError::into_inner);
266            once.retain(|(until, _)| *until > now);
267            if once.iter().any(|(_, made)| made == key) {
268                return false;
269            }
270            once.push((now + within, key.to_owned()));
271        }
272        self.call(character, call);
273        true
274    }
275
276    /// `character`'s highlight called for `call`.
277    pub fn call(&self, character: &str, call: Call) {
278        if let Some(desk) = self.0.desk.get() {
279            desk(Called {
280                character: character.to_owned(),
281                call,
282                master: self.0.volume.load(Ordering::Relaxed),
283            });
284        }
285    }

288/// The highlights file as the window last read it: off its thread
289/// (`crate::disk`), once a second at most.
290#[derive(Debug, Default)]
291pub struct Watched {
292    /// The read last set, by its generation; `None`, none yet, or one is
293    /// wanted at once.
294    seen: Option<u64>,
295}
296
297impl Watched {
298    /// Read the highlights file in `dir` into `highlights` when it has come,
299    /// gone or changed, once a second at most; `repaint` drawn again when a
300    /// read is made. What to tell the player: why it did not read, or which
301    /// highlights cannot be used.
302    pub fn read_again(
303        &mut self,
304        dir: &Path,
305        highlights: &Highlights,
306        repaint: Option<&crate::disk::Repaint>,
307    ) -> Option<String> {
308        let path: PathBuf = dir.join(hydra_ui::highlight::FILE);
309        let key = format!("highlights:{}", path.display());
310        let disk = crate::disk::Disk::hydras();
311        if self.seen.is_none() {
312            disk.look_again(&key);
313        }
314        let owned = dir.to_owned();
315        let (generation, read) = disk.kept(&key, &path, repaint, move |_| {
316            Some(HighlightFile::load(&owned))
317        })?;
318        // A change of the editor's waits to be written: the file is not
319        // taken over it.
320        if self.seen == Some(generation) || crate::store::pending(&path) {
321            return None;
322        }
323        self.seen = Some(generation);
324        set_read(read, highlights)
325    }
326}

328/// The highlights file in `dir`, the data folder, set into `highlights`
329/// once, before any character starts: a Hydra without a window reads it
330/// here or not at all; a window reads it again as it changes
331/// ([`Watched::read_again`]). What to tell the player, as a read says it.
332#[must_use]
333pub fn load(dir: &Path, highlights: &Highlights) -> Option<String> {
334    let _ = highlights.0.data.set(dir.to_owned());
335    set_read(HighlightFile::load(dir), highlights)
336}
337
338/// `read`, the highlights file as it came, set into `highlights`: what to
339/// tell the player, why it did not read, or which highlights cannot be
340/// used.
341fn set_read(read: Result<HighlightFile, String>, highlights: &Highlights) -> Option<String> {
342    match read {
343        Ok(file) => {
344            let problems = Matcher::check(&file);
345            highlights.set(file);
346            (!problems.is_empty()).then(|| {
347                let more = problems.len().saturating_sub(3);
348                format!(
349                    "{} trigger(s) not used: {}{}",
350                    problems.len(),
351                    problems[..problems.len().min(3)].join("; "),
352                    if more > 0 {
353                        format!("; and {more} more")
354                    } else {
355                        String::new()
356                    }
357                )
358            })
359        }
360        Err(why) => Some(format!("The triggers were not read: {why}")),
361    }
362}

382    /// Changes made at once from two places each keep the other's: the
383    /// highlights every story looks through hold both.
384    #[test]
385    fn changes_at_once_lose_neither() {
386        let dir =
387            std::env::temp_dir().join(format!("hydra-highlights-race-{}", std::process::id()));
388        let _ = std::fs::remove_dir_all(&dir);
389        std::fs::create_dir_all(&dir).expect("the folder");
390        for round in 0..50 {
391            let highlights = Highlights::default();
392            let threads: Vec<_> = ["a", "b"]
393                .into_iter()
394                .map(|name| {
395                    let (highlights, dir) = (highlights.clone(), dir.clone());
396                    std::thread::spawn(move || {
397                        highlights
398                            .change(&dir, move |file| {
399                                // Long enough between the read and the setting
400                                // for the other to come in, unguarded.
401                                std::thread::sleep(std::time::Duration::from_millis(1));
402                                file.rules
403                                    .insert(name.to_owned(), hydra_ui::highlight::Rule::default());
404                                Ok(String::new())
405                            })
406                            .expect("changed");
407                    })
408                })
409                .collect();
410            for thread in threads {
411                thread.join().expect("ran");
412            }
413            assert_eq!(highlights.file().rules.len(), 2, "round {round}");
414        }
415        let _ = std::fs::remove_dir_all(&dir);
416    }

418    /// Every character's highlights are of one built set, each character's
419    /// its own of it; held only while a character holds it; built again
420    /// once the file changes.
421    #[test]
422    fn every_character_shares_one_built_set() {
423        use hydra_ui::StyledRun;
424        use hydra_ui::highlight::Rule;
425
426        let highlights = Highlights::default();
427        let mut file = HighlightFile::default();
428        file.rules.insert(
429            "everyone's".to_owned(),
430            Rule {
431                text: Some("kobold".to_owned()),
432                color: Some("#ff0000".to_owned()),
433                ..Rule::default()
434            },
435        );
436        file.rules.insert(
437            "Ashryn's".to_owned(),
438            Rule {
439                text: Some("Ashryn".to_owned()),
440                color: Some("#00ff00".to_owned()),
441                characters: vec!["Ashryn".to_owned()],
442                ..Rule::default()
443            },
444        );
445        highlights.set(file.clone());
446        let held = || {
447            highlights
448                .0
449                .compiled
450                .lock()
451                .unwrap()
452                .as_ref()
453                .map_or(0, |(_, compiled)| compiled.strong_count())
454        };
455        let coloured = |matcher: &Matcher, line: &str| {
456            matcher
457                .respond(&[StyledRun::plain(line)])
458                .runs
459                .iter()
460                .any(|run| run.color.is_some())
461        };
462
463        let (_, ashryn) = highlights.for_character("GS3:Ashryn");
464        let (_, dicate) = highlights.for_character("GS3:Dicate");
465        assert_eq!(held(), 2, "one set, both characters'");
466        assert!(coloured(&ashryn, "a kobold") && coloured(&dicate, "a kobold"));
467        assert!(coloured(&ashryn, "Ashryn waves."));
468        assert!(!coloured(&dicate, "Ashryn waves."), "Ashryn's own");
469
470        drop((ashryn, dicate));
471        assert_eq!(held(), 0, "let go with the characters");
472
473        highlights.set(file);
474        let (generation, _ashryn) = highlights.for_character("GS3:Ashryn");
475        assert_eq!(generation, 2);
476        assert_eq!(held(), 1, "built again for the new file");
477    }

1//! The performance log: while *Performance* is on -- switched from the
2//! windows' hub or a terminal's, one switch for both -- each live
3//! character's stages (`hydra_ui::perf`) every [`EVERY`], a row each, in
4//! `perf/<date>.csv` in the data folder, for diagnosing a slow window or
5//! terminal. The board's upkeep keeps it, whichever is open.
6//!
7//! A row says what happened since the character's last: how many game and
8//! Lich lines and typed lines, and each stage's count, median and 95th
9//! percentile over its samples since. So a burst -- a key held on `look` --
10//! is its own rows, not folded into all that came before.
11
12use std::io::Write as _;
13use std::path::Path;
14use std::sync::Arc;
15use std::time::{Duration, Instant};
16
17use hydra_session::State;
18use hydra_session::lich::View;

20use crate::seat::Seat;

22/// How often a row is written for each character.
23pub const EVERY: Duration = Duration::from_secs(10);
24
25/// When the log last wrote, while it writes.
26#[derive(Debug, Default)]
27pub struct PerfLog {
28    since: Option<Instant>,
29}
30
31impl PerfLog {
32    /// The log, on or off, at `now`: with it on, each of `seats`' rows once
33    /// [`EVERY`] has passed, in `data`. When the next is due, while it is on.
34    pub fn tick(
35        &mut self,
36        on: bool,
37        data: &Path,
38        seats: &[Arc<Seat>],
39        now: Instant,
40    ) -> Option<Duration> {
41        if !on {
42            if self.since.take().is_some() {
43                // Switched off: nothing more is kept for it.
44                for seat in seats {
45                    collect(&mut seat.story(), false);
46                }
47            }
48            return None;
49        }
50        let Some(since) = self.since else {
51            // Switched on: what came before is not this log's, and from now
52            // every sample is kept for its rows.
53            for seat in seats {
54                let mut story = seat.story();
55                let story = &mut *story;
56                let (lich, view) = (story.lich_running(), story.view());
57                let _ = story
58                    .perf
59                    .log_row(lich, view_name(view), &mut story.terminal);
60                collect(story, true);
61            }
62            self.since = Some(now);
63            return Some(EVERY);
64        };
65        // A character seated since the log went on collects too.
66        for seat in seats {
67            collect(&mut seat.story(), true);
68        }
69        let waited = now.saturating_duration_since(since);
70        if waited < EVERY {
71            return Some(EVERY.saturating_sub(waited));
72        }
73        self.since = Some(now);
74        let time = hydra_net::line_time();
75        let rows: Vec<String> = seats
76            .iter()
77            .filter(|seat| seat.lifecycle() != State::Closed)
78            .map(|seat| {
79                let mut story = seat.story();
80                let story = &mut *story;
81                let (lich, view) = (story.lich_running(), story.view());
82                let row = story
83                    .perf
84                    .log_row(lich, view_name(view), &mut story.terminal);
85                format!("{time},{}:{},{row}", seat.game, seat.name)
86            })
87            .collect();
88        let (folder, day) = (data.join("perf"), hydra_net::date_dir());
89        let written = move || {
90            if let Err(why) = write(&folder, &day, &rows) {
91                eprintln!("[perf] not written: {why}");
92            }
93        };
94        // Off the upkeep's thread; under test, at once.
95        if cfg!(test) {
96            written();
97        } else {
98            std::thread::spawn(move || hydra_mem::part(hydra_mem::Part::WindowThreads, written));
99        }
100        Some(EVERY)
101    }
102}

104/// `story`'s samples, the window's and a terminal's, kept for the log's
105/// rows from now, or not.
106fn collect(story: &mut hydra_ui::Story, on: bool) {
107    story.perf.collect(on);
108    story.terminal.collect(on);
109}
110
111/// A view as the log names it.
112const fn view_name(view: View) -> &'static str {
113    match view {
114        View::Lich => "lich",
115        View::Game => "game",
116    }
117}
118
119/// Append `rows` to `<folder>/<day>.csv`, its header first when it is new.
120fn write(folder: &Path, day: &str, rows: &[String]) -> std::io::Result<()> {
121    if rows.is_empty() {
122        return Ok(());
123    }
124    std::fs::create_dir_all(folder)?;
125    let path = folder.join(format!("{day}.csv"));
126    let new = !path.exists();
127    let mut file = std::fs::OpenOptions::new()
128        .create(true)
129        .append(true)
130        .open(path)?;
131    let mut text = String::new();
132    if new {
133        text.push_str("time,character,");
134        text.push_str(&hydra_ui::perf::Perf::log_header());
135        text.push('\n');
136    }
137    for row in rows {
138        text.push_str(row);
139        text.push('\n');
140    }
141    file.write_all(text.as_bytes())
142}

144#[cfg(test)]
145mod tests {
146    use super::*;
147
148    /// Switched on, nothing before is logged; a row each live character
149    /// once its time has come, under a header written once.
150    #[test]
151    fn a_row_each_character_every_ten_seconds() {
152        let data = std::env::temp_dir().join(format!("hydra-perf-{}", std::process::id()));
153        let _ = std::fs::remove_dir_all(&data);
154        let (seat, _lines) = Seat::idle(1, "Ashryn");
155        seat.story().perf.game_lines = 99;
156        let seats = vec![seat];
157        let mut log = PerfLog::default();
158        let start = Instant::now();
159        assert_eq!(log.tick(true, &data, &seats, start), Some(EVERY));
160        seats[0].story().perf.game_lines = 104;
161        assert_eq!(
162            log.tick(true, &data, &seats, start + Duration::from_secs(4)),
163            Some(Duration::from_secs(6))
164        );
165        log.tick(true, &data, &seats, start + EVERY);
166        log.tick(true, &data, &seats, start + EVERY * 2);
167        let written = std::fs::read_to_string(
168            data.join("perf")
169                .join(format!("{}.csv", hydra_net::date_dir())),
170        )
171        .unwrap();
172        let lines: Vec<&str> = written.lines().collect();
173        assert_eq!(lines.len(), 3, "{written}");
174        assert!(lines[0].starts_with("time,character,lich,view,game_lines"));
175        assert!(lines[1].contains(",GS3:Ashryn,off,lich,5,"), "{}", lines[1]);
176        assert!(lines[2].contains(",GS3:Ashryn,off,lich,0,"), "{}", lines[2]);
177        assert_eq!(log.tick(false, &data, &seats, start + EVERY * 3), None);
178        let _ = std::fs::remove_dir_all(&data);
179    }
180}

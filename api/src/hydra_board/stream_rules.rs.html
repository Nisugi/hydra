1//! The stream rules every character's session sorts its lines by
2//! (`hydra_session::stream_rules`), held by the board for every session it
3//! starts, and read from their file in the data folder again when it
4//! changes, once a second at most, off the window's thread
5//! ([`crate::disk`]). An editor's change ([`change`]) is made to them at
6//! once, and to the file with the next batch of saves.
7//!
8//! A highlight that sends its line to another stream, from a file written
9//! before the stream rules or an import, is made a stream rule as the
10//! highlights come ([`Folding`]): the rules alone move lines.
11
12use std::path::{Path, PathBuf};
13
14use hydra_session::stream_rules::{self, Rule, Rules, StreamRules};
15
16use crate::highlights::Highlights;
17
18/// The rules file, watched.
19#[derive(Debug, Default)]
20pub struct Watched {
21    /// The read last set, by its generation; `None`, none yet, or one is
22    /// wanted at once.
23    seen: Option<u64>,
24}
25
26impl Watched {
27    /// Read the rules file in `dir` into `rules` when it has come, gone or
28    /// changed, once a second at most; `repaint` drawn again when a read is
29    /// made. What to tell the player: why it did not read, or which rules
30    /// cannot be used.
31    pub fn read_again(
32        &mut self,
33        dir: &Path,
34        rules: &StreamRules,
35        repaint: Option<&crate::disk::Repaint>,
36    ) -> Option<String> {
37        let path: PathBuf = dir.join(stream_rules::FILE);
38        let key = format!("stream rules:{}", path.display());
39        let disk = crate::disk::Disk::hydras();
40        if self.seen.is_none() {
41            disk.look_again(&key);
42        }
43        let owned = dir.to_owned();
44        let (generation, read) =
45            disk.kept(&key, &path, repaint, move |_| Some(Rules::load(&owned)))?;
46        if self.seen == Some(generation) || crate::store::pending(&path) {
47            return None;
48        }
49        self.seen = Some(generation);
50        match read {
51            Ok(file) => said(&rules.set(&file)),
52            Err(why) => Some(format!("The stream rules were not read: {why}")),
53        }
54    }
55}
56
57/// The stream rules in `dir`, the data folder, set into `rules`: what to
58/// tell the player, as [`Watched::read_again`] says it.
59#[must_use]
60pub fn load(dir: &Path, rules: &StreamRules) -> Option<String> {
61    match Rules::load(dir) {
62        Ok(file) => said(&rules.set(&file)),
63        Err(why) => Some(format!("The stream rules were not read: {why}")),
64    }
65}

67/// `change` made to `rules`, those every session sorts by, at once, and to
68/// their file in `dir`, the data folder, with the next batch of saves: the
69/// file as it is then, or the rules built in while there is none. What to
70/// say of it, and of the rules that cannot be used.
71///
72/// # Errors
73///
74/// Why it cannot be made: nothing is changed.
75pub fn change<F>(dir: &Path, rules: &StreamRules, change: F) -> Result<String, String>
76where
77    F: Fn(&mut Rules) -> Result<String, String> + Send + 'static,
78{
79    let mut now = rules.current().rules().clone();
80    let done = change(&mut now)?;
81    let problems = rules.set(&now);
82    let path = dir.join(stream_rules::FILE);
83    let none = !path.exists();
84    crate::store::edit(
85        dir,
86        &path,
87        Box::new(move |text: &str| {
88            let mut kept = if none && text.trim().is_empty() {
89                Rules::default()
90            } else {
91                Rules::parse(text)?
92            };
93            change(&mut kept)?;
94            kept.to_toml()
95        }),
96    );
97    Ok(match said(&problems) {
98        Some(more) => format!("{done} {more}"),
99        None => done,
100    })
101}

103/// The highlights' redirects, made stream rules as they come: each time the
104/// highlights change, from their file or an editor.
105#[derive(Debug, Default)]
106pub struct Folding {
107    /// The highlights' generation last looked at.
108    seen: Option<u64>,
109}
110
111impl Folding {
112    /// The redirects of `highlights` made stream rules among `rules`, their
113    /// files in `dir`, the data folder, when the highlights have changed
114    /// since last looked at ([`fold_redirects`]). What to tell the player.
115    pub fn fold(
116        &mut self,
117        dir: &Path,
118        highlights: &Highlights,
119        rules: &StreamRules,
120    ) -> Option<String> {
121        let generation = highlights.generation();
122        if self.seen == Some(generation) {
123            return None;
124        }
125        self.seen = Some(generation);
126        fold_redirects(dir, highlights, rules)
127    }
128}
129
130/// Every highlight's redirect made a stream rule, first among `rules`, in
131/// the highlights' names' order -- one the rules hold already is not made
132/// twice -- and taken off the highlight, a highlight that did nothing else
133/// going with it (`HighlightFile::redirects`). Both made at once and to
134/// their files in `dir`, the stream rules first, so a redirect is never
135/// lost. What to tell the player: which were made, or why none was.
136#[must_use]
137pub fn fold_redirects(dir: &Path, highlights: &Highlights, rules: &StreamRules) -> Option<String> {
138    let file = highlights.file();
139    if !file
140        .rules
141        .values()
142        .any(|rule| rule.redirect.is_some() || rule.copy)
143    {
144        return None;
145    }
146    let made = file.redirects();
147    let mut said = String::new();
148    if !made.is_empty() {
149        let adding = made.clone();
150        let added = change(dir, rules, move |rules| {
151            let new: Vec<Rule> = (adding.iter())
152                .filter(|rule| !rules.rules.contains(rule))
153                .cloned()
154                .collect();
155            rules.rules.splice(0..0, new);
156            Ok(String::new())
157        });
158        match added {
159            Ok(more) => said = more,
160            Err(why) => {
161                return Some(format!(
162                    "The triggers that move lines were not made stream rules: {why}"
163                ));
164            }
165        }
166    }
167    let taken = highlights.change(dir, |file| {
168        file.take_redirects();
169        Ok(String::new())
170    });
171    if let Err(why) = taken {
172        return Some(format!(
173            "The triggers that move lines are stream rules now, but kept their own: {why}"
174        ));
175    }
176    let names: Vec<&str> = made.iter().map(|rule| rule.name.as_str()).collect();
177    (!names.is_empty()).then(|| {
178        format!(
179            "{} trigger(s) that moved lines to another stream are stream rules now, for every \
180             character, on the Streams page: {}.{said}",
181            names.len(),
182            names.join(", ")
183        )
184    })
185}
186
187/// The rules that cannot be used, `problems`, as the player is told: the
188/// first three, and how many more.
189fn said(problems: &[String]) -> Option<String> {
190    (!problems.is_empty()).then(|| {
191        let more = problems.len().saturating_sub(3);
192        format!(
193            "{} stream rule(s) not used: {}{}",
194            problems.len(),
195            problems[..problems.len().min(3)].join("; "),
196            if more > 0 {
197                format!("; and {more} more")
198            } else {
199                String::new()
200            }
201        )
202    })
203}

209    /// The rules in the data folder are set into the rules every session
210    /// sorts by; one that cannot be used is said, and a file that is not
211    /// one leaves the rules as they were.
212    #[test]
213    fn the_rules_file_is_set_into_the_rules_shared() {
214        let dir = std::env::temp_dir().join(format!("hydra-board-rules-{}", std::process::id()));
215        let _ = std::fs::remove_dir_all(&dir);
216        std::fs::create_dir_all(&dir).expect("the folder");
217        let shared = StreamRules::default();
218        std::fs::write(
219            dir.join(stream_rules::FILE),
220            "[[rule]]\nname = \"Rocks\"\ntext = \"rock\"\nto = \"rocks\"\n\n\
221             [[rule]]\nname = \"Broken\"\nregex = \"(\"\nto = \"x\"\n",
222        )
223        .expect("write");
224        let told = load(&dir, &shared).expect("one not used");
225        assert!(
226            told.starts_with("1 stream rule(s) not used: Broken"),
227            "{told}"
228        );
229        let line = stream_rules::Line {
230            stream: "main",
231            text: "You see a rock.",
232            links: &[],
233            fighting: false,
234            marked: false,
235        };
236        assert_eq!(
237            shared
238                .current()
239                .sort(&line)
240                .map(|sorted| sorted.to.to_owned()),
241            Some("rocks".to_owned())
242        );
243        std::fs::write(dir.join(stream_rules::FILE), "not = [toml").expect("write");
244        assert!(
245            load(&dir, &shared).is_some_and(|told| told.contains("not read")),
246            "said"
247        );
248        assert!(shared.current().sort(&line).is_some(), "left as they were");
249        let _ = std::fs::remove_dir_all(&dir);
250    }

252    /// A highlight that moves its line is made a stream rule, first among
253    /// the rules, and taken off the highlight, which goes when it did
254    /// nothing else; both files say so once saved. Looked at again, nothing
255    /// more is made.
256    #[test]
257    fn a_highlights_redirect_is_made_a_stream_rule() {
258        use hydra_ui::highlight::{HighlightFile, Rule as Highlight};
259        let dir = std::env::temp_dir().join(format!("hydra-board-fold-{}", std::process::id()));
260        let _ = std::fs::remove_dir_all(&dir);
261        std::fs::create_dir_all(&dir).expect("the folder");
262        let mut file = HighlightFile::default();
263        file.rules.insert(
264            "lnet".to_owned(),
265            Highlight {
266                text: Some("[LNet]".to_owned()),
267                redirect: Some("lnet".to_owned()),
268                ..Highlight::default()
269            },
270        );
271        file.rules.insert(
272            "loot".to_owned(),
273            Highlight {
274                text: Some("silver".to_owned()),
275                color: Some("#ecc013".to_owned()),
276                redirect: Some("loot".to_owned()),
277                copy: true,
278                ..Highlight::default()
279            },
280        );
281        file.save(&dir).expect("written");
282        let highlights = Highlights::default();
283        highlights.set(file);
284        let rules = StreamRules::default();
285        let mut folding = Folding::default();
286        let said = folding.fold(&dir, &highlights, &rules).expect("told");
287        assert!(said.starts_with("2 trigger(s)"), "{said}");
288        let names: Vec<String> = (rules.current().rules().rules.iter())
289            .map(|rule| rule.name.clone())
290            .collect();
291        assert_eq!(names, ["lnet", "loot", "Combat"]);
292        let now = highlights.file();
293        assert!(!now.rules.contains_key("lnet"), "it did nothing else");
294        assert_eq!(now.rules["loot"].redirect, None);
295        assert_eq!(folding.fold(&dir, &highlights, &rules), None, "none left");
296        crate::store::flush();
297        let written = Rules::load(&dir).expect("the stream rules");
298        assert_eq!(written.rules.len(), 3);
299        let kept = HighlightFile::load(&dir).expect("the highlights");
300        assert_eq!(kept.rules.keys().collect::<Vec<_>>(), ["loot"]);
301        assert!(kept.rules["loot"].redirect.is_none());
302        let _ = std::fs::remove_dir_all(&dir);
303    }

305    /// A change is made to the rules every session sorts by at once, and
306    /// to the file once the batch is saved: with no file, to the rules
307    /// built in; a change that cannot be made changes nothing.
308    #[test]
309    fn a_change_is_made_at_once_and_to_the_file() {
310        let dir = std::env::temp_dir().join(format!("hydra-board-change-{}", std::process::id()));
311        let _ = std::fs::remove_dir_all(&dir);
312        std::fs::create_dir_all(&dir).expect("the folder");
313        let shared = StreamRules::default();
314        let rocks = stream_rules::Rule {
315            text: Some("rock".to_owned()),
316            regex: None,
317            ..stream_rules::Rule::like(Some("main"), "x", &[], "rocks")
318        };
319        let added = rocks.clone();
320        let said = change(&dir, &shared, move |rules| {
321            rules.rules.insert(0, added.clone());
322            Ok("Added.".to_owned())
323        })
324        .expect("made");
325        assert_eq!(said, "Added.");
326        assert_eq!(
327            shared.current().rules().rules,
328            [rocks.clone(), stream_rules::Rule::fighting()]
329        );
330        crate::store::flush();
331        let written = Rules::load(&dir).expect("written");
332        assert_eq!(written.rules, [rocks, stream_rules::Rule::fighting()]);
333        let refused = change(&dir, &shared, |_| Err("No.".to_owned()));
334        assert_eq!(refused, Err("No.".to_owned()));
335        assert_eq!(shared.current().rules().rules.len(), 2, "unchanged");
336        let _ = std::fs::remove_dir_all(&dir);
337    }

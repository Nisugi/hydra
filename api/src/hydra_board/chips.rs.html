1//! Effect chips: a chip is a small summary of the effects it gathers, a
2//! mark and a few words, coloured by how soon the first of them runs out.
3//! Each chip is a widget of its own, shaped by the player: which of the
4//! game's lists it gathers, whether the player's own timers and the
5//! effects tracked as missing, and any effect by name, so one spell can be
6//! a chip. The window and the page draw the same chips from the same
7//! layout file; the gathering here is what both show.
8

16/// Under this many seconds left, a chip is critical, by default.
17pub const CRITICAL_SECONDS: u32 = 30;
18/// Under this many seconds left, a chip warns, by default.
19pub const WARN_SECONDS: u32 = 120;
20
21/// What a chip says beside its mark.
22#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
23#[serde(rename_all = "snake_case")]
24pub enum Says {
25    /// How many are up, and the soonest to run out as a clock.
26    #[default]
27    Count,
28    /// The soonest to run out, as a clock.
29    Soonest,
30    /// The soonest to run out, by name, with its clock.
31    Name,
32}
33
34impl Says {
35    /// Every way, in the order a page lists them.
36    pub const ALL: [Says; 3] = [Says::Count, Says::Soonest, Says::Name];
37
38    /// As a layout file writes it.
39    #[must_use]
40    pub const fn as_str(self) -> &'static str {
41        match self {
42            Says::Count => "count",
43            Says::Soonest => "soonest",
44            Says::Name => "name",
45        }
46    }
47}

49/// What a chip does with nothing to show.
50#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
51#[serde(rename_all = "snake_case")]
52pub enum Empty {
53    /// Not drawn at all.
54    #[default]
55    Hidden,
56    /// Drawn with a count of none.
57    Zero,
58}

60/// How urgently a chip is coloured.
61#[derive(Clone, Copy, Debug, PartialEq, Eq)]
62pub enum Level {
63    /// Nothing soon.
64    Plain,
65    /// The soonest is under the warn seconds.
66    Warn,
67    /// The soonest is under the critical seconds.
68    Critical,
69}

71/// One chip, as the layout file keeps it.
72#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
73pub struct Chip {
74    /// What the player calls it; the widget's title.
75    pub name: String,
76    /// The game's lists it gathers, by their dialog ids: `Active Spells`,
77    /// `Buffs`, `Debuffs`, `Cooldowns`.
78    #[serde(default, skip_serializing_if = "Vec::is_empty")]
79    pub lists: Vec<String>,
80    /// The player's own timers, the ones filed with no game list.
81    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
82    pub timers: bool,
83    /// The effects tracked as missing that are not up.
84    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
85    pub missing: bool,
86    /// Effects by name or number, from any list: `640`, `Barkskin`.
87    #[serde(default, skip_serializing_if = "Vec::is_empty")]
88    pub named: Vec<String>,
89    /// The mark before the words: a symbol or an emoji.
90    #[serde(default)]
91    pub mark: String,
92    /// Its colour; none, the theme's accent.
93    #[serde(default, skip_serializing_if = "Option::is_none")]
94    pub color: Option<[u8; 3]>,
95    /// What it says beside the mark.
96    #[serde(default, skip_serializing_if = "is_default_says")]
97    pub says: Says,
98    /// Under this many seconds left on the soonest, it warns.
99    #[serde(default = "warn_seconds", skip_serializing_if = "is_warn_seconds")]
100    pub warn: u32,
101    /// Under this many seconds left on the soonest, it is critical.
102    #[serde(
103        default = "critical_seconds",
104        skip_serializing_if = "is_critical_seconds"
105    )]
106    pub critical: u32,
107    /// With nothing to show.
108    #[serde(default, skip_serializing_if = "is_default_empty")]
109    pub empty: Empty,
110}

170/// One effect a chip gathered.
171#[derive(Clone, Debug, PartialEq, Eq)]
172pub struct Row {
173    /// The list it came from, by its dialog id; `Timers` for the player's
174    /// own; `Missing` for one tracked as missing.
175    pub list: String,
176    /// As the game shows it, or the timer's name.
177    pub name: String,
178    /// Seconds left; none for one with no end, or no clock.
179    pub left: Option<u32>,
180    /// Up now. A named effect not up, or a missing one, is listed down.
181    pub up: bool,
182}

184/// What a chip gathered: its rows, up first and soonest first, and the
185/// soonest to run out.
186#[derive(Clone, Debug, Default, PartialEq, Eq)]
187pub struct Gathered {
188    /// Every effect gathered, the ones up first, soonest first.
189    pub rows: Vec<Row>,
190    /// Seconds left on the soonest effect that has an end.
191    pub soonest: Option<u32>,
192}
193
194impl Gathered {
195    /// How many are up.
196    #[must_use]
197    pub fn up(&self) -> usize {
198        self.rows.iter().filter(|row| row.up).count()
199    }
200}

202impl Chip {
203    /// The chips a layout file starts with, when none are placed: the two
204    /// the page has always shown, the good and the bad.
205    #[must_use]
206    pub fn defaults() -> Vec<Chip> {
207        vec![
208            Chip {
209                name: "Spells and buffs".to_owned(),
210                lists: vec!["Active Spells".to_owned(), "Buffs".to_owned()],
211                mark: "\u{2726}".to_owned(),
212                ..Chip::default()
213            },
214            Chip {
215                name: "Debuffs and cooldowns".to_owned(),
216                lists: vec!["Debuffs".to_owned(), "Cooldowns".to_owned()],
217                mark: "\u{26a0}".to_owned(),
218                ..Chip::default()
219            },
220        ]
221    }

223    /// Gather over `state` at the game's second `now`, with `missing` the
224    /// names tracked as missing that are not up (the window knows them;
225    /// the page is told). Each effect once: the listing with the most time
226    /// left when a list names one twice.
227    #[must_use]
228    pub fn gather(&self, state: &GameState, now: Option<u32>, missing: &[String]) -> Gathered {
229        let mut rows: Vec<Row> = Vec::new();
230        let mut put = |row: Row| match rows
231            .iter_mut()
232            .find(|had| had.name.eq_ignore_ascii_case(&row.name) && had.list == row.list)
233        {
234            Some(had) => {
235                if row.up && (!had.up || row.left > had.left) {
236                    *had = row;
237                }
238            }
239            None => rows.push(row),
240        };
241        for list in &self.lists {
242            for (id, effect) in state.effects.in_category(list) {
243                if now.is_some_and(|now| state.effects.active(id, now) == Some(false)) {
244                    continue;
245                }
246                put(Row {
247                    list: list.clone(),
248                    name: effect.text.clone(),
249                    left: now.and_then(|now| state.effects.remaining(id, now)),
250                    up: true,
251                });
252            }
253            // The player's own timers filed with the list.
254            if let Some(filed) = List::ALL
255                .into_iter()
256                .find(|l| l.game_id() == Some(list.as_str()))
257            {
258                for row in own_timers(state, now, filed, list) {
259                    put(row);
260                }
261            }
262        }
263        if self.timers {
264            for row in own_timers(state, now, List::Custom, "Timers") {
265                put(row);
266            }
267        }
268        if self.missing {
269            for name in missing {
270                put(Row {
271                    list: "Missing".to_owned(),
272                    name: name.clone(),
273                    left: None,
274                    up: false,
275                });
276            }
277        }
278        for wanted in &self.named {
279            put(named(state, now, wanted));
280        }
281        rows.sort_by_key(|row| (!row.up, row.left.map_or(u32::MAX, |left| left)));
282        let soonest = rows
283            .iter()
284            .filter(|row| row.up)
285            .filter_map(|row| row.left)
286            .min();
287        Gathered { rows, soonest }
288    }
289
290    /// The words beside the mark for `gathered`.
291    #[must_use]
292    pub fn words(&self, gathered: &Gathered) -> String {
293        let clock_of = |left: Option<u32>| left.map(clock).unwrap_or_default();
294        // One named effect alone: its clock, or that it is down.
295        if self.lists.is_empty() && !self.timers && !self.missing && self.named.len() == 1 {
296            return match gathered.rows.first() {
297                Some(row) if row.up => match row.left {
298                    Some(left) => clock(left),
299                    None => "up".to_owned(),
300                },
301                _ => "down".to_owned(),
302            };
303        }
304        match self.says {
305            Says::Count => {
306                let count = gathered.up();
307                match gathered.soonest {
308                    Some(left) => format!("{count} \u{b7} {}", clock(left)),
309                    None => count.to_string(),
310                }
311            }
312            Says::Soonest => clock_of(gathered.soonest),
313            Says::Name => gathered
314                .rows
315                .iter()
316                .find(|row| row.up && row.left == gathered.soonest)
317                .map(|row| match row.left {
318                    Some(left) => format!("{} {}", row.name, clock(left)),
319                    None => row.name.clone(),
320                })
321                .unwrap_or_default(),
322        }
323    }
324
325    /// How urgently `gathered` colours the chip.
326    #[must_use]
327    pub fn level(&self, gathered: &Gathered) -> Level {
328        match gathered.soonest {
329            Some(left) if left < self.critical => Level::Critical,
330            Some(left) if left < self.warn => Level::Warn,
331            _ => Level::Plain,
332        }
333    }

335    /// Whether the chip is drawn at all for `gathered`.
336    #[must_use]
337    pub fn shown(&self, gathered: &Gathered) -> bool {
338        self.empty == Empty::Zero || gathered.rows.iter().any(|row| row.up)
339    }

341    /// Every chip the layout file keeps for `character` on `instance`, in
342    /// the order placed; the defaults when it keeps none. The file's
343    /// `chips` map is by widget id, as the window writes it.
344    #[must_use]
345    pub fn load(layouts: &Path, instance: Option<&str>, character: &str) -> Vec<Chip> {
346        let read = |path: std::path::PathBuf| -> Option<Vec<Chip>> {
347            let text = std::fs::read_to_string(path).ok()?;
348            let layout: serde_json::Value = serde_json::from_str(&text).ok()?;
349            Self::in_layout(&layout)
350        };
351        read(crate::kept::file(layouts, instance, character))
352            .or_else(|| instance.and_then(|_| read(crate::kept::file(layouts, None, character))))
353            .unwrap_or_else(Self::defaults)
354    }
355
356    /// The chips `layout` keeps, by the widget ids' order; none when it has
357    /// no `chips` map or an empty one.
358    fn in_layout(layout: &serde_json::Value) -> Option<Vec<Chip>> {
359        let map: BTreeMap<u32, Chip> = serde_json::from_value(layout.get("chips")?.clone()).ok()?;
360        (!map.is_empty()).then(|| map.into_values().collect())
361    }
362}

364/// The player's own timers filed with `list`, running at `now`, as rows of
365/// `label`.
366fn own_timers(state: &GameState, now: Option<u32>, list: List, label: &str) -> Vec<Row> {
367    let Some(now) = now else {
368        return Vec::new();
369    };
370    state
371        .custom_timers
372        .timers()
373        .filter(|(_, timer)| timer.list == list)
374        .filter_map(|(name, timer)| {
375            let left = timer.ends.checked_sub(now).filter(|left| *left > 0)?;
376            Some(Row {
377                list: label.to_owned(),
378                name: name.to_owned(),
379                left: Some(left),
380                up: true,
381            })
382        })
383        .collect()
384}

386/// The effect `wanted` names, by its text or its number in any list, a
387/// timer by its name, or the bark by its lines: up with its time left, or
388/// listed down.
389fn named(state: &GameState, now: Option<u32>, wanted: &str) -> Row {
390    let wanted_trim = wanted.trim();
391    let is_bark = wanted_trim == "605" || wanted_trim.eq_ignore_ascii_case("Barkskin");
392    let listed = state
393        .effects
394        .iter()
395        .filter(|(id, effect)| *id == wanted_trim || effect.text.eq_ignore_ascii_case(wanted_trim))
396        .filter(|(id, _)| now.is_none_or(|now| state.effects.active(id, now) != Some(false)))
397        .map(|(id, effect)| Row {
398            list: effect.category.clone(),
399            name: effect.text.clone(),
400            left: now.and_then(|now| state.effects.remaining(id, now)),
401            up: true,
402        })
403        .max_by_key(|row| row.left);
404    if let Some(row) = listed {
405        return row;
406    }
407    if let (Some(now), Some(timer)) = (now, state.custom_timers.timer(wanted_trim))
408        && let Some(left) = timer.ends.checked_sub(now).filter(|left| *left > 0)
409    {
410        return Row {
411            list: timer.list.game_id().unwrap_or("Timers").to_owned(),
412            name: wanted_trim.to_owned(),
413            left: Some(left),
414            up: true,
415        };
416    }
417    if is_bark && state.bark_up() {
418        return Row {
419            list: "Active Spells".to_owned(),
420            name: "Barkskin".to_owned(),
421            left: None,
422            up: true,
423        };
424    }
425    Row {
426        list: String::new(),
427        name: if is_bark {
428            "Barkskin".to_owned()
429        } else {
430            wanted_trim.to_owned()
431        },
432        left: None,
433        up: false,
434    }
435}

437/// Seconds as a clock: `0:45`, `12:03`, `1:02:03`.
438#[must_use]
439pub fn clock(secs: u32) -> String {
440    let (h, m, s) = (secs / 3600, secs % 3600 / 60, secs % 60);
441    if h > 0 {
442        format!("{h}:{m:02}:{s:02}")
443    } else {
444        format!("{m}:{s:02}")
445    }
446}

462    /// A chip over two lists counts what is up once each, says the
463    /// soonest, and colours by its own seconds; one over a single name says
464    /// its clock or that it is down.
465    #[test]
466    fn a_chip_gathers_its_lists_and_a_named_one_says_its_clock() {
467        let mut state = GameState::default();
468        state.effects.insert(
469            "640".to_owned(),
470            effect("Active Spells", "Wall of Thorns", Some(1_100)),
471        );
472        state
473            .effects
474            .insert("506".to_owned(), effect("Buffs", "Celerity", Some(1_045)));
475        state
476            .effects
477            .insert("9000".to_owned(), effect("Buffs", "Celerity", Some(1_020)));
478        state
479            .effects
480            .insert("77".to_owned(), effect("Debuffs", "Stunned", Some(1_010)));
481        let chip = Chip {
482            lists: vec!["Active Spells".to_owned(), "Buffs".to_owned()],
483            ..Chip::default()
484        };
485        let got = chip.gather(&state, Some(1_000), &[]);
486        assert_eq!(got.up(), 2, "{got:?}");
487        assert_eq!(
488            got.soonest,
489            Some(45),
490            "the longer Celerity listing is the one kept"
491        );
492        assert_eq!(got.rows[0].name, "Celerity");
493        assert_eq!(chip.words(&got), "2 \u{b7} 0:45");
494        assert_eq!(chip.level(&got), Level::Warn);
495        let tight = Chip {
496            warn: 40,
497            ..chip.clone()
498        };
499        assert_eq!(tight.level(&got), Level::Plain);
500        let one = Chip {
501            named: vec!["640".to_owned()],
502            ..Chip::default()
503        };
504        let got = one.gather(&state, Some(1_000), &[]);
505        assert_eq!(one.words(&got), "1:40");
506        let down = Chip {
507            named: vec!["Barkskin".to_owned()],
508            ..Chip::default()
509        };
510        let got = down.gather(&state, Some(1_000), &[]);
511        assert_eq!(down.words(&got), "down");
512        assert!(!down.shown(&got));
513        assert!(
514            Chip {
515                empty: Empty::Zero,
516                ..down.clone()
517            }
518            .shown(&got)
519        );
520    }

538    /// A layout file's chips come back as written, by widget id; a file
539    /// with none gives the two the page always had.
540    #[test]
541    fn chips_round_trip_through_a_layout_and_default_to_two() {
542        let chip = Chip {
543            name: "Thorns".to_owned(),
544            named: vec!["640".to_owned()],
545            mark: "\u{1f33f}".to_owned(),
546            color: Some([10, 200, 30]),
547            says: Says::Name,
548            warn: 60,
549            ..Chip::default()
550        };
551        let text = serde_json::to_string(&chip).expect("written");
552        assert!(text.contains("\"warn\":60"), "{text}");
553        assert!(
554            !text.contains("critical"),
555            "the default is not written: {text}"
556        );
557        let back: Chip = serde_json::from_str(&text).expect("read");
558        assert_eq!(back, chip);
559        let layout = serde_json::json!({ "chips": { "7": chip, "3": Chip::default() } });
560        let read = Chip::in_layout(&layout).expect("two");
561        assert_eq!(read.len(), 2);
562        assert_eq!(read[0].name, "Chip", "the lower widget id first");
563        assert!(Chip::in_layout(&serde_json::json!({})).is_none());
564        assert_eq!(Chip::defaults().len(), 2);
565        assert_eq!(clock(45), "0:45");
566        assert_eq!(clock(3_723), "1:02:03");
567    }

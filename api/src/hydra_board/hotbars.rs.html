14/// Whether `rest`, after `macro`, names a macro to run, rather than one of
15/// the words the command line answers itself (`hydra-table`'s `.macro`).
16#[must_use]
17pub fn is_macro_run(rest: &str) -> bool {
18    let first = rest.split_whitespace().next().unwrap_or_default();
19    !first.is_empty()
20        && !["help", "list", "show", "alias"].contains(&first.to_ascii_lowercase().as_str())
21}

29/// What pressing the bar `bar`'s button `key` (an id or a nickname) sends
30/// in `world`; or why nothing is.
31///
32/// # Errors
33///
34/// No such bar or button, or a button that sends nothing.
35pub fn press(
36    file: &HotbarFile,
37    (macros, runs): (&MacroFile, &mut Runs),
38    bar: &str,
39    key: &str,
40    world: &World<'_>,
41) -> Result<String, String> {
42    let found: &Bar = file
43        .bar(bar)
44        .ok_or_else(|| format!("There is no hotbar {bar}."))?;
45    let button = found
46        .button(key)
47        .ok_or_else(|| format!("The hotbar {} has no button {key}.", found.name))?;
48    // A button that names a macro runs the macro: its chain, its states.
49    if let Some(name) = button
50        .macro_name
51        .as_deref()
52        .map(str::trim)
53        .filter(|name| !name.is_empty())
54    {
55        return crate::macros::macro_chain(macros, name, world, (runs, std::time::Instant::now()));
56    }
57    let command = button.look(world).command;
58    if command.trim().is_empty() {
59        return Err(format!("The button {} sends nothing.", button.id));
60    }
61    Ok(command)
62}

64/// A character's hotbars as its files have them now, for the web page:
65/// every character's in `data`, the character `name`'s own (on `instance`)
66/// over it; a file that does not read gives none, and says why.
67#[must_use]
68pub fn hotbars_for(data: &Path, instance: Option<&str>, name: &str) -> (HotbarFile, Vec<String>) {
69    let mut problems = Vec::new();
70    let mut read = |path: &Path| {
71        HotbarFile::load(path).unwrap_or_else(|why| {
72            problems.push(why);
73            HotbarFile::default()
74        })
75    };
76    let [every, own] = hotbars_paths(data, instance, name);
77    let every = read(&every);
78    let own = read(&own);
79    (every.merged(own), problems)
80}

82/// Every hotbars file in `data`, everyone's and each character's, whose
83/// buttons are not yet only places for macros, made so: each button of its
84/// own kept as a macro in the macros file beside it
85/// ([`hydra_ui::macros::adopt`]). The two files as they were are kept
86/// first, beside them, ending `.before-macros`. What was done, a line a
87/// file.
88#[must_use]
89pub fn adopt_files(data: &Path) -> Vec<String> {
90    let mut pairs = vec![(data.join(FILE), data.join(hydra_ui::macros::FILE))];
91    if let Ok(entries) = std::fs::read_dir(data.join("layouts")) {
92        for entry in entries.flatten() {
93            let path = entry.path();
94            let name = path
95                .file_name()
96                .map(|name| name.to_string_lossy().into_owned());
97            if let Some(stem) = name
98                .as_deref()
99                .and_then(|name| name.strip_suffix(".hotbars.toml"))
100            {
101                let macros = path.with_file_name(format!("{stem}.macros.toml"));
102                pairs.push((path, macros));
103            }
104        }
105    }
106    pairs
107        .into_iter()
108        .filter_map(|(bars, macros)| adopt_pair(&bars, &macros))
109        .collect()
110}

112/// [`adopt_files`] for the bars at `bars` and the macros at `macros`: what
113/// was done, or why not; none when there was nothing to do.
114fn adopt_pair(bars: &Path, macros: &Path) -> Option<String> {
115    let shown = bars.display();
116    let mut file = HotbarFile::load(bars).ok()?;
117    if !hydra_ui::macros::unadopted(&file) {
118        return None;
119    }
120    let mut kept = match MacroFile::load(macros) {
121        Ok(kept) => kept,
122        Err(why) => return Some(format!("{shown}: its buttons left as they are: {why}")),
123    };
124    let dir = bars.parent().unwrap_or(Path::new("."));
125    let before = |path: &Path| {
126        let mut name = path.as_os_str().to_owned();
127        name.push(".before-macros");
128        PathBuf::from(name)
129    };
130    for path in [bars, macros] {
131        if path.exists()
132            && let Err(why) = std::fs::copy(path, before(path))
133        {
134            return Some(format!(
135                "{shown}: its buttons left as they are: {} could not be kept: {why}",
136                path.display()
137            ));
138        }
139    }
140    let made = hydra_ui::macros::adopt(&mut file, &mut kept);
141    let written = hydra_session::store::save_bytes(dir, macros, kept.to_toml().as_bytes())
142        .and_then(|()| hydra_session::store::save_bytes(dir, bars, file.to_toml().as_bytes()));
143    Some(match written {
144        Ok(()) => format!(
145            "{shown}: its buttons are macros' places now, {made} macro(s) made in {}; both files as they were end .before-macros.",
146            macros.display()
147        ),
148        Err(why) => format!("{shown}: its buttons could not be made macros: {why}"),
149    })
150}

162/// `bar` put in `file` in place of the bar named `replacing`, or of the
163/// bar of its own name, or after the rest; its name trimmed.
164///
165/// # Errors
166///
167/// It has no name, or another bar has it, whatever the case: a bar is
168/// pressed by its name, case aside.
169pub fn put_bar(file: &mut HotbarFile, mut bar: Bar, replacing: Option<&str>) -> Result<(), String> {
170    let end = bar.name.trim_end().len();
171    bar.name.truncate(end);
172    let start = bar.name.len() - bar.name.trim_start().len();
173    bar.name.drain(..start);
174    if bar.name.is_empty() {
175        return Err("Give the bar a name.".to_owned());
176    }
177    let place = file
178        .bars
179        .iter()
180        .position(|had| had.name == replacing.unwrap_or(&bar.name));
181    if let Some(other) = file
182        .bars
183        .iter()
184        .position(|had| had.name.eq_ignore_ascii_case(&bar.name))
185        && Some(other) != place
186    {
187        return Err(format!(
188            "There is a hotbar {} already.",
189            file.bars[other].name
190        ));
191    }
192    match place {
193        Some(place) => file.bars[place] = bar,
194        None => file.bars.push(bar),
195    }
196    Ok(())
197}

199/// The bar named `name` taken out of `file`: whether it was there.
200pub fn take_bar(file: &mut HotbarFile, name: &str) -> bool {
201    let before = file.bars.len();
202    file.bars.retain(|bar| bar.name != name);
203    file.bars.len() != before
204}

206/// `bar` kept in the hotbars file at `path` as [`put_bar`] puts it, with
207/// the next batch of saves (`crate::store`): made to the file as it then
208/// stands, read and written whole, so what it says that Hydra does not
209/// read is kept and its comments are not. A change refused then is the
210/// store's to tell; [`put_bar`] on the bars as last read says so first.
211pub fn save_bar(path: &Path, bar: &Bar, replacing: Option<&str>) {
212    let (bar, replacing) = (bar.clone(), replacing.map(str::to_owned));
213    edit_bars(path, move |file| put_bar(file, bar, replacing.as_deref()));
214}

216/// The bar named `name` taken out of the hotbars file at `path`, with the
217/// next batch of saves.
218pub fn remove_bar(path: &Path, name: &str) {
219    let name = name.to_owned();
220    edit_bars(path, move |file| {
221        take_bar(file, &name);
222        Ok(())
223    });
224}

226/// `change` made to the hotbars file at `path` with the next batch of
227/// saves, the file read and written whole.
228fn edit_bars(
229    path: &Path,
230    change: impl FnOnce(&mut HotbarFile) -> Result<(), String> + Send + 'static,
231) {
232    let dir = path.parent().unwrap_or_else(|| Path::new(".")).to_owned();
233    crate::store::edit(
234        &dir,
235        path,
236        Box::new(move |text: &str| {
237            let mut file = HotbarFile::parse(text)?;
238            change(&mut file)?;
239            Ok(file.to_toml())
240        }),
241    );
242}

244/// What pressing the bar `bar`'s button `key` sends in `world`, for the
245/// web page as for `.hotbar`; never a press of another button, `symbol`
246/// being the character's command symbol.
247///
248/// # Errors
249///
250/// No such bar or button, one that sends nothing, or one that would press
251/// another.
252pub fn hotbar_command(
253    file: &HotbarFile,
254    macros: (&MacroFile, &mut Runs),
255    (bar, key): (&str, &str),
256    world: &World<'_>,
257    symbol: char,
258) -> Result<String, String> {
259    let command = press(file, macros, bar, key, world)?;
260    if presses_another(&command, symbol) {
261        return Err("Not sent: a hotbar button never presses another.".to_owned());
262    }
263    Ok(command)
264}

266/// Whether `command` is `.hotbar …`, `symbol` being the character's command
267/// symbol: a button that would press a hotbar.
268#[must_use]
269pub fn presses_another(command: &str, symbol: char) -> bool {
270    command
271        .trim()
272        .strip_prefix(symbol)
273        .and_then(|rest| rest.split_whitespace().next())
274        .is_some_and(|word| word.eq_ignore_ascii_case("hotbar"))
275}

277/// The longest command a button of the page's own may send.
278const BUTTON_MOST: usize = 1024;
279
280/// The lines a button of the web page's own sends, `command` being what
281/// the page keeps for it: one line per `\r` or `\n`, each trimmed, empty
282/// ones dropped. `symbol` is the character's command symbol.
283///
284/// # Errors
285///
286/// A command that is empty, too long, carries other control characters,
287/// would press a hotbar, or names a thing in focus (`<target_id>`), which
288/// the page has none of.
289pub fn button_lines(command: &str, symbol: char) -> Result<Vec<String>, String> {
290    if command.len() > BUTTON_MOST {
291        return Err("Not sent: the button's command is too long.".to_owned());
292    }
293    if command
294        .chars()
295        .any(|c| c.is_control() && c != '\r' && c != '\n' && c != '\t')
296    {
297        return Err(
298            "Not sent: the button's command has characters that cannot be typed.".to_owned(),
299        );
300    }
301    let lines: Vec<String> = command
302        .split(['\r', '\n'])
303        .map(str::trim)
304        .filter(|line| !line.is_empty())
305        .map(str::to_owned)
306        .collect();
307    if lines.is_empty() {
308        return Err("Not sent: the button sends nothing.".to_owned());
309    }
310    if lines.iter().any(|line| presses_another(line, symbol)) {
311        return Err("Not sent: a button never presses a hotbar.".to_owned());
312    }
313    if lines
314        .iter()
315        .any(|line| crate::pad::interact::placed(line, None).is_none())
316    {
317        return Err(
318            "Not sent: the button names a thing in focus, and the page has none.".to_owned(),
319        );
320    }
321    Ok(lines)
322}

324/// The bar and the button a `.hotbar` line names, its command symbol and
325/// word taken off already: the bar, then the button, which may be words.
326pub fn named(rest: &str) -> Option<(&str, &str)> {
327    let (bar, key) = rest.trim().split_once(char::is_whitespace)?;
328    let key = key.trim();
329    (!key.is_empty()).then_some((bar, key))
330}

336    /// Everyone's bars and a character's are made macros' places, each
337    /// into the macros file beside it, the files as they were kept; done,
338    /// a second look does nothing.
339    #[test]
340    fn the_files_buttons_are_made_macros_once() {
341        let dir = std::env::temp_dir().join(format!("hydra-board-adopt-{}", std::process::id()));
342        let _ = std::fs::remove_dir_all(&dir);
343        std::fs::create_dir_all(dir.join("layouts")).unwrap();
344        let bars = "[[bars]]\nname = 'combat'\n\
345                    [[bars.buttons]]\nid = 'hide'\nlabel = 'Hide'\ncommand = 'hide'\n";
346        std::fs::write(dir.join(FILE), bars).unwrap();
347        let own = dir.join("layouts").join("prime_ashryn.hotbars.toml");
348        std::fs::write(&own, bars).unwrap();
349        let said = adopt_files(&dir);
350        assert_eq!(said.len(), 2, "{said:?}");
351        let every = MacroFile::load(&dir.join(hydra_ui::macros::FILE)).unwrap();
352        assert_eq!(
353            every.named("Hide").map(|made| made.chain.as_str()),
354            Some("hide")
355        );
356        let mine = MacroFile::load(&dir.join("layouts").join("prime_ashryn.macros.toml")).unwrap();
357        assert!(mine.named("Hide").is_some());
358        let bar = &HotbarFile::load(&own).unwrap().bars[0];
359        assert_eq!(bar.buttons[0].macro_name.as_deref(), Some("Hide"));
360        assert_eq!(
361            std::fs::read_to_string(dir.join(format!("{FILE}.before-macros"))).unwrap(),
362            bars
363        );
364        assert!(adopt_files(&dir).is_empty(), "done once");
365        let _ = std::fs::remove_dir_all(&dir);
366    }

368    /// A bar saved goes in place of the one it was, renamed, what the file
369    /// says that Hydra does not read kept; a new one after the rest; one
370    /// taken away is gone; a name another bar has is refused.
371    #[test]
372    fn a_bar_is_saved_in_place_added_and_taken_away() {
373        let dir = std::env::temp_dir().join(format!("hydra-board-hotbars-{}", std::process::id()));
374        let _ = std::fs::remove_dir_all(&dir);
375        std::fs::create_dir_all(&dir).unwrap();
376        let path = dir.join(FILE);
377        std::fs::write(
378            &path,
379            "[[bars]]\nname = 'combat'\nshape = 'round'\n\
380             [[bars.buttons]]\nid = 'hide'\nlabel = 'Hide'\ncommand = 'hide'\n\
381             [[bars]]\nname = 'travel'\n",
382        )
383        .unwrap();
384        let names = |file: &HotbarFile| -> Vec<String> {
385            file.bars.iter().map(|bar| bar.name.clone()).collect()
386        };
387        let mut bar = HotbarFile::load(&path).unwrap().bars[0].clone();
388        bar.name = " fight ".to_owned();
389        bar.buttons[0].command = "hide\rstalk".to_owned();
390        save_bar(&path, &bar, Some("combat"));
391        let file = HotbarFile::load(&path).unwrap();
392        assert_eq!(names(&file), ["fight", "travel"], "in its place");
393        assert_eq!(file.bars[0].buttons[0].command, "hide\rstalk");
394        assert!(
395            file.bars[0].rest.contains_key("shape"),
396            "what Hydra does not read kept"
397        );
398        let new = Bar {
399            name: "new".to_owned(),
400            ..Bar::default()
401        };
402        save_bar(&path, &new, None);
403        assert_eq!(
404            names(&HotbarFile::load(&path).unwrap()),
405            ["fight", "travel", "new"]
406        );
407        let mut taken = HotbarFile::load(&path).unwrap();
408        let clash = Bar {
409            name: "TRAVEL".to_owned(),
410            ..Bar::default()
411        };
412        assert!(
413            put_bar(&mut taken, clash, Some("new")).is_err(),
414            "a name taken"
415        );
416        assert!(
417            put_bar(&mut taken, Bar::default(), None).is_err(),
418            "no name"
419        );
420        remove_bar(&path, "travel");
421        assert_eq!(names(&HotbarFile::load(&path).unwrap()), ["fight", "new"]);
422        let _ = std::fs::remove_dir_all(&dir);
423    }

425    /// A button of the page's own sends its lines as typed, and never a
426    /// hotbar press, a thing in focus, or what cannot be typed.
427    #[test]
428    fn a_pages_button_sends_its_lines_and_never_a_hotbar() {
429        assert_eq!(button_lines("hide", '.'), Ok(vec!["hide".to_owned()]));

435        assert!(
436            button_lines(".hotbar combat hide", '.').is_err(),

439        assert!(
440            button_lines("look\n.HOTBAR combat hide", '.').is_err(),
441            "a hotbar on a later line"

443        assert!(
444            button_lines(";hotbar combat hide", '.').is_ok(),

452        assert!(

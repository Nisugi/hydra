1//! Keybinds: a key, and the macro it does on the character whose window
2//! has the keyboard -- commands sent as if typed there, text put in the
3//! command input, or one of Hydra's actions ([`crate::binding`]).
4//!
5//! **Hydra's defaults live in the code** (`defaults.rs`); the keybinds file
6//! holds only what the player added or changed, a key written `""` unbinding
7//! a default. A character's own keys file goes over it, and each holds ten
8//! macro sets ([`mod@file`]).
9//!
10//! **A key is named by its winit code** -- `Numpad8`, `F13`, `KeyA`,
11//! `ArrowUp` -- with `ctrl+`, `shift+`, `alt+` or `cmd+` before it. The
12//! physical key, not the character it types, so a binding means the same key
13//! on every keyboard layout, and a numpad key is never its digit. A key that
14//! types -- a letter, a digit, a mark, with no Ctrl, Alt or Cmd -- never
15//! fires a binding, or nobody could type it.
16//!
17//! **`NumLock` is nobody's.** On Windows and Linux it is the operating
18//! system's switch for the numpad; on macOS, which has none, winit calls the
19//! Clear key `NumLock`, and Clear is Hydra's switch there, the numpad typing
20//! or sending its keys.
21

35/// Ctrl, in [`Chord::held`].
36pub const CTRL: u8 = 1;
37/// Shift.
38pub const SHIFT: u8 = 2;
39/// Alt, Option on macOS.
40pub const ALT: u8 = 4;
41/// Cmd, on macOS. Nowhere else: a window sees the Command key only on
42/// macOS, so a chord holding it is refused off macOS ([`Chord::refused`])
43/// rather than written and never fired. It is not the Windows key elsewhere.
44pub const CMD: u8 = 8;

58/// A key with the modifiers held.
59#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
60pub struct Chord {
61    /// The key, as winit names its code.
62    pub key: String,
63    /// The modifiers held: [`CTRL`], [`SHIFT`], [`ALT`], [`CMD`].
64    pub held: u8,
65}
66
67impl Chord {
68    /// `key`, by its winit name, with the modifiers `held`: [`CTRL`],
69    /// [`SHIFT`], [`ALT`], [`CMD`].
70    #[must_use]
71    pub fn new(key: &str, held: u8) -> Self {
72        Self {
73            key: key.to_owned(),
74            held,
75        }
76    }
77
78    /// A chord as written, `ctrl+shift+F1`: modifiers in any order and case,
79    /// the key last, by its winit name or the fork's numpad name (`num_8`).
80    ///
81    /// # Errors
82    ///
83    /// What is wrong with it, in words for the player.
84    pub fn parse(text: &str) -> Result<Self, String> {
85        let mut chord = Self {
86            key: String::new(),
87            held: 0,
88        };
89        let parts: Vec<&str> = text.split('+').map(str::trim).collect();
90        let Some((key, held)) = parts.split_last() else {
91            return Err(format!("`{text}` names no key."));
92        };
93        for modifier in held {
94            match modifier.to_ascii_lowercase().as_str() {
95                "ctrl" | "control" => chord.held |= CTRL,
96                "shift" => chord.held |= SHIFT,
97                "alt" | "option" => chord.held |= ALT,
98                "cmd" | "command" | "super" | "win" => chord.held |= CMD,
99                other => {
100                    return Err(format!(
101                        "`{other}` in `{text}` is not Ctrl, Shift, Alt or Cmd."
102                    ));
103                }
104            }
105        }
106        chord.key = known(key).map_err(|why| format!("`{text}`: {why}"))?;
107        Ok(chord)
108    }
109
110    /// The chord as the file and the Keys page write it: its modifiers in
111    /// one order, then its key, `Ctrl+Shift+F1`. [`Chord::parse`] reads it
112    /// back.
113    #[must_use]
114    pub fn written(&self) -> String {
115        let mut written = String::new();
116        for (bit, name) in [(CTRL, "Ctrl"), (SHIFT, "Shift"), (ALT, "Alt"), (CMD, "Cmd")] {
117            if self.held & bit != 0 {
118                written.push_str(name);
119                written.push('+');
120            }
121        }
122        written.push_str(&self.key);
123        written
124    }
125
126    /// Why this chord cannot be bound, in words for the player: it types
127    /// something rather than commanding -- a letter, a digit, a mark or the
128    /// space, with no Ctrl, Alt or Cmd held -- or it is Enter alone, which
129    /// sends what is typed (a key that cannot be bound should not look
130    /// bindable); or it holds Cmd where there is none ([`CMD`]).
131    #[must_use]
132    pub fn refused(&self) -> Option<String> {
133        let written = self.written();
134        if self.held & CMD != 0 && !cfg!(target_os = "macos") {
135            return Some(format!(
136                "{written}: Cmd is macOS's Command key; Hydra does not see the Windows key."
137            ));
138        }
139        if self.held & (CTRL | ALT | CMD) == 0 && types(&self.key) {
140            return Some(format!(
141                "{written} types: hold Ctrl, Alt or Cmd with it, or it could not be typed."
142            ));
143        }
144        (self.held == 0 && self.key == "Enter")
145            .then(|| "Enter sends what is typed: hold Ctrl, Shift, Alt or Cmd with it.".to_owned())
146    }
147}

149/// A key as the Keys page lists it: what it does now, from whose file, and
150/// what Hydra's default is.
151#[derive(Clone, Debug, PartialEq, Eq)]
152pub struct KeyRow {
153    /// The key, as the file writes it: `Ctrl+F1`, `Numpad8`.
154    pub key: String,
155    /// What it does now; `None` where a file unbinds it.
156    pub does: Option<Macro>,
157    /// What Hydra binds it to, if anything.
158    pub default: Option<Macro>,
159    /// Whose file binds or unbinds it; `None` for Hydra's default.
160    pub from: Option<Whose>,
161    /// What it would do without its file's line, if anything.
162    pub beneath: Option<Macro>,
163}

165/// The keybinds: Hydra's defaults, and every character's file on top. A
166/// character's own file goes over them in [`Keybinds::of`].
167#[derive(Clone, Debug, PartialEq, Eq)]
168pub struct Keybinds {
169    /// Hydra's own (`defaults.rs`), all in set 0.
170    defaults: BTreeMap<Chord, Macro>,
171    /// Every character's: the keybinds file.
172    every: KeyFile,
173}
174
175impl Default for Keybinds {
176    /// Hydra's defaults alone: no file.
177    fn default() -> Self {
178        Self {
179            defaults: defaults::defaults(),
180            every: KeyFile::default(),
181        }
182    }
183}

185impl Keybinds {

250    /// What `chord` does in set 0 as the file `whose` has it now: its own
251    /// line, binding or unbinding it, or else what is beneath it. A
252    /// character's own file is `mine`.
253    #[must_use]
254    pub fn in_file<'a>(
255        &'a self,
256        whose: Whose,
257        mine: Option<&'a KeyFile>,
258        chord: &Chord,
259    ) -> Option<&'a Macro> {
260        let own = match whose {
261            Whose::Character => mine.and_then(|mine| mine.sets[0].get(chord)),
262            Whose::Every => self.every.sets[0].get(chord),
263        };
264        match own {
265            Some(made) => made.as_ref(),
266            None => self.beneath(whose, chord),
267        }
268    }
269
270    /// What `chord` does in set 0 beneath a file's own line: under every
271    /// character's, Hydra's default; under a character's, every
272    /// character's, then Hydra's.
273    #[must_use]
274    pub fn beneath(&self, whose: Whose, chord: &Chord) -> Option<&Macro> {
275        if whose == Whose::Character
276            && let Some(made) = self.every.sets[0].get(chord)
277        {
278            return made.as_ref();
279        }
280        self.defaults.get(chord)
281    }
282
283    /// Every key set `set` binds or unbinds -- in `mine`, a character's own
284    /// file, when the page is a character's, and in every character's --
285    /// with what it does, whose file that came from, and Hydra's default:
286    /// what the Keys page lists. Set 0's hold Hydra's defaults too.
287    #[must_use]
288    pub fn rows(&self, set: u8, mine: Option<&KeyFile>) -> Vec<KeyRow> {
289        let set = usize::from(set.min(SETS - 1));
290        let layers: Vec<(Whose, &Layer)> = mine
291            .map(|mine| (Whose::Character, &mine.sets[set]))
292            .into_iter()
293            .chain(std::iter::once((Whose::Every, &self.every.sets[set])))
294            .collect();
295        let empty = BTreeMap::new();
296        let defaults = if set == 0 { &self.defaults } else { &empty };
297        let chords: BTreeSet<&Chord> = layers
298            .iter()
299            .flat_map(|(_, layer)| layer.keys())
300            .chain(defaults.keys())
301            .collect();
302        chords
303            .into_iter()
304            .map(|chord| {
305                // What the first layer after `from` binds it to, or Hydra.
306                let after = |from: usize| {
307                    layers[from..]
308                        .iter()
309                        .find_map(|(_, layer)| layer.get(chord).cloned())
310                        .unwrap_or_else(|| defaults.get(chord).cloned())
311                };
312                let (does, from, beneath) = match layers
313                    .iter()
314                    .position(|(_, layer)| layer.contains_key(chord))
315                {
316                    Some(at) => (after(at), Some(layers[at].0), after(at + 1)),
317                    None => (defaults.get(chord).cloned(), None, None),
318                };
319                KeyRow {
320                    key: chord.written(),
321                    does,
322                    default: defaults.get(chord).cloned(),
323                    from,
324                    beneath,
325                }
326            })
327            .collect()
328    }
329}

331/// The keys in effect for one character, or for none: a press looks in the
332/// chosen set, the character's own file and then every character's; then
333/// in set 0 the same way; then at Hydra's defaults. The first that binds or
334/// unbinds the key has it.
335#[derive(Clone, Copy, Debug)]
336pub struct Keys<'a> {
337    binds: &'a Keybinds,
338    mine: Option<&'a KeyFile>,
339}
340
341impl<'a> Keys<'a> {
342    /// The files' sets a press looks in, in order, before Hydra's defaults.
343    fn layers(self) -> impl Iterator<Item = &'a Layer> {
344        let chosen = self.mine.map_or(0, |mine| mine.chosen.min(SETS - 1));
345        let sets = if chosen == 0 {
346            vec![0]
347        } else {
348            vec![chosen, 0]
349        };
350        sets.into_iter().flat_map(move |set| {
351            let set = usize::from(set);
352            self.mine
353                .map(|mine| &mine.sets[set])
354                .into_iter()
355                .chain(std::iter::once(&self.binds.every.sets[set]))
356        })
357    }
358
359    /// What `chord` does, if it is bound.
360    #[must_use]
361    pub fn does(self, chord: &Chord) -> Option<&'a Macro> {
362        for layer in self.layers() {
363            if let Some(made) = layer.get(chord) {
364                return made.as_ref();
365            }
366        }
367        self.binds.defaults.get(chord)
368    }
369
370    /// Each key bound, and what it does.
371    pub fn bound(self) -> impl Iterator<Item = (&'a Chord, &'a Macro)> {
372        let chords: BTreeSet<&'a Chord> = self
373            .layers()
374            .flat_map(BTreeMap::keys)
375            .chain(self.binds.defaults.keys())
376            .collect();
377        chords
378            .into_iter()
379            .filter_map(move |chord| self.does(chord).map(|made| (chord, made)))
380    }
381
382    /// The fork's names for the numpad keys that are bound, whatever the
383    /// modifiers ([`NUMPAD`]): what it is told to catch, so an unbound
384    /// numpad key still types.
385    #[must_use]
386    pub fn numpad_caught(self) -> HashSet<String> {
387        self.bound()
388            .filter_map(|(chord, _)| NUMPAD.iter().find(|(winit, _)| *winit == chord.key))
389            .map(|(_, fork)| (*fork).to_owned())
390            .collect()
391    }
392}

394/// Whether a held key's repeats do `made` again. A send does, as holding a
395/// direction keeps walking; so does an action that steps -- scrolling, the
396/// history, the next window, tab, line found or target -- as a held arrow
397/// keeps moving. A switch does not, so a held one ends where one press
398/// leaves it: a macro set, the Find bar, a drawer, Lock, Arrange, a
399/// character's window. Nor does a fill, which would only write the same
400/// words over.
401#[must_use]
402pub fn repeats(made: &Macro) -> bool {
403    match made {
404        Macro::Send(_) => true,
405        Macro::Fill(_) | Macro::Hotbar { .. } | Macro::Run(_) => false,
406        Macro::Act(action) => !matches!(
407            action,
408            Action::Set(_)
409                | Action::Find
410                | Action::DrawerTop
411                | Action::DrawerBottom
412                | Action::DrawerLeft
413                | Action::DrawerRight
414                | Action::Lock
415                | Action::Arrange
416                | Action::Character(_)
417                | Action::TargetNext
418                | Action::TargetPrevious
419                | Action::TargetClear
420        ),
421    }
422}

1//! Push to phone: a trigger's alert sent to each phone paired for it, as a
2//! banner from HydraFE, through the push relay (`hydra-push`).
3//!
4//! **Pairing.** A phone already paired with the page says its device token,
5//! whether it is a build run from Xcode (Apple's sandbox), a name, and a
6//! sealing key it made. The desktop keeps them in `push.json` in the data
7//! folder, with its own id, the one the phone knows it by, and its token
8//! from the relay, asked for the first time a phone pairs.
9//!
10//! **Sealing.** Each alert is sealed for its phone with that phone's key,
11//! ChaCha20-Poly1305, laid out as Apple's `CryptoKit` lays a sealed box: a
12//! 12-byte nonce, the ciphertext, the 16-byte tag, as base64. The relay and
13//! Apple see only that. The app opens it on the phone and shows the title
14//! and text, and a tap opens this desktop on the character.
15//!
16//! **Sending** is a task of its own, never waited on: a relay down or slow
17//! holds no character up. A phone whose token Apple says is gone is
18//! forgotten; a desktop token the relay no longer knows is asked for again,
19//! once, however many pushes found it so (`desktop_token`). An alert is
20//! cut to fit a push before it is sealed ([`fitted`]).
21//!
22//! **Forgetting.** A page pairs its phone each time it connects, so a phone
23//! forgotten by `.push forget` is remembered by its key, and its pairing
24//! turned away until `.push allow` takes it again.
25

131/// What `push.json` in `data` keeps; nothing, when there is none or it does
132/// not read.
133#[must_use]
134pub fn read(data: &Path) -> Kept {
135    std::fs::read_to_string(path(data))
136        .ok()
137        .and_then(|text| serde_json::from_str(&text).ok())
138        .unwrap_or_default()
139}

141fn write(data: &Path, kept: &Kept) -> Result<(), String> {
142    let text = serde_json::to_string_pretty(kept).map_err(|why| why.to_string())?;
143    hydra_session::store::save_bytes(data, &path(data), text.as_bytes())
144        .map_err(|why| format!("{}: {why}", path(data).display()))
145}

147/// `push.json` changed by `change`, read and written under one lock.
148fn change<T>(data: &Path, change: impl FnOnce(&mut Kept) -> T) -> Result<T, String> {
149    let _held = FILE_LOCK.lock().unwrap_or_else(PoisonError::into_inner);
150    let mut kept = read(data);
151    let made = change(&mut kept);
152    write(data, &kept)?;
153    Ok(made)
154}

156fn random_hex(bytes: usize) -> Result<String, String> {
157    use std::fmt::Write as _;
158    let mut random = vec![0u8; bytes];
159    getrandom::fill(&mut random).map_err(|why| why.to_string())?;
160    Ok(random.iter().fold(String::new(), |mut out, byte| {
161        let _ = write!(out, "{byte:02x}");
162        out
163    }))
164}

166/// Why a phone's pairing is refused, if it is.
167fn refused(phone: &Phone) -> Option<&'static str> {
168    if phone.device.is_empty()
169        || phone.device.len() > 200
170        || !phone.device.bytes().all(|b| b.is_ascii_hexdigit())
171    {
172        return Some("its device token is not one");
173    }
174    if key_of(&phone.key).is_none() {
175        return Some("its sealing key is not 32 bytes");
176    }
177    None
178}

180fn key_of(key: &str) -> Option<[u8; 32]> {
181    STANDARD.decode(key).ok()?.try_into().ok()
182}

184/// A phone's pairing taken.
185#[derive(Clone, Debug, PartialEq, Eq)]
186pub struct Paired {
187    /// This desktop's id, for the phone to keep.
188    pub desktop: String,
189    /// It was not paired just so before: a new phone, a new token, or a
190    /// new name. A page pairs each time it connects, so only this is said.
191    pub fresh: bool,
192    /// The player forgot this phone: not paired, and nothing said.
193    pub forgotten: bool,
194}

196/// `phone` paired for push in `data`, in place of any with its device token
197/// or its key -- the same phone, its token changed, or a key made again.
198///
199/// # Errors
200///
201/// It is not a phone's pairing, too many are paired, or the file could not
202/// be written.
203pub fn pair(data: &Path, mut phone: Phone) -> Result<Paired, String> {
204    if let Some(why) = refused(&phone) {
205        return Err(format!("Push: not paired: {why}."));
206    }
207    phone.name = phone.name.trim().chars().take(60).collect();
208    phone.added = now();
209    let desktop = random_hex(16)?;
210    change(data, |kept| {
211        if kept.desktop.is_empty() {
212            kept.desktop = desktop;
213        }
214        if kept.forgotten.iter().any(|gone| gone.key == phone.key) {
215            return Ok(Paired {
216                desktop: kept.desktop.clone(),
217                fresh: false,
218                forgotten: true,
219            });
220        }
221        let same = |kept: &Phone| {
222            kept.device == phone.device && kept.key == phone.key && kept.name == phone.name
223        };
224        if let Some(at) = kept.phones.iter().position(same) {
225            phone.added = kept.phones[at].added;
226            kept.phones[at] = phone;
227            return Ok(Paired {
228                desktop: kept.desktop.clone(),
229                fresh: false,
230                forgotten: false,
231            });
232        }
233        kept.phones
234            .retain(|kept| kept.device != phone.device && kept.key != phone.key);
235        if kept.phones.len() >= PHONES_MOST {
236            return Err(format!("Push: {PHONES_MOST} phones are paired already."));
237        }
238        kept.phones.push(phone);
239        Ok(Paired {
240            desktop: kept.desktop.clone(),
241            fresh: true,
242            forgotten: false,
243        })
244    })?
245}

247/// The phone the player calls `name` forgotten: no longer pushed to, and its
248/// pairing turned away until [`allow`]. Its name as kept, if one was.
249///
250/// # Errors
251///
252/// The file could not be written.
253pub fn forget_named(data: &Path, name: &str) -> Result<Option<String>, String> {
254    change(data, |kept| {
255        let at = kept
256            .phones
257            .iter()
258            .position(|phone| phone.name.eq_ignore_ascii_case(name.trim()))?;
259        let phone = kept.phones.remove(at);
260        kept.forgotten.retain(|gone| gone.key != phone.key);
261        kept.forgotten.push(Forgotten {
262            name: phone.name.clone(),
263            key: phone.key,
264        });
265        Some(phone.name)
266    })
267}

269/// The phone forgotten as `name` let pair again, as its page next connects.
270/// Its name as kept, if one was forgotten so.
271///
272/// # Errors
273///
274/// The file could not be written.
275pub fn allow(data: &Path, name: &str) -> Result<Option<String>, String> {
276    change(data, |kept| {
277        let at = kept
278            .forgotten
279            .iter()
280            .position(|gone| gone.name.eq_ignore_ascii_case(name.trim()))?;
281        Some(kept.forgotten.remove(at).name)
282    })

285/// The phone with device token `device` no longer pushed to, as Apple or the
286/// phone itself said: it may pair again. Whether it was paired.
287///
288/// # Errors
289///
290/// The file could not be written.
291pub fn forget(data: &Path, device: &str) -> Result<bool, String> {
292    change(data, |kept| {
293        let before = kept.phones.len();
294        kept.phones.retain(|phone| phone.device != device);
295        kept.phones.len() != before
296    })
297}

299/// `text` cut to at most `most` bytes at a character's end, an ellipsis
300/// after it when cut.
301fn cut(text: &str, most: usize) -> String {
302    const ELLIPSIS: char = '\u{2026}';
303    if text.len() <= most {
304        return text.to_owned();
305    }
306    let mut end = most.saturating_sub(ELLIPSIS.len_utf8());
307    while !text.is_char_boundary(end) {
308        end -= 1;
309    }
310    format!("{}{ELLIPSIS}", text[..end].trim_end())
311}

313/// `alert` cut to fit a push: its title to [`TITLE_MOST`] and its text to
314/// [`BODY_MOST`]; whose it is, which the tap opens, kept whole.
315#[must_use]
316pub fn fitted(mut alert: Alert) -> Alert {
317    alert.title = cut(&alert.title, TITLE_MOST);
318    alert.body = cut(&alert.body, BODY_MOST);
319    alert

322/// `alert` sealed with `key`, as `CryptoKit`'s `ChaChaPoly.SealedBox` lays
323/// it out, in base64.
324///
325/// # Errors
326///
327/// No randomness for the nonce.
328pub fn seal(key: &[u8; 32], alert: &Alert) -> Result<String, String> {
329    let mut nonce = [0u8; 12];
330    getrandom::fill(&mut nonce).map_err(|why| why.to_string())?;
331    let plain = serde_json::to_vec(alert).map_err(|why| why.to_string())?;
332    let cipher = ChaCha20Poly1305::new(Key::from_slice(key));
333    let sealed = cipher
334        .encrypt(Nonce::from_slice(&nonce), plain.as_slice())
335        .map_err(|_| "not sealed".to_owned())?;
336    let mut boxed = nonce.to_vec();
337    boxed.extend_from_slice(&sealed);
338    Ok(STANDARD.encode(boxed))
339}

341/// `sealed` opened with `key`, as the app opens it.
342///
343/// # Errors
344///
345/// It is not sealed with that key.
346pub fn open(key: &[u8; 32], sealed: &str) -> Result<Alert, String> {
347    let boxed = STANDARD.decode(sealed).map_err(|why| why.to_string())?;
348    if boxed.len() < 12 + 16 {
349        return Err("too short".to_owned());
350    }
351    let (nonce, sealed) = boxed.split_at(12);
352    let cipher = ChaCha20Poly1305::new(Key::from_slice(key));
353    let plain = cipher
354        .decrypt(Nonce::from_slice(nonce), sealed)
355        .map_err(|_| "not this key's".to_owned())?;
356    serde_json::from_slice(&plain).map_err(|why| why.to_string())
357}

359/// What the relay said of one push.
360#[derive(Clone, Debug, PartialEq, Eq)]
361enum Pushed {
362    Taken,
363    /// The phone's token is gone: forget it.
364    Gone,
365    /// The relay does not know this desktop's token.
366    Stranger,
367    Refused(String),
368}

370fn client() -> Result<reqwest::Client, String> {
371    reqwest::Client::builder()
372        .timeout(TIMEOUT)
373        .build()
374        .map_err(|why| why.to_string())
375}

377async fn register(client: &reqwest::Client, relay: &str) -> Result<String, String> {
378    let response = client
379        .post(format!("{relay}/register"))
380        .send()
381        .await
382        .map_err(|why| format!("the relay could not be reached: {}", why.without_url()))?;
383    let status = response.status();
384    let body = response.bytes().await.map_err(|why| why.to_string())?;
385    if !status.is_success() {
386        return Err(format!(
387            "the relay refused to register this desktop: {status}"
388        ));
389    }
390    serde_json::from_slice::<serde_json::Value>(&body)
391        .ok()
392        .and_then(|said| said["desktop"].as_str().map(str::to_owned))
393        .ok_or_else(|| "the relay gave no token".to_owned())
394}

396/// One desktop token asked of the relay at a time: pushes that find none,
397/// or the same stale one, wait for the first to ask, then take what it
398/// kept.
399static REGISTERING: tokio::sync::Mutex<()> = tokio::sync::Mutex::const_new(());
400
401/// This desktop's token from the relay: the one kept in `data`, unless it is
402/// `stale` (the relay said it does not know it), when another is asked for
403/// and kept.
404async fn desktop_token(
405    client: &reqwest::Client,
406    relay: &str,
407    data: &Path,
408    stale: Option<&str>,
409) -> Result<String, String> {
410    let _one_at_a_time = REGISTERING.lock().await;
411    if let Some(kept) = read(data).token
412        && Some(kept.as_str()) != stale
413    {
414        return Ok(kept);
415    }
416    let made = register(client, relay).await?;
417    change(data, |kept| kept.token = Some(made.clone()))?;
418    Ok(made)
419}

421async fn send_one(
422    client: &reqwest::Client,
423    (relay, token): (&str, &str),
424    phone: &Phone,
425    sealed: String,
426) -> Result<Pushed, String> {
427    let note = serde_json::json!({
428        "device": phone.device,
429        "sandbox": phone.sandbox,
430        "sealed": sealed,
431    });
432    let response = client
433        .post(format!("{relay}/push"))
434        .header("authorization", format!("Bearer {token}"))
435        .header("content-type", "application/json")
436        .body(note.to_string())
437        .send()
438        .await
439        .map_err(|why| format!("the relay could not be reached: {}", why.without_url()))?;
440    Ok(match response.status().as_u16() {
441        200..=299 => Pushed::Taken,
442        410 => Pushed::Gone,
443        401 => Pushed::Stranger,
444        status => {
445            let said = response.text().await.unwrap_or_default();
446            Pushed::Refused(format!("{status} {said}"))
447        }
448    })
449}

451/// `alert` (its desktop filled in here) sent to every phone paired in
452/// `data`. What went wrong, if anything, for the log.
453pub async fn send(data: &Path, alert: Alert) -> Vec<String> {
454    let kept = read(data);
455    if kept.phones.is_empty() {
456        return Vec::new();
457    }
458    let relay = kept.relay.clone().unwrap_or_else(|| RELAY.to_owned());
459    let relay = relay.trim_end_matches('/').to_owned();
460    let mut alert = fitted(alert);
461    alert.desktop.clone_from(&kept.desktop);
462    let client = match client() {
463        Ok(client) => client,
464        Err(why) => return vec![why],
465    };
466    let mut stale: Option<String> = None;
467    let mut said = Vec::new();
468    let mut gone = Vec::new();
469    for phone in &kept.phones {
470        let Some(key) = key_of(&phone.key) else {
471            continue;
472        };
473        for _ in 0..2 {
474            let known = match desktop_token(&client, &relay, data, stale.as_deref()).await {
475                Ok(known) => known,
476                Err(why) => {
477                    said.push(why);
478                    return said;
479                }
480            };
481            let sealed = match seal(&key, &alert) {
482                Ok(sealed) => sealed,
483                Err(why) => {
484                    said.push(why);
485                    break;
486                }
487            };
488            match send_one(&client, (&relay, &known), phone, sealed).await {
489                Ok(Pushed::Taken) => break,
490                Ok(Pushed::Gone) => {
491                    gone.push(phone.device.clone());
492                    said.push(format!("{} no longer takes pushes: forgotten", phone.name));
493                    break;
494                }
495                Ok(Pushed::Stranger) => stale = Some(known),
496                Ok(Pushed::Refused(why)) => {
497                    said.push(format!("{}: {why}", phone.name));
498                    break;
499                }
500                Err(why) => {
501                    said.push(why);
502                    break;
503                }
504            }
505        }
506    }
507    for device in gone {
508        let _ = forget(data, &device);
509    }
510    said
511}

616        // What the relay wraps it in is a little over a hundred bytes.
617        assert!(sealed.len() + 200 < 4096, "{} bytes sealed", sealed.len());
618        assert_eq!(open(&[5u8; 32], &sealed).expect("opened"), fitted);
619        assert_eq!(cut("short", 10), "short");
620    }

622    /// A relay as the tests have it, on a port of its own: how many times a
623    /// desktop registered, and every push taken.
624    async fn relay() -> (String, std::sync::Arc<std::sync::atomic::AtomicUsize>) {
625        use std::sync::atomic::Ordering;
626        use tokio::io::{AsyncReadExt as _, AsyncWriteExt as _};
627        let listener = tokio::net::TcpListener::bind("127.0.0.1:0")
628            .await
629            .expect("bound");
630        let at = listener.local_addr().expect("an address");
631        let registered = std::sync::Arc::new(std::sync::atomic::AtomicUsize::new(0));
632        let counted = std::sync::Arc::clone(&registered);
633        tokio::spawn(async move {
634            while let Ok((mut stream, _)) = listener.accept().await {
635                let counted = std::sync::Arc::clone(&counted);
636                tokio::spawn(async move {
637                    let mut asked = Vec::new();
638                    let mut buf = [0u8; 4096];
639                    while let Ok(n @ 1..) = stream.read(&mut buf).await {
640                        asked.extend_from_slice(&buf[..n]);
641                        let text = String::from_utf8_lossy(&asked);
642                        let Some(head) = text.find("\r\n\r\n") else {
643                            continue;
644                        };
645                        let length = text[..head]
646                            .lines()
647                            .find_map(|line| {
648                                line.to_ascii_lowercase()
649                                    .strip_prefix("content-length:")
650                                    .and_then(|n| n.trim().parse::<usize>().ok())
651                            })
652                            .unwrap_or(0);
653                        if asked.len() >= head + 4 + length {
654                            break;
655                        }
656                    }
657                    let text = String::from_utf8_lossy(&asked);
658                    let answer = if text.starts_with("POST /register") {
659                        // Slow, as a relay far away is: the others wait meanwhile.
660                        tokio::time::sleep(std::time::Duration::from_millis(50)).await;
661                        let n = counted.fetch_add(1, Ordering::SeqCst);
662                        let body = format!("{{\"desktop\":\"token{n}\"}}");
663                        format!(
664                            "HTTP/1.1 201 Created\r\ncontent-length: {}\r\nconnection: close\r\n\r\n{body}",
665                            body.len()
666                        )
667                    } else {
668                        "HTTP/1.1 202 Accepted\r\ncontent-length: 0\r\nconnection: close\r\n\r\n"
669                            .to_owned()
670                    };
671                    let _ = stream.write_all(answer.as_bytes()).await;
672                });
673            }
674        });
675        (format!("http://{at}"), registered)
676    }

678    /// Five pushes at once with no desktop token yet register once: the
679    /// others wait and take the token the first kept.
680    #[tokio::test(flavor = "multi_thread", worker_threads = 4)]
681    async fn pushes_at_once_register_the_desktop_once() {
682        let data = dir("once");
683        let (url, registered) = relay().await;
684        pair(&data, phone("iPhone", "ab12")).expect("paired");
685        change(&data, |kept| kept.relay = Some(url)).expect("pointed at the relay");
686        let mut sends = tokio::task::JoinSet::new();
687        for _ in 0..5 {
688            let data = data.clone();
689            sends.spawn(async move { send(&data, alert()).await });
690        }
691        while let Some(said) = sends.join_next().await {
692            assert_eq!(said.expect("ran"), Vec::<String>::new());
693        }
694        assert_eq!(registered.load(std::sync::atomic::Ordering::SeqCst), 1);
695        assert_eq!(read(&data).token.as_deref(), Some("token0"));
696        let _ = std::fs::remove_dir_all(&data);
697    }

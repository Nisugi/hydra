1//! The command wheel: a ring of slices held open with a button, one aimed
2//! with a stick and fired.
3//!
4//! **The file.** The default wheel is `[[controller_wheel]]`, each entry a
5//! slice; a named one `[[controller_wheels.<name>]]`, opened by a button
6//! bound to `controller_wheel:<name>`; `[controller_wheels_meta.<name>]`
7//! (`default` for the default wheel) gives a wheel the stick that aims it
8//! and the angle its first slice sits at. A slice has a label and a
9//! command, or `slices` of its own: a folder, opened by aiming at it. It
10//! may have a colour, a width in degrees (`span`), a floor the stick must
11//! be pushed past to aim it (`inner`, a percent), and how it fires
12//! (`fire_type`: `release`, `edge`, `retract`, or `none` for a slice that
13//! only holds its place).
14//!
15//! **The ring.** Slices with a width take it, at least [`MIN_SPAN`]; the
16//! rest share what is left; and the whole is scaled to close at 360. The
17//! first slice is centred straight up, or at the wheel's `start`, and the
18//! rest follow clockwise. Inside a folder a Back slice is added last, and
19//! the ring turned so Back sits where `back_slice` says (down, unless it
20//! says otherwise or `none`); a folder with a slice of its own marked
21//! `back` is used as it is. A folder's own `start` turns its ring, and only
22//! its, that much further: the wheel's turn is the top ring's alone.
23//!
24//! **Aiming.** The slice under the stick, past its floor, is the
25//! candidate. Held on it [`Timing::aim_ms`] a leaf is committed; held on a
26//! folder or Back [`Timing::nav_ms`] the wheel goes into it or up out of
27//! it, and the stick must come back to the middle before anything is aimed
28//! again. A leaf fires as its fire type says: `release`, the committed one
29//! when the wheel's button is let go; `edge`, the moment the stick is
30//! pushed past [`Timing::edge_threshold`], with no wait; `retract`, once
31//! committed, when the stick is drawn back [`Timing::retract_delta`] from
32//! the furthest it reached. The stick back in the middle before the
33//! release fires nothing.
34

47/// One slice of a wheel.
48#[derive(Clone, Debug, Default, PartialEq, serde::Deserialize)]
49pub struct Slice {
50    /// What it shows.
51    pub label: String,
52    /// What it sends; empty for a folder.
53    #[serde(default)]
54    pub command: String,
55    /// A tint, `#rrggbb`: faint at rest, strong while aimed.
56    #[serde(default)]
57    pub color: Option<String>,
58    /// Its width in degrees; none, a share of what is left.
59    #[serde(default)]
60    pub span: Option<f32>,
61    /// How far, in percent, the stick is pushed before it can be aimed.
62    #[serde(default)]
63    pub inner: Option<u8>,
64    /// Its own slices: a folder.
65    #[serde(default)]
66    pub slices: Vec<Slice>,
67    /// A folder's own turn, in degrees clockwise, on top of where its ring
68    /// is placed; none, 0.
69    #[serde(default)]
70    pub start: Option<f32>,
71    /// Aimed, it goes up out of the folder.
72    #[serde(default)]
73    pub back: bool,
74    /// How it fires: `release`, `edge`, `retract`, or `none`.
75    #[serde(default)]
76    pub fire_type: Option<String>,
77    /// Its command is one of Hydra's own, sent with the command symbol
78    /// before it: the portals wheel's `go2` for a scripted way.
79    #[serde(skip)]
80    pub hydra: bool,
81}

109/// The wheels a controller file holds.
110#[derive(Clone, Debug, Default, PartialEq)]
111pub struct Wheels {
112    /// Each by name, the default as [`DEFAULT`].
113    pub rings: BTreeMap<String, Vec<Slice>>,
114    /// Each wheel's own settings, by name; a wheel without any uses Hydra's.
115    pub meta: BTreeMap<String, Meta>,
116}
117
118impl Wheels {
119    /// The wheels in `table`, and what is wrong with them.
120    pub fn read(table: &toml::Table) -> (Self, Vec<String>) {
121        let mut wheels = Self::default();
122        let mut problems = Vec::new();
123        let mut ring = |name: &str, value: &toml::Value| match value.clone().try_into() {
124            Ok(slices) => {
125                wheels.rings.insert(name.to_owned(), slices);
126            }
127            Err(why) => problems.push(format!("the {name} wheel: {}", why.message())),
128        };
129        if let Some(value) = table.get("controller_wheel") {
130            ring(DEFAULT, value);
131        }
132        if let Some(named) = table
133            .get("controller_wheels")
134            .and_then(toml::Value::as_table)
135        {
136            for (name, value) in named {
137                // [[controller_wheel]] wins over a [controller_wheels.default].
138                if name != DEFAULT || !table.contains_key("controller_wheel") {
139                    ring(name, value);
140                }
141            }
142        }
143        if let Some(meta) = table
144            .get("controller_wheels_meta")
145            .and_then(toml::Value::as_table)
146        {
147            for (name, value) in meta {
148                match value.clone().try_into() {
149                    Ok(read) => {
150                        wheels.meta.insert(name.clone(), read);
151                    }
152                    Err(why) => {
153                        problems.push(format!("the {name} wheel's settings: {}", why.message()));
154                    }
155                }
156            }
157        }
158        (wheels, problems)
159    }
160
161    /// `own`'s wheels over these, name by name.
162    pub fn over(&mut self, own: &Self) {
163        self.rings.extend(
164            own.rings
165                .iter()
166                .map(|(name, ring)| (name.clone(), ring.clone())),
167        );
168        self.meta.extend(
169            own.meta
170                .iter()
171                .map(|(name, meta)| (name.clone(), meta.clone())),
172        );
173    }
174
175    /// The slices of wheel `name` at the folder `path`.
176    #[must_use]
177    pub fn level(&self, name: &str, path: &[usize]) -> Option<&[Slice]> {
178        let mut level = self.rings.get(name)?.as_slice();
179        for at in path {
180            level = &level.get(*at)?.slices;
181        }
182        Some(level)
183    }
184
185    /// The turn of the ring at `path` of wheel `name`: the wheel's own at
186    /// the top, the folder's own inside one, each the others' none of.
187    #[must_use]
188    pub fn start_of(&self, name: &str, path: &[usize]) -> f32 {
189        match path.split_last() {
190            None => self.meta.get(name).and_then(|meta| meta.start),
191            Some((at, above)) => self
192                .level(name, above)
193                .and_then(|ring| ring.get(*at))
194                .and_then(|folder| folder.start),
195        }
196        .unwrap_or(0.0)
197    }
198}

200/// How a committed leaf fires.
201#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
202pub enum FireMode {
203    /// When the wheel's button is let go.
204    #[default]
205    Release,
206    /// The moment the stick is pushed far enough, with no wait.
207    Edge,
208    /// Once committed, when the stick is drawn back from its furthest.
209    Retract,
210}
211
212impl FireMode {
213    /// As written; anything else is [`Self::Release`], so a slip never
214    /// leaves a wheel that cannot fire.
215    #[must_use]
216    pub fn named(name: &str) -> Self {
217        match name {
218            "edge" => Self::Edge,
219            "retract" => Self::Retract,
220            _ => Self::Release,
221        }
222    }
223}
224
225/// How `slice` fires: its own fire type, else `default`; none for one that
226/// only holds its place.
227fn fire_mode(slice: &Slice, default: FireMode) -> Option<FireMode> {
228    match slice.fire_type.as_deref() {
229        Some("none") => None,
230        Some(named) => Some(FireMode::named(named)),
231        None => Some(default),
232    }
233}

235/// One slice's place on the ring, in degrees from straight up, clockwise:
236/// its leading edge and its width.
237#[derive(Clone, Copy, Debug, PartialEq)]
238pub struct Seat {
239    /// Its leading edge.
240    pub start: f32,
241    /// Its width.
242    pub span: f32,
243}

245impl Seat {
246    /// The middle of its arc.
247    #[must_use]
248    pub fn centre(self) -> f32 {
249        self.start + self.span / 2.0
250    }
251}

253/// The widths `spans` ask for, none a share of what is left, laid out as
254/// abutting seats closing at 360, the first centred at `start`.
255#[must_use]
256pub fn lay_out(spans: &[Option<f32>], start: f32) -> Vec<Seat> {
257    let count = spans.len();
258    if count == 0 {
259        return Vec::new();
260    }
261    let explicit: Vec<Option<f32>> = spans
262        .iter()
263        .map(|span| span.map(|width| width.max(MIN_SPAN)))
264        .collect();
265    let taken: f32 = explicit.iter().flatten().sum();
266    let free = explicit.iter().filter(|span| span.is_none()).count();
267    #[expect(clippy::cast_precision_loss, reason = "a few slices")]
268    let share = if free > 0 {
269        ((360.0 - taken) / free as f32).max(0.0)
270    } else {
271        0.0
272    };
273    let mut widths: Vec<f32> = explicit.iter().map(|span| span.unwrap_or(share)).collect();
274    let total: f32 = widths.iter().sum();
275    if total <= f32::EPSILON {
276        #[expect(clippy::cast_precision_loss, reason = "a few slices")]
277        let even = 360.0 / count as f32;
278        widths = vec![even; count];
279    } else if (total - 360.0).abs() > 1e-3 {
280        let scale = 360.0 / total;
281        for width in &mut widths {
282            *width *= scale;
283        }
284    }
285    let mut edge = start - widths[0] / 2.0;
286    widths
287        .into_iter()
288        .map(|span| {
289            let seat = Seat { start: edge, span };
290            edge += span;
291            seat
292        })
293        .collect()
294}

296/// The seat whose arc the stick at `(x, y)` (y up) points into, whatever
297/// how far out; none with no seats.
298#[must_use]
299pub fn seat_at_angle(x: f32, y: f32, seats: &[Seat]) -> Option<usize> {
300    let first = seats.first()?;
301    let rel = (x.atan2(y).to_degrees() - first.start).rem_euclid(360.0);
302    let mut reached = 0.0;
303    for (at, seat) in seats.iter().enumerate() {
304        reached += seat.span;
305        if rel < reached {
306            return Some(at);
307        }
308    }
309    Some(seats.len() - 1)
310}

312/// The angle, from straight up and clockwise, of a side of the screen.
313fn anchor(side: &str) -> f32 {
314    match side {
315        "up" => 0.0,
316        "up-right" => 45.0,
317        "right" => 90.0,
318        "down-right" => 135.0,
319        "left" => 270.0,
320        "up-left" => 315.0,
321        "down-left" => 225.0,
322        _ => 180.0,
323    }
324}

326/// The smallest gap between two angles, 0 to 180.
327#[cfg(test)]
328#[must_use]
329pub fn gap(a: f32, b: f32) -> f32 {
330    let between = (a - b).rem_euclid(360.0);
331    between.min(360.0 - between)
332}

364/// A ring as it is shown: its slices, Back added inside a folder, each
365/// slice's place among the real ones (none for the added Back), and the
366/// seats.
367#[derive(Clone, Debug, PartialEq)]
368pub struct View {
369    /// The slices shown, Back among them inside a folder.
370    pub slices: Vec<Slice>,
371    real: Vec<Option<usize>>,
372    /// Each slice's place on the ring, in the slices' order.
373    pub seats: Vec<Seat>,
374}
375
376impl View {
377    /// The ring of `real` slices, turned by `start`, its own turn ([`Wheels::start_of`]):
378    /// at the top, laid out from it; in a folder (`in_folder`), with Back
379    /// added last and the ring turned so it sits toward `back_side`, unless
380    /// that is `none` or a slice is marked `back` already, and then turned
381    /// by `start` further.
382    #[must_use]
383    pub fn build(real: &[Slice], in_folder: bool, back_side: &str, start: f32) -> Self {
384        let spans: Vec<Option<f32>> = real.iter().map(|slice| slice.span).collect();
385        if !in_folder || back_side == "none" || real.iter().any(|slice| slice.back) {
386            return Self {
387                slices: real.to_vec(),
388                real: (0..real.len()).map(Some).collect(),
389                seats: lay_out(&spans, start),
390            };
391        }
392        let mut slices = real.to_vec();
393        slices.push(Slice {
394            label: "< Back".to_owned(),
395            command: BACK.to_owned(),
396            ..Slice::default()
397        });
398        let mut indexes: Vec<Option<usize>> = (0..real.len()).map(Some).collect();
399        indexes.push(None);
400        let mut spans = spans;
401        spans.push(None);
402        let unturned = lay_out(&spans, 0.0);
403        let turn = anchor(back_side) - unturned[slices.len() - 1].centre() + start;
404        Self {
405            slices,
406            real: indexes,
407            seats: lay_out(&spans, turn),
408        }
409    }
410
411    /// What is shown at `shown`: a real slice, by its place, or the added
412    /// Back.
413    #[must_use]
414    pub fn real(&self, shown: usize) -> Option<Place> {
415        self.real
416            .get(shown)
417            .map(|real| real.map_or(Place::Back, Place::Real))
418    }
419
420    /// The seat under the stick at `(x, y)`, past its slice's floor (its
421    /// own `inner`, else `deadzone`); none for a slice that only holds its
422    /// place.
423    fn seat_at(&self, x: f32, y: f32, deadzone: f32) -> Option<usize> {
424        let seat = seat_at_angle(x, y, &self.seats)?;
425        let slice = self.slices.get(seat);
426        if slice.is_some_and(Slice::is_none_type) {
427            return None;
428        }
429        let floor = slice
430            .and_then(|slice| slice.inner)
431            .map_or(deadzone, |inner| f32::from(inner) / 100.0);
432        (x.hypot(y) >= floor).then_some(seat)
433    }
434
435    /// What the leaf shown at `shown` sends: none for Back, a folder, a
436    /// slice that only holds its place, or one with nothing to send.
437    #[must_use]
438    pub fn leaf(&self, shown: usize) -> Option<&Slice> {
439        if !matches!(self.real(shown), Some(Place::Real(_))) {
440            return None;
441        }
442        let slice = self.slices.get(shown)?;
443        let fires = !slice.is_folder()
444            && !is_back(slice)
445            && !slice.is_none_type()
446            && !slice.command.is_empty();
447        fires.then_some(slice)
448    }
449
450    fn is_back_seat(&self, shown: usize) -> bool {
451        matches!(self.real(shown), Some(Place::Back)) || self.slices.get(shown).is_some_and(is_back)
452    }
453}

455/// The feel the wheel is aimed with.
456#[derive(Clone, Copy, Debug, PartialEq)]
457pub struct Timing {
458    /// How far out, 0 to 1, the stick is pushed before a slice is aimed.
459    pub deadzone: f32,
460    /// Milliseconds on a leaf before it is committed.
461    pub aim_ms: u128,
462    /// On a folder or Back before the wheel goes into it or out.
463    pub nav_ms: u128,
464    /// How a leaf with no fire type of its own fires.
465    pub fire_mode: FireMode,
466    /// How far out, 0 to 1, an edge leaf fires.
467    pub edge_threshold: f32,
468    /// How far back from its furthest, 0 to 1, a retract leaf fires.
469    pub retract_delta: f32,
470}

472/// A wheel open: which, the folders gone into, and what is aimed.
473#[derive(Clone, Debug, PartialEq)]
474pub struct Open {
475    /// The wheel's name.
476    pub name: String,
477    /// The folders gone into, by their real places.
478    pub path: Vec<usize>,
479    /// The slice shown committed: what a release fires.
480    pub aimed: Option<usize>,
481    /// The slice shown under the stick, and since when.
482    candidate: Option<usize>,
483    since: Option<Instant>,
484    /// Nothing is aimed until the stick is back in the middle: after the
485    /// wheel went into a folder or out, or a fire.
486    rearm: bool,
487    /// How far out the stick has been since the leaf was committed.
488    peak: f32,
489}
490
491impl Open {
492    /// The wheel `name` opened; `spent`, nothing aimed until the stick is
493    /// back in the middle.
494    #[must_use]
495    pub fn new(name: &str, spent: bool) -> Self {
496        Self {
497            name: name.to_owned(),
498            path: Vec::new(),
499            aimed: None,
500            candidate: None,
501            since: None,
502            rearm: spent,
503            peak: 0.0,
504        }
505    }
506
507    /// Gone up out of a folder: whether there was one.
508    pub fn up(&mut self) -> bool {
509        if self.path.pop().is_none() {
510            return false;
511        }
512        self.settled();
513        true
514    }
515
516    fn settled(&mut self) {
517        self.aimed = None;
518        self.candidate = None;
519        self.since = None;
520        self.rearm = true;
521        self.peak = 0.0;
522    }
523
524    /// The stick at `(x, y)` at `now`, over `view`: a folder or Back gone
525    /// into or out of, a leaf committed; the leaf shown that fires now,
526    /// for an edge or retract one.
527    pub fn aim(
528        &mut self,
529        view: &View,
530        timing: &Timing,
531        (x, y): (f32, f32),
532        now: Instant,
533    ) -> Option<usize> {
534        let magnitude = x.hypot(y);
535        let candidate = view.seat_at(x, y, timing.deadzone);
536        let centred = candidate.is_none();
537        self.rearm &= !centred;
538        if self.candidate != candidate {
539            self.candidate = candidate;
540            self.since = Some(now);
541            self.aimed = None;
542            self.peak = 0.0;
543        }
544        if self.rearm {
545            return None;
546        }
547        let shown = candidate?;
548        let dwelt = self
549            .since
550            .map_or(0, |since| now.saturating_duration_since(since).as_millis());
551        if view.is_back_seat(shown) {
552            if dwelt >= timing.nav_ms {
553                self.path.pop();
554                self.settled();
555            }
556            return None;
557        }
558        let Some(Place::Real(real)) = view.real(shown) else {
559            return None;
560        };
561        let slice = view.slices.get(shown)?;
562        if slice.is_folder() {
563            if dwelt >= timing.nav_ms {
564                self.path.push(real);
565                self.settled();
566            }
567            return None;
568        }
569        match fire_mode(slice, timing.fire_mode)? {
570            FireMode::Edge => (magnitude >= timing.edge_threshold).then_some(shown),
571            FireMode::Retract => {
572                if dwelt < timing.aim_ms {
573                    return None;
574                }
575                if self.aimed != Some(shown) {
576                    self.aimed = Some(shown);
577                    self.peak = magnitude;
578                }
579                self.peak = self.peak.max(magnitude);
580                (magnitude <= self.peak - timing.retract_delta + 1e-4).then_some(shown)
581            }
582            FireMode::Release => {
583                if dwelt >= timing.aim_ms {
584                    self.aimed = Some(shown);
585                }
586                None
587            }
588        }
589    }
590
591    /// South, while the wheel is open: on what the stick is over now, with
592    /// no wait -- a leaf fired, a folder gone into, Back gone up out of.
593    /// The leaf shown that fires.
594    pub fn south(&mut self, view: &View) -> Option<usize> {
595        let shown = self.candidate.or(self.aimed)?;
596        if view.is_back_seat(shown) {
597            self.up();
598            return None;
599        }
600        let Some(Place::Real(real)) = view.real(shown) else {
601            return None;
602        };
603        let slice = view.slices.get(shown)?;
604        if slice.is_folder() {
605            self.path.push(real);
606            self.settled();
607            return None;
608        }
609        view.leaf(shown).map(|_| shown)
610    }
611}

1//! A controller file: `[controller]`, each bind's key and what it does,
2//! and `[controller_overlay]`, the binds the legend lists.
3//!
4//! ```toml
5//! [controller]
6//! south = { macro_text = "look\r" }          # commands, sent
7//! east = { hotbar = "combat", button = "hide" }  # a hotbar's button
8//! l2 = "controller_modifier"                 # held for chords
9//! "l2+dpad_up" = "scroll_current_window_up_page"
10//! select = "controller_overlay"
11//!
12//! [controller_overlay]
13//! buttons = ["l2", "l2+south"]
14//! ```
15//!
16//! A macro's text is sent as typed, a trailing `\r` or none alike, and a
17//! `\r` inside it parts two commands. An action is one of Hydra's, by its
18//! keybinds name (`scroll_page_up`), or by the name a controller file
19//! written for Vellum gives it (`scroll_current_window_up_page`), so such
20//! a file works as it is. A hotbar's button is pressed as a click on it
21//! would be, by its bar and its id or nickname; a wheel's slice presses
22//! one with the command `.hotbar <bar> <button>`. The file's other tables are the wheel's, rumble's
23//! and tuning's, read where those are.

34/// A menu's key, pressed with the pad while a menu is open.
35#[derive(Clone, Copy, Debug, PartialEq, Eq)]
36pub enum Menu {
37    /// The entry above.
38    Up,
39    /// The entry below.
40    Down,
41    /// The entry above; in interact mode, the kind before.
42    Left,
43    /// The entry below; in interact mode, the kind after.
44    Right,
45    /// The menu closed with nothing chosen.
46    Cancel,
47}

49/// What a bind does.
50#[derive(Clone, Debug, PartialEq, Eq)]
51pub enum Bound {
52    /// What a key's macro does: commands sent, or one of Hydra's actions.
53    Does(Macro),
54    /// Nothing alone: held, it makes another button's chord.
55    Modifier,
56    /// The legend shown, or put away.
57    Overlay,
58    /// Interact mode begun, or left.
59    InteractMode,
60    /// The thing interact mode is on, acted on; a menu's entry chosen.
61    InteractSelect,
62    /// A menu moved in, or closed.
63    Menu(Menu),
64    /// The wheel held open: the default one, or the one named.
65    Wheel(Option<String>),
66}
67
68impl Bound {
69    /// What it does, for the legend: a macro's text, or the action's name.
70    #[must_use]
71    pub fn said(&self) -> String {
72        match self {
73            Self::Does(Macro::Send(text) | Macro::Fill(text)) => {
74                text.replace(['\r', '\n'], " ").trim().to_owned()
75            }
76            Self::Does(Macro::Act(action)) => action.name().to_owned(),
77            Self::Does(Macro::Hotbar { bar, button }) => format!("hotbar {bar} {button}"),
78            Self::Modifier => "controller_modifier".to_owned(),
79            Self::Overlay => "controller_overlay".to_owned(),
80            Self::InteractMode => "interact_mode".to_owned(),
81            Self::InteractSelect => "interact_select".to_owned(),
82            Self::Menu(menu) => format!("menu_{}", format!("{menu:?}").to_lowercase()),
83            Self::Wheel(None) => "controller_wheel".to_owned(),
84            Self::Wheel(Some(name)) => format!("controller_wheel:{name}"),
85        }
86    }
87}

89/// The action `name` names: Hydra's, the controller's own, or Vellum's
90/// name for one of Hydra's.
91///
92/// # Errors
93///
94/// Why it is none: one Hydra does not have.
95fn named(name: &str) -> Result<Bound, String> {
96    let name = name.trim();
97    if let Some(action) = Action::named(name) {
98        return Ok(Bound::Does(Macro::Act(action)));
99    }
100    if let Some(wheel) = name.strip_prefix("controller_wheel:") {
101        return Ok(Bound::Wheel(Some(wheel.trim().to_owned())));
102    }
103    let act = |action| Bound::Does(Macro::Act(action));
104    Ok(match name.to_ascii_lowercase().as_str() {
105        "controller_modifier" | "controller_shift" => Bound::Modifier,
106        "controller_overlay" => Bound::Overlay,
107        "controller_wheel" => Bound::Wheel(None),
108        "interact_mode" => Bound::InteractMode,
109        "interact_select" => Bound::InteractSelect,
110        "menu_up" => Bound::Menu(Menu::Up),
111        "menu_down" => Bound::Menu(Menu::Down),
112        "menu_left" => Bound::Menu(Menu::Left),
113        "menu_right" => Bound::Menu(Menu::Right),
114        "menu_cancel" => Bound::Menu(Menu::Cancel),
115        "scroll_current_window_up_one" => act(Action::ScrollLineUp),
116        "scroll_current_window_down_one" => act(Action::ScrollLineDown),
117        "scroll_current_window_up_page" => act(Action::ScrollPageUp),
118        "scroll_current_window_down_page" => act(Action::ScrollPageDown),
119        "scroll_current_window_home" => act(Action::ScrollTop),
120        "scroll_current_window_end" => act(Action::ScrollBottom),
121        "send_last_command" => act(Action::RepeatLast),
122        "send_second_last_command" => act(Action::RepeatSecondLast),
123        "prev_tab" => act(Action::PreviousTab),
124        _ => return Err(format!("{name} is not one of Hydra's actions")),
125    })
126}

128/// One controller file: its binds by key, and the legend's entries when it
129/// names them.
130#[derive(Clone, Debug, Default, PartialEq)]
131pub struct ControllerFile {
132    /// What each key does, by its key as written in order (`l2+south`).
133    pub binds: BTreeMap<String, Bound>,
134    /// The keys the legend lists; `None` when the file names none.
135    pub overlay: Option<Vec<String>>,
136    /// How the controller feels; `None` when the file says nothing of it.
137    pub tuning: Option<super::tuning::Tuning>,
138    /// Its wheels (`wheel.rs`).
139    pub wheels: super::wheel::Wheels,
140    /// Its rumble; `None` when the file says nothing of it.
141    pub rumble: Option<super::rumble::Rumble>,
142}
143
144impl ControllerFile {
145    /// Read `path`; `None` when there is no file. What is wrong in it is
146    /// said, and that bind left out.
147    #[must_use]
148    pub fn load(path: &Path) -> Option<(Self, Vec<String>)> {
149        let text = std::fs::read_to_string(path).ok()?;
150        let (file, problems) = Self::read(&text);
151        let problems = problems
152            .into_iter()
153            .map(|problem| format!("{}: {problem}.", path.display()))
154            .collect();
155        Some((file, problems))
156    }
157
158    /// Hydra's own binds.
159    #[must_use]
160    pub fn hydras() -> (Self, Vec<String>) {
161        Self::read(HYDRAS)
162    }
163
164    /// The file `text`, and what is wrong in it.
165    pub fn read(text: &str) -> (Self, Vec<String>) {
166        let mut file = Self::default();
167        let table: toml::Table = match text.parse() {
168            Ok(table) => table,
169            Err(why) => return (file, vec![format!("not read: {}", why.message())]),
170        };
171        let mut problems = Vec::new();
172        if let Some(binds) = table.get("controller").and_then(toml::Value::as_table) {
173            for (written, value) in binds {
174                match bind(written, value) {
175                    Ok((key, bound)) => {
176                        file.binds.insert(key, bound);
177                    }
178                    Err(why) => problems.push(format!("{written}: {why}")),
179                }
180            }
181        }
182        let (wheels, wrong) = super::wheel::Wheels::read(&table);
183        file.wheels = wheels;
184        problems.extend(wrong);
185        match super::rumble::Rumble::read(&table) {
186            Some(Ok(rumble)) => file.rumble = Some(rumble),
187            Some(Err(why)) => problems.push(why),
188            None => {}
189        }
190        if table.contains_key("controller_tuning") {
191            file.tuning = Some(super::tuning::Tuning::read(&table));
192        }
193        if let Some(overlay) = table
194            .get("controller_overlay")
195            .and_then(toml::Value::as_table)
196        {
197            let buttons = overlay
198                .get("buttons")
199                .and_then(toml::Value::as_array)
200                .map(|buttons| {
201                    buttons
202                        .iter()
203                        .filter_map(toml::Value::as_str)
204                        .filter_map(BindKey::parse)
205                        .map(|key| key.canonical())
206                        .collect()
207                });
208            file.overlay = buttons;
209        }
210        (file, problems)
211    }
212}

214/// A bind to a hotbar's button: `{ hotbar = "combat", button = "hide" }`,
215/// the button by its id or nickname.
216fn hotbar(table: &toml::Table) -> Result<Macro, String> {
217    let word = |key: &str| {
218        table
219            .get(key)
220            .and_then(toml::Value::as_str)
221            .map(str::to_owned)
222    };
223    match (word("hotbar"), word("button")) {
224        (Some(bar), Some(button)) => Ok(Macro::Hotbar { bar, button }),
225        _ => Err("a hotbar bind names its hotbar and its button".to_owned()),
226    }
227}

229/// The bind written `written = value`: its key in order, and what it does.
230fn bind(written: &str, value: &toml::Value) -> Result<(String, Bound), String> {
231    let key = BindKey::parse(written).ok_or("no button")?;
232    if !key.known() {
233        return Err("names a button a controller does not have".to_owned());
234    }
235    let bound = match value {
236        toml::Value::String(name) => named(name)?,
237        toml::Value::Table(table) if table.contains_key("hotbar") => {
238            let made = hotbar(table)?;
239            made.check()?;
240            Bound::Does(made)
241        }
242        toml::Value::Table(table) => {
243            let text = table
244                .get("macro_text")
245                .and_then(toml::Value::as_str)
246                .ok_or("neither an action nor a macro_text")?;
247            let text = text.trim_end_matches(['\r', '\n']);
248            if text.trim().is_empty() {
249                return Err("an empty macro".to_owned());
250            }
251            let made = Macro::Send(text.to_owned());
252            made.steps()?;
253            Bound::Does(made)
254        }
255        _ => return Err("neither an action nor a macro_text".to_owned()),
256    };
257    if bound == Bound::Modifier && !key.mods.is_empty() {
258        return Err("a modifier is a button alone".to_owned());
259    }
260    Ok((key.canonical(), bound))

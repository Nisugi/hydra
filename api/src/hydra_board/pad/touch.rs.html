1//! The web page's touch wheel: a character's wheel as its controller files
2//! hold it -- every character's, its own over it -- for a phone to hold
3//! open with its thumb.
4//!
5//! The page is given the wheel's shape alone: labels, colours, widths,
6//! folders, where Back sits, and the feel it aims with. A slice fired is
7//! named by its place, and its command is read here, from the files, so
8//! nothing the page says is sent as a command of its own. A slice that
9//! needs interact mode's focus, which a phone has not, is refused.
10
11use std::path::Path;

13use super::interact::placed;
14use super::wheel::{DEFAULT, Slice};
15use super::{Binds, ControllerFile, FILE, character_path};

17/// The wheel a phone opens: this one when a controller file has it, else
18/// the default.
19pub const TOUCH: &str = "touch";
20
21/// A wheel as the page draws and aims it.
22#[derive(Clone, Debug, PartialEq, serde::Serialize)]
23pub struct TouchWheel {
24    /// The wheel's name, which a slice fired is named under.
25    pub name: String,
26    /// Where its first slice sits, in degrees from straight up, clockwise.
27    pub start: f32,
28    /// Where Back sits inside a folder: `down`, `up-left`, `none`.
29    pub back: String,
30    /// How far from the middle, of the wheel's radius, a slice can be
31    /// aimed.
32    pub deadzone: f32,
33    /// How long a leaf is held to be committed, in milliseconds.
34    pub aim_ms: u32,
35    /// How long a folder or Back is held to go into it or out.
36    pub nav_ms: u32,
37    /// Its slices.
38    pub slices: Vec<TouchSlice>,
39}
40
41/// One slice, without its command.
42#[derive(Clone, Debug, PartialEq, serde::Serialize)]
43pub struct TouchSlice {
44    /// What it says.
45    pub label: String,
46    /// Its tint, `#rrggbb`.
47    pub color: Option<String>,
48    /// Its width in degrees; none, a share of what is left.
49    pub span: Option<f32>,
50    /// How far, in percent, it is pushed before it can be aimed.
51    pub inner: Option<u8>,
52    /// Aimed, it goes up out of the folder.
53    pub back: bool,
54    /// It only holds its place.
55    pub none: bool,
56    /// Its own slices: a folder.
57    pub slices: Vec<TouchSlice>,
58}
59
60impl TouchSlice {
61    fn of(slice: &Slice) -> Self {
62        Self {
63            label: slice.label.clone(),
64            color: slice.color.clone(),
65            span: slice.span,
66            inner: slice.inner,
67            back: slice.back,
68            none: slice.is_none_type(),
69            slices: slice.slices.iter().map(Self::of).collect(),
70        }
71    }
72}

74/// The binds for the character `name` on the game whose instance is
75/// `instance`, from `data`: every character's file, or Hydra's own when
76/// there is none, and the character's own over it.
77fn binds(data: &Path, instance: Option<&str>, name: &str) -> Binds {
78    let every = ControllerFile::load(&data.join(FILE))
79        .map_or_else(|| ControllerFile::hydras().0, |(file, _)| file);
80    let own_path = character_path(&data.join("layouts"), instance, name);
81    let own = std::fs::read_to_string(own_path)
82        .ok()
83        .map(|text| ControllerFile::read(&text).0);
84    Binds::of(&every, own.as_ref())
85}

87/// The touch wheel of `name` on `instance`, from `data`: `touch` when
88/// there is one, else the default; none when neither has a slice.
89#[must_use]
90pub fn touch_wheel(data: &Path, instance: Option<&str>, name: &str) -> Option<TouchWheel> {
91    let binds = binds(data, instance, name);
92    let wheel = [TOUCH, DEFAULT].into_iter().find(|wheel| {
93        binds
94            .wheels
95            .rings
96            .get(*wheel)
97            .is_some_and(|ring| !ring.is_empty())
98    })?;
99    let tuning = &binds.tuning;
100    Some(TouchWheel {
101        name: wheel.to_owned(),
102        start: binds
103            .wheels
104            .meta
105            .get(wheel)
106            .and_then(|meta| meta.start)
107            .unwrap_or(0.0),
108        back: tuning.back_slice.clone(),
109        deadzone: f32::from(tuning.deadzone) / 100.0,
110        aim_ms: tuning.aim_dwell_ms,
111        nav_ms: tuning.nav_dwell_ms,
112        slices: binds.wheels.rings[wheel]
113            .iter()
114            .map(TouchSlice::of)
115            .collect(),
116    })
117}

119/// The command of the slice at `path` -- each folder's place, then the
120/// slice's -- of wheel `wheel`, as `name`'s files have it.
121///
122/// # Errors
123///
124/// Why it is not sent: no such slice, or one that is a folder, Back, only
125/// holds its place, or needs interact mode's focus.
126pub fn touch_command(
127    data: &Path,
128    (instance, name): (Option<&str>, &str),
129    wheel: &str,
130    path: &[usize],
131) -> Result<String, String> {
132    let binds = binds(data, instance, name);
133    let (last, folders) = path.split_last().ok_or("No slice was named.")?;
134    let slice = binds
135        .wheels
136        .level(wheel, folders)
137        .and_then(|level| level.get(*last))
138        .ok_or("That slice is no longer on the wheel.")?;
139    if slice.is_folder() || slice.back || slice.is_none_type() || slice.command.is_empty() {
140        return Err(format!("{} sends nothing.", slice.label));
141    }
142    placed(&slice.command, None).ok_or_else(|| {
143        format!(
144            "{} needs a thing in interact mode's focus, which a phone has not.",
145            slice.label
146        )
147    })
148}

150#[cfg(test)]
151mod tests {
152    use super::*;
153
154    const FILE_TEXT: &str = r#"
155[[controller_wheel]]
156label = "Default"
157command = "look"
158
159[[controller_wheels.touch]]
160label = "Hide"
161command = "hide"
162
163[[controller_wheels.touch]]
164label = "Spells"
165[[controller_wheels.touch.slices]]
166label = "Heal"
167command = "prep 1101\rcast"
168
169[[controller_wheels.touch]]
170label = "Aimed"
171command = "attack #<target_id>"
172"#;
173
174    fn data() -> std::path::PathBuf {
175        let data = std::env::temp_dir().join(format!("hydra-touch-{}", std::process::id()));
176        let _ = std::fs::create_dir_all(&data);
177        std::fs::write(data.join(FILE), FILE_TEXT).expect("written");
178        data
179    }
180
181    /// The touch wheel before the default; its shape without its commands;
182    /// a slice's command read from the file by its place; one needing a
183    /// focus, a folder and one not there refused.
184    #[test]
185    fn the_touch_wheel_and_its_commands() {
186        let data = data();
187        let wheel = touch_wheel(&data, Some("Prime"), "Tester").expect("a wheel");
188        assert_eq!(wheel.name, TOUCH);
189        assert_eq!(
190            wheel
191                .slices
192                .iter()
193                .map(|s| s.label.as_str())
194                .collect::<Vec<_>>(),
195            ["Hide", "Spells", "Aimed"]
196        );
197        assert_eq!(wheel.slices[1].slices[0].label, "Heal");
198        let command = |path: &[usize]| touch_command(&data, (Some("Prime"), "Tester"), TOUCH, path);
199        assert_eq!(command(&[0]), Ok("hide".to_owned()));
200        assert_eq!(command(&[1, 0]), Ok("prep 1101\rcast".to_owned()));
201        assert!(command(&[1]).is_err(), "a folder");
202        assert!(command(&[2]).is_err(), "needs a focus");
203        assert!(command(&[9]).is_err(), "not there");
204        let _ = std::fs::remove_dir_all(&data);
205    }

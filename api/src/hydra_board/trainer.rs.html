1//! One character's skill trainer: what `GOALS` was asked for, and the
2//! skill manager it opened.
3//!
4//! `goals` typed asks the game for the skill manager's one-time link, and
5//! the link it answers with (`<LaunchURL>`) is the trainer's: Hydra reads
6//! the page itself and the Skill trainer widget shows it. `goals web` asks
7//! the same and opens the link in the browser, the website's own trainer:
8//! typed on the web page, the page's browser, where the page offers it for
9//! a tap. A link nothing asked for opens in the browser too, as the game
10//! means, and only when it is play.net's.
11//!
12//! **Profiles.** A set of goals saved under a name (`normal`, `ensorcell`)
13//! and loaded again later, each goal stepped to its ranks as the website
14//! would. Kept per character beside its layout, in
15//! `<game>_<name>.goals.json`.
16

23/// What the next link is for.
24#[derive(Clone, Copy, Debug, PartialEq, Eq)]
25pub enum Asked {
26    /// The trainer.
27    Trainer,
28    /// The browser.
29    Web,
30    /// The browser of the web page it was typed on: the page offers it.
31    Page,
32}

34/// Where the trainer is.
35#[derive(Clone, Debug, Default, PartialEq, Eq)]
36pub enum Status {
37    /// Nothing asked yet.
38    #[default]
39    Idle,
40    /// `goals` sent; the link, then the page, to come.
41    Loading,
42    /// The goals being applied.
43    Applying,
44    /// Something to say: the goals were saved, or why not.
45    Said(String),
46}

48/// What the trainer shows: where it is, and the skill manager read.
49#[derive(Clone, Debug, Default, PartialEq, Eq)]
50pub struct TrainerView {
51    /// Where the trainer is.
52    pub status: Status,
53    /// The skill manager as it was read; none until it is.
54    pub goals: Option<SkillGoals>,
55    /// The last apply saved the goals: what a window opened for them
56    /// closes on.
57    pub saved: bool,
58}
59
60impl TrainerView {
61    /// What the trainer says of where it is, under its buttons: a hint of
62    /// Hydra's own (`false`), or what was said of the goals saved (`true`);
63    /// nothing once read and at rest.
64    #[must_use]
65    pub fn said(&self) -> Option<(&str, bool)> {
66        match &self.status {
67            Status::Idle if self.goals.is_none() => {
68                Some(("Load reads your skills from play.net.", false))
69            }
70            Status::Idle => None,
71            Status::Loading => Some(("Reading the skill manager…", false)),
72            Status::Applying => Some(("Applying…", false)),
73            Status::Said(said) => Some((said.as_str(), true)),
74        }
75    }
76}

120/// What a line typed for the game becomes: the line to send, and what the
121/// link it brings is for, when it is `goals` or `goals web`.
122#[must_use]
123pub fn typed(line: &str) -> (String, Option<Asked>) {
124    let words: Vec<&str> = line.split_whitespace().collect();
125    match words.as_slice() {
126        [goals] if goals.eq_ignore_ascii_case("goals") => (line.to_owned(), Some(Asked::Trainer)),
127        [goals, web] if goals.eq_ignore_ascii_case("goals") && web.eq_ignore_ascii_case("web") => {
128            ("goals".to_owned(), Some(Asked::Web))
129        }
130        _ => (line.to_owned(), None),
131    }
132}

134/// What to do with a link the game sent.
135#[derive(Clone, Debug, PartialEq, Eq)]
136pub enum Link {
137    /// Read it as the skill manager.
138    Read(String),
139    /// Open it in the browser.
140    Open(String),
141    /// Offer it on the web page that asked.
142    Page(String),
143    /// Nothing: not play.net's.
144    Refused,
145}
146
147/// `url` made whole, when it is play.net's: a path on it, or a link to it.
148fn play_net(url: &str) -> Option<String> {
149    let url = url.trim();
150    if url.starts_with('/') && !url.starts_with("//") {
151        return Some(format!("https://www.play.net{url}"));
152    }
153    url.strip_prefix("https://www.play.net/")
154        .map(|_| url.to_owned())
155}

157/// What the trainer's widget asks.
158#[derive(Clone, Debug, PartialEq, Eq)]
159pub enum Ask {
160    /// Send `goals`, for the trainer.
161    Load,
162    /// Send `goals`, for the browser.
163    Web,
164    /// Change the goals.
165    Change(Change),
166    /// Apply them on the website.
167    Apply,
168    /// Save, load or delete a profile.
169    Profile(Profiled),
170}

172/// What is done with a profile.
173#[derive(Clone, Debug, PartialEq, Eq)]
174pub enum Profiled {
175    /// The goals now, saved as `name`.
176    Save {
177        /// The profile's name.
178        name: String,
179        /// Each goal, by the skill's id.
180        goals: BTreeMap<u32, i64>,
181    },
182    /// The profile `name` loaded into the trainer.
183    Load(String),
184    /// The profile `name` deleted.
185    Delete(String),
186}

188/// A change to the goals asked from the trainer's widget.
189#[derive(Clone, Debug, PartialEq, Eq)]
190pub enum Change {
191    /// `id` up or down `times` ranks.
192    Step {
193        /// The skill's id.
194        id: u32,
195        /// Up rather than down.
196        up: bool,
197        /// How many ranks.
198        times: u32,
199    },
200    /// Every goal back to the ranks it has.
201    Reset,
202    /// Every goal stepped to the profile `name`'s.
203    Aim {
204        /// The profile's name.
205        name: String,
206        /// The profile's goals, by the skill's id.
207        goals: BTreeMap<u32, i64>,
208    },
209}

211/// The file's version: one that reads another is not this one's.
212const VERSION: u32 = 1;

214/// A character's profiles: each a goal by the id it is set on, by name.
215#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
216pub struct GoalProfiles {
217    version: u32,
218    /// Each profile's goals, by the skill's id, by the profile's name.
219    #[serde(default)]
220    pub profiles: BTreeMap<String, BTreeMap<u32, i64>>,
221}
222
223impl Default for GoalProfiles {
224    fn default() -> Self {
225        Self {
226            version: VERSION,
227            profiles: BTreeMap::new(),
228        }
229    }
230}

232impl GoalProfiles {
233    /// The profiles' names, in order.
234    #[must_use]
235    pub fn names(&self) -> Vec<String> {
236        self.profiles.keys().cloned().collect()
237    }
238
239    /// Read from `dir`, as `character` on `instance` keeps them; none when
240    /// there is no file, or one of another version.
241    #[must_use]
242    pub fn load(dir: &Path, instance: Option<&str>, character: &str) -> Self {
243        std::fs::read_to_string(file(dir, instance, character))
244            .ok()
245            .and_then(|text| serde_json::from_str::<Self>(&text).ok())
246            .filter(|kept| kept.version == VERSION)
247            .unwrap_or_default()
248    }
249
250    /// Write to `dir`. A file already there that [`Self::load`] could not
251    /// read is kept beside it as `.unread`.
252    ///
253    /// # Errors
254    ///
255    /// The file could not be written, or an unread file could not be kept.
256    pub fn save(&self, dir: &Path, instance: Option<&str>, character: &str) -> std::io::Result<()> {
257        let path = file(dir, instance, character);
258        let text = serde_json::to_string_pretty(self).map_err(std::io::Error::other)?;
259        // With the next batch of saves, an unreadable file kept aside first.
260        crate::store::whole_keeping_unread(dir, &path, text.into_bytes(), |text| {
261            serde_json::from_str::<Self>(text).is_ok_and(|kept| kept.version == VERSION)
262        });
263        Ok(())
264    }
265}

267impl GoalProfiles {
268    /// The profile `name` kept as `goals`, or deleted for none, in the file
269    /// `character` on `instance` keeps in `dir`: made to the file as it
270    /// stands when it is written, so what the window and a web page each
271    /// change of it is kept, neither writing over the other. A file that
272    /// does not read is left as it is, and the change not made.
273    pub fn change(
274        dir: &Path,
275        instance: Option<&str>,
276        character: &str,
277        (name, goals): (&str, Option<BTreeMap<u32, i64>>),
278    ) {
279        let name = name.to_owned();
280        crate::store::edit(
281            dir,
282            &file(dir, instance, character),
283            Box::new(move |text: &str| {
284                let mut kept = if text.trim().is_empty() {
285                    Self::default()
286                } else {
287                    serde_json::from_str::<Self>(text)
288                        .ok()
289                        .filter(|kept| kept.version == VERSION)
290                        .ok_or_else(|| "the goal profiles do not read".to_owned())?
291                };
292                match goals {
293                    Some(goals) => {
294                        kept.profiles.insert(name, goals);
295                    }
296                    None => {
297                        kept.profiles.remove(&name);
298                    }
299                }
300                serde_json::to_string_pretty(&kept).map_err(|why| why.to_string())
301            }),
302        );
303    }
304}

306/// The file `character` on `instance` keeps its profiles in.
307fn file(dir: &Path, instance: Option<&str>, character: &str) -> PathBuf {
308    let layout = crate::kept::file(dir, instance, character);
309    layout.with_extension("goals.json")
310}

312impl Trainer {
313    /// The page was read, in `session`, or why not.
314    pub fn read(&mut self, read: Result<(hydra_net::trainer::Trainer, SkillGoals), String>) {
315        match read {
316            Ok((session, goals)) => {
317                self.session = Some(session);
318                self.view.goals = Some(goals);
319                self.view.status = Status::Idle;
320            }
321            Err(why) => self.view.status = Status::Said(format!("Not read: {why}.")),
322        }
323    }
324
325    /// The goals were applied, or why not. Saved, they are the ranks the
326    /// character is to have: the trainer starts from them.
327    pub fn applied(&mut self, applied: Result<hydra_net::trainer::Applied, String>) {
328        self.view.saved = applied.is_ok();
329        let said = match applied {
330            Ok(applied) => {
331                if let Some(goals) = &mut self.view.goals {
332                    goals.ranks.clone_from(&goals.goals);
333                    goals.circle_ranks.clone_from(&goals.circle_goals);
334                    goals.lore_ranks.clone_from(&goals.lore_goals);
335                    goals.physical_spent = 0;
336                    goals.mental_spent = 0;
337                }
338                match applied {
339                    hydra_net::trainer::Applied::Saved => "Goals saved, and the game told.",
340                    hydra_net::trainer::Applied::NotTold => {
341                        "Goals saved on play.net, but the game could not be told: be logged in, \
342                         and apply again."
343                    }
344                }
345                .to_owned()
346            }
347            Err(why) => format!("Not applied: {why}."),
348        };
349        self.view.status = Status::Said(said);
350    }
351
352    /// A change to the goals: `id` stepped up or down `times` ranks, every
353    /// goal put back, or a profile's aimed at.
354    pub fn change(&mut self, change: Change) {
355        let Some(goals) = self.view.goals.as_mut() else {
356            return;
357        };
358        match change {
359            Change::Step { id, up, times } => {
360                let _ = goals.step(id, times, up);
361            }
362            Change::Reset => goals.reset(),
363            Change::Aim { name, goals: aimed } => {
364                let said = if goals.aim(&aimed) {
365                    format!("Profile {name} loaded: Apply saves it.")
366                } else {
367                    format!(
368                        "Profile {name} loaded, but not every goal: not points enough, or past \
369                         the most ranks."
370                    )
371                };
372                self.view.status = Status::Said(said);
373            }
374        }
375    }
376
377    /// A link for the page that typed `goals web`, kept until a page takes
378    /// it ([`Trainer::page_link`]).
379    pub fn offer(&mut self, url: String) {
380        self.for_page = Some(url);
381    }
382
383    /// The link a page's `goals web` brought, taken: none once taken.
384    pub fn page_link(&mut self) -> Option<String> {
385        self.for_page.take()
386    }
387
388    /// `goals` was typed, for `asked`.
389    pub fn ask(&mut self, asked: Asked) {
390        self.asked = Some(asked);
391        if asked == Asked::Trainer {
392            self.view.status = Status::Loading;
393        }
394    }
395
396    /// The game sent the link `url`: what to do with it.
397    pub fn launched(&mut self, url: &str) -> Link {
398        let asked = self.asked.take();
399        let Some(url) = play_net(url) else {
400            if asked == Some(Asked::Trainer) {
401                self.view.status = Status::Said("the game's link was not play.net's".to_owned());
402            }
403            return Link::Refused;
404        };
405        match asked {
406            Some(Asked::Trainer) => Link::Read(url),
407            Some(Asked::Page) => Link::Page(url),
408            Some(Asked::Web) | None => Link::Open(url),
409        }
410    }
411}

1//! A Wizard front end folder brought into Hydra: its scripts as aliases,
2//! and where its profile's macros and highlights are, for the keys and the
3//! highlights to be imported by their own readers.
4//!
5//! # Scripts
6//!
7//! A Wizard script (`.Cmd` or `.Wiz`) of the kind a player writes by hand
8//! is a list of lines to send, with waits between:
9//!
10//! ```text
11//! #pool-return.Cmd
12//! put ;go bank
13//! pause 2
14//! put ASK attend TO RETURN %1
15//! waitrt
16//! put stow all box
17//! ```
18//!
19//! That is what an alias's target is ([`hydra_session::alias`]): the
20//! commands with `\r` between, a wait as `s2`, roundtime as `rt`, `%1` and
21//! `%2` as the words typed after the alias, `\1` and `\2`. So each script
22//! becomes an alias named after its file, typed with the command symbol
23//! first as the Wizard ran it: `.pool-return` with the box's name after
24//! it, the file's name alone left to the game. A `.Wiz` whose name a `.Cmd` also has is named
25//! `<name>-wiz`.
26//!
27//! Two more shapes are read. A script that runs another (`put .nexus`)
28//! has that script's lines put in at that point, the `.Cmd` of the name
29//! first and the `.Wiz` when there is no `.Cmd`. A script that is one loop,
30//! a label on its first line and `goto` it on its last with a pause before
31//! the jump, becomes the alias of its lines, to be run on the clock as
32//! `.every <pause> .<name>`, which the import says.
33//!
34//! The Wizard's scripting language also has matching, other labels,
35//! counters and variables, which an alias has not; a script using them
36//! needs Lich, which runs Wizard scripts as they are, and is said, not
37//! imported. A line the reader does not know (a typo, `fput`) is said and
38//! skipped.
39
40use std::collections::BTreeMap;
41use std::path::{Path, PathBuf};

43/// The words between two commands in an alias's target.
44const BREAK: &str = "\\r";
45
46/// How many scripts deep one may run another before the reader stops.
47const DEEPEST: usize = 4;
48
49/// A script read: its alias target, and what else it needs.
50#[derive(Debug, Clone, PartialEq)]
51pub struct Script {
52    /// The file's name: the alias's trigger after the command symbol.
53    pub trigger: String,
54    /// Its target: the commands and waits.
55    pub target: String,
56    /// Lines skipped, each said.
57    pub skipped: Vec<String>,
58    /// The seconds the script waited before looping, when it looped: the
59    /// alias is to be run as `.every <seconds> <trigger>`.
60    pub every: Option<f32>,
61}
62
63/// The scripts of a folder by name, lower case, for one to run another.
64pub type Others<'a> = &'a dyn Fn(&str) -> Option<String>;
65
66/// The script `text`, named `name`, as an alias; `others` answers the text
67/// of another script in the folder by its name, for one this one runs.
68///
69/// # Errors
70///
71/// The script needs Lich: it matches, jumps other than back to its top,
72/// counts or waits for text; it runs a script the folder has not, or runs
73/// scripts too deep; or it sends nothing.
74pub fn script(name: &str, text: &str, others: Others<'_>) -> Result<Script, String> {
75    let mut read = Read::default();
76    let lines: Vec<&str> = text.lines().collect();
77    let every = loop_of(&lines);
78    let body = match every {
79        Some((_, first, last)) => &lines[first..last],
80        None => &lines[..],
81    };
82    read.lines(body, others, 0)?;
83    if !read.pieces.iter().any(|piece| !is_wait(piece)) {
84        return Err("it sends nothing".to_owned());
85    }
86    Ok(Script {
87        trigger: name.to_owned(),
88        target: read.pieces.join(BREAK),
89        skipped: read.skipped,
90        every: every.map(|(seconds, _, _)| seconds),
91    })
92}

94/// A script read so far.
95#[derive(Default)]
96struct Read {
97    pieces: Vec<String>,
98    skipped: Vec<String>,
99}
100
101impl Read {
102    /// `lines` read, `depth` scripts in from the one asked for.
103    fn lines(&mut self, lines: &[&str], others: Others<'_>, depth: usize) -> Result<(), String> {
104        for (at, raw) in lines.iter().enumerate() {
105            let number = at + 1;
106            let line = raw.trim();
107            if line.is_empty() || line.starts_with('#') {
108                continue;
109            }
110            let (word, rest) = line.split_once(char::is_whitespace).unwrap_or((line, ""));
111            let rest = rest.trim();
112            match word.to_ascii_lowercase().as_str() {
113                "put" | "move" => {
114                    if rest.is_empty() {
115                        self.skipped
116                            .push(format!("line {number}: `put` with nothing to send"));
117                    } else if let Some(other) = rest.strip_prefix('.') {
118                        self.other(other.trim(), number, others, depth)?;
119                    } else {
120                        self.pieces.push(argument(rest, number)?);
121                    }
122                }
123                "pause" => {
124                    let seconds = if rest.is_empty() {
125                        1.0
126                    } else {
127                        rest.parse::<f32>()
128                            .map_err(|_| format!("line {number}: `pause {rest}` is not a time"))?
129                    };
130                    self.pieces.push(format!("s{seconds}"));
131                }
132                "waitrt" => self.pieces.push("rt".to_owned()),
133                "echo" => {}
134                "match" | "matchre" | "matchwait" | "goto" | "waitfor" | "waitforre" | "wait"
135                | "nextroom" | "counter" | "save" | "setvariable" | "deletevariable" | "shift"
136                | "exit" => {
137                    return Err(format!(
138                        "line {number} is `{word}`: the script needs Lich, which runs it as it is"
139                    ));
140                }
141                _ if word.ends_with(':') => {
142                    return Err(format!(
143                        "line {number} is a label: the script needs Lich, which runs it as it is"
144                    ));
145                }
146                _ => self
147                    .skipped
148                    .push(format!("line {number}: `{line}` is not understood")),
149            }
150        }
151        Ok(())
152    }
153
154    /// Another script, `name`, run from line `number`: its lines in here.
155    fn other(
156        &mut self,
157        name: &str,
158        number: usize,
159        others: Others<'_>,
160        depth: usize,
161    ) -> Result<(), String> {
162        if depth >= DEEPEST {
163            return Err(format!(
164                "line {number} runs `.{name}`, {DEEPEST} scripts deep: too deep to follow"
165            ));
166        }
167        let Some(text) = others(&name.to_ascii_lowercase()) else {
168            return Err(format!(
169                "line {number} runs `.{name}`, which the folder has not"
170            ));
171        };
172        let lines: Vec<&str> = text.lines().collect();
173        if loop_of(&lines).is_some() {
174            return Err(format!(
175                "line {number} runs `.{name}`, which loops: a loop runs on its own, as `.every`"
176            ));
177        }
178        self.lines(&lines, others, depth + 1)
179            .map_err(|why| format!("in `.{name}`, {why}"))
180    }
181}

183/// Whether `lines` are one loop: a label first, `goto` it last, and a
184/// `pause` just before the jump. The pause's seconds, and the lines
185/// between, as a range.
186fn loop_of(lines: &[&str]) -> Option<(f32, usize, usize)> {
187    let meaningful: Vec<(usize, &str)> = lines
188        .iter()
189        .enumerate()
190        .map(|(at, line)| (at, line.trim()))
191        .filter(|(_, line)| !line.is_empty() && !line.starts_with('#'))
192        .collect();
193    let (&(first, label), &(last, jump)) = (meaningful.first()?, meaningful.last()?);
194    let label = label.strip_suffix(':')?;
195    let target = jump
196        .split_once(char::is_whitespace)
197        .filter(|(word, _)| word.eq_ignore_ascii_case("goto"))
198        .map(|(_, target)| target.trim())?;
199    if !target.eq_ignore_ascii_case(label) {
200        return None;
201    }
202    let (before, pause) = meaningful[..meaningful.len() - 1].last().copied()?;
203    let seconds = pause
204        .split_once(char::is_whitespace)
205        .filter(|(word, _)| word.eq_ignore_ascii_case("pause"))
206        .and_then(|(_, seconds)| seconds.trim().parse::<f32>().ok())?;
207    Some((seconds, first + 1, before)).filter(|_| before > first + 1 && last > before)
208}

210/// `rest` with the Wizard's `%1`, `%2`... as the alias's `\1`, `\2`...
211///
212/// # Errors
213///
214/// A counter or a saved value: the script needs Lich.
215fn argument(rest: &str, number: usize) -> Result<String, String> {
216    let mut out = String::with_capacity(rest.len());
217    let mut letters = rest.chars().peekable();
218    while let Some(letter) = letters.next() {
219        if letter == '%' {
220            match letters.peek().copied() {
221                Some(digit) if digit.is_ascii_digit() && digit != '0' => {
222                    letters.next();
223                    out.push('\\');
224                    out.push(digit);
225                    // `%1%` is the Wizard's other spelling of `%1`.
226                    if letters.peek() == Some(&'%') {
227                        letters.next();
228                    }
229                }
230                _ => {
231                    return Err(format!(
232                        "line {number} uses a variable, a counter or a saved value: the script needs Lich, which runs it as it is"
233                    ));
234                }
235            }
236            continue;
237        }
238        out.push(letter);
239    }
240    Ok(out)
241}

243/// Whether a piece is a wait rather than a command.
244fn is_wait(piece: &str) -> bool {
245    piece == "rt"
246        || piece
247            .strip_prefix('s')
248            .is_some_and(|rest| rest.parse::<f32>().is_ok())
249}

251/// Every script in `folder`, by trigger, with what could not be read, a
252/// sentence each. Folders one deep are looked in too, as an unpacked
253/// archive has them.
254#[must_use]
255pub fn scripts(folder: &Path) -> (Vec<Script>, Vec<String>) {
256    let mut read = Vec::new();
257    let mut said = Vec::new();
258    let mut cmd: BTreeMap<String, String> = BTreeMap::new();
259    let mut wiz: BTreeMap<String, String> = BTreeMap::new();
260    for path in files(folder) {
261        let (Some(ext), Some(stem)) = (
262            path.extension().and_then(|e| e.to_str()),
263            path.file_stem().and_then(|s| s.to_str()),
264        ) else {
265            continue;
266        };
267        let is_cmd = ext.eq_ignore_ascii_case("cmd");
268        let is_wiz = ext.eq_ignore_ascii_case("wiz");
269        if !is_cmd && !is_wiz {
270            continue;
271        }
272        let Ok(text) = std::fs::read_to_string(&path) else {
273            said.push(format!("{} could not be read.", path.display()));
274            continue;
275        };
276        let name = stem.to_ascii_lowercase();
277        if is_cmd {
278            cmd.insert(name, text);
279        } else {
280            wiz.insert(name, text);
281        }
282    }
283    // One script runs another by name: the `.Cmd` of the name, else the `.Wiz`.
284    let others = |name: &str| cmd.get(name).or_else(|| wiz.get(name)).cloned();
285    for (name, text) in &cmd {
286        match script(name, text, &others) {
287            Ok(script) => read.push(script),
288            Err(why) => said.push(format!("{name}.Cmd: {why}.")),
289        }
290    }
291    for (name, text) in &wiz {
292        let trigger = if cmd.contains_key(name) {
293            format!("{name}-wiz")
294        } else {
295            name.clone()
296        };
297        match script(&trigger, text, &others) {
298            Ok(script) => read.push(script),
299            Err(why) => said.push(format!("{name}.Wiz: {why}.")),
300        }
301    }
302    read.sort_by(|a, b| a.trigger.cmp(&b.trigger));
303    (read, said)
304}

306/// A Wizard profile's files in `folder`, by profile name: the `.Mac`,
307/// `.His`, `.Nam` and `.Colors`, whichever are there.
308#[must_use]
309pub fn profiles(folder: &Path) -> BTreeMap<String, Profile> {
310    let mut found: BTreeMap<String, Profile> = BTreeMap::new();
311    for path in files(folder) {
312        let (Some(stem), Some(ext)) = (
313            path.file_stem().and_then(|s| s.to_str()),
314            path.extension().and_then(|e| e.to_str()),
315        ) else {
316            continue;
317        };
318        let slot = match ext.to_ascii_lowercase().as_str() {
319            "mac" => 0,
320            "his" => 1,
321            "nam" => 2,
322            "colors" => 3,
323            _ => continue,
324        };
325        let profile = found.entry(stem.to_owned()).or_default();
326        let place = [
327            &mut profile.macros,
328            &mut profile.strings,
329            &mut profile.names,
330            &mut profile.colors,
331        ];
332        *place[slot] = Some(path);
333    }
334    found
335}

337/// One Wizard profile's files, by path.
338#[derive(Debug, Default, Clone, PartialEq, Eq)]
339pub struct Profile {
340    /// `<name>.Mac`: the macros.
341    pub macros: Option<PathBuf>,
342    /// `<name>.His`: the highlight strings.
343    pub strings: Option<PathBuf>,
344    /// `<name>.Nam`: the highlighted names.
345    pub names: Option<PathBuf>,
346    /// `<name>.Colors`: the palette.
347    pub colors: Option<PathBuf>,
348}
349
350/// The files of `folder` and of the folders one deep in it.
351fn files(folder: &Path) -> Vec<PathBuf> {
352    let mut out = Vec::new();
353    let Ok(entries) = std::fs::read_dir(folder) else {
354        return out;
355    };
356    let mut entries: Vec<PathBuf> = entries.flatten().map(|e| e.path()).collect();
357    entries.sort();
358    for path in entries {
359        if path.is_dir() {
360            if let Ok(inner) = std::fs::read_dir(&path) {
361                let mut inner: Vec<PathBuf> = inner
362                    .flatten()
363                    .map(|e| e.path())
364                    .filter(|p| p.is_file())
365                    .collect();
366                inner.sort();
367                out.extend(inner);
368            }
369        } else {
370            out.push(path);
371        }
372    }
373    out
374}

384    #[test]
385    fn a_flat_script_is_an_alias() {
386        let text = "#pool-return.Cmd\nput ;go bank\npause 2\nput withdr 20000 silver\npause\nput ASK attend TO RETURN %1\nwaitrt\nput stow all box\necho done\n";
387        let read = script("pool-return", text, &none).unwrap_or_else(|e| panic!("{e}"));
388        assert_eq!(read.trigger, "pool-return");
389        assert_eq!(
390            read.target,
391            ";go bank\\rs2\\rwithdr 20000 silver\\rs1\\rASK attend TO RETURN \\1\\rrt\\rstow all box"
392        );
393        assert!(read.skipped.is_empty());
394        assert_eq!(read.every, None);
395        let steps = hydra_session::steps::steps(&read.target.replace("\\r", "\r"));
396        assert!(steps.is_ok(), "{steps:?}");
397    }

461    fn a_folder_s_scripts_and_profiles_are_found_one_deep() {
462        let dir = std::env::temp_dir().join(format!("hydra-wizard-{}", std::process::id()));
463        let _ = std::fs::remove_dir_all(&dir);
464        let inner = dir.join("Wizard Scripts");
465        std::fs::create_dir_all(&inner).unwrap_or_else(|e| panic!("{e}"));

467            inner.join("601.Cmd"),
468            "put prep 601\npause 3\nput evoke 601\n",
469        )
470        .unwrap_or_else(|e| panic!("{e}"));
471        std::fs::write(inner.join("601.Wiz"), "put prep 601\n").unwrap_or_else(|e| panic!("{e}"));
472        std::fs::write(inner.join("Pile.Cmd"), "put jump\nput .nexus\n")
473            .unwrap_or_else(|e| panic!("{e}"));
474        std::fs::write(inner.join("nexus.Wiz"), "put stance def\n")
475            .unwrap_or_else(|e| panic!("{e}"));
476        std::fs::write(inner.join("Loop.Cmd"), "top:\ngoto top\n")
477            .unwrap_or_else(|e| panic!("{e}"));
478        std::fs::write(dir.join("Widmer.His"), "014 \"wand\"\n").unwrap_or_else(|e| panic!("{e}"));

495        let profiles = profiles(&dir);
496        assert_eq!(profiles.len(), 1);
497        let widmer = &profiles["Widmer"];
498        assert!(widmer.strings.is_some() && widmer.macros.is_some());
499        assert!(widmer.names.is_none() && widmer.colors.is_none());
500        let _ = std::fs::remove_dir_all(&dir);
501    }

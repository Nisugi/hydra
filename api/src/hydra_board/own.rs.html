1//! Hydra's own settings, one set for every character, kept in the window's
2//! own file, `window.toml` in the data folder.
3//!
4//! They are:
5//!
6//! - **Recording each session's raw stream** to `.bytes` in the logs folder:
7//!   on in a development build, off in a release one, until the switch says
8//!   otherwise. `HYDRA_RECORD=1` or `0`
9//!   overrides the file, for a run from the terminal.
10//! - **Keeping the player log**: on unless switched off.
11//! - **The theme**: the one worn, Despana unless chosen;
12//!   and, with *Follow the computer* on, a light one worn while the
13//!   computer is in light mode.
14//! - **Zoom**: how large everything is drawn, 1 unless chosen; the title
15//!   bar's − and + step it by a tenth, between [`ZOOM_LEAST`] and
16//!   [`ZOOM_MOST`]. One size for every window: egui draws them all at one
17//!   zoom.
18//! - **Volume**: how loud every highlight's sound plays, a percent, 100
19//!   unless chosen; a highlight's own percent is of this.
20//! - **Ebon Gate merchants**: when one opens for business, a banner in the
21//!   window of each character the game tells, and a notification from the
22//!   computer, once however many it tells (`crate::merchants`); each on
23//!   unless switched off.
24//! - **Pictures in the story**: the pictures scripts place there, on unless
25//!   switched off; and the **pictures folder** a script's files may be read
26//!   from, with the folders inside it, none unless chosen
27//!   (`hydra_ui::picture`).
28//! - **The web page**: off unless switched on, and the port it is served
29//!   on, [`WEB_PORT`] unless chosen. On, any device on the home network
30//!   that has the pairing can play; the hub shows the address.
31//! - **Starting fresh**: how many minutes after the last character stops,
32//!   with no window open, Hydra starts itself again in the background,
33//!   giving back all the memory its play held; [`FRESH_AFTER`] unless
34//!   chosen, 0 never.
35//! - **Updates**: Hydra looks for a newer release as it starts and every
36//!   few hours, on unless switched off, and the hub says when one is out.
37//!   With *Update on its own* on, off unless switched on, it fetches one as
38//!   soon as it is out, to run from Hydra's next start.
39//! - **Graphics**: on Windows, how the windows are drawn: Vulkan where a
40//!   graphics card has it and DirectX 12 otherwise, unless one is chosen
41//!   ([`Graphics`]). Takes from Hydra's next start; `WGPU_BACKEND` set in
42//!   the environment still wins.
43//!
44//! The settings menu shows them as Hydra's own pages, *General* and *Hub*
45//! ([`Own::pages`]), drawn as any other; no behavior is behind them, so the
46//! window makes a change itself ([`Own::change`]).
47

128/// How the windows are drawn on Windows: the graphics API wgpu opens.
129#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
130pub enum Graphics {
131    /// Vulkan where a graphics card has it, DirectX 12 otherwise.
132    #[default]
133    Auto,
134    /// Vulkan, whatever the probe would find.
135    Vulkan,
136    /// DirectX 12. Slower to open on some computers, with overlays in its
137    /// way, but clear of the Vulkan driver's own presenting.
138    DirectX12,
139}
140
141impl Graphics {
142    /// Each choice, as the file writes it and as a player reads it.
143    pub const CHOICES: [(Graphics, &'static str, &'static str); 3] = [
144        (
145            Graphics::Auto,
146            "auto",
147            "Vulkan where a card has it, DirectX 12 otherwise",
148        ),
149        (Graphics::Vulkan, "vulkan", "Vulkan"),
150        (Graphics::DirectX12, "dx12", "DirectX 12"),
151    ];
152
153    /// As the file writes it.
154    #[must_use]
155    pub const fn key(self) -> &'static str {
156        match self {
157            Graphics::Auto => "auto",
158            Graphics::Vulkan => "vulkan",
159            Graphics::DirectX12 => "dx12",
160        }
161    }
162
163    /// The choice `key` names, as the file writes it; `None` for a word
164    /// that is none of them.
165    #[must_use]
166    pub fn of(key: &str) -> Option<Self> {
167        Self::CHOICES
168            .iter()
169            .find(|(_, name, _)| *name == key)
170            .map(|(choice, _, _)| *choice)
171    }
172}

316    /// How large everything is drawn: 1 unless chosen, and held between
317    /// [`ZOOM_LEAST`] and [`ZOOM_MOST`] whatever the file says -- a hand
318    /// edit's 0 or `nan` would otherwise be what every window is drawn at.
319    #[must_use]
320    pub fn zoom(&self) -> f32 {
321        self.file
322            .zoom
323            .filter(|zoom| zoom.is_finite())
324            .map_or(1.0, |zoom| zoom.clamp(ZOOM_LEAST, ZOOM_MOST))
325    }

327    /// Draw everything at `zoom`, held between [`ZOOM_LEAST`] and
328    /// [`ZOOM_MOST`] and rounded to a hundredth, and keep it. What it is.
329    ///
330    /// # Errors
331    ///
332    /// As [`Self::set_record_sessions`].
333    pub fn set_zoom(&mut self, zoom: f32) -> Result<f32, String> {
334        let zoom = (zoom.clamp(ZOOM_LEAST, ZOOM_MOST) * 100.0).round() / 100.0;
335        let mut file = self.file.clone();
336        file.zoom = ((zoom - 1.0).abs() > f32::EPSILON).then_some(zoom);
337        self.keep(file)?;
338        Ok(zoom)
339    }

370    /// Hydra's own pages: *General* -- what it records, the web page, the
371    /// windows' zoom, sound, merchants, pictures, and Hydra itself -- and
372    /// *Hub*, the theme it wears, `themes` the ones to choose from and

376        let mut pages = vec![
377            self.general_page(),

379            self.file.banner.page(FILE),
380        ];
381        // The lookout's, kept in a file of its own beside this one.
382        if let Some(data) = &self.data {
383            pages.push(crate::lookout::page(data, &hydra_lookout::counts_all()));
384        }
385        pages
386    }

572    /// The graphics choice set to `named`, or back to its own. What was done.
573    fn set_graphics_named(&mut self, named: Option<&str>) -> Result<String, String> {
574        let choice =
575            match named {
576                None => None,
577                Some(typed) => Some(Graphics::of(typed).ok_or_else(|| {
578                    format!("The graphics are auto, vulkan or dx12, not {typed}.")
579                })?),
580            };
581        let mut file = self.file.clone();
582        file.graphics = choice
583            .filter(|choice| *choice != Graphics::Auto)
584            .map(|choice| choice.key().to_owned());
585        self.keep(file)?;
586        Ok(match self.graphics() {
587            Graphics::Auto => {
588                "The windows open on Vulkan where a card has it, DirectX 12 otherwise, from \
589                 Hydra's next start."
590                    .to_owned()
591            }
592            Graphics::Vulkan => "The windows open on Vulkan, from Hydra's next start.".to_owned(),
593            Graphics::DirectX12 => {
594                "The windows open on DirectX 12, from Hydra's next start.".to_owned()
595            }
596        })
597    }

819    /// The minutes before starting fresh as the page writes them, or back
820    /// to their own. What was done.
821    fn set_fresh_named(&mut self, named: Option<&str>) -> Result<String, String> {
822        let seconds = match named {
823            None => None,
824            Some(typed) => Some(
825                typed
826                    .parse::<u32>()
827                    .ok()
828                    .filter(|seconds| *seconds <= FRESH_MOST)
829                    .ok_or_else(|| {
830                        format!("Starting fresh is after 0 to {FRESH_MOST} seconds, not {typed}.")
831                    })?,
832            ),
833        };
834        let mut file = self.file.clone();
835        file.fresh_seconds = seconds.filter(|seconds| *seconds != FRESH_AFTER);
836        self.keep(file)?;
837        let after = match self.fresh_seconds() {
838            0 => return Ok("Hydra never starts fresh on its own.".to_owned()),
839            60 => "a minute".to_owned(),
840            n if n % 60 == 0 => format!("{} minutes", n / 60),
841            1 => "a second".to_owned(),
842            n => format!("{n} seconds"),
843        };
844        Ok(format!(
845            "Hydra starts fresh {after} after the last character stops."
846        ))
847    }

881    /// Set `key` on Hydra's own pages to `to`, as the menu writes it, or
882    /// back to its default (`None`), and keep it. What was done.
883    ///
884    /// # Errors
885    ///
886    /// Why nothing was changed.
887    pub fn change(&mut self, key: &str, to: Option<&str>) -> Result<String, String> {
888        if let Some(done) = self.banner_change(key, to) {
889            return done;
890        }
891        if let Some(lookout) = key.strip_prefix(crate::lookout::KEY) {
892            let data = self.data.as_deref().ok_or_else(|| {
893                "There is no data folder to keep the lookout's choices in.".to_owned()
894            })?;
895            return crate::lookout::change(data, lookout, to);
896        }
897        let named = to.map(str::trim).filter(|name| !name.is_empty());
898        if let Some(done) = self.drawn_change(key, named) {
899            return done;
900        }
901        match key {
902            "theme" => {
903                let mut file = self.file.clone();
904                file.theme = named.map(str::to_owned);
905                self.keep(file)?;
906                return Ok(format!("The theme is {}.", self.theme()));
907            }
908            "light_theme" => {
909                let mut file = self.file.clone();
910                file.light_theme = named.map(str::to_owned);
911                self.keep(file)?;
912                return Ok(format!("The light theme is {}.", self.light_theme()));
913            }
914            "web_port" => {
915                let port = match named {
916                    None => None,
917                    Some(typed) => Some(
918                        typed
919                            .parse::<u16>()
920                            .ok()
921                            .filter(|port| *port >= 1024)
922                            .ok_or_else(|| {
923                                format!("The web page's port is from 1024 to 65535, not {typed}.")
924                            })?,
925                    ),
926                };
927                let mut file = self.file.clone();
928                file.web_port = port.filter(|port| *port != WEB_PORT);
929                self.keep(file)?;
930                return Ok(format!("The web page's port is {}.", self.web_port()));
931            }
932            _ => {}
933        }
934        let on = match to {
935            None => None,
936            Some("on") => Some(true),
937            Some("off") => Some(false),
938            Some(other) => return Err(format!("{key} is on or off, not {other}.")),
939        };
940        if let Some(done) = self.update_change(key, on) {
941            return done;
942        }
943        if let Some(done) = self.merchant_change(key, on) {
944            return done;
945        }
946        match key {
947            "follow_computer" => {
948                let mut file = self.file.clone();
949                file.follow_computer = on.filter(|on| *on);
950                self.keep(file)?;
951                Ok(format!(
952                    "The theme {} the computer's dark or light mode.",
953                    if self.follow_computer() {
954                        "follows"
955                    } else {
956                        "does not follow"
957                    }
958                ))
959            }
960            "web_page" => {
961                let mut file = self.file.clone();
962                file.web_page = on.filter(|on| *on);
963                self.keep(file)?;
964                Ok(if self.web().is_some() {
965                    "The web page is served: the hub shows its address.".to_owned()
966                } else {
967                    "The web page is not served.".to_owned()
968                })
969            }
970            "player_log" => {
971                self.set_player_log(on.unwrap_or(true))?;
972                Ok(format!(
973                    "The player log is {}kept, from each character's next login.",
974                    if self.player_log() { "" } else { "not " }
975                ))
976            }
977            "record_sessions" => {
978                self.set_record_sessions(on.unwrap_or(RECORD_BY_DEFAULT))?;
979                Ok(format!(
980                    "Sessions are {}recorded, from each character's next login.",
981                    if self.record_sessions() { "" } else { "not " }
982                ))
983            }
984            other => Err(format!("Hydra's settings have no setting {other}.")),
985        }
986    }

988    /// `file` written over the window's own, and in use.
989    fn keep(&mut self, file: File) -> Result<(), String> {
990        if let Some(why) = &self.problem {
991            return Err(format!("{FILE} was not changed: {why}"));
992        }
993        if let Some(data) = &self.data {
994            let text = toml::to_string(&file).map_err(|why| why.to_string())?;
995            crate::store::whole(data, &data.join(FILE), text.into_bytes());
996        }
997        self.file = file;
998        Ok(())
999    }
1000}

1033/// The file in `data`, and why it does not read, if it does not.
1034fn read(data: &Path) -> (File, Option<String>) {
1035    match std::fs::read_to_string(data.join(FILE)) {
1036        Ok(text) => match toml::from_str::<File>(&text) {
1037            // A wait kept in minutes, before it was in seconds, kept as
1038            // seconds from now.
1039            Ok(mut file) => {
1040                let minutes = file.fresh_after.take();
1041                file.fresh_seconds = file
1042                    .fresh_seconds
1043                    .or_else(|| minutes.map(|minutes| minutes.saturating_mul(60)));
1044                (file, None)
1045            }
1046            Err(why) => (File::default(), Some(format!("it does not read: {why}"))),
1047        },
1048        Err(why) if why.kind() == std::io::ErrorKind::NotFound => (File::default(), None),
1049        Err(why) => (File::default(), Some(format!("it cannot be read: {why}"))),
1050    }
1051}

192/// The winit name `key` is written as, checked: a key this build can see.
193pub(super) fn known(key: &str) -> Result<String, String> {
194    if let Some((winit, _)) = NUMPAD
195        .iter()
196        .find(|(winit, fork)| winit.eq_ignore_ascii_case(key) || fork.eq_ignore_ascii_case(key))
197    {
198        return Ok((*winit).to_owned());
199    }
200    if let Some(name) = KNOWN.iter().find(|name| name.eq_ignore_ascii_case(key)) {
201        return Ok((*name).to_owned());
202    }
203    if let Some(name) = CAPTURED.iter().find(|name| name.eq_ignore_ascii_case(key)) {
204        return Ok((*name).to_owned());
205    }
206    if key.eq_ignore_ascii_case("NumLock") || key.eq_ignore_ascii_case("Clear") {
207        return Err(format!(
208            "{key} switches the numpad between typing and its keys (NumLock; Clear on a Mac), and is bound to nothing."
209        ));
210    }
211    Err(format!(
212        "{key} is not a key winit names; a key is named by its winit code, such as Numpad8, F13 or KeyA."
213    ))
214}

216/// Whether the key of this name types something when pressed. A bound
217/// key that does is taken with what it typed: egui sends a letter's text
218/// beside its press unless Ctrl or Cmd is held, so Alt+C did its macro and
219/// typed a `c` as well.
220#[must_use]
221pub fn types(key: &str) -> bool {
222    TYPING.contains(&key)
223}

1//! Hydra's default keys: set 0 as a player first finds it. They live here,
2//! in the code, and the keybinds file holds only what the player added or
3//! changed, so a default added in a later Hydra reaches every player who
4//! never changed that key.
5//!
6//! The numpad walks as Wrayth's stock key set has it; Shift with it peers
7//! that way. Alt with a digit chooses that macro set, as Wrayth's does, the
8//! one action with a key of Hydra's. The sending actions have Wrayth's
9//! keys: `NumpadEnter` sends or repeats, Ctrl with an Enter repeats the
10//! last command and Alt the one before it, Up and Down walk what was typed,
11//! and Escape clears the input. The window in use scrolls with the page
12//! keys: a page, with Shift a line, with Ctrl to the oldest or newest line.
13//! Ctrl+Tab turns the tabs of the window in use, with Shift back; choosing
14//! the window has no key, a click chooses it. Ctrl+F finds, F3 the one
15//! found before and Shift+F3 the one after. Tab targets the next creature
16//! and Shift+Tab the one before.
17

20use super::{ALT, CTRL, Chord, SHIFT};

23/// Each numpad key that walks, and where.
24const WALKS: [(&str, &str); 11] = [
25    ("Numpad8", "north"),
26    ("Numpad2", "south"),
27    ("Numpad6", "east"),
28    ("Numpad4", "west"),
29    ("Numpad9", "northeast"),
30    ("Numpad7", "northwest"),
31    ("Numpad3", "southeast"),
32    ("Numpad1", "southwest"),
33    ("Numpad5", "out"),
34    ("Numpad0", "down"),
35    ("NumpadDecimal", "up"),
36];

38/// The numpad's four marks, as Wrayth binds them.
39const MARKS: [(&str, &str); 4] = [
40    ("NumpadAdd", "look"),
41    ("NumpadSubtract", "info"),
42    ("NumpadMultiply", "exp"),
43    ("NumpadDivide", "health"),
44];

46/// The sending actions' keys, and the modifiers held.
47const SENDING: [(&str, u8, Action); 8] = [
48    ("NumpadEnter", 0, Action::SendOrRepeat),
49    ("Enter", CTRL, Action::RepeatLast),
50    ("NumpadEnter", CTRL, Action::RepeatLast),
51    ("Enter", ALT, Action::RepeatSecondLast),
52    ("NumpadEnter", ALT, Action::RepeatSecondLast),
53    ("ArrowUp", 0, Action::HistoryBack),
54    ("ArrowDown", 0, Action::HistoryForward),
55    ("Escape", 0, Action::ClearInput),
56];
57
58/// The scrolling actions' keys, and the modifiers held.
59const SCROLLING: [(&str, u8, Action); 8] = [
60    ("PageUp", 0, Action::ScrollPageUp),
61    ("PageDown", 0, Action::ScrollPageDown),
62    ("PageUp", SHIFT, Action::ScrollLineUp),
63    ("PageDown", SHIFT, Action::ScrollLineDown),
64    ("Home", CTRL, Action::ScrollTop),
65    ("PageUp", CTRL, Action::ScrollTop),
66    ("End", CTRL, Action::ScrollBottom),
67    ("PageDown", CTRL, Action::ScrollBottom),
68];
69
70/// The tabs' keys, Find's and the targets', and the modifiers held.
71const TABS: [(&str, u8, Action); 7] = [
72    ("Tab", CTRL, Action::NextTab),
73    ("Tab", CTRL | SHIFT, Action::PreviousTab),
74    ("KeyF", CTRL, Action::Find),
75    ("F3", 0, Action::FindNext),
76    ("F3", SHIFT, Action::FindPrevious),
77    ("Tab", 0, Action::TargetNext),
78    ("Tab", SHIFT, Action::TargetPrevious),
79];

81/// Every key Hydra binds, and what each does.
82pub(super) fn defaults() -> BTreeMap<Chord, Macro> {
83    let chord = |key: &str, held: u8| Chord {
84        key: key.to_owned(),
85        held,
86    };
87    let mut keys = BTreeMap::new();
88    for (key, way) in WALKS {
89        keys.insert(chord(key, 0), Macro::Send(way.to_owned()));
90        keys.insert(chord(key, SHIFT), Macro::Send(format!("peer {way}")));
91    }
92    for (key, line) in MARKS {
93        keys.insert(chord(key, 0), Macro::Send(line.to_owned()));
94    }
95    for (key, held, action) in SENDING.into_iter().chain(SCROLLING).chain(TABS) {
96        keys.insert(chord(key, held), Macro::Act(action));
97    }
98    for set in 0..super::SETS {
99        keys.insert(
100            chord(&format!("Digit{set}"), ALT),
101            Macro::Act(Action::Set(set)),
102        );
103    }
104    keys
105}

107#[cfg(test)]
108mod tests {
109    use super::*;
110
111    /// Every default is a key this build can see, one that does not type,
112    /// and a macro that can be done as written: a default the file would
113    /// refuse would be a default nobody gets.
114    #[test]
115    fn every_default_is_one_the_file_would_take() {
116        let defaults = defaults();
117        assert_eq!(
118            defaults.len(),
119            59,
120            "eleven walks, eleven peers, four marks, eight sending, eight scrolling, two tabs, three finding, two targets, ten sets"
121        );
122        for (chord, made) in &defaults {
123            let written = chord.written();
124            assert_eq!(Chord::parse(&written).as_ref(), Ok(chord), "{written}");
125            assert_eq!(chord.refused(), None, "{written}");
126            assert_eq!(made.check(), Ok(()), "{written}");
127        }
128        let peer = Chord::parse("Shift+Numpad7").expect("a key");
129        assert_eq!(
130            defaults.get(&peer),
131            Some(&Macro::Send("peer northwest".to_owned()))
132        );
133        assert_eq!(
134            defaults.get(&Chord::parse("Alt+Digit3").expect("a key")),
135            Some(&Macro::Act(Action::Set(3)))
136        );
137    }
138}

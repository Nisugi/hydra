20/// The page, as `dir`'s file has the lookout's choices, with how much it
21/// has learned of `games` (each `(game, learned, learning, bytes)`).
22#[must_use]
23#[expect(
24    clippy::too_many_lines,
25    reason = "the page's rows, one after another, each with its words"
26)]
27pub fn page(dir: &Path, games: &[(String, hydra_lookout::Counts)]) -> Page {
28    let (settings, problem) = match Settings::read(dir) {
29        Ok(settings) => (settings, None),
30        Err(why) => (
31            Settings::default(),
32            Some(format!("Nothing here is changed while {why}")),
33        ),
34    };
35    let defaults = Settings::default();
36    let whole = |key: &str, label: &str, help: &str, value: u32, default: u32, (min, max)| Row {
37        key: format!("{KEY}{key}"),
38        label: label.to_owned(),
39        help: help.to_owned(),
40        kind: RowKind::Whole { min, max },
41        value: Value::Text(value.to_string()),
42        here: value != default,
43        from: None,
44        needs: None,
45    };
46    let toggle = |key: &str, label: &str, help: &str, on: bool, default: bool| Row {
47        key: format!("{KEY}{key}"),
48        label: label.to_owned(),
49        help: help.to_owned(),
50        kind: RowKind::Toggle,
51        value: Value::On(on),
52        here: on != default,
53        from: None,
54        needs: None,
55    };
56    let mut rows = vec![
57        toggle(
58            "paused",
59            "Paused",
60            "On, nothing is read, learned or shown, for every character.",
61            settings.paused,
62            defaults.paused,
63        ),
64        toggle(
65            "mark",
66            "Mark in the story",
67            "Each new line marked where it came in the story as well, in the lookout's colour: \
68             best left off while the lookout is still learning most lines. A line whose stat is \
69             absurd is marked always.",
70            settings.mark,
71            defaults.mark,
72        ),
73        Row {
74            key: format!("{KEY}off"),
75            label: "Characters not read".to_owned(),
76            help: "Their lines are not read, learned or shown: names, apart by commas.".to_owned(),
77            kind: RowKind::Words,
78            value: Value::List(settings.off.clone()),
79            here: !settings.off.is_empty(),
80            from: None,
81            needs: None,
82        },
83        Row::heading("Learning"),
84        whole(
85            "occasions",
86            "Occasions to learn a line",
87            "How many times a line, or a thing in one, comes before it is learned and no \
88             longer shown.",
89            u32::from(settings.occasions),
90            u32::from(defaults.occasions),
91            (1, 50),
92        ),
93        Row {
94            key: format!("{KEY}window"),
95            label: "Same moment, seconds".to_owned(),
96            help: "Seconds in which one line heard by several of your characters counts as one \
97                   occasion: a group must not learn a line the first time it comes."
98                .to_owned(),
99            kind: RowKind::Number {
100                min: 0.0,
101                max: 30.0,
102            },
103            value: Value::Text(settings.window.to_string()),
104            here: (settings.window - defaults.window).abs() > f64::EPSILON,
105            from: None,
106            needs: None,
107        },
108        whole(
109            "forget_days",
110            "Forget one still learning after, days",
111            "Days: a line not yet learned that has not come in this long is dropped, which keeps \
112             the file small. 0, never.",
113            settings.forget_days,
114            defaults.forget_days,
115            (0, 3650),
116        ),
117        whole(
118            "quiet_secs",
119            "Quiet after login, seconds",
120            "Seconds after a character is in the game in which its lines are learned and not \
121             shown: the login's burst of lines said once.",
122            settings.quiet_secs,
123            defaults.quiet_secs,
124            (0, 3600),
125        ),
126        Row::heading("Combat"),
127        whole(
128            "stat_floor",
129            "Stat floor",
130            "A resolution stat -- AS, DS, CS, TD, UAF, UDF, AvD, CvA -- at or over this is shown \
131             always, marked, however often its line has come: one no character reaches is a \
132             script check's. A floor, not a most: a capped, buffed character can reach 1300 AS.",
133            u32::try_from(settings.stat_floor).unwrap_or(u32::MAX),
134            u32::try_from(defaults.stat_floor).unwrap_or(u32::MAX),
135            (100, 1_000_000),
136        ),
137        Row::heading("Learned"),
138    ];
139    if games.is_empty() {
140        rows.push(note(
141            "Nothing yet: a character's lines are learned once it is in the game.",
142        ));
143    }
144    for (game, counts) in games {
145        rows.push(note(&format!(
146            "{game}: {} learned, {} still learning, {} on disk.",
147            counts.learned,
148            counts.learning,
149            size(counts.bytes)
150        )));
151    }
152    Page {
153        id: PAGE.to_owned(),
154        title: "Lookout".to_owned(),
155        file: hydra_lookout::FILE.to_owned(),
156        takes: "at once".to_owned(),
157        problem,
158        rows,
159    }
160}

162/// A line of words on the page, with nothing to change: a heading of its
163/// own, the page having no plainer row.
164fn note(said: &str) -> Row {
165    Row::heading(said)
166}

168/// `bytes`, as a person reads a file's size.
169fn size(bytes: u64) -> String {
170    match bytes {
171        0..1024 => format!("{bytes} bytes"),
172        1024..1_048_576 => format!("{} KB", bytes / 1024),
173        _ => {
174            #[expect(clippy::cast_precision_loss, reason = "a size said to a tenth")]
175            let megabytes = bytes as f64 / 1_048_576.0;
176            format!("{megabytes:.1} MB")
177        }
178    }
179}

181/// The row `key` (after [`KEY`]) changed to `to`, `None` its default:
182/// written to `dir`'s file, and taken by every lookout open. What was done.
183///
184/// # Errors
185///
186/// The value is not one the row takes, or the file does not read or could
187/// not be written.
188pub fn change(dir: &Path, key: &str, to: Option<&str>) -> Result<String, String> {
189    let mut settings = Settings::read(dir)?;
190    let defaults = Settings::default();
191    let typed = to.map(str::trim).filter(|typed| !typed.is_empty());
192    let on = |typed: Option<&str>, default: bool| match typed {
193        None => Ok(default),
194        Some("on") => Ok(true),
195        Some("off") => Ok(false),
196        Some(other) => Err(format!("{key} is on or off, not {other}.")),
197    };
198    let number = |typed: Option<&str>| -> Result<Option<f64>, String> {
199        typed
200            .map(|typed| {
201                typed
202                    .parse::<f64>()
203                    .ok()
204                    .filter(|value| value.is_finite() && *value >= 0.0)
205                    .ok_or_else(|| format!("{key} is a number, not {typed}."))
206            })
207            .transpose()
208    };
209    let said = match key {
210        "paused" => {
211            settings.paused = on(typed, defaults.paused)?;
212            if settings.paused {
213                "The lookout is paused.".to_owned()
214            } else {
215                "The lookout is watching.".to_owned()
216            }
217        }
218        "mark" => {
219            settings.mark = on(typed, defaults.mark)?;
220            if settings.mark {
221                "Each new line is marked in the story too.".to_owned()
222            } else {
223                "New lines are marked in the story only when their stat is absurd.".to_owned()
224            }
225        }
226        "off" => {
227            // The menu writes a list as the file would, `["Zoleta"]`; a
228            // command, as typed.
229            settings.off = typed
230                .unwrap_or_default()
231                .trim_start_matches('[')
232                .trim_end_matches(']')
233                .split(',')
234                .map(|name| name.trim().trim_matches(['"', '\'']).trim())
235                .filter(|name| !name.is_empty())
236                .map(str::to_owned)
237                .collect();
238            if settings.off.is_empty() {
239                "Every character's lines are read.".to_owned()
240            } else {
241                format!("Not read: {}.", settings.off.join(", "))
242            }
243        }
244        "occasions" => {
245            let value = number(typed)?.map_or(f64::from(defaults.occasions), f64::round);
246            settings.occasions = (1.0..=50.0)
247                .contains(&value)
248                .then_some(value)
249                .and_then(|value| format!("{value}").parse().ok())
250                .ok_or_else(|| format!("Occasions are from 1 to 50, not {value}."))?;
251            format!("A line is learned after {} occasions.", settings.occasions)
252        }
253        "window" => {
254            settings.window = number(typed)?.unwrap_or(defaults.window).min(30.0);
255            format!("One moment is {} seconds.", settings.window)
256        }
257        "forget_days" | "quiet_secs" | "stat_floor" => {
258            let whole = number(typed)?.map(f64::round);
259            let value = |default: u32| -> Result<u32, String> {
260                whole.map_or(Ok(default), |whole| {
261                    format!("{whole}")
262                        .parse::<u32>()
263                        .map_err(|_| format!("{key} is a whole number, not {whole}."))
264                })
265            };
266            match key {
267                "forget_days" => {
268                    settings.forget_days = value(defaults.forget_days)?;
269                    format!(
270                        "A line still learning is forgotten after {} days.",
271                        settings.forget_days
272                    )
273                }
274                "quiet_secs" => {
275                    settings.quiet_secs = value(defaults.quiet_secs)?;
276                    format!("Quiet for {} seconds after a login.", settings.quiet_secs)
277                }
278                _ => {
279                    let default = u32::try_from(defaults.stat_floor).unwrap_or(u32::MAX);
280                    settings.stat_floor = i64::from(value(default)?.max(100));
281                    format!("A stat of {} or more is shown always.", settings.stat_floor)
282                }
283            }
284        }
285        other => return Err(format!("The lookout has no setting {other}.")),
286    };
287    settings.write(dir)?;
288    hydra_lookout::reload(dir);
289    Ok(said)
290}

292/// The trigger the lookout comes with, by its name in the triggers file.
293pub const TRIGGER: &str = "Lookout";
294
295/// The sound it plays, in the sounds folder.
296pub const SOUND: &str = "lookout.wav";

298/// The key of the page's row that switches the trigger on or off.
299pub const TRIGGER_KEY: &str = "lookout.trigger";
300
301/// The trigger the lookout comes with: a sound and a banner for each line
302/// put in its stream, ten seconds apart at most. Switched off until the
303/// player switches it on: while the lookout is still learning, most lines
304/// are new.
305#[must_use]
306pub fn trigger() -> hydra_ui::highlight::Rule {
307    hydra_ui::highlight::Rule {
308        off: true,
309        category: "Lookout".to_owned(),
310        stream: Some(hydra_lookout::STREAM.to_owned()),
311        sound: Some(SOUND.to_owned()),
312        alert: Some(hydra_ui::highlight::Say::Line(true)),
313        cooldown: Some(10.0),
314        ..hydra_ui::highlight::Rule::default()
315    }
316}

318/// `pages`, Hydra's own, with the Lookout page's row that switches its
319/// trigger, as `highlights` have it: on, off, or gone from the triggers.
320pub fn with_trigger_row(pages: &mut [Page], highlights: &crate::highlights::Highlights) {
321    let Some(page) = pages.iter_mut().find(|page| page.id == PAGE) else {
322        return;
323    };
324    let file = highlights.file();
325    let rule = file.rules.get(TRIGGER);
326    let on = rule.is_some_and(|rule| !rule.off);
327    let row = Row {
328        key: TRIGGER_KEY.to_owned(),
329        label: "Sound and banner".to_owned(),
330        help: format!(
331            "The {TRIGGER} trigger: a sound and a banner for each new line, at most one every \
332             ten seconds. Off unless switched on: while the lookout is still learning, most \
333             lines are new. Its sound and words are changed in Editors > Triggers{}.",
334            if rule.is_none() {
335                "; it is not there now, and switching this on puts it back"
336            } else {
337                ""
338            }
339        ),
340        kind: RowKind::Toggle,
341        value: Value::On(on),
342        here: on,
343        from: None,
344        needs: None,
345    };
346    // Under Paused, the first row.
347    let at = page.rows.len().min(1);
348    page.rows.insert(at, row);
349}

351/// Switch the lookout's trigger on or off in `highlights` and its file in
352/// `dir`; switched on when it is gone, it is put back. What was done.
353///
354/// # Errors
355///
356/// The triggers could not be changed.
357pub fn set_trigger(
358    dir: &Path,
359    highlights: &crate::highlights::Highlights,
360    on: bool,
361) -> Result<String, String> {
362    highlights.change(dir, move |file| {
363        let rule = file.rules.entry(TRIGGER.to_owned()).or_insert_with(trigger);
364        rule.off = !on;
365        Ok(if on {
366            "The lookout's sound and banner are on.".to_owned()
367        } else {
368            "The lookout's sound and banner are off.".to_owned()
369        })
370    })
371}

373/// Add the lookout's trigger to `highlights` and its file in `dir`, and its
374/// sound to the sounds folder, once: one the player has since removed or
375/// changed is left as it is. What was done, when something was.
376#[must_use]
377pub fn add_trigger(dir: &Path, highlights: &crate::highlights::Highlights) -> Option<String> {
378    let mut settings = Settings::read(dir).ok()?;
379    if settings.trigger_added {
380        return None;
381    }
382    let sound = dir.join("sounds").join(SOUND);
383    if !sound.exists() {
384        if let Some(folder) = sound.parent() {
385            let _ = std::fs::create_dir_all(folder);
386        }
387        if let Err(why) = std::fs::write(&sound, chime()) {
388            return Some(format!(
389                "The lookout's sound was not written to {}: {why}",
390                sound.display()
391            ));
392        }
393    }
394    let added = highlights.change(dir, |file| {
395        file.rules.entry(TRIGGER.to_owned()).or_insert_with(trigger);
396        Ok(String::new())
397    });
398    if let Err(why) = added {
399        return Some(format!("The lookout's trigger was not added: {why}"));
400    }
401    settings.trigger_added = true;
402    match settings.write(dir) {
403        Ok(()) => Some(format!(
404            "The trigger {TRIGGER} is added, switched off: Settings > Lookout switches it on."
405        )),
406        Err(why) => Some(format!("The lookout's choices were not written: {why}")),
407    }
408}

410/// The lookout's sound: two short rising notes, as a WAV file's bytes.
411#[must_use]
412pub fn chime() -> Vec<u8> {
413    const RATE: u32 = 22_050;
414    let mut samples: Vec<i16> = Vec::new();
415    // Each note's pitch and length in samples: a tenth of a second and a
416    // bit, then a fifth.
417    for (pitch, count) in [(880.0_f64, 2_646_u32), (1320.0, 4_410)] {
418        for at in 0..count {
419            let at = f64::from(at);
420            let t = at / f64::from(RATE);
421            // Up quickly, down slowly: a bell's shape, with no click.
422            let rise = (at / 200.0).min(1.0);
423            let fall = 1.0 - at / f64::from(count);
424            let wave = (std::f64::consts::TAU * pitch * t).sin();
425            #[expect(
426                clippy::cast_possible_truncation,
427                reason = "a sample held to a third of the loudest a WAV takes"
428            )]
429            samples.push((wave * rise * fall * f64::from(i16::MAX) / 3.0) as i16);
430        }
431    }
432    let data = u32::try_from(samples.len() * 2).unwrap_or(0);
433    let mut bytes = Vec::with_capacity(44 + samples.len() * 2);
434    bytes.extend_from_slice(b"RIFF");
435    bytes.extend_from_slice(&(36 + data).to_le_bytes());
436    bytes.extend_from_slice(b"WAVEfmt ");
437    bytes.extend_from_slice(&16u32.to_le_bytes());
438    bytes.extend_from_slice(&1u16.to_le_bytes()); // PCM
439    bytes.extend_from_slice(&1u16.to_le_bytes()); // one channel
440    bytes.extend_from_slice(&RATE.to_le_bytes());
441    bytes.extend_from_slice(&(RATE * 2).to_le_bytes());
442    bytes.extend_from_slice(&2u16.to_le_bytes());
443    bytes.extend_from_slice(&16u16.to_le_bytes());
444    bytes.extend_from_slice(b"data");
445    bytes.extend_from_slice(&data.to_le_bytes());
446    for sample in samples {
447        bytes.extend_from_slice(&sample.to_le_bytes());
448    }
449    bytes
450}

456    #[test]
457    fn the_trigger_is_added_once() {
458        let dir = std::env::temp_dir().join(format!(
459            "hydra-board-lookout-trigger-{}",
460            std::process::id()
461        ));
462        let _ = std::fs::remove_dir_all(&dir);
463        std::fs::create_dir_all(&dir).unwrap();
464        let highlights = crate::highlights::Highlights::default();
465        assert!(add_trigger(&dir, &highlights).is_some());
466        assert_eq!(highlights.file().rules.get(TRIGGER), Some(&trigger()));
467        assert!(dir.join("sounds").join(SOUND).is_file());
468        assert!(hydra_ui::highlight::Matcher::check_rule(TRIGGER, &trigger()).is_ok());
469        // Removed by the player: not added again.
470        let _ = highlights.change(&dir, |file| {
471            file.rules.remove(TRIGGER);
472            Ok(String::new())
473        });
474        assert!(add_trigger(&dir, &highlights).is_none());
475        assert!(!highlights.file().rules.contains_key(TRIGGER));
476        // Switched on from the page: put back, on.
477        assert!(set_trigger(&dir, &highlights, true).is_ok());
478        assert_eq!(
479            highlights.file().rules.get(TRIGGER).map(|rule| rule.off),
480            Some(false)
481        );
482        let mut pages = vec![page(&dir, &[])];
483        with_trigger_row(&mut pages, &highlights);
484        let row = pages[0]
485            .rows
486            .iter()
487            .find(|row| row.key == TRIGGER_KEY)
488            .unwrap();
489        assert_eq!(row.value, Value::On(true));
490        assert!(set_trigger(&dir, &highlights, false).is_ok());
491        assert_eq!(
492            highlights.file().rules.get(TRIGGER).map(|rule| rule.off),
493            Some(true)
494        );
495        let _ = std::fs::remove_dir_all(&dir);
496    }

499    fn the_chime_is_a_wav() {
500        let chime = chime();
501        assert_eq!(&chime[..4], b"RIFF");
502        assert_eq!(&chime[8..16], b"WAVEfmt ");
503        assert_eq!(chime.len(), 44 + (2646 + 4410) * 2);
504    }

506    #[test]
507    fn a_change_is_written_and_read_by_the_page() {
508        let dir = std::env::temp_dir().join(format!("hydra-board-lookout-{}", std::process::id()));
509        let _ = std::fs::remove_dir_all(&dir);
510        std::fs::create_dir_all(&dir).unwrap();
511        assert!(change(&dir, "mark", Some("on")).is_ok());
512        assert!(change(&dir, "occasions", Some("7")).is_ok());
513        assert!(change(&dir, "off", Some(r#"["Zoleta", "Sugiin"]"#)).is_ok());
514        assert!(change(&dir, "occasions", Some("0")).is_err());
515        assert!(change(&dir, "nothing", Some("1")).is_err());
516        let settings = Settings::read(&dir).unwrap();
517        assert!(settings.mark);
518        assert_eq!(settings.occasions, 7);
519        assert_eq!(settings.off, ["Zoleta", "Sugiin"]);
520        let page = page(&dir, &[]);
521        let mark = page
522            .rows
523            .iter()
524            .find(|row| row.key == "lookout.mark")
525            .unwrap();
526        assert_eq!(mark.value, Value::On(true));
527        assert!(mark.here);
528        // Back to its default.
529        assert!(change(&dir, "occasions", None).is_ok());
530        assert_eq!(Settings::read(&dir).unwrap().occasions, 5);
531        let _ = std::fs::remove_dir_all(&dir);
532    }

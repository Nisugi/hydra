1//! Banners: a highlight's alert, and Hydra's own, shown over the play
2//! window and the web page while they last. Where they float, how wide,
3//! how solid, their colours and how long they stay are Hydra's to say for
4//! every character, on Settings' *Banners* page: the `[banner]` table of
5//! Hydra's own settings file.
6
7use hydra_ui::settings::{Page, Row, RowKind, Value};
8use hydra_ui::theme::{Rgb, hex, parse_hex};
9use serde::{Deserialize, Serialize};
10
11/// The page's id.
12pub const PAGE: &str = "hydra_banners";
13
14/// What starts each of the page's keys, so Hydra's settings send them here.
15pub const KEY: &str = "banner_";
16
17/// Across the window: where a banner sits from left to right.
18#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
19pub enum Across {
20    /// At the left edge.
21    Left,
22    /// In the middle.
23    #[default]
24    Centre,
25    /// At the right edge.
26    Right,
27}
28
29/// Down the window: where a banner sits from top to bottom.
30#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
31pub enum Down {
32    /// Under the top bar.
33    #[default]
34    Top,
35    /// In the middle.
36    Middle,
37    /// Over the command line.
38    Bottom,
39}
40
41/// How a banner is shown.
42#[derive(Clone, Copy, Debug, PartialEq)]
43pub struct Look {
44    /// Where across.
45    pub across: Across,
46    /// Where down.
47    pub down: Down,
48    /// Its width, a share of the window's, 0.1 to 1.
49    pub width: f32,
50    /// How solid it is, 0.1 to 1.
51    pub opacity: f32,
52    /// Its ground; none, the theme's warning colour.
53    pub ground: Option<Rgb>,
54    /// Its words' colour; none, whichever of black or white reads on its
55    /// ground.
56    pub text: Option<Rgb>,
57    /// How long one stays, in seconds.
58    pub seconds: u16,
59}
60
61/// The defaults a page without its own says nothing of.
62const WIDTH: u8 = 60;
63const OPACITY: u8 = 90;
64const SECONDS: u16 = 10;
65
66/// The `[banner]` table as the file writes it: a key left out is the
67/// default.
68#[derive(Clone, Debug, Default, PartialEq, Eq, Deserialize, Serialize)]
69#[serde(deny_unknown_fields)]
70pub struct BannerFile {
71    #[serde(default, skip_serializing_if = "Option::is_none")]
72    across: Option<String>,
73    #[serde(default, skip_serializing_if = "Option::is_none")]
74    down: Option<String>,
75    #[serde(default, skip_serializing_if = "Option::is_none")]
76    width: Option<u8>,
77    #[serde(default, skip_serializing_if = "Option::is_none")]
78    opacity: Option<u8>,
79    #[serde(default, skip_serializing_if = "Option::is_none")]
80    ground: Option<String>,
81    #[serde(default, skip_serializing_if = "Option::is_none")]
82    text: Option<String>,
83    #[serde(default, skip_serializing_if = "Option::is_none")]
84    seconds: Option<u16>,
85}
86
87impl BannerFile {
88    /// Whether it says nothing: left out of the file.
89    #[must_use]
90    pub fn is_empty(&self) -> bool {
91        *self == Self::default()
92    }
93
94    /// How a banner is shown, as it says.
95    #[must_use]
96    pub fn look(&self) -> Look {
97        let across = match self.across.as_deref() {
98            Some("left") => Across::Left,
99            Some("right") => Across::Right,
100            _ => Across::Centre,
101        };
102        let down = match self.down.as_deref() {
103            Some("middle") => Down::Middle,
104            Some("bottom") => Down::Bottom,
105            _ => Down::Top,
106        };
107        let share = |percent: Option<u8>, default: u8| {
108            f32::from(percent.unwrap_or(default).clamp(10, 100)) / 100.0
109        };
110        Look {
111            across,
112            down,
113            width: share(self.width, WIDTH),
114            opacity: share(self.opacity, OPACITY),
115            ground: self.ground.as_deref().and_then(parse_hex),
116            text: self.text.as_deref().and_then(parse_hex),
117            seconds: self.seconds.unwrap_or(SECONDS).clamp(1, 600),
118        }
119    }
120
121    /// The *Banners* page.
122    #[must_use]
123    #[expect(
124        clippy::too_many_lines,
125        reason = "one page's rows in order, each a few lines of words"
126    )]
127    pub fn page(&self, file: &str) -> Page {
128        let look = self.look();
129        let row =
130            |key: &str, label: &str, help: &str, kind: RowKind, value: Value, here: bool| Row {
131                key: format!("{KEY}{key}"),
132                label: label.to_owned(),
133                help: help.to_owned(),
134                kind,
135                value,
136                here,
137                from: None,
138                needs: None,
139            };
140        let choice = |pairs: &[(&str, &str)]| {
141            RowKind::Choice(
142                pairs
143                    .iter()
144                    .map(|(value, said)| ((*value).to_owned(), (*said).to_owned()))
145                    .collect(),
146            )
147        };
148        let percent = |share: f32| format!("{}", (share * 100.0).round());
149        Page {
150            id: PAGE.to_owned(),
151            title: "Banners".to_owned(),
152            file: file.to_owned(),
153            takes: "at once".to_owned(),
154            problem: None,
155            rows: vec![
156                row(
157                    "across",
158                    "Across",
159                    "Where a banner floats from left to right, over the windows.",
160                    choice(&[("left", "Left"), ("centre", "Centre"), ("right", "Right")]),
161                    Value::Text(
162                        match look.across {
163                            Across::Left => "left",
164                            Across::Centre => "centre",
165                            Across::Right => "right",
166                        }
167                        .to_owned(),
168                    ),
169                    self.across.is_some(),
170                ),
171                row(
172                    "down",
173                    "Down",
174                    "Where it floats from top to bottom: under the top bar, the middle, or over \
175                     the command line.",
176                    choice(&[("top", "Top"), ("middle", "Middle"), ("bottom", "Bottom")]),
177                    Value::Text(
178                        match look.down {
179                            Down::Top => "top",
180                            Down::Middle => "middle",
181                            Down::Bottom => "bottom",
182                        }
183                        .to_owned(),
184                    ),
185                    self.down.is_some(),
186                ),
187                row(
188                    "width",
189                    "Width",
190                    "How wide, in percent of the window.",
191                    RowKind::Slider {
192                        min: 10.0,
193                        max: 100.0,
194                        step: 5.0,
195                    },
196                    Value::Text(percent(look.width)),
197                    self.width.is_some(),
198                ),
199                row(
200                    "opacity",
201                    "Opacity",
202                    "How solid, in percent: below 100 what is under it shows through.",
203                    RowKind::Slider {
204                        min: 10.0,
205                        max: 100.0,
206                        step: 5.0,
207                    },
208                    Value::Text(percent(look.opacity)),
209                    self.opacity.is_some(),
210                ),
211                row(
212                    "ground",
213                    "Ground",
214                    "Its colour; empty, the theme's warning colour.",
215                    RowKind::Color,
216                    Value::Text(look.ground.map(hex).unwrap_or_default()),
217                    self.ground.is_some(),
218                ),
219                row(
220                    "text",
221                    "Words",
222                    "Its words' colour; empty, black or white, whichever reads on its ground.",
223                    RowKind::Color,
224                    Value::Text(look.text.map(hex).unwrap_or_default()),
225                    self.text.is_some(),
226                ),
227                row(
228                    "seconds",
229                    "Stays for",
230                    "How long one stays, in seconds. A click on it takes it away sooner.",
231                    RowKind::Whole { min: 1, max: 600 },
232                    Value::Text(look.seconds.to_string()),
233                    self.seconds.is_some(),
234                ),
235            ],
236        }
237    }
238
239    /// `key`, without [`KEY`], set to `to`, or back to its default. What was
240    /// done.
241    ///
242    /// # Errors
243    ///
244    /// No such key, or a value it does not take.
245    pub fn change(&mut self, key: &str, to: Option<&str>) -> Result<String, String> {
246        let to = to.map(str::trim).filter(|to| !to.is_empty());
247        let percent = |typed: &str| {
248            typed
249                .parse::<f32>()
250                .ok()
251                .filter(|n| (10.0..=100.0).contains(n))
252                .map(|n| {
253                    #[expect(
254                        clippy::cast_possible_truncation,
255                        clippy::cast_sign_loss,
256                        reason = "10 to 100, checked"
257                    )]
258                    let n = n.round() as u8;
259                    n
260                })
261                .ok_or_else(|| format!("A percent from 10 to 100, not {typed}."))
262        };
263        let colour = |typed: &str| {
264            parse_hex(typed)
265                .map(hex)
266                .ok_or_else(|| format!("`{typed}` is not a colour: say #rrggbb."))
267        };
268        match key {
269            "across" => {
270                self.across = match to {
271                    None | Some("centre") => None,
272                    Some(side @ ("left" | "right")) => Some(side.to_owned()),
273                    Some(other) => return Err(format!("Left, centre or right, not {other}.")),
274                };
275            }
276            "down" => {
277                self.down = match to {
278                    None | Some("top") => None,
279                    Some(side @ ("middle" | "bottom")) => Some(side.to_owned()),
280                    Some(other) => return Err(format!("Top, middle or bottom, not {other}.")),
281                };
282            }
283            "width" => self.width = to.map(percent).transpose()?.filter(|n| *n != WIDTH),
284            "opacity" => self.opacity = to.map(percent).transpose()?.filter(|n| *n != OPACITY),
285            "ground" => self.ground = to.map(colour).transpose()?,
286            "text" => self.text = to.map(colour).transpose()?,
287            "seconds" => {
288                self.seconds = to
289                    .map(|typed| {
290                        typed
291                            .parse::<u16>()
292                            .ok()
293                            .filter(|n| (1..=600).contains(n))
294                            .ok_or_else(|| format!("Seconds from 1 to 600, not {typed}."))
295                    })
296                    .transpose()?
297                    .filter(|n| *n != SECONDS);
298            }
299            other => return Err(format!("The Banners page has no setting {other}.")),
300        }
301        Ok("Banners are shown so from now.".to_owned())
302    }
303}

305#[cfg(test)]
306mod tests {
307    use super::*;
308
309    /// Unset, a banner floats at the top in the middle, 60% wide and 90%
310    /// solid, in the theme's colours, for ten seconds; each setting changes
311    /// its own part, and a default set is left out of the file.
312    #[test]
313    fn a_banner_looks_as_its_settings_say() {
314        let mut file = BannerFile::default();
315        let look = file.look();
316        assert_eq!((look.across, look.down), (Across::Centre, Down::Top));
317        assert!((look.width - 0.6).abs() < 0.001 && (look.opacity - 0.9).abs() < 0.001);
318        assert_eq!((look.ground, look.text, look.seconds), (None, None, 10));
319        file.change("across", Some("right")).unwrap();
320        file.change("down", Some("bottom")).unwrap();
321        file.change("width", Some("40")).unwrap();
322        file.change("ground", Some("#102030")).unwrap();
323        file.change("seconds", Some("4")).unwrap();
324        let look = file.look();
325        assert_eq!((look.across, look.down), (Across::Right, Down::Bottom));
326        assert!((look.width - 0.4).abs() < 0.001);
327        assert_eq!(look.ground, Some([0x10, 0x20, 0x30]));
328        assert_eq!(look.seconds, 4);
329        assert!(file.change("width", Some("5")).is_err());
330        file.change("opacity", Some("90")).unwrap();
331        assert_eq!(file.opacity, None, "the default is not written");
332        file.change("width", None).unwrap();
333        assert_eq!(file.width, None);
334    }

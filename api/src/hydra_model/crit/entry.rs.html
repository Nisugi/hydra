1//! `CritEntry`: one row of one critical-hit table.
2//!
3//! The enums in `crit/types.rs` are the vocabulary; this is the record built
4//! out of them.
5//!
6//! The source tables give the damage type and location twice -- once as a
7//! table key and once inside the record. The extractor asserts the two agree
8//! for all 2,394 entries and keeps one copy, in `damage_type` and `location`.
9
10use super::types::{DamageType, Location, Position, SecondaryWound};
11
12/// The sentinel `stunned` value meaning "this crit stuns, for an unknown
13/// number of rounds".
14///
15/// 999 is the convention the source tables use for "unknown".
16pub const STUN_UNKNOWN: u16 = 999;
17
18/// One row of one critical-hit table: what this crit does, and the message
19/// that announces it.
20///
21/// The damage type and location appear once, in `damage_type` and
22/// `location`; the extractor checked that the source's duplicate copies agree.
23// Eight of the nineteen fields are booleans, and clippy's
24// `struct_excessive_bools` is right that this usually means a missing enum.
25// Here it does not: these eight are INDEPENDENT effects that co-occur -- a
26// single crit can be fatal and amputate and stun and silence at once, and 255
27// entries do set `amputated` alongside other flags. Collapsing them into an
28// enum would forbid combinations the data contains. The fields also mirror
29// the source tables one for one, and the digest test asserts field-for-field
30// equality with them, so a "tidier" shape here would stop matching the data.
31#[allow(clippy::struct_excessive_bools)]
32#[derive(Debug, Clone, PartialEq, Eq)]
33pub struct CritEntry {
34    /// Which table this row is from: column 1 of the TSV.
35    pub damage_type: DamageType,
36    /// Where the hit landed: the row's key within its table.
37    pub location: Location,
38    /// Measured 0..=11.
39    pub rank: u8,
40    /// Extra damage. Measured 0..=88.
41    pub damage: u16,
42    /// The posture the crit knocks the target into; `None` when it does not
43    /// change posture, which is the common case.
44    pub position: Option<Position>,
45    /// The crit kills the target outright.
46    pub fatal: bool,
47    /// Rounds of stun. Measured 0..=20, plus `STUN_UNKNOWN`.
48    ///
49    /// **999 means "unknown", not 999 rounds.** Exactly one entry uses it --
50    /// `generic/unspecified/0`, whose message says the target is stunned
51    /// without saying for how long. A caller that treats this as a duration
52    /// will wait a very long time, so `stun_is_known` exists to ask.
53    pub stunned: u16,
54    /// The crit severs the limb at `location`.
55    pub amputated: bool,
56    /// **`false` in all 2,394 entries.** Kept because it is part of the table
57    /// schema and a future table may set it, but it carries no information
58    /// today. Measured:
59    /// `cut -f9 crates/hydra-model/data/crit_tables.tsv | tail -n +2 | sort -u`
60    /// prints `0` and nothing else.
61    pub crippled: bool,
62    /// The crit leaves the target unconscious or asleep.
63    pub sleeping: bool,
64    /// The crit leaves the target dazed; the tables use this for
65    /// unarmed-combat crits.
66    pub dazed: bool,
67    /// The target favors the wounded limb. The tables leave its finer meaning
68    /// undecided, so read it as a bare flag only.
69    pub limb_favored: bool,
70    /// Extra roundtime in seconds. Measured domain {0, 2, 5, 10, 20}.
71    pub roundtime: u8,
72    /// The crit prevents the target speaking or casting.
73    pub silenced: bool,
74    /// The crit makes the target's actions slower.
75    pub slowed: bool,
76    /// Rank of the wound at `location`. Measured 0..=3.
77    pub wound_rank: u8,
78    /// A second wound at another location; `None` for all but 156 entries.
79    pub secondary_wound: Option<SecondaryWound>,
80    /// The regex source that matches this crit's message, as the table
81    /// writes it.
82    ///
83    /// Kept beside the compiled form so a caller can report *why* a line
84    /// matched, and so the digest test can compare against the TSV.
85    pub pattern: String,
86}
87
88impl CritEntry {
89    /// Whether `stunned` is a real duration rather than the unknown sentinel.
90    #[must_use]
91    pub const fn stun_is_known(&self) -> bool {
92        self.stunned != STUN_UNKNOWN
93    }
94
95    /// The key this entry is sorted and looked up by.
96    ///
97    /// VERIFIED unique across all 2,394 entries by the extractor, which
98    /// refuses to write a TSV with a duplicate.
99    #[must_use]
100    pub const fn key(&self) -> (DamageType, Location, u8) {
101        (self.damage_type, self.location, self.rank)
102    }
103}

1//! What the spell table did not keep, cut by `tools/extract_spell_extras.rb`
2//! into `data/spell_extras.tsv` and joined to it by number.
3//!
4//! Each `<duration>`'s shape -- whether a cast **stacks** onto the time left
5//! or **refreshes** it, whether it can be multicast, its ceiling, whether it
6//! outlasts death and whether it runs in real time -- the `<spell>`'s own
7//! `incant`, `stance` and `channel` flags, every `<cost>` as written, and the
8//! `<cast-proc>`, a Ruby snippet for how the spell is cast when `incant` will
9//! not do. The flags are yes-or-absent in the file, and `incant` is no-or-absent.
10
11use std::collections::BTreeMap;
12
13use super::{CastType, REC, UNIT};

15const EXTRAS_TSV: &str = include_str!("../../data/spell_extras.tsv");

17/// How another cast of a spell already up adds to it.
18#[derive(Clone, Copy, Debug, PartialEq, Eq)]
19pub enum Span {
20    /// Its time is added to what is left, up to a ceiling.
21    Stackable,
22    /// It restarts the time.
23    Refreshable,
24}
25
26/// One `<duration>`'s shape.
27#[derive(Clone, Debug, PartialEq, Eq)]
28pub struct Shape {
29    /// Which casting it describes.
30    pub cast: CastType,
31    /// Stacks or refreshes; `None` for neither (a cast while up does nothing).
32    pub span: Option<Span>,
33    /// Whether it can be multicast; `None` when the file does not say.
34    pub multicastable: Option<bool>,
35    /// The ceiling in minutes, as written (`max='250'`).
36    pub max: Option<String>,
37    /// Whether it outlasts death; `None` when the file does not say.
38    pub persist_on_death: Option<bool>,
39    /// It runs in real time, not game time.
40    pub real_time: bool,
41}
42
43/// One `<cost>`, as written.
44#[derive(Clone, Debug, PartialEq, Eq)]
45pub struct Cost {
46    /// `mana`, `spirit`, `stamina` or `renew`.
47    pub kind: String,
48    /// The cost: a number, or Ruby for one.
49    pub text: String,
50    /// The file's own note on it, when it has one.
51    pub fixme: Option<String>,
52}

54/// What the table did not keep about one spell.
55#[derive(Clone, Debug, PartialEq, Eq)]
56pub struct Extras {
57    /// `incant` casts it; `false` for the seventeen marked `incant='no'`.
58    pub incant: bool,
59    /// Wants an offensive stance to cast (`stance='yes'`).
60    pub stance: bool,
61    /// Can be channeled (`channel='yes'`).
62    pub channel: bool,
63    /// Each `<duration>`'s shape, in the file's order.
64    pub shapes: Vec<Shape>,
65    /// Every `<cost>`, in the file's order.
66    pub costs: Vec<Cost>,
67    /// The `<cast-proc>`, when the spell has one.
68    pub cast_proc: Option<String>,
69    /// Every `<message>`, by type (`start`, `end`, ...), in the file's
70    /// order: the table keeps one per type.
71    pub messages: Vec<(String, String)>,
72}
73
74impl Default for Extras {
75    fn default() -> Self {
76        Self {
77            incant: true,
78            stance: false,
79            channel: false,
80            shapes: Vec::new(),
81            costs: Vec::new(),
82            cast_proc: None,
83            messages: Vec::new(),
84        }
85    }
86}

88impl Extras {
89    /// The shape of this casting, when the file gives one.
90    #[must_use]
91    pub fn shape(&self, cast: CastType) -> Option<&Shape> {
92        self.shapes.iter().find(|s| s.cast == cast)
93    }

95    /// What the cast proc sends the game, read off it; `None` for a spell
96    /// with no proc. See [`ProcLine`].
97    #[must_use]
98    pub fn proc_line(&self) -> Option<ProcLine> {
99        self.cast_proc.as_deref().map(ProcLine::read)
100    }

103/// What a `<cast-proc>` sends the game: the first string it hands to
104/// `fput`, `put`, `multifput` or `dothistimeout`, which is how every society
105/// power's proc is written (`awk -F'\t' '$1>=9700 {print $7}'
106/// data/spell_extras.tsv`). The proc is Ruby; Hydra reads only the line,
107/// so a proc's loops and waits (9918's `sign of wracking` until mana is
108/// full) are not carried, only what it says.
109#[derive(Clone, Debug, PartialEq, Eq)]
110pub enum ProcLine {
111    /// This line. `targeted`: the proc appends the target to it
112    /// (`"symbol of submission#{target}"`, `"smite#{target}"`), as
113    /// ` #<id>` for a game object and ` <words>` otherwise.
114    Sends {
115        /// The line, without the target.
116        line: String,
117        /// The proc appends `#{target}`.
118        targeted: bool,
119    },
120    /// It sends an empty line: `fput ""`, Kai's Strike (9808) and Symbol of
121    /// Thought (9803), powers the Order of Voln made passive.
122    Nothing,
123    /// It sends nothing at all: 9920 (Sign of Hopelessness) is an `echo`
124    /// asking what is wrong with whoever casts it.
125    Refuses,
126}
127
128impl ProcLine {
129    /// Read a proc as written.
130    #[must_use]
131    pub fn read(proc: &str) -> Self {
132        let Some(quoted) = first_sent(proc) else {
133            return Self::Refuses;
134        };
135        let targeted = quoted.contains("#{target}");
136        let line = quoted.replace("#{target}", "").trim().to_owned();
137        if line.is_empty() {
138            Self::Nothing
139        } else {
140            Self::Sends { line, targeted }
141        }
142    }
143}

145/// The string literal handed to the first `fput`, `put`, `multifput` or
146/// `dothistimeout` in `proc`, quotes off.
147fn first_sent(proc: &str) -> Option<&str> {
148    const SENDS: [&str; 4] = ["dothistimeout", "multifput", "fput", "put"];
149    let word_start = |at: usize| {
150        proc[..at]
151            .chars()
152            .next_back()
153            .is_none_or(|c| !(c.is_ascii_alphanumeric() || c == '_'))
154    };
155    proc.char_indices().find_map(|(at, _)| {
156        let verb = SENDS
157            .iter()
158            .find(|verb| proc[at..].starts_with(*verb) && word_start(at))?;
159        let rest = proc[at + verb.len()..].trim_start_matches([' ', '(']);
160        let quote = rest.chars().next().filter(|c| *c == '"' || *c == '\'')?;
161        let body = &rest[1..];
162        body.find(quote).map(|end| &body[..end])
163    })
164}

166/// Every spell's extras, by number.
167pub(super) fn read() -> BTreeMap<u16, Extras> {
168    EXTRAS_TSV
169        .lines()
170        .filter(|line| !line.starts_with('#') && !line.starts_with("number\t"))
171        .filter_map(read_row)
172        .collect()
173}
174
175fn read_row(line: &str) -> Option<(u16, Extras)> {
176    let mut field = line.split('\t');
177    let number = field.next()?.parse().ok()?;
178    let incant = field.next()? != "no";
179    let stance = field.next()? == "yes";
180    let channel = field.next()? == "yes";
181    let shapes = field
182        .next()?
183        .split(REC)
184        .filter(|cell| !cell.is_empty())
185        .filter_map(read_shape)
186        .collect();
187    let costs = field
188        .next()?
189        .split(REC)
190        .filter(|cell| !cell.is_empty())
191        .filter_map(|cell| {
192            let mut part = cell.split(UNIT);
193            Some(Cost {
194                kind: part.next()?.to_owned(),
195                text: part.next()?.to_owned(),
196                fixme: part.next().filter(|f| !f.is_empty()).map(str::to_owned),
197            })
198        })
199        .collect();
200    let cast_proc = field.next().filter(|p| !p.is_empty()).map(str::to_owned);
201    let messages = field
202        .next()
203        .unwrap_or_default()
204        .split(REC)
205        .filter_map(|cell| cell.split_once(UNIT))
206        .map(|(kind, text)| (kind.to_owned(), text.to_owned()))
207        .collect();
208    Some((
209        number,
210        Extras {
211            incant,
212            stance,
213            channel,
214            shapes,
215            costs,
216            cast_proc,
217            messages,
218        },
219    ))
220}

222/// `cast-type~span~multicastable~max~persist-on-death~real-time`.
223fn read_shape(cell: &str) -> Option<Shape> {
224    let mut part = cell.split(UNIT);
225    let cast = match part.next()? {
226        "target" => CastType::Target,
227        _ => CastType::SelfCast,
228    };
229    let span = match part.next()? {
230        "stackable" => Some(Span::Stackable),
231        "refreshable" => Some(Span::Refreshable),
232        _ => None,
233    };
234    let yes_no = |value: &str| match value {
235        "yes" => Some(true),
236        "no" => Some(false),
237        _ => None,
238    };
239    let multicastable = yes_no(part.next()?);
240    let max = part.next().filter(|m| !m.is_empty()).map(str::to_owned);
241    let persist_on_death = yes_no(part.next()?);
242    let real_time = part.next() == Some("yes");
243    Some(Shape {
244        cast,
245        span,
246        multicastable,
247        max,
248        persist_on_death,
249        real_time,
250    })

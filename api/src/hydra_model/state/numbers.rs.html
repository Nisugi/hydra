1//! The one reader for a comma-grouped integer the game printed.
2//!
3//! # Why one, and why strict
4//!
5//! Every reader of a game figure -- currency, experience, the bank, standing,
6//! the bounty, `info` -- uses this one, so they cannot disagree about the one
7//! thing that matters. Keeping **every digit and dropping everything else**
8//! would read `"12a3"` as `123` and `"-5 (was 9)"` as `59`.
9//!
10//! That is the partial parse `hydra-protocol`'s `numbers.rs` refuses in its
11//! own `signed`: *"a partial parse of something unexpected, which is precisely
12//! how a wrong number reaches a behavior that acts on it."* A figure the wire
13//! did not send is worse than no figure, because a caller cannot tell the two
14//! apart -- `None` at least says "not understood".
15//!
16//! So this accepts exactly `-?[\d,]+`, and nothing else: the whole token must
17//! be a number, commas are the only separator, and the sign survives. `T`
18//! decides the range, so an unsigned
19//! target refuses a minus sign by construction rather than by a check here.
20
21use std::str::FromStr;
22
23/// `"1,024"` -> `1024`, `"-3"` -> `-3`; `None` for anything that is not a
24/// comma-grouped integer **end to end** (`"12a3"`, `"1 024"`, `""`, `","`).
25///
26/// Surrounding whitespace is trimmed; nothing else is forgiven.
27pub(crate) fn grouped<T: FromStr>(text: &str) -> Option<T> {
28    let text = text.trim();
29    let magnitude = text.strip_prefix('-').unwrap_or(text);
30    // At least one digit, and nothing but digits and commas. `FromStr` then
31    // rejects what is left over -- a `-` into an unsigned type, or a value out
32    // of `T`'s range -- rather than wrapping or clamping it.
33    if !magnitude.bytes().any(|b| b.is_ascii_digit())
34        || !magnitude.bytes().all(|b| b.is_ascii_digit() || b == b',')
35    {
36        return None;
37    }
38    text.replace(',', "").parse().ok()
39}

41#[cfg(test)]
42mod tests {
43    use super::grouped;
44
45    #[test]
46    fn a_grouped_figure_reads_whole() {
47        assert_eq!(grouped::<u64>("1,453,539,090"), Some(1_453_539_090));
48        assert_eq!(grouped::<u32>(" 3673 "), Some(3673));
49        assert_eq!(grouped::<i64>("-1,234"), Some(-1234));
50        assert_eq!(grouped::<u32>("0"), Some(0));
51    }
52
53    #[test]
54    fn a_partial_number_is_refused_not_repaired() {
55        // Filtering to digits would turn these into numbers the wire never
56        // sent.
57        assert_eq!(grouped::<u64>("12a3"), None);
58        assert_eq!(grouped::<u64>("1 024"), None);
59        assert_eq!(grouped::<i64>("-5 (was 9)"), None);
60        assert_eq!(grouped::<u64>(""), None);
61        assert_eq!(grouped::<u64>(","), None);
62        assert_eq!(grouped::<u64>("-"), None);
63    }
64
65    #[test]
66    fn the_target_type_decides_sign_and_range() {
67        assert_eq!(grouped::<u64>("-5"), None, "unsigned refuses a sign");
68        assert_eq!(grouped::<u32>("4,294,967,296"), None, "out of range");
69        assert_eq!(grouped::<i32>("-2,147,483,648"), Some(i32::MIN));
70    }
71}

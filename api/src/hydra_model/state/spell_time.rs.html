1//! A spell's duration and cost for this character: the spell table's
2//! expressions, evaluated against what the model knows.
3//!
4//! The names resolved: `Spells.<list>` (a circle's ranks, matched to the
5//! printed circle name without spaces or case), `Skills.<skill>` (by its
6//! snake-case key or its short form, `emc`, `slreligion`), `Stats.level`,
7//! `Society.rank`, `Spellsong.timeleft`, and `Spell[N].known?`, `.active?`,
8//! `.timeleft`. Anything the model has not been told, or a name it has no
9//! word for, leaves the whole answer `None`.
10//!
11//! # A bard's songs
12//!
13//! The songs' time left and renewal cost live here rather than beside the
14//! rest of the song arithmetic (`character/spellsong.rs`), because both read
15//! the effects list and the spell table, which a character alone does not
16//! hold.
17//!
18//! **Time left reads the game, not a timestamp.** The songs are renewed
19//! together, so the time left on an active song is the time left on the
20//! cycle, and the effects list states it. Nothing has to remember when the
21//! songs were last renewed; the model reads the game's own dialog each time.
22
23use crate::spells::expr::{self, Name, Value};

25/// The songs renewed and priced together.
26pub const SONGS: [u16; 9] = [1003, 1006, 1009, 1010, 1012, 1014, 1018, 1019, 1025];

32    /// Minutes one cast of spell `number` lasts, cast `cast`. A target
33    /// duration the table does not give falls back to the self one.
34    #[must_use]
35    pub fn spell_minutes(&self, number: u16, cast: CastType) -> Option<f64> {
36        let spell = spells::spell(number)?;
37        let duration = spell
38            .duration(cast)
39            .or_else(|| spell.duration(CastType::SelfCast))?;
40        match duration {
41            Duration::Fixed(text) => text.parse().ok(),
42            Duration::Derived(source) => self.evaluate(source),
43            Duration::Unknown(_) => None,
44        }
45    }
46
47    /// Spell `number`'s cost of `kind` (`mana`, `spirit`, `stamina`,
48    /// `renew`): the table's number, else its expression evaluated. `Some(0)`
49    /// when the spell states no cost of that kind.
50    #[must_use]
51    pub fn spell_cost(&self, number: u16, kind: &str) -> Option<f64> {
52        let spell = spells::spell(number)?;
53        match spell.extras.costs.iter().find(|c| c.kind == kind) {
54            None => Some(0.0),
55            Some(cost) => cost
56                .text
57                .trim()
58                .parse()
59                .ok()
60                .or_else(|| self.evaluate(&cost.text)),
61        }
62    }

64    /// Minutes until a bard's songs lapse. `0.0` for any other profession.
65    /// For a bard, the least time left on an active bard spell (`10xx`) in
66    /// the effects list; `None` while no
67    /// song is up, or the profession or the clock has not been told.
68    #[must_use]
69    pub fn spellsong_timeleft(&self) -> Option<f64> {
70        let profession = self.character.identity.profession.as_deref()?;
71        if !profession.eq_ignore_ascii_case("bard") {
72            return Some(0.0);
73        }
74        let now = self.game_time_now()?;
75        self.effects
76            .iter()
77            .filter(|(id, _)| is_song(id))
78            .filter(|(id, _)| self.effects.active(id, now) == Some(true))
79            .filter_map(|(id, _)| self.effects.remaining(id, now))
80            .min()
81            .map(|secs| f64::from(secs) / 60.0)
82    }
83
84    /// Mana to renew every song that is up: each active song's `renew` cost
85    /// from the spell table, summed. `0.0` with none up. `None` when the
86    /// clock has not been told, or an active song's cost cannot be worked
87    /// out.
88    ///
89    /// UNVERIFIED whether renewing several songs at once carries a penalty;
90    /// the sum is the plain sum.
91    #[must_use]
92    pub fn spellsong_renew_cost(&self) -> Option<f64> {
93        let now = self.game_time_now()?;
94        let mut total = 0.0;
95        for song in SONGS {
96            if self.effects.active(&song.to_string(), now) == Some(true) {
97                total += self.spell_cost(song, "renew")?;
98            }
99        }
100        Some(total)
101    }

103    /// Evaluate a spell-table expression against this character.
104    #[must_use]
105    pub fn evaluate(&self, source: &str) -> Option<f64> {
106        expr::evaluate(source, &|name| self.resolve(name))
107    }
108
109    /// Every name [`Self::evaluate`] can work out, for a front end to offer:
110    /// the fixed ones, each skill, and each spell circle the `skills` table
111    /// has listed.
112    #[must_use]
113    pub fn value_names(&self) -> Vec<String> {
114        let mut names = vec![
115            "Stats.level".to_owned(),
116            "Society.rank".to_owned(),
117            "Spellsong.timeleft".to_owned(),
118        ];
119        names.extend(
120            SkillKind::ALL
121                .into_iter()
122                .map(|kind| format!("Skills.{}", kind.key())),
123        );
124        names.extend(
125            self.character
126                .skills
127                .circles()
128                .map(|(printed, _)| format!("Spells.{}", squash(printed))),
129        );
130        names
131    }
132
133    fn resolve(&self, name: &Name) -> Option<Value> {
134        let num = |n: u16| Some(Value::Int(i64::from(n)));
135        match name {
136            Name::Path(module, method) => match (module.as_str(), method.as_str()) {
137                ("Spells", list) => self
138                    .character
139                    .skills
140                    .circles()
141                    .find(|(printed, _)| squash(printed) == list.replace('_', ""))
142                    .and_then(|(_, ranks)| num(ranks))
143                    // A circle the `skills` table did not list has no ranks.
144                    .or_else(|| {
145                        self.character
146                            .skills
147                            .known_count()
148                            .gt(&0)
149                            .then_some(Value::Int(0))
150                    }),
151                ("Skills", short) => {
152                    let kind = SkillKind::ALL
153                        .into_iter()
154                        .find(|k| k.key() == short || lich_short(&k.key()) == short)?;
155                    let ranks = self.character.skills.ranks(kind)?;
156                    num(ranks)
157                }
158                ("Stats", "level") => {
159                    // `Level 100`, verbatim: the number is the last word.
160                    let label = self.character.experience.level.as_deref()?;
161                    let level: u16 = label.split_whitespace().last()?.parse().ok()?;
162                    num(level)
163                }
164                ("Society", "rank") => num(u16::from(self.character.standing.society_rank?)),
165                ("Spellsong", "timeleft") => self.spellsong_timeleft().map(Value::Num),
166                _ => None,
167            },
168            Name::Spell(number, method) => {
169                let id = number.to_string();
170                match method.as_str() {
171                    "known?" => self.known_spells.knows(u32::from(*number)).map(Value::Bool),
172                    "active?" => {
173                        let now = self.game_time_now()?;
174                        self.effects.active(&id, now).map(Value::Bool)
175                    }
176                    "timeleft" => {
177                        let now = self.game_time_now()?;
178                        let secs = self.effects.remaining(&id, now).unwrap_or(0);
179                        Some(Value::Num(f64::from(secs) / 60.0))
180                    }
181                    _ => None,
182                }
183            }
184        }
185    }
186}

188/// A bard spell's id: `10` and two digits.
189fn is_song(id: &str) -> bool {
190    id.len() == 4 && id.starts_with("10") && id.bytes().all(|b| b.is_ascii_digit())
191}
192
193/// A printed name without spaces, punctuation or case.
194fn squash(printed: &str) -> String {
195    printed
196        .chars()
197        .filter(char::is_ascii_alphanumeric)
198        .map(|c| c.to_ascii_lowercase())
199        .collect()
200}
201
202/// The short form of a skill key, as the spell table's expressions write it:
203/// underscores out, then the lore and mana-control prefixes shortened.
204fn lich_short(key: &str) -> String {
205    key.replace('_', "")
206        .replace("elementallore", "el")
207        .replace("spirituallore", "sl")
208        .replace("sorcerouslore", "sl")
209        .replace("mentallore", "ml")
210        .replace("elementalmanacontrol", "emc")
211        .replace("spiritmanacontrol", "smc")
212        .replace("mentalmanacontrol", "mmc")
213}

222    /// A bard at game second 1000 with these songs up, `secs` left on each.
223    fn bard(songs: &[(u16, u32)]) -> GameState {
224        let mut state = GameState::default();
225        state.apply(&Frame::Prompt {
226            time: "1000".into(),
227            text: ">".into(),
228        });
229        state.character.identity.profession = Some("Bard".to_owned());
230        for (song, secs) in songs {
231            state.effects.insert_lasting(
232                song.to_string(),
233                Effect {
234                    category: "Active Spells".to_owned(),
235                    text: format!("song {song}"),
236                    ends_at: None,
237                    percent: 100,
238                },
239                *secs,
240                Some(1000),
241            );
242        }
243        state
244    }

246    /// The time left on the songs is the time left on an active bard spell;
247    /// the least of them.
248    #[test]
249    fn a_bards_song_time_is_read_off_the_effects() {
250        let state = bard(&[(1003, 600), (1010, 540)]);
251        assert_eq!(state.spellsong_timeleft(), Some(9.0));
252        assert_eq!(bard(&[]).spellsong_timeleft(), None, "no song up");
253    }
254
255    /// Not a bard, no song time.
256    #[test]
257    fn anyone_else_has_no_song_time() {
258        let mut state = bard(&[(1003, 600)]);
259        state.character.identity.profession = Some("Wizard".to_owned());
260        assert_eq!(state.spellsong_timeleft(), Some(0.0));
261        state.character.identity.profession = None;
262        assert_eq!(state.spellsong_timeleft(), None, "not yet told");
263    }

265    /// Each active song's renew cost, summed; none up, nothing to pay.
266    #[test]
267    fn the_renew_cost_is_the_active_songs_summed() {
268        assert_eq!(bard(&[]).spellsong_renew_cost(), Some(0.0));
269        // Fortitude Song renews for 1, Sonic Weapon Song for 4 (the table's
270        // `renew_cost`, `data/spells.tsv`).
271        assert_eq!(bard(&[(1003, 600)]).spellsong_renew_cost(), Some(1.0));
272        assert_eq!(
273            bard(&[(1003, 600), (1012, 600)]).spellsong_renew_cost(),
274            Some(5.0)
275        );
276        // Song of Valor's is `3+((Spells.bard-10)/10)`: without the Bard
277        // circle's ranks, the sum is not known either.
278        assert_eq!(
279            bard(&[(1003, 600), (1010, 600)]).spellsong_renew_cost(),
280            None
281        );
282    }

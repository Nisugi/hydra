1//! The lines a hunt reacts to that are not combat: a weapon knocked away, a
2//! curse or infection taken, a trap sprung, an ambusher, rooted, a bless
3//! gone, an arrow stuck, a charge counter, a mark on a creature, the weapon
4//! reaction offered.
5//!
6//! 21 events in seven families, gathered into one table. The state machine
7//! never reads them; the hunt does.
8//!
9//! # Plain text and the line's objects, not the markup
10//!
11//! Each pattern matches the line's plain text and the id and noun come
12//! from the line's objects (`text.rs`): an item is the first
13//! unbolded object, a creature the first bolded one (bold is the wire's
14//! mark for a creature). The weapon reaction's command is the `<d cmd=>`
15//! link's own. So nothing is re-parsed.
16//!
17//! **The weapon is the object right after `your`, or the line is not one.**
18//! The game's weapon lines carry the weapon's link in that place, which is
19//! the half of them a player's words cannot supply. The plain text alone
20//! matches `Bob says, "Your sword tears free from your hands and floats
21//! away."`, and the first unbolded object then is Bob. Spoken lines do not
22//! reach here at all (`ChunkLine::is_spoken`); the anchor is kept as well.
23//!
24//! # Queued, and drained by whoever acts
25//!
26//! A chunk's incidents are queued at its prompt ([`Incidents::take`]),
27//! capped so a session with no hunt running holds a bounded few; the hunt's
28//! driver drains its own fold's queue each turn.
29

42/// How a weapon left the hand (the `disarm` family).
43#[derive(Clone, Copy, Debug, PartialEq, Eq)]
44pub enum Disarm {
45    /// Knocked from the grasp, or wrenched away into the shadows: `recover
46    /// item` where it fell (ecleanse's `recover`).
47    Knocked,
48    /// Torn free and floating (Telekinetic Disarm, 1406): `get` it.
49    Telekinetic,
50    /// Stuck in webbing: `pry` it free.
51    Webbed,
52}

54/// Which hive trap (`hive_trap`'s `kind`).
55#[derive(Clone, Copy, Debug, PartialEq, Eq)]
56pub enum HiveTrap {
57    /// The flickering apparatus.
58    Apparatus,
59    /// The churning ground.
60    Ground,

63/// One thing that happened.
64#[derive(Clone, Debug, PartialEq, Eq)]
65pub enum Incident {
66    /// `disarm_seen`: the weapon left the hand. `None` when the line named
67    /// no object.
68    Disarmed {
69        /// How.
70        how: Disarm,
71        /// The weapon.
72        weapon: Option<ItemRef>,
73    },
74    /// `sanctum_transform`: a Sanctum of Scales sentinel turned the weapon
75    /// into a snake (`clench` recovers it).
76    SanctumSnake {
77        /// What the weapon became.
78        snake: Option<ItemRef>,
79    },
80    /// `itchy_curse`.
81    ItchyCurse,
82    /// `infected_wound`.
83    InfectedWound,
84    /// `hive_trap`.
85    HiveTrap(HiveTrap),
86    /// `entangled`: an unseen force.
87    Entangled,
88    /// `ambusher`: something leapt from hiding. Its noun when the line
89    /// named it.
90    Ambusher(Option<String>),
91    /// `bolted`: `You bolt`.
92    Bolted,
93    /// `rooted`, with the snake's id when a snake holds you.
94    Rooted(Option<String>),
95    /// `unrooted`: free of the snake, by id.
96    Unrooted(Option<String>),
97    /// `item_limit`: too much held to pick more up.
98    ItemLimit,
99    /// `bless_shrugged`: the weapon struck true and the creature shrugged
100    /// off some of the damage.
101    BlessShrugged(Option<ItemRef>),
102    /// `bless_expired`: the weapon returns to normal.
103    BlessExpired(Option<ItemRef>),
104    /// `arrow_stuck`: an arrow in a creature, by the creature's id, and
105    /// where.
106    ArrowStuck {
107        /// The creature.
108        creature: Option<ItemRef>,
109        /// Where it stuck: `neck`, `left arm`'s `arm`, ...
110        at: String,
111    },
112    /// `aiming`: aiming at this part now, or at nothing in particular.
113    Aiming(Option<String>),
114    /// `bond_return`: a bonded weapon flew back to the hand.
115    BondReturn(String),
116    /// `haze_703`: Sounds's blood red haze on or off a creature.
117    Haze {
118        /// The creature.
119        creature: Option<ItemRef>,
120        /// On, or dissipated.
121        on: bool,
122    },
123    /// `rebuke_1614`: struggling against, in awe of, or recovered from the
124    /// radiant aura.
125    Rebuke {
126        /// The creature.
127        creature: Option<ItemRef>,
128        /// On, or recovered.
129        on: bool,
130    },
131    /// `swift_justice`: the charges now.
132    SwiftJustice(u32),
133    /// `arcane_reflex`: on, or back to normal.
134    ArcaneReflex(bool),
135    /// `weapon_reaction`: `You could use this opportunity to <d cmd='WEAPON
136    /// <reaction> #<id>'>`. The reaction and its target, as the link's
137    /// command has them after `WEAPON`: `tackle #1234`.
138    WeaponReaction(String),
139}

317/// One line's incident, if it states one. Each line states at most one:
318/// no two of the patterns match the same line.
319#[must_use]
320pub fn read(line: &ChunkLine) -> Option<Incident> {
321    let p = patterns();
322    let text = line.text();
323    let t = text.as_str();
324    let disarmed = |how| {
325        weapon(line).map(|weapon| Incident::Disarmed {
326            how,
327            weapon: Some(weapon),
328        })
329    };
330    if p.knocked.is_match(t)
331        || p.wrenched.is_match(t)
332        || p.protrusion.is_match(t)
333        || p.swing_protrusion.is_match(t)
334    {
335        return disarmed(Disarm::Knocked);
336    }
337    if p.floats.is_match(t) {
338        return disarmed(Disarm::Telekinetic);
339    }
340    if p.webbing.is_match(t) {
341        return disarmed(Disarm::Webbed);
342    }
343    if p.sanctum.is_match(t) {
344        // The last object on the line is what the weapon became.
345        let snake = super::text::objects(line)
346            .into_iter()
347            .map(|(object, _)| object)
348            .next_back();
349        return Some(Incident::SanctumSnake { snake });
350    }
351    if p.itchy.is_match(t) {
352        return Some(Incident::ItchyCurse);
353    }
354    if p.infected.is_match(t) {
355        return Some(Incident::InfectedWound);
356    }
357    if p.glint.is_match(t) || p.apparatus.is_match(t) {
358        return Some(Incident::HiveTrap(HiveTrap::Apparatus));
359    }
360    if p.churns.is_match(t) || p.frenzy.is_match(t) || p.helpless.is_match(t) {
361        return Some(Incident::HiveTrap(HiveTrap::Ground));
362    }
363    if p.entangled.is_match(t) {
364        return Some(Incident::Entangled);
365    }
366    if p.figure.is_match(t) || p.shadows.is_match(t) {
367        return Some(Incident::Ambusher(None));
368    }
369    if p.leaps.is_match(t) {
370        return Some(Incident::Ambusher(creature(line).map(|c| c.noun)));
371    }
372    if p.bolt.is_match(t) {
373        return Some(Incident::Bolted);
374    }
375    if p.rooted.is_match(t) {
376        return Some(Incident::Rooted(None));
377    }
378    if p.coils.is_match(t) {
379        return Some(Incident::Rooted(creature(line).map(|c| c.id)));
380    }
381    if p.free.is_match(t) {
382        return Some(Incident::Unrooted(creature(line).map(|c| c.id)));
383    }
384    if p.unable_hold.is_match(t) || p.treasure.is_match(t) || p.sigils.is_match(t) {
385        return Some(Incident::ItemLimit);
386    }
387    bless_and_marks(line, t, p)
388}

390/// The weapon a weapon line names: the object whose link comes right after
391/// `your`, where the game puts `<a exist=... noun=...>`. `None` when no
392/// object follows `your`: that line is not the game's, whatever its words.
393fn weapon(line: &ChunkLine) -> Option<ItemRef> {
394    line.runs.runs.windows(2).find_map(|pair| {
395        if !pair[0].text.to_ascii_lowercase().ends_with("your ") {
396            return None;
397        }
398        let object = pair[1].object()?;
399        let LinkKind::Exist { id, noun } = &object.kind else {
400            return None;
401        };
402        Some(ItemRef {
403            id: id.clone(),
404            noun: noun.clone(),
405            text: object.text.clone(),
406        })
407    })
408}

410/// The `bless`, `archery`, `marks` and `reaction` families.
411fn bless_and_marks(line: &ChunkLine, t: &str, p: &Patterns) -> Option<Incident> {
412    if p.shrugs.is_match(t) {
413        return Some(Incident::BlessShrugged(item(line)));
414    }
415    if p.reflex_off.is_match(t) {
416        return Some(Incident::ArcaneReflex(false));
417    }
418    if p.normal.is_match(t) {
419        return weapon(line).map(|weapon| Incident::BlessExpired(Some(weapon)));
420    }
421    if let Some(caps) = p.sticks.captures(t) {
422        return Some(Incident::ArrowStuck {
423            creature: creature(line),
424            at: caps.get(1).map_or("", |m| m.as_str()).to_owned(),
425        });
426    }
427    if let Some(caps) = p.aiming.captures(t) {
428        return Some(Incident::Aiming(caps.get(1).map(|m| m.as_str().to_owned())));
429    }
430    if p.not_aiming.is_match(t) {
431        return Some(Incident::Aiming(None));
432    }
433    if let Some(caps) = p.bond.captures(t) {
434        return Some(Incident::BondReturn(
435            caps.get(1).map_or("", |m| m.as_str()).to_owned(),
436        ));
437    }
438    if p.haze_on.is_match(t) || p.haze_off.is_match(t) {
439        return Some(Incident::Haze {
440            creature: creature(line),
441            on: p.haze_on.is_match(t),
442        });
443    }
444    if p.struggling.is_match(t) || p.awe.is_match(t) || p.recovers.is_match(t) {
445        return Some(Incident::Rebuke {
446            creature: creature(line),
447            on: !p.recovers.is_match(t),
448        });
449    }
450    if let Some(caps) = p
451        .justice_up
452        .captures(t)
453        .or_else(|| p.justice_down.captures(t))
454    {
455        let charges = caps.get(1)?.as_str().parse().ok()?;
456        return Some(Incident::SwiftJustice(charges));
457    }
458    if p.reflex_on.is_match(t) {
459        return Some(Incident::ArcaneReflex(true));
460    }
461    if p.reaction.is_match(t) {
462        let reaction = line.links().find_map(|link| match &link.kind {
463            LinkKind::Direct { cmd } => {
464                let (verb, rest) = cmd.split_once(' ')?;
465                verb.eq_ignore_ascii_case("weapon")
466                    .then(|| rest.trim().to_owned())
467            }
468            _ => None,
469        })?;
470        return Some(Incident::WeaponReaction(reaction));
471    }
472    None
473}

475/// The incidents waiting for whoever acts on them.
476#[derive(Clone, Debug, Default, PartialEq, Eq)]
477pub struct Incidents {
478    held: VecDeque<Incident>,
479}
480
481impl super::GameState {
482    /// The chunk's `incidents`, kept for whoever reads them, and where the
483    /// character aims taken from them: the one fact among them that lasts,
484    /// which the Aim widget displays.
485    pub(super) fn note_incidents(&mut self, incidents: Vec<Incident>) {
486        for incident in &incidents {
487            if let Incident::Aiming(at) = incident {
488                self.aiming.clone_from(at);
489                self.aim_said = true;
490            }
491        }
492        self.incidents.push(incidents);
493    }
494}
495
496impl Incidents {
497    /// Queue a chunk's incidents; past the cap, the oldest go.
498    pub fn push(&mut self, incidents: Vec<Incident>) {
499        for incident in incidents {
500            if self.held.len() >= MAX_HELD {
501                self.held.pop_front();
502            }
503            self.held.push_back(incident);
504        }
505    }
506
507    /// Every waiting incident, oldest first, leaving none.
508    pub fn take(&mut self) -> Vec<Incident> {
509        self.held.drain(..).collect()
510    }
511}

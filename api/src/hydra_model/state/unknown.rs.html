1//! Unknown tags: the evidence of a protocol change, bounded.
2//!
3//! Kept apart from `state.rs` so that file stays under its cap, which
4//! `split_parents_stay_facades` enforces: move code down into a submodule
5//! instead of raising the cap.
6
7use super::GameState;
8
9/// How many distinct unknown-tag sightings [`GameState`] keeps the raw bytes of.
10///
11/// # Why it is bounded at all
12///
13/// Unbounded, a bare `Vec::push` per unknown tag would grow forever, carried
14/// across every reconnect and deep-cloned by every `subscribe()` -- on the
15/// production path, in a client designed for 3-25 simultaneous characters.
16///
17/// # Why 512, and why that is generous rather than tight
18///
19/// MEASURED over a 3 MB capture: **zero** unknown tags. The table covers what
20/// the wire sends today, so nothing is growing. The failure needs Simutronics
21/// to add a tag that rides every prompt -- ~86,000 a day per session -- which
22/// is precisely the event this evidence exists for.
23///
24/// 512 distinct raws is far more than a human will read and far less than a day
25/// of one tag. It is a ceiling on a cost we are not yet paying, not a trade
26/// against one we are.
27pub const MAX_UNKNOWN_TAGS: usize = 512;
28
29/// A tag `hydra-protocol` has no variant for, kept for display and for the log.
30#[derive(Clone, Debug, PartialEq, Eq)]
31pub struct UnknownTag {
32    /// The element name.
33    pub name: String,
34    /// The bytes the game actually sent. The raw form IS the diagnostic: a
35    /// reader has to see what arrived, not a summary of it.
36    pub raw: String,
37}
38
39impl GameState {
40    /// Record one unknown tag: always count it, keep its raw bytes while there
41    /// is room.
42    ///
43    /// The count comes first and is unconditional -- it is the fact that
44    /// survives the bound.
45    pub(super) fn record_unknown_tag(&mut self, name: &str, raw: &str) {
46        *self.unknown_tag_counts.entry(name.to_owned()).or_insert(0) += 1;
47        // The raws are a CENSUS, not a transcript, so the ring keeps the FIRST
48        // occurrences: the interesting fact is that a new tag appeared and what
49        // it looked like. `Recorder` keeps the newest because it is a transcript
50        // being replayed; this is the other kind of log and takes the other end.
51        if self.unknown_tags.len() < MAX_UNKNOWN_TAGS {
52            self.unknown_tags.push(UnknownTag {
53                name: name.to_owned(),
54                raw: raw.to_owned(),
55            });
56        }
57    }
58
59    /// How many times an unknown tag of this name has been seen.
60    ///
61    /// **Not bounded by [`MAX_UNKNOWN_TAGS`].** The ring keeps a census of raw
62    /// bytes and stops; this keeps counting, so a tag on every prompt never looks
63    /// as rare as a tag seen twice. Zero for a name never seen.
64    #[must_use]
65    pub fn unknown_tag_count(&self, name: &str) -> u64 {
66        self.unknown_tag_counts.get(name).copied().unwrap_or(0)
67    }
68}

1//! The player's own timers: each a name, the list it shows in, and the
2//! second it ends, by the game's clock, as a roundtime's end is.
3//!
4//! A Lich script sets one with `<vellumTimer id= value=>`, which reaches the
5//! state as the script's dialogs do; `.timer` and a highlight set one the
6//! same way. A value of 0 clears it. A timer run out is kept until it is
7//! set again or cleared: what shows it decides how long a finished one
8//! stays.
9

16/// The list a timer shows in: one of the game's effect lists, or the
17/// player's own.
18#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
19pub enum List {
20    /// Active Spells.
21    ActiveSpells,
22    /// Buffs.
23    Buffs,
24    /// Debuffs.
25    Debuffs,
26    /// Cooldowns.
27    Cooldowns,
28    /// The player's own, apart from the game's.
29    #[default]
30    Custom,
31}
32
33impl List {
34    /// Every list, the game's in their order, then the player's own.
35    pub const ALL: [List; 5] = [
36        List::ActiveSpells,
37        List::Buffs,
38        List::Debuffs,
39        List::Cooldowns,
40        List::Custom,
41    ];
42
43    /// The list `words` name, either case and either number: `buff`,
44    /// `Buffs`, `active spell`, `spell`, the game's own `Active Spells`;
45    /// none for words that name no list.
46    #[must_use]
47    pub fn named(words: &str) -> Option<List> {
48        let words = words.trim().to_ascii_lowercase();
49        let words = words.strip_suffix('s').unwrap_or(&words);
50        Some(match words {
51            "active spell" | "active_spell" | "spell" | "active" => List::ActiveSpells,
52            "buff" => List::Buffs,
53            "debuff" => List::Debuffs,
54            "cooldown" => List::Cooldowns,
55            "custom" | "own" => List::Custom,
56            _ => return None,
57        })
58    }
59
60    /// The game's dialog id for its effect list, as the effects are filed;
61    /// none for the player's own.
62    #[must_use]
63    pub const fn game_id(self) -> Option<&'static str> {
64        match self {
65            List::ActiveSpells => Some("Active Spells"),
66            List::Buffs => Some("Buffs"),
67            List::Debuffs => Some("Debuffs"),
68            List::Cooldowns => Some("Cooldowns"),
69            List::Custom => None,
70        }
71    }
72
73    /// What a player calls it.
74    #[must_use]
75    pub const fn name(self) -> &'static str {
76        match self {
77            List::ActiveSpells => "active spell",
78            List::Buffs => "buff",
79            List::Debuffs => "debuff",
80            List::Cooldowns => "cooldown",
81            List::Custom => "custom",
82        }
83    }
84}

86/// One timer: when it was set, when it ends, by the game's clock, and the
87/// list it shows in.
88#[derive(Clone, Copy, Debug, PartialEq, Eq)]
89pub struct Timer {
90    /// The second it was set, when the clock was known.
91    pub set_at: Option<u32>,
92    /// The second it ends.
93    pub ends: u32,
94    /// The list it shows in.
95    pub list: List,
96}
97
98/// The timers, by name.
99#[derive(Clone, Debug, Default, PartialEq, Eq)]
100pub struct CustomTimers {
101    by_name: BTreeMap<String, Timer>,
102}
103
104impl CustomTimers {
105    /// When the timer `name` ends, if it is set.
106    #[must_use]
107    pub fn get(&self, name: &str) -> Option<u32> {
108        self.by_name.get(name).map(|timer| timer.ends)
109    }
110
111    /// The timer `name`, if it is set.
112    #[must_use]
113    pub fn timer(&self, name: &str) -> Option<&Timer> {
114        self.by_name.get(name)
115    }
116
117    /// Every timer set, by name, with the second it ends.
118    pub fn iter(&self) -> impl Iterator<Item = (&str, u32)> {
119        self.by_name
120            .iter()
121            .map(|(name, timer)| (name.as_str(), timer.ends))
122    }
123
124    /// Every timer set, by name.
125    pub fn timers(&self) -> impl Iterator<Item = (&str, &Timer)> {
126        self.by_name
127            .iter()
128            .map(|(name, timer)| (name.as_str(), timer))
129    }
130
131    /// Fold `frame` in, when it sets or clears a timer, the game's second
132    /// being `now`. A list it does not name is the player's own.
133    pub fn read(&mut self, frame: &Frame, now: Option<u32>) {
134        if let Frame::Timer {
135            id,
136            value,
137            category,
138        } = frame
139        {
140            if *value == 0 {
141                self.by_name.remove(id);
142            } else {
143                let list = category
144                    .as_deref()
145                    .and_then(List::named)
146                    .unwrap_or_default();
147                self.by_name.insert(
148                    id.clone(),
149                    Timer {
150                        set_at: now,
151                        ends: *value,
152                        list,
153                    },
154                );
155            }
156        }
157    }
158}

160impl GameState {
161    /// The second a timer of the player's own started now ends, `seconds`
162    /// on: by the game's clock, or this machine's before the game has said.
163    #[must_use]
164    pub fn timer_ends_in(&self, seconds: u32) -> u32 {
165        let now = self.game_time_now().unwrap_or_else(|| {
166            let since = std::time::SystemTime::now()
167                .duration_since(std::time::UNIX_EPOCH)
168                .map_or(0, |since| since.as_secs());
169            u32::try_from(since).unwrap_or(u32::MAX)
170        });
171        now.saturating_add(seconds)
172    }
173}

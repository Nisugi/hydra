1//! Reading a command report out of a completed chunk.
2//!
3//! A chunk consumer: the prompt is the report's boundary (see
4//! `state/chunks.rs`), and the chunk holds parsed lines with `exist`, `noun`
5//! and bold already typed, so this file needs no pattern for markup at all.
6//!
7//! # No block state machine
8//!
9//! **The chunk is already the report's boundary**, so "which block is open"
10//! is answered by looking at the chunk's own lines rather than remembered
11//! across them. There is no buffer, no opener that sets an open block, no
12//! terminator that commits one.
13//!
14//! What remains is a pure function of one chunk, which is a better shape for
15//! the same reason a classifier is: same chunk in, same answer out, and nothing
16//! left dirty when a connection drops mid-report.
17
18use super::stats::{Identity, Stat, StatKind, StatLine};
19use crate::state::lines::Chunk;
20
21/// A complete `info` report read from one chunk.
22#[derive(Debug, Clone, PartialEq, Eq)]
23pub struct InfoReport {
24    /// The stat lines, in wire order, each with whether its enhanced column
25    /// arrived bolded.
26    pub stats: Vec<(StatKind, StatLine, bool)>,
27    /// Race, profession, gender and age.
28    pub identity: Identity,
29}
30
31impl InfoReport {
32    /// Read an `info` report out of a chunk, or `None` if it is not one.
33    ///
34    /// **The header is required.** `Name: ... Race: ... Profession: ...` is
35    /// `info`'s first line of prose and the report's opener. Without it, a
36    /// chunk that
37    /// merely contains a stat-shaped line -- a player typing one into a channel
38    /// -- is not a report. That is the rule `state.rs:306-309` records for the
39    /// idle warning: *"a player can say anything"*.
40    ///
41    /// **Nothing is returned for a chunk with no stat lines.** A caller must
42    /// not be handed an empty update to apply.
43    #[must_use]
44    pub fn read(chunk: &Chunk) -> Option<Self> {
45        let mut identity = None;
46        let mut stats = Vec::new();
47
48        for line in chunk.lines() {
49            if let Some(found) = classify_identity(&line.text()) {
50                // A second header in one chunk means two reports ran with no
51                // prompt between. The later one wins, as it would if they had
52                // arrived in separate chunks.
53                identity = Some(found);
54                stats.clear();
55                continue;
56            }
57            if identity.is_none() {
58                continue;
59            }
60            if let Some((stat, bolded)) =
61                StatLine::classify_with_bold(&line.text(), &line.bold_refs())
62            {
63                stats.push((stat.kind, stat, bolded));
64                continue;
65            }
66            if let (Some((gender, age)), Some(id)) =
67                (classify_gender_age(&line.text()), identity.as_mut())
68            {
69                id.gender = Some(gender);
70                id.age = Some(age);
71            }
72        }
73
74        let identity = identity?;
75        if stats.is_empty() {
76            return None;
77        }
78        Some(Self { stats, identity })
79    }
80
81    /// Fold one stat line into a [`Stat`], preserving columns the line did not
82    /// carry.
83    ///
84    /// **`normal` is only overwritten by a line that had it.** `info` sends two
85    /// columns and `info full` three, so a plain `info` run after an `info full`
86    /// must not erase the base values it never mentioned. That is the difference
87    /// between "unknown" and "unchanged".
88    #[must_use]
89    pub fn merge_into(line: &StatLine, bolded: bool, previous: Stat) -> Stat {
90        Stat {
91            normal: line.normal().or(previous.normal),
92            ascended: line.ascended().or(previous.ascended),
93            enhanced: line.enhanced().or(previous.enhanced),
94            // **Replaced, not `or`-ed.** Bold is a per-report observation: a
95            // stat that stopped arriving bolded stopped being enhanced, because
96            // the item was removed or paused. Carrying the old `true` forward
97            // would make an expired enhancive permanent.
98            enhanced_is_bolded: bolded,
99        }
100    }
101}

103/// Classify `info`'s opening line, which is also where race and profession live.
104///
105/// ```text
106/// Name: Ashryn Race: Half-Elf  Profession: Ranger (shown as: Hero)
107/// ```
108///
109/// # Three things deliberately not taken from this line
110///
111/// * **The name.** `<playerID>` and the `<a exist>` link are authoritative;
112///   this line is not.
113/// * **`(shown as: Hero)`.** A title, not a profession, so it is stripped.
114/// * **Anything at all, while Shroud of Deception is up.** The spell falsifies
115///   race, profession, gender and age, so they are not stored while spell
116///   1212 is active -- while the numbers, which the shroud does not touch,
117///   still are. **This function cannot see the
118///   effect list, so the caller enforces it**; see `Character::consume_chunk`.
119fn classify_identity(line: &str) -> Option<Identity> {
120    let rest = line.strip_prefix("Name: ")?;
121    let (_name, rest) = rest.split_once(" Race: ")?;
122    let (race, profession) = rest.split_once(" Profession: ")?;
123    let race = race.trim();
124    // `(shown as: X)` and `(not shown)` are display titles; the profession is
125    // whatever precedes the parenthesis.
126    let profession = profession
127        .split_once('(')
128        .map_or(profession, |(before, _)| before)
129        .trim();
130    if race.is_empty() || profession.is_empty() {
131        return None;
132    }
133    Some(Identity {
134        race: Some(race.to_owned()),
135        profession: Some(profession.to_owned()),
136        ..Identity::default()
137    })
138}

140/// Classify `info`'s second line: gender and age.
141///
142/// ```text
143/// Gender: Male    Age: 36    Expr: 43,904,921    Level:  100
144/// ```
145///
146/// **`Expr:` and `Level:` are read and discarded.** They change continuously
147/// and `info` is a snapshot;
148/// `<dialogData id='expr'>` carries the live values.
149fn classify_gender_age(line: &str) -> Option<(String, u32)> {
150    let rest = line.strip_prefix("Gender: ")?;
151    let (gender, rest) = rest.split_once(" Age: ")?;
152    let age = crate::state::numbers::grouped(rest.split_whitespace().next()?)?;
153    let gender = gender.trim();
154    if gender.is_empty() {
155        return None;
156    }
157    Some((gender.to_owned(), age))
158}

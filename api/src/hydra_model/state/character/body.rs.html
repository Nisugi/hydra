1//! Body parts, and the wound/scar accessors over them.
2//!
3//! Wounds and scars are the same shape read two ways: one type, and a
4//! [`Track`] to say which. The one real difference is which the game's
5//! injury mode shows ([`InjuryMode`]).
6//!
7//! # Sixteen parts, not fourteen
8//!
9//! The game's documentation lists fourteen body locations. The wire carries
10//! **sixteen**: it adds `leftFoot` and `rightFoot`, which the sneaking rule
11//! reads (`injured.rs`). The documentation is authoritative where it speaks
12//! and incomplete elsewhere.
13//!
14//! # The composites are not the obvious groupings
15//!
16//! Three of them, and two are surprising enough that the tests assert their
17//! membership rather than their results:
18//!
19//! | Composite | Parts | Note |
20//! |---|---|---|
21//! | `arms` | arms **and hands** | four parts, not two |
22//! | `limbs` | arms, hands, legs | **no feet**, though the wire has them |
23//! | `torso` | chest, abdomen, back **and both eyes** | eyes are not torso anatomically |
24//!
25//! The shapes carry meaning: a rank-3 `limbs` scar is a severed limb and a
26//! rank-3 eye scar a missing eye, which is why `limbs` excluding feet matters
27//! -- a severed foot is not a severed limb for that purpose.
28//!
29//! Each takes the **maximum**, not a sum. A character with four rank-1 wounds
30//! has `arms == 1`.
31

36/// Which injuries the game's injury window shows: its three radio buttons,
37/// `injrRad`, `scarRad` and `bothRad`, the one set with `value='1'`.
38/// VERIFIED in a month of captures: Both 140 times, Scars 29, Injuries
39/// never; the radios come in the window's `openDialog`, before any image.
40/// Both is the useful mode, since a wound covers a scar otherwise.
41///
42/// **What a mode hides, the model keeps.** In those captures a hand wounded
43/// and scarred shows `Injury1` under Both and `Scar1` under Scars, and a part
44/// only wounded shows whole under Scars. Read as if every image said
45/// everything, switching to Scars would erase the wounds, and back to Both
46/// the scars under them: wounds and scars would flash.
47#[derive(Debug, Clone, Copy, PartialEq, Eq)]
48pub enum InjuryMode {
49    /// `injrRad`: wounds.
50    Wounds,
51    /// `scarRad`: scars.
52    Scars,
53    /// `bothRad`: both, a wound over a scar.
54    Both,
55}
56
57impl InjuryMode {
58    /// `wounds`, `scars` or `both`.
59    #[must_use]
60    pub const fn as_str(self) -> &'static str {
61        match self {
62            Self::Wounds => "wounds",
63            Self::Scars => "scars",
64            Self::Both => "both",
65        }
66    }
67}

69/// Read the injury window's radio buttons: the one set names the mode.
70pub(crate) fn read_radios(
71    character: &mut super::Character,
72    widgets: &hydra_protocol::frame::DialogWidgets,
73) {
74    for attrs in &widgets.widgets {
75        let get = |name: &str| {
76            attrs
77                .iter()
78                .find(|(key, _)| key == name)
79                .map(|(_, value)| value.as_str())
80        };
81        let mode = match get("id") {
82            Some("injrRad") => InjuryMode::Wounds,
83            Some("scarRad") => InjuryMode::Scars,
84            Some("bothRad") => InjuryMode::Both,
85            _ => continue,
86        };
87        if get("value") == Some("1") {
88            character.injury_mode = Some(mode);
89        }
90    }
91}

93/// Fold one injury image into `character`, as its injury window's mode
94/// allows ([`InjuryMode`]), and note the part as observed. An unrecognized
95/// image clears the part, as it always has, but cannot prove it healthy.
96pub(super) fn apply_image(character: &mut super::Character, part: &str, name: &str) {
97    if let Some(index) = ALL_PARTS.iter().position(|id| *id == part) {
98        if name == part
99            || matches!(
100                name,
101                "Injury1"
102                    | "Injury2"
103                    | "Injury3"
104                    | "Scar1"
105                    | "Scar2"
106                    | "Scar3"
107                    | "Nsys1"
108                    | "Nsys2"
109                    | "Nsys3"
110            )
111        {
112            character.observed_body_parts |= 1 << index;
113        } else {
114            character.observed_body_parts &= !(1 << index);
115        }
116    }
117    let rank = |prefix: &str| -> Option<u8> { name.strip_prefix(prefix)?.parse().ok() };
118    if part == "nsys" {
119        // The nerves' image says one rank, wound or scar as the window's
120        // mode has it. Under Scars it says nothing of the wound: a script
121        // that switches the window to Scars and back every move would
122        // otherwise heal the nerves and hurt them again each time. Under
123        // Wounds a whole image says the wound is healed, and nothing of a
124        // scar.
125        let nerve = match character.injury_mode {
126            Some(InjuryMode::Scars) => None,
127            Some(InjuryMode::Wounds) if name == part => {
128                character.nerves_wound_healed();
129                None
130            }
131            _ if name == part => Some(0),
132            _ => rank("Nsys"),
133        };
134        if let Some(nerve) = nerve {
135            character.apply_nerve_rank(nerve);
136        }
137        if name == part || name.starts_with("Nsys") {
138            return;
139        }
140    }
141    let known = character.injuries.get(part).copied().unwrap_or_default();
142    let scars_only = character.injury_mode == Some(InjuryMode::Scars);
143    let injury = if let Some(wound) = rank("Injury") {
144        // A wound covers a scar; it is not evidence the old scar healed.
145        Injury {
146            wound,
147            scar: known.scar,
148        }
149    } else if let Some(scar) = rank("Scar") {
150        // Shown both ways, a scar image means no wound remains over it;
151        // showing scars only, it says nothing about the wound.
152        Injury {
153            wound: if scars_only { known.wound } else { 0 },
154            scar,
155        }
156    } else {
157        // Whole, in what the window shows: under Scars no scar, under
158        // Wounds no wound, the other track kept; shown both ways, whole.
159        match character.injury_mode {
160            Some(InjuryMode::Scars) => Injury {
161                wound: known.wound,
162                scar: 0,
163            },
164            Some(InjuryMode::Wounds) => Injury {
165                wound: 0,
166                scar: known.scar,
167            },
168            Some(InjuryMode::Both) | None => Injury::default(),
169        }
170    };
171    if injury.is_hurt() {
172        character.injuries.insert(part.to_owned(), injury);
173    } else {
174        character.injuries.remove(part);
175    }
176}

201/// `arms`: both arms **and both hands**.
202pub const ARMS: [&str; 4] = ["leftArm", "rightArm", "leftHand", "rightHand"];

204/// `limbs`: arms, hands and legs -- **not feet**.
205///
206/// The omission is load-bearing rather than an oversight to correct: a
207/// rank-3 `limbs` scar is a severed limb ([`Body::has_severed_limb`]), and a

218/// `torso`: chest, abdomen, back **and both eyes**.
219///
220/// The eyes are not anatomically torso, but the herbs group them the same
221/// way -- `'minor organ wound'` covers the same set -- so the grouping is the
222/// game's.
223pub const TORSO: [&str; 5] = ["rightEye", "leftEye", "chest", "abdomen", "back"];

225/// Which of an [`Injury`]'s two tracks to read.
226///
227/// The one thing wounds and scars differ in.
228#[derive(Debug, Clone, Copy, PartialEq, Eq)]
229pub enum Track {
230    /// Fresh injuries. Heal to nothing, or to a scar.
231    Wound,
232    /// What a partially-healed wound left behind.
233    ///
234    /// **Only visible once no wound covers it** in the game's default injury
235    /// mode, which is what `Scars.all_scars` swaps modes to defeat. The model
236    /// retains the last scar it was shown (`character.rs`'s
237    /// `apply_injury_image`), so this reports what is known rather than what
238    /// is currently displayed.
239    Scar,
240}

242/// Wound and scar ranks by body part.
243///
244/// A thin reader over [`Character::injuries`](super::Character::injuries),
245/// so it holds a borrow rather than a copy.
246#[derive(Debug, Clone, Copy)]
247pub struct Body<'a> {
248    parts: &'a BTreeMap<String, Injury>,
249}
250
251impl<'a> Body<'a> {
252    /// Read over one character's injuries.
253    #[must_use]
254    pub const fn new(parts: &'a BTreeMap<String, Injury>) -> Self {
255        Self { parts }
256    }
257
258    /// One part's rank on one track.
259    ///
260    /// **Zero for a part that is whole**, and zero for a part the wire has
261    /// never mentioned. Those are the same thing here: the dialog sends every
262    /// part it knows about, and `apply_injury_image` removes a part that heals
263    /// rather than storing a zero, so absence IS "no injury".
264    #[must_use]
265    pub fn rank(&self, part: &str, track: Track) -> u8 {
266        let injury = self.parts.get(part).copied().unwrap_or_default();
267        match track {
268            Track::Wound => injury.wound,
269            Track::Scar => injury.scar,
270        }
271    }
272
273    /// The worst rank across a group, as the composites do.
274    ///
275    /// **Maximum, not sum.** Four rank-1 wounds give 1.
276    #[must_use]
277    pub fn worst(&self, parts: &[&str], track: Track) -> u8 {
278        parts.iter().map(|p| self.rank(p, track)).max().unwrap_or(0)
279    }
280
281    /// `arms`: the worst of both arms and both hands.
282    #[must_use]
283    pub fn arms(&self, track: Track) -> u8 {
284        self.worst(&ARMS, track)
285    }
286
287    /// `limbs`: the worst of arms, hands and legs. **Feet excluded.**
288    #[must_use]
289    pub fn limbs(&self, track: Track) -> u8 {
290        self.worst(&LIMBS, track)
291    }
292
293    /// `torso`: the worst of chest, abdomen, back and both eyes.
294    #[must_use]
295    pub fn torso(&self, track: Track) -> u8 {
296        self.worst(&TORSO, track)
297    }
298
299    /// Every part's rank on one track, including the whole ones.
300    ///
301    /// Unlike
302    /// [`Character::injuries`](super::Character::injuries) -- which holds only
303    /// hurt parts, so `is_empty` answers "unhurt" without a scan -- this lists
304    /// all sixteen, because a caller asking for "all wounds" wants a complete
305    /// picture rather than a sparse one.
306    #[must_use]
307    pub fn all(&self, track: Track) -> BTreeMap<&'static str, u8> {
308        ALL_PARTS
309            .into_iter()
310            .map(|part| (part, self.rank(part, track)))
311            .collect()
312    }
313
314    /// Is anything hurt on this track?
315    #[must_use]
316    pub fn any(&self, track: Track) -> bool {
317        ALL_PARTS.iter().any(|p| self.rank(p, track) > 0)
318    }
319
320    /// **A severed limb.**
321    ///
322    /// A rank-3 limb SCAR, which the game describes as *"a missing right
323    /// arm"*. Named here because the herb that treats it is its own type
324    /// (`'severed limb'`) and a caller should not have to know the magic
325    /// number.
326    #[must_use]
327    pub fn has_severed_limb(&self) -> bool {
328        self.limbs(Track::Scar) == 3
329    }
330
331    /// **A missing eye.**
332    ///
333    /// A rank-3 eye scar -- *"a missing right eye"*. Note this reads the eyes
334    /// directly rather than through [`Self::torso`], which would also match a
335    /// rank-3 chest scar.
336    #[must_use]
337    pub fn has_missing_eye(&self) -> bool {
338        self.rank("leftEye", Track::Scar) == 3 || self.rank("rightEye", Track::Scar) == 3
339    }
340}

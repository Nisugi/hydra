1//! Silver, notes, and the eleven event currencies.
2//!
3//! # All single lines, all from ordinary play
4//!
5//! Like `standing.rs` and for the same reason: each line stands alone and most
6//! arrive without a command. `wealth` states your silver; a festival merchant
7//! tells you your bloodscrip while you shop.
8//!
9//! # Why so many currencies
10//!
11//! The game's paid events each mint their own scrip, and they accumulate: a
12//! character who has attended Duskruin, Reim, Ebon Gate and Rumor Woods carries
13//! four balances that do not convert. MEASURED against one character's
14//! store, which holds five at once:
15//!
16//! ```text
17//! currency.silver_total : 81
18//! currency.tickets : 17
19//! currency.bloodscrip : 1024
20//! currency.ethereal_scrip : 7117
21//! currency.soul_shards : 9963
22//! currency.gemstone_dust : 7
23//! ```
24//!
25//! So this is a named field per currency rather than a map: a closed
26//! vocabulary gets typed fields, and this vocabulary is closed by what
27//! Simutronics has run. A new event adds a field, which is a visible change.
28

31/// Every balance a character carries.
32///
33/// All `Option`: a character who has never been to Duskruin has **unknown**
34/// bloodscrip, not zero. That distinction is why a consumer can tell
35/// "you have none" from "nobody has said", and why a display can omit a
36/// currency the player has never earned rather than showing a row of zeroes.
37#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
38pub struct Currency {
39    /// Silver on your person, from `wealth`.
40    pub silver: Option<u64>,
41    /// Silver in a container you are carrying.
42    pub silver_container: Option<u64>,
43    /// The whole: `You are carrying a total of N silver.`, or `N silver
44    /// between notes and coins.`
45    pub silver_total: Option<u64>,
46    /// Promissory notes, not coins: `Total note value: N`, or `wealth`'s
47    /// `<town> promissory notes valued at a total of N silver.`
48    pub notes: Option<u64>,
49    /// Voln favor. Signed: the wire can send a negative.
50    pub voln_favor: Option<i64>,
51    /// Simucoin general tickets.
52    pub tickets: Option<u64>,
53    /// Gold ("Gold - N gold.").
54    pub gold: Option<u64>,
55    /// Troubled Waters.
56    pub blackscrip: Option<u64>,
57    /// Duskruin Arena.
58    pub bloodscrip: Option<u64>,
59    /// Reim.
60    pub ethereal_scrip: Option<u64>,
61    /// Ebon Gate.
62    pub soul_shards: Option<u64>,
63    /// Rumor Woods.
64    pub raikhen: Option<u64>,
65    /// Inquisitor.
66    pub aevit: Option<u64>,
67    /// Gigas artifact fragments.
68    pub gigas_artifact_fragments: Option<u64>,
69    /// Redsteel marks.
70    pub redsteel_marks: Option<u64>,
71    /// `Dust`, held "in your reserves".
72    ///
73    /// **Named for the item, not the game.** `gemstone_dust` would name the
74    /// GAME rather than what you are carrying -- and the game-name scanner
75    /// flags that spelling in a shared module, correctly: the wire says only `N
76    /// Dust in your reserves`, so
77    /// `dust` is both the accurate name and the one that passes.
78    pub dust: Option<u64>,
79}
80
81impl Currency {
82    /// Take whatever balances a chunk's lines state.
83    ///
84    /// Returns whether anything moved, so a caller marks a group dirty only
85    /// when there is something to write. Silver changes on every purchase, so
86    /// a `wealth` that restates it must not schedule a save.
87    pub fn absorb_chunk(&mut self, chunk: &Chunk) -> bool {
88        let mut changed = false;
89        for line in chunk.lines() {
90            changed |= self.absorb(&line.text());
91        }
92        changed
93    }
94
95    /// One line.
96    fn absorb(&mut self, line: &str) -> bool {
97        let before = *self;
98
99        // `You have no silver with you.` / `but one` / `N`.
100        //
101        // The word forms are the game's, and they mean 0 and 1. Reading them
102        // as numbers would silently store
103        // nothing for a character who is broke, which reads the same as never
104        // having asked.
105        if let Some(rest) = line.strip_prefix("You have ")
106            && let Some(amount) = rest.strip_suffix(" silver with you.")
107        {
108            self.silver = match amount {
109                "no" => Some(0),
110                "but one" => Some(1),
111                other => number(other),
112            };
113        }
114
115        if let Some(rest) = line.strip_prefix("You are carrying ") {
116            if let Some((amount, _)) = rest.split_once(" silver stored within your ") {
117                self.silver_container = number(amount);
118            } else if let Some(amount) = rest.strip_suffix(" gigas artifact fragments.") {
119                self.gigas_artifact_fragments = number(amount);
120            } else if let Some(amount) = rest.strip_suffix(" gigas artifact fragment.") {
121                self.gigas_artifact_fragments = number(amount);
122            } else if let Some(amount) = rest.strip_suffix(" Dust in your reserves.") {
123                self.dust = number(amount);
124            } else if let Some(amount) = rest.strip_suffix(" Dust in your reserve.") {
125                self.dust = number(amount);
126            } else if let Some(amount) = rest
127                .strip_suffix(" redsteel marks.")
128                .or_else(|| rest.strip_suffix(" redsteel mark."))
129            {
130                self.redsteel_marks = number(amount);
131            } else if let Some(amount) = rest
132                .strip_prefix("a total of ")
133                .and_then(|r| r.strip_suffix(" silver."))
134            {
135                self.silver_total = number(amount);
136            } else if let Some(amount) = rest.strip_suffix(" silver between notes and coins.") {
137                // The whole, notes and coins.
138                self.silver_total = number(amount);
139            } else if let Some((_, amount)) = rest
140                .split_once(" promissory notes valued at a total of ")
141                .and_then(|(town, amount)| Some((town, amount.strip_suffix(" silver.")?)))
142            {
143                // `<town> promissory notes`: the whole is the coins and these,
144                // which a report with no coins does not state on a line of
145                // its own.
146                self.notes = number(amount);
147                if let (Some(coins), Some(notes)) = (self.silver, self.notes) {
148                    self.silver_total = Some(coins + notes);
149                }
150            }
151        }
152
153        if let Some(amount) = line.strip_prefix("Total note value: ") {
154            self.notes = number(amount);
155        }
156        if let Some(amount) = line.strip_prefix("Voln Favor: ") {
157            self.voln_favor = signed(amount);
158        }
159        // The indented `Redsteel Marks:` form, the same line in another
160        // layout.
161        if let Some(amount) = line.trim_start().strip_prefix("Redsteel Marks:") {
162            self.redsteel_marks = number(amount);
163        }
164
165        self.absorb_ticket(line.trim_start());
166        before != *self
167    }
168
169    /// The `<Event> - N <currency>.` lines a ticket balance arrives on.
170    fn absorb_ticket(&mut self, line: &str) {
171        const TICKETS: [(&str, &str); 8] = [
172            ("General - ", " tickets."),
173            ("Gold - ", " gold."),
174            ("Troubled Waters - ", " blackscrip."),
175            ("Duskruin Arena - ", " bloodscrip."),
176            ("Reim - ", " ethereal scrip."),
177            ("Ebon Gate - ", " soul shards."),
178            ("Rumor Woods - ", " raikhen."),
179            ("Inquisitor - ", " aevit."),
180        ];
181        for (index, (prefix, suffix)) in TICKETS.into_iter().enumerate() {
182            let Some(rest) = line.strip_prefix(prefix) else {
183                continue;
184            };
185            // The singular, because the game says "1 ticket." not "1 tickets."
186            let singular = suffix.trim_end_matches('.').trim_end_matches('s');
187            let Some(amount) = rest
188                .strip_suffix(suffix)
189                .or_else(|| rest.strip_suffix(&format!("{singular}.")))
190            else {
191                continue;
192            };
193            let value = number(amount);
194            match index {
195                0 => self.tickets = value,
196                1 => self.gold = value,
197                2 => self.blackscrip = value,
198                3 => self.bloodscrip = value,
199                4 => self.ethereal_scrip = value,
200                5 => self.soul_shards = value,
201                6 => self.raikhen = value,
202                _ => self.aevit = value,
203            }
204            return;
205        }
206    }
207}

209/// `"1,024"` -> `1024`. The shared, strict reader: see `state/numbers.rs` for
210/// why `"12a3"` is `None` rather than `123`.
211fn number(text: &str) -> Option<u64> {
212    crate::state::numbers::grouped(text)
213}

215/// Same, keeping a leading `-`.
216fn signed(text: &str) -> Option<i64> {

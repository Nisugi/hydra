1//! Society, citizenship, warcries and resources: the facts one line teaches.
2//!
3//! # Why these are together and separate from `blocks.rs`
4//!
5//! `blocks.rs` reads a command's *report* out of a completed chunk -- `info`
6//! is fifteen lines and means nothing until they are read together. Everything
7//! here is a **single line that stands alone**, and most of them arrive during
8//! ordinary play rather than during a sync:
9//!
10//! ```text
11//! The Grandmaster says, "Welcome to the Order of Voln."
12//! You have now achieved rank 5 of Shield Bash, costing 20 Combat points.
13//! ```
14//!
15//! That is the difference that matters for the store. A sync BOOTSTRAPS a
16//! character the client has never seen, because a gameplay message only tells
17//! you about a *change* and a character who trained before the client existed
18//! never emits those lines again. After that, these keep the store current
19//! incrementally, and a resync is a repair tool rather than the source.
20//!
21//! # Classifiers
22//!
23//! Each is a free function, `&str` in, typed `Option` out, remembering
24//! nothing. A line that is not one of these returns `None` and the caller is
25//! unchanged -- so a player quoting "Welcome to the Order" in a channel
26//! changes no state, because the quote arrives inside a speech frame and
27//! never reaches this as a bare line.
28

33/// Society membership, citizenship, warcries and resources.
34///
35/// Everything [`society_line`] and its siblings teach, stored. Grouped in one
36/// struct rather than five fields on [`Character`](super::Character) because
37/// they share a refresh: `Group::Standing` is one line of a sync and one entry
38/// in the dirty set.
39///
40/// **Every field is `Option` or empty-by-default**: a character the client
41/// has never synced has not "no society", it has an unknown one, and the two
42/// must not read the same. A `'None'` string sentinel for no society would
43/// blur them, which is why the society is an `Option<Society>`.
44#[derive(Debug, Clone, Default, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
45pub struct Standing {
46    /// The society, once stated. `Some(None)` is "stated: none".
47    pub society: Option<Option<Society>>,
48    /// The rank within it.
49    ///
50    /// For a Master the wire sends no number and this holds
51    /// [`Society::max_rank`], applied by [`Self::apply_society`] -- VERIFIED
52    /// against a live character, whose `society` output says only
53    /// "Master of the Guardians of Sunfist" while the store holds rank 20.
54    pub society_rank: Option<u8>,
55    /// The citizenship town. `Some(None)` is "stated: none".
56    pub citizenship: Option<Option<String>>,
57    /// Whether the urchin guides serve this character, once `urchin status`
58    /// has said ([`urchin_line`]). Kept, so travel asks only while this is
59    /// unknown.
60    #[serde(default)]
61    pub urchin_access: Option<bool>,
62    /// When that access ends, by the game's clock, when the game gave a
63    /// date ([`urchin_until`]). Read through [`Self::urchins`].
64    #[serde(default)]
65    pub urchin_until: Option<u32>,
66    /// The warcries known, in [`Warcry::ALL`] order.
67    ///
68    /// **A set, not a count.** Recording presence and absence under different
69    /// keys would let a warcry, once learned, never be un-learned. A set cannot
70    /// have that failure: [`Self::set_warcries`] replaces it wholesale.
71    pub warcries: BTreeSet<Warcry>,
72    /// Which resource this profession earns, from a `Suffused` line.
73    pub resource_type: Option<ResourceType>,
74    /// Weekly and total earned.
75    pub resources: Option<ResourceAmounts>,
76    /// The amount currently suffused.
77    pub suffused: Option<u32>,
78    /// Covert Arts charges, out of 200. Signed: the wire can send a negative.
79    pub covert_arts_charges: Option<i32>,
80    /// Accumulated shadow essence, 0 to 5 (`character/shadow.rs`).
81    pub shadow_essence: Option<u8>,
82}

84impl Standing {
85    /// Whether the urchin guides serve this character at game second `now`.
86    /// `None`: nobody has said, or the date the game gave has passed, and
87    /// it is to be asked again.
88    #[must_use]
89    pub fn urchins(&self, now: Option<u32>) -> Option<bool> {
90        let lapsed = self.urchin_access == Some(true)
91            && self
92                .urchin_until
93                .zip(now)
94                .is_some_and(|(until, now)| now >= until);
95        self.urchin_access.filter(|_| !lapsed)
96    }
97
98    /// Apply one society line's meaning.
99    ///
100    /// Returns whether anything changed, so a caller can mark a group dirty
101    /// only when there is something to write.
102    ///
103    /// # The arithmetic the line does not carry
104    ///
105    /// A Master's line has no rank, so [`Society::max_rank`] fills it. A step
106    /// line says only that one happened, so the rank rises by one -- **and an
107    /// unknown rank stays unknown**, rather than becoming 1: adding one to an
108    /// absent rank would turn "we never asked" into "rank 1".
109    pub fn apply_society(&mut self, event: SocietyEvent) -> bool {
110        let before = (self.society, self.society_rank);
111        match event {
112            SocietyEvent::Report {
113                society,
114                rank,
115                master,
116            } => {
117                self.society = Some(society);
118                self.society_rank = match (society, rank, master) {
119                    (Some(s), None, true) => Some(s.max_rank()),
120                    (Some(_), rank, _) => rank,
121                    (None, _, _) => Some(0),
122                };
123            }
124            SocietyEvent::Joined(society) => {
125                self.society = Some(Some(society));
126                // Voln and the Council start at step 1; Sunfist at rank 0.
127                self.society_rank = Some(u8::from(society != Society::GuardiansOfSunfist));
128            }
129            SocietyEvent::Resigned => {
130                self.society = Some(None);
131                self.society_rank = Some(0);
132            }
133            SocietyEvent::Stepped => {
134                // Unknown stays unknown. Saturating because a step past a
135                // society's maximum is not a thing the game does, and
136                // wrapping to 0 would be worse than standing still.
137                self.society_rank = self.society_rank.map(|r| r.saturating_add(1));
138            }
139        }
140        before != (self.society, self.society_rank)
141    }
142
143    /// Replace the known warcries wholesale.
144    ///
145    /// Wholesale because a `warcry` report states the complete set, and
146    /// merging would keep a warcry the character no longer has.
147    pub fn set_warcries(&mut self, warcries: BTreeSet<Warcry>) -> bool {
148        let changed = self.warcries != warcries;
149        self.warcries = warcries;
150        changed
151    }
152}

154/// What a society line stated.
155///
156/// **Four shapes, not one.** They answer different questions: a report states
157/// the standing, a join/resign/step announces a change. Collapsing them would
158/// lose the difference between "you are rank 5" and "you just became rank 5",
159/// and only the second is a reason to write anything down.
160#[derive(Debug, Clone, Copy, PartialEq, Eq)]
161pub enum SocietyEvent {
162    /// The `society` command's report: membership and rank as they stand.
163    Report {
164        /// Which society, or `None` for "not a member of any society".
165        society: Option<Society>,
166        /// The rank, or `None` when the line is a Master's.
167        ///
168        /// **Absent is not zero.** A Master's line carries no number; the rank
169        /// comes from
170        /// [`Society::max_rank`] instead, which the caller applies because it
171        /// is knowledge about the game rather than about the line.
172        rank: Option<u8>,
173        /// Whether the line said "Master" rather than "member".
174        master: bool,
175    },
176    /// Just joined. The rank that follows is the society's first step.
177    Joined(Society),
178    /// Just resigned. Every society fact is now false.
179    Resigned,
180    /// Advanced one step. The line does not say to what.
181    ///
182    /// **The new rank is the old one plus one**, which is the caller's
183    /// arithmetic and not this line's -- and an unknown rank stays unknown
184    /// instead of becoming 1.
185    Stepped,
186}

188/// Read a society line.
189///
190/// The three join welcomes are matched directly, each to its society. The
191/// Council of Light's welcome names the Lodge, not the Council, so scanning a
192/// welcome for the society's name would miss it.
193#[must_use]
194pub fn society_line(line: &str) -> Option<SocietyEvent> {
195    let trimmed = line.trim_start();
196
197    // The REPORT lines are indented; the announcements are not. Requiring the
198    // indent is load-bearing rather than cosmetic -- VERIFIED against a live
199    // `society` output, whose report line is indented under a `<popBold/>`:
200    //
201    // ```text
202    // <popBold/>   You are a Master of the Guardians of Sunfist.
203    // ```
204    //
205    // Without the indent requirement, a player typing the same sentence into a
206    // channel would restate the character's society. With it, they cannot: a
207    // speech frame carries no leading run of spaces.
208    let indented = line.len() != trimmed.len();
209
210    if indented && trimmed.starts_with("You are not a member of any society at this time") {
211        return Some(SocietyEvent::Report {
212            society: None,
213            rank: None,
214            master: false,
215        });
216    }
217
218    if indented
219        && let Some((rest, master)) = trimmed
220            .strip_prefix("You are a Master ")
221            .map(|rest| (rest, true))
222            .or_else(|| {
223                trimmed
224                    .strip_prefix("You are a member ")
225                    .map(|r| (r, false))
226            })
227    {
228        return society_report(rest, master);
229    }
230
231    // A join. There are three welcomes; the Lodge one belongs to the
232    // Council of Light, whose Grand Poohbah is its own greeter.
233    for (prefix, society) in [
234        (
235            "The Grandmaster says, \"Welcome to the Order",
236            Society::OrderOfVoln,
237        ),
238        (
239            "The Grandmaster says, \"You are now a member of the Guardians of Sunfist",
240            Society::GuardiansOfSunfist,
241        ),
242        (
243            "The Grand Poohbah smiles broadly.  \"Welcome to the Lodge,\" he cries",
244            Society::CouncilOfLight,
245        ),
246    ] {
247        if trimmed.starts_with(prefix) {
248            return Some(SocietyEvent::Joined(society));
249        }
250    }
251
252    if is_resignation(trimmed) {
253        return Some(SocietyEvent::Resigned);
254    }
255
256    if is_step(trimmed) {
257        return Some(SocietyEvent::Stepped);
258    }
259
260    None
261}

263/// The tail of "You are a {Master,member} ..." -- `in the X at rank N.`
264fn society_report(rest: &str, master: bool) -> Option<SocietyEvent> {
265    let rest = rest
266        .strip_prefix("in the ")
267        .or_else(|| rest.strip_prefix("of the "))?;
268    let (name, tail) = match rest.find(" at ") {
269        Some(at) => (&rest[..at], &rest[at..]),
270        None => (rest.trim_end_matches('.'), ""),
271    };
272    let society = Society::parse(name)?;
273    // "at rank 5." or "at step 5." -- both spellings are read.
274    let rank = tail
275        .strip_prefix(" at rank ")
276        .or_else(|| tail.strip_prefix(" at step "))
277        .and_then(|n| n.trim_end_matches('.').parse().ok());
278    Some(SocietyEvent::Report {
279        society: Some(society),
280        rank,
281        master,
282    })
283}

285/// The three resignation lines.
286fn is_resignation(line: &str) -> bool {
287    line.starts_with(
288        "The Grandmaster says, \"I'm sorry to hear that.  You are no longer in our service.",
289    ) || line.starts_with("The Poohbah looks at you sternly.  \"I had high hopes for you,\"")
290        || (line.starts_with("The Grandmaster says, \"I'm sorry to hear that,")
291            && line.contains("I wish you well with any of your future endeavors."))
292}

294/// The three advancement lines.
295fn is_step(line: &str) -> bool {
296    const SIGIL_TRACERS: [&str; 9] = [
297        "Zarak", "Faylanna", "Draelox", "Marl", "Vindar", "Taryn", "Meaha", "Oxanna", "Cyndelle",
298    ];
299    if line.starts_with("The High Taskmaster looks at you, consults ")
300        || line.starts_with("The monk concludes ceremoniously,")
301    {
302        return true;
303    }
304    SIGIL_TRACERS.iter().any(|who| {
305        line.strip_prefix(who).is_some_and(|rest| {
306            rest.starts_with(" traces the outline of a sigil into the air before you and says")
307        })
308    })
309}

311/// Read a citizenship line: the town, or `None` for no citizenship.
312///
313/// Returns `Some(None)` for "you don't seem to have
314/// citizenship" and `Some(Some(town))` for a town -- the outer `Option` is
315/// "was this a citizenship line at all".
316///
317/// The town stays a `String`: inventing a list of towns would refuse a town
318/// the game adds tomorrow.
319#[must_use]
320pub fn citizenship_line(line: &str) -> Option<Option<String>> {
321    if line.starts_with("You don't seem to have citizenship.") {
322        return Some(None);
323    }
324    let rest = line.strip_prefix("You currently have ")?;
325    let town = rest.split_once(" citizenship in ")?.1.strip_suffix('.')?;
326    Some(Some(town.to_owned()))
327}

329/// Read `urchin status`'s answer: access until a date or
330/// for good is access; the game does not say "until" of a day that has
331/// passed. `None`: not one of the three lines.
332#[must_use]
333pub fn urchin_line(line: &str) -> Option<bool> {
334    if line.starts_with("You will have access to the urchin guides")
335        || line.starts_with("You have permanent access to the urchin guides")
336    {
337        Some(true)
338    } else if line.starts_with("You currently have no access to the urchin guides") {
339        Some(false)
340    } else {
341        None
342    }
343}

345/// The date in `You will have access to the urchin guides until 1/18/2038
346/// 21:14:07 CST.`, as the game's clock counts: seconds since 1970. The game
347/// writes its own time, US Central: `CST` is six hours behind, `CDT` five.
348/// `None`: not that line, or a date that does not read.
349#[must_use]
350pub fn urchin_until(line: &str) -> Option<u32> {
351    let rest = line.strip_prefix("You will have access to the urchin guides until ")?;
352    let mut words = rest.trim_end_matches('.').split_whitespace();
353    let mut date = words.next()?.split('/').map(str::parse::<i64>);
354    let (month, day, year) = (date.next()?.ok()?, date.next()?.ok()?, date.next()?.ok()?);
355    let mut time = words.next()?.split(':').map(str::parse::<i64>);
356    let (hour, minute, second) = (time.next()?.ok()?, time.next()?.ok()?, time.next()?.ok()?);
357    let behind = if words.next() == Some("CDT") { 5 } else { 6 };
358    if !(1..=12).contains(&month) || !(1..=31).contains(&day) {
359        return None;
360    }
361    // Days since 1970-01-01 of a civil date (Howard Hinnant's algorithm).
362    let year = if month <= 2 { year - 1 } else { year };
363    let era = year.div_euclid(400);
364    let of_era = year.rem_euclid(400);
365    let of_year = (153 * ((month + 9) % 12) + 2) / 5 + day - 1;
366    let days = era * 146_097 + of_era * 365 + of_era / 4 - of_era / 100 + of_year - 719_468;
367    u32::try_from(days * 86_400 + (hour + behind) * 3_600 + minute * 60 + second).ok()
368}

370/// Read one `AFFILIATIONS` line from `profile`.
371///
372/// **`profile` words these differently from the reports**, which is why
373/// [`society_line`] and [`citizenship_line`] cannot read them and this exists:
374///
375/// | report | `profile` |
376/// |---|---|
377/// | `   You are a Master of the Guardians of Sunfist.` | `Master of the Guardians of Sunfist` |
378/// | `You currently have full citizenship in Kraken's Fall.` | `Full citizen of Kraken's Fall` |
379///
380/// MEASURED against `fixtures/character_profile.xml`. Found by a test that
381/// asserted the profile's society reached `Standing` and got `None`: the first
382/// version of the profile reader assumed the report readers would take these,
383/// on no evidence.
384///
385/// Returns nothing for the lines `Standing` has no field for -- `Follower of
386/// Zelia`, `Attuned to the Element of Earth`, `Member of House of Paupers`.
387/// Those stay in [`Profile::affiliations`](super::profile::Profile).
388#[must_use]
389pub fn profile_affiliation(line: &str) -> Option<Affiliation> {
390    let text = line.trim();
391    // A society, `Master of` or `Member of`. The rank is never on this line,
392    // so a Master's is `max_rank` and a member's is unknown -- NOT 1, for the
393    // reason `apply_society` gives.
394    for (prefix, master) in [("Master of the ", true), ("Member of the ", false)] {
395        if let Some(name) = text.strip_prefix(prefix)
396            && let Some(society) = Society::parse(name)
397        {
398            return Some(Affiliation::Society(SocietyEvent::Report {
399                society: Some(society),
400                rank: master.then(|| society.max_rank()),
401                master,
402            }));
403        }
404    }
405    // Citizenship. `Full citizen of X`, and the game's other wordings for a
406    // partial one are unmeasured -- so only the form in the capture is read,
407    // and an unfamiliar one falls through to the kept lines rather than being
408    // guessed at.
409    if let Some(town) = text.strip_prefix("Full citizen of ") {
410        return Some(Affiliation::Citizenship(town.to_owned()));
411    }
412    None
413}

426/// Read a warcry line from the `warcry` report.
427///
428/// `Some(None)` is the "you are not a Warrior Guild member"
429/// line, which states that the character has **none** -- distinct from having
430/// simply not been asked.
431#[must_use]
432pub fn warcry_line(line: &str) -> Option<Option<Warcry>> {
433    if line.starts_with("You must be an active member of the Warrior Guild to use this skill.") {
434        return Some(None);
435    }
436    // The report indents each name and carries nothing else on the line.
437    let name = line.trim();
438    if name.len() == line.trim_end().len() && !line.starts_with(char::is_whitespace) {
439        return None;
440    }
441    Warcry::parse(name).map(Some)
442}

444/// One PSM rank change, from ordinary play rather than from a table.
445///
446/// Five line shapes, one meaning: a PSM, a category and the rank it now
447/// holds. Unlearning and losing a technique differ only in what the new rank
448/// is, so that arithmetic is done here and the caller stores a number.
449#[derive(Debug, Clone, PartialEq, Eq)]
450pub struct PsmChange {
451    /// The PSM's display name, e.g. `"Shield Bash"`.
452    pub name: String,
453    /// The category word the line used, e.g. `"Combat"` or `"Shield"`.
454    ///
455    /// **Not parsed into a `PsmCategory` here.** The
456    /// line's word is the game's, and mapping it is the caller's business --
457    /// a classifier that refused an unrecognised category would silently drop
458    /// a PSM the game added.
459    pub category: String,
460    /// The rank the PSM now holds. Zero means no longer trained.
461    pub rank: u16,
462}
463
464/// Read a PSM rank change.
465#[must_use]
466pub fn psm_change(line: &str) -> Option<PsmChange> {
467    // "You have now achieved rank 5 of Shield Bash, costing 20 Combat points."
468    if let Some(rest) = line.strip_prefix("You have now achieved rank ") {
469        let (rank, rest) = rest.split_once(" of ")?;
470        let (name, rest) = rest.split_once(", costing ")?;
471        let category = rest.split_whitespace().nth(1)?;
472        return Some(PsmChange {
473            name: name.to_owned(),
474            category: category.to_owned(),
475            rank: rank.parse().ok()?,
476        });
477    }
478
479    // "You decide to unlearn rank 5 of Shield Bash, regaining 20 Combat points."
480    //
481    // The rank named is the one being REMOVED, so what remains is one less.
482    if let Some(rest) = line.strip_prefix("You decide to unlearn rank ") {
483        let (rank, rest) = rest.split_once(" of ")?;
484        let (name, rest) = rest.split_once(", regaining ")?;
485        let category = rest.split_whitespace().nth(1)?;
486        return Some(PsmChange {
487            name: name.to_owned(),
488            category: category.to_owned(),
489            rank: rank.parse::<u16>().ok()?.saturating_sub(1),
490        });
491    }
492
493    bracketed_technique(line)
494}

496/// The `[...]` technique lines, which name the category before the PSM.
497fn bracketed_technique(line: &str) -> Option<PsmChange> {
498    let inner = line.strip_prefix('[')?.strip_suffix(']')?;
499
500    // "[You are no longer trained in Shield Specialization: Block Mastery.]"
501    if let Some(rest) = inner.strip_prefix("You are no longer trained in ") {
502        let (category, name) = split_technique(rest)?;
503        return Some(PsmChange {
504            name,
505            category,
506            rank: 0,
507        });
508    }
509
510    // "[You have gained rank 2 of Shield Specialization: Block Mastery.]"
511    // "[You have increased to rank 2 of ...]" / "decreased to rank 1 of ..."
512    let rest = inner
513        .strip_prefix("You have gained rank ")
514        .or_else(|| inner.strip_prefix("You have increased to rank "))
515        .or_else(|| inner.strip_prefix("You have decreased to rank "))?;
516    let (rank, rest) = rest.split_once(" of ")?;
517    let (category, name) = split_technique(rest)?;
518    Some(PsmChange {
519        name,
520        category,
521        rank: rank.parse().ok()?,
522    })
523}

525/// `Shield Specialization: Block Mastery.` -> `("Shield", "Block Mastery")`.
526fn split_technique(rest: &str) -> Option<(String, String)> {
527    let (head, name) = rest.split_once(": ")?;
528    let category = head.split_whitespace().next()?;
529    Some((category.to_owned(), name.trim_end_matches('.').to_owned()))
530}

541/// Read the `resource` report's amounts line.
542///
543/// **The line does not say which resource it is**, so only the numbers come
544/// back. The type is
545/// learned from [`suffused_line`], which is why `vocabulary.rs` records that
546/// the two are stored independently.
547#[must_use]
548pub fn resource_line(line: &str) -> Option<ResourceAmounts> {
549    let (name, rest) = line.split_once(": ")?;
550    ResourceType::parse(name)?;
551    let (weekly, rest) = rest.split_once("/50,000 (Weekly)")?;
552    let (total, _) = rest.trim_start().split_once("/200,000 (Total)")?;
553    Some(ResourceAmounts {
554        weekly: parse_comma_number(weekly)?,
555        total: parse_comma_number(total)?,
556    })
557}

567/// Read the `Covert Arts Charges: N/200` line.
568///
569/// VERIFIED in a live `resource` output, where it
570/// follows the suffused line:
571///
572/// ```text
573/// Covert Arts Charges: 164/200
574/// ```
575///
576/// The denominator is fixed at 200 and is required -- a different
577/// denominator would mean the game changed the cap, and
578/// silently reading the numerator anyway would store a number against the
579/// wrong scale.
580///

589}

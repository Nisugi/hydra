1//! The nervous system's rank, and whether it is a wound or a scar.
2//!
3//! The injury window reports the nerves as `<image id='nsys' name='Nsys2'>`:
4//! a rank, **not** `Injury2` or `Scar2`, so the image alone does not say which
5//! it is (a doll that reads only the `Injury`/`Scar` prefix never shows
6//! convulsions). Sending `health` each time a rank above 0 arrives, and
7//! reading its nerve line, would answer it at the cost of a command.
8//!
9//! # Hydra works it out instead
10//!
11//! Hydra sees every line, so it can follow the rank as it changes.
12//!
13//! Two facts decide it. **Damage only makes a wound**: lightning, an
14//! overcast, a blow. **Only a herb makes a scar**: eating one heals a wound a
15//! rank and leaves a scar as deep as the wound was, and the scar heals a rank
16//! at a time after. The window shows the wound over the scar, so healing a
17//! rank-2 wound reads `Nsys2`, `Nsys1`, `Nsys2`, `Nsys1`, `Nsys0`:
18//!
19//! | Rank | What happened | Wound | Scar |
20//! |---|---|---|---|
21//! | 2 | damage: a rise is a wound | 2 | 0 |
22//! | 1 | a herb: the wound down a rank, a scar of 2 under it | 1 | 2 |
23//! | 2 | a herb: the wound gone, the scar showing | 0 | 2 |
24//! | 1 | a herb: the scar down a rank | 0 | 1 |
25//! | 0 | a herb: whole | 0 | 0 |
26//!
27//! So a rank is read at the prompt that ends its chunk, when the chunk's
28//! lines say whether a herb was eaten. Only when that does not explain it --
29//! the first rank after logging in, which may be an old scar; a fall with no
30//! herb, which an empath or a spell may make; a herb step that is not the
31//! next one -- is Hydra **confused**, the rank shown as a wound (wounds over
32//! scars) and `health` asked once ([`Character::take_nerve_question`]). A
33//! player's own `health` settles it the same way.
34
35use super::{Character, Injury};

37/// The nerves as the injury window last ranked them.
38#[derive(Clone, Debug, Default, PartialEq, Eq)]
39pub struct Nerves {
40    /// The last `Nsys<n>` rank read; 0 when whole or never shown.
41    pub rank: u8,
42    /// Whether the wound and scar under that rank are known, worked out or
43    /// said by `health`, rather than guessed.
44    pub settled: bool,
45    /// A rank has been read since the character logged in: the next one is a
46    /// step from it, not a first sighting.
47    known: bool,
48    /// Ranks the window gave in the chunk under way, read at its prompt.
49    seen: Vec<u8>,
50    /// Confused since last asked: `health` should be sent, once.
51    asking: bool,
52}
53
54/// The six lines `health` gives the nerves, each with its kind and rank
55/// (verbatim fragments).
56const LINES: [(&str, Track, u8); 6] = [
57    ("a case of uncontrollable convulsions", Track::Wound, 3),
58    ("a case of sporadic convulsions", Track::Wound, 2),
59    ("a strange case of muscle twitching", Track::Wound, 1),
60    ("a very difficult time with muscle control", Track::Scar, 3),
61    ("constant muscle spasms", Track::Scar, 2),
62    ("developed slurred speech", Track::Scar, 1),
63];
64
65/// Which of the two a nerve line names.
66#[derive(Clone, Copy, Debug, PartialEq, Eq)]
67enum Track {
68    Wound,
69    Scar,
70}
71
72/// Where the `health` report's table starts: its description lines are the
73/// ones above.
74const TABLE: &str = "Maximum Health Points:";
75
76/// How eating or drinking a herb begins.
77const HERB: [&str; 2] = ["You take a bite of your ", "You take a drink from your "];
78
79/// The nerve line among a `health` report's description, if the lines are
80/// one and it has one.
81fn nerve_line(lines: &[String]) -> Option<(Track, u8)> {
82    let table = lines.iter().position(|line| line.contains(TABLE))?;
83    lines[..table].iter().find_map(|line| {
84        let line = line.trim_start();
85        if !line.starts_with("You") {
86            return None;
87        }
88        LINES
89            .iter()
90            .find(|(text, ..)| line.contains(text))
91            .map(|(_, track, rank)| (*track, *rank))
92    })
93}
94
95/// Whether `line` is one of `health`'s six nerve lines about the character:
96/// what a session leaves out of the reply to a `health` it sent itself.
97#[must_use]
98pub fn is_nerve_line(line: &str) -> bool {
99    let line = line.trim_start();
100    line.starts_with("You") && LINES.iter().any(|(text, ..)| line.contains(text))
101}
102
103/// The wound and scar a herb leaves, from `wound` and `scar`, if the window's
104/// `rank` is the step it shows next.
105fn herb_step(wound: u8, scar: u8, rank: u8) -> Option<(u8, u8)> {
106    if wound > 0 {
107        // The wound down a rank; a scar as deep as it was.
108        let scar = scar.max(wound);
109        let wound = wound - 1;
110        let shown = if wound > 0 { wound } else { scar };
111        (shown == rank).then_some((wound, scar))
112    } else {
113        (scar > 0 && rank + 1 == scar).then_some((0, rank))
114    }
115}

117impl Character {
118    /// `Nsys<rank>` from the injury window: kept until the chunk's prompt.
119    pub(super) fn apply_nerve_rank(&mut self, rank: u8) {
120        self.nerves.seen.push(rank);
121    }
122
123    /// The injury window, showing wounds alone, shows the nerves whole: no
124    /// wound remains, whatever scar is known stays. Nothing is asked.
125    pub(super) fn nerves_wound_healed(&mut self) {
126        self.nerves.rank = 0;
127        self.nerves.known = true;
128        self.nerves.settled = true;
129        let scar = self.nsys().scar;
130        self.set_nsys(Injury { wound: 0, scar });
131    }
132
133    /// Read the chunk's nerve ranks, knowing whether it ate a herb; then a
134    /// `health` report among `lines`, if it is one and says.
135    pub(crate) fn consume_nerves(&mut self, lines: &[String]) {
136        let herb = lines.iter().any(|line| {
137            let line = line.trim_start();
138            HERB.iter().any(|start| line.starts_with(start))
139        });
140        let (was, seen) = (self.nerves.rank, std::mem::take(&mut self.nerves.seen));
141        let moved = seen.iter().any(|rank| *rank != was);
142        for rank in seen {
143            self.read_nerve_rank(rank, herb);
144        }
145        // **A herb that leaves the rank showing where the nerves' herb would
146        // too** cannot be worked out: a rank-1 wound healed leaves a rank-1
147        // scar, `Nsys1` either way, and the herb may have been another
148        // part's. Kept as the wound, a heal would eat wound herbs for a scar
149        // again and again. Asked instead.
150        let Injury { wound, scar } = self.nsys();
151        if herb
152            && !moved
153            && self.nerves.known
154            && self.nerves.settled
155            && herb_step(wound, scar, was).is_some_and(|step| step != (wound, scar))
156        {
157            self.nerves.settled = false;
158            self.nerves.asking = true;
159        }
160        if let Some((track, rank)) = nerve_line(lines) {
161            let known = self.nsys();
162            self.set_nsys(match track {
163                Track::Wound => Injury {
164                    wound: rank,
165                    scar: known.scar,
166                },
167                Track::Scar => Injury {
168                    wound: 0,
169                    scar: rank,
170                },
171            });
172            self.nerves.settled = true;
173            self.nerves.asking = false;
174        }
175    }
176
177    /// One rank, as [`nerves`](self)'s table reads it.
178    fn read_nerve_rank(&mut self, rank: u8, herb: bool) {
179        let (was, first) = (self.nerves.rank, !self.nerves.known);
180        self.nerves.rank = rank;
181        self.nerves.known = true;
182        if rank == 0 {
183            self.set_nsys(Injury::default());
184            self.nerves.settled = true;
185            return;
186        }
187        if rank == was && !first {
188            return;
189        }
190        let Injury { wound, scar } = self.nsys();
191        let worked_out = if first {
192            None
193        } else if herb {
194            herb_step(wound, scar, rank)
195        } else if rank > was {
196            Some((rank, scar))
197        } else {
198            None
199        };
200        if let Some((wound, scar)) = worked_out {
201            self.set_nsys(Injury { wound, scar });
202            self.nerves.settled = true;
203        } else {
204            self.set_nsys(Injury { wound: rank, scar });
205            self.nerves.settled = false;
206            self.nerves.asking = true;
207        }
208    }
209
210    fn nsys(&self) -> Injury {
211        self.injuries.get("nsys").copied().unwrap_or_default()
212    }
213
214    fn set_nsys(&mut self, injury: Injury) {
215        if injury.is_hurt() {
216            self.injuries.insert("nsys".to_owned(), injury);
217        } else {
218            self.injuries.remove("nsys");
219        }
220    }
221
222    /// Whether the nerves are confused since last asked, and `health` should
223    /// be sent to settle them: `true` once per confusion.
224    pub fn take_nerve_question(&mut self) -> bool {
225        std::mem::take(&mut self.nerves.asking)
226    }
227}

229#[cfg(test)]
230mod tests {
231    use super::{Track, herb_step, nerve_line};
232
233    fn lines(text: &str) -> Vec<String> {
234        text.lines().map(str::to_owned).collect()
235    }
236
237    #[test]
238    fn each_line_names_its_kind_and_rank_only_above_the_table() {
239        let report = "You have a case of sporadic convulsions.\n\n   Maximum Health Points:   158";
240        assert_eq!(nerve_line(&lines(report)), Some((Track::Wound, 2)));
241        let scar = "You have developed slurred speech.\n   Maximum Health Points:   158";
242        assert_eq!(nerve_line(&lines(scar)), Some((Track::Scar, 1)));
243        // Not a report: someone else's convulsions, in the room.
244        assert_eq!(
245            nerve_line(&lines("Ada has a case of sporadic convulsions.")),
246            None
247        );
248        // Below the table is not a description.
249        let after = "   Maximum Health Points:   158\nYou have constant muscle spasms.";
250        assert_eq!(nerve_line(&lines(after)), None);
251    }
252
253    /// The healing sequence, step by step, and a step that is not next.
254    #[test]
255    fn a_herb_takes_the_next_step_or_none() {
256        assert_eq!(herb_step(2, 0, 1), Some((1, 2)));
257        assert_eq!(herb_step(1, 2, 2), Some((0, 2)));
258        assert_eq!(herb_step(0, 2, 1), Some((0, 1)));
259        assert_eq!(herb_step(3, 0, 1), None, "two ranks at once is not a step");
260        assert_eq!(herb_step(0, 0, 1), None, "nothing to heal");
261    }

1//! Flare announce lines: weapon scripts, enchants, GEFs, flourishes.
2//!
3//! Reads the `flare` family. Each def carries three flags the state machine
4//! routes on:
5//!
6//! - `damaging` -- a damage line is expected to follow; a buff flare
7//!   (acuity, tailwind) never produces one;
8//! - `aoe` -- false when the flare is single-target even if its parent is not;
9//! - `spawns` -- the flare casts an imbedded spell as a SEPARATE attack
10//!   (Blink), so what follows in the chunk is its child.
11//!
12//! **Timing is not declared.** Whether a flare resolves before its swing
13//! (dispel gloves) or after (most) is positional: a flare seen with no open
14//! attack event is held and claimed by the next swing. That is the state
15//! machine's job; this only says what the line is.
16//!
17//! # The flaring weapon is a link after `Your `
18//!
19//! A 2p flare names its weapon as a link (`Your <a exist="..." ...>`), and
20//! the id is the join key for claiming pre-flares by weapon. Written by hand
21//! (`defs::HAND_PORTED`): the weapon is the first link whose run follows the
22//! text `Your `.
23//!
24//! # The mirror table, after the main flares
25//!
26//! The main flare table misses most custom and festival flare texts,
27//! purified metals, lore-flare repeats, weapon and armor scripts, Covert Arts
28//! poisons, the `_GS` jewel properties and ensorcell. A second table covers
29//! them: family `flare_mirror` (`data/combat_flare_mirror.tsv`, generated by
30//! `tools/extract_flare_mirror.rb`), read only when no main flare matches. A
31//! row that reuses a main flare's name carries that flare's flags, so a
32//! custom acid text is an `acid` flare. Six rows are left out: ensorcell's
33//! benefit and the spirit animal's prefix are lines of a proc a main flare
34//! already counts, and read here too each proc would count as two flares
35//! (the extractor's `NOT_PORTED`). Ensorcell's benefit is read into its
36//! flare instead, as the flare's kind (`parse/flares.rs`).
37//!
38//! **Not on an attack or a damage line.** Three mirror rows re-read a line
39//! the attack and damage grammar own: the Wither lore benefit is the
40//! `wither` initiation, the flaming aura's lash is `flaming_aura`, and
41//! Sanguine Sacrifice's *"X suffers an additional N damage!"* is a bleed
42//! tick (`tests/combat_flare_mirror.rs`). Read as a flare as well, the attack
43//! would claim its own announce as a pre-flare, and the tick would be
44//! counted twice -- once as the flare's inline damage and once as the damage
45//! line -- as the replay blob `weapon_cast 47a49e78552be40a` shows. So a
46//! `flare_mirror` row never classifies a line that [`AttackLine`] or
47//! [`DamageLine`] reads.

55/// One flare announce line, classified.
56#[allow(clippy::struct_excessive_bools)] // see `event::Crit`
57#[derive(Debug, Clone, PartialEq, Eq)]
58pub struct FlareLine {
59    /// The def name: `acid`, `ensorcell`, `blink`, `dispel_flux`, ...
60    pub name: String,
61    /// A damage line is expected to follow.
62    pub damaging: bool,
63    /// May strike more than one target.
64    pub aoe: bool,
65    /// Casts an imbedded spell as a separate attack.
66    pub spawns: bool,
67    /// The target named, as captured.
68    pub target_text: Option<String>,
69    /// That target, resolved to its link.
70    pub target: Option<Actor>,
71    /// Whose item flared, for third-person forms. `None` for our own.
72    pub attacker: Option<Actor>,
73    /// The flaring weapon, when the line links it (2p forms).
74    pub weapon: Option<Actor>,
75    /// Read by the main flare table rather than the mirror: only these
76    /// resume a swing a creature's attack interrupted
77    /// (`parse/flares.rs`).
78    pub shipped: bool,
79    /// Set off by a creature's attack on us, not by a swing of ours: a block,
80    /// bark absorbing the blow, an image taking it. It stays on that attack,
81    /// and never resumes a swing of ours the attack interrupted.
82    pub reactive: bool,
83    /// What its definition says of the line beyond the name, when the
84    /// same flare has more than one wording worth telling apart: barkskin's
85    /// `physical_endured`, `magical_crumbled` and the two between.
86    pub kind: Option<String>,
87}

89/// The link whose run follows `Your ` -- the weapon a 2p flare names.
90fn weapon_link(line: &ChunkLine) -> Option<Actor> {
91    let text = line.text();
92    let mut at = 0;
93    for run in &line.runs.runs {
94        let start = at;
95        at += run.text.len();
96        if run.link.is_none() {
97            continue;
98        }
99        if text[..start].ends_with("Your ") {
100            return target::link_in(line, start..at, Pick::First);
101        }
102    }
103    None
104}

106impl FlareLine {
107    /// Classify one line as a flare announcement: the main flares first,
108    /// then the mirror's on a line that is neither an attack nor damage (see
109    /// the module doc).
110    #[must_use]
111    pub fn classify(line: &ChunkLine) -> Option<Self> {
112        let text = line.text();
113        let (def, caps) = defs().first_match("flare", &text).or_else(|| {
114            defs().first_match("flare_mirror", &text).filter(|_| {
115                AttackLine::classify(line).is_none() && DamageLine::classify(line).is_none()
116            })
117        })?;
118        // A lazy attacker capture can land on the FIRST-PERSON form ("from
119        // your hands", boil_blood): that is ours, not an attacker.
120        let attacker = caps.name("attacker").and_then(|m| {
121            let who = m.as_str().trim();
122            if matches!(
123                who.to_ascii_lowercase().as_str(),
124                "you" | "your" | "yourself"
125            ) {
126                return None;
127            }
128            target::link_in(line, m.range(), Pick::Last).or_else(|| Some(Actor::unlinked(who)))
129        });
130        let weapon = defs()
131            .any_match("flare_weapon_link", &text)
132            .then(|| weapon_link(line))
133            .flatten();
134        Some(Self {
135            name: def.name.clone(),
136            damaging: def.flag("damaging"),
137            aoe: def.flag("aoe"),
138            spawns: def.flag("spawns"),
139            target_text: caps.name("target").map(|m| m.as_str().to_owned()),
140            target: caps
141                .name("target")
142                .and_then(|m| target::link_in(line, m.range(), Pick::First)),
143            attacker,
144            weapon,
145            shipped: def.family == "flare",
146            reactive: def.flag("reactive"),
147            kind: def.extra("kind").map(str::to_owned),
148        })
149    }
150
151    /// Our own item fired, rather than a nearby player's or a creature's.
152    #[must_use]
153    pub const fn is_ours(&self) -> bool {
154        self.attacker.is_none()
155    }
156}

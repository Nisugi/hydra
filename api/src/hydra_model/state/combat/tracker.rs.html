1//! The combat consumer: what survives between chunks, and the entry point.
2//!
3//! The cross-chunk state of the combat parse, and its one real job: feeding
4//! a prompt-bounded chunk to the state machine. Buffers, worker threads and
5//! settings stores are plumbing the chunk mechanism and the session already
6//! own.
7//!
8//! # Three facts cross a chunk boundary
9//!
10//! | Field | Why it must survive the prompt |
11//! |---|---|
12//! | `held_cast` | *"You gesture at X."* ends a chunk; the spell's result opens the next. Held for ONE chunk, then superseded or emitted as itself. |
13//! | `held_pre_flares` | A bow's dispel fires on the NOCK and resolves before *"You fire"* prints -- in the next chunk. |
14//! | [`active_assault`](CombatTracker::active_assault) | A flurry's rounds arrive over several chunks; the opener named the only target they can strike. |
15//!
16//! The death watch (a creature an event touched, watched until a room
17//! refresh flags it dead) belongs to the creature registry. Position
18//! recovery is reset at the top of every chunk and is therefore chunk-local,
19//! on the cursor.
20//!
21//! # No settings
22//!
23//! Damage, wounds, statuses, UCS and the event payload are always tracked:
24//! a config option with only one sensible value is not an option, and a
25//! wrong gate leaves a crit-shaped hole in what is emitted.
26//!
27//! # The crit tables are a handle, not a global
28//!
29//! `crit.rs` holds `CritTables` behind an `Arc` and forbids a static, for
30//! the reason multi-session depends on. So the tracker is *handed* its tables
31//! by whoever built the `GameState` -- [`CombatTracker::set_crit_tables`] --
32//! and without them the crit lookahead is skipped and every [`Hit`](super::event::Hit) carries
33//! `crit: None`. A test that asserts crits sets them; a session sets them
34//! once.
35

45/// The open assault bracket, across chunks.
46#[derive(Debug, Clone, PartialEq, Eq)]
47pub struct ActiveAssault {
48    /// Which assault.
49    pub name: AssaultName,
50    /// The only target its rounds can strike. `None` until the first named
51    /// round backfills it (barrage's opener names nobody).
52    pub target: Option<Actor>,
53}
54
55/// How many chunks' facts are kept for a consumer to drain.
56///
57/// A recorder reads them as they arrive; a bounded queue means a session
58/// with no recorder attached does not grow without limit. Oldest dropped
59/// first, and counted: nothing is dropped without saying so.
60pub const MAX_PENDING_CHUNKS: usize = 64;
61
62/// The combat consumer's state between chunks.
63#[derive(Clone, Default)]
64pub struct CombatTracker {
65    /// A bare `cast` gesture whose spell result has not arrived yet.
66    pub(super) held_cast: Option<AttackEvent>,
67    /// Pre-flares no swing claimed, held for the next chunk's first swing.
68    pub(super) held_pre_flares: Vec<FlareEvent>,
69    /// The open single-target multi-round attack.
70    pub(super) active_assault: Option<ActiveAssault>,
71    /// The volley under way: how many hails have fallen since it was
72    /// loosed, which is the round its arrows are in. `None` between
73    /// volleys.
74    pub(super) volley_round: Option<u8>,
75    /// The last creature stomp: when, and who. Every shake of the ground
76    /// within its tremor is that creature's warding roll on the room.
77    pub(super) tremors: Option<(u32, Actor)>,
78    /// The crit tables, when the session has provided them.
79    crit: Option<Arc<CritTables>>,
80    /// Facts not yet drained by a consumer, oldest first.
81    pending: VecDeque<ChunkFacts>,
82    /// Chunks' facts dropped to the cap before anyone drained them.
83    dropped: usize,
84    /// How many chunks have been consumed, ever.
85    chunks_seen: u64,
86}

88impl std::fmt::Debug for CombatTracker {
89    /// `CritTables` is 2,394 compiled patterns and does not print itself;
90    /// whether the handle is set is the fact worth showing.
91    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
92        f.debug_struct("CombatTracker")
93            .field("held_cast", &self.held_cast)
94            .field("held_pre_flares", &self.held_pre_flares)
95            .field("active_assault", &self.active_assault)
96            .field("volley_round", &self.volley_round)
97            .field("tremors", &self.tremors)
98            .field("crit_tables", &self.crit.is_some())
99            .field("pending", &self.pending.len())
100            .field("dropped", &self.dropped)
101            .field("chunks_seen", &self.chunks_seen)
102            .finish()
103    }
104}

106impl PartialEq for CombatTracker {
107    /// Everything but the tables themselves: whether they are set is
108    /// compared, their 2,394 rows are not. `GameState` derives `Eq` over
109    /// this, and no `f64` here comes from anywhere but parsed digits.
110    fn eq(&self, other: &Self) -> bool {
111        self.held_cast == other.held_cast
112            && self.held_pre_flares == other.held_pre_flares
113            && self.active_assault == other.active_assault
114            && self.crit.is_some() == other.crit.is_some()
115            && self.pending == other.pending
116            && self.dropped == other.dropped
117            && self.chunks_seen == other.chunks_seen
118    }
119}

123impl CombatTracker {
124    /// Give the tracker its crit tables.
125    ///
126    /// The session builds one `CritTables` and shares it as an `Arc`. Until
127    /// this is called, hits carry no crit.
128    pub fn set_crit_tables(&mut self, tables: Arc<CritTables>) {
129        self.crit = Some(tables);
130    }
131
132    /// The crit tables, if provided.
133    #[must_use]
134    pub fn crit_tables(&self) -> Option<&CritTables> {
135        self.crit.as_deref()
136    }
137
138    /// The open assault, if any.
139    #[must_use]
140    pub const fn active_assault(&self) -> Option<&ActiveAssault> {
141        self.active_assault.as_ref()
142    }
143
144    /// Is a bare cast being held for the next chunk?
145    #[must_use]
146    pub const fn holds_a_cast(&self) -> bool {
147        self.held_cast.is_some()
148    }
149
150    /// How many pre-flares are held for the next chunk.
151    #[must_use]
152    pub fn held_pre_flare_count(&self) -> usize {
153        self.held_pre_flares.len()
154    }
155
156    /// Chunks consumed so far.
157    #[must_use]
158    pub const fn chunks_seen(&self) -> u64 {
159        self.chunks_seen
160    }
161
162    /// Facts dropped to the pending cap before being drained.
163    #[must_use]
164    pub const fn dropped(&self) -> usize {
165        self.dropped
166    }
167
168    /// Take every pending chunk's facts, oldest first.
169    pub fn take_facts(&mut self) -> Vec<ChunkFacts> {
170        self.pending.drain(..).collect()
171    }
172
173    /// The most recent chunk's facts, without taking them.
174    #[must_use]
175    pub fn last_facts(&self) -> Option<&ChunkFacts> {
176        self.pending.back()
177    }
178
179    /// Parse one completed chunk into its facts, without publishing them.
180    ///
181    /// `at` is the chunk's prompt time -- the server epoch second the prompt
182    /// carried -- which stamps every event. `GameState::close_chunk` applies
183    /// the facts to the creature registry between this and [`Self::publish`],
184    /// in that order: parse, persist, then emit.
185    pub fn parse_chunk(&mut self, chunk: &Chunk, at: Option<u32>) -> ChunkFacts {
186        self.chunks_seen = self.chunks_seen.saturating_add(1);
187        super::parse::parse_chunk(self, chunk, at)
188    }
189
190    /// Queue a chunk's facts for a consumer to drain. Empty facts are not
191    /// queued.
192    pub fn publish(&mut self, facts: ChunkFacts) {
193        if facts.is_empty() {
194            return;
195        }
196        if self.pending.len() >= MAX_PENDING_CHUNKS {
197            self.pending.pop_front();
198            self.dropped = self.dropped.saturating_add(1);
199        }
200        self.pending.push_back(facts);
201    }
202
203    /// Parse and publish in one step: the registry-less path.
204    pub fn consume_chunk(&mut self, chunk: &Chunk, at: Option<u32>) {
205        let facts = self.parse_chunk(chunk, at);
206        self.publish(facts);
207    }
208
209    /// Forget everything a new connection has not re-taught.
210    ///
211    /// A held cast or pre-flare belongs to a chunk the old connection never
212    /// finished; an assault bracket cannot outlive the fight it was in.
213    /// The model's own reconnect invalidation, run on the session's
214    /// reconnect, calls this.
215    pub fn invalidate_for_reconnect(&mut self) {
216        self.held_cast = None;
217        self.held_pre_flares.clear();
218        self.active_assault = None;
219        self.volley_round = None;
220        self.tremors = None;
221    }
222}

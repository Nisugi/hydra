1//! Roll lines: the numbers that decide an attack.
2//!
3//! Eight roll grammars, every named capture converted to a number.
4//!
5//! # One shape for eight grammars
6//!
7//! Each grammar captures its own names (`as`, `ds`, `avd` for one; `cs`,
8//! `td`, `cva` for another). They are folded here, once, into one row shape
9//! -- `attacker_stat`, `defender_stat`, `modifier`, `roll`, `bonus`,
10//! `penalty`, `result`, `total` -- so every consumer sees the same eight
11//! fields whatever the grammar:
12//!
13//! | kind | attacker | defender | modifier | extra |
14//! |---|---|---|---|---|
15//! | `as_ds` | AS | DS | `AvD` | `bonus` = True Strike |
16//! | `cs_td` | CS | TD | `CvA` | `bonus` / `penalty` |
17//! | `uaf_udf` | UAF | UDF | MM | `total`, fractional |
18//! | `fear` | FS | FD | `FvP` | |
19//! | `activation` | | | Modifiers | |
20//! | `smr` / `ssr` / `maneuver_roll` | | | | `bonus` / `penalty` |
21//!
22//! **`roll` is kept apart from `result`** so `result - roll` is the
23//! deterministic margin -- what a "use a setup first" tactic reads.
24
25use super::defs::defs;

28/// Which roll grammar a line is.
29#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
30pub enum ResolutionKind {
31    /// Melee and ranged: `AS: +351 vs DS: +299 with AvD: +22 + d100 roll: +88 = +162`.
32    AsDs,
33    /// Warding spells: `CS: 384 - TD: 419 + CvA: 13 + d100: 59 == 37`.
34    CsTd,
35    /// Unarmed combat: `UAF: 681 vs UDF: 575 = 1.18 * MM: 110 + d100: 32 = 130`.
36    UafUdf,
37    /// Standard maneuver roll: `[SMR result: 191 (Open d100: 13, Bonus: 56)]`.
38    Smr,
39    /// Standard save roll: `[SSR result: N (Open d100: N)]`.
40    Ssr,
41    /// Imbed or crystal activation: `1d100: 30 + Modifiers: 375 == 405`.
42    Activation,
43    /// The legacy maneuver form: `[Roll result: N (open d100: N)]`.
44    ManeuverRoll,
45    /// Sheer fear: `FS: N - FD: N + FvP: N + d100(L): N = N`.
46    Fear,
47}
48
49impl ResolutionKind {
50    /// Every kind.
51    pub const ALL: [Self; 8] = [
52        Self::AsDs,
53        Self::CsTd,
54        Self::UafUdf,
55        Self::Smr,
56        Self::Ssr,
57        Self::Activation,
58        Self::ManeuverRoll,
59        Self::Fear,
60    ];
61
62    /// The TSV and recorder spelling.
63    #[must_use]
64    pub const fn as_str(self) -> &'static str {
65        match self {
66            Self::AsDs => "as_ds",
67            Self::CsTd => "cs_td",
68            Self::UafUdf => "uaf_udf",
69            Self::Smr => "smr",
70            Self::Ssr => "ssr",
71            Self::Activation => "activation",
72            Self::ManeuverRoll => "maneuver_roll",
73            Self::Fear => "fear",
74        }
75    }
76
77    /// Parse the TSV spelling.
78    #[must_use]
79    pub fn parse(text: &str) -> Option<Self> {
80        Self::ALL.into_iter().find(|k| k.as_str() == text)
81    }
82
83    /// **Does this roll precede its attack line?**
84    ///
85    /// Maneuver-class rolls (SMR/SSR/fear) PRECEDE their per-target line;
86    /// AS/DS-class rolls FOLLOW their attack line. The state machine's
87    /// held-roll logic keys on this set (`smr`, `ssr`, `maneuver_roll`).
88    #[must_use]
89    pub const fn precedes_its_line(self) -> bool {
90        matches!(
91            self,
92            Self::Smr | Self::Ssr | Self::ManeuverRoll | Self::Fear
93        )
94    }
95}

97/// One roll line, folded into the recorder's shape.
98#[derive(Debug, Clone, PartialEq)]
99pub struct Resolution {
100    /// Which grammar.
101    pub kind: ResolutionKind,
102    /// AS, CS, UAF or FS.
103    pub attacker_stat: Option<i64>,
104    /// DS, TD, UDF or FD.
105    pub defender_stat: Option<i64>,
106    /// `AvD`, `CvA`, MM, `FvP` or an activation's Modifiers.
107    pub modifier: Option<i64>,
108    /// The die, kept separate.
109    pub roll: Option<i64>,
110    /// A bonus term.
111    pub bonus: Option<i64>,
112    /// A penalty term.
113    pub penalty: Option<i64>,
114    /// The endroll.
115    pub result: Option<i64>,
116    /// UCS's pre-MM total, the one fractional value.
117    pub total: Option<f64>,
118}
119
120impl Resolution {
121    /// Classify one line as a roll.
122    #[must_use]
123    pub fn classify(line: &ChunkLine) -> Option<Self> {
124        let text = line.text();
125        let (def, caps) = defs().first_match("resolution", &text)?;
126        let kind = ResolutionKind::parse(&def.name)?;
127        let int = |name: &str| caps.name(name).and_then(|m| m.as_str().parse::<i64>().ok());
128        Some(Self {
129            kind,
130            attacker_stat: int("as")
131                .or_else(|| int("cs"))
132                .or_else(|| int("uaf"))
133                .or_else(|| int("fs")),
134            defender_stat: int("ds")
135                .or_else(|| int("td"))
136                .or_else(|| int("udf"))
137                .or_else(|| int("fd")),
138            modifier: int("avd")
139                .or_else(|| int("cva"))
140                .or_else(|| int("mm"))
141                .or_else(|| int("fvp"))
142                .or_else(|| int("mods")),
143            roll: int("roll"),
144            bonus: int("bonus"),
145            penalty: int("penalty"),
146            result: int("result"),
147            total: caps
148                .name("total")
149                .and_then(|m| m.as_str().parse::<f64>().ok()),
150        })
151    }
152
153    /// `result - roll`: the margin the dice did not supply.
154    #[must_use]
155    pub fn margin(&self) -> Option<i64> {
156        Some(self.result? - self.roll?)
157    }
158}

160/// Is this the knockdown rider a leg crit rolls after its damage settles?
161///
162/// That roll belongs to the hit that caused it, not to the next attack --
163/// without this it would be orphaned into a synthetic unknown attack against
164/// whatever creature was current.
165#[must_use]
166pub fn crit_rider_line(line: &ChunkLine) -> bool {
167    defs().any_match("crit_rider", &line.text())

1//! Attack initiations: the branch that opens events.
2//!
3//! Runs in both states -- a new attack while seeking damage closes the
4//! previous event -- and is where every modifier armed by an earlier line
5//! (ambush, redirect, nock, held pre-flares, held rolls) is claimed. One
6//! branch, in phases.

25/// A switch artifact's inherited lineage, carried to the event that replaces
26/// it on the same line.
27struct Lineage {
28    root: Option<EvId>,
29    parent: Option<EvId>,
30    confidence: Option<Confidence>,
31}

34    /// The `(spell, victim)` key of the `DoT` ownership table.
35    fn cast_owner_key(name: &str, target: &EventTarget) -> (String, String) {
36        let victim = match target {
37            EventTarget::Creature(a) => a.id.map_or_else(|| a.name.clone(), |i| i.to_string()),
38            EventTarget::Foreign(n) => n.clone(),
39            EventTarget::None => String::new(),
40        };
41        (name.to_owned(), victim)
42    }

44    /// The attack branch. `atk` is the line's initiation, already gated:
45    /// `None` when an ambush or redirect prefix took the line.
46    pub fn handle_attack(&mut self, ctx: &LineCtx, line: &ChunkLine, atk: AttackLine) {
47        let superseded_cast = self.supersede_bare_cast(ctx, &atk);
48        self.note_single_hit_parent(ctx, &atk);
49        self.close_previous(ctx, line, &atk);
50        let artifact_lineage = self.take_artifact_lineage(ctx);
51        let id = self.open_attack_event(ctx, line, atk, superseded_cast);
52        let name = self.ev(id).name.clone();
53        self.stamp_lineage(id, &ctx.text, artifact_lineage);
54
55        // Record ownership of a DoT from its CAST line. Cast lines carry no
56        // inline damage, and inline damage is added below, so this always
57        // runs; a "no hits yet" guard would be a tautology here.
58        if UNOWNED_TICK_ATTACKS.contains(&name.as_str()) {
59            let e = self.ev(id);
60            let owner = if e.foreign_caster {
61                Some(false)
62            } else if e.attacker.is_none() && starts_you(&ctx.text) {
63                Some(true)
64            } else {
65                None
66            };
67            self.cast_owner
68                .insert(Self::cast_owner_key(&name, &e.target), owner);
69        }
70
71        // Claimed: the ambush belongs to this attack only.
72        self.pending_ambush = None;
73        self.current_target = self
74            .ev(id)
75            .target
76            .creature()
77            .filter(|a| a.id.is_some())
78            .cloned();
79        self.bind_assault(id, &name);
80
81        // Inline damage: DoT ticks are the whole event, message and damage in
82        // one. Summary-damage attacks print the total the next line repeats.
83        if !SUMMARY_DAMAGE_ATTACKS.contains(&name.as_str())
84            && let Some(inline) = DamageLine::classify(line)
85        {
86            self.ev_mut(id).hits.push(Hit {
87                damage: inline.amount,
88                crit: None,
89                line: ctx.index,
90            });
91        }
92        if let Some(o) = ctx.same_line_outcome {
93            self.ev_mut(id).outcomes.push(o);
94        }
95
96        // A new swing claims held pre-flares and held rolls. An inbound
97        // attack claims neither: both were produced by OUR weapon. It does
98        // claim its own line's reactive flare (the bark's absorb written on
99        // the attack's line): that flare is this attack's, not a later
100        // swing's, and not an attack of its own.
101        if self.ev(id).inbound {
102            while self
103                .pending_flares
104                .last()
105                .is_some_and(|f| f.line == ctx.index && f.reactive)
106            {
107                if let Some(f) = self.pending_flares.pop() {
108                    self.ev_mut(id).flares.push(f);
109                }
110            }
111        } else {
112            self.claim_pending_flares(id);
113            let mut rolls = std::mem::take(&mut self.pending_resolutions);
114            self.ev_mut(id).resolutions.append(&mut rolls);
115        }
116        self.claim_maneuver_rolls(id);
117        self.flare_ctx = None;
118        self.seeking_damage = true;
119    }

121    /// A bare gesture `cast` is the WRAPPER for the spell-specific
122    /// initiation that follows: hand its rolls and fact-less flares to the
123    /// new event and discard it. A HELD cast is superseded only by a
124    /// spell-result line, never by a fresh 2p initiation of our own.
125    fn supersede_bare_cast(&mut self, ctx: &LineCtx, atk: &AttackLine) -> bool {
126        let Some(cur) = self.current else {
127            return false;
128        };
129        let e = self.ev(cur);
130        let same_attacker = match (&atk.attacker, &e.attacker) {
131            (None, _) => true,
132            (Some(a), Some(c)) => a.name == c.name,
133            (Some(_), None) => false,
134        };
135        let supersedes =
136            self.bare_cast(cur) && same_attacker && (!e.held || !starts_you(&ctx.text));
137        if !supersedes {
138            return false;
139        }
140        // The spell line that follows a creature's gesture seldom names
141        // it: the gesture's caster is the spell's.
142        if e.attacker.as_ref().is_some_and(|a| !a.is_player()) {
143            let caster = e.attacker.clone();
144            self.last_inbound_attacker = caster;
145        }
146        let e = self.ev_mut(cur);
147        let mut rolls = std::mem::take(&mut e.resolutions);
148        let mut flares = std::mem::take(&mut e.flares);
149        rolls.append(&mut self.pending_resolutions);
150        self.pending_resolutions = rolls;
151        self.pending_flares.append(&mut flares);
152        self.current = None;
153        true
154    }

156    /// A one-hit side effect remembers the event it cut in front of; its
157    /// own line names ITS victim, so the switcher has already saved the
158    /// real parent and left a same-line artifact in `current`.
159    fn note_single_hit_parent(&mut self, ctx: &LineCtx, atk: &AttackLine) {
160        self.single_hit_parent = None;
161        if !SINGLE_HIT_ATTACKS.contains(&atk.name.as_str()) {
162            return;
163        }
164        let cand = match self.current {
165            Some(c) if self.ev(c).born_line == Some(ctx.index) => self.events.last().copied(),
166            other => other,
167        };
168        if let Some(c) = cand {
169            let e = self.ev(c);
170            if !e.inbound && !e.foreign_caster && !e.is_foreign_target() {
171                self.single_hit_parent = Some(c);
172            }
173        }
174    }

176    /// Save the previous event -- unless the switcher created it on this
177    /// very line -- and remember what the new line does to it.
178    fn close_previous(&mut self, ctx: &LineCtx, line: &ChunkLine, atk: &AttackLine) {
179        let Some(cur) = self.current else {
180            return;
181        };
182        if !self.savable(cur) || self.ev(cur).born_line == Some(ctx.index) {
183            return;
184        }
185        self.save_event(cur);
186        let e = self.ev(cur);
187        let ours = !(e.inbound || e.foreign_caster || e.is_foreign_target());
188        // an inbound attack cutting into our own open swing
189        if atk.inbound && ours {
190            self.interrupted_own = Some(cur);
191        }
192        // our swing carrying an UNCLAIMED spell-releasing flare: a weaponless
193        // CAST opening now is that flare's spell. One flare releases one spell.
194        if !atk.inbound && ours && atk.weapon.is_none() && swing_weapon(line).is_none() {
195            let e = self.ev_mut(cur);
196            if let Some(rel) = e
197                .flares
198                .iter_mut()
199                .find(|f| Self::releasing(&f.name) && !f.release_claimed)
200            {
201                rel.release_claimed = true;
202                let (flare, weapon) = (rel.name.clone(), rel.weapon.clone());
203                self.interrupted_own = Some(cur);
204                self.released_parent = Some(ReleasedParent {
205                    event: cur,
206                    flare,
207                    weapon,
208                });
209            }
210        }
211        let e = self.ev(cur);
212        if e.inbound && e.attacker.is_some() {
213            let attacker = e.attacker.clone();
214            self.last_inbound_attacker = attacker;
215        }
216    }

218    /// A same-line switch artifact may have claimed held rolls and inherited
219    /// lineage: carry both across the replacement.
220    fn take_artifact_lineage(&mut self, ctx: &LineCtx) -> Option<Lineage> {
221        let cur = self.current?;
222        if self.ev(cur).born_line != Some(ctx.index) {
223            return None;
224        }
225        let e = self.ev_mut(cur);
226        if !e.resolutions.is_empty() {
227            let mut rolls = std::mem::take(&mut e.resolutions);
228            rolls.append(&mut self.pending_resolutions);
229            self.pending_resolutions = rolls;
230        }
231        let e = self.ev(cur);
232        e.root.is_some().then_some(Lineage {
233            root: e.root,
234            parent: e.parent,
235            confidence: e.parent_confidence,
236        })
237    }

239    /// `cur` is a creature's inbound attack with nothing resolved yet, and
240    /// `atk` the same creature's next line: the spell its gesture led to.
241    /// `cur` takes the spell's name and stays current.
242    fn spell_of_the_cast(&mut self, cur: EvId, atk: &AttackLine) -> Option<EvId> {
243        let e = self.ev(cur);
244        // The spell line often names no caster ("The radiant burst of
245        // light engulfs you!"): the gesture before it did.
246        let same = match (&e.attacker, &atk.attacker) {
247            (Some(a), Some(b)) => a.id.is_some() && a.id == b.id,
248            (Some(_), None) => true,
249            _ => false,
250        };
251        if !(same
252            && e.attacker.as_ref().is_some_and(|a| !a.is_player())
253            && e.outcomes.is_empty()
254            && e.resolutions.is_empty()
255            && e.hits.is_empty()
256            && e.flares.is_empty())
257        {
258            return None;
259        }
260        let e = self.ev_mut(cur);
261        e.name.clone_from(&atk.name);
262        // A gesture at another creature whose spell reaches us: at us.
263        e.inbound = true;
264        e.foreign_caster = false;
265        e.target = EventTarget::None;
266        Some(cur)
267    }

269    /// Build the event from the line, and make it current.
270    fn open_attack_event(
271        &mut self,
272        ctx: &LineCtx,
273        line: &ChunkLine,
274        atk: AttackLine,
275        superseded_cast: bool,
276    ) -> EvId {
277        let text = &ctx.text;
278        // Foreign-attacker latch: a 2p "You..." attack reclaims ownership; a
279        // foreign_caster attack arms it, so the actor-less swings of a nearby
280        // player's AoE inherit their ownership.
281        let our_2p = starts_you(text) && atk.attacker.is_none() && !atk.foreign_caster;
282        if our_2p {
283            self.foreign_latch = None;
284        }
285        if atk.foreign_caster
286            && let Some(a) = &atk.attacker
287        {
288            self.foreign_latch = Some(a.name.clone());
289        }
290        let eff_foreign = atk.foreign_caster
291            || (self.foreign_latch.is_some() && atk.attacker.is_none() && !our_2p);
292        let (eff_foreign, carried_caster, nobodys) =
293            self.foreign_spell_round(&atk, text, eff_foreign);
294
295        // A creature's cast is two lines, the gesture and the spell, and
296        // the roll follows the second: the second line is the same attack,
297        // not a new one, so the first does not stand with no result.
298        if atk.inbound
299            && let Some(cur) = self.current
300            && let Some(id) = self.spell_of_the_cast(cur, &atk)
301        {
302            return id;
303        }
304
305        let target = match atk.target {
306            TargetKind::Creature(a) => EventTarget::Creature(a),
307            TargetKind::Foreign(n) => EventTarget::Foreign(n),
308            TargetKind::None => EventTarget::None,
309        };
310        // A DoT tick with no owning cast of that spell on THIS victim is
311        // unowned: real damage to the creature, not our deal.
312        let unowned = UNOWNED_TICK_ATTACKS.contains(&atk.name.as_str())
313            && atk.attacker.is_none()
314            && !eff_foreign
315            && !starts_you(text)
316            && self
317                .cast_owner
318                .get(&Self::cast_owner_key(&atk.name, &target))
319                .copied()
320                .flatten()
321                != Some(true);
322        let mut ev = self.blank(&atk.name, target);
323        ev.attacker.clone_from(&atk.attacker);
324        if carried_caster.is_some() {
325            ev.attacker = carried_caster;
326        }
327        ev.inbound = atk.inbound;
328        ev.foreign_caster = eff_foreign;
329        // A shake of the ground is the last stomp's warding roll on us.
330        if atk.name == "tremors"
331            && atk.inbound
332            && ev.attacker.is_none()
333            && let Some((at, who)) = &self.tracker.tremors
334            && self
335                .at
336                .is_some_and(|now| now.saturating_sub(*at) <= TREMOR_SECONDS)
337        {
338            ev.attacker = Some(who.clone());
339        }
340        // A creature's spell line at another creature, after its gesture:
341        // the creature's, not ours.
342        if !atk.inbound
343            && atk.attacker.is_none()
344            && !starts_you(text)
345            && atk.group.contains("spell")
346            && let Some(c) = &self.last_creature_cast
347        {
348            ev.attacker = Some(c.clone());
349            ev.foreign_caster = true;
350        }
351        ev.unowned = unowned || nobodys;
352        ev.ambush = self.pending_ambush.as_ref().is_some_and(|a| a.hidden);
353        ev.attack_kind = self.pending_ambush.as_ref().map(|a| a.kind);
354        ev.redirect = self.pending_redirect.take();
355        ev.aimed = atk.aimed;
356        ev.weapon = swing_weapon(line).or(atk.weapon);
357        ev.via_cast = superseded_cast;
358        ev.parent_flare = self.active_spawn.as_ref().map(|s| ParentFlare {
359            flare: s.flare.clone(),
360            weapon: s.weapon.clone(),
361        });
362        ev.attack_born = true;
363        let id = self.open(ev);
364        self.current = Some(id);
365        if our_2p {
366            self.last_creature_cast = None;
367        } else if matches!(atk.name.as_str(), "cast" | "area_cast")
368            && let Some(a) = &atk.attacker
369            && !a.is_player()
370        {
371            self.last_creature_cast = Some(a.clone());
372        }
373
374        // Only OUR own attack can be the swing the nock announced.
375        if !(atk.inbound || eff_foreign) {
376            self.nocked = None;
377        }
378        // A creature's warding-spell effect line that follows its cast is
379        // that caster's.
380        if atk.inbound
381            && atk.attacker.is_none()
382            && (superseded_cast || !attackerless_line(line))
383            && let Some(lia) = self.last_inbound_attacker.clone()
384        {
385            self.ev_mut(id).attacker = Some(lia);
386        }
387        id
388    }

390    /// Spawn-tree lineage: only what can be asserted, never a positional
391    /// guess. Foreign/inbound/unowned/orphan events never join our tree.
392    fn stamp_lineage(&mut self, id: EvId, text: &str, artifact: Option<Lineage>) {
393        let not_ours = Self::not_ours(self.ev(id));
394        let name = self.ev(id).name.clone();
395        if let Some(l) = artifact.filter(|_| !not_ours) {
396            // Same-line AoE per-target line: the same attack striking another
397            // creature sits at the SAME spawn-tree node as its sibling.
398            let e = self.ev_mut(id);
399            e.root = l.root;
400            e.parent = l.parent;
401            e.parent_confidence = l.confidence;
402        } else if let (Some(_), Some(root), false) = (&self.active_spawn, self.spawn_root, not_ours)
403        {
404            // blink's bracketed cast: a child DECLARED by the game
405            let e = self.ev_mut(id);
406            e.root = Some(root);
407            e.parent = Some(root);
408            e.parent_confidence = Some(Confidence::Bracket);
409        } else if let (Some(rp), false) = (self.released_parent.take(), not_ours) {
410            // Child of the swing whose flare released it: asserted by
411            // adjacency, the same count constraint an echo uses.
412            let owner_root = self.ev(rp.event).root.unwrap_or(rp.event);
413            let e = self.ev_mut(id);
414            e.root = Some(owner_root);
415            e.parent = Some(rp.event);
416            e.parent_confidence = Some(Confidence::Count);
417            e.parent_flare = Some(ParentFlare {
418                flare: rp.flare,
419                weapon: rp.weapon,
420            });
421            e.released = true;
422        } else if !not_ours
423            && self.spawn_root.is_some()
424            && starts_you(text)
425            && self.pending_echoes.front().is_some_and(|echo| {
426                let owner = echo.owner.or(self.spawn_root).unwrap_or_default();
427                self.ev(owner).name == name
428            })
429        {
430            // A mirror/afterimage ECHO: N echo flares, N echo swings, paired
431            // FIFO. The flare is the attack.
432            let echo = self
433                .pending_echoes
434                .pop_front()
435                .unwrap_or_else(|| unreachable!());
436            let owner = echo.owner.or(self.spawn_root).unwrap_or_default();
437            let root = self.spawn_root;
438            let e = self.ev_mut(id);
439            e.root = root;
440            e.parent = Some(owner);
441            e.parent_confidence = Some(Confidence::Count);
442            e.parent_flare = Some(ParentFlare {
443                flare: echo.flare,
444                weapon: echo.weapon,
445            });
446        } else {
447            // every other own attack becomes the root of its OWN tree
448            if !not_ours {
449                self.spawn_root = Some(id);
450            }
451            self.ev_mut(id).root = Some(id);
452        }
453        // A weaponless cast opening while a spell-releasing pre-flare is
454        // still unclaimed IS that flare's spell (proc -> spell -> swing).
455        if !not_ours
456            && !self.ev(id).released
457            && self.ev(id).weapon.is_none()
458            && self.pending_flares.iter().any(|f| Self::releasing(&f.name))
459        {
460            self.ev_mut(id).released = true;
461            self.pending_release_cast = Some(id);
462        }
463    }

465    /// A creature MANEUVER rolls before its line prints: that held maneuver
466    /// roll is its own, not our next swing's -- unless we are mid-sequence
467    /// (volley, barrage), which announce themselves. Every OTHER assault
468    /// rolls AFTER its round line, so an open one must not block the claim.
469    fn claim_maneuver_rolls(&mut self, id: EvId) {
470        let barrage_open = self
471            .tracker
472            .active_assault
473            .as_ref()
474            .is_some_and(|a| a.name.rolls_before_round());
475        if !self.ev(id).inbound
476            || self.pending_resolutions.is_empty()
477            || self.active_sequence.is_some()
478            || self.nocked.is_some()
479            || barrage_open
480        {
481            return;
482        }
483        let (mut mine, rest): (Vec<_>, Vec<_>) = std::mem::take(&mut self.pending_resolutions)
484            .into_iter()
485            .partition(|r| {
486                matches!(
487                    r.kind,
488                    ResolutionKind::Smr | ResolutionKind::Ssr | ResolutionKind::ManeuverRoll
489                )
490            });
491        self.pending_resolutions = rest;
492        self.ev_mut(id).resolutions.append(&mut mine);
493    }

495    /// Assault binding: while an assault is open, its own targetless rounds
496    /// strike the assault target by definition. Any OTHER outbound attack
497    /// def mid-assault means the bracket state is stale: drop it.
498    fn bind_assault(&mut self, id: EvId, name: &str) {
499        let Some(aa) = self.tracker.active_assault.clone() else {
500            return;
501        };
502        let e = self.ev(id);
503        if e.inbound || e.is_foreign_target() {
504            return;
505        }
506        if name == aa.name.as_str() || name == "ambush" {
507            if self.current_target.is_none() {
508                if let Some(t) = aa.target.filter(|a| a.id.is_some()) {
509                    self.ev_mut(id).target = EventTarget::Creature(t.clone());
510                    self.current_target = Some(t);
511                }
512            } else if aa.target.is_none()
513                && let Some(open) = self.tracker.active_assault.as_mut()
514            {
515                // barrage's opener names no target: the first named round
516                // inside the bracket backfills it
517                open.target.clone_from(&self.current_target);
518            }
519        } else if e.attacker.is_none() && !matches!(name, "unknown" | "companion") && !e.released {
520            self.tracker.active_assault = None;
521        }
522    }

524    /// Whether a nearby player's cast or multi-round spell is still fresh
525    /// at this chunk's time.
526    fn fresh(then: Option<u32>, now: Option<u32>) -> bool {
527        then.zip(now)
528            .is_none_or(|(then, now)| now.saturating_sub(then) <= FOREIGN_SPELL_SECONDS)
529    }

531    /// A nearby player's gesture arms a latch for the multi-round spell it
532    /// may be: the rounds name nobody and come chunks apart, between our
533    /// own swings, so a round of a spell the `foreign_spell` table knows
534    /// binds the latch to that spell, and every round after is the
535    /// caster's while the latch holds. A round of such a spell is ours only
536    /// after our own opening of it; with no opening and no caster in sight
537    /// it is nobody's, as a storm walked into is. Whether this line is
538    /// foreign, the caster it carries when the latch says so, and whether
539    /// it is nobody's.
540    fn foreign_spell_round(
541        &mut self,
542        atk: &AttackLine,
543        text: &str,
544        eff_foreign: bool,
545    ) -> (bool, Option<Actor>, bool) {
546        let now = self.at;
547        // The gesture: a nearby player's cast arms the latch, or keeps the
548        // caster's own latch fresh. Another caster's bound spell in flight
549        // is left to finish.
550        if atk.foreign_caster
551            && atk.name == "cast"
552            && let Some(caster) = atk.attacker.as_ref().filter(|a| a.is_player())
553        {
554            match self.tracker.foreign_spell.as_mut() {
555                Some(f) if f.caster.name == caster.name => f.at = now,
556                Some(f) if f.name.is_some() && Self::fresh(f.at, now) => {}
557                _ => {
558                    self.tracker.foreign_spell = Some(ForeignSpell {
559                        caster: caster.clone(),
560                        name: None,
561                        at: now,
562                    });
563                }
564            }
565            return (eff_foreign, None, false);
566        }
567        let spell_round =
568            atk.group.contains("spell") && atk.attacker.is_none() && !starts_you(text);
569        if !spell_round {
570            return (eff_foreign, None, false);
571        }
572        let latchable = defs()
573            .family("foreign_spell")
574            .iter()
575            .any(|d| d.name == atk.name);
576        // A round that speaks to us ("the rainbow around you") is the
577        // caster's own view of it: ours, whoever gestured nearby and
578        // whether or not our opening was seen.
579        if latchable && speaks_to_us(text) {
580            return (false, None, false);
581        }
582        if eff_foreign {
583            // Foreign by the chunk's own latch: a multi-round spell's round
584            // binds the latch to the caster the gesture named. Any other
585            // spell of theirs leaves the latch as it is.
586            let caster = self.foreign_latch.as_deref().map(Actor::unlinked);
587            if latchable && let Some(caster) = &caster {
588                self.tracker.foreign_spell = Some(ForeignSpell {
589                    caster: caster.clone(),
590                    name: Some(atk.name.clone()),
591                    at: now,
592                });
593            }
594            return (true, caster, false);
595        }
596        // Our own opening of this spell, fresh: its rounds are ours, and
597        // each keeps the next fresh, so a long spell stays ours to its end.
598        if let Some((name, at)) = self.tracker.own_spell.as_mut()
599            && *name == atk.name
600            && Self::fresh(Some(*at), now)
601        {
602            if let Some(now) = now {
603                *at = now;
604            }
605            return (false, None, false);
606        }
607        if latchable
608            && let Some(f) = self.tracker.foreign_spell.as_mut()
609            && f.name.as_deref().is_none_or(|n| n == atk.name)
610            && Self::fresh(f.at, now)
611        {
612            f.name = Some(atk.name.clone());
613            f.at = now;
614            return (true, Some(f.caster.clone()), false);
615        }
616        // A round of a spell only a caster's opening starts, with neither
617        // ours nor theirs in sight: nobody's.
618        (false, None, latchable)
619    }

621    /// A nearby player's multi-round spell let go: the caster's closing
622    /// line names them; our own opening of that spell makes its rounds
623    /// ours for a while, whoever gestured nearby.
624    pub fn handle_foreign_spell(&mut self, line: &ChunkLine) {
625        let text = line.text();
626        if let Some((def, caps)) = defs().first_match_with_role("foreign_spell", Role::End, &text) {
627            let caster = caps.name("attacker").map(|m| m.as_str().trim());
628            if self.tracker.foreign_spell.as_ref().is_some_and(|f| {
629                f.name.as_deref() == Some(def.name.as_str())
630                    && caster.is_none_or(|c| c == f.caster.name)
631            }) {
632                self.tracker.foreign_spell = None;
633            }
634        } else if let Some((def, _)) =
635            defs().first_match_with_role("foreign_spell", Role::Own, &text)
636        {
637            if self
638                .tracker
639                .foreign_spell
640                .as_ref()
641                .is_some_and(|f| f.name.as_deref() == Some(def.name.as_str()))
642            {
643                self.tracker.foreign_spell = None;
644            }
645            self.tracker.own_spell = Some((def.name.clone(), self.at.unwrap_or(0)));
646        }
647    }

649    /// The pre-flare claim.
650    ///
651    /// A bow's pre-flare names the BOW while the shot names the ARROW, so a
652    /// weapon match cannot be required: a pre-flare belongs to this swing
653    /// unless a flare from a DIFFERENT weapon already sits on it. A
654    /// spell-RELEASING pre-flare rode a swing, never the spell it released:
655    /// only an attack that names a weapon may claim it.
656    fn claim_pending_flares(&mut self, id: EvId) {
657        let pending = std::mem::take(&mut self.pending_flares);
658        let mut still = Vec::new();
659        for mut f in pending {
660            // A creature's attack set it off; it is never a swing's of ours.
661            if f.reactive {
662                still.push(f);
663                continue;
664            }
665            let releasing = Self::releasing(&f.name);
666            let e = self.ev(id);
667            let claim = if releasing {
668                e.weapon.is_some() && !Self::flare_contradicts_weapon(&f, e)
669            } else {
670                Self::flare_matches_weapon(&f, e.weapon.as_deref())
671                    || !Self::flare_contradicts_weapon(&f, e)
672            };
673            if !claim {
674                still.push(f);
675                continue;
676            }
677            // fired BEFORE the swing line: never the cause of a status the
678            // swing's own crit inflicts afterwards
679            f.pre = true;
680            let cast = if releasing {
681                self.pending_release_cast.take()
682            } else {
683                None
684            };
685            if cast.is_some() {
686                f.release_claimed = true;
687            }
688            let (flare, weapon) = (f.name.clone(), f.weapon.clone());
689            self.ev_mut(id).flares.push(f);
690            // ...and the cast that flare released, which printed before this
691            // swing, is this swing's child: re-emitted after the swing.
692            if let Some(cast) = cast {
693                let root = self.ev(id).root.unwrap_or(id);
694                let c = self.ev_mut(cast);
695                c.root = Some(root);
696                c.parent = Some(id);
697                c.parent_confidence = Some(Confidence::Count);
698                c.parent_flare = Some(ParentFlare { flare, weapon });
699                self.events.retain(|&e| e != cast);
700                self.released_children.entry(id).or_default().push(cast);
701                self.deferred_casts.push(cast);
702            }
703        }
704        self.pending_flares = still;
705    }

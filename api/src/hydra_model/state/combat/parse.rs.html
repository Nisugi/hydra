1//! The combat state machine over one chunk.
2//!
3//! Turns a prompt-bounded chunk into attack events in one pass over the
4//! lines, with the branches in a fixed order -- because that order IS the
5//! behaviour: the status handler runs before the flare branch so a
6//! flare-and-status line files on the flare it announces; the switcher runs
7//! before the outcome branch so an outcome lands on the target it names; the
8//! attack branch runs last so a same-line outcome belongs to the event the
9//! line opens. Any other order misfiles a line.
10//!
11//! Each branch is a method on `cursor::Cursor` in its own module, split
12//! by what the branch handles rather than by size:
13//!
14//! | Module | Handles |
15//! |---|---|
16//! | `cursor` | the working state, the pool, the predicates |
17//! | `facts` | statuses, spell losses, UCS |
18//! | `flares` | flare announces, sequence and assault brackets |
19//! | `switch` | target switching |
20//! | `rolls` | roll lines, outcome lines |
21//! | `attack` | initiations, and every claim they make |
22//! | `damage` | damage, the coup, the crit lookahead |
23//! | `finish` | the hold, the orphan sinks, the emit |
24//!
25//! # What is not here
26//!
27//! - **Settings.** Everything is tracked, and the recorder is always
28//!   subscribed.
29//! - **Ingestion provenance.** A chunk belongs to one session's one
30//!   connection by construction.
31//! - **Room components.** They never reach the chunk; they are routed to the
32//!   room model by the stream layer.
33//! - **The registry.** Applying these events to creatures -- persisting
34//!   them, the death sweep and position recovery -- is the consumer above
35//!   this one (`creatures/apply.rs`). `LossCause::Death` is decided from this
36//!   chunk's kills alone.
37//!
38//! # A second-person status target
39//!
40//! A status def that captures a second-person target (`You are stunned!`
41//! through a `(?<target>...)` group) would otherwise go to a name lookup and
42//! be dropped by the registry. It is emitted here as `Subject::Us`
43//! (`facts.rs`), so nothing is dropped.

86/// What one line established before the branches ran.
87pub(super) struct LineCtx {
88    pub index: usize,
89    pub text: String,
90    /// ONE attack scan per line, shared by every consumer.
91    pub line_attack: Option<AttackLine>,
92    /// The line's bolded creature; `None` on an inbound line, whose only
93    /// creature link is the attacker.
94    pub line_target: Option<Actor>,
95    pub line_redirect: Option<RedirectPrefix>,
96    /// The id a status applied to on THIS line.
97    pub line_status_id: Option<i64>,
98    /// An outcome printed ON an initiation line.
99    pub same_line_outcome: Option<OutcomeKind>,
100    /// This line was a flare announce.
101    pub line_flare: bool,
102}

104/// Parse one chunk into its events and facts.
105pub(super) fn parse_chunk(
106    tracker: &mut CombatTracker,
107    chunk: &Chunk,
108    at: Option<u32>,
109) -> ChunkFacts {
110    let lines: &[ChunkLine] = chunk.lines();
111    let mut c = Cursor::new(tracker, lines, at);
112
113    // A bare gesture held over from the previous chunk: re-open it so the
114    // spell-result line that begins THIS chunk can supersede it.
115    if let Some(mut held) = c.tracker.held_cast.take() {
116        held.held = true;
117        held.born_line = None;
118        let id = c.open(held);
119        c.ev_mut(id).root = Some(id);
120        c.resume(id);
121        c.seeking_damage = true;
122    }
123    // Pre-flares held over: re-seeded so this chunk's swing claims them.
124    let mut held_flares = std::mem::take(&mut c.tracker.held_pre_flares);
125    for f in &mut held_flares {
126        f.held = true;
127    }
128    c.pending_flares.append(&mut held_flares);
129
130    for (index, line) in lines.iter().enumerate() {
131        let text = line.text();
132        if text.trim().is_empty() || defs().is_narration(&text) {
133            continue;
134        }
135        let line_attack = AttackLine::classify(line);
136        let inbound_line = line_attack.as_ref().is_some_and(|a| a.inbound);
137        let mut ctx = LineCtx {
138            index,
139            line_target: if inbound_line {
140                None
141            } else {
142                target::bolded_link(line)
143            },
144            line_redirect: RedirectPrefix::classify(line),
145            line_attack,
146            line_status_id: None,
147            same_line_outcome: None,
148            line_flare: false,
149            text,
150        };
151
152        c.handle_status(&mut ctx, line);
153        c.handle_spell_loss(line);
154        c.handle_ucs(line);
155        c.handle_flare(&mut ctx, line);
156        c.handle_ensorcell_benefit(&ctx, line);
157        c.handle_sequence(line);
158        c.handle_assault(&ctx, line);
159        c.handle_target_switch(&ctx);
160        c.handle_rolls(&mut ctx, line);
161
162        // Prefixes are modifiers on the attack line that follows, never
163        // events of their own: arm and move on.
164        let amb = AmbushPrefix::classify(line);
165        if let Some(a) = amb.clone() {
166            c.pending_ambush = Some(a);
167        }
168        if let Some(rdr) = ctx.line_redirect.clone() {
169            c.handle_redirect(&ctx, rdr);
170        }
171        let attack = if amb.is_some() || ctx.line_redirect.is_some() {
172            None
173        } else {
174            ctx.line_attack.take()
175        };
176
177        if let Some(w) = defs().nock_weapon(&ctx.text) {
178            c.nocked = Some(w);
179        }
180        if let Some(atk) = attack {
181            c.handle_attack(&ctx, line, atk);
182        } else if c.flare_ctx.is_some() || c.seeking_damage {
183            c.handle_damage(&ctx, line);
184        } else if let Some(d) = crate::state::combat::damage::DamageLine::classify(line) {
185            // Damage while seeking an attack: its initiation had no def, or
186            // lived in a prior chunk. Sink it rather than drop the fact.
187            c.orphan_hits.push(crate::state::combat::event::Hit {
188                damage: d.amount,
189                crit: None,
190                line: index,
191            });
192        }
193
194        // The line's status belongs to whichever event now holds that
195        // target: flag it so the save predicate counts it as a fact.
196        if let (Some(sid), Some(cur)) = (ctx.line_status_id, c.current)
197            && c.ev(cur).target.id() == Some(sid)
198        {
199            c.ev_mut(cur).had_status = true;
200        }
201    }
202    c.finish()
203}

205impl Cursor<'_> {
206    /// A guardian redirect prefix. UAC shape (corpus: 21 of 130): the
207    /// announce comes AFTER the attack line and the roll still lands on the
208    /// intended victim -- stamp the open attack as unhonored and arm nothing.
209    fn handle_redirect(&mut self, ctx: &LineCtx, rdr: RedirectPrefix) {
210        let interceptor = ctx
211            .line_target
212            .clone()
213            .unwrap_or_else(|| Actor::unlinked(&rdr.interceptor));
214        let unhonored = self.current.filter(|&cur| {
215            let e = self.ev(cur);
216            let open_noun = e.target.creature().map(|a| {
217                a.noun
218                    .clone()
219                    .unwrap_or_else(|| a.name.rsplit(' ').next().unwrap_or("").to_owned())
220            });
221            e.redirect.is_none()
222                && e.attack_born
223                && open_noun.as_deref().is_none_or(|n| n == rdr.intended)
224                && e.resolutions.is_empty()
225                && e.hits.is_empty()
226                && e.outcomes.is_empty()
227        });
228        let redirect = Redirect {
229            interceptor,
230            intended: rdr.intended,
231            honored: unhonored.is_none(),
232        };
233        match unhonored {
234            Some(cur) => self.ev_mut(cur).redirect = Some(redirect),
235            None => self.pending_redirect = Some(redirect),
236        }
237    }
238}

1//! The definition tables: four TSVs, compiled once, matched in data order.
2//!
3//! See the parent module for what the files are and how they were made. This
4//! file owns loading them, substituting the hand-written patterns, and the one
5//! operation every family needs: *the first def in this family that matches
6//! this text, with its captures*.
7//!
8//! # The gate
9//!
10//! Each family carries a [`RegexSet`] over its compiled patterns. One pass of
11//! the set says *which* defs match a line; only those are then run for their
12//! captures, lowest order first, so first-match-wins is exactly what a linear
13//! scan gives. Gating on a union of each pattern's longest literal would do
14//! the same job; the set does it with no literal extraction to get wrong.
15//!
16//! MEASURED (release, warm, the 85 replay blobs, `first_match` once per
17//! family per line): **105 us/line linear, 3.6 us/line gated**; the `attack`
18//! family alone went 42.6 -> 0.9. `tests/combat_gate.rs` prints its own
19//! figure and proves the two agree on every fixture line in every family.
20//!
21//! # Nothing is dropped without saying so
22//!
23//! A row whose pattern will not compile under this crate's `regex` is kept
24//! with `regex: None` and listed in [`Defs::compile_failures`]; a flagged
25//! `markup=1` row with no hand-written replacement likewise. Both are
26//! counted, and `tests/combat_defs.rs` asserts the counts are what is
27//! expected -- zero and zero -- so a data update that adds a lookbehind or a
28//! new tag-reading pattern goes red rather than silently shrinking the
29//! grammar.
30

43/// A pattern re-expressed by hand, replacing the data's row at `(family, order)`.
44#[derive(Debug, Clone, Copy)]
45pub struct HandPort {
46    /// The family whose row this replaces.
47    pub family: &'static str,
48    /// The row's position in that family's lookup, 1-based -- so first-match
49    /// order stays exactly the data's.
50    pub order: u32,
51    /// The plain-text pattern.
52    pub pattern: &'static str,
53    /// A line-level veto: when it matches, the row does not.
54    ///
55    /// The shape `crit/match_index.rs` built for the one lookahead in the crit
56    /// tables, reused for the one lookbehind here. Rust's `regex` has neither,
57    /// by design (linear time), and a positive match plus a separate veto says
58    /// the same thing.
59    pub veto: Option<&'static str>,
60}

62/// The nine patterns the extractor could not make plain, and their
63/// replacements.
64///
65/// Eight read an `exist` id out of the tag (`markup=1` in the data); the
66/// ninth is the single lookbehind in 954 rows. The id each tag-reader needs
67/// comes from the line's links (`target.rs`), which is the point: the
68/// pattern finds the sentence, the runs say who.
69///
70/// | Data | Here |
71/// |---|---|
72/// | `(?<target>.+?</a>) .+?!` (energy, sprite) | `(?<target>[^!]+)!` -- capture to the bang; the link inside it is the target. The data pattern's `</a>` was the boundary; overcapturing the text is harmless because the id is the link's, not the text's. |
73/// | `Your <a exist="(?<id>\d+)" ...>(?<name>[^<]+)</a>` | `^\*\* Your ` / `^Your ` -- the weapon is the first link whose run follows `Your `; see `flare.rs` |
74/// | the five UCS `<a exist="([0-9]+)"` forms | the sentence without the tag; the creature is the line's first link |
75/// | `(?<target>.+?)(?<! dazed and)(?<! paralyzed and) is (?:knocked\|driven) to ... knees!` | the same without the lookbehinds, plus a veto on `(?:dazed\|paralyzed) and is ...` -- a line where the "target" would have ended in the excluded words is refused whole |
76///
77/// The orders are read from the data, not remembered: the loader's own test
78/// checks them.

142/// What part a pattern plays within its family.
143///
144/// Most families have none. Statuses are `add`/`remove` pairs; brackets
145/// (assaults, sequences) are `start`/`end` pairs; the `attack_class` rows are
146/// name lists under three roles.
147#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
148pub enum Role {
149    /// No role: the family is flat.
150    None,
151    /// A status onset line.
152    Add,
153    /// A status expiry line.
154    Remove,
155    /// A bracket's opening line.
156    Start,
157    /// A bracket's closing line.
158    End,
159    /// `attack_class`: the world did it (frigid wind).
160    Environmental,
161    /// `attack_class`: our own gear did it (thorn recoil).
162    SelfInflicted,
163    /// `attack_class`: a creature maneuver aimed at the whole room.
164    RoomTargeted,
165    /// `attack_class`: another player's attack told without naming them
166    /// (a shot from hiding): never ours, never at us.
167    Anothers,
168}
169
170impl Role {
171    fn parse(text: &str) -> Self {
172        match text {
173            "add" => Self::Add,
174            "remove" => Self::Remove,
175            "start" => Self::Start,
176            "end" => Self::End,
177            "environmental" => Self::Environmental,
178            "self_inflicted" => Self::SelfInflicted,
179            "room_targeted" => Self::RoomTargeted,
180            "anothers" => Self::Anothers,
181            _ => Self::None,
182        }
183    }
184}

186/// One definition row.
187#[derive(Debug)]
188pub struct Def {
189    /// The family: `attack`, `flare`, `status`, ...
190    pub family: String,
191    /// The def's name within its family: an attack name, a status, a
192    /// resolution type, a spell number.
193    pub name: String,
194    /// Its role, where the family has roles.
195    pub role: Role,
196    /// Position in the family's first-match-wins lookup, 1-based.
197    pub order: u32,
198    /// Family-specific facts: `damaging`, `aoe`, `spawns`, `spell_name`,
199    /// `group`, `tier`.
200    pub extra: BTreeMap<String, String>,
201    /// The plain-text pattern, after the extractor's stripping or the
202    /// hand-port's substitution.
203    pub pattern: String,
204    /// Compiled, or `None` when the pattern could not be (see
205    /// [`Defs::compile_failures`]) or the row is a name list with no pattern.
206    pub regex: Option<Regex>,
207    /// A hand-port's veto: when it matches the line, this def does not.
208    pub veto: Option<Regex>,
209}
210
211impl Def {
212    /// A family-specific fact, as text.
213    #[must_use]
214    pub fn extra(&self, key: &str) -> Option<&str> {
215        self.extra.get(key).map(String::as_str)
216    }
217
218    /// A family-specific flag: `1` is true, anything else false.
219    #[must_use]
220    pub fn flag(&self, key: &str) -> bool {
221        self.extra(key) == Some("1")
222    }
223
224    /// This def's captures over `text`, unless its veto refuses the line.
225    fn captures<'t>(&self, text: &'t str) -> Option<Captures<'t>> {
226        let caps = self.regex.as_ref()?.captures(text)?;
227        if self.veto.as_ref().is_some_and(|v| v.is_match(text)) {
228            return None;
229        }
230        Some(caps)
231    }
232}

234/// One family: its defs in first-match order, and the gate over them.
235#[derive(Debug, Default)]
236struct Family {
237    defs: Vec<Def>,
238    /// The set over every def that compiled, or `None` when it would not
239    /// build -- the family then scans linearly and [`Defs::ungated`] says so.
240    gate: Option<RegexSet>,
241    /// Set index -> index into `defs`.
242    gated: Vec<usize>,
243}
244
245impl Family {
246    /// The gate's size limits. The defaults refuse the 362-pattern `attack`
247    /// family. The lazy DFA's cache keeps what it learns until this limit, a
248    /// copy for every thread that ever matches, never given back, so it is
249    /// the memory a family costs. Over a four-character afternoon's
250    /// recordings (135 MB of game stream, through the whole model), 8 MB
251    /// against 32 took each matching thread's gates from about 50 MB to
252    /// about 40 for a quarter more time, a few seconds an hour of play; at
253    /// 2 MB the gates thrash, five times slower.
254    const SIZE_LIMIT: usize = 1 << 30;
255    const DFA_SIZE_LIMIT: usize = 1 << 23;
256
257    fn build_gate(&mut self) {
258        self.gated = (0..self.defs.len())
259            .filter(|&i| self.defs[i].regex.is_some())
260            .collect();
261        let sources = self
262            .gated
263            .iter()
264            .filter_map(|&i| self.defs[i].regex.as_ref().map(Regex::as_str));
265        self.gate = RegexSetBuilder::new(sources)
266            .size_limit(Self::SIZE_LIMIT)
267            .dfa_size_limit(Self::DFA_SIZE_LIMIT)
268            .build()
269            .ok();
270    }
271
272    /// The defs that can match `text`, in first-match order. Empty -- and
273    /// unallocated -- for the great majority of lines.
274    fn candidates(&self, text: &str) -> Vec<&Def> {
275        match &self.gate {
276            // `SetMatches` iterates ascending, and `gated` is ascending.
277            Some(gate) => hydra_mem::part(hydra_mem::Part::CombatGates, || gate.matches(text))
278                .into_iter()
279                .filter_map(|i| self.gated.get(i).and_then(|&d| self.defs.get(d)))
280                .collect(),
281            None => self.defs.iter().collect(),
282        }
283    }
284}

309    DEFS.get_or_init(build)

318/// Every name a flare is read as, from the flare families' rows, without
319/// compiling a pattern: what a frontend lists to give each flare a colour
320/// of its own.
321#[must_use]
322pub fn flare_names() -> std::collections::BTreeSet<&'static str> {
323    [EFFECTS_TSV, FLARE_MIRROR_TSV]
324        .into_iter()
325        .flat_map(|tsv| tsv.lines().skip(1))
326        .filter_map(|line| {
327            let cols: Vec<&str> = line.splitn(3, '\t').collect();
328            let family = column(&cols, 0)?;
329            matches!(family, "flare" | "flare_mirror").then_some(column(&cols, 1)?)
330        })
331        .collect()
332}

334/// A TSV column, or `None` when absent or empty.
335fn column<'a>(cols: &[&'a str], index: usize) -> Option<&'a str> {
336    cols.get(index)
337        .map(|c| c.trim_end_matches('\r'))
338        .filter(|c| !c.is_empty())
339}

341fn build() -> Defs {
342    let mut out = Defs::default();
343    for tsv in [ATTACKS_TSV, RESULTS_TSV, EFFECTS_TSV, FLARE_MIRROR_TSV] {
344        for line in tsv.lines().skip(1).filter(|l| !l.is_empty()) {
345            let cols: Vec<&str> = line.splitn(7, '\t').collect();
346            let (Some(family), Some(name)) = (column(&cols, 0), column(&cols, 1)) else {
347                continue;
348            };
349            let role = column(&cols, 2).map_or(Role::None, Role::parse);
350            let order = column(&cols, 3)
351                .and_then(|o| o.parse::<u32>().ok())
352                .unwrap_or(0);
353            let flags = column(&cols, 4).unwrap_or("");
354            let extra: BTreeMap<String, String> = column(&cols, 5)
355                .unwrap_or("")
356                .split(';')
357                .filter_map(|kv| kv.split_once('='))
358                .map(|(k, v)| (k.to_owned(), v.to_owned()))
359                .collect();
360            let mut pattern = column(&cols, 6).unwrap_or("").to_owned();
361
362            // A hand-written row replaces the data's at the same position. A
363            // tag-reading row with no replacement is a fact, not an error,
364            // until the test says otherwise.
365            let port = HAND_PORTED
366                .iter()
367                .find(|p| p.family == family && p.order == order);
368            let mut veto = None;
369            match port {
370                Some(p) => {
371                    p.pattern.clone_into(&mut pattern);
372                    veto = p.veto.and_then(|v| crate::pattern::compile_gated(v).ok());
373                }
374                None if extra.get("markup").map(String::as_str) == Some("1") => {
375                    out.unported.push((family.to_owned(), order));
376                    continue;
377                }
378                None => {}
379            }
380
381            let regex = if pattern.is_empty() {
382                None
383            } else {
384                let source = if flags.contains('i') {
385                    format!("(?i){pattern}")
386                } else {
387                    pattern.clone()
388                };
389                match crate::pattern::compile_gated(&source) {
390                    Ok(re) => Some(re),
391                    Err(e) => {
392                        out.failures.push((
393                            family.to_owned(),
394                            name.to_owned(),
395                            pattern.clone(),
396                            e.to_string(),
397                        ));
398                        None
399                    }
400                }
401            };
402
403            out.by_family
404                .entry(family.to_owned())
405                .or_default()
406                .defs
407                .push(Def {
408                    family: family.to_owned(),
409                    name: name.to_owned(),
410                    role,
411                    order,
412                    extra,
413                    pattern,
414                    regex,
415                    veto,
416                });
417        }
418    }
419    for family in out.by_family.values_mut() {
420        family.defs.sort_by_key(|d| d.order);
421        family.build_gate();
422    }
423    out.addresses_self =
424        crate::pattern::compile(r"(?i)\b(?:at|towards?|upon|around|near|on)\s+your?\b").ok();
425    // The capture ends at the ` at ` that precedes the target. In plain text
426    // the target simply follows; there is no link tag to stop at.
427    out.swing_weapon = crate::pattern::compile(
428        r"^You(?: take aim and)? (?:swing|fire) (?:an? |your |some )?(?<weapon>.+?) at \S",
429    )
430    .ok();
431    out.narration = [
432        r" leaps from the back of .+? as .+? topples, narrowly avoiding being pinned",
433        r" looks a little bit more wary after that display!",
434        // a hidden creature revealed by an AoE flare: printed BEFORE the
435        // bloom line that names it (seen in a recorded hunt)
436        r" is forced out of hiding!",
437        r" gurgles out an animalistic shriek of rage, ",
438        // a sanguine ooze splitting on the hit (seen in a recorded hunt)
439        r" splatter across the ground, wobbling disconcertingly as they twitch together to form ",
440        r" pulses larger, bubbling grotesquely with new mass\.",
441        r" pulses monstrously as .+? swells to its full size!",
442        r"^You are now targeting ",
443    ]
444    .iter()
445    .filter_map(|p| crate::pattern::compile(p).ok())
446    .collect();
447    out.nock = crate::pattern::compile(r"^You nock .+? in your (?<weapon>[^.]+)\.\s*$").ok();
448    out
449}

451impl Defs {
452    /// Every def in a family, in first-match order. Empty for an unknown family.
453    #[must_use]
454    pub fn family(&self, name: &str) -> &[Def] {
455        self.by_family.get(name).map_or(&[], |f| f.defs.as_slice())
456    }

458    /// How the barrier flare `name` stops an attack (`barrier=` on its
459    /// row): `block`, before any roll, so the attack may or may not have
460    /// hit (the thorns, the mirror image); `absorb`, a hit taken instead
461    /// (bark). `None` for a flare that is no barrier, or one not yet known.
462    #[must_use]
463    pub fn barrier(&self, name: &str) -> Option<&str> {
464        ["flare", "flare_mirror"]
465            .into_iter()
466            .flat_map(|family| self.family(family))
467            .find_map(|def| (def.name == name).then(|| def.extra("barrier")).flatten())
468    }
469
470    /// Whether the flare `name` fires on a first flight only
471    /// (`first_flight=1` on its row): an arrow of a volley's later rounds
472    /// cannot carry it, where an older flare rides every round.
473    #[must_use]
474    pub fn first_flight(&self, name: &str) -> bool {
475        ["flare", "flare_mirror"]
476            .into_iter()
477            .flat_map(|family| self.family(family))
478            .any(|def| def.name == name && def.flag("first_flight"))
479    }
480
481    /// Whether the flare `name` is a gemstone's (`gemstone=1` on its row,
482    /// in the main table or the mirror), and the property's mnemonic when
483    /// the row names it (`property=`): `Some(None)` for a gemstone flare
484    /// whose property is not yet known by its game word.
485    #[must_use]
486    pub fn gemstone_property(&self, name: &str) -> Option<Option<&str>> {
487        ["flare", "flare_mirror"]
488            .into_iter()
489            .flat_map(|family| self.family(family))
490            .find(|def| def.name == name && def.flag("gemstone"))
491            .map(|def| def.extra("property"))
492    }
493
494    /// The defs in `family` that can match `text`, in first-match order.
495    fn candidates(&self, family: &str, text: &str) -> Vec<&Def> {
496        self.by_family
497            .get(family)
498            .map_or_else(Vec::new, |f| f.candidates(text))
499    }

501    /// Families with patterns whose gate did not build: they scan linearly.
502    /// `tests/combat_gate.rs` asserts there are none.
503    pub fn ungated(&self) -> impl Iterator<Item = &str> {
504        self.by_family
505            .iter()
506            .filter(|(_, f)| f.gate.is_none() && !f.gated.is_empty())
507            .map(|(name, _)| name.as_str())
508    }

510    /// The family names the tables declare.

512        self.by_family.keys().map(String::as_str)
513    }

518        &self.failures
519    }

521    /// Flagged `markup=1` rows with no hand-port: `(family, order)`.

523    pub fn unported(&self) -> &[(String, u32)] {
524        &self.unported
525    }

527    /// The first def in `family` whose pattern matches `text`, with captures.
528    ///
529    /// First match wins, in data order.
530    #[must_use]
531    pub fn first_match<'t>(&self, family: &str, text: &'t str) -> Option<(&Def, Captures<'t>)> {
532        self.candidates(family, text)
533            .into_iter()
534            .find_map(|d| d.captures(text).map(|c| (d, c)))
535    }
536
537    /// The first def in `family` with the given role that matches `text`.
538    #[must_use]
539    pub fn first_match_with_role<'t>(
540        &self,
541        family: &str,
542        role: Role,
543        text: &'t str,
544    ) -> Option<(&Def, Captures<'t>)> {
545        self.candidates(family, text)
546            .into_iter()
547            .filter(|d| d.role == role)
548            .find_map(|d| d.captures(text).map(|c| (d, c)))
549    }

551    /// Does any def in `family` match `text`?
552    #[must_use]
553    pub fn any_match(&self, family: &str, text: &str) -> bool {
554        self.candidates(family, text)
555            .into_iter()
556            .any(|d| d.captures(text).is_some())
557    }

559    /// Does a def's own pattern text address us?
560    ///
561    /// `\b(?:at|towards?|upon|around|near|on)\s+your?\b`, case-insensitive,
562    /// tested against the def's **source**: some inbound defs name us in the
563    /// literal rather than a capture (*"springs from the shadows and strikes
564    /// at you!"*), and have an attacker capture and no target capture.
565    #[must_use]
566    pub fn pattern_addresses_self(&self, pattern: &str) -> bool {
567        self.addresses_self
568            .as_ref()
569            .is_some_and(|re| re.is_match(pattern))
570    }

572    /// The weapon a 2p swing or fire line names in prose.

581            .captures(text)?

586    /// Is this narration that links a creature but states no combat fact?
587    ///
588    /// A rider leaping clear of its toppling mount, bystanders growing wary,
589    /// our own target-set echo. Fed to the target switcher, these would split
590    /// the open attack into phantom per-creature events.
591    #[must_use]
592    pub fn is_narration(&self, text: &str) -> bool {
593        self.narration.iter().any(|re| re.is_match(text))
594    }

596    /// The bow a *"You nock ... in your `<bow>`."* line names.
597    #[must_use]
598    pub fn nock_weapon(&self, text: &str) -> Option<String> {
599        self.nock
600            .as_ref()?
601            .captures(text)?
602            .name("weapon")
603            .map(|m| m.as_str().to_owned())
604    }

606    /// The names in an `attack_class` role: which attacks are environmental,
607    /// self-inflicted, or room-targeted.
608    pub fn attack_class(&self, role: Role) -> impl Iterator<Item = &str> {
609        self.family("attack_class")
610            .iter()
611            .filter(move |d| d.role == role)
612            .map(|d| d.name.as_str())
613    }

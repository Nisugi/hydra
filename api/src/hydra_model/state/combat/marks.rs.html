1//! A closed chunk's lines as the combat reader read them, for a frontend to
2//! colour: a flare, a status landing on the character, a killing blow. The
3//! reader has already matched every line; this only names what it found,
4//! so a frontend colours a flare without matching the line again.
5//!
6//! Each mark carries its line's place in the chunk and its text: a frontend
7//! that keeps lines of its own finds the line by its text.
8
9use super::event::{AttackEvent, Fact, Subject};
10use super::status::StatusAction;
11use crate::state::lines::ChunkLine;
12
13/// One line of a closed chunk, and what the combat reader found it was.
14#[derive(Debug, Clone, PartialEq, Eq)]
15pub struct LineMark {
16    /// Where the line sat in the chunk.
17    pub line: usize,
18    /// Its text.
19    pub text: String,
20    /// What it was.
21    pub kind: MarkKind,
22}
23
24/// What a marked line was.
25#[derive(Debug, Clone, PartialEq, Eq)]
26pub enum MarkKind {
27    /// A flare, by its name in the combat tables: `acid`, `air_flourish`.
28    Flare(String),
29    /// A status landing on the character, by its name: `stunned`, `webbed`.
30    StatusYou(&'static str),
31    /// A blow that killed: a crit its table calls fatal.
32    Fatal,
33}
34
35/// The marks of a chunk of `lines`, which the reader made `events` and
36/// `facts` of and on which it read each flare `announced` (the line and the
37/// flare): in line order, one a line, a killing blow over a status, a status
38/// over a flare, where one line is more than one.
39///
40/// A flare is marked by where it was announced, not by the line its event
41/// carries: a flare moves between events as it is claimed, and one held
42/// over from the chunk before keeps that chunk's line.
43pub(super) fn line_marks(
44    events: &[AttackEvent],
45    facts: &[Fact],
46    announced: &[(usize, String)],
47    lines: &[ChunkLine],
48) -> Vec<LineMark> {
49    let mut found: std::collections::BTreeMap<usize, MarkKind> = std::collections::BTreeMap::new();
50    let mut mark = |line: usize, kind: MarkKind| {
51        found.entry(line).or_insert(kind);
52    };
53    // An event held over from the chunk before -- a bare gesture re-opened
54    // here -- keeps that chunk's lines with what it had then; so does a
55    // flare held over on its own. Their crits are not this chunk's.
56    for event in events.iter().filter(|event| !event.held) {
57        let flares_hits = event
58            .flares
59            .iter()
60            .filter(|flare| !flare.held)
61            .flat_map(|flare| flare.hits.iter());
62        for hit in event.hits.iter().chain(flares_hits) {
63            if let Some(crit) = hit.crit.as_ref().filter(|crit| crit.fatal) {
64                mark(crit.line, MarkKind::Fatal);
65            }
66        }
67    }
68    for fact in facts {
69        if let Fact::Status {
70            subject: Subject::Us,
71            status,
72            action: StatusAction::Add,
73            line,
74            ..
75        } = fact
76        {
77            mark(*line, MarkKind::StatusYou(status.as_str()));
78        }
79    }
80    for (line, name) in announced {
81        mark(*line, MarkKind::Flare(name.clone()));
82    }
83    found
84        .into_iter()
85        .filter_map(|(line, kind)| {
86            Some(LineMark {
87                line,
88                text: lines.get(line)?.text(),
89                kind,
90            })
91        })
92        .collect()
93}

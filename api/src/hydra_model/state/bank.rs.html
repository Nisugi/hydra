1//! What the bank says you have.
2//!
3//! **Reading only**: the account listing and note values, plus the helpers
4//! that turn them into an answer. Depositing and withdrawing send commands
5//! and wait, so they belong above this crate.
6//!
7//! # A bank name can have `Bank` in the middle
8//!
9//! A `bank account` listing, VERIFIED on a live account:
10//!
11//! ```text
12//!      First Elanith Secured Bank: 586,836,811
13//!              Icemule Trace Bank: 3,800,267
14//!       Vornavis Bank of Solhaven: 10,627,060
15//!                 Four Winds Bank: 425,802,149
16//!              Kraken's Fall Bank: 26,743,838
17//!                          Total: 1,053,810,125
18//! ```
19//!
20//! A rule requiring the name to **end** in `Bank` would miss `Vornavis Bank
21//! of Solhaven`, and with it **10,627,060 silver**. Worse than a missing
22//! row: a balance that falls back to 0 for an unknown bank would have a
23//! character standing in Solhaven read a balance of **0** while holding 10.6
24//! million, and anything trusting that figure would behave as though the
25//! account were empty.
26//!
27//! So the separator is the **last** `": "` on the line, which is what the
28//! wire's fixed-width layout actually guarantees, and the name is whatever
29//! precedes it. No bank name has to be any particular shape.
30//!
31//! # What is read without a capture
32//!
33//! The free-to-play lines (the balance, the maximum, and the single-account
34//! `You currently have an account` opener) do **not** appear in the corpus
35//! -- no recorded account is free-to-play. They are read anyway and marked
36//! UNVERIFIED in the tests, rather than left out: a capped account reading
37//! its balance as "unknown" would be worse than reading it from an untested
38//! rule.

45/// The town a bank serves, for the banks whose name does not say it.
46pub const TOWN_OF_BANK: [(&str, &str); 1] = [("First Elanith Secured Bank", "Wehnimer's Landing")];

48/// Words of a bank's name that name no town.
49const BANK_WORDS: [&str; 5] = ["bank", "of", "the", "secured", "first"];

51/// One town's balance.
52#[derive(Clone, Debug, PartialEq, Eq)]
53pub struct BankBalance {
54    /// The bank's name as the listing spells it, e.g.
55    /// `"Vornavis Bank of Solhaven"`.
56    pub bank: String,
57    /// Silver on deposit.
58    pub silver: u64,

61/// What `bank account` reported.
62#[derive(Clone, Debug, Default, PartialEq, Eq)]
63pub struct Account {
64    /// Every town's balance, in the order the listing gave them.
65    balances: Vec<BankBalance>,
66    /// The `Total:` line, when the listing carried one.
67    ///
68    /// **Read, not summed.** A sum of the rows read would hide a dropped row
69    /// -- it is self-consistent and wrong. Keeping the game's own figure
70    /// makes a disagreement visible; see [`Self::agrees`].
71    total: Option<u64>,
72    /// A single-account balance, from `in the amount of N silvers`.
73    single: Option<u64>,
74    /// The free-to-play cap, from `a maximum of N silvers`.
75    max: Option<u64>,
76    /// Whether this bank refused access.
77    no_access: bool,
78    /// Whether a listing has been read at all.
79    known: bool,
80}

82impl Account {
83    /// Read a whole `bank account` response.
84    pub fn read_chunk(&mut self, chunk: &Chunk) -> bool {
85        let texts: Vec<String> = chunk.lines().iter().map(ChunkLine::text).collect();
86        self.read_lines(chunk.lines(), &texts)
87    }
88
89    /// [`Self::read_chunk`], given each line's text already rendered.
90    ///
91    /// # A refusal is not a listing
92    ///
93    /// This model sees no commands, only what came back -- and the refusal
94    /// line answers a `WITHDRAW` at a bank holding no account for you as
95    /// well as a `bank account`. If a chunk that merely CONTAINED the refusal
96    /// opened a fresh, empty account and replaced the listing, one refused
97    /// withdrawal would erase every balance.
98    ///
99    /// So only a listing's own opener replaces the account. A refusal outside
100    /// one sets [`Self::no_access`] -- which is true in both contexts: this
101    /// bank will not serve you -- and keeps the balances, which a refusal says
102    /// nothing about. A line someone SAID is neither.
103    ///
104    /// **Open question**: true command scoping needs the session to tell the
105    /// model which command a chunk answers, and no such input exists.
106    pub(crate) fn read_lines(&mut self, lines: &[ChunkLine], texts: &[String]) -> bool {
107        let heard = || {
108            lines
109                .iter()
110                .zip(texts)
111                .filter(|(line, _)| !line.is_spoken())
112                .map(|(_, text)| text.as_str())
113        };
114        if !heard().any(opens_account) {
115            if heard().any(|text| no_access(text.trim())) && !self.no_access {
116                self.no_access = true;
117                return true;
118            }
119            return false;
120        }
121        let mut fresh = Self {
122            known: true,
123            ..Self::default()
124        };
125        for text in heard() {
126            fresh.read_line(text);
127        }
128        let changed = *self != fresh;
129        *self = fresh;
130        changed
131    }
132
133    /// Read one line of the response.
134    fn read_line(&mut self, text: &str) {
135        let trimmed = text.trim();
136        if no_access(trimmed) {
137            self.no_access = true;
138            return;
139        }
140        // **The listing's rows AND its total are indented; the prose around
141        // them is not.** That is what separates a balance from a sentence
142        // quoting a figure, and it applies to `Total:` for the same reason it
143        // applies to a town row -- an unindented `Total: 9` is something
144        // being said, not the listing speaking.
145        let indented = text.len() != trimmed.len();
146        if !indented {
147            return;
148        }
149        if let Some(silver) = trimmed
150            .strip_prefix("Total:")
151            .and_then(|rest| number(rest.trim()))
152        {
153            self.total = Some(silver);
154            return;
155        }
156        // `in the amount of N silvers` / `a maximum of N silvers`.
157        if let Some(silver) = after(trimmed, "in the amount of") {
158            self.single = Some(silver);
159        }
160        if let Some(silver) = after(trimmed, "a maximum of") {
161            self.max = Some(silver);
162        }
163        // A town row. **Split on the LAST `": "`**, not on a `" Bank: "`
164        // literal -- see the module doc for the 10.6 million silver that
165        // costs.
166        if let Some((bank, amount)) = trimmed.rsplit_once(": ")
167            && let Some(silver) = number(amount)
168            && !bank.is_empty()
169        {
170            self.balances.push(BankBalance {
171                bank: bank.to_owned(),
172                silver,
173            });
174        }
175    }
176
177    /// Every town's balance, as the listing gave them.
178    #[must_use]
179    pub fn balances(&self) -> &[BankBalance] {
180        &self.balances
181    }
182
183    /// The balance at a named bank.
184    #[must_use]
185    pub fn at(&self, bank: &str) -> Option<u64> {
186        self.balances
187            .iter()
188            .find(|b| b.bank == bank)
189            .map(|b| b.silver)
190    }
191
192    /// The bank serving a town, by the room's location.
193    ///
194    /// Three passes: an exact prefix first, then a looser containment
195    /// either way, then the town's own word in the location (`Solhaven` of
196    /// `the free port of Solhaven`, for `Vornavis Bank of Solhaven`). A bank
197    /// whose name says no town is known by [`TOWN_OF_BANK`]. An empty
198    /// location matches nothing.
199    #[must_use]
200    pub fn local(&self, location: &str) -> Option<&BankBalance> {
201        let location = location.trim();
202        if location.is_empty() {
203            return None;
204        }
205        // Matched against the bank's name with `Bank` REMOVED, because that
206        // word is in the name and never in the location: the row says
207        // `Four Winds Bank` and the room says `Four Winds Isle`.
208        //
209        // Keeping the whole name in the stored fact is what lets `Vornavis
210        // Bank of Solhaven` be read at all (see the module doc), so the
211        // stripping happens here, the one place that needs it.
212        self.balances
213            .iter()
214            .find(|b| location.starts_with(town_of(&b.bank)))
215            .or_else(|| {
216                self.balances.iter().find(|b| {
217                    let town = town_of(&b.bank);
218                    location.contains(town) || town.contains(location)
219                })
220            })
221            .or_else(|| {
222                let lower = location.to_lowercase();
223                self.balances.iter().find(|b| {
224                    let town = TOWN_OF_BANK
225                        .iter()
226                        .find_map(|(bank, town)| (*bank == b.bank).then_some(*town));
227                    match town {
228                        Some(town) => lower.contains(&town.to_lowercase()),
229                        None => b
230                            .bank
231                            .split_whitespace()
232                            .filter(|word| !BANK_WORDS.contains(&word.to_lowercase().as_str()))
233                            .any(|word| lower.contains(&word.to_lowercase())),
234                    }
235                })
236            })
237    }

239    /// The total the game stated, or `None` if it stated none.
240    ///
241    /// **Not a sum of [`Self::balances`].** See the field doc.
242    #[must_use]
243    pub fn total(&self) -> Option<u64> {
244        self.total
245    }
246
247    /// The sum of the rows this listing was read from.
248    #[must_use]
249    pub fn sum_of_balances(&self) -> u64 {
250        self.balances.iter().map(|b| b.silver).sum()
251    }
252
253    /// Whether the stated total matches the rows read.
254    ///
255    /// `None` when the game stated no total. **`Some(false)` means a row was
256    /// missed** -- the condition that would hide a bank's silver, and which a
257    /// sum-based total cannot express.
258    #[must_use]
259    pub fn agrees(&self) -> Option<bool> {
260        self.total.map(|total| total == self.sum_of_balances())
261    }
262
263    /// A single-account balance, where the bank reports one figure.
264    #[must_use]
265    pub fn single(&self) -> Option<u64> {
266        self.single
267    }
268
269    /// The free-to-play cap, when the bank stated one.
270    #[must_use]
271    pub fn max(&self) -> Option<u64> {
272        self.max
273    }
274
275    /// Whether this bank refused access.
276    #[must_use]
277    pub fn no_access(&self) -> bool {
278        self.no_access
279    }
280
281    /// Whether any `bank account` response has been read.
282    #[must_use]
283    pub fn is_known(&self) -> bool {
284        self.known
285    }
286}

288/// A bank note the game has described.
289#[derive(Clone, Debug, PartialEq, Eq)]
290pub struct Note {
291    /// The note's `exist` id, when the line carried a link.
292    pub id: Option<String>,
293    /// What it is worth.
294    pub silver: u64,
295}
296
297/// Read a `read <note>` reply.
298///
299/// VERIFIED on real wire -- the corpus carries the form with a link, and the
300/// note's id is kept from it:
301///
302/// ```text
303/// The Mist Harbor <a exist="46215417" noun="note">promissory note</a>
304///   has a value of 42,170 silver and reads, "Hold in right hand to use."
305/// ```
306#[must_use]
307pub fn note_line(line: &ChunkLine) -> Option<Note> {
308    let text = line.text();
309    let silver = after(&text, "has a value of")?;
310    // `and reads` is what separates a note's own value from any other
311    // sentence carrying the phrase.
312    if !text.contains("and reads") {
313        return None;
314    }
315    let id = line
316        .objects()
317        .find(|link| {
318            matches!(&link.kind, hydra_protocol::frame::LinkKind::Exist { noun, .. }
319                if NOTE_NOUNS.contains(&noun.as_str()))
320        })
321        .and_then(|link| match &link.kind {
322            hydra_protocol::frame::LinkKind::Exist { id, .. } => Some(id.clone()),
323            _ => None,
324        });
325    Some(Note { id, silver })
326}

328/// A bank's name with the word `Bank` taken out.
329///
330/// `"Four Winds Bank"` -> `"Four Winds"`, and
331/// `"Vornavis Bank of Solhaven"` -> `"Vornavis of Solhaven"`. Used only for
332/// matching a room's location, never for the name a caller sees.
333fn town_of(bank: &str) -> &str {
334    bank.strip_suffix(" Bank").unwrap_or(bank)
335}
336
337/// Whether a line opens a `bank account` LISTING.
338///
339/// The refusal is not an opener: see [`Account::read_lines`] for why a
340/// refusal alone does not open one.
341fn opens_account(text: &str) -> bool {
342    let trimmed = text.trim();
343    trimmed.starts_with("You currently have the following amounts on deposit")
344        || trimmed.starts_with("You currently have an account")
345}
346
347/// The no-access refusal, case-insensitive.
348fn no_access(text: &str) -> bool {
349    text.to_lowercase().contains("you don't have access")
350}
351
352/// The number following a phrase, e.g. `"a maximum of 50,000 silvers"` -> 50000.
353fn after(text: &str, phrase: &str) -> Option<u64> {
354    let (_, rest) = text.split_once(phrase)?;
355    number(rest.split_whitespace().next()?)
356}
357
358/// A comma-grouped silver figure, forgiving only the sentence's own trailing
359/// punctuation (`"50,000."`). The number itself goes through the shared strict
360/// reader (`state/numbers.rs`).
361fn number(text: &str) -> Option<u64> {
362    crate::state::numbers::grouped(text.trim().trim_end_matches(['.', ',']))
363}

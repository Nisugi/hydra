1//! What a new connection has not been told: the reconnect invalidation.
2//!
3//! A dropped connection is followed by a new one whose login burst restates
4//! most of the character, but not all of it, and not at once. Anything that
5//! could have changed while the character was out of the world, and that the
6//! burst does not restate before it is read, is forgotten here so it reads as
7//! "not told", never as what was true before the drop. What the character is
8//! (stats, skills, standing, the world around it) is kept.
9//!
10//! The session's supervisor calls this when it reconnects.
11
12use super::GameState;
13use super::group::Group;
14
15impl GameState {
16    /// Forget what a new connection has not been told.
17    ///
18    /// Every field named, so a new one is a compile error here and whoever
19    /// adds it decides which side it belongs on.
20    #[expect(
21        clippy::too_many_lines,
22        reason = "names every field, so a new one must be placed on a side"
23    )]
24    pub fn invalidate_for_reconnect(&mut self) {
25        // The server's second as the connection ended: what effects have
26        // left is counted to here, not to the last prompt, which may be
27        // long before it.
28        let ended_at = self.game_time_now();
29        let Self {
30            room,
31            ship,
32            arrivals,
33            cooldowns,
34            messages,
35            overwatch,
36            known_spells,
37            worn,
38            reserve,
39            society_said,
40            bounty,
41            kits,
42            order_menu,
43            targeting,
44            maneuvers,
45            cast_time_ends,
46            stun_ends,
47            aiming,
48            aim_said,
49            prepared,
50            prepared_id,
51            prompt,
52            left_hand,
53            right_hand,
54            roundtime_ends,
55            vitals,
56            objectives,
57            status,
58            effects,
59            game_time,
60            game_time_received,
61            unknown_tags,
62            unknown_tag_counts,
63            idle_warning,
64            streams,
65            retained,
66            stream_windows,
67            tally,
68            combat,
69            incidents,
70            world,
71            creatures,
72            pending,
73            chunk,
74            gemstones_published,
75            worn_published,
76            effects_published,
77            bark_up,
78            character,
79            inventory,
80            inventory_snapshot,
81            group,
82            containers,
83            bank,
84            learned_commands,
85            dialogs,
86            custom_timers,
87        } = self;
88        // --- Forgotten: who and what is here, and what was under way -------
89        room.forget_contents();
90        // A ship left at sea is back at its port by the next login.
91        ship.lapse();
92        *group = Group::default();
93        containers.lapse();
94        cooldowns.clear();
95        overwatch.clear();
96        known_spells.clear();
97        bounty.clear();
98        // The game says the society again at the next login.
99        society_said.clear();
100        // Kits are known by item id, and every item id changes at a login.
101        kits.clear();
102        targeting.clear();
103        maneuvers.clear();
104        *stun_ends = None;
105        (*prepared, *prepared_id) = (None, None);
106        // Effects keep what they had LEFT, dated again by the next prompt.
107        if let Some(then) = ended_at {
108            effects.unanchor(then);
109        }
110        for id in super::afflictions::Affliction::ALL {
111            status.forget(id.id());
112        }
113        // Who leads goes with the group: unknown until a line says again.
114        status.forget("leader");
115        inventory_snapshot.lapse();
116        worn.drop_partial();
117        // Known by item id, which a login changes, and not listed again by
118        // it: unknown, to be asked for.
119        *reserve = super::worn::Reserve::default();
120        *game_time = None;
121        *game_time_received = super::Received(None);
122        *idle_warning = super::IdleWarning::None;
123        streams.clear();
124        pending.clear();
125        *stream_windows = super::stream_windows::Windows::default();
126        combat.invalidate_for_reconnect();
127        creatures.invalidate_for_reconnect();
128        *chunk = super::lines::Chunk::default();
129        *prompt = None;
130        character.invalidate_for_reconnect();
131        *inventory = super::Inventory::default();
132        world.pulse = None;
133
134        // --- Kept: facts the drop did not change, or the burst restates -----
135        let _ = (
136            gemstones_published,
137            worn_published,
138            effects_published,
139            bark_up,
140            arrivals,
141            messages,
142            bank,
143            order_menu,
144            aiming,
145            aim_said,
146            left_hand,
147            right_hand,
148            roundtime_ends,
149            // An absolute server time, as roundtime is: cleared,
150            // `in_casttime` would say no during a live one.
151            cast_time_ends,
152            vitals,
153            objectives,
154            learned_commands,
155            dialogs,
156            // The player's own, which the game neither ends nor restates.
157            custom_timers,
158            tally,
159            // A consumer's choice, which the drop does not change.
160            retained,
161            incidents,
162            unknown_tags,
163            unknown_tag_counts,
164        );
165    }
166}

168#[cfg(test)]
169mod tests {
170    use std::time::{Duration, Instant};
171
172    use hydra_protocol::Parser;
173
174    use super::GameState;
175
176    fn feed(state: &mut GameState, parser: &mut Parser, wire: &str) {
177        for frame in parser.push_bytes(wire.as_bytes()) {
178            state.apply(&frame);
179        }
180    }
181
182    /// An effect keeps what it had left when the connection ended, not what
183    /// it had at the last prompt: time spent since is spent. Celerity, 104
184    /// seconds at the prompt, has run out two minutes later, and stays out.
185    #[test]
186    fn an_effect_keeps_what_it_had_left_when_the_connection_ended() {
187        let mut state = GameState::default();
188        let mut parser = Parser::new();
189        feed(
190            &mut state,
191            &mut parser,
192            "<dialogData id='Buffs'><progressBar id='506' value='100' text=\"Celerity\" \
193             time='00:01:44'/></dialogData>\n<prompt time=\"500\">&gt;</prompt>\n",
194        );
195        // The connection ends two minutes after that prompt.
196        state.game_time_received =
197            super::super::Received(Instant::now().checked_sub(Duration::from_mins(2)));
198        state.invalidate_for_reconnect();
199        // The new connection's first prompt, an hour of server time later.
200        feed(
201            &mut state,
202            &mut parser,
203            "<prompt time=\"4100\">&gt;</prompt>\n",
204        );
205        assert_eq!(state.effects.active("506", 4100), Some(false));
206    }

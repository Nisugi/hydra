1//! Hazards: what in a room hurts whoever stays, whether the game draws it as
2//! an object or as a creature.
3//!
4//! # Two tables, because the game draws hazards two ways
5//!
6//! | drawn as | table |
7//! |---|---|
8//! | a room object, unbolded in `room objs` | [`OBJECTS`], 13 names |
9//! | a creature, bolded, with a `<crtrStatus>` | [`CREATURES`], 2 names |
10//!
11//! The object-type table (`data/gameobj-data.tsv`) has no hazard category
12//! (`cut -f2 gameobj-data.tsv | sort -u`), which is why these lists exist. A
13//! `gas cloud` object is one of [`OBJECTS`]' `cloud`.
14//!
15//! # The two creatures a hunt would otherwise attack
16//!
17//! Both are ordinary bestiary rows with hit points (`data/creatures.tsv`,
18//! `shimmering_fungus` and `wasp_nest`), and `gameobj-data.tsv`'s
19//! `aggressive npc` names both, so a target list holding `any` would attack
20//! them. A hunt should refuse them: drop them from its targets and leave
21//! when nothing else is left, or walk away from a wasp nest. The fungus
22//! summons maw spores (`creatures.tsv`'s `special_other`), and its shudder
23//! warns that the ground is about to erupt.
24//!
25//! # Not here
26//!
27//! The gas-cloud warning lines (a cloud forming above you, sparks, rumbling,
28//! dispersing). They are a room's state over time, which the hunt's flee
29//! messages already read as phrases.
30

38/// The room objects that are hazards. Each matches as whole words anywhere in the name, ignoring
39/// case, so `cloud` is also a gas cloud and `web` a sticky web.
40pub const OBJECTS: [&str; 13] = [
41    "acidic cloud of mist",
42    "glimmering boltstone apparatus",
43    "pale hovering runestone",
44    "cloud",
45    "unearthly silvery blue globe",
46    "spiraling ghostly rift",
47    "sandstorm",
48    "vine",
49    "black metal bomb",
50    "black void",
51    "windy vortex",
52    "web",
53    "whirlwind",
54];
55
56/// The creatures a hunt never targets, matched as the whole name, ignoring
57/// case.
58pub const CREATURES: [&str; 2] = ["shimmering fungus", "wasp nest"];
59
60fn objects() -> &'static Pat {
61    static P: OnceLock<Pat> = OnceLock::new();
62    P.get_or_init(|| {
63        let names = OBJECTS.map(regex::escape).join("|");
64        Pat::new(&format!(r"(?i)\b(?:{names})\b"))
65    })
66}
67
68/// Is a room object with this name a hazard? One of [`OBJECTS`] as whole
69/// words, and not a disk: any name holding `disk` is excluded.
70#[must_use]
71pub fn is_object(name: &str) -> bool {
72    !name.to_ascii_lowercase().contains("disk") && objects().is_match(name)
73}
74
75/// Is a creature with this display name a hazard? The whole name is one of
76/// [`CREATURES`], ignoring case.
77#[must_use]
78pub fn is_creature(name: &str) -> bool {
79    let name = name.trim();
80    CREATURES.iter().any(|c| c.eq_ignore_ascii_case(name))
81}
82
83impl Room {
84    /// The hazard objects lying in the room. Empty is the
85    /// ordinary answer, and also the answer before `room objs` has been seen.
86    pub fn hazards(&self) -> impl Iterator<Item = &RoomItem> {
87        self.objects.iter().filter(|item| is_object(&item.text))
88    }
89}
90
91impl CreatureInstance {
92    /// **Is it a hazard rather than a target?** A wasp nest or a shimmering
93    /// fungus, by its name or its name with a boon adjective stripped, as the
94    /// bestiary retries a lookup. `false` is every other creature: the list
95    /// is two names long, so `false` means "not one of those", not "safe".
96    #[must_use]
97    pub fn hazard(&self) -> bool {
98        is_creature(&self.name) || without_boon(&self.name).is_some_and(|n| is_creature(&n))
99    }
100}

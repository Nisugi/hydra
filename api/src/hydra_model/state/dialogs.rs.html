1//! The game's own dialogs that nothing else in the model reads: what they
2//! hold, kept so a viewer can draw them.
3//!
4//! The parser already types every dialog frame, and the model reads a few
5//! by name: the vitals, experience, stance,
6//! encumbrance, the injuries and the effects, each into its own place. The
7//! rest reach the catch-all in [`GameState::apply`](super::GameState::apply):
8//! the Betrayer panel's blood points and items, the combat
9//! panel's buttons, and any panel an event adds. Here each is kept by its
10//! id, its title, and its parts in the order they came, as the wire said
11//! them: a label's text, a bar's percent and words, and any other widget as
12//! its attributes (`cmdButton`, `link`, `dropDownBox`, `image`, ...).
13//!
14//! **Only dialogs no one else claims**: a reader claims exact ids only, and
15//! anything else falls through to a generic renderer. A claimed dialog has a
16//! better home already, and keeping it twice would keep the busiest frames
17//! on the wire twice.
18//!
19//! The aim timer is the one dialog read as a fact rather than kept: its
20//! `<timer>` is an absolute server end time, `0` once it is over, as the
21//! cast time is.
22//!
23//! Bounded: [`MAX_DIALOGS`] dialogs, the one changed longest ago let go
24//! first, and [`MAX_PARTS`] parts in each.
25
26use hydra_protocol::frame::{Attrs, Frame};
27
28/// Dialogs kept at most.
29pub const MAX_DIALOGS: usize = 32;
30
31/// Parts kept in one dialog at most.
32pub const MAX_PARTS: usize = 128;
33
34/// The dialog the aim timer comes in.
35const AIM: &str = "AimTimerDialog";
36
37/// Whether the model reads `id` elsewhere, so it is not kept here.
38fn claimed(id: &str) -> bool {
39    matches!(
40        id,
41        "minivitals" | "expr" | "encum" | "stance" | "injuries" | "mapViewMain" | AIM
42    ) || crate::state::effects::is_effect_dialog(id)
43        // Another character's injuries, one dialog each: the doll of
44        // whoever was appraised, left out of the model on purpose
45        // (`state/vitals.rs`).
46        || id.starts_with("injuries-")
47}

49/// One part of a dialog, as the game sent it.
50#[derive(Clone, Debug, PartialEq, Eq)]
51pub enum Part {
52    /// `<label>`: its text, and every attribute as sent: where it sits
53    /// (`top`, `left`, `align`, its anchors) and how it is aligned
54    /// (`justify`).
55    Label {
56        /// `value=`.
57        text: String,
58        /// Its attributes, in wire order.
59        attrs: Attrs,
60    },
61    /// `<progressBar>`: its percent, its words, and every attribute as sent.
62    Bar {
63        /// `value=`.
64        percent: u32,
65        /// `text=`.
66        text: String,
67        /// Its attributes, in wire order.
68        attrs: Attrs,
69    },
70    /// Any other widget: its tag, and every attribute as sent.
71    Widget {
72        /// `cmdButton`, `link`, `dropDownBox`, `editBox`, ...
73        kind: String,
74        /// Its attributes, in wire order.
75        attrs: Attrs,
76    },
77}

79/// One dialog: its title, whether it is open, and its parts by id.
80#[derive(Clone, Debug, Default, PartialEq, Eq)]
81pub struct Dialog {
82    /// `title=` from its `<openDialog>`, once one came.
83    pub title: Option<String>,
84    /// Open: the game opened or filled it and has not closed it.
85    pub open: bool,
86    /// Every attribute of its last `<openDialog>`, in wire order: where it
87    /// goes (`location=`), its kind, its size.
88    pub attrs: Vec<(String, String)>,
89    /// Opened by a script of the player's Lich, not by the game
90    /// ([`Dialogs::scripted`]): its links are the player's own, as typed.
91    pub scripted: bool,
92    /// Its parts, by their `id=`, in the order each first came.
93    pub parts: Vec<(String, Part)>,
94    /// When it last changed, in dialogs changed: which one goes first.
95    changed: u64,
96}

98/// The dialogs no one else reads, and the aim timer.
99#[derive(Clone, Debug, Default, PartialEq, Eq)]
100pub struct Dialogs {
101    by_id: std::collections::BTreeMap<String, Dialog>,
102    changes: u64,
103    /// When an aimed shot's aim is done, by the server's clock.
104    pub aim_ends: Option<u32>,
105    /// The quick bar the game last asked to show (`<switchQuickBar id=>`):
106    /// `quick`, `quick-combat` or `quick-simu`. `None` until it has said.
107    pub quick_bar: Option<String>,
108}

110impl Dialog {
111    /// Where the game put it, `location=`: `quickBar` for a quick bar.
112    #[must_use]
113    pub fn location(&self) -> Option<&str> {
114        self.attrs
115            .iter()
116            .find(|(name, _)| name == "location")
117            .map(|(_, value)| value.as_str())
118    }
119}
120
121impl Dialogs {
122    /// The dialog `id`, if the game has sent it.
123    #[must_use]
124    pub fn get(&self, id: &str) -> Option<&Dialog> {
125        self.by_id.get(id)
126    }
127
128    /// Every dialog kept, by id.
129    pub fn iter(&self) -> impl Iterator<Item = (&str, &Dialog)> {
130        self.by_id.iter().map(|(id, dialog)| (id.as_str(), dialog))
131    }
132
133    /// `frame`, folded in already, came from a script of the player's Lich
134    /// and not the game: a dialog it opens is the script's.
135    pub fn scripted(&mut self, frame: &Frame) {
136        if let Frame::DialogOpen { id, .. } | Frame::DialogPanelOpen { id, .. } = frame
137            && let Some(dialog) = self.by_id.get_mut(id)
138        {
139            dialog.scripted = true;
140        }
141    }
142
143    /// Fold `frame` in, when it is one of a dialog kept here.
144    pub fn read(&mut self, frame: &Frame) {
145        match frame {
146            Frame::DialogOpen { id, title, attrs }
147            | Frame::DialogPanelOpen { id, title, attrs } => {
148                if let Some(dialog) = self.touch(id) {
149                    dialog.open = true;
150                    dialog.scripted = false;
151                    if title.is_some() {
152                        dialog.title.clone_from(title);
153                    }
154                    // `<dialogData>` opens one too, with attributes of its
155                    // own: each kept by its name, the rest left as they were.
156                    for (name, value) in attrs {
157                        match dialog.attrs.iter_mut().find(|(each, _)| each == name) {
158                            Some((_, was)) => was.clone_from(value),
159                            None => dialog.attrs.push((name.clone(), value.clone())),
160                        }
161                    }
162                }
163            }
164            Frame::ClearDialogData { id } => {
165                if let Some(dialog) = self.touch(id) {
166                    dialog.parts.clear();
167                }
168            }
169            Frame::CloseDialog { id } => {
170                if let Some(dialog) = self.by_id.get_mut(id) {
171                    dialog.open = false;
172                }
173            }
174            Frame::Label {
175                id,
176                value,
177                dialog: Some(dialog),
178                attrs,
179            } => self.put(
180                dialog,
181                id,
182                Part::Label {
183                    text: value.clone(),
184                    attrs: attrs.clone(),
185                },
186            ),
187            Frame::ProgressBar(bar) => {
188                if let Some(dialog) = &bar.dialog {
189                    let part = Part::Bar {
190                        percent: bar.percent,
191                        text: bar.text.clone(),
192                        attrs: bar.attrs.clone(),
193                    };
194                    self.put(dialog, &bar.id, part);
195                }
196            }
197            Frame::DialogWidgets(widgets) => {
198                let Some(dialog) = &widgets.dialog else {
199                    return;
200                };
201                for attrs in &widgets.widgets {
202                    let id = attrs
203                        .iter()
204                        .find(|(name, _)| name == "id")
205                        .map_or("", |(_, value)| value.as_str());
206                    let part = Part::Widget {
207                        kind: widgets.kind.clone(),
208                        attrs: attrs.clone(),
209                    };
210                    self.put(dialog, id, part);
211                }
212            }
213            // An image is a button when it carries `cmd=`: the combat
214            // panel's sheathe and shield buttons are.
215            Frame::InjuryImage {
216                id,
217                dialog: Some(dialog),
218                attrs,
219                ..
220            } => {
221                let part = Part::Widget {
222                    kind: "image".to_owned(),
223                    attrs: attrs.clone(),
224                };
225                self.put(dialog, id, part);
226            }
227            Frame::AimTime { value } => self.aim_ends = (*value != 0).then_some(*value),
228            Frame::QuickbarSwitch { id } if !id.is_empty() => self.quick_bar = Some(id.clone()),
229            _ => {}
230        }
231    }
232
233    /// `part` as `id` in the dialog `dialog`, replacing the one of that id,
234    /// or after the rest.
235    fn put(&mut self, dialog: &str, id: &str, part: Part) {
236        let Some(dialog) = self.touch(dialog) else {
237            return;
238        };
239        dialog.open = true;
240        // A part with no id is one of its own, a script's separator: after
241        // the rest, never in another's place.
242        if let Some((_, was)) = dialog
243            .parts
244            .iter_mut()
245            .find(|(each, _)| !id.is_empty() && each == id)
246        {
247            *was = part;
248        } else if dialog.parts.len() < MAX_PARTS {
249            dialog.parts.push((id.to_owned(), part));
250        }
251    }
252
253    /// The dialog `id`, made if new and marked changed; `None` when it is
254    /// claimed or has no id. Making one past [`MAX_DIALOGS`] lets go of the
255    /// one changed longest ago.
256    fn touch(&mut self, id: &str) -> Option<&mut Dialog> {
257        if id.is_empty() || claimed(id) {
258            return None;
259        }
260        if !self.by_id.contains_key(id) && self.by_id.len() >= MAX_DIALOGS {
261            let oldest = self
262                .by_id
263                .iter()
264                .min_by_key(|(_, dialog)| dialog.changed)
265                .map(|(id, _)| id.clone());
266            if let Some(oldest) = oldest {
267                self.by_id.remove(&oldest);
268            }
269        }
270        self.changes += 1;
271        let changed = self.changes;
272        let dialog = self.by_id.entry(id.to_owned()).or_default();
273        dialog.changed = changed;
274        Some(dialog)
275    }
276}

278impl super::GameState {
279    /// Seconds until an aimed shot's aim is done: `Some(0)` with none
280    /// running, `None` before the game has said what time it is. As
281    /// [`Self::roundtime_remaining`] reads the roundtime.
282    #[must_use]
283    pub fn aim_remaining(&self) -> Option<u32> {
284        let now = self.game_time_now()?;
285        Some(
286            self.dialogs
287                .aim_ends
288                .map_or(0, |ends| ends.saturating_sub(now)),
289        )
290    }

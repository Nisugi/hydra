1//! What the game says of the world rather than the character: when the next
2//! pulse comes, the world events under way, and the Ebon Gate merchants at
3//! work ([`merchants`]).
4//!
5//! A `<pulse min= max=>` bounds when the *next* pulse arrives, anchored on the
6//! server clock at arrival; a `<worldEvent expires=>` lapses that many
7//! minutes after it arrived, and lapsed events are pruned when a new one
8//! comes. Here they are kept whole and filtered when read, so nothing is
9//! lost between two events.
10
11pub mod merchants;
12
13/// The last pulse the game announced, and so when the next is due.
14#[derive(Clone, Debug, PartialEq, Eq)]
15pub struct Pulse {
16    /// The server second it arrived; `None` before the game's clock is
17    /// known.
18    pub at: Option<u32>,
19    /// The fewest seconds until the next pulse.
20    pub min: u32,
21    /// The most seconds until the next pulse.
22    pub max: u32,
23    /// Whether the next pulse is a mana pulse.
24    pub mana: bool,
25}
26
27impl Pulse {
28    /// The seconds until the next pulse can come and must have come, at
29    /// server second `now`: `(0, 0)` once it is overdue.
30    #[must_use]
31    pub fn due(&self, now: u32) -> Option<(u32, u32)> {
32        let at = self.at?;
33        let passed = now.saturating_sub(at);
34        Some((
35            self.min.saturating_sub(passed),
36            self.max.saturating_sub(passed),
37        ))
38    }
39}

41/// One world event the game announced.
42#[derive(Clone, Debug, PartialEq, Eq)]
43pub struct WorldEvent {
44    /// Where, as the game names it; `None` when it did not say.
45    pub realm: Option<String>,
46    /// What, in the game's words.
47    pub text: String,
48    /// The server second it lapses; `None` when the game gave no expiry, or
49    /// its clock was not yet known.
50    pub expires_at: Option<u32>,
51}
52
53/// The world's pulse and events, as the game has told them.
54#[derive(Clone, Debug, Default, PartialEq, Eq)]
55pub struct World {
56    /// The last pulse; `None` before the first.
57    pub pulse: Option<Pulse>,
58    /// Every event announced, oldest first.
59    events: Vec<WorldEvent>,
60    /// The Ebon Gate merchants at work.
61    pub merchants: merchants::Merchants,
62}
63
64/// The most events kept, lapsed or not: a bound on what a long run holds.
65const MAX_EVENTS: usize = 32;
66
67impl World {
68    /// A `<pulse>` arrived at server second `now`.
69    pub(crate) fn pulsed(&mut self, now: Option<u32>, min: u32, max: u32, mana: bool) {
70        self.pulse = Some(Pulse {
71            at: now,
72            min,
73            max,
74            mana,
75        });
76    }
77
78    /// A `<worldEvent>` arrived at server second `now`: lapsed ones go.
79    pub(crate) fn announced(
80        &mut self,
81        now: Option<u32>,
82        realm: &str,
83        expires_min: Option<u32>,
84        text: &str,
85    ) {
86        if let Some(now) = now {
87            self.events
88                .retain(|event| event.expires_at.is_none_or(|at| at > now));
89        }
90        self.events.push(WorldEvent {
91            realm: (!realm.is_empty()).then(|| realm.to_owned()),
92            text: text.to_owned(),
93            expires_at: now
94                .zip(expires_min)
95                .map(|(now, min)| now.saturating_add(min.saturating_mul(60))),
96        });
97        if self.events.len() > MAX_EVENTS {
98            self.events.remove(0);
99        }
100    }
101
102    /// The events under way at server second `now`, oldest first: every
103    /// one, when the clock is not known.
104    pub fn events(&self, now: Option<u32>) -> impl Iterator<Item = &WorldEvent> {
105        self.events
106            .iter()
107            .filter(move |event| now.zip(event.expires_at).is_none_or(|(now, at)| at > now))
108    }
109}

111#[cfg(test)]
112mod tests {
113    use super::*;
114
115    /// A pulse bounds the next: due between its least and most seconds,
116    /// counting down, and overdue at none.
117    #[test]
118    fn a_pulse_says_when_the_next_is_due() {
119        let mut world = World::default();
120        assert_eq!(world.pulse, None, "before the first");
121        world.pulsed(Some(1_000), 46, 75, true);
122        let pulse = world.pulse.clone().expect("pulsed");
123        assert!(pulse.mana);
124        assert_eq!(pulse.due(1_000), Some((46, 75)));
125        assert_eq!(pulse.due(1_050), Some((0, 25)));
126        assert_eq!(pulse.due(2_000), Some((0, 0)));
127        world.pulsed(None, 46, 75, false);
128        assert_eq!(
129            world.pulse.and_then(|pulse| pulse.due(1_000)),
130            None,
131            "no clock"
132        );
133    }
134
135    /// An event lapses its minutes after it came; a lapsed one is not
136    /// under way, and goes when the next arrives.
137    #[test]
138    fn a_world_event_lapses() {
139        let mut world = World::default();
140        world.announced(
141            Some(1_000),
142            "Wehnimer's Landing",
143            Some(10),
144            "An invasion begins!",
145        );
146        world.announced(Some(1_000), "", None, "The moons align.");
147        let under_way = |world: &World, now| -> Vec<String> {
148            world.events(now).map(|event| event.text.clone()).collect()
149        };
150        assert_eq!(
151            under_way(&world, Some(1_599)),
152            ["An invasion begins!", "The moons align."]
153        );
154        assert_eq!(under_way(&world, Some(1_600)), ["The moons align."]);
155        assert_eq!(under_way(&world, None).len(), 2, "no clock: every one");
156        let realms: Vec<Option<&str>> = world
157            .events(None)
158            .map(|event| event.realm.as_deref())
159            .collect();
160        assert_eq!(realms, [Some("Wehnimer's Landing"), None], "none said");
161        // Minutes no clock can hold: never lapsing, not a wrapped second
162        // in the past, nor a panic in a debug build.
163        world.announced(Some(1_000), "", Some(u32::MAX), "Forever.");
164        assert!(under_way(&world, Some(u32::MAX - 1)).contains(&"Forever.".to_owned()));
165        world.events.pop();
166        // The second it lapses, it goes.
167        world.announced(Some(1_600), "", None, "Another.");
168        assert_eq!(world.events.len(), 2, "the lapsed one went");
169        for n in 0..MAX_EVENTS {
170            world.announced(None, "", None, &format!("event {n}"));
171        }
172        assert_eq!(world.events.len(), MAX_EVENTS);
173    }
174
175    /// The game state keeps the frames it once dropped; a reconnect forgets
176    /// the pulse, timed from its connection, and keeps the world's events;
177    /// two states told different worlds are not the same state.
178    #[test]
179    fn the_game_state_keeps_the_world() {
180        use crate::state::GameState;
181        use hydra_protocol::Frame;
182        let mut state = GameState::default();
183        state.apply(&Frame::Pulse {
184            mana: true,
185            min: 46,
186            max: 75,
187        });
188        state.apply(&Frame::WorldEvent {
189            realm: "Icemule Trace".to_owned(),
190            expires_min: None,
191            text: "Snow falls.".to_owned(),
192        });
193        assert_eq!(
194            state.world.pulse.as_ref().map(|pulse| pulse.mana),
195            Some(true)
196        );
197        assert_eq!(state.world.events(None).count(), 1);
198        assert_ne!(
199            state,
200            GameState::default(),
201            "the world is part of the state"
202        );
203        state.invalidate_for_reconnect();
204        assert_eq!(state.world.pulse, None, "a new connection pulses anew");
205        assert_eq!(state.world.events(None).count(), 1, "the world went on");
206    }

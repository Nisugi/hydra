1//! Reading the things a line names: the creature, the item, every object.
2//!
3//! Shared by every classifier that reads a line's objects.
4//!
5//! Bold is the wire's mark for a creature, in a room listing and in a line
6//! alike, and the pronoun forms (`He`, `her`) carry the same id and the same
7//! bold. So the creature on a line is its first bolded object, and an item the
8//! first that is not.
9
10use hydra_protocol::frame::LinkKind;
11use regex::Regex;
12
13use super::containers::ItemRef;
14use super::lines::ChunkLine;
15
16/// Every game object on the line, in order, with whether it arrived bolded.
17///
18/// The object a run names even inside a clickable command (`Run::object`),
19/// not only the link a click acts on.
20#[must_use]
21pub fn objects(line: &ChunkLine) -> Vec<(ItemRef, bool)> {
22    line.runs
23        .runs
24        .iter()
25        .filter_map(|run| {
26            let object = run.object()?;
27            let LinkKind::Exist { id, noun } = &object.kind else {
28                return None;
29            };
30            Some((
31                ItemRef {
32                    id: id.clone(),
33                    noun: noun.clone(),
34                    text: object.text.clone(),
35                },
36                run.style.bold_depth > 0,
37            ))
38        })
39        .collect()
40}
41
42/// The creature on the line: the first bolded object.
43#[must_use]
44pub fn creature(line: &ChunkLine) -> Option<ItemRef> {
45    objects(line)
46        .into_iter()
47        .find_map(|(item, bold)| bold.then_some(item))
48}
49
50/// The item on the line: the first object that is not bolded.
51#[must_use]
52pub fn item(line: &ChunkLine) -> Option<ItemRef> {
53    objects(line)
54        .into_iter()
55        .find_map(|(item, bold)| (!bold).then_some(item))
56}

58/// A pattern that either compiled or never matches. A test per module that
59/// uses one checks its literals compile, so a typo is a red test rather than a
60/// silent miss.
61pub(crate) struct Pat(Option<Regex>);
62
63impl Pat {
64    pub(crate) fn new(pattern: &str) -> Self {
65        Self(crate::pattern::compile(pattern).ok())
66    }
67
68    pub(crate) fn is_match(&self, text: &str) -> bool {
69        self.0.as_ref().is_some_and(|re| re.is_match(text))
70    }
71
72    pub(crate) fn captures<'t>(&self, text: &'t str) -> Option<regex::Captures<'t>> {
73        self.0.as_ref()?.captures(text)
74    }
75
76    #[cfg(test)]
77    pub(crate) const fn compiled(&self) -> bool {
78        self.0.is_some()
79    }

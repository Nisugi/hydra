1//! The server clock, and what is derived from it.
2//!
3//! [`GameState::game_time_now`], [`GameState::in_roundtime`] and
4//! [`GameState::roundtime_remaining`] -- one `impl` block, kept apart from
5//! `state.rs` so that file stays under its size cap: **move code down, do not
6//! raise the cap.**
7//!
8//! # Why these three and not some other three
9//!
10//! Because they are the only methods on [`GameState`] that answer a question
11//! about **now**. Everything else in `state.rs` reports what the wire said;
12//! these compare it against a clock that keeps running between frames. That is
13//! also why they are the methods a roundtime gate calls, and why their
14//! `Option` semantics needed a page of justification apiece.
15

21    /// A prompt: the game's clock, and the end of the worn and reserve
22    /// lists and of the chunk. Its own method so `GameState::apply` stays
23    /// under clippy's 100 lines.
24    pub(super) fn apply_prompt(&mut self, time: &str, text: &str) {
25        self.prompt = Some(text.to_owned());
26        // The server's clock, and when it reached us. Together these are what
27        // make `game_time_now()` keep counting between prompts -- which
28        // matters because a prompt is only sent when something happens, so
29        // an idle client gets none at all (MEASURED).
30        if let Ok(t) = time.parse::<u32>() {
31            // A prompt carries a whole second, and lands anywhere inside
32            // it. Counting from where it landed would start the second
33            // over at each prompt, and a clock read between prompts would
34            // skip or repeat a step. So the count keeps its footing while
35            // the prompt agrees with it to the second, and moves only when
36            // the server's clock has drifted from it. Kept, its start moves
37            // on by as many seconds as the base does, or the time since the
38            // last prompt would be counted twice.
39            let kept = match (self.game_time, self.game_time_received.0) {
40                (Some(base), Some(at)) if self.game_time_now() == Some(t) => {
41                    at.checked_add(Duration::from_secs(u64::from(t.saturating_sub(base))))
42                }
43                _ => None,
44            };
45            self.game_time_received = super::Received(Some(kept.unwrap_or_else(Instant::now)));
46            self.game_time = Some(t);
47            // Effects that arrived with no clock -- the login burst precedes
48            // its first prompt -- get their end time now (`effects.rs`,
49            // `Effects::pending`).
50            self.effects.anchor(t);
51        }
52        // The worn and reserve lists end here, popped or not (`worn.rs`).
53        self.close_lists();
54        // **The chunk closes here**, and this is the only place it does. See
55        // `state/chunks.rs`: a command's output has no terminator of its own,
56        // so the prompt is the only boundary there is.
57        self.close_chunk();
58    }

60    /// The last prompt's own server time, unextrapolated.
61    ///
62    /// What a RECORD of the past wants: it is a fact of the stream, so a
63    /// replay reads the same value. [`Self::game_time_now`] moves with the
64    /// wall clock between prompts, which is what "is my roundtime over"
65    /// wants and what a recorder must not use.
66    #[must_use]
67    pub const fn game_time(&self) -> Option<u32> {
68        self.game_time
69    }

71    /// The server's clock **now**, extrapolated.
72    ///
73    /// The last prompt's timestamp plus how long ago it arrived on the local
74    /// monotonic clock. The single most important idea in this file.
75    ///
76    /// # Why extrapolate rather than read a clock
77    ///
78    /// **Timers must keep flowing between prompts.** A prompt is sent when
79    /// something happens; nothing is sent when a roundtime merely ends.
80    /// MEASURED: a session with a
81    /// behavior firing once a second produced 58 prompts, and an idle one
82    /// would produce none -- so anything that waits for a prompt to learn a
83    /// roundtime ended waits forever in a quiet room.
84    ///
85    /// **Both sides stay in server time**, so clock skew cancels instead of
86    /// needing correction. This is what made an elaborate skew-calibration
87    /// design unnecessary: there is no comparison between two clocks anywhere
88    /// in it, only an interval measured on one.
89    ///
90    /// Returns `None` until the first prompt arrives -- unknown, not a
91    /// fabricated zero.
92    #[must_use]
93    pub fn game_time_now(&self) -> Option<u32> {
94        self.game_time_after(self.elapsed_since_prompt())
95    }

97    /// [`Self::game_time_now`] with the elapsed interval **supplied** rather
98    /// than read from the local clock.
99    ///
100    /// # Why this exists
101    ///
102    /// `game_time_now` reads `Instant::elapsed`, and **a test cannot control
103    /// it.** `hydra-model` deliberately has no tokio dependency, so
104    /// `start_paused` -- the usual lever for this in async code -- is not
105    /// available here: pulling an async runtime into the
106    /// model crate to make one assertion deterministic buys determinism with
107    /// the layering the workspace exists to enforce.
108    ///
109    /// So the interval becomes a parameter. Production still measures it;
110    /// a test states it, and states `0` when it means "at the instant the
111    /// prompt arrived".
112    ///
113    /// **This is not a test-only method.** A caller that already knows how
114    /// much time has passed -- a replay, a recorded session, anything driving
115    /// the model from a timeline rather than from the wall clock -- wants
116    /// exactly this, and running many sessions in one process makes that
117    /// likelier, not less.
118    ///
119    /// # Why a test needs it
120    ///
121    /// A test that pins a prompt second and asserts the clock equals it,
122    /// through the wall clock, holds only while the suite runs inside one
123    /// second of wall time: a slow machine crosses a boundary and reads one
124    /// second later. **Not a flake** -- a real dependency on how long the
125    /// process has been running.
126    #[must_use]
127    pub fn game_time_after(&self, elapsed: u32) -> Option<u32> {
128        Some(self.game_time?.saturating_add(elapsed))
129    }

131    /// Whether the character is in roundtime.
132    ///
133    /// Compares the extrapolated server clock against
134    /// [`Self::roundtime_ends`], both in server epoch seconds.
135    ///
136    /// # This test is EXACT, not approximate
137    ///
138    /// MEASURED: across three roundtimes the
139    /// first plain `>` prompt landed **precisely** on `<roundTime value=>`, so
140    /// `value` is the end instant rather than "somewhere inside that second".
141    ///
142    /// # `None` means unknown, and unknown is not `false`
143    ///
144    /// Returns `None` when **the clock** is unobserved -- no prompt has
145    /// arrived, so there is nothing to compare against. A caller that treats
146    /// that as "not in roundtime" is inventing a belief, and the type makes
147    /// them write the decision down.
148    ///
149    /// # A roundtime that was never reported is OVER, not unknown
150    ///
151    /// When `roundtime_ends` is `None`, no roundtime has ever been reported,
152    /// and that is the normal state at login: a character who has done
153    /// nothing has no roundtime, and the first `<roundTime>` may not arrive
154    /// for minutes. Reading it as unknown would have every `send_now` refuse
155    /// for the whole of that window -- the feature disabled exactly when a
156    /// fresh character is most likely to want a sigil. An unreported
157    /// roundtime is *in the past*, not unheard-of.
158    ///
159    /// The asymmetry is the point: the **clock** is a fact about the
160    /// connection that must be observed before anything can be compared, and
161    /// the **roundtime** is a fact about the character that has a meaningful
162    /// default. Concluding "a roundtime nobody has ever mentioned is not
163    /// running" invents nothing.
164    #[must_use]
165    pub fn in_roundtime(&self) -> Option<bool> {
166        self.in_roundtime_after(self.elapsed_since_prompt())
167    }
168
169    /// [`Self::in_roundtime`] against a supplied interval. See
170    /// [`Self::game_time_after`] for why the seam exists.
171    #[must_use]
172    pub fn in_roundtime_after(&self, elapsed: u32) -> Option<bool> {
173        let now = self.game_time_after(elapsed)?;
174        Some(self.roundtime_ends.is_some_and(|ends| now < ends))
175    }

177    /// How many seconds of roundtime remain, or `None` if the clock is
178    /// unknown.
179    ///
180    /// Saturates at zero rather than going negative: a roundtime that has
181    /// passed has no remainder. A roundtime that was **never reported** also
182    /// has none -- `Some(0)`, for the reason [`Self::in_roundtime`] gives at
183    /// length. The two methods must agree, or a caller could be told it is not
184    /// in roundtime and then handed a remainder.
185    #[must_use]
186    pub fn roundtime_remaining(&self) -> Option<u32> {
187        self.roundtime_remaining_after(self.elapsed_since_prompt())
188    }

190    /// [`Self::roundtime_remaining`] against a supplied interval. See
191    /// [`Self::game_time_after`] for why the seam exists.
192    #[must_use]
193    pub fn roundtime_remaining_after(&self, elapsed: u32) -> Option<u32> {
194        let now = self.game_time_after(elapsed)?;
195        Some(
196            self.roundtime_ends
197                .map_or(0, |ends| ends.saturating_sub(now)),
198        )

237    /// Seconds since the last prompt landed, on the local monotonic clock.
238    ///
239    /// The **one** place the wall clock is read. Every extrapolating method
240    /// goes through here, so a caller driving the model from a timeline has a
241    /// single thing to replace -- one home per fact.
242    /// How long until the clock's next whole second, for whoever draws it
243    /// to draw again then; `None` with no clock yet.
244    #[must_use]
245    pub fn until_next_second(&self) -> Option<std::time::Duration> {
246        let at = self.game_time_received.0?;
247        let elapsed = at.elapsed();
248        let into = elapsed.as_secs_f64().fract();
249        Some(std::time::Duration::from_secs_f64((1.0 - into).max(0.01)))
250    }

252    /// When the server's clock reaches `second`, on the local monotonic
253    /// clock: the tick the prompts place, counted on from the last. A
254    /// second already reached gives an instant in the past; `None` with no
255    /// clock yet.
256    ///
257    /// The game lets a command through the instant its clock reaches a
258    /// roundtime's end, and keeps no time finer than a second, so this
259    /// instant is when a wait for that end is over.
260    #[must_use]
261    pub fn instant_of(&self, second: u32) -> Option<Instant> {
262        let at = self.game_time_received.0?;
263        let ahead = i64::from(second) - i64::from(self.game_time?);
264        match u64::try_from(ahead) {
265            Ok(ahead) => at.checked_add(Duration::from_secs(ahead)),
266            Err(_) => at.checked_sub(Duration::from_secs(ahead.unsigned_abs())),
267        }
268    }

270    /// How long until the roundtime and the cast roundtime are both over,
271    /// to the tick: zero when neither runs, `None` with no clock yet.
272    ///
273    /// A wait for a roundtime sleeps this long and is over then, rather
274    /// than looking again now and then and sending up to a beat late.
275    #[must_use]
276    pub fn until_out_of_roundtime(&self) -> Option<Duration> {
277        self.until_tick(self.roundtime_ends.max(self.cast_time_ends))
278    }
279
280    /// [`Self::until_out_of_roundtime`] for the roundtime alone: what a
281    /// move waits for, a cast roundtime not stopping one.
282    #[must_use]
283    pub fn until_roundtime_over(&self) -> Option<Duration> {
284        self.until_tick(self.roundtime_ends)
285    }

287    /// How long until the clock reaches `end`: zero once it has, or with
288    /// no end; `None` with no clock yet.
289    fn until_tick(&self, end: Option<u32>) -> Option<Duration> {
290        let now = self.game_time_now()?;
291        Some(
292            end.filter(|end| *end > now)
293                .and_then(|end| self.instant_of(end))
294                .map_or(Duration::ZERO, |at| {
295                    at.saturating_duration_since(Instant::now())
296                }),
297        )
298    }
299
300    pub(super) fn elapsed_since_prompt(&self) -> u32 {
301        self.game_time_received.0.map_or(0, |at| {
302            u32::try_from(at.elapsed().as_secs()).unwrap_or(u32::MAX)
303        })
304    }

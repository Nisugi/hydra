1//! Which characters a spell has locked out.
2//!
3//! Tracks the spell cooldowns a casting puts on the characters it lands on.
4//!
5//! # Two cast mechanics, two ways of learning the target
6//!
7//! The spells that look at the group members are the ones that are self cast
8//! but affect your group.
9//!
10//! | Kind | Cast | How the target is learned |
11//! |---|---|---|
12//! | [`CooldownKind::Group`] | **on yourself**, lands on everyone grouped | it is not — stamp every member |
13//! | [`CooldownKind::Target`] | at one character | the `target-start` message names them |
14//!
15//! Five spells declare one, and the invariant is checked over the whole table
16//! in `spells.rs`: every `target` cooldown has a `target-start` message and no
17//! `group` cooldown has one, because a self-cast names nobody.
18//!
19//! # The group case is optimistic, and bounded
20//!
21//! The caster's **only** notice that a group casting landed is a clause inside
22//! their own start message — each of the three is one pattern with `your
23//! group` as an optional part, and the presence of `your group` tells the two
24//! views apart.
25//!
26//! So everyone grouped at the time is stamped. A member who was out of range,
27//! or who was already on cooldown from an earlier casting, gets a stamp that
28//! is too early — the next casting corrects it, so the error is bounded by
29//! one cooldown and does not accumulate.
30//!
31//! **A member still locked out is skipped rather than re-stamped**: they did
32//! not receive this casting, so their own cooldown keeps running. Getting
33//! that wrong would extend a cooldown every time someone else got a buff.
34//!
35//! # Keyed by noun, and the target case needs no group
36//!
37//! A target cooldown counts whoever the spell landed on, in or out of the
38//! group: seeing someone else put a target on cooldown is as useful as doing
39//! it yourself.
40//!
41//! # Nothing here sends
42//!
43//! Reading the group's members must never send `GROUP` and wait for a reply
44//! that only the reading thread could parse. Nothing here sends anything, so
45//! the hazard does not arise -- but it is the kind of constraint worth
46//! keeping in mind for any caller.
47

54/// When each character becomes castable again, by spell number then noun.
55///
56/// Expiries in server seconds.
57#[derive(Clone, Debug, Default, PartialEq, Eq)]
58pub struct Cooldowns {
59    /// Spell number -> member noun -> the server second it expires.
60    expiries: BTreeMap<u16, BTreeMap<String, u32>>,
61}
62
63impl Cooldowns {
64    /// Stamp every member for a group casting.
65    ///
66    /// Returns how many were stamped. A member **already locked out is
67    /// skipped**, not re-stamped — see the module doc.
68    pub fn record_group(&mut self, spell: u16, members: &[Member], now: u32) -> usize {
69        let Some(seconds) = spells::spell(spell).and_then(|s| s.cooldown(CooldownKind::Group))
70        else {
71            return 0;
72        };
73        let expires = now.saturating_add(seconds);
74        let held = self.expiries.entry(spell).or_default();
75        let mut stamped = 0;
76        for member in members {
77            // Still locked out: they did not receive this casting, so their
78            // own cooldown keeps running rather than being extended by it.
79            if held.get(&member.noun).is_some_and(|at| *at > now) {
80                continue;
81            }
82            held.insert(member.noun.clone(), expires);
83            stamped += 1;
84        }
85        stamped
86    }
87
88    /// Stamp one character a spell landed on.
89    ///
90    /// **Not restricted to the group**: the cooldown belongs to the character,
91    /// so it counts whoever cast it and whether or not they are grouped.
92    pub fn record_target(&mut self, spell: u16, noun: &str, now: u32) -> bool {
93        let Some(seconds) = spells::spell(spell).and_then(|s| s.cooldown(CooldownKind::Target))
94        else {
95            return false;
96        };
97        self.expiries
98            .entry(spell)
99            .or_default()
100            .insert(noun.to_owned(), now.saturating_add(seconds));
101        true
102    }
103
104    /// Seconds before a character can receive this spell again.
105    ///
106    /// **Zero when nothing is recorded**, which is the right answer: a
107    /// character nobody has seen
108    /// buffed is castable, and `None` would make every caller handle a case
109    /// that means "yes".
110    #[must_use]
111    pub fn remaining(&self, spell: u16, noun: &str, now: u32) -> u32 {
112        self.expiries
113            .get(&spell)
114            .and_then(|held| held.get(noun))
115            .map_or(0, |at| at.saturating_sub(now))
116    }
117
118    /// Whether this character can receive the spell now.
119    #[must_use]
120    pub fn ready(&self, spell: u16, noun: &str, now: u32) -> bool {
121        self.remaining(spell, noun, now) == 0
122    }
123
124    /// Forget everything.
125    ///
126    /// Called on a reconnect: these are stamps on **other people**, taken
127    /// against a clock this session was keeping, and both assumptions break
128    /// while we are away.
129    pub fn clear(&mut self) {
130        self.expiries.clear();
131    }
132
133    /// Whether anything is recorded.
134    #[must_use]
135    pub fn is_empty(&self) -> bool {
136        self.expiries.is_empty()
137    }
138}

140impl GameState {
141    /// Stamp a cooldown a closing chunk's line started.
142    ///
143    /// # What drives [`Cooldowns`]
144    ///
145    /// This is the production caller of `record_group` and `record_target`.
146    /// A cooldown tracker that is never fed answers "ready" forever, which is
147    /// the one answer that makes a buff behavior waste casts.
148    ///
149    /// The input is the caster's start message, when it says `your group`, and the third-person
150    /// message naming a target. [`crate::spells::cooldown_landing`] reads
151    /// both from the spell table's messages, and `close_chunk` calls this per
152    /// line, after that line's group event. `now` is the prompt's server
153    /// second, the clock every other stamp in this model uses.
154    pub(super) fn record_cooldown(&mut self, landing: &crate::spells::CooldownLanding, now: u32) {
155        match landing {
156            crate::spells::CooldownLanding::Group { spell } => {
157                self.cooldowns
158                    .record_group(*spell, self.group.members(), now);
159            }
160            crate::spells::CooldownLanding::Target { spell, noun } => {
161                self.cooldowns.record_target(*spell, noun, now);
162            }
163        }
164    }
165
166    /// Which of the current group a spell would actually affect.
167    ///
168    /// Empty when the group is
169    /// empty or every member is locked out — and **also** when the server
170    /// time is unknown, because a cooldown cannot be judged without a clock.
171    #[must_use]
172    pub fn spell_cooldown_ready(&self, spell: u16) -> Vec<&Member> {
173        let Some(now) = self.game_time_now() else {
174            return Vec::new();
175        };
176        self.group
177            .members()
178            .iter()
179            .filter(|member| self.cooldowns.ready(spell, &member.noun, now))
180            .collect()
181    }
182
183    /// Seconds before a character can receive this spell again.
184    ///
185    /// `None` when the server time is unknown, which is distinct from the
186    /// `0` that means "ready now".
187    #[must_use]
188    pub fn spell_cooldown_left(&self, spell: u16, noun: &str) -> Option<u32> {
189        Some(self.cooldowns.remaining(spell, noun, self.game_time_now()?))
190    }
191}

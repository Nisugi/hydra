1//! Which container holds what, and which weapon comes to hand.
2//!
3//! Two lists the game keeps for you, taught by `stow list` and `ready list`
4//! and by the one-line confirmations that follow a `stow set` or `ready`.
5//!
6//! # Patterns match the prose, not the markup
7//!
8//! Each line names its container or item as a link. [`ChunkLine::links`]
9//! yields [`LinkKind::Exist`] typed, so a pattern here only has to say
10//! **which slot** a line is about; nothing re-tokenizes `<a exist= noun=>`.
11//!
12//! # One store-mode, two spellings
13//!
14//! The two messages that teach the store-mode disagree about its wording:
15//!
16//! | Taught by | "wear it if you can" is spelled |
17//! |---|---|
18//! | `ready list` | `worn if possible, stowed otherwise` |
19//! | `store set` | `worn if possible and stowed if not` |
20//!
21//! Same state, two spellings. Kept as a raw string, a consumer comparing it
22//! would get a different answer depending on which message happened to teach
23//! it last. The sheath spelling splits the same way -- `put in secondary
24//! sheath` against `stored in your secondary sheath`. [`StoreMode`] reads
25//! both and stores one value, so the question "will this be worn?" has one
26//! answer.
27//!
28//! # What is deliberately NOT here
29//!
30//! Re-checking every held id against the inventory, and clearing the
31//! `checked` flag if one has gone, is a *cache-coherence* check, and it
32//! belongs to whoever owns inventory -- not to a record of what the game
33//! said. `Containers`'s `checked` flag reports whether the list was ever
34//! taught; a consumer that needs "and the items still exist" asks inventory,
35//! which is the layer that knows.
36

41/// An item the game named, held by the id a command targets it with.
42#[derive(Clone, Debug, PartialEq, Eq)]
43pub struct ItemRef {
44    /// `exist=`, verbatim. The identity.
45    pub id: String,
46    /// `noun=`. What a command targets it by.
47    pub noun: String,
48    /// The link's display text, e.g. `"leather backpack"`.
49    pub text: String,
50}
51
52impl ItemRef {
53    /// Read the first game object named on a line, if there is one.
54    ///
55    /// [`ChunkLine::objects`], not `links`: in `ready list` the item's own
56    /// text is clickable and clicking it sends `store WEAPON clear`, so the
57    /// markup states both what the text NAMES and what a click SENDS. `links`
58    /// gives the command; `objects` gives the thing. Asking for links here
59    /// would report no item at all.
60    fn first_on(line: &ChunkLine) -> Option<Self> {
61        line.objects().find_map(|link| match &link.kind {
62            LinkKind::Exist { id, noun } => Some(Self {
63                id: id.clone(),
64                noun: noun.clone(),
65                text: link.text.clone(),
66            }),
67            _ => None,
68        })
69    }
70}

72/// A stow category: which container a kind of loot goes into.
73///
74/// All of the game's categories. A new
75/// category added by the game is a compile error at [`StowSlot::parse`]'s
76/// match rather than a silent miss.
77#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
78pub enum StowSlot {
79    /// `box`: boxes.
80    Box,
81    /// `gem`: gems.
82    Gem,
83    /// `herb`: herbs.
84    Herb,
85    /// `skin`: skins.
86    Skin,
87    /// `wand`: wands.
88    Wand,
89    /// `scroll`: scrolls.
90    Scroll,
91    /// `potion`: potions.
92    Potion,
93    /// `trinket`: trinkets.
94    Trinket,
95    /// `reagent`: reagents.
96    Reagent,
97    /// `lockpick`: lockpicks.
98    Lockpick,
99    /// `treasure`: treasure.
100    Treasure,
101    /// `forageable`: foraged items.
102    Forageable,
103    /// `collectible`: collectibles.
104    Collectible,
105    /// Where anything uncategorised goes -- and where the bank's notes go
106    /// when nothing better is set.
107    Default,
108}

134            Self::Gem => "gem",
135            Self::Herb => "herb",
136            Self::Skin => "skin",
137            Self::Wand => "wand",
138            Self::Scroll => "scroll",
139            Self::Potion => "potion",
140            Self::Trinket => "trinket",
141            Self::Reagent => "reagent",
142            Self::Lockpick => "lockpick",
143            Self::Treasure => "treasure",
144            Self::Forageable => "forageable",
145            Self::Collectible => "collectible",
146            Self::Default => "default",
147        }
148    }
149
150    /// Read a slot from the wire's word.
151    ///
152    /// **Case-insensitive**, because the game shouts it in one message and
153    /// whispers it in the other: `stow list` prints `(box)` and the
154    /// confirmation prints `your STOW BOX container`. Folding case here
155    /// rather than at each call site keeps the knowledge in the type.
156    #[must_use]
157    pub fn parse(word: &str) -> Option<Self> {
158        Self::ALL
159            .into_iter()
160            .find(|slot| slot.as_str().eq_ignore_ascii_case(word))
161    }
162}

168pub enum ReadySlot {

178    AmmoBundle,

180    Ammo2Bundle,

211            Self::AmmoBundle => "ammo bundle",

213            Self::Sheath => "sheath",
214            Self::SecondarySheath => "secondary sheath",
215            Self::Wand => "wand",
216        }

219    /// Read a slot from the wire's phrase.
220    ///
221    /// A table rather than a string rewrite (downcasing and turning spaces
222    /// into underscores, `"ammo bundle"` into `ammo_bundle`), so a phrase the
223    /// game adds later returns `None` instead of inventing a key nothing
224    /// reads.
225    #[must_use]
226    pub fn parse(phrase: &str) -> Option<Self> {
227        Self::ALL
228            .into_iter()
229            .find(|slot| slot.as_str().eq_ignore_ascii_case(phrase))
230    }
231
232    /// Whether this slot has a store-mode.
233    ///
234    /// Six of the nine have one: the two
235    /// sheaths and the second ammo bundle are *where things go*, so asking
236    /// where to store them is not a question the game offers.
237    #[must_use]
238    pub fn has_store_mode(self) -> bool {
239        matches!(
240            self,
241            Self::Shield
242                | Self::Weapon
243                | Self::SecondaryWeapon
244                | Self::RangedWeapon
245                | Self::AmmoBundle
246                | Self::Wand
247        )
248    }
249}

251/// Where an item goes when you store it.
252///
253/// **Three states, two vocabularies.** See the module doc: `ready list` and
254/// `store set` word the same fact differently. This type reads both
255/// spellings into one value.
256#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
257pub enum StoreMode {
258    /// Worn if it can be, stowed if not.
259    WornIfPossible,
260    /// Stowed, into the stow container for its kind.
261    Stowed,
262    /// Into the sheath.
263    Sheath,
264    /// Into the secondary sheath.
265    SecondarySheath,
266}
267
268impl StoreMode {
269    /// Read either vocabulary.
270    ///
271    /// The secondary spellings are tested **before** their primaries, because
272    /// `"stored in your secondary sheath"` contains `"sheath"` and would
273    /// otherwise answer [`Self::Sheath`].
274    #[must_use]
275    pub fn parse(phrase: &str) -> Option<Self> {
276        let phrase = phrase.trim();
277        if phrase.starts_with("worn if possible") {
278            return Some(Self::WornIfPossible);
279        }
280        if phrase == "stowed" {
281            return Some(Self::Stowed);
282        }
283        if phrase.ends_with("secondary sheath") {
284            return Some(Self::SecondarySheath);
285        }
286        if phrase.ends_with("sheath") {
287            return Some(Self::Sheath);
288        }
289        None
290    }
291}

293/// What one line said about the two lists.
294#[derive(Clone, Debug, PartialEq, Eq)]
295pub enum ContainerEvent {
296    /// `You have the following containers set as stow targets:` -- the list
297    /// is about to be restated, so forget what is held.
298    StowListBegins,
299    /// One row of `stow list`, or a `stow set` confirmation.
300    StowSet {
301        /// Which category.
302        slot: StowSlot,
303        /// The container it goes in.
304        item: ItemRef,
305    },
306    /// `Your current settings are:` -- `ready list` is about to restate.
307    ReadyListBegins,
308    /// One row of `ready list`, or a `ready` confirmation.
309    ReadySet {
310        /// Which slot.
311        slot: ReadySlot,
312        /// The item, or `None` where the row said `none`.
313        item: Option<ItemRef>,
314        /// The store-mode, where the row carried one.
315        store: Option<StoreMode>,
316    },
317    /// `Cleared your default <slot>.`
318    ReadyCleared(ReadySlot),
319    /// `When storing your <slot>, it will be <mode>.`
320    StoreModeSet {
321        /// Which slot.
322        slot: ReadySlot,
323        /// Where it will go.
324        store: StoreMode,
325    },
326    /// The closing line of `ready list`, which is how we know it was whole.
327    ReadyListEnds,
328}

330/// Read a line as a container event, or `None` if it is not one.
331#[must_use]
332pub fn classify(line: &ChunkLine) -> Option<ContainerEvent> {
333    classify_text(line, &line.text())
334}

336/// [`classify`], given the line's text already rendered (the chunk renders
337/// each line once and shares it).
338#[allow(clippy::too_many_lines)]
339pub(crate) fn classify_text(line: &ChunkLine, text: &str) -> Option<ContainerEvent> {
340    let trimmed = text.trim();
341
342    if trimmed == "You have the following containers set as stow targets:" {
343        return Some(ContainerEvent::StowListBegins);
344    }
345    if trimmed == "Your current settings are:" {
346        return Some(ContainerEvent::ReadyListBegins);
347    }
348    if trimmed.starts_with("To change your default item for a category") {
349        return Some(ContainerEvent::ReadyListEnds);
350    }
351
352    // `Cleared your default shield.` -- no link, so it is checked before the
353    // arms that need one.
354    if let Some(rest) = trimmed.strip_prefix("Cleared your default ") {
355        let slot = ReadySlot::parse(rest.trim_end_matches('.'))?;
356        return Some(ContainerEvent::ReadyCleared(slot));
357    }
358
359    // `When storing your weapon, it will be stowed.`
360    if let Some(rest) = trimmed.strip_prefix("When storing your ") {
361        let (slot, mode) = rest.split_once(", it will be ")?;
362        return Some(ContainerEvent::StoreModeSet {
363            slot: ReadySlot::parse(slot)?,
364            store: StoreMode::parse(mode.trim_end_matches('.'))?,
365        });
366    }
367
368    // **The two list rows are INDENTED, and that is required.** Both
369    // listings open each row with two spaces, their own layout. Matching the
370    // trimmed text would let any main-window line ending `(gem)` with a link
371    // on it teach a stow container, and any line shaped `shield: ...` teach a
372    // ready slot. Indentation is what separates the listing from prose that
373    // happens to look like a row.
374    let indented = text.starts_with("  ");
375
376    // A `stow list` row: `  <a ...>my backpack</a> (gem)`. The category is
377    // the parenthesised word at the end, and the container is the link.
378    if indented && let Some(slot) = trailing_parenthetical(trimmed).and_then(StowSlot::parse) {
379        return Some(ContainerEvent::StowSet {
380            slot,
381            item: ItemRef::first_on(line)?,
382        });
383    }
384
385    // `Set "<a ...>a pouch</a>" to be your STOW GEM container.` and the
386    // `default` spelling. Both begin `Set "`, so both require it here.
387    if trimmed.contains("Set \"")
388        && trimmed.contains("\" to be your ")
389        && trimmed.ends_with(" STOW container.")
390    {
391        let slot = trimmed
392            .rsplit_once("\" to be your ")
393            .map(|(_, tail)| tail.trim_end_matches(" STOW container."))
394            .and_then(StowSlot::parse)?;
395        return Some(ContainerEvent::StowSet {
396            slot,
397            item: ItemRef::first_on(line)?,
398        });
399    }
400    if trimmed.starts_with("Set \"")
401        && let Some((_, tail)) = trimmed.rsplit_once("\" to be your STOW ")
402    {
403        let slot = StowSlot::parse(tail.trim_end_matches(" container."))?;
404        return Some(ContainerEvent::StowSet {
405            slot,
406            item: ItemRef::first_on(line)?,
407        });
408    }
409
410    // `Setting <a ...>a sword</a> to be your default weapon.`
411    //
412    // The link is required, so a line with none is not this event -- `?`, not
413    // an `Option` that would clear.
414    if trimmed.starts_with("Setting ")
415        && let Some((_, tail)) = trimmed.rsplit_once(" to be your default ")
416    {
417        return Some(ContainerEvent::ReadySet {
418            slot: ReadySlot::parse(tail.trim_end_matches('.'))?,
419            item: Some(ItemRef::first_on(line)?),
420            store: None,
421        });
422    }
423
424    // A `ready list` row: `  weapon: (<a ...>a sword</a>) (stowed)`, or with
425    // `none` where nothing is set, or with no store-mode at all for the
426    // sheaths and the second ammo bundle.
427    //
428    // Indented, and carrying the row's own `<d cmd='store ...'>` or
429    // `<d cmd='ready ...'>` link around the item or the `none`: that command
430    // is what makes the row a row of THIS listing.
431    if !indented || !has_slot_command(line) {
432        return None;
433    }
434    let (label, rest) = trimmed.split_once(": ")?;
435    let slot = ReadySlot::parse(label)?;
436    let store = trailing_parenthetical(rest).and_then(StoreMode::parse);
437    Some(ContainerEvent::ReadySet {
438        slot,
439        item: ItemRef::first_on(line),
440        store,
441    })
442}
443
444/// Whether the line carries a `ready list` row's slot command:
445/// `<d cmd='store WEAPON clear'>`, `<d cmd='ready SHIELD'>`. The store-mode
446/// link, `<d cmd='store set'>`, is on every row too and is not this.
447fn has_slot_command(line: &ChunkLine) -> bool {
448    line.links().any(|link| match &link.kind {
449        LinkKind::Direct { cmd } => {
450            (cmd.starts_with("store ") || cmd.starts_with("ready ")) && cmd != "store set"
451        }
452        _ => false,
453    })
454}
455
456/// The contents of a trailing `(...)`, if the line ends in one.
457fn trailing_parenthetical(text: &str) -> Option<&str> {
458    let inner = text.strip_suffix(')')?;
459    let (_, last) = inner.rsplit_once('(')?;
460    Some(last)
461}

463/// The two lists the game keeps: stow targets and ready items.
464#[derive(Clone, Debug, Default, PartialEq, Eq)]
465pub struct Containers {
466    stow: Vec<(StowSlot, ItemRef)>,
467    ready: Vec<(ReadySlot, ItemRef)>,
468    store: Vec<(ReadySlot, StoreMode)>,
469    stow_checked: bool,
470    ready_checked: bool,
471}
472
473impl Containers {
474    /// A new connection: every id in both lists is dead (every item id
475    /// changes at a login), so both are emptied and unchecked, to be asked
476    /// again. The store modes are settings and are kept (`reconnect.rs`).
477    pub(super) fn lapse(&mut self) {
478        self.stow.clear();
479        self.ready.clear();
480        self.stow_checked = false;
481        self.ready_checked = false;
482    }
483
484    /// Apply one event. Returns whether anything changed.
485    pub fn apply(&mut self, event: &ContainerEvent) -> bool {
486        let before = self.clone();
487        match event {
488            // A restated list replaces the old one: a slot the game no
489            // longer lists has been cleared, and keeping it would report a
490            // container that is not a target any more.
491            ContainerEvent::StowListBegins => {
492                self.stow.clear();
493                self.stow_checked = true;
494            }
495            ContainerEvent::ReadyListBegins => {
496                self.ready.clear();
497                self.store.clear();
498                self.ready_checked = false;
499            }
500            // `checked` is set by the listing's closing line, not its start:
501            // a list cut off by a disconnect is not a list you have read.
502            ContainerEvent::ReadyListEnds => self.ready_checked = true,
503            ContainerEvent::StowSet { slot, item } => set(&mut self.stow, *slot, item.clone()),
504            ContainerEvent::ReadySet { slot, item, store } => {
505                // **A row with no item leaves the slot as it was**: it writes
506                // only when the row named something. A `none` row inside a
507                // listing finds the slot already emptied by the opener, so
508                // nothing is lost; outside one, it is not evidence enough to
509                // throw away an item a confirmation taught.
510                if let Some(item) = item {
511                    set(&mut self.ready, *slot, item.clone());
512                }
513                if let Some(store) = store {
514                    set(&mut self.store, *slot, *store);
515                }
516            }
517            // **Clearing an item clears its store-mode too.** Leaving the
518            // mode behind would have a slot with no item still report where
519            // that item would be stored -- an answer about nothing.
520            ContainerEvent::ReadyCleared(slot) => {
521                self.ready.retain(|(held, _)| held != slot);
522                self.store.retain(|(held, _)| held != slot);
523            }
524            ContainerEvent::StoreModeSet { slot, store } => set(&mut self.store, *slot, *store),
525        }
526        before != *self
527    }
528
529    /// The container a kind of loot stows into, if the game has said.
530    #[must_use]
531    pub fn stow(&self, slot: StowSlot) -> Option<&ItemRef> {
532        find(&self.stow, slot)
533    }
534
535    /// The item readied for a slot, if the game has said.
536    #[must_use]
537    pub fn ready(&self, slot: ReadySlot) -> Option<&ItemRef> {
538        find(&self.ready, slot)
539    }
540
541    /// Where a slot's item goes when stored, if the game has said.
542    #[must_use]
543    pub fn store_mode(&self, slot: ReadySlot) -> Option<StoreMode> {
544        find(&self.store, slot).copied()
545    }
546
547    /// Whether a whole `stow list` has been read.
548    ///
549    /// **Not "and the containers still exist"** -- see the module doc on
550    /// `valid?`. This says the game stated the list; whether those ids are
551    /// still in inventory is inventory's question.
552    #[must_use]
553    pub fn stow_checked(&self) -> bool {
554        self.stow_checked
555    }
556
557    /// Whether a whole `ready list` has been read, closing line included.
558    #[must_use]
559    pub fn ready_checked(&self) -> bool {
560        self.ready_checked
561    }
562
563    /// Forget everything.
564    ///
565    /// Both lists are per-character settings the game holds, so a reconnect
566    /// does **not** clear them -- they are re-taught by the sync, not
567    /// invalidated by the disconnect.
568    pub fn reset(&mut self) {
569        *self = Self::default();
570    }
571}

573/// Set a key in an association list, replacing any value already held.
574fn set<K: PartialEq, V>(list: &mut Vec<(K, V)>, key: K, value: V) {
575    match list.iter_mut().find(|(held, _)| *held == key) {
576        Some(slot) => slot.1 = value,
577        None => list.push((key, value)),
578    }
579}
580
581/// Look a key up in an association list.
582fn find<K: PartialEq + Copy, V>(list: &[(K, V)], key: K) -> Option<&V> {
583    list.iter()
584        .find(|(held, _)| *held == key)
585        .map(|(_, value)| value)
586}

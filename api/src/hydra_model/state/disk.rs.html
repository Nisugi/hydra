1//! Floating disks: whose they are, and what they are called.
2//!
3//! # A disk is a room object named for someone
4//!
5//! `Bagagwa disk`, `fiery red Vasstryke disk`, `four-toned Desorceri coffret`.
6//! Eleven nouns carry them ([`DISK_NOUNS`]), and the display text puts the owner's
7//! name, capitalised, right before the noun -- which is what makes a disk
8//! findable without the game ever labelling one as such. The noun comes
9//! typed off the link, so only the owner's word is read: the word before the
10//! noun, a capital then lowercase letters (`[A-Z][a-z]+`). The game allows no
11//! other shape of name: no `McTavish`, no `D'iel`.
12//!
13//! **There is no possessive.** MEASURED over 45 wire logs (`logs/**/*.bytes`,
14//! every `<a ... noun="<disk noun>">...</a>`): 2,953 links, 83 distinct;
15//! **0** possessive (`Ryeka's disk`), 76 a capitalised name before the noun
16//! (`Bagagwa disk`, `rusty iron Duffield disk`), and 7 furniture or loot
17//! (`iron-bound walnut chest`, `crimson sphere`). A test written from an
18//! invented possessive line would agree with code written from the same
19//! invention and find no disk on the wire; the tests here use the wire's
20//! shape (`faenor Demandred disk`).
21
22use super::room::RoomItem;
23
24/// The eleven nouns a floating disk can be.
25///
26/// All eleven. A twelfth would be a visible change here rather than a
27/// silent miss.
28pub const DISK_NOUNS: [&str; 11] = [
29    "cassone", "chest", "coffer", "coffin", "coffret", "disk", "hamper", "saucer", "sphere",
30    "trunk", "tureen",
31];
32
33/// One floating disk seen in the room.
34#[derive(Clone, Debug, PartialEq, Eq)]
35pub struct Disk {
36    /// `exist=`, verbatim -- what a command targets.
37    pub id: String,
38    /// The noun, one of [`DISK_NOUNS`].
39    pub noun: String,
40    /// Whose it is, the name before the noun: `Bagagwa disk` -> `Bagagwa`.
41    pub owner: String,
42}
43
44impl Disk {
45    /// Read a room object as a disk, or `None` if it is not one.
46    ///
47    /// Requires BOTH halves: a disk noun, and an owner's name before it. A
48    /// plain `chest` sitting in a room is furniture, not somebody's disk, and
49    /// treating it as one would have a behavior try to loot the scenery.
50    #[must_use]
51    pub fn read(item: &RoomItem) -> Option<Self> {
52        if !DISK_NOUNS.contains(&item.noun.as_str()) {
53            return None;
54        }
55        let owner = owner_of(&item.text, &item.noun)?;
56        Some(Self {
57            id: item.id.clone(),
58            noun: item.noun.clone(),
59            owner,
60        })
61    }
62}
63
64/// `"fiery red Vasstryke disk"` -> `"Vasstryke"`, given the noun: the word
65/// before it, when it is a name, `[A-Z][a-z]+`.
66fn owner_of(text: &str, noun: &str) -> Option<String> {
67    let before = text.strip_suffix(noun)?.strip_suffix(' ')?;
68    let owner = before.rsplit(' ').next()?;
69    let mut letters = owner.chars();
70    let named = letters
71        .next()
72        .is_some_and(|first| first.is_ascii_uppercase())
73        && owner.len() > 1
74        && letters.all(|letter| letter.is_ascii_lowercase());
75    named.then(|| owner.to_owned())
76}
77
78impl crate::state::Room {
79    /// Every disk in the room, in wire order.
80    pub fn disks(&self) -> impl Iterator<Item = Disk> + '_ {
81        self.objects.iter().filter_map(Disk::read)
82    }
83
84    /// The disk belonging to `owner`, if it is here.
85    ///
86    /// Exact, not a substring: with a substring match a character called
87    /// `Rye` would find `Ryeka disk` and try to loot a stranger's property.
88    #[must_use]
89    pub fn disk_of(&self, owner: &str) -> Option<Disk> {
90        self.disks().find(|disk| disk.owner == owner)
91    }
92}

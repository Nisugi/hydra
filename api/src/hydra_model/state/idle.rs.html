1//! The server's idle warning: reading it, and clearing it.
2//!
3//! Kept apart from `state.rs` beside [`clock`](super::clock) and
4//! [`reconnect`](super::reconnect) -- move code down, do not raise the cap.
5//!
6//! # What this is for
7//!
8//! `MAX_UNATTENDED_LOSSES = 2` exists because the game idle-kicks after ~30
9//! minutes, and an idle kick is the one disconnect where **the connection worked
10//! perfectly** -- so the supervisor's `worked` test resets the backoff ladder and
11//! it reconnects at the one-second rung, idles for another 30 minutes, and is
12//! kicked again. The cap stops that only after two full cycles.
13//!
14//! The server announces it in advance, in plain text. Full evidence and
15//! rationale in `crates/hydra-model/tests/idle_warning.rs`.
16//!
17//! # The three states, and why the field is private
18//!
19//! | [`GameState::idle_warned`] | [`GameState::idle_warned_at`] | Means |
20//! |---|---|---|
21//! | `false` | `None` | never warned, or answered since |
22//! | `true` | `None` | warned before any prompt, so no server clock yet |
23//! | `true` | `Some(t)` | warned at server second `t` |
24//!
25//! Three states, held in a named `IdleWarning` enum rather than an
26//! `Option<Option<u32>>`, which clippy's `option_option` rightly objects to: a
27//! nested `Option` makes a reader decode the nesting to find the third state.
28//! The field stays private and these methods are the contract.
29
30use super::GameState;

32/// Whether the server has warned about idling, and when.
33///
34/// **Three states, so a named enum rather than `Option<Option<u32>>`**, which
35/// clippy's `option_option` rightly objects to: an `Option` read as two
36/// states where three exist, nested, makes a reader decode the nesting to
37/// find the third. Naming them costs nine lines and removes the decoding.
38///
39/// The type lives here, with the three methods that read it, rather than in
40/// `state.rs`, which is at its line cap.
41#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
42pub(super) enum IdleWarning {
43    /// Never warned, or warned and since answered by an outbound command.
44    #[default]
45    None,
46    /// Warned before any `<prompt>` arrived, so there was no server clock to
47    /// stamp. Real, and not the same as never warned -- the login burst carries
48    /// no prompt until its end.
49    Unstamped,
50    /// Warned at this server epoch second.
51    At(u32),
52}
53
54impl GameState {
55    /// Has the server warned that this character is idle, without an answer
56    /// since?
57    ///
58    /// The supervisor's question. `true` means the next disconnect is most likely
59    /// an idle kick rather than a lost network, which is the difference between
60    /// "back off and retry" and "stop, nobody is here".
61    #[must_use]
62    pub const fn idle_warned(&self) -> bool {
63        !matches!(self.idle_warning, IdleWarning::None)
64    }
65
66    /// The server clock when the warning arrived, if there was one.
67    ///
68    /// `None` when not warned **or** when warned with no clock yet, which is why
69    /// [`Self::idle_warned`] is the liveness test and this is only the timestamp.
70    /// Conflating them would make a warning during the login burst -- before the
71    /// first prompt -- read as no warning at all.
72    #[must_use]
73    pub const fn idle_warned_at(&self) -> Option<u32> {
74        match self.idle_warning {
75            IdleWarning::At(at) => Some(at),
76            IdleWarning::None | IdleWarning::Unstamped => None,
77        }
78    }
79
80    /// The character acted, so the warning is answered.
81    ///
82    /// Called by the session actor on every outbound write, because **only an
83    /// outbound command answers an idle warning** and this model is inbound-only.
84    ///
85    /// The obvious alternative -- clear it on the next `<prompt>` -- is wrong, and
86    /// measurably so: in a recorded session the character idled on for
87    /// minutes after the warning while prompts arrived every few seconds, driven
88    /// by `dialogData id='Buffs'` refreshes and a bystander's emote. A prompt
89    /// means something happened, not that the player did something.
90    ///
91    /// Idempotent, and cheap enough to call unconditionally: it is one enum store
92    /// on a path that is already doing a socket write.
93    pub const fn answer_idle_warning(&mut self) {
94        self.idle_warning = IdleWarning::None;
95    }
96}

98/// The server's idle warning, exactly as it arrives once the parser has stripped
99/// the bell characters that wrap it on the wire.
100///
101/// MEASURED: 6 occurrences across 6 characters in a log archive,
102/// byte-identical every time. Matched whole, never as a substring -- see
103/// `GameState::apply`.
104///
105/// # Nothing inbound clears it
106///
107/// A measurement rather than an omission. The obvious guess -- "the next
108/// prompt means the player answered" -- is false: MEASURED in a recorded
109/// session, which idled on for minutes after the warning with
110/// prompts arriving every few seconds, driven by `dialogData id='Buffs'`
111/// refreshes and by a bystander emoting. **A prompt is sent when ANYTHING
112/// happens, not when the player acts.**
113///
114/// Only an OUTBOUND command answers an idle warning, and this model is
115/// inbound-only by design. So the session's actor clears it on write
116/// ([`GameState::answer_idle_warning`], called from its one write path,
117/// which every send passes through).
118///
119/// # Why this one line is text-derived
120///
121/// For the reasons `tests/idle_warning.rs` records: it is an exact string with
122/// nothing to extract, and the supervisor cannot otherwise tell an idle kick
123/// from a network drop.
124///
125/// Matched with `trim() ==`, not `contains`: a player can say anything, and a
126/// `contains` would let one make the supervisor stop reconnecting by typing
127/// it.
128pub(crate) const IDLE_WARNING: &str = "YOU HAVE BEEN IDLE TOO LONG. PLEASE RESPOND.";

1//! What the game says when a move does not simply work: a **stateless
2//! classifier** over one line of text, for travel's move recovery.
3//!
4//! # The ladder
5//!
6//! The patterns are the game's own text, gathered over years of play, and
7//! are tried as an ordered ladder: a line two patterns could claim is claimed
8//! by the earlier one, and the order is deliberate (see
9//! `the_first_pattern_that_matches_wins`).
10//!
11//! # What this does not do
12//!
13//! It names what a line *is*. What to do about it -- which remedy, how many
14//! tries, whether the exit is banned -- is the walker's, which has the state
15//! this file must not.
16

21/// What one line says about the move that was just sent.
22#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
23pub enum MoveFeedback {
24    /// Hidden or invisible, and this way in wants the walker seen. Remedy:
25    /// `unhide`.
26    MustUnhide,
27    /// One of several doors: this was the wrong one. Remedy: the next
28    /// ordinal (`go door` -> `go second door`).
29    WrongDoor,
30    /// **The exit is wrong for the map**: there is no such way from here. A
31    /// caller may drop the exit.
32    NoSuchWay,
33    /// Refused, but the exit is real: a guard, a ticket, a closed shop. Keep
34    /// the exit, give up on it for now.
35    Denied,
36    /// A skill roll failed and may succeed next time: a climb lost, a swim
37    /// pushed back, vertigo, fog. Remedy: wait out the roundtime, try again.
38    /// `fell` is true where the failure leaves the walker on the ground.
39    FailedRoll {
40        /// True when the message says the walker ended up on the ground.
41        fell: bool,
42    },
43    /// The move worked, though the room did not change in the usual way:
44    /// the long swims of Sailor's Grief.
45    Swam,
46    /// A climb that needs the hands free as well as another try.
47    FailedRollHandsFull,
48    /// Too injured to climb. Remedy: Sigil of Resolve, if known.
49    TooInjured,
50    /// `go` was sent and the game wants `climb`.
51    NeedsClimb,
52    /// `climb` was sent and the game wants `go`.
53    NeedsGo,
54    /// Dragging someone, and it did not work.
55    CannotDrag,
56    /// Remedy: empty the hands, and give them back after.
57    HandsFull,
58    /// Remedy: `open` it, once.
59    Closed,
60    /// `...wait N seconds.` -- roundtime, with the seconds.
61    Wait(u32),
62    /// Remedy: `stand`.
63    MustStand,
64    /// "still recovering from your recent..." Remedy: wait two seconds.
65    StillRecovering,
66    /// Too many commands in flight. Remedy: wait a second.
67    TypeAhead,
68    /// Remedy: wait for the stun to pass.
69    Stunned,
70    /// A floating disk would not follow. Remedy: just try again.
71    DiskWobbled,
72    /// Something of the walker's is on the ground. Remedy: `stow feet`.
73    ItemAtFeet,
74    /// A boltstone trap fired: stunned, knocked down.
75    Trapped,
76    /// Held fast. Remedy: wait to regain control.
77    Held,
78    /// No light. **The move succeeded**: the walker arrived, it just cannot
79    /// see the room.
80    PitchDark,
81}

186fn ladder() -> &'static Vec<(Regex, MoveFeedback)> {
187    static LADDER_BUILT: OnceLock<Vec<(Regex, MoveFeedback)>> = OnceLock::new();
188    LADDER_BUILT.get_or_init(|| {
189        LADDER
190            .iter()
191            .filter_map(|(pattern, feedback)| Some((Regex::new(pattern).ok()?, *feedback)))
192            .collect()
193    })
194}

196/// How many of `LADDER`'s patterns compiled. A test holds this to the
197/// ladder's length: a pattern that failed to compile would otherwise be a
198/// class of line the walker silently never sees.
199#[must_use]
200pub fn patterns_compiled() -> (usize, usize) {
201    (ladder().len(), LADDER.len())
202}
203
204/// What this line says about a move, if anything. The first pattern of the
205/// ladder that matches wins.
206#[must_use]
207pub fn classify(line: &str) -> Option<MoveFeedback> {
208    for (pattern, feedback) in ladder() {
209        let Some(found) = pattern.captures(line) else {
210            continue;
211        };
212        return Some(match feedback {
213            MoveFeedback::Wait(_) => MoveFeedback::Wait(found.get(1)?.as_str().parse().ok()?),
214            // `^You are already(?! as far away as you can get)`.
215            MoveFeedback::NoSuchWay
216                if line.starts_with("You are already as far away as you can get") =>
217            {
218                continue;
219            }
220            other => *other,
221        });
222    }
223    None
224}

226#[cfg(test)]
227mod tests {
228    use super::*;
229
230    #[test]
231    fn every_pattern_compiles() {
232        let (built, written) = patterns_compiled();
233        assert_eq!(built, written);
234    }
235
236    #[test]
237    fn lines_are_classified() {
238        for (line, expected) in [
239            ("You can't go there.", MoveFeedback::NoSuchWay),
240            (
241                "I could not find what you were referring to.",
242                MoveFeedback::NoSuchWay,
243            ),
244            ("You may not pass.", MoveFeedback::NoSuchWay),
245            ("An unseen force prevents you.", MoveFeedback::Denied),
246            ("The gate appears to be closed.", MoveFeedback::Closed),
247            ("...wait 3 seconds.", MoveFeedback::Wait(3)),
248            ("Wait 1 sec.", MoveFeedback::Wait(1)),
249            ("You must be standing first.", MoveFeedback::MustStand),
250            (
251                "You can't jump when you're sitting down!",
252                MoveFeedback::MustStand,
253            ),
254            ("You are still stunned.", MoveFeedback::Stunned),
255            (
256                "You'll need empty hands to climb that.",
257                MoveFeedback::HandsFull,
258            ),
259            (
260                "You're going to have to climb that.",
261                MoveFeedback::NeedsClimb,
262            ),
263            ("You can't climb that.", MoveFeedback::NeedsGo),
264            (
265                "Sorry, you may only type ahead 1 command.",
266                MoveFeedback::TypeAhead,
267            ),
268            (
269                "It's pitch dark and you can't see a thing!",
270                MoveFeedback::PitchDark,
271            ),
272            (
273                "You can't enter the bank and remain hidden or invisible.",
274                MoveFeedback::MustUnhide,
275            ),
276            (
277                "You slip in the mud and fall flat on your back!",
278                MoveFeedback::FailedRoll { fell: true },
279            ),
280            (
281                "You struggle against the swift current to swim north.",
282                MoveFeedback::FailedRoll { fell: false },
283            ),
284        ] {
285            assert_eq!(classify(line), Some(expected), "{line}");
286        }
287    }
288
289    /// Order decides. "appears to be closed, perhaps you
290    /// should try again later?" is a shop that is shut (keep the exit), not a
291    /// door to open -- and only the ladder's order says so.
292    #[test]
293    fn the_first_pattern_that_matches_wins() {
294        assert_eq!(
295            classify("The shop appears to be closed, perhaps you should try again later?"),
296            Some(MoveFeedback::Denied)
297        );
298    }
299
300    #[test]
301    fn the_lookahead_is_kept() {
302        assert_eq!(
303            classify("You are already standing."),
304            Some(MoveFeedback::NoSuchWay)
305        );
306        assert_eq!(
307            classify("You are already as far away as you can get!"),
308            None
309        );
310    }

312    #[test]
313    fn ordinary_text_says_nothing_about_a_move() {
314        for line in ["[Town Square, Central]", "Obvious paths: north, south", ""] {
315            assert_eq!(classify(line), None, "{line}");
316        }

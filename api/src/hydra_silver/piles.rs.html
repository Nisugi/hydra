1//! Coin piles on the ground, by their id, shared by every character's
2//! watcher in the process.
3//!
4//! A search or a mug states its silver only to the one who did it, and the
5//! pile it drops gets one id everyone in the room sees. When another
6//! character of the household picks that pile up, by hand or by the fossil
7//! charm, the pile's birth says whose it was and how much it held. A pile
8//! nobody here dropped is unknown, and the picker's own line has to say
9//! what it held, if it does.
10
11use std::collections::HashMap;
12use std::sync::{Mutex, OnceLock};
13
14use hydra_ledger::Who;
15
16/// How long a pile is remembered before it is dropped as stale: picked up
17/// by someone outside the household, or left to the game's cleanup.
18const KEEP_SECONDS: u64 = 60 * 60;
19
20/// A pile's birth: who dropped it, and what it held.
21#[derive(Debug, Clone, PartialEq, Eq)]
22pub struct Birth {
23    /// Whose search or mug dropped it.
24    pub who: Who,
25    /// The silver it held.
26    pub amount: i64,
27}
28
29/// The piles known, by the game's id for each.
30#[derive(Debug, Default)]
31pub struct Piles {
32    known: Mutex<HashMap<String, (Birth, u64)>>,
33}
34
35impl Piles {
36    /// The process's one table: every character's watcher shares it.
37    pub fn shared() -> &'static Self {
38        static SHARED: OnceLock<Piles> = OnceLock::new();
39        SHARED.get_or_init(Self::default)
40    }
41
42    /// A pile dropped: `who`'s, holding `amount`, with the id `id`.
43    pub fn born(&self, id: &str, who: &Who, amount: i64) {
44        let mut known = self
45            .known
46            .lock()
47            .unwrap_or_else(std::sync::PoisonError::into_inner);
48        let now = seconds_now();
49        known.retain(|_, (_, at)| now.saturating_sub(*at) < KEEP_SECONDS);
50        known.insert(
51            id.to_owned(),
52            (
53                Birth {
54                    who: who.clone(),
55                    amount,
56                },
57                now,
58            ),
59        );
60    }
61
62    /// Whether the pile `id` has a birth here.
63    #[must_use]
64    pub fn knows(&self, id: &str) -> bool {
65        self.known
66            .lock()
67            .unwrap_or_else(std::sync::PoisonError::into_inner)
68            .contains_key(id)
69    }
70
71    /// The pile `id` grew by `amount`: a second search or mug dropped its
72    /// silver onto a pile already there, which the game keeps as one. The
73    /// pile stays its first finder's. Whether the pile was known.
74    pub fn grew(&self, id: &str, amount: i64) -> bool {
75        let mut known = self
76            .known
77            .lock()
78            .unwrap_or_else(std::sync::PoisonError::into_inner);
79        match known.get_mut(id) {
80            Some((birth, _)) => {
81                birth.amount += amount;
82                true
83            }
84            None => false,
85        }
86    }
87
88    /// The pile `id` picked up: its birth, and it is forgotten. `None` for
89    /// a pile nobody here dropped.
90    pub fn taken(&self, id: &str) -> Option<Birth> {
91        let mut known = self
92            .known
93            .lock()
94            .unwrap_or_else(std::sync::PoisonError::into_inner);
95        known.remove(id).map(|(birth, _)| birth)
96    }
97}
98
99fn seconds_now() -> u64 {
100    std::time::SystemTime::now()
101        .duration_since(std::time::UNIX_EPOCH)
102        .map_or(0, |d| d.as_secs())
103}

105#[cfg(test)]
106mod tests {
107    use super::*;
108
109    #[test]
110    fn a_pile_is_known_once_from_its_birth_to_its_pickup() {
111        let piles = Piles::default();
112        let zoleta = Who::new("prime", "Zoleta");
113        piles.born("892676299", &zoleta, 686);
114        assert_eq!(
115            piles.taken("892676299"),
116            Some(Birth {
117                who: zoleta,
118                amount: 686
119            })
120        );
121        assert_eq!(piles.taken("892676299"), None);
122        assert_eq!(piles.taken("1"), None);
123    }
124
125    #[test]
126    fn a_pile_grows_and_keeps_its_first_finder() {
127        let piles = Piles::default();
128        let zoleta = Who::new("prime", "Zoleta");
129        piles.born("1", &zoleta, 499);
130        assert!(piles.knows("1"));
131        assert!(piles.grew("1", 396));
132        assert!(!piles.grew("2", 5));
133        assert_eq!(
134            piles.taken("1"),
135            Some(Birth {
136                who: zoleta,
137                amount: 895
138            })
139        );
140    }
141}

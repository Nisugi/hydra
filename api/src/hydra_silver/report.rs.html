20/// The words after `silver`, answered for `who` at `now`, as lines.

22/// # Errors

24/// A word that is not one of the commands, or the ledger cannot be read.
25pub fn answer(ledger: &Ledger, who: &Who, rest: &str, now: i64) -> Result<Vec<String>, String> {
26    answer_report(ledger, who, rest, now).map(|report| report.text())

29/// The words after `silver`, answered for `who` at `now`, as blocks.
30///
31/// # Errors
32///
33/// A word that is not one of the commands, or the ledger cannot be read.
34pub fn answer_report(ledger: &Ledger, who: &Who, rest: &str, now: i64) -> Result<Report, String> {
35    let word = rest.split_whitespace().next().map(str::to_ascii_lowercase);
36    let today = eastern::format(eastern::date(now));
37    let reader = ledger.reader().map_err(|e| e.to_string())?;
38    match word.as_deref() {
39        None => {
40            let sums = reader
41                .transaction_sums(who, &today)
42                .map_err(|e| e.to_string())?;
43            let recent = reader.transactions(who, SHOWN).map_err(|e| e.to_string())?;
44            Ok(own(who, &today, &sums, &recent))
45        }
46        Some("all") => {
47            let everyone = reader
48                .transaction_sums_everyone(&today)
49                .map_err(|e| e.to_string())?;
50            Ok(all(&today, &everyone))
51        }
52        Some("help") => Ok(Report::from_lines(
53            HELP.iter().map(|&line| line.to_owned()).collect(),
54        )),
55        Some(other) => Err(format!(
56            "`{other}` is not a silver word; silver help lists them."
57        )),
58    }
59}

61/// A report asked for on a command line, `silver ...`, answered for `who`
62/// at `now`, as lines.
63///
64/// # Errors
65///
66/// The line is not a silver line, or the ledger cannot be read.
67pub fn run(ledger: &Ledger, who: &Who, line: &str, now: i64) -> Result<Vec<String>, String> {
68    run_report(ledger, who, line, now).map(|report| report.text())
69}

71/// A report asked for on a command line, `silver ...`, answered for `who`
72/// at `now`, as blocks: what a window that draws the reports runs.
73///
74/// # Errors
75///
76/// The line is not a silver line, or the ledger cannot be read.
77pub fn run_report(ledger: &Ledger, who: &Who, line: &str, now: i64) -> Result<Report, String> {
78    let line = line.trim().trim_start_matches(['.', ';']);
79    let (word, rest) = line.split_once(char::is_whitespace).unwrap_or((line, ""));
80    if !word.eq_ignore_ascii_case("silver") {
81        return Err(format!("{word} is not a silver report."));
82    }
83    answer_report(ledger, who, rest, now)
84}

86/// A day's sums, each kind at hand.
87struct Day<'a> {
88    sums: &'a [(TransactionKind, i64)],
89}
90
91impl Day<'_> {
92    fn of(&self, kind: TransactionKind) -> i64 {
93        self.sums
94            .iter()
95            .find(|(k, _)| *k == kind)
96            .map_or(0, |(_, sum)| *sum)
97    }
98
99    fn total(&self, pick: fn(TransactionKind) -> bool) -> i64 {
100        self.sums
101            .iter()
102            .filter(|(k, _)| pick(*k))
103            .map(|(_, sum)| *sum)
104            .sum()
105    }
106
107    /// Silver in (found and sold), out (bought, paid, fees and tips), net,
108    /// banked (deposits less withdrawals), and bounty points.
109    fn headline(&self) -> [(&'static str, i64); 5] {
110        let came = self.total(TransactionKind::is_income);
111        let went = self.total(TransactionKind::is_outgo);
112        [
113            ("in", came),
114            ("out", went),
115            ("net", came - went),
116            (
117                "banked",
118                self.of(TransactionKind::Deposit) - self.of(TransactionKind::Withdrawal),
119            ),
120            ("bounty points", self.of(TransactionKind::Bounty)),
121        ]
122    }
123}

125/// One character's report: today's sums as a card, each silver kind as a
126/// bar, then the newest transactions as a table.
127#[must_use]
128pub fn own(
129    who: &Who,
130    today: &str,
131    sums: &[(TransactionKind, i64)],
132    recent: &[Transaction],
133) -> Report {
134    let day = Day { sums };
135    let mut report = Report::default();
136    report.heading(format!("{}, today ({today})", who.character));
137    report.push(Block::Facts(
138        day.headline()
139            .iter()
140            .map(|(name, n)| ((*name).to_owned(), Cell::Number(*n)))
141            .collect(),
142    ));
143    // Each silver kind with a row today, in against out by colour.
144    let kinds: Vec<TransactionKind> = TransactionKind::ALL
145        .into_iter()
146        .filter(|k| *k != TransactionKind::Bounty && sums.iter().any(|(s, _)| s == k))
147        .collect();
148    if !kinds.is_empty() {
149        #[allow(clippy::cast_precision_loss)]
150        let mut chart = Chart::one(
151            ChartKind::Rows,
152            "silver by kind",
153            "silver",
154            kinds.iter().map(|k| k.as_str().to_owned()).collect(),
155            kinds.iter().map(|k| day.of(*k) as f64).collect(),
156            Hue::Primary,
157        );
158        chart.series[0].hues = Some(
159            kinds
160                .iter()
161                .map(|k| {
162                    if k.is_income() {
163                        Hue::Good
164                    } else if k.is_outgo() {
165                        Hue::Bad
166                    } else {
167                        Hue::Secondary
168                    }
169                })
170                .collect(),
171        );
172        report.push(Block::Chart(chart));
173    }
174    if recent.is_empty() {
175        report.note("No transaction is on file.");
176        return report;
177    }
178    let mut table = Table::new(vec![
179        Column::left("when"),
180        Column::left("kind"),
181        Column::right("amount"),
182        Column::left("with"),
183        Column::left("the line"),
184    ]);
185    for t in recent {
186        let party = match (t.party.as_deref(), t.found_by.as_deref()) {
187            (Some(party), Some(finder)) => format!("{party}, {finder}'s"),
188            (Some(party), None) => party.to_owned(),
189            (None, Some(finder)) => format!("{finder}'s"),
190            (None, None) => String::new(),
191        };
192        table.rows.push(vec![
193            Cell::Time(format!("{} {}", t.day, clock(t.stamp.at)), t.stamp.at),
194            Cell::Text(t.kind.as_str().to_owned()),
195            Cell::Number(t.amount),
196            Cell::Text(party),
197            Cell::Text(t.line.clone()),
198        ]);
199    }
200    report.heading(format!("The last {}", recent.len()));
201    report.push(Block::Table(table));
202    report
203}

205/// Every character's sums today as one table.
206#[must_use]
207pub fn all(today: &str, everyone: &[(Who, Sums)]) -> Report {
208    let mut report = Report::default();
209    if everyone.is_empty() {
210        report.note(format!("No transaction is on file today ({today})."));
211        return report;
212    }
213    report.heading(format!("Silver today ({today})"));
214    let mut table = Table::new(vec![
215        Column::left("character"),
216        Column::right("in"),
217        Column::right("out"),
218        Column::right("net"),
219        Column::right("banked"),
220        Column::right("bounty points"),
221    ]);
222    table.sortable = true;
223    let mut totals = [0i64; 5];
224    for (who, sums) in everyone {
225        let day = Day { sums };
226        let line = day.headline();
227        for (t, (_, n)) in totals.iter_mut().zip(line.iter()) {
228            *t += n;
229        }
230        let mut row = vec![Cell::Text(who.character.clone())];
231        row.extend(line.iter().map(|(_, n)| Cell::Number(*n)));
232        table.rows.push(row);
233    }
234    let mut footer = vec![Cell::Text("all".to_owned())];
235    footer.extend(totals.iter().map(|n| Cell::Number(*n)));
236    table.footer = Some(footer);
237    report.push(Block::Table(table));
238    report
239}

241/// `HH:MM` Eastern.
242fn clock(at: i64) -> String {
243    let seconds = (at + eastern::offset(at)).rem_euclid(eastern::DAY);
244    format!("{:02}:{:02}", seconds / 3600, seconds % 3600 / 60)
245}

249pub fn commas(n: i64) -> String {
250    grouped(n)

257    /// The facts of a report, as words.
258    fn facts(report: &Report) -> Vec<(String, String)> {
259        report
260            .blocks
261            .iter()
262            .find_map(|b| match b {
263                Block::Facts(f) => Some(f.iter().map(|(n, c)| (n.clone(), c.words())).collect()),
264                _ => None,
265            })
266            .unwrap_or_default()
267    }

269    #[test]
270    fn sums_lead_with_what_was_banked_and_the_bounty_points() {
271        let who = Who::new("prime", "Nisugi");
272        let sums = [
273            (TransactionKind::Deposit, 10_000),
274            (TransactionKind::Withdrawal, 2_500),
275            (TransactionKind::Bounty, 1609),
276            (TransactionKind::Found, 3_000),
277            (TransactionKind::Sold, 5_000),
278            (TransactionKind::Fee, 1_200),
279        ];
280        let report = own(&who, "2026-10-28", &sums, &[]);
281        assert!(
282            matches!(&report.blocks[0], Block::Heading(h) if h == "Nisugi, today (2026-10-28)")
283        );
284        let facts = facts(&report);
285        let said = |name: &str| {
286            facts
287                .iter()
288                .find(|(n, _)| n == name)
289                .map_or("", |(_, v)| v.as_str())
290        };
291        assert_eq!(said("in"), "8,000");
292        assert_eq!(said("out"), "1,200");
293        assert_eq!(said("net"), "6,800");
294        assert_eq!(said("banked"), "7,500");
295        assert_eq!(said("bounty points"), "1,609");
296        let Some(Block::Chart(chart)) = report.blocks.get(2) else {
297            panic!("a chart: {:?}", report.blocks);
298        };
299        assert_eq!(
300            chart.categories,
301            ["found", "sold", "fee", "deposit", "withdrawal"]
302        );
303        assert_eq!(
304            chart.series[0].values,
305            [3000.0, 5000.0, 1200.0, 10000.0, 2500.0]
306        );
307        assert_eq!(
308            chart.series[0].hues.as_deref(),
309            Some(
310                &[
311                    Hue::Good,
312                    Hue::Good,
313                    Hue::Bad,
314                    Hue::Secondary,
315                    Hue::Secondary
316                ][..]
317            )
318        );
319        assert!(
320            matches!(report.blocks.last(), Some(Block::Note(n)) if n == "No transaction is on file.")
321        );
322        let text = report.text();
323        assert_eq!(text[0], "Nisugi, today (2026-10-28):");
324        assert!(text[1].starts_with("  in    "), "{text:?}");
325    }

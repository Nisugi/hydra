1//! The reports named, for a window that lists them: each with the words
2//! that ask for it and a line saying what it shows. Only the reports.
3
4/// One report.
5#[derive(Debug, Clone, Copy, PartialEq, Eq)]
6pub struct Entry {
7    /// What the list calls it.
8    pub name: &'static str,
9    /// The words after `silver` that ask for it; none for the character's own.
10    pub words: &'static str,
11    /// A line on what it shows.
12    pub blurb: &'static str,
13}
14
15/// Every report, in the order a list shows them.
16pub const CATALOG: &[Entry] = &[
17    Entry {
18        name: "Silver today",
19        words: "",
20        blurb: "Today's silver in and out: found, sold, bought, paid, fees, banked, bounty points, and the last ten lines",
21    },
22    Entry {
23        name: "Silver today, everyone",
24        words: "all",
25        blurb: "Every character's silver in and out today",
26    },
27];
28
29impl Entry {
30    /// Whether `query`'s words all appear in its name or its blurb.
31    #[must_use]
32    pub fn matches(&self, query: &str) -> bool {
33        let text = format!("{} {} {}", self.name, self.words, self.blurb).to_lowercase();
34        query
35            .split_whitespace()
36            .all(|word| text.contains(&word.to_lowercase()))
37    }
38
39    /// The command line that asks for it: `silver`, `silver all`.
40    #[must_use]
41    pub fn command(&self) -> String {
42        if self.words.is_empty() {
43            "silver".to_owned()
44        } else {
45            format!("silver {}", self.words)
46        }
47    }
48}

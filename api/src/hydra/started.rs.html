1//! How long Hydra takes to start, stage by stage, written to its log: the
2//! wall clock as `main` began, then each stage with the time since, up to
3//! the first frame the windows draw. A start that is slow on one computer
4//! says in its log where the time went, before `main` ran included: that
5//! is the clock's first line against the moment Hydra was asked for.
6
7use std::path::{Path, PathBuf};
8use std::sync::{Mutex, OnceLock};
9use std::time::Instant;
10
11/// When `main` began, and the log once the data folder is known.
12struct Start {
13    began: Instant,
14    log: Option<PathBuf>,
15}
16
17static START: OnceLock<Mutex<Start>> = OnceLock::new();
18
19/// `main` has begun: the clock starts, before anything else.
20pub(crate) fn begin() {
21    let _ = START.set(Mutex::new(Start {
22        began: Instant::now(),
23        log: None,
24    }));
25}
26
27/// The data folder is known: the log is there, and its first line is when
28/// `main` began, by the wall clock, and how long ago that was.
29pub(crate) fn log_in(data: &Path) {
30    let now = jiff::Zoned::now();
31    let Some(start) = START.get() else { return };
32    let Ok(mut start) = start.lock() else { return };
33    let since = start.began.elapsed();
34    start.log = Some(data.join(crate::startup::LOG));
35    let began = now
36        .checked_sub(jiff::SignedDuration::try_from(since).unwrap_or_default())
37        .unwrap_or(now);
38    write(
39        start.log.as_deref(),
40        &format!(
41            "[start] main began at {}",
42            began.strftime("%Y-%m-%d %H:%M:%S%.3f")
43        ),
44    );
45    write(
46        start.log.as_deref(),
47        &line(since, "the data folder is claimed"),
48    );
49}
50
51/// `what` is done: a line with the time since `main` began.
52pub(crate) fn stage(what: &str) {
53    let Some(start) = START.get() else { return };
54    let Ok(start) = start.lock() else { return };
55    write(start.log.as_deref(), &line(start.began.elapsed(), what));
56}
57
58fn line(since: std::time::Duration, what: &str) -> String {
59    format!("[start] +{:>7.3}s {what}", since.as_secs_f64())
60}
61
62/// A line to the log, and to stderr: a terminal sees it too. Nothing is
63/// said when the log cannot be written; a start is not stopped for it.
64fn write(log: Option<&Path>, text: &str) {
65    eprintln!("{text}");
66    if let Some(log) = log
67        && let Ok(mut file) = crate::startup::open_log(log)
68    {
69        use std::io::Write as _;
70        let _ = writeln!(file, "{text}");
71    }
72}

74#[cfg(test)]
75mod tests {
76    use super::*;
77
78    /// A stage's line carries the seconds since the start, to the
79    /// millisecond, aligned.
80    #[test]
81    fn a_stage_says_how_long_since_the_start() {
82        assert_eq!(
83            line(std::time::Duration::from_millis(1234), "tray started"),
84            "[start] +  1.234s tray started"
85        );

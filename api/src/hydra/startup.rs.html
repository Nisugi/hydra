1//! Hydra starting on its own: `hydra serve --detach`, which leaves the
2//! console and rests in the background; on Windows, the sign-in entry that
3//! runs it; and Hydra starting fresh, which starts the next process so, with
4//! `--again` (`rest`).
5//!
6//! `--in <dir>` says the working directory, which the data folder and the
7//! logs are found from: at sign-in the system starts Hydra elsewhere. The
8//! sign-in entry is the user's Run key, holding the command that starts
9//! Hydra here, in the directory it ran in when the entry was made.
10//!
11//! On a Mac, Hydra started from its app runs in the user's Application
12//! Support folder, `Hydra` in it ([`in_the_apps_home`]): the system starts
13//! an app at the disk's root, where nothing can be made, and an app's
14//! files belong there. Hydra started as a program alone, not from an app,
15//! runs where it is started, as everywhere else.
16

25/// Hydra's own log while it rests with no console, in the data folder.
26pub(crate) const LOG: &str = "hydra.log";
27
28/// How long the log may grow before a start begins it afresh.
29const LOG_MOST: u64 = 8 * 1024 * 1024;
30
31/// The program running, as it was found when Hydra started.
32static EXE: OnceLock<Result<PathBuf, String>> = OnceLock::new();
33
34/// The program running, as it was found when Hydra started: taken first
35/// thing in `main` and kept. An update renames the running program aside,
36/// and some systems then name it by the name it was given, which is no
37/// longer the program to start again.
38///
39/// # Errors
40///
41/// Hydra cannot find itself: why.
42pub(crate) fn exe() -> Result<PathBuf, String> {
43    EXE.get_or_init(|| {
44        std::env::current_exe().map_err(|why| format!("Hydra cannot find itself: {why}"))
45    })
46    .clone()
47}

49/// The mark of a Hydra started in the background ([`detach`]): it takes no
50/// terminal's console as its own, so closing the terminal it was started
51/// from, which closes every program on its console, leaves it playing.
52pub(crate) const DETACHED: &str = "--detached";
53
54/// Whether this Hydra was started in the background ([`DETACHED`]).
55#[must_use]
56pub(crate) fn detached() -> bool {
57    std::env::args().any(|arg| arg == DETACHED)
58}

60/// `serve`'s arguments read: whether to detach, the directory said with
61/// `--in`, whether the process before started this one (`--again`), the
62/// port a team started for the web had the page on (`--web`), whether the
63/// hub opens at once (`--open`), each character a restart starts again
64/// (`--resume`), and the rest, the characters to start. [`DETACHED`] is
65/// passed by.
66#[derive(Debug, Default, PartialEq, Eq)]
67pub(crate) struct Serve {
68    pub(crate) detach: bool,
69    pub(crate) dir: Option<PathBuf>,
70    pub(crate) again: bool,
71    pub(crate) web: Option<u16>,
72    pub(crate) open: bool,
73    pub(crate) resume: Vec<Resume>,
74    pub(crate) rest: Vec<String>,
75}
76
77impl Serve {
78    /// Read `args`, everything after `serve`.
79    pub(crate) fn read(args: &[String]) -> Self {
80        let mut serve = Self::default();
81        let mut args = args.iter();
82        while let Some(arg) = args.next() {
83            match arg.as_str() {
84                "--detach" => serve.detach = true,
85                DETACHED => {}
86                "--in" => serve.dir = args.next().map(PathBuf::from),
87                "--again" => serve.again = true,
88                "--web" => serve.web = args.next().and_then(|port| port.parse().ok()),
89                "--open" => serve.open = true,
90                "--resume" => serve
91                    .resume
92                    .extend(args.next().map(|arg| Resume::read(arg))),
93                _ => serve.rest.push(arg.clone()),
94            }
95        }
96        serve
97    }
98
99    /// What a detached start is told: all of this but `--detach` and
100    /// `--in`, the directory being the one it starts in.
101    pub(crate) fn args(&self) -> Vec<String> {
102        let mut args = self.rest.clone();
103        if self.again {
104            args.push("--again".to_owned());
105        }
106        if let Some(port) = self.web {
107            args.extend(["--web".to_owned(), port.to_string()]);
108        }
109        if self.open {
110            args.push("--open".to_owned());
111        }
112        for resume in &self.resume {
113            args.extend(["--resume".to_owned(), resume.arg()]);
114        }
115        args
116    }
117}

119/// Start `hydra serve` with `args` again, with no console, from the
120/// working directory, and leave it running: its log goes to the data
121/// folder's [`LOG`]. On Windows it leaves the job of the terminal it was
122/// started from, so closing that terminal, which ends every process in its
123/// job, leaves it playing; a job that will not let it leave keeps it, and
124/// the log says so.
125///
126/// # Errors
127///
128/// The log could not be opened or Hydra could not be started again: why.
129pub(crate) fn detach(args: &[String]) -> Result<(), String> {
130    let data = crate::account::data_dir();
131    std::fs::create_dir_all(&data)
132        .map_err(|why| format!("The data folder {} cannot be made: {why}", data.display()))?;
133    let path = data.join(LOG);
134    let exe = exe()?;
135    let start = |leave_job: bool| -> Result<(), String> {
136        let log =
137            open_log(&path).map_err(|why| format!("{} cannot be opened: {why}", path.display()))?;
138        let mut command = Command::new(&exe);
139        command
140            .arg("serve")
141            .arg(DETACHED)
142            .args(args)
143            .stdin(Stdio::null())
144            .stdout(
145                log.try_clone()
146                    .map_err(|why| format!("{} cannot be shared: {why}", path.display()))?,
147            )
148            .stderr(log);
149        #[cfg(windows)]
150        {
151            use std::os::windows::process::CommandExt as _;
152            // No console at all, and out of this one's group, so a Ctrl-C
153            // here does not reach it; and out of the terminal's job.
154            const DETACHED_PROCESS: u32 = 0x0000_0008;
155            const CREATE_NEW_PROCESS_GROUP: u32 = 0x0000_0200;
156            const CREATE_BREAKAWAY_FROM_JOB: u32 = 0x0100_0000;
157            let leave = if leave_job {
158                CREATE_BREAKAWAY_FROM_JOB
159            } else {
160                0
161            };
162            command.creation_flags(DETACHED_PROCESS | CREATE_NEW_PROCESS_GROUP | leave);
163        }
164        #[cfg(not(windows))]
165        let _ = leave_job;
166        command
167            .spawn()
168            .map(drop)
169            .map_err(|why| format!("Hydra cannot be started again: {why}"))
170    };
171    // Said in the log too: the console this is said on may be cleared.
172    let said = match start(true) {
173        Ok(()) => "[hydra] started out of the terminal's job: closing it leaves Hydra".to_owned(),
174        Err(why) => {
175            start(false)?;
176            format!(
177                "[hydra] could not leave the terminal's job ({why}): closing that terminal                  ends Hydra"
178            )
179        }
180    };
181    eprintln!("{said}");
182    if let Ok(mut log) = open_log(&path) {
183        let _ = writeln!(log, "{said}");
184    }
185    eprintln!(
186        "[hydra] resting in the background; its log is {}",
187        path.display()
188    );
189    Ok(())
190}

192/// The log, to add to; begun afresh once it has grown past [`LOG_MOST`].
193pub(crate) fn open_log(path: &Path) -> std::io::Result<std::fs::File> {
194    let long = std::fs::metadata(path).is_ok_and(|kept| kept.len() > LOG_MOST);
195    let mut options = OpenOptions::new();
196    options.create(true);
197    if long {
198        options.write(true).truncate(true);
199    } else {
200        options.append(true);
201    }
202    options.open(path)
203}

205/// The command the sign-in entry runs: `exe` resting, detached, in `dir`.
206#[cfg(any(windows, test))]
207fn sign_in_command(exe: &Path, dir: &Path) -> String {
208    format!(
209        "{} serve --detach --in {}",
210        quoted(exe.display()),
211        quoted(dir.display())
212    )
213}

215/// `word`, a path among them, in double quotes, for a Windows command line.
216#[cfg(any(windows, test))]
217pub(crate) fn quoted(word: impl std::fmt::Display) -> String {
218    let word = word.to_string();
219    // A backslash before the closing quote would escape it: `G:\` is `G:\.`.
220    if word.ends_with('\\') {
221        format!("\"{word}.\"")
222    } else {
223        format!("\"{word}\"")
224    }
225}

227/// The user's Run key, which the system runs at sign-in, and Hydra's entry
228/// in it.
229#[cfg(windows)]
230const RUN: &str = r"Software\Microsoft\Windows\CurrentVersion\Run";
231#[cfg(windows)]
232const ENTRY: &str = "Hydra";
233
234/// Whether Hydra starts at sign-in.
235#[cfg(windows)]
236pub(crate) fn starts_at_sign_in() -> bool {
237    windows_registry::CURRENT_USER
238        .open(RUN)
239        .and_then(|key| key.get_string(ENTRY))
240        .is_ok()
241}
242
243/// Start Hydra at sign-in, resting, as it runs now -- this program, in this
244/// directory -- or no longer: what was done, in words.
245///
246/// # Errors
247///
248/// The Run key would not change, or Hydra cannot find itself: why.
249#[cfg(windows)]
250pub(crate) fn start_at_sign_in(on: bool) -> Result<String, String> {
251    let key = windows_registry::CURRENT_USER
252        .create(RUN)
253        .map_err(|why| format!("The sign-in entry cannot be changed: {why}"))?;
254    if on {
255        let exe = exe()?;
256        let dir = std::env::current_dir()
257            .map_err(|why| format!("Hydra cannot find where it runs: {why}"))?;
258        key.set_string(ENTRY, sign_in_command(&exe, &dir))
259            .map_err(|why| format!("The sign-in entry cannot be made: {why}"))?;
260        Ok("Hydra starts at sign-in, resting.".to_owned())
261    } else {
262        if starts_at_sign_in() {
263            key.remove_value(ENTRY)
264                .map_err(|why| format!("The sign-in entry cannot be taken away: {why}"))?;
265        }
266        Ok("Hydra no longer starts at sign-in.".to_owned())
267    }
268}

270/// Whether `exe` is an app's own program, `<Name>.app/Contents/MacOS/<exe>`.
271fn in_an_app(exe: &Path) -> bool {
272    let named = |dir: Option<&Path>, name: &str| {
273        dir.and_then(Path::file_name).is_some_and(|dir| dir == name)
274    };
275    let mut up = exe.ancestors().skip(1);
276    let (macos, contents, app) = (up.next(), up.next(), up.next());
277    named(macos, "MacOS")
278        && named(contents, "Contents")
279        && app
280            .and_then(Path::extension)
281            .is_some_and(|extension| extension.eq_ignore_ascii_case("app"))
282}

284/// Started from its app, Hydra runs in `~/Library/Application Support/Hydra`,
285/// made if it is not there yet; a program started alone stays where it is.
286/// Only a Mac's program is ever inside an app, so this is asked everywhere.
287///
288/// # Errors
289///
290/// Hydra cannot find itself or the home folder, or that folder cannot be
291/// made or entered: why.
292pub(crate) fn in_the_apps_home() -> Result<(), String> {
293    let exe = exe()?;
294    if !in_an_app(&exe) {
295        return Ok(());
296    }
297    let home =
298        std::env::var_os("HOME").ok_or("Hydra cannot find the home folder: HOME is not set")?;
299    let dir = Path::new(&home)
300        .join("Library")
301        .join("Application Support")
302        .join("Hydra");
303    std::fs::create_dir_all(&dir)
304        .map_err(|why| format!("{} cannot be made: {why}", dir.display()))?;
305    std::env::set_current_dir(&dir)
306        .map_err(|why| format!("Hydra cannot run in {}: {why}", dir.display()))
307}

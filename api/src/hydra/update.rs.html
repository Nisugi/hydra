1//! Hydra updating itself (`hydra_update`): a newer release looked for as
2//! Hydra starts and every six hours while *Check for updates* is on, said
3//! on the hub and in the tray, and fetched when either asks for it, or as
4//! soon as it is found with *Update on its own* on. Fetched, it is in place
5//! of the program running, and runs from Hydra's next start: when Hydra
6//! starts fresh, or is started again.
7//!
8//! The settings are read again every minute, so a switch turned on takes
9//! within one. A look that failed is made again in a quarter of an hour. A
10//! fetch that failed is not tried again on its own; the hub offers it
11//! again.
12
13use std::path::{Path, PathBuf};
14use std::time::Duration;
15
16use hydra_board::{Fetch, Sessions, Update};
17use hydra_update::{Offer, Releases, Version};
18use tokio::time::Instant;
19
20/// How often a newer release is looked for.
21const EVERY: Duration = Duration::from_hours(6);
22
23/// How soon a look that failed is made again: started at sign-in, Hydra
24/// may look before the network is up.
25const AGAIN: Duration = Duration::from_mins(15);
26
27/// How often the settings are read again.
28const TICK: Duration = Duration::from_mins(1);
29
30/// As Hydra starts on `data`, the data folder: the program an update set
31/// aside goes, the process before having left, and a newer release is
32/// looked for on `runtime` for the rest of the run.
33pub(crate) fn start(runtime: &tokio::runtime::Runtime, sessions: &Sessions, data: &Path) {
34    match crate::startup::exe() {
35        Ok(exe) => {
36            hydra_update::clear_old(&exe);
37            runtime.spawn(keep(sessions.clone(), data.to_path_buf(), exe));
38        }
39        Err(why) => eprintln!("[hydra] no updates: {why}"),
40    }
41}

43/// Look for, say and fetch an update for the rest of the run: `data` holds
44/// the settings, and `exe` is the program running, which an update takes
45/// the place of.
46async fn keep(sessions: Sessions, data: PathBuf, exe: PathBuf) {
47    let own = match Version::parse(hydra_table::VERSION) {
48        Ok(own) => own,
49        Err(why) => {
50            eprintln!(
51                "[hydra] no updates: this Hydra's version, {}, is no version: {why}",
52                hydra_table::VERSION
53            );
54            return;
55        }
56    };
57    let releases = match Releases::new() {
58        Ok(releases) => releases,
59        Err(why) => {
60            eprintln!("[hydra] no updates: {why}");
61            return;
62        }
63    };
64    let asked = sessions.update_asked();
65    let mut offer: Option<Offer> = None;
66    let mut next_look = Instant::now();
67    // The release last fetched on its own, so a failure is not tried again
68    // every minute.
69    let mut tried: Option<Version> = None;
70    let mut ready = false;
71    loop {
72        let (check, on_its_own) = hydra_board::updates(&data);
73        if !ready && check && Instant::now() >= next_look {
74            match releases.look(&own).await {
75                Ok(found) => {
76                    next_look = Instant::now() + EVERY;
77                    let was = offer.as_ref().map(|offer| &offer.version);
78                    if found.as_ref().map(|found| &found.version) != was {
79                        if let Some(found) = &found {
80                            eprintln!("[hydra] Hydra {} is out", found.version);
81                        }
82                        sessions.update(found.as_ref().map(|found| said(found, Fetch::Out)));
83                    }
84                    offer = found;
85                }
86                Err(why) => {
87                    next_look = Instant::now() + AGAIN;
88                    eprintln!("[hydra] no newer release could be looked for: {why}");
89                }
90            }
91        }
92        let fetch_now = !ready
93            && on_its_own
94            && offer
95                .as_ref()
96                .is_some_and(|offer| tried.as_ref() != Some(&offer.version));
97        if !fetch_now {
98            tokio::select! {
99                () = asked.notified() => {}
100                () = tokio::time::sleep(TICK) => continue,
101            }
102        }
103        let Some(offer) = offer.as_ref().filter(|_| !ready) else {
104            continue;
105        };
106        tried = Some(offer.version.clone());
107        ready = fetch(&sessions, &releases, offer, &exe).await;
108    }
109}

111/// Fetch `offer` into the place of `exe`, saying how far it has come:
112/// whether it is in place.
113async fn fetch(sessions: &Sessions, releases: &Releases, offer: &Offer, exe: &Path) -> bool {
114    sessions.update(Some(said(offer, Fetch::Fetching(0, 0))));
115    // Said again at each whole percent, not at every piece that comes.
116    let mut percent = 0;
117    let got = releases
118        .fetch(offer, exe, |got, total| {
119            let now = got.saturating_mul(100) / total.max(1);
120            if now != percent {
121                percent = now;
122                sessions.update(Some(said(offer, Fetch::Fetching(got, total))));
123            }
124        })
125        .await;
126    match got {
127        Ok(()) => {
128            eprintln!(
129                "[hydra] Hydra {} is in place: it runs from Hydra's next start",
130                offer.version
131            );
132            sessions.update(Some(said(offer, Fetch::Ready)));
133            true
134        }
135        Err(why) => {
136            eprintln!(
137                "[hydra] Hydra {} could not be fetched: {why}",
138                offer.version
139            );
140            sessions.update(Some(said(offer, Fetch::Failed(why))));
141            false
142        }
143    }
144}

146/// `offer`, as the hub and the tray say it, `fetch` far along.
147fn said(offer: &Offer, fetch: Fetch) -> Update {
148    Update {
149        version: offer.version.to_string(),
150        page: offer.page.clone(),
151        fetch,
152    }

1//! `hydra`, `hydra gui` and `hydra serve`: Hydra's own process, every
2//! character on a supervised session in one host table, and its windows
3//! whenever they are wanted.
4//!
5//! Hydra runs once on a data folder at a time (`instance`). Another launch
6//! on it opens the windows of the process running, and leaves.
7//!
8//! Hydra rests between its windows: its characters play on, the web page is
9//! served while its setting is on (`hydra_web::serve`), and the computer is
10//! kept awake while a character plays (`rest`). On Windows its icon is in
11//! the tray throughout (`tray`). `hydra serve` starts it resting; `hydra`
12//! and `hydra gui` open the windows at once. The windows take the main
13//! thread while they are open, so the sessions and their feeds run on the
14//! runtime's workers.
15//!
16//! Characters named on the command line start at once, their passwords asked
17//! at the terminal when none is kept; the hub and the web page start the
18//! roster's. Closing the hub hides it while another window is open, and
19//! once none is the windows close and Hydra rests. Its Shut down, or Ctrl-C,
20//! stops every character in order: `quit` to each, a wait for the game to
21//! close each connection, then the sessions end. A second Ctrl-C leaves at
22//! once.
23//!
24//! Rested long enough with nothing playing, Hydra starts fresh
25//! (`rest::until_fresh`): it starts the next process in the background,
26//! resting, and leaves, the lock held to its end; the next waits for it to
27//! go. A web page a team started for the web had is served by the next as
28//! well. A page open on a phone loses its connection for a moment and finds
29//! the next by itself. An update fetched (`update`) is the program the next
30//! process starts as.
31//!
32//! Each character's session runs on a runtime of its own, two threads
33//! (`game_runtime`), apart from the windows', the page's and the rest of
34//! Hydra's.
35//!
36//! The hub's *Restart now*, once an update is fetched, shuts Hydra down in
37//! order as Shut down does and starts the next process the same way, but
38//! with the hub open, and, if the player kept the box ticked, the
39//! characters that were playing started again, each with its team
40//! (`Table::resume`).

55/// What a start was told besides its characters and whether its windows
56/// open at once.
57#[derive(Clone, Copy, Debug, Default)]
58pub(crate) struct Told {
59    /// The process before started it fresh, and is leaving (`--again`).
60    pub(crate) again: bool,
61    /// The web page is served on this port for the run, whatever its
62    /// setting, as a team started for the web has it (`--web`).
63    pub(crate) web: Option<u16>,
64}
65
66/// How many threads the characters' sessions run on, at most.
67const GAME_THREADS: usize = 2;
68
69/// How long the characters' runtime is given to finish, once every
70/// character has stopped, before Hydra goes on without it.
71const GAME_STOPS: std::time::Duration = std::time::Duration::from_secs(5);
72
73/// The runtime the characters' sessions run on, apart from the windows',
74/// the page's and the rest of Hydra's: their game text is read on its
75/// threads alone, two (one on a machine of one core), however many
76/// characters play. Each pattern the model matches keeps a copy of its
77/// caches for every thread that ever runs it, so this bounds the copies.
78///
79/// # Errors
80///
81/// The runtime could not start: why.
82fn game_runtime() -> Result<tokio::runtime::Runtime, String> {
83    let cores = std::thread::available_parallelism().map_or(1, std::num::NonZero::get);
84    tokio::runtime::Builder::new_multi_thread()
85        .worker_threads(GAME_THREADS.min(cores))
86        .thread_name("game")
87        .enable_all()
88        .on_thread_start(|| hydra_mem::thread(hydra_mem::Part::Background))
89        .build()
90        .map_err(|why| format!("the characters' runtime cannot start: {why}"))
91}

93/// How a rest between the windows ended.
94enum Rested {
95    /// The windows were asked for.
96    Open(Open),
97    /// Hydra is shut down.
98    Closed,
99    /// Nothing has played for long enough: Hydra starts fresh.
100    Fresh,
101}

103/// Log every target in, start again each character of `resume`, which a
104/// restart named, and run Hydra until it is shut down: its windows open at
105/// once when `open`, and whenever another launch asks for them.
106///
107/// # Errors
108///
109/// What failed before a named character's session could start, or the data
110/// folder could not be claimed.
111pub(crate) fn run(
112    runtime: &tokio::runtime::Runtime,
113    targets: Vec<Target>,
114    open: bool,
115    told: Told,
116    resume: Vec<Resume>,
117) -> Result<(), String> {
118    let data = crate::account::data_dir();
119    let Some((held, asks)) = claim(&data, open, told, !targets.is_empty())? else {
120        return Ok(());
121    };
122    crate::started::log_in(&data);
123    // Every password the command line needs, asked before anything opens.
124    let mut logins = Vec::new();
125    for target in targets {
126        let login = target.resolve()?;
127        let password = crate::account::password(&login.account).map_err(|e| e.to_string())?;
128        logins.push((login, password));
129    }
130    let game = game_runtime()?;
131    let (sessions, table) = seat(runtime, game.handle(), logins, resume);
132    crate::started::stage("the roster is read and its passwords looked for");
133
134    serve_page(runtime, &sessions, &data, told);
135    crate::started::stage("the web page is served");
136    let windows = Arc::new(AtomicBool::new(false));
137    runtime.spawn(crate::rest::watch(sessions.clone(), Arc::clone(&windows)));
138    crate::update::start(runtime, &sessions, &data);
139    // Another launch, the tray, a character started or shown again, or a
140    // team started with its play windows asks for the windows: open ones
141    // show the hub in front, or a seat's play window, or leave the hub be
142    // for play windows; or the main thread is told to open them. The
143    // sender is held here as well, so the asking never ends while Hydra
144    // runs.
145    let (ask, mut opens) = mpsc::unbounded_channel();
146    let opener: Opener = {
147        let (wake, asking) = (sessions.wake(), ask.clone());
148        Arc::new(move |open| {
149            let open_already = match open {
150                Open::Hub => wake.raise(),
151                // Open already, each play window opens as it is seated.
152                Open::PlayWindows => wake.is_open(),
153                Open::Seat(id) => wake.raise_seat(id),
154            };
155            if !open_already {
156                let _ = asking.send(open);
157            }
158        })
159    };
160    sessions.opener(Arc::clone(&opener));
161    let bring_up = move || opener(Open::Hub);
162    runtime.spawn(instance::answer(asks, bring_up.clone()));
163    #[cfg(windows)]
164    let tray = crate::tray::Tray::start(sessions.clone(), runtime.handle().clone(), bring_up);
165    crate::started::stage("the tray is up");
166    runtime.spawn(stop_on_ctrl_c(Arc::clone(&table)));
167
168    let mut closes = sessions.closes();
169    let mut open_now = open.then_some(Open::Hub);
170    // The windows have been open in this run: theirs to give back.
171    let mut used = false;
172    let fresh = loop {
173        if let Some(how) = open_now {
174            used = true;
175            windows.store(true, Ordering::Relaxed);
176            crate::started::stage("the windows are opening");
177            hydra_gui::opening(crate::started::stage);
178            let shown = hydra_mem::part(hydra_mem::Part::Windows, || {
179                hydra_gui::run(sessions.clone(), runtime.handle().clone(), &data, how)
180            });
181            windows.store(false, Ordering::Relaxed);
182            if let Err(e) = shown {
183                eprintln!("[hydra] the windows cannot open: {e}");
184            }
185            // An ask that came as they closed is not a reason to open them
186            // again at once.
187            while opens.try_recv().is_ok() {}
188        }
189        if *closes.borrow() {
190            break false;
191        }
192        eprintln!("[hydra] resting: `hydra` opens the windows, Ctrl-C shuts Hydra down");
193        let rested = runtime.block_on(async {
194            tokio::select! {
195                // An ask for the windows as the time comes is answered.
196                biased;
197                asked = opens.recv() => asked.map_or(Rested::Closed, Rested::Open),
198                _ = closes.wait_for(|closing| *closing) => Rested::Closed,
199                () = crate::rest::until_fresh(&sessions, &data, used) => Rested::Fresh,
200            }
201        });
202        match rested {
203            Rested::Open(how) => open_now = Some(how),
204            Rested::Closed => break false,
205            Rested::Fresh => break true,
206        }
207    };
208
209    if fresh {
210        eprintln!(
211            "[hydra] nothing has played for a while: starting fresh, giving back what play held"
212        );
213    } else {
214        eprintln!("[hydra] shutting down; quitting every character");
215    }
216    runtime.block_on(table.stop_everything());
217    game.shutdown_timeout(GAME_STOPS);
218    #[cfg(windows)]
219    if let Some(tray) = tray {
220        tray.close();
221    }
222    if fresh {
223        start_next(&sessions, None);
224    } else if let Some(resume) = table.restarting() {
225        start_next(&sessions, Some(resume));
226    }
227    drop((ask, held));
228    Ok(())
229}

231/// Claim `data`, the data folder, for this process: its lock, and where
232/// other launches ask it for its windows. `None` when Hydra already runs
233/// there and this process leaves: having asked for the windows when `open`,
234/// or because it was started fresh (`told.again`) and another took its
235/// place.
236///
237/// # Errors
238///
239/// The data folder could not be claimed, or Hydra already runs there and
240/// this process was started to rest.
241fn claim(
242    data: &Path,
243    open: bool,
244    told: Told,
245    characters: bool,
246) -> Result<Option<(Held, Asks)>, String> {
247    let claimed = if told.again {
248        instance::claim_once_left(data)?
249    } else {
250        instance::claim(data)?
251    };
252    match claimed {
253        Claim::First(held, asks) => {
254            if told.again {
255                eprintln!("[hydra] started fresh by the process before this one");
256            }
257            Ok(Some((held, asks)))
258        }
259        Claim::Running if told.again => {
260            eprintln!(
261                "[hydra] Hydra already runs on {}: this process leaves",
262                data.display()
263            );
264            Ok(None)
265        }
266        Claim::Running if open => {
267            instance::ask_open(data)?;
268            if characters {
269                eprintln!("[hydra] Hydra is already running: start characters from its hub.");
270            }
271            Ok(None)
272        }
273        Claim::Running => Err(format!(
274            "Hydra is already running on {}: `hydra` opens its windows.",
275            data.display()
276        )),
277    }
278}

280/// Serve the web page on `runtime` while it is wanted: by the settings the
281/// window keeps, kept in `data`, so with no window open too, and on the
282/// port `told.web` says for the run.
283/// The sessions and the table over them, answering the hub, with each
284/// character of `logins` started, each of `resume` started again, and the
285/// roster offered.
286fn seat(
287    runtime: &tokio::runtime::Runtime,
288    game: &tokio::runtime::Handle,
289    logins: Vec<(crate::account::Login, String)>,
290    resume: Vec<Resume>,
291) -> (Sessions, Arc<Table>) {
292    let sessions = Sessions::new(Wake::default());
293    let table = Table::with_game_runtime(sessions.clone(), runtime.handle().clone(), game.clone());
294    let answering = Arc::clone(&table);
295    let control: HubControl = Arc::new(move |request| {
296        let table = Arc::clone(&answering);
297        Box::pin(async move { table.answer(request).await })
298    });
299    sessions.control(control);
300    runtime.block_on(async {
301        for (login, password) in logins {
302            if let Err(e) = table.start(login, password, false, None).await {
303                eprintln!("[hydra] {e}");
304            }
305        }
306        table.resume(resume).await;
307        table.offer().await;
308    });
309    (sessions, table)
310}

312fn serve_page(runtime: &tokio::runtime::Runtime, sessions: &Sessions, data: &Path, told: Told) {
313    sessions.want_web(hydra_board::web_page(data));
314    if let Some(port) = told.web {
315        sessions.serve_web(port);
316    }
317    // The process before lets the page's port go as it ends.
318    let wanted = *sessions.web_wanted().borrow();
319    if told.again
320        && let Some(port) = wanted
321    {
322        instance::until_free(port);
323    }
324    let page = hydra_web::serve(sessions.clone(), data.to_path_buf());
325    runtime.spawn(hydra_mem::task(hydra_mem::Part::WebPage, page));
326}

328/// Start the next process and leave: fresh and resting, or, restarting,
329/// with the hub open and each character of `restart` started again. The
330/// web page a team started for the web had served goes on being served.
331/// Said, when it cannot be started, and this one ends all the same.
332fn start_next(sessions: &Sessions, restart: Option<Vec<Resume>>) {
333    if restart.is_some() {
334        eprintln!("[hydra] starting again");
335    }
336    let next = Serve {
337        again: true,
338        web: sessions.web_team(),
339        open: restart.is_some(),
340        resume: restart.unwrap_or_default(),
341        ..Serve::default()
342    };
343    match crate::startup::detach(&next.args()) {
344        // The lock held to the end, so the next has it only once this one,
345        // and the web page's port, are gone.
346        Ok(()) => std::process::exit(0),
347        Err(why) => eprintln!("[hydra] {why}: Hydra has shut down"),
348    }
349}

351/// Ctrl-C shuts Hydra down in order, as the hub's Shut down does; a second
352/// leaves at once.
353async fn stop_on_ctrl_c(table: Arc<Table>) {
354    if tokio::signal::ctrl_c().await.is_err() {
355        return;
356    }
357    eprintln!("[hydra] Ctrl-C: shutting down (Ctrl-C again leaves at once)");
358    table.answer(HubRequest::Shutdown).await;
359    if tokio::signal::ctrl_c().await.is_ok() {
360        eprintln!("[hydra] leaving now");
361        std::process::exit(1);
362    }

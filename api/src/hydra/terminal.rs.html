1//! A terminal opened by Hydra itself: `hydra-tui` in a console window of its
2//! own, on a character shown in a terminal as it starts, or on the hub from
3//! the tray.
4//!
5//! It runs the `hydra-tui` beside this executable from Hydra's working
6//! directory, where the data folder is, and attaches as any `hydra-tui`
7//! does.
8//!
9//! On Windows `cmd`'s `start` opens the console: the program it starts has
10//! the new console's own input and output, where one started directly
11//! would write to wherever Hydra's own output goes, its log or its console.
12//! Windows 11 hands a new console to the terminal its settings choose.
13//! Elsewhere the system's terminal is asked: Terminal on macOS,
14//! `x-terminal-emulator` on Linux and the like.
15

21/// Open a terminal on the seat numbered `id`, while it is seated: why not,
22/// when it cannot be, said to the player.
23pub(crate) fn open_on(sessions: &Sessions, id: u32) {
24    let Some(seat) = sessions.seats().into_iter().find(|seat| seat.id == id) else {
25        return;
26    };
27    if let Err(why) = open(Some(&format!("{}:{}", seat.game, seat.name))) {
28        eprintln!("[hydra] {why}");
29        sessions.tell(why);
30    }
31}

33/// Open a terminal attached to Hydra, on `seat` (`GAME:Name`) when one is
34/// named.
35///
36/// # Errors
37///
38/// Hydra cannot find itself or where it runs, or no terminal opens: why.
39pub(crate) fn open(seat: Option<&str>) -> Result<(), String> {
40    let exe = std::env::current_exe()
41        .map_err(|why| format!("Hydra cannot find itself: {why}"))?
42        .with_file_name(format!("hydra-tui{}", std::env::consts::EXE_SUFFIX));
43    let dir =
44        std::env::current_dir().map_err(|why| format!("Hydra cannot find where it runs: {why}"))?;
45    let mut child = command(&exe, &dir, seat)
46        .stdin(Stdio::null())
47        .spawn()
48        .map_err(|why| format!("No terminal could be opened: {why}"))?;
49    // What opens the terminal leaves at once; waited for, it leaves nothing
50    // behind.
51    let _ = std::thread::Builder::new()
52        .name("terminal".to_owned())
53        .spawn(move || child.wait());
54    Ok(())
55}

57/// What `hydra-tui` is told: to run in `dir`, on `seat` when one is named.
58fn tui_args(dir: &Path, seat: Option<&str>) -> Vec<String> {
59    let mut args = vec!["--in".to_owned(), dir.display().to_string()];
60    args.extend(seat.map(str::to_owned));
61    args
62}

64/// What opens the terminal: `cmd`, itself with no window, starting `exe` in
65/// a console of its own.
66#[cfg(windows)]
67fn command(exe: &Path, dir: &Path, seat: Option<&str>) -> Command {
68    use std::os::windows::process::CommandExt as _;
69    const CREATE_NO_WINDOW: u32 = 0x0800_0000;
70    let mut command = Command::new("cmd.exe");
71    command
72        .raw_arg(start_line(exe, dir, seat))
73        .creation_flags(CREATE_NO_WINDOW);
74    command
75}
76
77/// `cmd`'s command line, starting `exe`, `hydra-tui`, in a console of its
78/// own, its window titled Hydra. Each word quoted: `cmd` leaves what is
79/// quoted as it is, and the program reads its words out of the quotes.
80#[cfg(any(windows, test))]
81fn start_line(exe: &Path, dir: &Path, seat: Option<&str>) -> String {
82    let words: Vec<String> = std::iter::once(exe.display().to_string())
83        .chain(tui_args(dir, seat))
84        .map(crate::startup::quoted)
85        .collect();
86    format!("/d /c start \"Hydra\" {}", words.join(" "))
87}

89/// What opens the terminal: Terminal, told by `osascript` to run `exe`.
90#[cfg(target_os = "macos")]
91fn command(exe: &Path, dir: &Path, seat: Option<&str>) -> Command {
92    let mut command = Command::new("osascript");
93    command.args(apple_script(exe, dir, seat));
94    command
95}
96
97/// `osascript`'s arguments: Terminal runs `exe` with its words, each in
98/// single quotes for the shell, and comes to the front.
99#[cfg(any(target_os = "macos", test))]
100fn apple_script(exe: &Path, dir: &Path, seat: Option<&str>) -> Vec<String> {
101    let shell = |word: &str| format!("'{}'", word.replace('\'', r"'\''"));
102    let line = std::iter::once(exe.display().to_string())
103        .chain(tui_args(dir, seat))
104        .map(|word| shell(&word))
105        .collect::<Vec<_>>()
106        .join(" ");
107    let line = line.replace('\\', r"\\").replace('"', "\\\"");
108    vec![
109        "-e".to_owned(),
110        format!("tell application \"Terminal\" to do script \"{line}\""),
111        "-e".to_owned(),
112        "tell application \"Terminal\" to activate".to_owned(),
113    ]
114}

116/// What opens the terminal: the system's own, running `exe`.
117#[cfg(all(unix, not(target_os = "macos")))]
118fn command(exe: &Path, dir: &Path, seat: Option<&str>) -> Command {
119    let mut command = Command::new("x-terminal-emulator");
120    command.arg("-e").arg(exe).args(tui_args(dir, seat));
121    command
122}

124#[cfg(test)]
125mod tests {
126    use super::*;
127
128    /// `cmd` starts Hydra in a console of its own, every word quoted, a
129    /// root's backslash kept from escaping its quote.
130    #[test]
131    fn cmd_starts_hydra_in_a_console_of_its_own() {
132        let exe = Path::new("C:\\Program Files\\Hydra\\hydra.exe");
133        assert_eq!(
134            start_line(exe, Path::new("G:\\dev\\hydra"), Some("GS3:Ashryn")),
135            "/d /c start \"Hydra\" \"C:\\Program Files\\Hydra\\hydra.exe\" \"--in\" \
136             \"G:\\dev\\hydra\" \"GS3:Ashryn\""
137        );
138        assert_eq!(
139            start_line(exe, Path::new("G:\\"), None),
140            "/d /c start \"Hydra\" \"C:\\Program Files\\Hydra\\hydra.exe\" \"--in\" \
141             \"G:\\.\""
142        );
143    }
144
145    /// `hydra-tui` is told where to run, and the character when there is
146    /// one.
147    #[test]
148    fn the_terminal_runs_in_hydras_directory_on_its_character() {
149        let dir = Path::new("data");
150        assert_eq!(tui_args(dir, None), ["--in", "data"]);
151        assert_eq!(
152            tui_args(dir, Some("GS3:Ashryn")),
153            ["--in", "data", "GS3:Ashryn"]
154        );
155    }
156
157    /// Terminal is told the command line, each word quoted for the shell
158    /// and the whole for `osascript`.
159    #[test]
160    fn terminal_is_told_the_command_line() {
161        let script = apple_script(
162            Path::new("/Applications/Hydra's/hydra-tui"),
163            Path::new("/Users/me/hydra"),
164            Some("GS3:Ashryn"),
165        );
166        assert_eq!(
167            script[1],
168            "tell application \"Terminal\" to do script \"'/Applications/Hydra'\\\\''s/hydra-tui' \
169             '--in' '/Users/me/hydra' 'GS3:Ashryn'\""
170        );
171        assert_eq!(script[3], "tell application \"Terminal\" to activate");
172    }
173
174    /// On Windows the command is `cmd`, told the line as it is written.
175    #[cfg(windows)]
176    #[test]
177    fn the_command_is_cmd_with_its_line() {
178        let built = command(Path::new("C:\\hydra.exe"), Path::new("G:\\dev"), None);
179        assert_eq!(built.get_program(), "cmd.exe");
180        let args: Vec<_> = built.get_args().collect();
181        assert_eq!(
182            args,
183            ["/d /c start \"Hydra\" \"C:\\hydra.exe\" \"--in\" \"G:\\dev\""]
184        );
185    }

1//! `hydra play`: one character in the terminal, on a supervised session.
2//!
3//! The session holds the connection, the parser and the game state, and
4//! reconnects when the connection drops. This shows its events as readable
5//! text, sends each typed line as a command, and answers `;status` from the
6//! game state without sending it. Ctrl-C quits; a second Ctrl-C drops the
7//! connection.
8
9use std::time::{Duration, Instant};
10
11use hydra_protocol::Frame;
12use hydra_session::{Event, State, SupervisedSession};
13use tokio::io::{AsyncBufReadExt, AsyncWriteExt};
14use tokio::sync::broadcast::error::RecvError;
15
16use crate::account::{self, Login};
17use crate::connect::Target;
18use crate::connector::LiveConnector;
19use crate::render::Renderer;

21/// How long a quit waits for the game to close the connection.
22const QUIT_DEADLINE: Duration = Duration::from_secs(10);
23
24/// Log in and play until the session ends.
25///
26/// # Errors
27///
28/// What [`supervised`] says.
29pub async fn run(target: Target) -> Result<(), String> {
30    let (login, session) = supervised(target)?;
31    let handle = session.handle();
32    let alone = crate::relay::alone(&handle, &login.game_code, &login.character);
33    let lichs = crate::lich_command::Lichs::default();
34    crate::commands::install(&handle, &login, alone, &lichs);
35    crate::table::give_typing(&handle, &login.game_code, &login.character);
36    lichs.once_ready(&handle, &format!("{}:{}", login.game_code, login.character));
37    let cancel = session.cancel_token();
38    let mut events = handle.subscribe();
39    let mut running = tokio::spawn(session.run());
40
41    // Typed lines go to the session in order, one round trip at a time.
42    let (typed, mut lines) = tokio::sync::mpsc::unbounded_channel::<String>();
43    let sender = tokio::spawn({
44        let handle = handle.clone();
45        async move {
46            while let Some(line) = lines.recv().await {
47                let _ = handle.send_typed(&line).await;
48            }
49        }
50    });
51
52    let mut stdout = tokio::io::stdout();
53    let mut stdin = tokio::io::BufReader::new(tokio::io::stdin()).lines();
54    let mut renderer = Renderer::default();
55    let mut last_prompt = Instant::now();
56    let mut typing = true;
57    let mut quitting = false;
58    let end = loop {
59        tokio::select! {
60            end = &mut running => break end,
61            event = events.recv() => match event {
62                Ok(Event::Frame(frame)) => {
63                    if matches!(*frame, Frame::Prompt { .. }) {
64                        last_prompt = Instant::now();
65                    }
66                    let text = renderer.frame(&frame);
67                    let _ = stdout.write_all(text.as_bytes()).await;
68                    let _ = stdout.flush().await;
69                }
70                Ok(Event::StateChanged(State::Reconnecting)) => {
71                    eprintln!("\n[hydra] the connection dropped; reconnecting");
72                }
73                Ok(Event::ConnectFailed { attempt, delay, detail }) => {
74                    eprintln!(
75                        "[hydra] {detail}; try {} in {:.1}s",
76                        attempt + 1,
77                        delay.as_secs_f64()
78                    );
79                }
80                Ok(Event::Notice(notice)) => {
81                    let text = format!("
82{}
83", notice.lines().join("
84"));
85                    let _ = stdout.write_all(text.as_bytes()).await;
86                    let _ = stdout.flush().await;
87                }
88                Err(RecvError::Lagged(missed)) => {
89                    eprintln!("\n[hydra] the terminal fell behind; {missed} events not shown");
90                }
91                Ok(_) | Err(RecvError::Closed) => {}
92            },
93            line = stdin.next_line(), if typing => match line {
94                Ok(Some(line)) if line.trim() == ";status" => {
95                    let elapsed = u32::try_from(last_prompt.elapsed().as_secs()).unwrap_or(u32::MAX);
96                    let status = match handle.snapshot().await {
97                        Some(snapshot) => crate::status::line(&snapshot.state, elapsed),
98                        None => "[status] the session has ended".to_owned(),
99                    };
100                    let _ = stdout.write_all(format!("\n{status}\n").as_bytes()).await;
101                    let _ = stdout.flush().await;
102                }
103                Ok(Some(line)) => {
104                    let _ = typed.send(line.trim_end().to_owned());
105                }
106                // The terminal closed its input; keep showing the game.
107                Ok(None) | Err(_) => typing = false,
108            },
109            _ = tokio::signal::ctrl_c() => {
110                if quitting {
111                    eprintln!("\n[hydra] dropping the connection");
112                    cancel.cancel();
113                } else {
114                    quitting = true;
115                    eprintln!("\n[hydra] sending quit (Ctrl-C again to drop the connection)");
116                    let handle = handle.clone();
117                    tokio::spawn(async move { handle.quit(QUIT_DEADLINE).await });
118                }
119            }
120        }
121    };
122    sender.abort();
123    // Its Lich asked to stop and waited for, not killed as the process ends.
124    lichs.shutdown().await;
125    match end {
126        Ok(end) => eprintln!("\n[hydra] session ended: {:?}", end.stopped_because),
127        Err(e) => eprintln!("\n[hydra] the session task failed: {e}"),
128    }
129    Ok(())
130}

132/// The character's login and a supervised session for it, not yet running:
133/// its password read, its session log open, the crit tables loaded.
134///
135/// # Errors
136///
137/// A message naming what failed before the session could start: the
138/// arguments, the password, the data folder or the session log.
139pub fn supervised(target: Target) -> Result<(Login, SupervisedSession<LiveConnector>), String> {
140    let login = target.resolve()?;
141    let password = account::password(&login.account).map_err(|e| e.to_string())?;
142    let session = crate::session::supervised_with(&login, password, crate::session::crit_tables())?;
143    Ok((login, session))
144}

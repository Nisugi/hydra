1//! The sorter: a container look shown as its contents by item type.
2//!
3//! A look such as
4//!
5//! ```text
6//! In the mahogany box you see a bright gold ingot, some silver coins, a smooth amber wand, a piece of brown jade, a steel lockpick, a pinch of electrum dust and a tar black tourmaline.
7//! ```
8//!
9//! is shown as
10//!
11//! ```text
12//! In the mahogany box:
13//! valuable (1): a bright gold ingot
14//! other (1): some silver coins
15//! wand (1): a smooth amber wand
16//! gem (3): a pinch of electrum dust, a piece of brown jade, a tar black tourmaline
17//! lockpick (1): a steel lockpick
18//! ```
19//!
20//! or, given a width, each type's label on a line of its own and its items
21//! under it in columns that fit the width.
22//!
23//! The look carries every item as an `<a exist= noun=>` link, so the line is
24//! what is sorted, and each item keeps its link. The look leaves out what
25//! the game says after an item's name (`a pair of heavy pewter earrings with
26//! a cluster of jasper accents`); the `<inv>` lines the game sends before it
27//! have it, and [`Listed`] keeps it for the look that follows.
28//!
29//! - **What is a look:** a line starting `In the ` or `On the ` with
30//!   ` you see `, or `Peering into ` with `you see `, and never one naming
31//!   both an `In the` and an `On the` ([`is_container_look`]). The first
32//!   object link is the container; the rest are the items.
33//! - **Only a whole list is sorted:** after the container ` you see ` and an
34//!   article; between items `, ` or ` and ` and an article; after the last a
35//!   `.` that ends the line. Any other shape is shown as it came: the herb
36//!   kit's look goes on past its list (`The kit contains DOSEs of ...`), and
37//!   sorting it would lose the doses. Text after an item's link that is not
38//!   a sentence stays with the item: `leather journal embossed with a
39//!   peacock feather`.
40//! - **The type** is every type the item's noun and name have
41//!   ([`classify`]), joined by commas, or `other`. Types join in name order.
42//! - **Names:** each item whole, as the `<inv>` lines say it: its article,
43//!   its name, and what follows the name; or, with long names off, its name
44//!   alone and what the look says after it. An item `containing` something
45//!   says it in brackets, `a glass vial (some red liquid)`; a crystalline
46//!   flask is named by what it holds.
47//! - **Order:** types as first seen; within one, items by the last word of
48//!   the whole name; the same item twice is one, with its count.
49

61/// Runs a line may have and still be sorted; past it the line is shown as
62/// it came. The longest look in a month of play named 108 items, 217 runs.
63pub const MAX_SORTED_RUNS: usize = 1024;

65/// Spaces between two columns, and kept free at the width's edge.
66const GUTTER: usize = 8;

68/// The indent of a column's line.
69const INDENT: &str = "    ";

71/// A character's settings file's `sorter` section (`.sorter`, the General
72/// page): whether its looks are sorted, in columns how wide, and with each
73/// thing's long name or its short one.
74#[derive(Clone, Debug, Default, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
75pub struct Sorting {
76    /// Sorted; off when unset.
77    #[serde(default, skip_serializing_if = "Option::is_none")]
78    pub enabled: Option<bool>,
79    /// The width columns fit, in characters; none or 0, a list.
80    #[serde(default, skip_serializing_if = "Option::is_none")]
81    pub width: Option<u16>,
82    /// Each thing named long, its article and what follows its name
83    /// (`a pair of heavy pewter earrings with a cluster of jasper accents`);
84    /// off, short, its name alone (`pair of heavy pewter earrings`). Long
85    /// when unset.
86    #[serde(default, skip_serializing_if = "Option::is_none")]
87    pub long_names: Option<bool>,
88}
89
90impl Sorting {
91    /// The section's name in the settings file.
92    pub const SECTION: &str = "sorter";
93
94    /// How looks are sorted; `None` while off.
95    #[must_use]
96    pub fn sorted(&self) -> Option<Sorted> {
97        self.enabled.unwrap_or(false).then_some(Sorted {
98            layout: match self.width {
99                Some(width) if width > 0 => Layout::Columns(width),
100                _ => Layout::List,
101            },
102            long_names: self.long_names.unwrap_or(true),
103        })
104    }
105}

116/// How a sorted look lays out each type's items.
117#[derive(Clone, Copy, Debug, PartialEq, Eq)]
118pub enum Layout {
119    /// On the type's line, `, ` between.
120    List,
121    /// Under the type's label, in columns that fit this many characters.
122    Columns(u16),
123}
124
125impl Layout {
126    /// The width [`sort`] fits columns to; `None`, a list.
127    #[must_use]
128    pub const fn width(self) -> Option<u16> {
129        match self {
130            Self::List => None,
131            Self::Columns(width) => Some(width),
132        }
133    }
134}

136/// Whether `text`, a story line's plain text, is a container look: a cheap
137/// first test, [`sort`] also requires a whole list.
138#[must_use]
139pub fn is_container_look(text: &str) -> bool {
140    let sees = text.contains(" you see ");
141    let in_form = text.starts_with("In the ") && sees;
142    let on_form = text.starts_with("On the ") && sees;
143    let peering = text.starts_with("Peering into ") && text.contains("you see ");
144    if !(in_form || on_form || peering) {
145        return false;
146    }
147    !(text.contains("In the ") && text.contains("On the "))
148}

150/// Items past this many kept are dropped, all of them: a listing that big
151/// never came with one look.
152const MOST_LISTED: usize = 4096;
153
154/// What the game's `<inv>` lines say around each item's name, by its
155/// `exist` id: `a` before `pair of heavy pewter earrings`, `with a cluster
156/// of jasper accents` after. A look's container is emptied (`<clearContainer>`)
157/// and listed again just before the look, so each emptying starts afresh.
158#[derive(Clone, Debug, Default)]
159pub struct Listed {
160    words: HashMap<String, Around>,
161}
162
163/// The prose either side of one item's name, trimmed; `None` when blank.
164#[derive(Clone, Debug, Default, PartialEq, Eq)]
165struct Around {
166    before: Option<String>,
167    after: Option<String>,
168}
169
170impl Listed {
171    /// Take `frame`'s listing, if it is one.
172    pub fn hear(&mut self, frame: &Frame) {
173        match frame {
174            Frame::ClearContainer { .. } => self.words.clear(),
175            Frame::ContainerItem { content, .. } => {
176                if self.words.len() > MOST_LISTED {
177                    self.words.clear();
178                }
179                let runs = &content.runs;
180                for (index, run) in runs.iter().enumerate() {
181                    let Some(LinkKind::Exist { id, .. }) = run.link.as_ref().map(|link| &link.kind)
182                    else {
183                        continue;
184                    };
185                    let prose = |at: Option<usize>| {
186                        let run = runs.get(at?)?;
187                        let trimmed = run.text.trim();
188                        (run.link.is_none() && !trimmed.is_empty()).then(|| trimmed.to_owned())
189                    };
190                    self.words.insert(
191                        id.clone(),
192                        Around {
193                            before: prose(index.checked_sub(1)),
194                            after: prose(Some(index + 1)),
195                        },
196                    );
197                }
198            }
199            _ => {}
200        }
201    }
202
203    fn around(&self, run: &StyledRun) -> Option<&Around> {
204        match &run.link {
205            Some(RunLink::Object { exist, .. }) => self.words.get(exist),
206            _ => None,
207        }
208    }
209}

211/// One distinct item of a type.
212struct Entry {
213    /// Its name and trailing text, lowercased: what makes two the same.
214    key: String,
215    /// The last word of its name, lowercased: the order within a type.
216    last_word: String,
217    count: usize,
218    /// Its runs as shown, its link kept.
219    runs: Vec<StyledRun>,
220}

222impl Entry {
223    /// Its text as shown, without its count.
224    fn text(&self) -> String {
225        text(&self.runs)
226    }

235/// The lines a container look is shown as when sorted: a header, then each
236/// type's items, as a list or, given `width` in characters, in columns.
237/// Given `listing`, each item is named long, as it says, or as the look
238/// does where it says nothing; without, short, its name and the look's own
239/// text after it. `None` when `line` is not a look, or not a whole list.
240#[must_use]
241pub fn sort(
242    line: &[StyledRun],
243    width: Option<u16>,
244    listing: Option<&Listed>,
245) -> Option<Vec<Vec<StyledRun>>> {
246    if line.len() > MAX_SORTED_RUNS || !is_container_look(&text(line)) {
247        return None;
248    }
249    let (gaps, objects) = cut(line);
250    let (container, items) = objects.split_first()?;
251    if items.is_empty() {
252        return None;
253    }
254    let opening = text(&gaps[1]);
255    if !matches!(strip_article(&opening), " you see " | ", you see ") {
256        return None;
257    }
258    let mut buckets: Vec<Bucket> = Vec::new();
259    for (index, item) in items.iter().enumerate() {
260        // The gap after this item: another item follows it, or the line ends.
261        let after = &gaps[index + 2];
262        let after_text = text(after);
263        let tail_len = if index + 1 < items.len() {
264            tail_before_separator(&after_text)?
265        } else {
266            after_text.trim_end().strip_suffix('.')?.len()
267        };
268        if after_text[..tail_len].contains('.') {
269            return None;
270        }
271        let around = listing.and_then(|listing| listing.around(item.1));
272        let before = listing.and_then(|_| {
273            around
274                .and_then(|around| around.before.clone())
275                .or_else(|| article(&text(&gaps[index + 1])).map(str::to_owned))
276        });
277        let tail = match around.and_then(|around| around.after.as_ref()) {
278            Some(said) => vec![StyledRun::plain(&format!(" {said}"))],
279            None => leading(after, tail_len),
280        };
281        file(&mut buckets, *item, before.as_deref(), tail);
282    }
283    let mut header = gaps[0].clone();
284    header.push(container.1.clone());
285    header.push(StyledRun::plain(":"));
286    let mut lines = vec![header];
287    for bucket in &mut buckets {
288        bucket.entries.sort_by(|a, b| a.last_word.cmp(&b.last_word));
289    }
290    match width {
291        Some(width) => columns(&mut lines, buckets, usize::from(width)),
292        None => listed(&mut lines, buckets),
293    }
294    Some(lines)
295}

297/// The noun of the object `run` links to, if it links to one.
298fn noun(run: &StyledRun) -> Option<&str> {
299    match &run.link {
300        Some(RunLink::Object { noun, .. }) => Some(noun.as_str()),
301        _ => None,
302    }
303}

305/// `runs` cut at their objects: `gaps[k]` is the text before `objects[k]`,
306/// the last gap the text after the last object, so there is one more gap
307/// than objects. One run is one object.
308fn cut(runs: &[StyledRun]) -> (Vec<Vec<StyledRun>>, Vec<(&str, &StyledRun)>) {
309    let mut gaps: Vec<Vec<StyledRun>> = vec![Vec::new()];
310    let mut objects = Vec::new();
311    for run in runs {
312        if let Some(noun) = noun(run) {
313            objects.push((noun, run));
314            gaps.push(Vec::new());
315        } else if let Some(gap) = gaps.last_mut() {
316            gap.push(run.clone());
317        }
318    }
319    (gaps, objects)
320}

322/// File one item, with its article and its own trailing text, under its
323/// type.
324fn file(
325    buckets: &mut Vec<Bucket>,
326    (noun, run): (&str, &StyledRun),
327    before: Option<&str>,
328    tail: Vec<StyledRun>,
329) {
330    let name = run.text.trim();
331    let types = classify(noun, name).types;
332    let category = if types.is_empty() {
333        OTHER.to_owned()
334    } else {
335        types.into_iter().collect::<Vec<_>>().join(",")
336    };
337    let runs = named(before, run, tail);
338    let whole = text(&runs);
339    let key = whole.to_lowercase();
340    let index = if let Some(found) = buckets.iter().position(|b| b.category == category) {
341        found
342    } else {
343        buckets.push(Bucket {
344            category,
345            entries: Vec::new(),
346        });
347        buckets.len() - 1
348    };
349    let entries = &mut buckets[index].entries;
350    if let Some(entry) = entries.iter_mut().find(|entry| entry.key == key) {
351        entry.count += 1;
352        return;
353    }
354    entries.push(Entry {
355        key,
356        last_word: whole
357            .trim_end()
358            .rsplit(' ')
359            .next()
360            .unwrap_or_default()
361            .to_lowercase(),
362        count: 1,
363        runs,
364    });
365}

367/// An item's runs as shown: its article, its link, then its own text, with
368/// what it holds in brackets, or, for a crystalline flask, what it holds as
369/// its name and no article.
370fn named(before: Option<&str>, run: &StyledRun, mut tail: Vec<StyledRun>) -> Vec<StyledRun> {
371    const FLASK: &str = "crystalline flask";
372    let mut link = run.clone();
373    let mut flask = false;
374    let tail_text = text(&tail);
375    if let Some(held) = tail_text.strip_prefix(" containing ") {
376        if link.text.trim_end().ends_with(FLASK) {
377            held.clone_into(&mut link.text);
378            tail.clear();
379            flask = true;
380        } else {
381            tail = vec![StyledRun::plain(&format!(" ({held})"))];
382        }
383    } else if let Some((name, held)) = link.text.split_once(" containing ") {
384        flask = name.ends_with(FLASK);
385        link.text = if flask {
386            held.to_owned()
387        } else {
388            format!("{name} ({held})")
389        };
390    }
391    let mut runs = Vec::new();
392    if let Some(before) = before.filter(|_| !flask) {
393        runs.push(StyledRun::plain(&format!("{before} ")));
394    }
395    runs.push(link);
396    runs.extend(tail);
397    runs
398}

400/// The bold label of a type: `gem (3):`.
401fn label(bucket: &Bucket) -> StyledRun {
402    let total: usize = bucket.entries.iter().map(|entry| entry.count).sum();
403    StyledRun {
404        text: format!("{} ({total}):", bucket.category),
405        bold: true,
406        ..StyledRun::default()
407    }

410/// An entry's count, after a duplicate: ` (2)`.
411fn counted(entry: &Entry) -> String {
412    if entry.count > 1 {
413        format!(" ({})", entry.count)
414    } else {
415        String::new()
416    }
417}

419/// Each type on a line: its label, then its items, `, ` between, `.` after.
420fn listed(lines: &mut Vec<Vec<StyledRun>>, buckets: Vec<Bucket>) {
421    for bucket in buckets {
422        let mut line = vec![label(&bucket), StyledRun::plain(" ")];
423        for (index, entry) in bucket.entries.iter().enumerate() {
424            if index > 0 {
425                line.push(StyledRun::plain(", "));
426            }
427            line.extend(entry.runs.iter().cloned());
428            let count = counted(entry);
429            if !count.is_empty() {
430                line.push(StyledRun::plain(&count));
431            }
432        }
433        line.push(StyledRun::plain("."));
434        lines.push(line);
435    }
436}

438/// Each type's label on a line, its items under it in as many columns as
439/// fit `width` characters, down each column first, then a blank line.
440fn columns(lines: &mut Vec<Vec<StyledRun>>, buckets: Vec<Bucket>, width: usize) {
441    for bucket in buckets {
442        lines.push(vec![label(&bucket)]);
443        let names: Vec<usize> = bucket
444            .entries
445            .iter()
446            .map(|entry| entry.text().chars().count())
447            .collect();
448        let count = column_count(&names, width);
449        let rows = bucket.entries.len().div_ceil(count);
450        let columns: Vec<&[Entry]> = bucket.entries.chunks(rows.max(1)).collect();
451        let widths: Vec<usize> = columns
452            .iter()
453            .map(|column| {
454                column
455                    .iter()
456                    .map(|entry| entry.text().chars().count())
457                    .max()
458                    .unwrap_or_default()
459            })
460            .collect();
461        for row in 0..rows {
462            let mut line = vec![mono(INDENT.to_owned())];
463            let cells: Vec<(&Entry, usize)> = columns
464                .iter()
465                .zip(&widths)
466                .filter_map(|(column, width)| column.get(row).map(|entry| (entry, *width)))
467                .collect();
468            for (at, (entry, width)) in cells.iter().enumerate() {
469                line.extend(entry.runs.iter().cloned().map(|mut run| {
470                    run.monospace = true;
471                    run
472                }));
473                let count = counted(entry);
474                let shown = entry.text().chars().count() + count.chars().count();
475                let pad = if at + 1 < cells.len() {
476                    (width + GUTTER).saturating_sub(shown)
477                } else {
478                    0
479                };
480                let after = format!("{count}{}", " ".repeat(pad));
481                if !after.is_empty() {
482                    line.push(mono(after));
483                }
484            }
485            lines.push(line);
486        }
487        lines.push(vec![StyledRun::plain("")]);
488    }
489}

491/// How many columns of `names`, each this many characters, fit `width`:
492/// the most whose widest names and gutters stay inside it, at least one.
493fn column_count(names: &[usize], width: usize) -> usize {
494    let mut count = 1;
495    loop {
496        let per = names.len().div_ceil(count).max(1);
497        let total: usize = names
498            .chunks(per)
499            .map(|column| column.iter().copied().max().unwrap_or_default())
500            .sum::<usize>()
501            + (count - 1) * GUTTER;
502        if total > width.saturating_sub(GUTTER) {
503            count -= 1;
504            break;
505        }
506        if count >= names.len() {
507            break;
508        }
509        count += 1;
510    }
511    count.max(1)
512}

527/// `gap` less a closing article: each item comes after `a `, `an ` or
528/// `some `, or with none when the link's text carries its own (`some
529/// silver coins`).
530fn strip_article(gap: &str) -> &str {
531    for article in ["a ", "an ", "some "] {
532        if let Some(rest) = gap.strip_suffix(article)
533            && (rest.is_empty() || rest.ends_with(' '))
534        {
535            return rest;
536        }
537    }
538    gap
539}
540
541/// The article closing `gap`, the text before an item: `a`, `an` or `some`.
542fn article(gap: &str) -> Option<&str> {
543    let rest = strip_article(gap);
544    (rest.len() < gap.len()).then(|| gap[rest.len()..].trim_end())
545}
546
547/// How much of the gap between two items is the first one's own text: the
548/// gap must end as a list's does, `, ` or ` and ` then an article.
549fn tail_before_separator(gap: &str) -> Option<usize> {
550    let rest = strip_article(gap);
551    rest.strip_suffix(", ")
552        .or_else(|| rest.strip_suffix(" and "))
553        .map(str::len)
554}

556/// The first `len` bytes of `runs`, styles kept; `len` is a character
557/// boundary of their text together.
558fn leading(runs: &[StyledRun], len: usize) -> Vec<StyledRun> {
559    let mut left = len;
560    let mut out = Vec::new();
561    for run in runs {
562        if left == 0 {
563            break;
564        }
565        let take = left.min(run.text.len());
566        out.push(StyledRun {
567            text: run.text[..take].to_owned(),
568            ..run.clone()
569        });
570        left -= take;
571    }
572    out
573}

579    #[test]
580    fn columns_fit_the_width() {
581        // Ten names of ten: four fit 80 (4 x 10 + 3 x 8 = 64 <= 72); five
582        // do not (5 x 10 + 4 x 8 = 82).
583        assert_eq!(column_count(&[10; 10], 80), 4);
584        assert_eq!(column_count(&[10; 10], 10), 1);
585        assert_eq!(column_count(&[5], 80), 1);
586    }

588    #[test]
589    fn what_an_item_holds_is_bracketed_and_a_flask_named_by_it() {
590        let link = |text: &str| StyledRun {
591            text: text.to_owned(),
592            link: Some(RunLink::Object {
593                exist: "1".to_owned(),
594                noun: "vial".to_owned(),
595                coord: None,
596            }),
597            ..StyledRun::default()
598        };
599        let shown = |runs: Vec<StyledRun>| text(&runs);
600        let tail = vec![StyledRun::plain(" containing some red liquid")];
601        assert_eq!(
602            shown(named(Some("a"), &link("glass vial"), tail.clone())),
603            "a glass vial (some red liquid)"
604        );
605        assert_eq!(
606            shown(named(Some("a"), &link("crystalline flask"), tail)),
607            "some red liquid"
608        );
609        assert_eq!(
610            shown(named(None, &link("glass vial containing ale"), Vec::new())),
611            "glass vial (ale)"
612        );
613    }

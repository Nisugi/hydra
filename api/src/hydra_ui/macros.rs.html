1//! Macros: the player's chains, each with the logic, a key, an alias word
2//! and its states. A macro is the core a keybind, an alias word and a
3//! hotbar button are three doors into: the key for the hands, the word for
4//! the command line and Lich, the button for the eyes.
5//!
6//! **The file is `macros.toml`**, in the data folder for every character and
7//! beside a character's layout for that one; a character's macro replaces
8//! every character's macro of the same name, whole.
9//!
10//! ```toml
11//! [[macros]]
12//! name = "barrage"
13//! commands = "stance off, rt, weapon barrage" # the chain language (hydra_session::steps)
14//! key = "F5"                                 # a key that runs it, as the keybinds write one
15//! set = 1                                    # in this macro set; none, every set
16//! alias = "wb"                               # a word that runs it from the command line
17//! when = { type = "vital", vital = "stamina", cmp = ">=", value = 60, unit = "absolute" }
18//!                                            # runs only when this holds; else unavailable
19//!
20//! [[macros.states]]
21//! when = { type = "vital", vital = "stamina", cmp = "<", value = 60, unit = "absolute" }
22//! unavailable = true                         # a press runs nothing: a reserve kept back
23//!
24//! [[macros.states]]
25//! when = { type = "indicator", id = "stunned" }
26//! chain = "stance def"                       # runs this instead
27//! ```
28//!
29//! **`when` is the macro's condition**: not holding, the macro is
30//! unavailable and a press runs nothing. **Its states are more logic**, the
31//! first that holds deciding: unavailable, or another chain. How a state looks on a button
32//! is the hotbar's business (`crate::hotbar`).

45/// The file.
46#[derive(Clone, Debug, Default, PartialEq, Deserialize, Serialize)]
47pub struct MacroFile {
48    /// Every macro, in the file's order.
49    #[serde(default, skip_serializing_if = "Vec::is_empty")]
50    pub macros: Vec<MacroDef>,
51}

53/// One macro.
54#[derive(Clone, Debug, Default, PartialEq, Deserialize, Serialize)]
55pub struct MacroDef {
56    /// Its name, one word: what a button, a key and `.macro` run it by.
57    pub name: String,
58    /// Its commands: the chain, as the file's `commands` (or `chain`).
59    #[serde(rename = "commands", alias = "chain")]
60    pub chain: String,
61    /// A key that runs it, written as the keybinds file writes a key.
62    #[serde(default, skip_serializing_if = "Option::is_none")]
63    pub key: Option<String>,
64    /// The macro set its key is in, 0 to 9; none, every set.
65    #[serde(default, skip_serializing_if = "Option::is_none")]
66    pub set: Option<u8>,
67    /// A word that runs it from the command line, kept as an alias.
68    #[serde(default, skip_serializing_if = "Option::is_none")]
69    pub alias: Option<String>,
70    /// The condition it runs under; none, always. Not holding, the macro
71    /// is unavailable: a button greys, and a press runs nothing.
72    #[serde(default, skip_serializing_if = "Option::is_none")]
73    pub when: Option<Condition>,
74    /// Seconds that must pass after it ran before it runs again; none,
75    /// as often as pressed.
76    #[serde(default, skip_serializing_if = "Option::is_none")]
77    pub repeat_after: Option<f32>,
78    /// Runs once in a room: pressed again there, it runs nothing until
79    /// the character has moved.
80    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
81    pub once_per_room: bool,
82    /// What a hover says.
83    #[serde(default, skip_serializing_if = "Option::is_none")]
84    pub tooltip: Option<String>,
85    /// What it counts down, when a button shows it.
86    #[serde(default, skip_serializing_if = "Option::is_none")]
87    pub countdown: Option<Countdown>,
88    /// Its states, in order: the first whose condition holds decides.
89    #[serde(default, skip_serializing_if = "Vec::is_empty")]
90    pub states: Vec<MacroState>,
91    /// Anything a later Hydra keeps here, kept as it is.
92    #[serde(flatten)]
93    pub rest: BTreeMap<String, toml::Value>,
94}

96/// One state of a macro.
97#[derive(Clone, Debug, PartialEq, Deserialize, Serialize)]
98pub struct MacroState {
99    /// When it holds.
100    pub when: Condition,
101    /// A press runs nothing: a reserve kept back, a thing not to do now.
102    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
103    pub unavailable: bool,
104    /// Another chain run instead of the macro's.
105    #[serde(default, skip_serializing_if = "Option::is_none")]
106    pub chain: Option<String>,
107    /// What it counts down instead, when a button shows it.
108    #[serde(default, skip_serializing_if = "Option::is_none")]
109    pub countdown: Option<Countdown>,
110}

112impl MacroFile {
113    /// Read from its text.
114    ///
115    /// # Errors
116    ///
117    /// Why the text is not a macros file.
118    pub fn parse(text: &str) -> Result<Self, String> {
119        toml::from_str(text).map_err(|why| why.to_string())
120    }
121
122    /// As the file writes it.
123    #[must_use]
124    pub fn to_toml(&self) -> String {
125        toml::to_string(self).unwrap_or_default()
126    }
127
128    /// Read from `path`; an empty file when there is none.
129    ///
130    /// # Errors
131    ///
132    /// The file is there and does not read.
133    pub fn load(path: &Path) -> Result<Self, String> {
134        let name = path.file_name().map_or_else(
135            || FILE.to_owned(),
136            |name| name.to_string_lossy().into_owned(),
137        );
138        match std::fs::read_to_string(path) {
139            Ok(text) => Self::parse(&text).map_err(|why| format!("{name}: {why}")),
140            Err(why) if why.kind() == std::io::ErrorKind::NotFound => Ok(Self::default()),
141            Err(why) => Err(format!("{name} cannot be read: {why}")),
142        }
143    }
144
145    /// Everyone's with `character`'s over them: a macro of the same name
146    /// replaced whole.
147    #[must_use]
148    pub fn merged(mut self, character: Self) -> Self {
149        for made in character.macros {
150            match self
151                .macros
152                .iter_mut()
153                .find(|had| had.name.eq_ignore_ascii_case(&made.name))
154            {
155                Some(had) => *had = made,
156                None => self.macros.push(made),
157            }
158        }
159        self
160    }
161
162    /// The macro called `name`, without regard to case.
163    #[must_use]
164    pub fn named(&self, name: &str) -> Option<&MacroDef> {
165        let name = name.trim();
166        self.macros
167            .iter()
168            .find(|made| made.name.eq_ignore_ascii_case(name))
169    }
170}

172/// When each macro last ran, and where: what `repeat_after` and
173/// `once_per_room` are held against.
174#[derive(Debug, Default)]
175pub struct Runs {
176    last: BTreeMap<String, (std::time::Instant, Option<String>)>,
177}
178
179impl Runs {
180    /// Whether `made` may run at `now` in the room `room` (the game's id),
181    /// held against when it last ran; why not, for the player.
182    ///
183    /// # Errors
184    ///
185    /// It ran too recently, or in this room already.
186    pub fn allows(
187        &self,
188        made: &MacroDef,
189        now: std::time::Instant,
190        room: Option<&str>,
191    ) -> Result<(), String> {
192        let Some((at, there)) = self.last.get(&made.name.to_lowercase()) else {
193            return Ok(());
194        };
195        if made.once_per_room && there.as_deref() == room && room.is_some() {
196            return Err(format!(
197                "{} has run in this room already: it runs once a room.",
198                made.name
199            ));
200        }
201        if let Some(after) = made.repeat_after {
202            let since = now.saturating_duration_since(*at).as_secs_f32();
203            if since < after {
204                return Err(format!(
205                    "{} ran {since:.0} s ago: it runs again after {after:.0} s.",
206                    made.name
207                ));
208            }
209        }
210        Ok(())
211    }
212
213    /// `name` ran at `now` in `room`.
214    pub fn note(&mut self, name: &str, now: std::time::Instant, room: Option<&str>) {
215        self.last
216            .insert(name.to_lowercase(), (now, room.map(str::to_owned)));
217    }
218}

220/// How `button` looks in `world`, with the macro it names, if one, read
221/// from `macros`: dim when the macro is unavailable now, and counting the
222/// macro's countdown down when the button has none of its own.
223#[must_use]
224pub fn button_look(button: &Button, world: &World<'_>, macros: &MacroFile) -> Look {
225    let mut look = button.look(world);
226    let Some(made) = button
227        .macro_name
228        .as_deref()
229        .map(str::trim)
230        .filter(|name| !name.is_empty())
231        .and_then(|name| macros.named(name))
232    else {
233        return look;
234    };
235    if made.chain_in(world).is_err() {
236        look.dim = true;
237    }
238    if look.countdown.is_none() {
239        look.countdown = made.countdown_in(world);
240    }
241    look
242}

244impl MacroDef {
245    /// The state that holds in `world`, if any: the first.
246    #[must_use]
247    pub fn state_in(&self, world: &World<'_>) -> Option<&MacroState> {
248        self.states.iter().find(|state| state.when.holds(world))
249    }
250
251    /// The chain a press runs in `world`: the macro's, or what its state
252    /// says instead.
253    ///
254    /// # Errors
255    ///
256    /// The state that holds makes it unavailable: why, for the player.
257    pub fn chain_in(&self, world: &World<'_>) -> Result<&str, String> {
258        if !self.available_in(world) {
259            return Err(format!(
260                "{} runs only when its condition holds, and it does not now.",
261                self.name
262            ));
263        }
264        match self.state_in(world) {
265            Some(state) if state.unavailable => Err(format!("{} is not available now.", self.name)),
266            Some(state) => Ok(state.chain.as_deref().unwrap_or(&self.chain)),
267            None => Ok(&self.chain),
268        }
269    }
270
271    /// Whether its condition holds in `world`: with none, it does.
272    #[must_use]
273    pub fn available_in(&self, world: &World<'_>) -> bool {
274        self.when.as_ref().is_none_or(|when| when.holds(world))
275    }
276
277    /// What it counts down in `world`: its state's, else its own.
278    #[must_use]
279    pub fn countdown_in(&self, world: &World<'_>) -> Option<u32> {
280        self.state_in(world)
281            .and_then(|state| state.countdown.as_ref())
282            .or(self.countdown.as_ref())
283            .and_then(|countdown| countdown.left(world))
284    }
285}

308    /// The file reads, a character's macro replaces everyone's of the name,
309    /// and a name is found without regard to case.
310    #[test]
311    fn the_file_reads_and_a_characters_macro_replaces_everyones() {
312        let every = MacroFile::parse(FILE_TEXT).expect("reads");
313        let barrage = every.named("BARRAGE").expect("named");
314        assert_eq!(barrage.key.as_deref(), Some("F5"));
315        assert_eq!(barrage.set, Some(1));
316        assert_eq!(barrage.states.len(), 2);
317        let own = MacroFile::parse("[[macros]]\nname = \"Barrage\"\nchain = \"attack\"\n")
318            .expect("reads");
319        let merged = every.merged(own);
320        assert_eq!(merged.macros.len(), 1);
321        assert_eq!(
322            merged.named("barrage").map(|m| m.chain.as_str()),
323            Some("attack")
324        );
325        assert!(MacroFile::parse("[[macros]]\nname = 1\n").is_err());
326        let written = merged.to_toml();
327        assert!(written.contains("name = \"Barrage\""), "{written}");
328        assert!(written.contains("commands = \"attack\""), "{written}");
329    }

334    fn a_macros_condition_makes_it_unavailable() {
335        use crate::condition::Flags;
336        let file = MacroFile::parse(
337            "[[macros]]\nname = \"look\"\ncommands = \"look\"\nwhen = { type = \"rt_active\" }\n",

343            flags: &flags,
344            statuses: &flags,
345            unix: 0,
346        };
347        let look = file.named("look").expect("named");
348        assert!(!look.available_in(&world), "no roundtime: it does not hold");
349        assert!(look.chain_in(&world).is_err());

352            chain: "look".to_owned(),

392    /// A macro that runs once a room runs nothing pressed again there, and
393    /// runs in the next room; one with a wait between runs is held for it.
394    #[test]
395    fn a_macros_runs_are_throttled() {
396        use std::time::{Duration, Instant};
397        let once = MacroDef {
398            name: "Pick".to_owned(),
399            chain: "search".to_owned(),
400            once_per_room: true,
401            ..MacroDef::default()
402        };
403        let mut runs = Runs::default();
404        let now = Instant::now();
405        assert!(runs.allows(&once, now, Some("100")).is_ok());
406        runs.note("pick", now, Some("100"));
407        assert!(
408            runs.allows(&once, now, Some("100")).is_err(),
409            "the same room"
410        );
411        assert!(runs.allows(&once, now, Some("101")).is_ok(), "moved");
412        let spaced = MacroDef {
413            name: "shout".to_owned(),
414            chain: "cman shout".to_owned(),
415            repeat_after: Some(10.0),
416            ..MacroDef::default()
417        };
418        runs.note("shout", now, None);
419        assert!(
420            runs.allows(&spaced, now + Duration::from_secs(3), None)
421                .is_err()
422        );
423        assert!(
424            runs.allows(&spaced, now + Duration::from_secs(10), None)
425                .is_ok()
426        );
427    }

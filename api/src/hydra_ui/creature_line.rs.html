1//! A creature's own lines, marked by what they say: its death, a spell it is
2//! preparing, a special about to land, or its rising or shaking off a stun.
3//! A line that links a creature is matched against that creature's messages
4//! in the bestiary (`hydra_model::state::creature_message`), and one that is
5//! one of them takes the kind's preset, which the theme colours as it does
6//! the game's own presets: the player writes no highlight for them.
7//!
8//! The creature is found by its link's noun, every bestiary creature with
9//! that noun tried in turn, and only its handful of messages of the kinds
10//! here compared, as text with placeholders rather than as patterns: a line
11//! with no creature in it costs nothing, and one with costs microseconds.
12//!
13//! A creature the bestiary lacks, or a wording it does not record --
14//! a boon's adjective before the name, say -- leaves its line unmarked.
15
16use hydra_model::state::creature::{MessageKind, by_noun};
17use hydra_model::state::creature_message::classify;
18
19use crate::run::{RunLink, StyledRun};
20
21/// The preset of a creature's death.
22pub const DEATH: &str = "creature_death";
23/// The preset of a creature preparing a spell.
24pub const PREP: &str = "creature_prep";
25/// The preset of a creature about to land a special.
26pub const TRIGGER: &str = "creature_trigger";
27/// The preset of a creature standing up or shaking off a stun.
28pub const RECOVER: &str = "creature_recover";
29
30/// The kinds of message looked for, first match winning.
31const KINDS: [MessageKind; 5] = [
32    MessageKind::Death,
33    MessageKind::SpellPrep,
34    MessageKind::Trigger,
35    MessageKind::Stand,
36    MessageKind::StunBreak,
37];
38
39/// The preset `line` takes, if it is a death, a spell prep, a special or a
40/// recovery of a creature it links.
41#[must_use]
42pub fn preset(line: &[StyledRun]) -> Option<&'static str> {
43    let mut nouns: Vec<&str> = Vec::new();
44    for run in line {
45        if let Some(RunLink::Object { noun, .. }) = &run.link
46            && !nouns.iter().any(|seen| seen.eq_ignore_ascii_case(noun))
47        {
48            nouns.push(noun);
49        }
50    }
51    if nouns.is_empty() {
52        return None;
53    }
54    let text: String = line.iter().map(|run| run.text.as_str()).collect();
55    let found = nouns
56        .into_iter()
57        .flat_map(by_noun)
58        .find_map(|creature| classify(creature, &text, &KINDS))?;
59    Some(match found.kind {
60        MessageKind::Death => DEATH,
61        MessageKind::SpellPrep => PREP,
62        MessageKind::Trigger => TRIGGER,
63        _ => RECOVER,
64    })
65}

67/// Mark `line` with its preset, if it is a creature's line: every run of it
68/// without a preset of its own takes it. A link and a creature's name keep
69/// their own colours over it ([`crate::run_token`]).
70pub fn mark(line: &mut [StyledRun]) {
71    if let Some(preset) = preset(line) {
72        for run in line.iter_mut().filter(|run| run.preset.is_none()) {
73            run.preset = Some(preset.to_owned());
74        }
75    }
76}

78#[cfg(test)]
79mod tests {
80    use super::*;

82    /// A line as the game sends it: `before`, the creature's name linked by
83    /// `noun` and bold, then `after`.
84    fn line(before: &str, name: &str, noun: &str, after: &str) -> Vec<StyledRun> {
85        vec![
86            StyledRun::plain(before),
87            StyledRun {
88                bold: true,
89                link: Some(RunLink::Object {
90                    exist: "1234".to_owned(),
91                    noun: noun.to_owned(),
92                    coord: None,
93                }),
94                ..StyledRun::plain(name)
95            },
96            StyledRun::plain(after),
97        ]
98    }

100    /// Each kind of a creature's own line, as the bestiary records it, takes
101    /// its preset.
102    #[test]
103    fn each_kind_of_creature_line_takes_its_preset() {
104        for (want, line) in [
105            (
106                DEATH,
107                line(
108                    "A ",
109                    "grim gigas skald",
110                    "skald",
111                    " slumps, his eyes dull and unfocused.",
112                ),
113            ),
114            (
115                DEATH,
116                line(
117                    "The ",
118                    "kobold",
119                    "kobold",
120                    " crumples to a heap on the ground and dies.",
121                ),
122            ),
123            (
124                PREP,
125                line(
126                    "An ",
127                    "Agresh troll chieftain",
128                    "chieftain",
129                    " mutters, \"Srlarloror'rt srar 'mrosrdnragh srar 'r'rar s'r'vr'r'rawrd!\"",
130                ),
131            ),
132            (
133                TRIGGER,
134                line(
135                    "An ",
136                    "ashen patrician vampire",
137                    "vampire",
138                    " flicks a finger impatiently at you!",
139                ),
140            ),
141            (
142                RECOVER,
143                line("An ", "Agresh bear", "bear", " stands up and growls!"),
144            ),
145            (
146                RECOVER,
147                line("An ", "arctic titan", "titan", " shakes off the stun!"),
148            ),
149        ] {
150            assert_eq!(preset(&line), Some(want), "{line:?}");
151        }
152    }

154    /// A line that names a creature but is none of its own lines, a line
155    /// with no creature in it, and a creature the bestiary lacks are left
156    /// as they are.
157    #[test]
158    fn other_lines_are_left_unmarked() {
159        let mut lines = [
160            line(
161                "You notice a number of the briar's nettles scrape into a ",
162                "grim gigas skald",
163                "skald",
164                "'s skin.",
165            ),
166            vec![StyledRun::plain("Roundtime: 4 sec.")],
167            line(
168                "A ",
169                "glimmering moth",
170                "moth",
171                " slumps, its eyes dull and unfocused.",
172            ),
173        ];
174        for line in &mut lines {
175            assert_eq!(preset(line), None, "{line:?}");
176            let before = line.clone();
177            mark(line);
178            assert_eq!(*line, before);
179        }
180    }

182    /// Marked, the plain words take the preset; a run with a preset of its
183    /// own keeps it; the creature's name keeps its link and bold, so it is
184    /// still drawn as a creature.
185    #[test]
186    fn marking_leaves_a_runs_own_preset_and_the_name_its_colour() {
187        let mut marked = line(
188            "A ",
189            "grim gigas skald",
190            "skald",
191            " slumps, his eyes dull and unfocused.",
192        );
193        marked[2].preset = Some("speech".to_owned());
194        mark(&mut marked);
195        let presets: Vec<_> = marked.iter().map(|run| run.preset.as_deref()).collect();
196        assert_eq!(presets, [Some(DEATH), Some(DEATH), Some("speech")]);
197        assert_eq!(
198            crate::run_token(&marked[1]),
199            Some(crate::theme::Token::Creature)
200        );
201        assert_eq!(
202            crate::run_token(&marked[0]),
203            Some(crate::theme::Token::CreatureDeath)
204        );
205    }

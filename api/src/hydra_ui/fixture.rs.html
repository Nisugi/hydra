14fn bar(id: &str, percent: u32, current: i32, max: i32) -> Frame {
15    Frame::ProgressBar(ProgressBar {
16        id: id.to_owned(),
17        dialog: Some("minivitals".to_owned()),
18        percent,
19        text: format!("{id} {current}/{max}"),
20        amount: Some(Amount { current, max }),
21        time_remaining_secs: None,
22        attrs: Vec::new(),
23    })
24}

26fn item(noun: &str, text: &str) -> RoomItem {
27    RoomItem {
28        id: format!("-{}", noun.len()),
29        noun: noun.to_owned(),
30        text: text.to_owned(),
31        before: None,
32        after: None,
33        status: None,
34    }

41    let mut state = GameState::default();
42    state.apply(&bar("health", 87, 348, 400));

106/// `wire` heard by `story`, as the session publishes it: each frame the
107/// parser makes of it, and each prompt also as a prompt.
108pub fn heard(story: &mut Story, wire: &str) {
109    let mut parser = Parser::new();
110    for frame in parser.push_bytes(wire.as_bytes()) {
111        if let Frame::Prompt { text, .. } = &frame {
112            story.hear(&Event::Prompt(text.clone()));
113        }
114        story.hear(&Event::Frame(Box::new(frame)));
115    }
116}
117
118/// `text`, a line said on `stream` (`""` for the main window), heard by
119/// `story`.
120pub fn said(story: &mut Story, stream: &str, text: &str) {
121    let wire = if stream.is_empty() {
122        format!("{text}\n")
123    } else {
124        format!("<pushStream id=\"{stream}\"/>{text}\n<popStream/>")
125    };
126    heard(story, &wire);
127}
128
129/// A line of `runs` on `stream`, heard by `story`: a text frame for each
130/// run, the last ending the line, as the parser makes them.
131pub fn line(story: &mut Story, stream: &str, runs: &[Run]) {
132    let last = runs.len().saturating_sub(1);
133    for (at, run) in runs.iter().enumerate() {
134        story.hear(&Event::Frame(Box::new(Frame::Text(TextFrame {
135            content: run.text.clone(),
136            stream: stream.to_owned(),
137            style: run.style.clone(),
138            link: run.link.clone(),
139            inner_link: run.inner_link.clone(),
140            ends_line: at == last,
141        }))));
142    }
143}

145/// `line` sent on the character, as the player typed it.
146pub fn typed(story: &mut Story, line: &str) {
147    story.hear(&Event::Sent {
148        line: line.to_owned(),
149        origin: Origin::Player,
150    });
151}

156/// Ashryn's story: a swing, its prompt, `look` typed after it, and a word
157/// from Hydra.
158#[must_use]
159pub fn story() -> Story {
160    let mut story = Story::default();
161    said(&mut story, "", "You swing a steel broadsword at a kobold!");
162    story.hear(&Event::Prompt(">".to_owned()));
163    typed(&mut story, "look");
164    story.tell(TOLD);

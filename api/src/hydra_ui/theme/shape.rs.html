1//! A theme's shape: the corners of each kind of thing, the edges, how roomy
2//! the controls are, the scrollbar, the shadows, the panes' spacing, the
3//! bars' thickness, the title bars' height, and how a button is drawn.
4//!
5//! **Every default is Hydra's plain look**, so a theme that says nothing of
6//! its shape changes nothing: egui's own corners for controls
7//! (2) and windows (6), Hydra's 3 for its bars and panes, cells edge to
8//! edge, bars filling their panes, egui's title bars, filled buttons.
9
10use serde::{Deserialize, Serialize};

12/// How roomy the controls are: the gaps between them and the padding in
13/// them, as a multiple of egui's own.
14#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Deserialize, Serialize)]
15#[serde(rename_all = "snake_case")]
16pub enum Density {
17    /// Six tenths of egui's gaps.
18    Tight,
19    /// egui's own.
20    #[default]
21    Normal,
22    /// Half again egui's gaps.
23    Roomy,
24}
25
26impl Density {
27    /// Every density, in order.
28    pub const ALL: [Density; 3] = [Density::Tight, Density::Normal, Density::Roomy];
29
30    /// What egui's gaps and padding are multiplied by.
31    #[must_use]
32    pub const fn factor(self) -> f32 {
33        match self {
34            Density::Tight => 0.6,
35            Density::Normal => 1.0,
36            Density::Roomy => 1.5,
37        }
38    }
39
40    /// Its name, as a theme file writes it.
41    #[must_use]
42    pub const fn name(self) -> &'static str {
43        match self {
44            Density::Tight => "tight",
45            Density::Normal => "normal",
46            Density::Roomy => "roomy",
47        }
48    }
49}

107/// The corner radius of each kind of thing, in points.
108#[derive(Clone, Copy, Debug, PartialEq, Eq)]
109pub struct Corners {
110    /// Buttons, fields, checkboxes, dropdowns, tabs: egui's controls.
111    pub controls: u8,
112    /// Windows, menus and popups.
113    pub windows: u8,
114    /// The vitals and every other bar.
115    pub bars: u8,
116    /// What Hydra draws in a pane: an indicator, a compass cell, a find's
117    /// hit, the window in use's outline, a dragged widget.
118    pub panes: u8,
119}
120
121impl Default for Corners {
122    fn default() -> Self {
123        Self {
124            controls: 2,
125            windows: 6,
126            bars: 3,
127            panes: 3,
128        }
129    }
130}

132/// A theme's shape, whole.
133#[derive(Clone, Copy, Debug, PartialEq)]
134pub struct Shape {
135    /// Each kind of thing's corners.
136    pub corners: Corners,
137    /// An edge's width, in points.
138    pub stroke: f32,
139    /// How roomy the controls are.
140    pub density: Density,
141    /// The scrollbar's width, in points.
142    pub scrollbar: f32,
143    /// Whether windows and menus cast shadows.
144    pub shadows: bool,
145    /// The gap between the cells of a custom window, in points.
146    pub pane_gap: f32,
147    /// The room inside a pane round its widget, in points.
148    pub pane_padding: f32,
149    /// The most a bar is thick, in points, centred in its pane; 0 fills
150    /// the pane.
151    pub bar_thickness: f32,
152    /// A window's title bar's and a cell's tab strip's height, in points;
153    /// 0 is egui's title bar and Hydra's strip.
154    pub title_height: f32,
155    /// Where a window's title sits in its title bar.
156    pub title_align: TitleAlign,
157    /// How a button is drawn at rest.
158    pub buttons: Buttons,
159}

205/// The shape as a file writes it: a part left out is the base's.
206#[derive(Clone, Debug, Default, PartialEq, Deserialize, Serialize)]
207#[serde(deny_unknown_fields)]
208pub struct ShapeFile {
209    /// The corners, each kind by name.
210    #[serde(default, skip_serializing_if = "CornersFile::is_empty")]
211    pub corners: CornersFile,
212    /// The edge width.
213    #[serde(default, skip_serializing_if = "Option::is_none")]
214    pub stroke: Option<f32>,
215    /// The density, by name.
216    #[serde(default, skip_serializing_if = "Option::is_none")]
217    pub density: Option<Density>,
218    /// The scrollbar's width.
219    #[serde(default, skip_serializing_if = "Option::is_none")]
220    pub scrollbar: Option<f32>,
221    /// Shadows on or off.
222    #[serde(default, skip_serializing_if = "Option::is_none")]
223    pub shadows: Option<bool>,
224    /// The gap between a custom window's cells.
225    #[serde(default, skip_serializing_if = "Option::is_none")]
226    pub pane_gap: Option<f32>,
227    /// The room inside a pane.
228    #[serde(default, skip_serializing_if = "Option::is_none")]
229    pub pane_padding: Option<f32>,
230    /// The most a bar is thick; 0 fills its pane.
231    #[serde(default, skip_serializing_if = "Option::is_none")]
232    pub bar_thickness: Option<f32>,
233    /// The title bars' height; 0 is egui's.
234    #[serde(default, skip_serializing_if = "Option::is_none")]
235    pub title_height: Option<f32>,
236    /// Where a title sits, by name.
237    #[serde(default, skip_serializing_if = "Option::is_none")]
238    pub title_align: Option<TitleAlign>,
239    /// The button style, by name.
240    #[serde(default, skip_serializing_if = "Option::is_none")]
241    pub buttons: Option<Buttons>,
242}
243
244impl ShapeFile {
245    /// Whether it sets nothing.
246    #[must_use]
247    pub fn is_empty(&self) -> bool {
248        *self == Self::default()
249    }
250
251    /// What `shape` says that `base` does not, as a file writes it.
252    #[must_use]
253    pub fn differing(shape: &Shape, base: &Shape) -> Self {
254        let differs = |a: f32, b: f32| ((a - b).abs() > f32::EPSILON).then_some(a);
255        let corner = |a: u8, b: u8| (a != b).then_some(a);
256        let (mine, theirs) = (shape.corners, base.corners);
257        Self {
258            corners: CornersFile {
259                controls: corner(mine.controls, theirs.controls),
260                windows: corner(mine.windows, theirs.windows),
261                bars: corner(mine.bars, theirs.bars),
262                panes: corner(mine.panes, theirs.panes),
263            },
264            stroke: differs(shape.stroke, base.stroke),
265            density: (shape.density != base.density).then_some(shape.density),
266            scrollbar: differs(shape.scrollbar, base.scrollbar),
267            shadows: (shape.shadows != base.shadows).then_some(shape.shadows),
268            pane_gap: differs(shape.pane_gap, base.pane_gap),
269            pane_padding: differs(shape.pane_padding, base.pane_padding),
270            bar_thickness: differs(shape.bar_thickness, base.bar_thickness),
271            title_height: differs(shape.title_height, base.title_height),
272            title_align: (shape.title_align != base.title_align).then_some(shape.title_align),
273            buttons: (shape.buttons != base.buttons).then_some(shape.buttons),
274        }
275    }
276
277    /// `shape` with what this file sets laid over it, each within its
278    /// bounds.
279    #[must_use]
280    pub fn over(&self, shape: Shape) -> Shape {
281        let corner = |set: Option<u8>, base: u8| set.unwrap_or(base).min(32);
282        let corners = &self.corners;
283        Shape {
284            corners: Corners {
285                controls: corner(corners.controls, shape.corners.controls),
286                windows: corner(corners.windows, shape.corners.windows),
287                bars: corner(corners.bars, shape.corners.bars),
288                panes: corner(corners.panes, shape.corners.panes),
289            },
290            stroke: self.stroke.unwrap_or(shape.stroke).clamp(0.0, 8.0),
291            density: self.density.unwrap_or(shape.density),
292            scrollbar: self.scrollbar.unwrap_or(shape.scrollbar).clamp(2.0, 32.0),
293            shadows: self.shadows.unwrap_or(shape.shadows),
294            pane_gap: self.pane_gap.unwrap_or(shape.pane_gap).clamp(0.0, 32.0),
295            pane_padding: self
296                .pane_padding
297                .unwrap_or(shape.pane_padding)
298                .clamp(0.0, 32.0),
299            bar_thickness: self
300                .bar_thickness
301                .unwrap_or(shape.bar_thickness)
302                .clamp(0.0, 200.0),
303            title_height: self
304                .title_height
305                .unwrap_or(shape.title_height)
306                .clamp(0.0, 64.0),
307            title_align: self.title_align.unwrap_or(shape.title_align),
308            buttons: self.buttons.unwrap_or(shape.buttons),
309        }
310    }
311}

314mod tests {
315    use super::*;

317    #[test]
318    fn a_file_sets_what_it_says_and_keeps_the_rest_within_bounds() {
319        let file: ShapeFile = toml::from_str(
320            "density = \"roomy\"\nscrollbar = 1\nbuttons = \"outlined\"\ntitle_align = \"left\"\npane_gap = 99\n[corners]\nbars = 0\nwindows = 40\n",
321        )
322        .expect("reads");
323        let shape = file.over(Shape::default());
324        assert_eq!(shape.corners.bars, 0);
325        assert_eq!(shape.corners.windows, 32, "at most");
326        assert_eq!(shape.corners.controls, 2, "the base's");
327        assert_eq!(shape.density, Density::Roomy);
328        assert_eq!(shape.buttons, Buttons::Outlined);
329        assert_eq!(shape.title_align, TitleAlign::Left);
330        assert!((shape.scrollbar - 2.0).abs() < f32::EPSILON, "at least");
331        assert!((shape.pane_gap - 32.0).abs() < f32::EPSILON, "at most");
332        assert!((shape.stroke - 1.0).abs() < f32::EPSILON, "the base's");
333        assert!(shape.shadows);
334        assert!(toml::from_str::<ShapeFile>("density = \"loose\"\n").is_err());
335        assert!(toml::from_str::<ShapeFile>("bevel = 2\n").is_err());
336        assert!(toml::from_str::<ShapeFile>("[corners]\ntabs = 2\n").is_err());
337        assert!(ShapeFile::default().is_empty());
338    }
339
340    #[test]
341    fn only_what_differs_is_written_and_reads_back() {
342        let base = Shape::default();
343        let mut shape = base;
344        shape.corners.panes = 0;
345        shape.bar_thickness = 14.0;
346        let file = ShapeFile::differing(&shape, &base);
347        assert_eq!(file.corners.panes, Some(0));
348        assert_eq!(file.corners.controls, None);
349        assert_eq!(file.bar_thickness, Some(14.0));
350        assert_eq!(file.stroke, None);
351        let text = toml::to_string(&file).expect("written");
352        let read: ShapeFile = toml::from_str(&text).expect("read");
353        assert_eq!(read.over(base), shape);
354    }

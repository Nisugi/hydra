1//! A theme's type: the UI's font, the story's and the fixed-width one, and
2//! their sizes; the story's line spacing; the fonts tried in turn for a
3//! character a font lacks; and a font or size of its own for a kind of
4//! widget.
5//!
6//! A font is named by a file in the data folder's `fonts` (its stem:
7//! `Iosevka-Regular.ttf` is `Iosevka-Regular`) or by an installed font's
8//! family (`Consolas`); a name neither knows is egui's own. **Every default
9//! is egui's**, 13 points and its own fonts, so a theme that says nothing of
10//! its type changes nothing.
11
12use std::collections::BTreeMap;
13
14use serde::{Deserialize, Serialize};
15
16/// The least and most a size may be, in points.
17const SIZES: (f32, f32) = (6.0, 48.0);
18
19/// A font and a size of a kind of widget's own; what it leaves out is the
20/// theme's.
21#[derive(Clone, Debug, Default, PartialEq, Deserialize, Serialize)]
22#[serde(deny_unknown_fields)]
23pub struct Face {
24    /// The font, by its file's stem or its family.
25    #[serde(default, skip_serializing_if = "Option::is_none")]
26    pub font: Option<String>,
27    /// The size, in points.
28    #[serde(default, skip_serializing_if = "Option::is_none")]
29    pub size: Option<f32>,
30}
31
32impl Face {
33    /// Whether it sets nothing.
34    #[must_use]
35    pub fn is_empty(&self) -> bool {
36        self.font.is_none() && self.size.is_none()
37    }
38
39    /// `self` with what `over` sets laid over it.
40    #[must_use]
41    pub fn under(&self, over: &Face) -> Face {
42        Face {
43            font: over.font.clone().or_else(|| self.font.clone()),
44            size: over.size.or(self.size),
45        }
46    }
47
48    /// Kept within bounds, an empty font name none.
49    #[must_use]
50    pub fn bounded(&self) -> Face {
51        Face {
52            font: named(self.font.as_ref()),
53            size: self.size.map(|size| size.clamp(SIZES.0, SIZES.1)),
54        }
55    }
56}
57
58/// A theme's type, whole.
59#[derive(Clone, Debug, PartialEq)]
60pub struct Type {
61    /// The UI's font; `None`, egui's.
62    pub ui_font: Option<String>,
63    /// The UI's text size, in points.
64    pub ui_size: f32,
65    /// The story's font, its streams' and the room's; `None`, the UI's.
66    pub story_font: Option<String>,
67    /// The story's text size, in points.
68    pub story_size: f32,
69    /// The fixed-width font, for tables and the log window; `None`,
70    /// egui's.
71    pub mono_font: Option<String>,
72    /// The fixed-width text's size, in points.
73    pub mono_size: f32,
74    /// Points added between two lines of the story and its streams.
75    pub line_spacing: f32,
76    /// Fonts tried in turn for a character the font drawing it lacks.
77    pub fallbacks: Vec<String>,
78    /// A font or size of a kind of widget's own, by the kind's key:
79    /// `story`, `room`, a stream's id (`thoughts`).
80    pub widgets: BTreeMap<String, Face>,
81}
82
83impl Default for Type {
84    /// egui's own fonts at its own sizes.
85    fn default() -> Self {
86        Self {
87            ui_font: None,
88            ui_size: 13.0,
89            story_font: None,
90            story_size: 13.0,
91            mono_font: None,
92            mono_size: 13.0,
93            line_spacing: 0.0,
94            fallbacks: Vec::new(),
95            widgets: BTreeMap::new(),
96        }
97    }
98}
99
100impl Type {
101    /// Every font it names: what has to be loaded to draw it.
102    #[must_use]
103    pub fn fonts(&self) -> Vec<String> {
104        let mut fonts: Vec<String> = [&self.ui_font, &self.story_font, &self.mono_font]
105            .into_iter()
106            .flatten()
107            .cloned()
108            .chain(self.fallbacks.iter().cloned())
109            .chain(self.widgets.values().filter_map(|face| face.font.clone()))
110            .collect();
111        fonts.sort();
112        fonts.dedup();
113        fonts
114    }
115}

117/// The type as a file writes it: a part left out is the base's.
118#[derive(Clone, Debug, Default, PartialEq, Deserialize, Serialize)]
119#[serde(deny_unknown_fields)]
120pub struct TypeFile {
121    /// The UI's font; empty, egui's.
122    #[serde(default, skip_serializing_if = "Option::is_none")]
123    pub ui_font: Option<String>,
124    /// The UI's text size.
125    #[serde(default, skip_serializing_if = "Option::is_none")]
126    pub ui_size: Option<f32>,
127    /// The story's font; empty, the UI's.
128    #[serde(default, skip_serializing_if = "Option::is_none")]
129    pub story_font: Option<String>,
130    /// The story's text size.
131    #[serde(default, skip_serializing_if = "Option::is_none")]
132    pub story_size: Option<f32>,
133    /// The fixed-width font; empty, egui's.
134    #[serde(default, skip_serializing_if = "Option::is_none")]
135    pub mono_font: Option<String>,
136    /// The fixed-width text's size.
137    #[serde(default, skip_serializing_if = "Option::is_none")]
138    pub mono_size: Option<f32>,
139    /// The story's line spacing.
140    #[serde(default, skip_serializing_if = "Option::is_none")]
141    pub line_spacing: Option<f32>,
142    /// The fallbacks, in turn; in place of the base's.
143    #[serde(default, skip_serializing_if = "Option::is_none")]
144    pub fallbacks: Option<Vec<String>>,
145    /// Each kind of widget's own, laid over the base's kind by kind.
146    #[serde(default, skip_serializing_if = "BTreeMap::is_empty")]
147    pub widgets: BTreeMap<String, Face>,
148}

150/// A font name as kept: trimmed, and none when empty.
151fn named(name: Option<&String>) -> Option<String> {
152    name.map(|name| name.trim().to_owned())
153        .filter(|name| !name.is_empty())
154}
155
156impl TypeFile {
157    /// Whether it sets nothing.
158    #[must_use]
159    pub fn is_empty(&self) -> bool {
160        *self == Self::default()
161    }
162
163    /// What `kind` says that `base` does not, as a file writes it; a font
164    /// the base names and this does not is written empty, egui's own.
165    #[must_use]
166    pub fn differing(kind: &Type, base: &Type) -> Self {
167        let font = |a: &Option<String>, b: &Option<String>| {
168            (a != b).then(|| a.clone().unwrap_or_default())
169        };
170        let differs = |a: f32, b: f32| ((a - b).abs() > f32::EPSILON).then_some(a);
171        Self {
172            ui_font: font(&kind.ui_font, &base.ui_font),
173            ui_size: differs(kind.ui_size, base.ui_size),
174            story_font: font(&kind.story_font, &base.story_font),
175            story_size: differs(kind.story_size, base.story_size),
176            mono_font: font(&kind.mono_font, &base.mono_font),
177            mono_size: differs(kind.mono_size, base.mono_size),
178            line_spacing: differs(kind.line_spacing, base.line_spacing),
179            fallbacks: (kind.fallbacks != base.fallbacks).then(|| kind.fallbacks.clone()),
180            widgets: kind
181                .widgets
182                .iter()
183                .filter(|(key, face)| base.widgets.get(*key) != Some(*face))
184                .map(|(key, face)| (key.clone(), face.clone()))
185                .collect(),
186        }
187    }
188
189    /// `kind` with what this file sets laid over it, each within its
190    /// bounds; an empty font name is egui's own.
191    #[must_use]
192    pub fn over(&self, kind: Type) -> Type {
193        let font = |set: &Option<String>, base: Option<String>| match set {
194            Some(_) => named(set.as_ref()),
195            None => base,
196        };
197        let size = |set: Option<f32>, base: f32| set.unwrap_or(base).clamp(SIZES.0, SIZES.1);
198        let mut widgets = kind.widgets;
199        for (key, face) in &self.widgets {
200            let face = face.bounded();
201            if face.is_empty() {
202                widgets.remove(key);
203            } else {
204                widgets.insert(key.clone(), face);
205            }
206        }
207        Type {
208            ui_font: font(&self.ui_font, kind.ui_font),
209            ui_size: size(self.ui_size, kind.ui_size),
210            story_font: font(&self.story_font, kind.story_font),
211            story_size: size(self.story_size, kind.story_size),
212            mono_font: font(&self.mono_font, kind.mono_font),
213            mono_size: size(self.mono_size, kind.mono_size),
214            line_spacing: self
215                .line_spacing
216                .unwrap_or(kind.line_spacing)
217                .clamp(0.0, 24.0),
218            fallbacks: self.fallbacks.as_ref().map_or(kind.fallbacks, |fallbacks| {
219                fallbacks
220                    .iter()
221                    .filter_map(|name| named(Some(name)))
222                    .collect()
223            }),
224            widgets,
225        }
226    }
227}

229#[cfg(test)]
230mod tests {
231    use super::*;
232
233    #[test]
234    fn a_file_sets_what_it_says_within_bounds_and_an_empty_font_is_eguis() {
235        let file: TypeFile = toml::from_str(
236            "ui_font = \"Iosevka\"\nstory_size = 100\nline_spacing = 3\nfallbacks = [\"Segoe UI Symbol\", \" \"]\n[widgets.thoughts]\nsize = 11\n[widgets.room]\nfont = \"Consolas\"\n",
237        )
238        .expect("reads");
239        let kind = file.over(Type {
240            story_font: Some("Hack".to_owned()),
241            ..Type::default()
242        });
243        assert_eq!(kind.ui_font.as_deref(), Some("Iosevka"));
244        assert_eq!(kind.story_font.as_deref(), Some("Hack"), "the base's");
245        assert!((kind.story_size - 48.0).abs() < f32::EPSILON, "at most");
246        assert!((kind.ui_size - 13.0).abs() < f32::EPSILON, "the base's");
247        assert!((kind.line_spacing - 3.0).abs() < f32::EPSILON);
248        assert_eq!(kind.fallbacks, ["Segoe UI Symbol"]);
249        assert_eq!(kind.widgets["thoughts"].size, Some(11.0));
250        assert_eq!(kind.widgets["room"].font.as_deref(), Some("Consolas"));
251        assert_eq!(
252            kind.fonts(),
253            ["Consolas", "Hack", "Iosevka", "Segoe UI Symbol"]
254        );
255        let cleared: TypeFile = toml::from_str("story_font = \"\"\n").expect("reads");
256        assert_eq!(cleared.over(kind).story_font, None);
257        assert!(toml::from_str::<TypeFile>("weight = 700\n").is_err());
258        assert!(toml::from_str::<TypeFile>("[widgets.room]\nweight = 7\n").is_err());
259        assert!(TypeFile::default().is_empty());
260    }
261
262    #[test]
263    fn only_what_differs_is_written_and_reads_back() {
264        let base = Type::default();
265        let mut kind = base.clone();
266        kind.mono_font = Some("Consolas".to_owned());
267        kind.widgets.insert(
268            "thoughts".to_owned(),
269            Face {
270                font: None,
271                size: Some(11.0),
272            },
273        );
274        let file = TypeFile::differing(&kind, &base);
275        assert_eq!(file.mono_font.as_deref(), Some("Consolas"));
276        assert_eq!(file.ui_font, None);
277        assert_eq!(file.widgets.len(), 1);
278        let text = toml::to_string(&file).expect("written");
279        let read: TypeFile = toml::from_str(&text).expect("read");
280        assert_eq!(read.over(base), kind);
281    }

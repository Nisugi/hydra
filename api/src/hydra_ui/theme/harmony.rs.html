1//! The harmony generator: a whole palette from one seed colour against a
2//! background, covering every token of the palette. The engine is pure: a
3//! [`Recipe`] in, a [`Palette`] out, the same every time.
4//!
5//! The idea: the seed's hue anchors a [`Scheme`] of hues round the OKLCH
6//! circle, and each *free* token takes a slot in it, nudged in lightness and
7//! chroma so two on one hue stay apart. A token that *carries meaning* by
8//! its hue (health red, mana blue, a bank's mark gold) is *anchored*: it
9//! keeps the hue it has always had and harmony moves only its lightness and
10//! chroma. Every generated colour is lifted until it clears a contrast floor
11//! against the background, which is how a light theme costs one input, and
12//! kept a least distance from the others in its group. A *pin* is kept as
13//! written; the rest harmonise round it.
14

20/// Hue offsets in degrees round the OKLCH hue circle. The golden angle is
21/// the honest answer to "give me N distinct hues": 137.5 degrees never
22/// repeats and spreads evenly however many are taken.
23#[derive(Clone, Copy, Debug, PartialEq, Eq)]
24pub enum Scheme {
25    /// One hue, varied in lightness and chroma.
26    Monochrome,
27    /// Neighbours: calm, low contrast.
28    Analogous,
29    /// Opposites: a strong pairing.
30    Complementary,
31    /// Softer than complementary.
32    Split,
33    /// Even thirds: vivid, balanced.
34    Triadic,
35    /// Two complementary pairs.
36    Tetradic,
37    /// The golden angle: the most separation.
38    Golden,
39    /// An analogous pair plus its opposite.
40    Compound,
41}
42
43impl Scheme {
44    /// Every scheme, in the order the editor offers them.
45    pub const ALL: [Scheme; 8] = [
46        Scheme::Monochrome,
47        Scheme::Analogous,
48        Scheme::Complementary,
49        Scheme::Split,
50        Scheme::Triadic,
51        Scheme::Tetradic,
52        Scheme::Golden,
53        Scheme::Compound,
54    ];
55
56    /// Its hue offsets from the seed's hue, in degrees.
57    #[must_use]
58    pub const fn offsets(self) -> &'static [f64] {
59        match self {
60            Scheme::Monochrome => &[0.0],
61            Scheme::Analogous => &[0.0, 30.0, -30.0, 60.0, -60.0],
62            Scheme::Complementary => &[0.0, 180.0],
63            Scheme::Split => &[0.0, 150.0, 210.0],
64            Scheme::Triadic => &[0.0, 120.0, 240.0],
65            Scheme::Tetradic => &[0.0, 90.0, 180.0, 270.0],
66            Scheme::Golden => &[0.0, 137.5, 275.0, 52.5, 190.0, 327.5],
67            Scheme::Compound => &[0.0, 30.0, 180.0, 210.0],
68        }
69    }
70
71    /// Its name, as a theme file writes it.
72    #[must_use]
73    pub const fn name(self) -> &'static str {
74        match self {
75            Scheme::Monochrome => "monochrome",
76            Scheme::Analogous => "analogous",
77            Scheme::Complementary => "complementary",
78            Scheme::Split => "split",
79            Scheme::Triadic => "triadic",
80            Scheme::Tetradic => "tetradic",
81            Scheme::Golden => "golden",
82            Scheme::Compound => "compound",
83        }
84    }
85
86    /// What it does, in a few words.
87    #[must_use]
88    pub const fn description(self) -> &'static str {
89        match self {
90            Scheme::Monochrome => "one hue, varied lightness and chroma",
91            Scheme::Analogous => "neighbours: calm, low contrast",
92            Scheme::Complementary => "opposites: strong pairing",
93            Scheme::Split => "softer than complementary",
94            Scheme::Triadic => "even thirds: vivid, balanced",
95            Scheme::Tetradic => "two complementary pairs",
96            Scheme::Golden => "golden angle: most separation",
97            Scheme::Compound => "analogous pair plus its opposite",
98        }
99    }
100
101    /// The scheme named `name`, ignoring case.
102    #[must_use]
103    pub fn parse(name: &str) -> Option<Scheme> {
104        Scheme::ALL
105            .into_iter()
106            .find(|s| s.name().eq_ignore_ascii_case(name.trim()))
107    }
108}

110/// How a token takes its colour from the harmony.
111#[derive(Clone, Copy, Debug, PartialEq)]
112pub enum Role {
113    /// A slot in the scheme's hues, nudged in lightness and chroma.
114    Free {
115        /// Which of the scheme's offsets.
116        slot: usize,
117        /// Added to the seed's lightness.
118        dl: f64,
119        /// Added to the seed's chroma.
120        dc: f64,
121    },
122    /// The hue the token has always had ([`Token::bare`]'s), its lightness
123    /// and chroma the seed's, nudged.
124    Anchored {
125        /// Added to the seed's lightness.
126        dl: f64,
127        /// Added to the seed's chroma.
128        dc: f64,
129    },
130    /// Text on the surfaces: the seed's hue nearly greyed, its lightness
131    /// `dl` away from the background's, toward the light on a dark ground
132    /// and toward the dark on a light one.
133    Onto {
134        /// How far from the background's lightness.
135        dl: f64,
136        /// Added to the seed's chroma.
137        dc: f64,
138    },
139    /// A surface: the background, its lightness moved by `dl`. Never lifted
140    /// to the contrast floor, which is for what is drawn *on* a surface.
141    Surface {
142        /// Added to the background's lightness.
143        dl: f64,
144    },
145    /// The plate under the room's name: its hue, the lightness solved so the
146    /// name reads on it at the recipe's spread.
147    Plate,
148    /// Kept as it is ([`Token::bare`]) unless pinned: a veil or a grid line
149    /// drawn part clear, whose colour is not a colour.
150    Fixed,
151}

153/// Every group of tokens keeps its distances within itself: a vital and a
154/// map line are never on one screen at once, so they need not differ.
155#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
156pub enum Group {
157    /// The surfaces and the text on them.
158    Surfaces,
159    /// The story's and the room's text, the accents.
160    Text,
161    /// The lines said on the game's channels.
162    Channels,
163    /// The bars.
164    Vitals,
165    /// The doll.
166    Injuries,
167    /// The indicators and the lists of effects.
168    Status,
169    /// The map's drawing.
170    Map,
171    /// The map's marks.
172    Marks,
173    /// What is not a colour.
174    Chrome,
175}

177/// What a theme generates from: the seed, the background, a scheme and the
178/// dials, and the pins kept as written.
179#[derive(Clone, Debug, PartialEq)]
180pub struct Recipe {
181    /// Its OKLCH hue anchors the scheme; its lightness and chroma anchor the
182    /// candidate ladder.
183    pub seed: Rgb,
184    /// What every generated colour must read against.
185    pub background: Rgb,
186    /// The hues.
187    pub scheme: Scheme,
188    /// A multiplier on the scheme's offsets: low 0.7, medium 1.0, high 1.4.
189    pub variance: f64,
190    /// The WCAG contrast floor against the background: low 3.0, medium 4.5,
191    /// high 7.0.
192    pub contrast: f64,
193    /// The least `OKLab` distance wanted between two tokens of one group: low
194    /// 0.04, medium 0.09, high 0.15.
195    pub separation: f64,
196    /// The room's name's contrast against its plate: 2.5 reads as a subtle
197    /// plate, 7.0 as a hard label.
198    pub room_spread: f64,
199    /// Tokens kept as written; the rest harmonise round them.
200    pub pins: BTreeMap<Token, Rgb>,
201    /// Kinds of line drawn as the story's own text, as if they had no
202    /// colour of their own: a pin's `none`. Only a token that
203    /// [`Token::may_be_plain`] is.
204    pub plain: BTreeSet<Token>,
205    /// Lines of a kind named one by one (`flare:acid`) given a colour of
206    /// their own over their kind's, by their preset; `None`, drawn as the
207    /// story's own text.
208    pub names: BTreeMap<String, Option<Rgb>>,
209}

211impl Default for Recipe {
212    /// Despana's link blue on its canvas, triadic, every dial at medium.
213    fn default() -> Self {
214        Self {
215            seed: [0x47, 0x7a, 0xb3],
216            background: [0x0d, 0x11, 0x15],
217            scheme: Scheme::Triadic,
218            variance: 1.0,
219            contrast: 4.5,
220            separation: 0.09,
221            room_spread: 2.5,
222            pins: BTreeMap::new(),
223            plain: BTreeSet::new(),
224            names: BTreeMap::new(),
225        }
226    }

229/// A seed that does not parse as a colour would still be one; this is the
230/// ladder's start when nothing better is known.
231const FALLBACK_SEED: Lch = [0.7, 0.12, 250.0];
232
233fn hue_for(token: Token, recipe: &Recipe, seed_hue: f64) -> f64 {
234    match token.role() {
235        Role::Free { slot, .. } => {
236            let off = recipe.scheme.offsets();
237            (seed_hue + off[slot % off.len()] * recipe.variance).rem_euclid(360.0)
238        }
239        Role::Onto { .. } => seed_hue,
240        _ => to_lch(token.bare())[2],
241    }
242}

244/// Raise (or lower, on a light background) the lightness until the colour
245/// clears the contrast floor against the background. Cheaper and less
246/// destructive than desaturating.
247fn lift_to_contrast(mut l: f64, c: f64, h: f64, recipe: &Recipe) -> Rgb {
248    let mut rgb = from_lch([l, c, h]);
249    if contrast(rgb, recipe.background) >= recipe.contrast {
250        return rgb;
251    }
252    let step = if to_lch(recipe.background)[0] < 0.5 {
253        0.015
254    } else {
255        -0.015
256    };
257    for _ in 0..40 {
258        l = (l + step).clamp(0.05, 0.99);
259        rgb = from_lch([l, c, h]);
260        if contrast(rgb, recipe.background) >= recipe.contrast {
261            break;
262        }
263    }
264    rgb
265}

267/// A ladder of lightness and chroma round a nominal point, each rung lifted
268/// to the contrast floor. Tokens sharing a hue separate along lightness,
269/// which is what the eye reads first. Nearest the nominal point first.
270fn candidates_at(hue: f64, dl: f64, dc: f64, recipe: &Recipe, seed: Lch) -> Vec<Rgb> {
271    let mut out = Vec::new();
272    for step_l in [0.0, 0.09, -0.09, 0.18, -0.18, 0.27, -0.27] {
273        for step_c in [0.0, 0.05, -0.05, 0.10] {
274            let l = (seed[0] + dl + step_l).clamp(0.28, 0.94);
275            let c = (seed[1] + dc + step_c).clamp(0.015, 0.33);
276            let rgb = lift_to_contrast(l, c, hue, recipe);
277            if !out.contains(&rgb) {
278                out.push(rgb);
279            }
280        }
281    }
282    out
283}

285/// The first candidate (nearest the nominal point) that clears the
286/// separation floor and is not lost in the background; else the one
287/// farthest from what is used, so two tokens on one hue never come out the
288/// same.
289fn pick_color(cands: &[Rgb], used: &[Rgb], recipe: &Recipe) -> Rgb {
290    let min_dist = |rgb: Rgb| {
291        used.iter()
292            .map(|u| delta_e(*u, rgb))
293            .fold(f64::INFINITY, f64::min)
294    };
295    let visible = |rgb: &&Rgb| delta_e(**rgb, recipe.background) >= 0.06;
296    cands
297        .iter()
298        .find(|c| visible(c) && min_dist(**c) >= recipe.separation)
299        .or_else(|| {
300            cands
301                .iter()
302                .filter(visible)
303                .max_by(|a, b| min_dist(**a).total_cmp(&min_dist(**b)))
304        })
305        .or_else(|| cands.first())
306        .copied()
307        .unwrap_or(recipe.seed)
308}

310/// The plate under the room's name: the name's hue, a little of its chroma,
311/// the lightness solved by bisection so the pair hits the recipe's spread.
312fn plate(name: Rgb, recipe: &Recipe) -> Rgb {
313    let rl = to_lch(name);
314    let c = (rl[1] * 0.7).clamp(0.02, 0.2);
315    let (mut lo, mut hi) = (0.05, (rl[0] - 0.02).max(0.05));
316    let mut plate = from_lch([lo, c, rl[2]]);
317    for _ in 0..26 {
318        let mid = f64::midpoint(lo, hi);
319        plate = from_lch([mid, c, rl[2]]);
320        if contrast(name, plate) > recipe.room_spread {
321            lo = mid;
322        } else {
323            hi = mid;
324        }
325    }
326    plate
327}

329/// The whole palette from `recipe`. The same recipe gives the same palette.
330#[must_use]
331pub fn generate(recipe: &Recipe) -> Palette {
332    let seed = to_lch(recipe.seed);
333    let seed = if seed[1].is_finite() {
334        seed
335    } else {
336        FALLBACK_SEED
337    };
338    let background = to_lch(recipe.background);
339    let mut palette = Palette::bare();
340    let mut used: BTreeMap<Group, Vec<Rgb>> = BTreeMap::new();
341    for token in Token::ALL {
342        let unpinned = token.plain_unless_pinned() && !recipe.pins.contains_key(&token);
343        if token.may_be_plain() && (recipe.plain.contains(&token) || unpinned) {
344            palette.set(token, palette.get(Token::Text));
345            continue;
346        }
347        if let Some(pinned) = recipe.pins.get(&token) {
348            used.entry(token.group()).or_default().push(*pinned);
349            palette.set(token, *pinned);
350            continue;
351        }
352        let (dl, dc) = match token.role() {
353            Role::Free { dl, dc, .. } | Role::Anchored { dl, dc } => (dl, dc),
354            Role::Onto { dl, dc } => {
355                let away = if background[0] < 0.5 { dl } else { -dl };
356                (background[0] + away - seed[0], dc)
357            }
358            Role::Surface { dl } => {
359                let l = (background[0] + dl).clamp(0.0, 1.0);
360                palette.set(token, from_lch([l, background[1], background[2]]));
361                continue;
362            }
363            Role::Plate => {
364                palette.set(token, plate(palette.get(Token::RoomName), recipe));
365                continue;
366            }
367            Role::Fixed => continue,
368        };
369        let cands = candidates_at(hue_for(token, recipe, seed[2]), dl, dc, recipe, seed);
370        let group = used.entry(token.group()).or_default();
371        let pick = pick_color(&cands, group, recipe);
372        group.push(pick);
373        palette.set(token, pick);
374    }
375    let text = palette.get(Token::Text);
376    palette.set_names(
377        recipe
378            .names
379            .iter()
380            .map(|(name, color)| (name.clone(), color.unwrap_or(text)))
381            .collect(),
382    );
383    palette
384}

386/// For the editor's swatch explorer: `token`'s lightness and chroma held,
387/// its hue turned by noticeable steps round the wheel, so the player
388/// chooses between hues rather than shades of one. Empty for a token that
389/// is not free.
390#[must_use]
391pub fn hue_variants(token: Token, recipe: &Recipe) -> Vec<Rgb> {
392    let Role::Free { dl, dc, .. } = token.role() else {
393        return Vec::new();
394    };
395    let seed = to_lch(recipe.seed);
396    let base_h = hue_for(token, recipe, seed[2]);
397    let l = (seed[0] + dl).clamp(0.30, 0.92);
398    let c = (seed[1] + dc).clamp(0.03, 0.30);
399    let mut out = Vec::new();
400    for d in [
401        0.0, 30.0, -30.0, 60.0, -60.0, 90.0, -90.0, 140.0, 180.0, 220.0,
402    ] {
403        let rgb = lift_to_contrast(l, c, (base_h + d).rem_euclid(360.0), recipe);
404        if !out.contains(&rgb) {
405            out.push(rgb);
406        }
407    }
408    out
409}

411/// Seeds worth offering from `raw` colours (a theme's own, say): near-greys
412/// and colours lost in `background` dropped, perceptual duplicates dropped,
413/// the most vivid first, at most `cap`. Every swatch offered must be a safe
414/// seed, or choosing by eye is guesswork.
415#[must_use]
416pub fn seed_swatches(raw: &[Rgb], background: Rgb, cap: usize) -> Vec<Rgb> {
417    let pass = |min_chroma: f64| {
418        let mut swatches: Vec<(Rgb, f64)> = Vec::new();
419        for rgb in raw {
420            let [_, chroma, _] = to_lch(*rgb);
421            let lost = delta_e(*rgb, background) < 0.15;
422            let seen = swatches.iter().any(|(s, _)| delta_e(*s, *rgb) < 0.02);
423            if chroma >= min_chroma && !lost && !seen {
424                swatches.push((*rgb, chroma));
425            }
426        }
427        swatches
428    };
429    // An all-grey set has nothing vivid; a grey seed is then coherent, so
430    // the chroma floor is relaxed rather than nothing offered.
431    let mut swatches = pass(0.04);
432    if swatches.is_empty() {
433        swatches = pass(0.0);
434    }
435    swatches.sort_by(|a, b| b.1.total_cmp(&a.1));
436    swatches.truncate(cap);
437    swatches.into_iter().map(|(rgb, _)| rgb).collect()
438}

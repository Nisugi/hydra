1//! A theme as a file, and the themes Hydra has: a name, a base it takes
2//! what it does not say from, a recipe, and pins. The palette is computed
3//! from them when the theme is looked up, never saved.
4//!
5//! Two themes are built in: **Despana**, Hydra's plain colours with a dark
6//! slate chrome round them, and **Light**, generated from a light
7//! background. A file in the `themes` folder adds one more, holding only
8//! what it says.

143impl RecipeFile {
144    fn is_empty(&self) -> bool {
145        *self == Self::default()
146    }
147
148    /// What `recipe` says that `base` does not, as a file writes it; the
149    /// pins are the theme's own and not here.
150    #[must_use]
151    pub fn differing(recipe: &Recipe, base: &Recipe) -> Self {
152        let differs = |a: f64, b: f64| ((a - b).abs() > 1e-9).then_some(a);
153        Self {
154            seed: (recipe.seed != base.seed).then(|| hex(recipe.seed)),
155            background: (recipe.background != base.background).then(|| hex(recipe.background)),
156            scheme: (recipe.scheme != base.scheme).then(|| recipe.scheme.name().to_owned()),
157            variance: differs(recipe.variance, base.variance),
158            contrast: differs(recipe.contrast, base.contrast),
159            separation: differs(recipe.separation, base.separation),
160            room_spread: differs(recipe.room_spread, base.room_spread),
161        }
162    }
163
164    /// `recipe` with what this file sets laid over it.
165    ///
166    /// # Errors
167    ///
168    /// A colour or a scheme that does not read, named.
169    fn over(&self, mut recipe: Recipe) -> Result<Recipe, String> {
170        if let Some(seed) = &self.seed {
171            recipe.seed =
172                parse_hex(seed).ok_or_else(|| format!("seed `{seed}` is not a colour"))?;
173        }
174        if let Some(background) = &self.background {
175            recipe.background = parse_hex(background)
176                .ok_or_else(|| format!("background `{background}` is not a colour"))?;
177        }
178        if let Some(scheme) = &self.scheme {
179            recipe.scheme =
180                Scheme::parse(scheme).ok_or_else(|| format!("`{scheme}` is not a scheme"))?;
181        }
182        recipe.variance = self.variance.unwrap_or(recipe.variance);
183        recipe.contrast = self.contrast.unwrap_or(recipe.contrast);
184        recipe.separation = self.separation.unwrap_or(recipe.separation);
185        recipe.room_spread = self.room_spread.unwrap_or(recipe.room_spread);
186        Ok(recipe)
187    }
188}

196    /// **Despana**: a dark slate chrome round Hydra's plain colours
197    /// ([`Token::bare`]), every one pinned but those plain until a theme
198    /// pins them ([`Token::plain_unless_pinned`]).
199    #[must_use]
200    pub fn despana() -> Self {
201        let mut pins: BTreeMap<Token, Rgb> = Token::ALL
202            .into_iter()
203            .filter(|token| !token.plain_unless_pinned())
204            .map(|t| (t, t.bare()))
205            .collect();
206        for (token, css) in [
207            (Token::Canvas, [0x0d, 0x11, 0x15]),
208            (Token::Surface, [0x18, 0x1e, 0x24]),
209            (Token::Raised, [0x20, 0x27, 0x2e]),
210            (Token::Inset, [0x11, 0x16, 0x1b]),
211            (Token::Line, [0x35, 0x40, 0x4a]),
212            (Token::LineStrong, [0x76, 0x5f, 0x39]),
213            (Token::Text, [0xdd, 0xdc, 0xd7]),
214            (Token::Muted, [0xa4, 0xae, 0xb7]),
215            (Token::Selection, [0x3b, 0x4a, 0x5a]),
216        ] {
217            pins.insert(token, css);
218        }
219        Self {
220            name: Self::DEFAULT.to_owned(),
221            base: None,
222            recipe: RecipeFile {
223                seed: Some(hex([0xd7, 0xad, 0x63])),
224                background: Some(hex([0x0d, 0x11, 0x15])),
225                scheme: Some(Scheme::Compound.name().to_owned()),
226                ..RecipeFile::default()

228            pins,

260    /// Read from `text`, a theme file; `stem` names it when it does not
261    /// name itself.
262    ///
263    /// # Errors
264    ///
265    /// Why it does not read.
266    pub fn parse(text: &str, stem: &str) -> Result<Self, String> {
267        let file: File = toml::from_str(text).map_err(|why| why.to_string())?;
268        let mut pins = BTreeMap::new();
269        let mut plain = BTreeSet::new();
270        for (name, colour) in &file.pins {
271            let token = Token::named(name).ok_or_else(|| format!("`{name}` is not a token"))?;
272            if colour.eq_ignore_ascii_case("none") {
273                if !token.may_be_plain() {
274                    return Err(format!(
275                        "{name} = `none`: only a kind of line can be left plain"
276                    ));
277                }
278                plain.insert(token);
279                continue;
280            }
281            let rgb =
282                parse_hex(colour).ok_or_else(|| format!("{name} = `{colour}` is not a colour"))?;
283            pins.insert(token, rgb);
284        }
285        let names = names_read(&file.names)?;
286        let prompt = prompt_read(&file.prompt)?;
287        file.recipe.over(Recipe::default())?;
288        Ok(Self {
289            name: file.name.unwrap_or_else(|| stem.to_owned()),
290            base: file.base,
291            recipe: file.recipe,
292            pins,
293            plain,
294            names,
295            prompt,
296            shape: file.shape,
297            kind: file.kind,
298            frame: file.frame,
299        })
300    }
301
302    /// As a theme file writes it.
303    #[must_use]
304    pub fn to_toml(&self) -> String {
305        let file = File {
306            name: Some(self.name.clone()),
307            base: self.base.clone(),
308            recipe: self.recipe.clone(),
309            pins: self
310                .pins
311                .iter()
312                .map(|(token, rgb)| (token.name().to_owned(), hex(*rgb)))
313                .chain(
314                    self.plain
315                        .iter()
316                        .map(|token| (token.name().to_owned(), "none".to_owned())),
317                )
318                .collect(),
319            names: names_written(&self.names),
320            prompt: prompt_written(&self.prompt),
321            shape: self.shape.clone(),
322            kind: self.kind.clone(),
323            frame: self.frame.clone(),
324        };
325        toml::to_string(&file).unwrap_or_default()
326    }

404impl Themes {
405    /// The built-ins alone.
406    #[must_use]
407    pub fn built_in() -> Self {
408        let mut themes = vec![Theme::despana(), Theme::light()];
409        // Vellum's themes, each a recipe of its own; one that would not read
410        // is a fault of the build, caught by its test, and left out.
411        themes.extend(
412            VELLUM
413                .iter()
414                .filter_map(|(stem, text)| Theme::parse(text, stem).ok()),
415        );
416        Self {
417            themes,
418            problems: Vec::new(),
419        }
420    }

422    /// The built-ins and every `.toml` in `folder`. A file named as a
423    /// built-in is is read after it and wins.
424    #[must_use]
425    pub fn load(folder: &Path) -> Self {
426        let mut themes = Self::built_in();
427        let Ok(entries) = std::fs::read_dir(folder) else {
428            return themes;
429        };
430        let mut files: Vec<_> = entries
431            .filter_map(Result::ok)
432            .map(|entry| entry.path())
433            .filter(|path| path.extension().is_some_and(|ext| ext == "toml"))
434            .collect();
435        files.sort();
436        for path in files {
437            let stem = path
438                .file_stem()
439                .map(|s| s.to_string_lossy().into_owned())
440                .unwrap_or_default();
441            let read = std::fs::read_to_string(&path)
442                .map_err(|why| why.to_string())
443                .and_then(|text| Theme::parse(&text, &stem));
444            match read {
445                Ok(theme) => themes.add(theme),
446                Err(why) => themes.problems.push(format!("{}: {why}", path.display())),
447            }
448        }
449        themes
450    }

452    /// The `.toml` files in the themes folder `folder`, each with when it
453    /// was last changed, in order: what [`Themes::load`] reads, to see
454    /// whether it has changed before reading it whole.
455    #[must_use]
456    pub fn files(folder: &Path) -> Vec<(std::path::PathBuf, Option<std::time::SystemTime>)> {
457        let mut files: Vec<_> = std::fs::read_dir(folder)
458            .map(|entries| {
459                entries
460                    .filter_map(Result::ok)
461                    .map(|entry| {
462                        let modified = entry.metadata().and_then(|m| m.modified()).ok();
463                        (entry.path(), modified)
464                    })
465                    .filter(|(path, _)| path.extension().is_some_and(|ext| ext == "toml"))
466                    .collect()
467            })
468            .unwrap_or_default();
469        files.sort();
470        files
471    }

494    /// The recipe and pins of the theme named `name`, its base's under it,
495    /// and its shape and type the same way.
496    ///
497    /// # Errors
498    ///
499    /// No theme of that name, a base that is missing or circular, or a
500    /// value that does not read.
501    pub fn resolve(&self, name: &str) -> Result<(Recipe, Shape, Type), String> {
502        let chain = self.chain(name)?;
503        let mut recipe = Recipe::default();
504        let mut shape = Shape::default();
505        let mut kind = Type::default();
506        for theme in chain.iter().rev() {
507            recipe = theme
508                .recipe
509                .over(recipe)
510                .map_err(|why| format!("{}: {why}", theme.name))?;
511            recipe.pins.extend(theme.pins.iter().map(|(t, c)| (*t, *c)));
512            // A theme's own word on a kind of line, a colour or none, over
513            // its base's.
514            for token in theme.pins.keys() {
515                recipe.plain.remove(token);
516            }
517            for token in &theme.plain {
518                recipe.pins.remove(token);
519                recipe.plain.insert(*token);
520            }
521            recipe.names.extend(
522                theme
523                    .names
524                    .iter()
525                    .map(|(name, color)| (name.clone(), *color)),
526            );
527            recipe
528                .prompt
529                .extend(theme.prompt.iter().map(|(c, look)| (*c, *look)));
530            shape = theme.shape.over(shape);
531            kind = theme.kind.over(kind);
532        }
533        Ok((recipe, shape, kind))
534    }

536    /// The theme named `name` and its bases after it, the root last.
537    fn chain(&self, name: &str) -> Result<Vec<&Theme>, String> {
538        let mut chain: Vec<&Theme> = Vec::new();
539        let mut next = Some(name);
540        while let Some(name) = next {
541            if chain.len() >= 8 {
542                return Err(format!("`{name}`'s bases go round in a circle"));
543            }
544            let theme = self
545                .get(name)
546                .ok_or_else(|| format!("there is no theme `{name}`"))?;
547            next = theme.base.as_deref();
548            chain.push(theme);
549        }
550        Ok(chain)
551    }

583    /// What a character's window wears: `chosen`'s theme, or `hydras`, with
584    /// `chosen`'s accent pinned over it.
585    ///
586    /// # Errors
587    ///
588    /// As [`Themes::resolve`], or an accent that is not a colour.
589    pub fn outfit_for(&self, hydras: &str, chosen: &Chosen) -> Result<Outfit, String> {
590        let name = chosen.theme.as_deref().unwrap_or(hydras);
591        let (mut recipe, shape, kind) = self.resolve(name)?;
592        if let Some(accent) = &chosen.accent {
593            let rgb =
594                parse_hex(accent).ok_or_else(|| format!("accent `{accent}` is not a colour"))?;
595            recipe.pins.insert(Token::Accent, rgb);
596        }
597        Ok(Outfit {
598            palette: generate(&recipe),
599            shape,
600            kind,
601            frame: self.frame(name)?,
602        })
603    }
604}

606/// One `[prompt]` entry: a colour alone, or a colour and a ground.
607#[derive(Clone, Debug, PartialEq, Deserialize, Serialize)]
608#[serde(untagged)]
609enum PromptFile {
610    /// `R = "#e05050"`.
611    Fg(String),
612    /// `S = { fg = "#e0a030", bg = "#302010" }`.
613    Both {
614        #[serde(default, skip_serializing_if = "Option::is_none")]
615        fg: Option<String>,
616        #[serde(default, skip_serializing_if = "Option::is_none")]
617        bg: Option<String>,
618    },
619}
620
621/// A file's `[prompt]` table as each character's look.
622fn prompt_read(table: &BTreeMap<String, PromptFile>) -> Result<BTreeMap<char, PromptLook>, String> {
623    let colour = |key: &str, written: &str| {
624        parse_hex(written).ok_or_else(|| format!("[prompt] {key} = `{written}` is not a colour"))
625    };
626    let mut looks = BTreeMap::new();
627    for (key, entry) in table {
628        let mut chars = key.chars();
629        let (Some(character), None) = (chars.next(), chars.next()) else {
630            return Err(format!(
631                "[prompt] `{key}`: one character of the prompt, as R or !"
632            ));
633        };
634        let look = match entry {
635            PromptFile::Fg(fg) => PromptLook {
636                fg: Some(colour(key, fg)?),
637                bg: None,
638            },
639            PromptFile::Both { fg, bg } => PromptLook {
640                fg: fg.as_deref().map(|fg| colour(key, fg)).transpose()?,
641                bg: bg.as_deref().map(|bg| colour(key, bg)).transpose()?,
642            },
643        };
644        looks.insert(character, look);
645    }
646    Ok(looks)
647}
648
649/// Each character's look as a file's `[prompt]` table.
650fn prompt_written(looks: &BTreeMap<char, PromptLook>) -> BTreeMap<String, PromptFile> {
651    looks
652        .iter()
653        .map(|(character, look)| {
654            let entry = match (look.fg, look.bg) {
655                (Some(fg), None) => PromptFile::Fg(hex(fg)),
656                (fg, bg) => PromptFile::Both {
657                    fg: fg.map(hex),
658                    bg: bg.map(hex),
659                },
660            };
661            (character.to_string(), entry)
662        })
663        .collect()
664}

666/// A file's `[names.<kind>]` tables as presets and their colours.
667///
668/// # Errors
669///
670/// A kind not named one by one, or a colour that does not read.
671fn names_read(
672    file: &BTreeMap<String, BTreeMap<String, String>>,
673) -> Result<BTreeMap<String, Option<Rgb>>, String> {
674    let mut names = BTreeMap::new();
675    for (kind, colors) in file {
676        if !crate::combat_line::NAMED
677            .iter()
678            .any(|(named, _)| named == kind)
679        {
680            return Err(format!(
681                "[names.{kind}]: only a flare and a status on you are coloured by name"
682            ));
683        }
684        for (name, color) in colors {
685            let color =
686                if color.eq_ignore_ascii_case("none") {
687                    None
688                } else {
689                    Some(parse_hex(color).ok_or_else(|| {
690                        format!("[names.{kind}] {name} = `{color}` is not a colour")
691                    })?)
692                };
693            names.insert(format!("{kind}:{name}"), color);
694        }
695    }
696    Ok(names)
697}
698
699/// Presets and their colours as a file's `[names.<kind>]` tables.
700fn names_written(
701    names: &BTreeMap<String, Option<Rgb>>,
702) -> BTreeMap<String, BTreeMap<String, String>> {
703    let mut file: BTreeMap<String, BTreeMap<String, String>> = BTreeMap::new();
704    for (preset, color) in names {
705        let Some((kind, name)) = preset.split_once(':') else {
706            continue;
707        };
708        file.entry(kind.to_owned()).or_default().insert(
709            name.to_owned(),
710            color.map_or_else(|| "none".to_owned(), hex),
711        );
712    }
713    file
714}

716/// The file a theme named `name` is written to in the themes folder: its
717/// name in lower case, each character not a letter or digit a `-`.
718#[must_use]
719pub fn file_stem(name: &str) -> String {
720    name.trim()
721        .chars()
722        .map(|c| {
723            if c.is_alphanumeric() {
724                c.to_ascii_lowercase()
725            } else {
726                '-'
727            }
728        })
729        .collect()
730}

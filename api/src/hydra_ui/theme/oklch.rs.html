1//! Colour maths for the harmony generator: sRGB to `OKLab` and OKLCH and
2//! back (Ottosson's transform), a fit into the sRGB gamut that keeps hue and
3//! lightness, WCAG contrast and a perceptual distance; over [`Rgb`] rather
4//! than hex strings, since that is Hydra's colour.
5//!
6//! Equal OKLCH lightness reads equally bright across hues, which is why the
7//! generator works in it: a palette built there keeps every role readable.
8
9use super::Rgb;
10
11/// A colour in OKLCH: lightness 0 to 1, chroma 0 to about 0.33, hue in
12/// degrees.
13pub type Lch = [f64; 3];
14
15/// A colour in sRGB, each channel 0 to 1.
16type Srgb = [f64; 3];
17
18fn srgb_to_linear(c: f64) -> f64 {
19    if c <= 0.040_45 {
20        c / 12.92
21    } else {
22        ((c + 0.055) / 1.055).powf(2.4)
23    }
24}
25
26fn linear_to_srgb(c: f64) -> f64 {
27    if c <= 0.003_130_8 {
28        12.92 * c
29    } else {
30        1.055 * c.powf(1.0 / 2.4) - 0.055
31    }
32}
33
34fn to_srgb([red, green, blue]: Rgb) -> Srgb {
35    [
36        f64::from(red) / 255.0,
37        f64::from(green) / 255.0,
38        f64::from(blue) / 255.0,
39    ]
40}
41
42#[expect(
43    clippy::cast_possible_truncation,
44    clippy::cast_sign_loss,
45    reason = "clamped to 0..=1 and rounded, so the value is a byte"
46)]
47fn channel(v: f64) -> u8 {
48    (v.clamp(0.0, 1.0) * 255.0).round() as u8
49}
50
51fn to_rgb(srgb: Srgb) -> Rgb {
52    [channel(srgb[0]), channel(srgb[1]), channel(srgb[2])]
53}
54
55/// Ottosson's transform: linear sRGB to the cone response `lms`, cube
56/// rooted, then to `Lab`.
57fn rgb_to_oklab(rgb: Srgb) -> [f64; 3] {
58    let red = srgb_to_linear(rgb[0]);
59    let green = srgb_to_linear(rgb[1]);
60    let blue = srgb_to_linear(rgb[2]);
61    let long = (0.412_221_470_8 * red + 0.536_332_536_3 * green + 0.051_445_992_9 * blue).cbrt();
62    let medium = (0.211_903_498_2 * red + 0.680_699_545_1 * green + 0.107_396_956_6 * blue).cbrt();
63    let short = (0.088_302_461_9 * red + 0.281_718_837_6 * green + 0.629_978_700_5 * blue).cbrt();
64    [
65        0.210_454_255_3 * long + 0.793_617_785 * medium - 0.004_072_046_8 * short,
66        1.977_998_495_1 * long - 2.428_592_205 * medium + 0.450_593_709_9 * short,
67        0.025_904_037_1 * long + 0.782_771_766_2 * medium - 0.808_675_766 * short,
68    ]
69}
70
71/// The transform back: `Lab` to the cone response, cubed, to linear sRGB.
72fn oklab_to_rgb(lab: [f64; 3]) -> Srgb {
73    let [light, a_axis, b_axis] = lab;
74    let long = (light + 0.396_337_777_4 * a_axis + 0.215_803_757_3 * b_axis).powi(3);
75    let medium = (light - 0.105_561_345_8 * a_axis - 0.063_854_172_8 * b_axis).powi(3);
76    let short = (light - 0.089_484_177_5 * a_axis - 1.291_485_548 * b_axis).powi(3);
77    [
78        linear_to_srgb(4.076_741_662_1 * long - 3.307_711_591_3 * medium + 0.230_969_929_2 * short),
79        linear_to_srgb(
80            -1.268_438_004_6 * long + 2.609_757_401_1 * medium - 0.341_319_396_5 * short,
81        ),
82        linear_to_srgb(-0.004_196_086_3 * long - 0.703_418_614_7 * medium + 1.707_614_701 * short),
83    ]
84}

86fn lab_to_lch(lab: [f64; 3]) -> Lch {
87    let [l, a, b] = lab;
88    [l, a.hypot(b), (b.atan2(a).to_degrees() + 360.0) % 360.0]
89}
90
91fn lch_to_lab(lch: Lch) -> [f64; 3] {
92    let [l, c, h] = lch;
93    [l, c * h.to_radians().cos(), c * h.to_radians().sin()]
94}
95
96fn in_gamut(rgb: Srgb) -> bool {
97    rgb.iter().all(|v| (-0.001..=1.001).contains(v))
98}
99
100/// `rgb` in OKLCH.
101#[must_use]
102pub fn to_lch(rgb: Rgb) -> Lch {
103    lab_to_lch(rgb_to_oklab(to_srgb(rgb)))
104}
105
106/// `lch` as a colour. A hue turned often lands outside sRGB; the chroma is
107/// then reduced by bisection until it fits, keeping the hue and the
108/// lightness, which are what the eye keys on.
109#[must_use]
110pub fn from_lch(lch: Lch) -> Rgb {
111    let [l, c, h] = lch;
112    let rgb = oklab_to_rgb(lch_to_lab([l, c, h]));
113    if in_gamut(rgb) {
114        return to_rgb(rgb);
115    }
116    let (mut lo, mut hi) = (0.0, c);
117    for _ in 0..24 {
118        let mid = f64::midpoint(lo, hi);
119        if in_gamut(oklab_to_rgb(lch_to_lab([l, mid, h]))) {
120            lo = mid;
121        } else {
122            hi = mid;
123        }
124    }
125    to_rgb(oklab_to_rgb(lch_to_lab([l, lo, h])))
126}

128/// WCAG 2.x relative-luminance contrast ratio between two colours, 1 to 21.
129#[must_use]
130pub fn contrast(a: Rgb, b: Rgb) -> f64 {
131    let lum = |rgb: Rgb| {
132        let [r, g, b] = to_srgb(rgb);
133        0.2126 * srgb_to_linear(r) + 0.7152 * srgb_to_linear(g) + 0.0722 * srgb_to_linear(b)
134    };
135    let (l1, l2) = (lum(a), lum(b));
136    (l1.max(l2) + 0.05) / (l1.min(l2) + 0.05)
137}

139/// Perceptual distance: Euclidean in `OKLab`. About 0.02 is barely different;
140/// over 0.1 is clearly distinct.
141#[must_use]
142pub fn delta_e(a: Rgb, b: Rgb) -> f64 {
143    let la = rgb_to_oklab(to_srgb(a));
144    let lb = rgb_to_oklab(to_srgb(b));
145    ((la[0] - lb[0]).powi(2) + (la[1] - lb[1]).powi(2) + (la[2] - lb[2]).powi(2)).sqrt()
146}
147
148/// The shorter way round the hue circle between two hues, in degrees.
149#[must_use]
150pub fn hue_distance(a: f64, b: f64) -> f64 {
151    let d = (a - b).rem_euclid(360.0);
152    d.min(360.0 - d)
153}

155#[cfg(test)]
156mod tests {
157    use super::*;
158    use crate::theme::parse_hex;
159
160    fn hex(text: &str) -> Rgb {
161        parse_hex(text).unwrap()
162    }
163
164    #[test]
165    fn oklch_roundtrip_stays_close() {
166        for text in [
167            "#ff0000", "#00ff00", "#0000ff", "#808080", "#bf616a", "#123456",
168        ] {
169            let rgb = hex(text);
170            let back = from_lch(to_lch(rgb));
171            assert!(delta_e(rgb, back) < 0.01, "{text} -> {back:?} drifted");
172        }
173    }
174
175    #[test]
176    fn known_oklch_values() {
177        // White: L=1, C~0. Black: L=0.
178        let white = to_lch([0xff, 0xff, 0xff]);
179        assert!(
180            (white[0] - 1.0).abs() < 0.01 && white[1] < 0.01,
181            "{white:?}"
182        );
183        let black = to_lch([0, 0, 0]);
184        assert!(black[0].abs() < 0.01, "{black:?}");
185        // sRGB red: Ottosson's reference gives L~0.628, C~0.258, h~29.2.
186        let red = to_lch([0xff, 0, 0]);
187        assert!((red[0] - 0.628).abs() < 0.01, "{red:?}");
188        assert!((red[1] - 0.258).abs() < 0.01, "{red:?}");
189        assert!((red[2] - 29.23).abs() < 1.0, "{red:?}");
190    }
191
192    #[test]
193    fn gamut_fit_preserves_hue_and_lightness() {
194        // Chroma 0.4 at this hue is far outside sRGB; the fit must land in
195        // gamut with hue and lightness intact.
196        let lch = to_lch(from_lch([0.6, 0.4, 145.0]));
197        assert!((lch[0] - 0.6).abs() < 0.02, "lightness held: {lch:?}");
198        assert!((lch[2] - 145.0).abs() < 2.0, "hue held: {lch:?}");
199    }
200
201    #[test]
202    fn wcag_contrast_known_values() {
203        assert!((contrast([0xff; 3], [0; 3]) - 21.0).abs() < 0.01);
204        assert!((contrast([0; 3], [0xff; 3]) - 21.0).abs() < 0.01);
205        assert!((contrast([0x77; 3], [0x77; 3]) - 1.0).abs() < 0.01);
206    }
207
208    #[test]
209    fn hue_distance_goes_the_short_way_round() {
210        assert!((hue_distance(10.0, 350.0) - 20.0).abs() < 1e-9);
211        assert!((hue_distance(350.0, 10.0) - 20.0).abs() < 1e-9);
212        assert!((hue_distance(90.0, 270.0) - 180.0).abs() < 1e-9);
213    }

1//! The effect lists as their widgets show them, whatever draws them: the
2//! effects of a list that are up, one row a name, the player's own timers
3//! filed with them; and the colour an effect's bar wears for the spell
4//! circle its spell is in.
5//!
6//! **Circles.** A Minor Spirit spell's bar one colour, a Wizard spell's
7//! another, so a glance at a list tells which circles are up. The colours
8//! are the character's, one for every effect list, kept with the layout and
9//! shown on each list's page as a table of circle and colour. Hydra starts
10//! with a palette over the circles a character casts from; a row removed
11//! leaves its circle in the list's own colour, and the table emptied
12//! leaves every bar so. An effect's spell is found by the effect's name, as
13//! the spell table names it; an effect that is no spell, or a tracked
14//! effect given a colour of its own, is not coloured by its circle.
15

26/// One row of a list: an effect that is up, and its time left. A timer of
27/// the player's own is a row too, made an effect of its list.
28#[derive(Clone, Debug, PartialEq, Eq)]
29pub struct Up<'a> {
30    /// The list it is in.
31    pub category: Category,
32    /// The effect, as the game lists it or as a timer is made one.
33    pub effect: Cow<'a, Effect>,
34    /// Its seconds left, when the game's clock says.
35    pub left: Option<u32>,
36}
37
38/// The player's own timers in `list` that are running at the game's
39/// second now: each as an effect of the list `game_id`, filled by how much
40/// of it is left, and its seconds left.
41#[must_use]
42pub fn own_timers(state: &GameState, list: List, game_id: &str) -> Vec<(Effect, u32)> {
43    let Some(now) = state.game_time_now() else {
44        return Vec::new();
45    };
46    state
47        .custom_timers
48        .timers()
49        .filter(|(_, timer)| timer.list == list)
50        .filter_map(|(name, timer)| {
51            let left = timer.ends.checked_sub(now).filter(|left| *left > 0)?;
52            let whole = timer
53                .set_at
54                .and_then(|set_at| timer.ends.checked_sub(set_at))
55                .filter(|whole| *whole >= left);
56            let percent = whole.map_or(100, |whole| left * 100 / whole);
57            Some((
58                Effect {
59                    category: game_id.to_owned(),
60                    text: name.to_owned(),
61                    ends_at: Some(timer.ends),
62                    percent,
63                },
64                left,
65            ))
66        })
67        .collect()
68}

70/// The effects of `category` that are up, one per name: the one with the
71/// most time left, in the game's order; the player's own timers filed with
72/// the list after the game's. One still listed but run out by the clock is
73/// down, as Missing has it.
74#[must_use]
75pub fn up(state: &GameState, category: Category) -> Vec<Up<'_>> {
76    let now = state.game_time_now();
77    let mut rows: Vec<Up<'_>> = Vec::new();
78    for (id, effect) in state.effects.in_category(category.game_id()) {
79        if now.is_some_and(|now| state.effects.active(id, now) == Some(false)) {
80            continue;
81        }
82        let left = now.and_then(|now| state.effects.remaining(id, now));
83        match rows.iter_mut().find(|row| row.effect.text == effect.text) {
84            Some(row) if left > row.left => {
85                row.effect = Cow::Borrowed(effect);
86                row.left = left;
87            }
88            Some(_) => {}
89            None => rows.push(Up {
90                category,
91                effect: Cow::Borrowed(effect),
92                left,
93            }),
94        }
95    }
96    let list = List::named(category.game_id()).unwrap_or_default();
97    for (effect, left) in own_timers(state, list, category.game_id()) {
98        rows.push(Up {
99            category,
100            effect: Cow::Owned(effect),
101            left: Some(left),
102        });
103    }
104    rows
105}

113/// The palette Hydra starts with: the circles a character casts from,
114/// each a hue of its own.
115#[must_use]

118        (1, [0x8f, 0xb8, 0xde]),  // Minor Spirit: pale blue
119        (2, [0x4f, 0x7f, 0xc0]),  // Major Spirit: blue
120        (3, [0xe0, 0xd0, 0x80]),  // Cleric: gold
121        (4, [0x9f, 0xd4, 0x9f]),  // Minor Elemental: pale green
122        (5, [0x4f, 0xa8, 0x5f]),  // Major Elemental: green
123        (6, [0x8f, 0x6f, 0x3f]),  // Ranger: bark
124        (7, [0x9f, 0x5f, 0xbf]),  // Sorcerer: violet
125        (9, [0xd0, 0x70, 0x40]),  // Wizard: flame
126        (10, [0xe0, 0x90, 0xc0]), // Bard: rose
127        (11, [0x60, 0xc0, 0xc0]), // Empath: teal
128        (12, [0xa0, 0xa0, 0xe0]), // Minor Mental: lavender
129        (16, [0xf0, 0xf0, 0xf0]), // Paladin: white

139/// The colour `effect` wears for its spell's circle, when the effect names
140/// a spell whose circle has one. A percent or count after the name,
141/// `Resist Nature (50%)`, is not part of the name.
142#[must_use]
143pub fn circle_color(circles: &Circles, effect: &str) -> Option<Rgb> {
144    let name = effect.split(" (").next().unwrap_or(effect).trim();
145    let spell = spells::spell_named(name)?;
146    circles.get(&spell.circle()).copied()
147}

149/// What a player calls circle `circle`: its name, and the hundreds its
150/// spells are numbered in: `Minor Spirit (100s)`, `Council of Light
151/// (9900s)`.
152#[must_use]
153pub fn circle_called(circle: u16) -> String {
154    let hundreds = u32::from(circle) * 100;
155    spells::circle_name(circle).map_or_else(
156        || format!("{hundreds}s"),
157        |name| format!("{name} ({hundreds}s)"),
158    )
159}

161/// Every circle the spell table names, for the table's Circle column: each
162/// its number and what a player calls it.
163fn circle_choices() -> Vec<(String, String)> {
164    (1..=99)
165        .filter(|circle| spells::circle_name(*circle).is_some())
166        .map(|circle| (circle.to_string(), circle_called(circle)))
167        .collect()
168}

170/// The circles' table row: each circle, picked from those there are, and
171/// its colour; the palette until the player chose.
172#[must_use]
173pub fn circles_row(circles: Option<&Circles>) -> Row {
174    let palette = circle_palette();
175    let now = circles.unwrap_or(&palette);
176    Row {
177        key: CIRCLES.to_owned(),
178        label: "By circle".to_owned(),
179        help: "A bar in its spell's circle's colour, when the effect is a spell of a \
180               circle listed. Shared by every effects window. A tracked effect's own \
181               colour comes first; a circle not listed leaves the list's colour."
182            .to_owned(),
183        kind: RowKind::Table(vec![
184            Column::choice("Circle", circle_choices()),
185            Column::colour("Colour"),
186        ]),
187        value: Value::Table(
188            now.iter()
189                .map(|(circle, color)| vec![circle.to_string(), hex(*color)])
190                .collect(),
191        ),
192        here: circles.is_some(),
193        from: None,
194        needs: None,
195    }
196}

198/// The circles the table `written` says: each row a circle, by name or
199/// number, and its colour.
200///
201/// # Errors
202///
203/// Not rows of cells, a circle that is none, or a colour not `#rrggbb`.
204pub fn read_circles(written: &str) -> Result<Circles, String> {
205    let rows: Vec<Vec<String>> =
206        serde_json::from_str(written).map_err(|_| "not rows of cells".to_owned())?;
207    let mut circles = Circles::new();
208    for mut cells in rows {
209        cells.resize(2, String::new());
210        let (circle, color) = (cells[0].trim(), cells[1].trim());
211        if circle.is_empty() && color.is_empty() {
212            continue;
213        }
214        let number = circle.parse::<u16>().ok().or_else(|| {
215            (1..=99).find(|n| {
216                spells::circle_name(*n).is_some_and(|name| name.eq_ignore_ascii_case(circle))
217            })
218        });
219        let number = number.ok_or_else(|| format!("`{circle}` is not a spell circle."))?;
220        let color = crate::settings::rgb(color)
221            .ok_or_else(|| format!("`{color}` is not a colour #rrggbb."))?;
222        circles.insert(number, color);
223    }
224    Ok(circles)
225}

231    /// A spell's bar takes its circle's colour, by the effect's name
232    /// without what follows it; a tracked colour or no spell leaves none.
233    #[test]
234    fn an_effect_is_coloured_by_its_spells_circle() {
235        let circles = circle_palette();
236        assert_eq!(
237            circle_color(&circles, "Spirit Warding I"),
238            circles.get(&1).copied()
239        );
240        assert_eq!(
241            circle_color(&circles, "Elemental Defense I (50%)"),
242            circles.get(&4).copied()
243        );
244        assert_eq!(circle_color(&circles, "Stunned"), None, "not a spell");
245        assert_eq!(circle_color(&Circles::new(), "Spirit Warding I"), None);
246    }

248    /// The table reads back by name or number and refuses what is neither;
249    /// the row shows the palette until the player chose.
250    #[test]
251    fn the_table_names_circles_and_reads_them_back() {
252        let read =
253            read_circles(r##"[["Wizard","#d07040"],["1","#8fb8de"],["",""]]"##).expect("read");
254        assert_eq!(read.len(), 2);
255        assert_eq!(read.get(&9), Some(&[0xd0, 0x70, 0x40]));
256        assert!(read_err("[[\"Weather\",\"#000000\"]]").contains("spell circle"));
257        assert!(read_err("[[\"Wizard\",\"red\"]]").contains("#rrggbb"));
258        let row = circles_row(None);
259        assert!(!row.here);
260        assert!(matches!(&row.value, Value::Table(rows) if rows.iter().any(|r| r[0] == "9")));
261    }

283    /// A state whose `category` lists `names`, half full.
284    fn listing(category: Category, names: &[&str]) -> GameState {
285        let mut state = GameState::default();
286        state.effects.clear_category(category.game_id());
287        for name in names {
288            state.effects.insert(
289                format!("{}:{name}", category.game_id()),
290                Effect {
291                    category: category.game_id().to_owned(),
292                    text: (*name).to_owned(),
293                    ends_at: None,
294                    percent: 50,
295                },
296            );
297        }
298        state
299    }

301    /// A list's effects up are its rows, in the game's order, one a name;
302    /// another list's are not.
303    #[test]
304    fn a_lists_effects_up_are_its_rows() {
305        let state = listing(Category::Buffs, &["Rapid Fire", "Heroism"]);
306        let rows = up(&state, Category::Buffs);
307        let names: Vec<&str> = rows.iter().map(|row| row.effect.text.as_str()).collect();
308        assert_eq!(names.len(), 2);
309        assert!(names.contains(&"Heroism") && names.contains(&"Rapid Fire"));
310        assert!(up(&state, Category::Debuffs).is_empty());
311        assert!(
312            own_timers(&state, List::Custom, "").is_empty(),
313            "no clock, no timer"
314        );
315    }

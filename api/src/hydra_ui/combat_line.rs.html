1//! The combat reader's word on a line, as a preset the theme colours: a
2//! flare, a status landing on the character, a killing blow. The reader has
3//! already matched every line of the chunk its prompt closed
4//! (`hydra_model::state::combat::marks`); the story gives each line it
5//! named the preset of what it was, and matches nothing again.
6//!
7//! A preset carries the flare's or the status's name after a colon --
8//! `flare:acid`, `status_you:stunned` -- and is coloured as its kind
9//! ([`crate::run::preset_token`]). A killing blow -- a fatal critical hit --
10//! has a preset of its own, apart from a creature's death, since a creature
11//! immune to them dies by its death message alone: plain until a theme
12//! pins it.
13
14use std::sync::OnceLock;
15
16use hydra_model::StatusName;
17use hydra_model::state::combat::MarkKind;
18use hydra_model::state::combat::defs::flare_names;
19
20use crate::run::StyledRun;
21use crate::theme::Token;
22
23/// The preset of a flare.
24pub const FLARE: &str = "flare";
25/// The preset of a status landing on the character.
26pub const STATUS_YOU: &str = "status_you";
27/// The preset of a fatal critical hit.
28pub const FATAL: &str = "fatal_crit";
29
30/// The kinds of line named one by one, each with the token it is coloured
31/// by: a theme can give each of their names a colour of its own.
32pub const NAMED: [(&str, Token); 2] = [(FLARE, Token::Flare), (STATUS_YOU, Token::StatusYou)];
33
34/// Every name a kind of line named one by one (`kind`, one of [`NAMED`])
35/// can carry, as the combat reader's tables have them, sorted; none for
36/// another kind. Read once.
37#[must_use]
38pub fn names(kind: &str) -> &'static [&'static str] {
39    static FLARES: OnceLock<Vec<&'static str>> = OnceLock::new();
40    static STATUSES: OnceLock<Vec<&'static str>> = OnceLock::new();
41    match kind {
42        FLARE => FLARES.get_or_init(|| flare_names().into_iter().collect()),
43        STATUS_YOU => STATUSES.get_or_init(|| {
44            let mut names: Vec<&str> = StatusName::ALL
45                .into_iter()
46                .map(StatusName::as_str)
47                .collect();
48            names.sort_unstable();
49            names
50        }),
51        _ => &[],
52    }
53}
54
55/// The preset a line the combat reader marked takes.
56#[must_use]
57pub fn preset(kind: &MarkKind) -> String {
58    match kind {
59        MarkKind::Flare(name) => format!("{FLARE}:{name}"),
60        MarkKind::StatusYou(name) => format!("{STATUS_YOU}:{name}"),
61        MarkKind::Fatal => FATAL.to_owned(),
62    }
63}
64
65/// Mark `line` as the combat reader found it was: every run of it without a
66/// preset of its own takes `kind`'s.
67pub fn mark(line: &mut [StyledRun], kind: &MarkKind) {
68    let preset = preset(kind);
69    for run in line.iter_mut().filter(|run| run.preset.is_none()) {
70        run.preset = Some(preset.clone());
71    }
72}

74#[cfg(test)]
75mod tests {
76    use super::*;
77
78    /// Each kind's preset is coloured as its kind, whatever it is named.
79    #[test]
80    fn each_mark_is_coloured_as_its_kind() {
81        for (kind, token) in [
82            (MarkKind::Flare("acid".to_owned()), Token::Flare),
83            (MarkKind::StatusYou("stunned"), Token::StatusYou),
84            (MarkKind::Fatal, Token::FatalCrit),
85        ] {
86            assert_eq!(
87                crate::run::preset_token(&preset(&kind)),
88                Some(token),
89                "{kind:?}"
90            );
91        }
92        assert_eq!(preset(&MarkKind::Flare("acid".to_owned())), "flare:acid");
93    }
94
95    /// A named line takes the colour the theme gives its name, else its
96    /// kind's; a link in it keeps the link's.
97    #[test]
98    fn a_named_line_takes_its_names_colour_over_its_kinds() {
99        use crate::run::run_rgb;
100        use crate::theme::Palette;
101        let mut palette = Palette::bare();
102        palette.set_names([("flare:acid".to_owned(), [0x00, 0xff, 0x00])].into());
103        let marked = |name: &str| {
104            let mut line = vec![StyledRun {
105                text: "** Your bow flares! **".to_owned(),
106                ..StyledRun::default()
107            }];
108            mark(&mut line, &MarkKind::Flare(name.to_owned()));
109            line.remove(0)
110        };
111        assert_eq!(run_rgb(&marked("acid"), &palette), Some([0x00, 0xff, 0x00]));
112        assert_eq!(
113            run_rgb(&marked("fire"), &palette),
114            Some(palette.get(Token::Flare))
115        );
116        let linked = StyledRun {
117            link: Some(crate::RunLink::Command {
118                command: "look".to_owned(),
119            }),
120            ..marked("acid")
121        };
122        assert_eq!(
123            run_rgb(&linked, &palette),
124            Some(palette.get(Token::Command))
125        );
126    }
127
128    /// Each named kind lists its names from the tables, sorted; another
129    /// kind lists none.
130    #[test]
131    fn each_named_kind_lists_its_names() {
132        for (kind, _) in NAMED {
133            let listed = names(kind);
134            assert!(listed.is_sorted(), "{kind}");
135            assert!(!listed.is_empty(), "{kind}");
136        }
137        assert!(names(FLARE).contains(&"acid"));
138        assert!(names(STATUS_YOU).contains(&"stunned"));
139        assert!(names("speech").is_empty());
140    }

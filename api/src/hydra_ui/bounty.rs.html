1//! A bounty in brief: the task the guild gave, in up to four short lines --
2//! what kind, what to get or kill, and where -- for a bounty window kept
3//! small. The lines are minibounty.lic's, read from the task the model has
4//! already classified.
5

34/// `name` with a long creature name in it made short.
35fn short(name: &str) -> String {
36    SHORT_NAMES
37        .iter()
38        .find(|(long, _)| name.contains(long))
39        .map_or_else(
40            || name.to_owned(),
41            |(long, brief)| name.replace(long, brief),
42        )
43}
44
45/// `word` with its first letter a capital.
46fn capital(word: &str) -> String {
47    let mut letters = word.chars();
48    letters.next().map_or_else(String::new, |first| {
49        first.to_uppercase().chain(letters).collect()
50    })
51}

53/// `task` in brief, one to four lines.
54#[must_use]
55pub fn brief(task: &Task) -> Vec<String> {
56    let said = |key: &str| task.requirement(key).unwrap_or_default().to_owned();
57    let creature = || short(&said("creature"));
58    let area = || said("area");
59    let count = || said("number");
60    // Who the task is to help, when it is to help another adventurer.
61    let helping = |otherwise: &str| match task.requirement("assist") {
62        Some(name) => format!("Help {name}"),
63        None => otherwise.to_owned(),
64    };
65    let lines: Vec<String> = match task.kind {
66        TaskKind::None => vec!["No task currently".to_owned()],
67        TaskKind::Taskmaster => vec!["Turn in task at AG".to_owned()],
68        TaskKind::Failed => vec!["FAILED BOUNTY!!".to_owned()],
69        TaskKind::RescueSpawned => vec!["Return Child to Guard".to_owned()],
70        TaskKind::Guard => vec!["Visit Guard".to_owned(), "Report Success".to_owned()],
71        TaskKind::HeirloomFound => vec!["Visit Guard".to_owned(), "Return Heirloom".to_owned()],
72        TaskKind::BanditAssignment => vec!["Visit Guard".to_owned(), "Bandits".to_owned()],
73        TaskKind::CreatureAssignment => vec!["Visit Guard".to_owned(), "Creatures".to_owned()],
74        TaskKind::RescueAssignment => vec!["Visit Guard".to_owned(), "Rescue Kid".to_owned()],
75        TaskKind::HeirloomAssignment => {
76            vec!["Visit Guard".to_owned(), "Heirloom Task".to_owned()]
77        }
78        TaskKind::SkinAssignment => {
79            vec!["Visit Furrier".to_owned(), "Skinning Bounty".to_owned()]
80        }
81        TaskKind::GemAssignment => vec!["Visit Gemshop".to_owned(), "Gem Bounty".to_owned()],
82        TaskKind::HerbAssignment => {
83            // The healer, the herbalist or the alchemist, though the guild
84            // may send you to one's assistant.
85            let role = said("npc_role");
86            let who = ["healer", "herbalist", "alchemist"]
87                .into_iter()
88                .find(|who| role.contains(who))
89                .map_or_else(|| capital(&role), capital);
90            vec![format!("Visit {who}"), "Foraging Bounty".to_owned()]
91        }
92        TaskKind::Rescue => vec!["Rescue Child Task".to_owned(), creature(), area()],
93        TaskKind::Skin => {
94            let quality: String = said("quality").chars().take(4).collect();
95            vec![
96                "Skinning Task".to_owned(),
97                format!("{} {} ({quality})", count(), said("skin")),
98                creature(),
99            ]
100        }
101        TaskKind::Gem => vec![
102            "Gem Task".to_owned(),
103            format!("{} {}", count(), said("gem")),
104        ],
105        TaskKind::Herb => vec![
106            "Foraging Task".to_owned(),
107            format!("{} ({})", said("herb"), count()),
108            area(),
109        ],
110        TaskKind::Bandit => vec![
111            helping("Bandit Task"),
112            format!("{} Bandits", count()),
113            area(),
114        ],
115        TaskKind::Cull => vec![
116            helping("Culling Task"),
117            format!("{} {}", count(), creature()),
118            area(),
119        ],
120        TaskKind::Dangerous => vec!["Dangerous Creature Task".to_owned(), creature(), area()],
121        TaskKind::DangerousSpawned => vec!["Found Dangerous".to_owned(), creature(), area()],
122        TaskKind::Heirloom => {
123            let item = said("item");
124            let item = item.rsplit(' ').next().unwrap_or_default();
125            vec![
126                format!("{} Heirloom", said("action").to_uppercase()),
127                creature(),
128                format!("{item} ({})", said("initials")),
129                area(),
130            ]
131        }
132        TaskKind::Escort => vec![
133            "Escort Task".to_owned(),
134            format!("Escort to {}", said("destination")),
135        ],
136    };
137    lines
138        .into_iter()
139        .filter(|line| !line.trim().is_empty())
140        .collect()
141}

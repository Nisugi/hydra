1//! The room as its widgets show it, whatever draws it: its name and
2//! number, its description with what is here run on after it, who else is
3//! here, the ways out; the creatures to target, the others, and who is
4//! fighting.
5//!
6//! What the game wrote comes as runs, its links kept; each creature, object
7//! and player named apart is a link to it by its id, in the colour it is
8//! given. Which creatures are targets, what is said of each, and which side
9//! each fighter is on are read here off the model, so every front end lists

22/// The room's name without the number the game adds to it when the player
23/// has room numbers shown: `Rawknuckle's, Watering Hole - 7503251`.
24#[must_use]
25pub fn bare(title: &str) -> &str {
26    match title.rsplit_once(" - ") {
27        Some((name, number))
28            if !number.is_empty() && number.chars().all(|c| c.is_ascii_digit()) =>
29        {
30            name
31        }
32        _ => title,
33    }
34}

36/// The room's name as the game's room window heads it, with its number:
37/// `[Rawknuckle's, Watering Hole] (7503251)`.
38#[must_use]
39pub fn heading(room: &Room) -> String {
40    let title = room.title.as_deref().map_or("Room unknown", bare);
41    let number = room
42        .id
43        .as_deref()
44        .map_or_else(String::new, |id| format!(" ({id})"));
45    format!("[{title}]{number}")
46}

48/// One of the game's components as runs; `creatures`, its bold runs in
49/// the creatures' look, as the game's bold marks them.
50#[must_use]
51pub fn component(runs: &Runs, creatures: bool) -> Vec<StyledRun> {
52    runs.runs
53        .iter()
54        .map(|run| {
55            let bold = run.style.bold_depth > 0;
56            StyledRun {
57                text: run.text.clone(),
58                bold,
59                monospace: run.style.mono,
60                preset: if creatures && bold {
61                    Some("monsterbold".to_owned())
62                } else {
63                    run.style.preset.clone()
64                },
65                link: run.link.as_ref().and_then(RunLink::of),
66                ..StyledRun::default()
67            }
68        })
69        .collect()
70}

72/// The room's description, when `description`, with what is here run on
73/// after it, when `here`, as the game joins them: empty when neither is
74/// known.
75#[must_use]
76pub fn prose(room: &Room, (description, here): (bool, bool)) -> Vec<StyledRun> {
77    let mut runs: Vec<StyledRun> = Vec::new();
78    if description && let Some(description) = &room.description {
79        runs.extend(component(description, false));
80    }
81    if here && let Some(seen) = room.component("room objs") {
82        if !runs.is_empty() {
83            runs.push(StyledRun::plain("  "));
84        }
85        runs.extend(component(seen, true));
86    }
87    runs
88}

116/// The tag `.targetid` gives the creature `id` in `state`, in the look
117/// `tags`, when it shows them and the creature has one.
118#[must_use]
119pub fn tag(state: &GameState, id: &str, tags: Option<Tags>) -> Option<String> {
120    let style = tags?;
121    let creature = state.creatures().get(id.parse().ok()?)?;
122    let (mark, health) = hydra_model::targetid::of(creature)?;
123    Some(hydra_model::targetid::tag(mark, style, Some(health)))
124}

135/// A creature or an object in the room, in `color`, a link to it; its tag
136/// after it when `tags` (`.targetid`, in the state given), and what it is
137/// doing when the game says (`dead`, `lying down`), each in `aside`.
138#[must_use]
139pub fn item_runs(
140    item: &RoomItem,
141    color: Rgb,
142    tags: Option<(&GameState, Tags)>,
143    aside: Option<Rgb>,
144) -> Vec<StyledRun> {
145    let mut runs = vec![StyledRun {
146        text: item.text.clone(),
147        color: Some(hex(color)),
148        link: Some(object(item)),
149        ..StyledRun::default()
150    }];
151    let marked = tags.and_then(|(state, style)| tag(state, &item.id, Some(style)));
152    if let Some(tag) = marked {
153        runs.push(said_aside(format!(" ({tag})"), aside));
154    }
155    if let Some(status) = &item.status {
156        runs.push(said_aside(format!(" ({status})"), aside));
157    }
158    runs
159}

172/// Who else is here, `Also here: ` and each a link to them, in `color`.
173#[must_use]
174pub fn players(room: &Room, color: Rgb) -> Vec<StyledRun> {
175    let mut runs = vec![StyledRun::plain("Also here: ")];
176    for (at, player) in room.players.iter().enumerate() {
177        if at > 0 {
178            runs.push(StyledRun::plain(", "));
179        }
180        runs.extend(player_runs(player, color));
181    }
182    runs
183}

194/// The ways out of a room, as a widget shows them.
195#[derive(Clone, Debug, PartialEq, Eq)]
196pub enum WaysOut {
197    /// As the game gave them, or as the model holds them when it sent no
198    /// line of them, each a link that goes there.
199    Runs(Vec<StyledRun>),
200    /// The room has none: [`NO_EXITS`].
201    None,
202    /// Not known yet.
203    Unknown,
204}
205
206/// The ways out of `room`, as the game gave them, or as the model holds
207/// them.
208#[must_use]
209pub fn ways_out(room: Option<&Room>) -> WaysOut {
210    let Some(room) = room else {
211        return WaysOut::Unknown;
212    };
213    match (room.component("room exits"), &room.exits) {
214        (Some(exits), _) => WaysOut::Runs(component(exits, false)),
215        (None, Some(exits)) if exits.is_empty() => WaysOut::None,
216        (None, Some(exits)) => {
217            let mut runs = vec![StyledRun::plain("Obvious exits: ")];
218            for (at, exit) in exits.iter().enumerate() {
219                if at > 0 {
220                    runs.push(StyledRun::plain(", "));
221                }
222                runs.push(StyledRun {
223                    text: exit.clone(),
224                    link: Some(RunLink::Command {
225                        command: exit.clone(),
226                    }),
227                    ..StyledRun::default()
228                });
229            }
230            WaysOut::Runs(runs)
231        }
232        (None, None) => WaysOut::Unknown,
233    }
234}

236/// Whether `item` was hostile when the game first said, whatever it says
237/// now: a sympathied creature still is, and is the Targets list's.
238#[must_use]
239pub fn hostile(state: &GameState, item: &RoomItem) -> bool {
240    item.id
241        .parse()
242        .ok()
243        .and_then(|id| state.creatures().get(id))
244        .and_then(CreatureInstance::hostile_when_first_seen)
245        == Some(true)
246}

266/// A creature's health, 0-100, when the game states it.
267#[must_use]
268pub fn health(creature: &CreatureInstance) -> Option<u32> {
269    creature.hp_is_stated().then(|| {
270        #[allow(
271            clippy::cast_possible_truncation,
272            clippy::cast_sign_loss,
273            reason = "rounded and clamped to 0-100 in the same expression"
274        )]
275        let percent = creature.hp_percent().clamp(0.0, 100.0).round() as u32;
276        percent
277    })
278}

280/// The palette's colour for a creature's health at `percent`: health
281/// above half, warning to a quarter, the wound colour below.
282#[must_use]
283pub fn health_band(percent: u32) -> Token {
284    if percent > 50 {
285        Token::Health
286    } else if percent > 25 {
287        Token::Warning
288    } else {
289        Token::Wrong
290    }
291}

293/// One creature as the Targets list shows it.
294#[derive(Clone, Debug, PartialEq, Eq)]
295pub struct Target {
296    /// Named as the room's line names it, `a kobold`, article and all,
297    /// where it does; by its link's words alone otherwise.
298    pub name: String,
299    /// Its id, which its link is to.
300    pub id: i64,
301    /// Its noun, when the game named one.
302    pub noun: Option<String>,
303    /// Its tag (`.targetid`), when shown.
304    pub tag: Option<String>,
305    /// The one targeted now.
306    pub current: bool,
307    /// One the bestiary calls a boss.
308    pub boss: bool,
309    /// What the bestiary says of it, for a hint and its entry: its
310    /// template's name, level and hit points.
311    pub codex: Option<(String, Option<i32>, Option<String>)>,
312    /// Its statuses, each as `shown` said it.
313    pub statuses: Vec<String>,
314    /// Its health, 0-100, when the game states it.
315    pub health: Option<u32>,
316}

318/// The targets in the room, in the game's order -- the creatures hostile
319/// when first seen, less the dead, the decoys and the bare appendages (an
320/// arm, a claw, a tentacle) -- and how many appendages were left out; each
321/// status said as `shown` says it. `None` until the game has said what is
322/// here.
323#[must_use]
324pub fn targets(
325    state: &GameState,
326    tags: Option<Tags>,
327    shown: &dyn Fn(StatusName) -> String,
328) -> Option<(Vec<Target>, usize)> {
329    state.room.component("room objs")?;
330    let now = state.game_time_now();
331    let current = state.targeting.current();
332    let mut arms = 0;
333    let mut listed = Vec::new();
334    for creature in state.creatures().in_room() {
335        if creature.hostile_when_first_seen() != Some(true) {
336            continue;
337        }
338        if creature.appendage() {
339            arms += 1;
340            continue;
341        }
342        if !creature.valid_target() {
343            continue;
344        }
345        let tag = tags.and_then(|style| {
346            let (mark, health) = hydra_model::targetid::of(creature)?;
347            Some(hydra_model::targetid::tag(mark, style, Some(health)))
348        });
349        let said = state
350            .room
351            .creatures
352            .iter()
353            .find(|item| item.id.parse::<i64>().ok() == Some(creature.id));
354        listed.push(Target {
355            name: said.map_or_else(|| creature.name.clone(), |item| item.text.clone()),
356            id: creature.id,
357            noun: creature.noun.clone(),
358            tag,
359            current: current == Some(creature.id),
360            boss: creature.is_boss(),
361            codex: creature.template().map(|template| {
362                (
363                    template.name.clone(),
364                    template.level,
365                    template.max_hp.map(|hp| match hp {
366                        hydra_model::state::creature::Stat::Exact(hp) => hp.to_string(),
367                        hydra_model::state::creature::Stat::Range(low, high) => {
368                            format!("{low}-{high}")
369                        }
370                    }),
371                )
372            }),
373            statuses: creature.statuses(now).into_iter().map(shown).collect(),
374            health: health(creature),
375        });
376    }
377    Some((listed, arms))
378}

380/// One creature as the combat list shows it: read off the model's
381/// creature, so the list is drawn from what it says and nothing else.
382#[derive(Clone, Debug, PartialEq, Eq)]
383pub struct Fighter {
384    /// Its name.
385    pub name: String,
386    /// Its id, which its link is to.
387    pub id: i64,
388    /// Its noun, when the game named one.
389    pub noun: Option<String>,
390    /// Whether it is on the character's side.
391    pub friend: bool,
392    /// Whether it is dead.
393    pub dead: bool,
394    /// Its statuses, in words.
395    pub statuses: Vec<String>,
396    /// Its health, 0-100, when the server states it.
397    pub health: Option<u32>,
398}
399
400impl Fighter {
401    /// `creature` as the list shows it at `now`. A creature the game has
402    /// not said is friendly is a foe.
403    #[must_use]
404    pub fn of(creature: &CreatureInstance, now: Option<u32>) -> Self {
405        Self {
406            name: creature.name.clone(),
407            id: creature.id,
408            noun: creature.noun.clone(),
409            friend: creature.hostile() == Some(false),
410            dead: creature.dead(),
411            statuses: creature
412                .statuses(now)
413                .into_iter()
414                .map(status_word)
415                .collect(),
416            health: health(creature),
417        }
418    }
419}

421/// Everyone fighting in the room, in the game's order.
422#[must_use]
423pub fn fighters(state: &GameState) -> Vec<Fighter> {
424    let now = state.game_time_now();
425    state
426        .creatures()
427        .in_room()
428        .map(|creature| Fighter::of(creature, now))
429        .collect()
430}

440    /// The fixture's room: its name without the number, its number after;
441    /// what is here run on; the ways out as the game gave them.
442    #[test]
443    fn the_room_is_said_as_the_game_joins_it() {
444        let state = crate::fixture::snapshot().state;
445        assert_eq!(
446            bare("Rawknuckle's, Watering Hole - 7503251"),
447            "Rawknuckle's, Watering Hole"
448        );
449        assert_eq!(bare("Town Square - East"), "Town Square - East");
450        assert!(heading(&state.room).starts_with("[Rawknuckle's, Watering Hole]"));
451        assert_eq!(
452            text(&prose(&state.room, (false, true))),
453            "You also see a kobold"
454        );
455        let Some(kobold) = prose(&state.room, (true, true))
456            .into_iter()
457            .find(|run| run.bold)
458        else {
459            panic!("the kobold");
460        };
461        assert_eq!(kobold.preset.as_deref(), Some("monsterbold"));
462        match ways_out(Some(&state.room)) {
463            WaysOut::Runs(runs) => assert_eq!(text(&runs), "Obvious exits: north, out"),
464            other => panic!("{other:?}"),
465        }
466        assert_eq!(ways_out(None), WaysOut::Unknown);
467        assert_eq!(text(&players(&state.room, [1, 2, 3])), "Also here: Maravel");
468    }

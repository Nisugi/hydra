10/// Text and the game's structural styling. A preset is the game's own name
11/// for a look (`speech`, `roomName`), never a colour: the frontend picks the
12/// colour.
13#[derive(Clone, Debug, Default, PartialEq, Eq)]
14pub struct StyledRun {
15    /// The text.
16    pub text: String,
17    /// Inside at least one `<pushBold>`: the game's mark on a creature.
18    pub bold: bool,
19    /// Inside `<output class="mono">`: a table the game lined up.
20    pub monospace: bool,
21    /// The innermost `<preset id=>` or `<style id=>`.
22    pub preset: Option<String>,
23    /// A colour of its own, `#rrggbb`, over its preset's: one a widget
24    /// gives a name it lists (a creature, a player), or a highlight's paint
25    /// ([`crate::highlight`]).
26    pub color: Option<String>,
27    /// A ground of its own, `#rrggbb`: a highlight's.
28    pub background: Option<String>,
29    /// What a click on it does, when the game made it a link.
30    pub link: Option<RunLink>,
31}
32
33impl StyledRun {
34    /// `text` with no styling.
35    #[must_use]
36    pub fn plain(text: &str) -> Self {
37        Self {
38            text: text.to_owned(),
39            ..Self::default()
40        }
41    }
42
43    /// The game's prompt, `text`: drawn a character at a time in the
44    /// theme's prompt colours ([`prompt_coloured`]).
45    #[must_use]
46    pub fn prompt(text: &str) -> Self {
47        Self {
48            text: text.to_owned(),
49            preset: Some(PROMPT.to_owned()),
50            ..Self::default()
51        }
52    }
53
54    /// Whether it is the game's prompt ([`StyledRun::prompt`]).
55    #[must_use]
56    pub fn is_prompt(&self) -> bool {
57        self.preset.as_deref() == Some(PROMPT)
58    }
59}

64/// `runs` as they are drawn: each prompt among them a run a character,
65/// each character in its look from `palette`, one with none faint
66/// (`Token::Muted`). Borrowed when there is no prompt.
67#[must_use]
68pub fn prompt_coloured<'a>(
69    runs: &'a [StyledRun],
70    palette: &crate::theme::Palette,
71) -> std::borrow::Cow<'a, [StyledRun]> {
72    if !runs.iter().any(StyledRun::is_prompt) {
73        return std::borrow::Cow::Borrowed(runs);
74    }
75    let faint = crate::theme::hex(palette.get(crate::theme::Token::Muted));
76    let mut out = Vec::with_capacity(runs.len() + 4);
77    for run in runs {
78        if !run.is_prompt() {
79            out.push(run.clone());
80            continue;
81        }
82        for character in run.text.chars() {
83            let look = palette.prompt(character).unwrap_or_default();
84            out.push(StyledRun {
85                text: character.to_string(),
86                preset: None,
87                color: Some(look.fg.map_or_else(|| faint.clone(), crate::theme::hex)),
88                background: look.bg.map(crate::theme::hex),
89                ..run.clone()
90            });
91        }
92    }
93    std::borrow::Cow::Owned(out)
94}

96/// What a click on a run does: the game's `<a>` and `<d>` links as a
97/// frontend acts on them. The outermost link is the one a click takes.
98#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
99#[serde(tag = "kind", rename_all = "snake_case")]
100pub enum RunLink {
101    /// `<a exist= noun=>`: a game object. A click asks the game for its
102    /// menu, `_menu #<exist>`; or, when `coord` names a command in the
103    /// dictionary, sends that command.
104    Object {
105        /// Its id.
106        exist: String,
107        /// Its noun, which a menu command's `@` becomes.
108        noun: String,
109        /// `coord=`, when the link names its own command.
110        #[serde(default, skip_serializing_if = "Option::is_none")]
111        coord: Option<String>,
112    },
113    /// `<d cmd=>`, or a bare `<d>`: a click sends `command`.
114    Command {
115        /// What a click sends.
116        command: String,
117    },
118    /// A web address, which a click opens.
119    Url {
120        /// Where.
121        href: String,
122    },
123}
124
125impl RunLink {
126    /// What a run's `link` does when clicked; `None` for one that does
127    /// nothing (`LinkKind::NotActionable`).
128    #[must_use]
129    pub fn of(link: &Link) -> Option<Self> {
130        match &link.kind {
131            LinkKind::Exist { id, noun } => Some(Self::Object {
132                exist: id.clone(),
133                noun: noun.clone(),
134                coord: link.coord.clone(),
135            }),
136            // A web address alone: anything else handed to the system's
137            // opener could open a file or a program on a click.
138            LinkKind::Url { href } if is_web(href) => Some(Self::Url { href: href.clone() }),
139            LinkKind::Direct { .. } | LinkKind::DirectText => {
140                link.command().map(|command| Self::Command {
141                    command: command.to_owned(),
142                })
143            }
144            LinkKind::Url { .. } | LinkKind::NotActionable => None,
145        }
146    }
147}

149/// Which colour of the theme's draws `run`, when its own does not: the
150/// creature colour for a bold one (the game's bold marks a creature, as the
151/// `bold` preset colours it), else a link's ([`Token::Link`], or
152/// [`Token::Command`] for one that sends a command), else its preset's
153/// ([`preset_token`]); `None` for the text colour. A link comes before a
154/// preset so that one in a room's description or in speech is seen as a
155/// link.
156#[must_use]
157pub fn run_token(run: &StyledRun) -> Option<Token> {
158    if run.bold {
159        return Some(Token::Creature);
160    }
161    if let Some(link) = &run.link {
162        return Some(match link {
163            RunLink::Command { .. } => Token::Command,
164            _ => Token::Link,
165        });
166    }
167    run.preset.as_deref().and_then(preset_token)
168}

170/// `run`'s colour in `palette`: its token's ([`run_token`]), or, for a line
171/// named one by one (`flare:acid`), the colour the theme gives its name
172/// over its kind's; `None` for the text colour.
173#[must_use]
174pub fn run_rgb(run: &StyledRun, palette: &Palette) -> Option<Rgb> {
175    let token = run_token(run)?;
176    let by_preset = !run.bold && run.link.is_none();
177    by_preset
178        .then(|| {
179            run.preset
180                .as_deref()
181                .and_then(|preset| palette.named(preset))
182        })
183        .flatten()
184        .or_else(|| Some(palette.get(token)))
185}

191/// The theme's colour for a game preset; `None` for one it does not
192/// colour. Wrayth's presets, and the `roomDesc` style; `bold` is Wrayth's
193/// name for what the wire calls `monsterbold`.
194#[must_use]
195pub fn preset_token(name: &str) -> Option<Token> {
196    // A kind of line named further after a colon -- `flare:acid` -- is
197    // coloured as its kind.
198    let name = name.split_once(':').map_or(name, |(kind, _)| kind);
199    Some(match name {
200        "roomName" => Token::RoomName,
201        "roomDesc" => Token::RoomDesc,
202        "monsterbold" | "bold" => Token::Creature,
203        "speech" => Token::Speech,
204        "whisper" => Token::Whisper,
205        "thought" => Token::Thought,
206        "watching" => Token::Watching,
207        "command" => Token::Command,
208        "link" | "links" => Token::Link,
209        "selectedLink" => Token::SelectedLink,
210        crate::creature_line::DEATH => Token::CreatureDeath,
211        crate::creature_line::PREP => Token::CreaturePrep,
212        crate::creature_line::TRIGGER => Token::CreatureTrigger,
213        crate::creature_line::RECOVER => Token::CreatureRecover,
214        crate::combat_line::FLARE => Token::Flare,
215        crate::combat_line::STATUS_YOU => Token::StatusYou,
216        crate::combat_line::FATAL => Token::FatalCrit,
217        crate::lookout_line::PRESET => Token::Lookout,
218        TIMESTAMP => Token::Timestamp,
219        _ => return crate::channel_line::token(name),
220    })
221}

223/// A line the game gave to send on a click, checked: one line, no control
224/// characters, and not marked as Hydra's or Lich's (`symbols`), which a
225/// line from the game -- or from anyone between it and here, the stream
226/// being plain text -- must not be able to run. Its own `_` commands
227/// (`_menu`, `_drag`) are the game's, and go.
228///
229/// # Errors
230///
231/// Why it is not sent, for the player.
232pub fn game_line(line: &str, (hydra, lich): (char, char)) -> Result<&str, String> {
233    if line.chars().any(char::is_control) {
234        return Err(format!("the game's {line:?} is more than one line."));
235    }
236    let start = line.trim_start().chars().next();
237    if start == Some(hydra) || start == Some(lich) {
238        return Err(format!(
239            "the game's {line:?} is marked as Hydra's or Lich's, which it cannot run."
240        ));
241    }
242    Ok(line)
243}

245/// The game's runs as a frontend draws them.
246#[must_use]
247pub fn styled(runs: &[Run]) -> Vec<StyledRun> {
248    runs.iter()
249        .map(|run| StyledRun {
250            text: run.text.clone(),
251            bold: run.style.bold_depth > 0,
252            monospace: run.style.mono,
253            preset: run.style.preset.clone(),
254            color: None,
255            background: None,
256            link: run.link.as_ref().and_then(RunLink::of),
257        })
258        .collect()
259}

261/// Whether `href` is a web address, `http` or `https`, whatever its case:
262/// the only kind a click hands to the system's opener.
263#[must_use]
264pub fn is_web(href: &str) -> bool {
265    let href = href.trim_start();
266    ["http://", "https://"].iter().any(|scheme| {
267        href.get(..scheme.len())
268            .is_some_and(|start| start.eq_ignore_ascii_case(scheme))
269    })
270}

276    /// A prompt is drawn a character at a time in the theme's prompt
277    /// colours, one the theme gives none faint; other runs as they were.
278    #[test]
279    fn a_prompt_is_coloured_a_character_at_a_time() {
280        let palette = crate::theme::Palette::bare();
281        let runs = [StyledRun::plain("[7:08] "), StyledRun::prompt("R>")];
282        let drawn = prompt_coloured(&runs, &palette);
283        let texts: Vec<&str> = drawn.iter().map(|run| run.text.as_str()).collect();
284        assert_eq!(texts, ["[7:08] ", "R", ">"]);
285        let wrong = crate::theme::hex(palette.get(crate::theme::Token::Wrong));
286        let muted = crate::theme::hex(palette.get(crate::theme::Token::Muted));
287        assert_eq!(drawn[1].color.as_deref(), Some(wrong.as_str()));
288        assert_eq!(drawn[2].color.as_deref(), Some(muted.as_str()));
289        assert!(drawn[0].color.is_none(), "not a prompt: as it was");
290        let plain = [StyledRun::plain("no prompt")];
291        assert!(matches!(
292            prompt_coloured(&plain, &palette),
293            std::borrow::Cow::Borrowed(_)
294        ));
295    }

305    /// Each kind of link as a click acts on it: a command written in the
306    /// link or its own text, a web address; one that does nothing is none.
307    #[test]
308    fn each_link_does_what_a_click_on_it_should() {
309        let command = |command: &str| {
310            Some(RunLink::Command {
311                command: command.to_owned(),
312            })
313        };
314        let direct = LinkKind::Direct {
315            cmd: "go north".to_owned(),
316        };
317        assert_eq!(RunLink::of(&link(direct, "north")), command("go north"));
318        assert_eq!(
319            RunLink::of(&link(LinkKind::DirectText, "look")),
320            command("look")
321        );
322        let url = LinkKind::Url {
323            href: "https://play.net".to_owned(),
324        };
325        assert_eq!(
326            RunLink::of(&link(url, "play.net")),
327            Some(RunLink::Url {
328                href: "https://play.net".to_owned()
329            })
330        );
331        for href in ["file:///C:/Windows/notepad.exe", "C:/x.exe", "javascript:x"] {
332            let other = LinkKind::Url {
333                href: href.to_owned(),
334            };
335            assert_eq!(
336                RunLink::of(&link(other, "x")),
337                None,
338                "{href} is no web address"
339            );
340        }
341        let shouted = LinkKind::Url {
342            href: "HTTPS://play.net".to_owned(),
343        };
344        assert!(RunLink::of(&link(shouted, "play.net")).is_some());
345        assert_eq!(RunLink::of(&link(LinkKind::NotActionable, "Maravel")), None);
346        let kobold = LinkKind::Exist {
347            id: "123".to_owned(),
348            noun: "kobold".to_owned(),
349        };
350        assert_eq!(
351            RunLink::of(&link(kobold, "a kobold")),
352            Some(RunLink::Object {
353                exist: "123".to_owned(),
354                noun: "kobold".to_owned(),
355                coord: None,
356            })
357        );
358    }

1//! A dialog's controls placed as the game lays them out: each control's own
2//! `top` and `left` taken from where its `align` says (`nw` from the left,
3//! `n` from the middle, `ne` from the right), then its anchors followed
4//! against the controls they name: `anchor_top` hangs it below one,
5//! `anchor_left` puts it right of one, and with `anchor_right` too it
6//! stretches between two. A control that names a column with `anchor_left`
7//! and no `anchor_top` flows down that column, below the column's controls
8//! that are placed by their own top.
9//!
10//! The places are in the game's pixels, on a canvas the width of the
11//! game's side panels; the windows scale them to their width, the terminal
12//! turns them into rows and their order.
13//!
14//! ```text
15//! <label id='ublog' value='Today:' anchor_left='ubinjury' align='n' top='4' left='5'/>
16//! <label id='ublogv' value='1,234' anchor_left='ublog' top='4' left='0'/>
17//! <label id='ubhour' value='AVG/Hr:' anchor_left='ubinjury' anchor_top='ublog' top='2'/>
18//! ```
19
20use hydra_model::state::dialogs::{Dialog, Part};
21
22/// The width of the game's side panels, in pixels: what `n` and `ne` are
23/// measured across, unless the controls reach past it.
24pub const CANVAS_WIDTH: f32 = 190.0;
25
26/// A place as left, top, width and height, in pixels.
27pub type Place = (f32, f32, f32, f32);
28
29/// Each control placed, as its kind and its place, and the width and height
30/// they reach to.
31pub type Placed<K> = (Vec<(K, Place)>, (f32, f32));
32
33/// Where a control's tag says it goes: its offsets, size and alignment, and
34/// the controls it hangs from.
35#[derive(Clone, Debug, Default, PartialEq, Eq)]
36pub struct Hints {
37    /// Its offset down, in pixels; up when negative.
38    pub top: Option<i32>,
39    /// Its offset across, in pixels; left when negative.
40    pub left: Option<i32>,
41    /// Its width, in pixels.
42    pub width: Option<u16>,
43    /// Its height, in pixels.
44    pub height: Option<u16>,
45    /// Which point of the dialog it is placed from: `n`, `nw`, `ne`, `e`,
46    /// `w`, `c`, `s`, `sw` or `se`.
47    pub align: Option<String>,
48    /// The control it is placed below, by id.
49    pub anchor_top: Option<String>,
50    /// The control it is placed right of, by id.
51    pub anchor_left: Option<String>,
52    /// The control it is placed left of, by id.
53    pub anchor_right: Option<String>,
54}
55
56impl Hints {
57    /// The hints a tag's attributes give; none when they give none.
58    #[must_use]
59    pub fn from_attrs(attrs: &[(String, String)]) -> Option<Self> {
60        let named = |name: &str| {
61            attrs
62                .iter()
63                .find(|(key, _)| key == name)
64                .map(|(_, value)| value.as_str())
65                .filter(|value| !value.is_empty())
66        };
67        let hints = Self {
68            top: named("top").and_then(|value| value.parse().ok()),
69            left: named("left").and_then(|value| value.parse().ok()),
70            width: named("width").and_then(|value| value.parse().ok()),
71            height: named("height").and_then(|value| value.parse().ok()),
72            align: named("align").map(str::to_owned),
73            anchor_top: named("anchor_top").map(str::to_owned),
74            anchor_left: named("anchor_left").map(str::to_owned),
75            anchor_right: named("anchor_right").map(str::to_owned),
76        };
77        (!hints.is_empty()).then_some(hints)
78    }
79
80    /// Whether it gives no hint at all.
81    #[must_use]
82    pub fn is_empty(&self) -> bool {
83        *self == Self::default()
84    }
85
86    /// Whether it says where the control goes, not only its size.
87    #[must_use]
88    pub fn places(&self) -> bool {
89        self.top.is_some()
90            || self.left.is_some()
91            || self.anchor_top.is_some()
92            || self.anchor_left.is_some()
93            || self.anchor_right.is_some()
94    }
95}

97/// One control to be placed: its id, its hints, what it is to its owner,
98/// and its place before it is placed, its size the one its kind takes.
99#[derive(Clone, Debug)]
100pub struct Control<K> {
101    /// Its id, which other controls' anchors name.
102    pub id: String,
103    /// Where its tag says it goes.
104    pub hints: Option<Hints>,
105    /// What it is, to whoever draws it.
106    pub kind: K,
107    /// Its place so far.
108    pub place: Place,
109}
110
111/// The size a control of the tag `tag` takes when its tag gives none:
112/// a picture behind others takes 100 by 100, a picture that is only a
113/// point to hang others from takes nothing.
114#[must_use]
115pub fn default_size(tag: &str) -> (f32, f32) {
116    match tag {
117        "skin" => (100.0, 100.0),
118        "cmdButton" | "closeButton" | "radio" => (55.0, 20.0),
119        "dropDownBox" => (80.0, 20.0),
120        "progressBar" => (130.0, 16.0),
121        "label" => (60.0, 15.0),
122        "link" => (90.0, 20.0),
123        "upDownEditBox" => (100.0, 25.0),
124        _ => (0.0, 0.0),
125    }
126}

128/// A whole number of pixels as a float.
129#[expect(
130    clippy::cast_precision_loss,
131    reason = "a dialog's pixels are far below 2^24"
132)]
133fn pixels(value: i32) -> f32 {
134    value as f32
135}

137/// Each control placed, in the order given, with the extent they reach to;
138/// none when no control's hints say where it goes, so it is laid out in
139/// rows instead. `declared_height` is the dialog's own height when its tag
140/// gives one: `e`, `w` and `c` are measured from its middle, `s` from its
141/// foot.
142#[must_use]
143pub fn place<K>(mut controls: Vec<Control<K>>, declared_height: Option<f32>) -> Option<Placed<K>> {
144    if !controls
145        .iter()
146        .any(|control| control.hints.as_ref().is_some_and(Hints::places))
147    {
148        return None;
149    }
150    place_alone(
151        &mut controls,
152        declared_height.filter(|height| *height > 1.0),
153    );
154    let flows = flows_of(&controls);
155    settle(&mut controls, &flows);
156    let (mut right, mut foot) = (CANVAS_WIDTH, 0.0_f32);
157    for control in &controls {
158        right = right.max(control.place.0 + control.place.2);
159        foot = foot.max(control.place.1 + control.place.3);
160    }
161    let placed = controls
162        .into_iter()
163        .map(|control| (control.kind, control.place))
164        .collect();
165    Some((placed, (right + 4.0, foot + 4.0)))
166}

168/// Each control placed by its own alignment and offsets, before any anchor
169/// is followed.
170fn place_alone<K>(controls: &mut [Control<K>], declared_height: Option<f32>) {
171    for control in controls {
172        let Some(hints) = &control.hints else {
173            continue;
174        };
175        let width = hints.width.map_or(control.place.2, f32::from);
176        let height = hints.height.map_or(control.place.3, f32::from);
177        let left = pixels(hints.left.unwrap_or(0));
178        let top = pixels(hints.top.unwrap_or(0));
179        let align = hints.align.as_deref().unwrap_or("nw");
180        let x = match align {
181            "n" | "s" | "c" | "" => CANVAS_WIDTH / 2.0 - width / 2.0 + left,
182            "ne" | "e" | "se" => CANVAS_WIDTH - width - left,
183            _ => left,
184        };
185        let y = match (align, declared_height) {
186            ("e" | "w" | "c", Some(whole)) => whole / 2.0 - height / 2.0 + top,
187            ("s" | "se" | "sw", Some(whole)) => whole - height + top,
188            _ => top,
189        };
190        control.place = (x, y, width, height);
191    }
192}

194/// For each control, the control whose column it flows down: one that names
195/// a column with `anchor_left` and has no `anchor_top`, and that no other
196/// control hangs below, which would chase what hangs from it. None for the
197/// rest.
198fn flows_of<K>(controls: &[Control<K>]) -> Vec<Option<String>> {
199    let hung_from: std::collections::HashSet<&str> = controls
200        .iter()
201        .filter_map(|control| control.hints.as_ref()?.anchor_top.as_deref())
202        .collect();
203    controls
204        .iter()
205        .map(|control| {
206            let hints = control.hints.as_ref()?;
207            if hints.anchor_top.is_some()
208                || (!control.id.is_empty() && hung_from.contains(control.id.as_str()))
209            {
210                return None;
211            }
212            hints.anchor_left.clone()
213        })
214        .collect()
215}

217/// The anchors followed against the places of the controls they name, a few
218/// rounds, so a chain of anchors settles. A control that flows goes below
219/// the lowest of its column's controls that do not flow.
220fn settle<K>(controls: &mut [Control<K>], flows: &[Option<String>]) {
221    for _ in 0..4 {
222        for at in 0..controls.len() {
223            let Some(hints) = controls[at].hints.clone() else {
224                continue;
225            };
226            let find = |id: &str| {
227                controls
228                    .iter()
229                    .find(|control| !control.id.is_empty() && control.id == id)
230                    .map(|control| control.place)
231            };
232            let mut place = controls[at].place;
233            if let Some(above) = hints.anchor_top.as_deref().and_then(find) {
234                place.1 = above.1 + above.3 + pixels(hints.top.unwrap_or(2));
235            } else if let Some(column) = flows[at].as_deref() {
236                let floor = controls
237                    .iter()
238                    .enumerate()
239                    .filter(|(other, control)| {
240                        *other != at
241                            && flows[*other].is_none()
242                            && control
243                                .hints
244                                .as_ref()
245                                .and_then(|hints| hints.anchor_left.as_deref())
246                                == Some(column)
247                    })
248                    .map(|(_, control)| control.place.1 + control.place.3)
249                    .fold(f32::MIN, f32::max);
250                if floor > f32::MIN {
251                    place.1 = floor + pixels(hints.top.unwrap_or(2));
252                }
253            }
254            let gap = pixels(hints.left.unwrap_or(2));
255            match (
256                hints.anchor_left.as_deref().and_then(find),
257                hints.anchor_right.as_deref().and_then(find),
258            ) {
259                (Some(left_of), Some(right_of)) => {
260                    place.0 = left_of.0 + left_of.2 + gap;
261                    place.2 = (right_of.0 - place.0 - 2.0).max(10.0);
262                }
263                (Some(left_of), None) => place.0 = left_of.0 + left_of.2 + gap,
264                (None, Some(right_of)) => place.0 = right_of.0 - place.2 - gap,
265                (None, None) => {}
266            }
267            controls[at].place = place;
268        }
269    }
270}

272/// A model dialog's parts as controls, each by its place in `parts`: every
273/// part, its size its tag's when its tag gives none, a bar with no hints
274/// stacked below the bars before it.
275#[must_use]
276pub fn controls_of(dialog: &Dialog) -> Vec<Control<usize>> {
277    let mut bars = 0.0;
278    dialog
279        .parts
280        .iter()
281        .enumerate()
282        .map(|(at, (id, part))| {
283            let (tag, attrs) = match part {
284                Part::Label { attrs, .. } => ("label", attrs),
285                Part::Bar { attrs, .. } => ("progressBar", attrs),
286                Part::Widget { kind, attrs } => (kind.as_str(), attrs),
287            };
288            let (width, height) = default_size(tag);
289            let top = if tag == "progressBar" {
290                let top = bars * 20.0;
291                bars += 1.0;
292                top
293            } else {
294                0.0
295            };
296            Control {
297                id: id.clone(),
298                hints: Hints::from_attrs(attrs),
299                kind: at,
300                place: (0.0, top, width, height),
301            }
302        })
303        .collect()
304}

306/// The dialog's own height, when its tag gives one.
307#[must_use]
308pub fn declared_height(dialog: &Dialog) -> Option<f32> {
309    dialog
310        .attrs
311        .iter()
312        .find(|(name, _)| name == "height")
313        .and_then(|(_, value)| value.parse::<u16>().ok())
314        .map(f32::from)
315}

321    /// A control by its id, with the hints its attributes give.
322    fn control(id: &str, size: (f32, f32), pairs: &[(&str, &str)]) -> Control<String> {
323        let attrs: Vec<(String, String)> = pairs
324            .iter()
325            .map(|(name, value)| ((*name).to_owned(), (*value).to_owned()))
326            .collect();
327        Control {
328            id: id.to_owned(),
329            hints: Hints::from_attrs(&attrs),
330            kind: id.to_owned(),
331            place: (0.0, 0.0, size.0, size.1),
332        }
333    }

335    /// A table built of anchors: names in a column right of a picture, each
336    /// value right of its name, each row below the last; and a column of
337    /// bars that flows below the names.
338    #[test]
339    fn a_table_of_anchors_is_placed_as_rows_beside_a_picture() {
340        let controls = vec![
341            control(
342                "doll",
343                (0.0, 0.0),
344                &[
345                    ("top", "5"),
346                    ("left", "5"),
347                    ("width", "100"),
348                    ("height", "150"),
349                ],
350            ),
351            control(
352                "name1",
353                (60.0, 15.0),
354                &[("anchor_left", "doll"), ("top", "4"), ("left", "5")],
355            ),
356            control(
357                "value1",
358                (60.0, 15.0),
359                &[("anchor_left", "name1"), ("top", "4"), ("left", "0")],
360            ),
361            control(
362                "name2",
363                (60.0, 15.0),
364                &[
365                    ("anchor_left", "doll"),
366                    ("anchor_top", "name1"),
367                    ("top", "2"),
368                    ("left", "5"),
369                ],
370            ),
371            control(
372                "value2",
373                (60.0, 15.0),
374                &[
375                    ("anchor_left", "name2"),
376                    ("anchor_top", "value1"),
377                    ("top", "2"),
378                    ("left", "0"),
379                ],
380            ),
381        ];
382        let (placed, _) = place(controls, None).unwrap();
383        let at = |id: &str| placed.iter().find(|(kind, _)| kind == id).unwrap().1;
384        let (doll, name1, value1, name2, value2) = (
385            at("doll"),
386            at("name1"),
387            at("value1"),
388            at("name2"),
389            at("value2"),
390        );
391        assert!(
392            (name1.0 - (doll.0 + doll.2 + 5.0)).abs() < 0.5,
393            "right of the doll"
394        );
395        assert!(
396            (value1.0 - (name1.0 + name1.2)).abs() < 0.5,
397            "right of its name"
398        );
399        assert!((value1.1 - name1.1).abs() < 0.5, "on its name's row");
400        assert!(
401            (name2.1 - (name1.1 + name1.3 + 2.0)).abs() < 0.5,
402            "below the first"
403        );
404        assert!((value2.1 - name2.1).abs() < 0.5, "on its name's row");
405    }

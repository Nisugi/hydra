11/// What an alert or a notification says: the line itself, or words of its
12/// own. `false` says nothing, as if it were not there.
13#[derive(Clone, Debug, PartialEq, Eq, Deserialize, Serialize)]
14#[serde(untagged)]
15pub enum Say {
16    /// `true`: the line.
17    Line(bool),
18    /// Its own words.
19    Words(String),
20}
21
22impl Say {
23    /// What it says of `line`; `None` for `false`.
24    #[must_use]
25    pub fn of(&self, line: &str) -> Option<String> {
26        match self {
27            Self::Line(true) => Some(line.trim().to_owned()),
28            Self::Line(false) => None,
29            Self::Words(words) => Some(words.clone()),
30        }
31    }
32}

34/// A flag or a custom status a highlight sets: by its name alone, set
35/// until cleared; or `{ name = "ready", seconds = 30 }`, set for that long;
36/// or `{ name = "ready", clear = true }`, cleared.
37#[derive(Clone, Debug, PartialEq, Eq, Deserialize, Serialize)]
38#[serde(untagged)]
39pub enum Mark {
40    /// By its name alone: set until cleared.
41    Named(String),
42    /// With how long, or that it is cleared.
43    Full {
44        /// Its name.
45        name: String,
46        /// Set for this long; none, until cleared.
47        #[serde(default, skip_serializing_if = "Option::is_none")]
48        seconds: Option<u32>,
49        /// Cleared rather than set.
50        #[serde(default, skip_serializing_if = "is_false")]
51        clear: bool,
52    },
53}
54
55impl Mark {
56    /// Its name.
57    #[must_use]
58    pub fn name(&self) -> &str {
59        match self {
60            Self::Named(name) | Self::Full { name, .. } => name.trim(),
61        }
62    }
63
64    /// How long it is set for; `None` until cleared.
65    #[must_use]
66    pub const fn seconds(&self) -> Option<u32> {
67        match self {
68            Self::Named(_) => None,
69            Self::Full { seconds, .. } => *seconds,
70        }
71    }
72
73    /// It clears, rather than sets.
74    #[must_use]
75    pub const fn clears(&self) -> bool {
76        matches!(self, Self::Full { clear: true, .. })
77    }
78}

87pub struct Rule {
88    /// The group it is listed in, and switched off with.
89    #[serde(default = "general", skip_serializing_if = "is_general")]
90    pub category: String,
91    /// The words it looks for: several with `|` between them, each
92    /// trimmed (`Nisugi | Ashryn`); one alone, as written. A `|` to look
93    /// for takes a pattern, `\|`.
94    #[serde(default, skip_serializing_if = "Option::is_none")]
95    pub text: Option<String>,
96    /// A pattern it looks for instead (Rust's `regex`: no look-around).
97    #[serde(default, skip_serializing_if = "Option::is_none")]
98    pub regex: Option<String>,
99    /// Capitals count.
100    #[serde(default, skip_serializing_if = "is_false")]
101    pub case_sensitive: bool,
102    /// The words only where they stand alone, not inside another word.
103    #[serde(default, skip_serializing_if = "is_false")]
104    pub whole_word: bool,
105    /// The colour it paints, `#rrggbb`.
106    #[serde(default, skip_serializing_if = "Option::is_none")]
107    pub color: Option<String>,
108    /// The ground it paints under, `#rrggbb`.
109    #[serde(default, skip_serializing_if = "Option::is_none")]
110    pub background: Option<String>,
111    /// It paints the whole line, not only what it found.
112    #[serde(default, skip_serializing_if = "is_false")]
113    pub line: bool,
114    /// The line is hidden.
115    #[serde(default, skip_serializing_if = "is_false")]
116    pub squelch: bool,
117    /// The words found written instead as this: `$1`, `$2` the pattern's
118    /// groups.
119    #[serde(default, skip_serializing_if = "Option::is_none")]
120    pub substitute: Option<String>,
121    /// The line sent to this stream's windows: made a stream rule when Hydra
122    /// reads the file ([`HighlightFile::redirects`]), and taken off. It does
123    /// nothing here.
124    #[serde(default, skip_serializing_if = "Option::is_none")]
125    pub redirect: Option<String>,
126    /// The line sent is kept where it was as well: made a stream rule's
127    /// with [`Self::redirect`].
128    #[serde(default, skip_serializing_if = "is_false")]
129    pub copy: bool,
130    /// A flag set or cleared: what a condition's `flag` reads.
131    #[serde(default, skip_serializing_if = "Option::is_none")]
132    pub flag: Option<Mark>,
133    /// Game commands sent: `\r` between two, `$1` and `$2` the pattern's
134    /// groups, which may fill a command's words but never choose the
135    /// command. Never Hydra's or Lich's commands.
136    #[serde(default, skip_serializing_if = "Option::is_none")]
137    pub send: Option<String>,
138    /// For one an import brought: the send the player approved. A send is
139    /// sent only when it is the one approved.
140    #[serde(default, skip_serializing_if = "Option::is_none")]
141    pub approved: Option<String>,
142    /// A custom status lit or put out: shown in the Custom statuses widget,
143    /// and read by a condition's `indicator` as the game's own are.
144    #[serde(default, skip_serializing_if = "Option::is_none")]
145    pub status: Option<Mark>,
146    /// A sound to play: a file name in the sounds folder.
147    #[serde(default, skip_serializing_if = "Option::is_none")]
148    pub sound: Option<String>,
149    /// How loud the sound plays, a percent of the window's volume: 100
150    /// unless said.
151    #[serde(default, skip_serializing_if = "Option::is_none")]
152    pub volume: Option<u8>,
153    /// A banner over the story.
154    #[serde(default, skip_serializing_if = "Option::is_none")]
155    pub alert: Option<Say>,
156    /// A notification from the computer.
157    #[serde(default, skip_serializing_if = "Option::is_none")]
158    pub notify: Option<Say>,
159    /// A banner on each phone paired for push (HydraFE).
160    #[serde(default, skip_serializing_if = "Option::is_none")]
161    pub push: Option<Say>,
162    /// Words spoken aloud by the computer's voice.
163    #[serde(default, skip_serializing_if = "Option::is_none")]
164    pub speak: Option<Say>,
165    /// A rumble of the game controller, by its pattern's name: `short`,
166    /// `long`, `double`, or one of the controller file's.
167    #[serde(default, skip_serializing_if = "Option::is_none")]
168    pub rumble: Option<String>,
169    /// A picture placed in the story after the line it found: an `https://`
170    /// address, a data URI, or a file in the pictures folder, as a script's
171    /// is (`crate::picture`).
172    #[serde(default, skip_serializing_if = "Option::is_none")]
173    pub picture: Option<String>,
174    /// Seconds before its sound, banner or notification comes again; 3
175    /// unless said.
176    #[serde(default, skip_serializing_if = "Option::is_none")]
177    pub cooldown: Option<f64>,
178    /// A timer of the player's own, by name, that its words -- the timer's
179    /// up message -- start: the Timers widget counts it down. `.timer
180    /// <name>` starts it too.
181    #[serde(default, skip_serializing_if = "Option::is_none")]
182    pub timer: Option<String>,
183    /// How long its timer runs unless the up message says: its default
184    /// duration, in seconds.
185    #[serde(default, skip_serializing_if = "Option::is_none")]
186    pub seconds: Option<u32>,
187    /// `"$1"`: the up message says how long, the number its pattern's first
188    /// group caught (`"$2"` the second's), in place of the default.
189    #[serde(default, skip_serializing_if = "Option::is_none")]
190    pub caught: Option<String>,
191    /// What a number caught is multiplied by: 5 for rounds of five seconds.
192    #[serde(default, skip_serializing_if = "Option::is_none")]
193    pub times: Option<u32>,
194    /// The timer's down message: words that end it, as `text` finds them.
195    #[serde(default, skip_serializing_if = "Option::is_none")]
196    pub down: Option<String>,
197    /// The list its timer shows in: `active spell`, `buff`, `debuff`,
198    /// `cooldown`, or `custom`, the player's own, unless said.
199    #[serde(default, skip_serializing_if = "Option::is_none")]
200    pub list: Option<String>,
201    /// What must hold of the character for it to act: a line found while
202    /// it does not hold is left as it is. With no words to look for, it is
203    /// a condition alone, acting each time it comes to hold.
204    #[serde(default, skip_serializing_if = "Option::is_none")]
205    pub when: Option<crate::condition::Condition>,
206    /// A condition alone acts again only once it has not held this many
207    /// seconds; 3 unless said.
208    #[serde(default, skip_serializing_if = "Option::is_none")]
209    pub rearm: Option<f64>,
210    /// Whose it is, by name or `GAME:Name`; none, everyone's.
211    #[serde(default, skip_serializing_if = "Vec::is_empty")]
212    pub characters: Vec<String>,
213    /// Only the lines in this stream, as they reach it: those the game
214    /// sends there, those a stream rule puts there, and the lookout's
215    /// (`lookout`); never the main window's. With no words to look for, it
216    /// takes every line there.
217    #[serde(default, skip_serializing_if = "Option::is_none")]
218    pub stream: Option<String>,
219    /// Where it came from, when an import brought it: importing that file
220    /// again replaces what it brought.
221    #[serde(default, skip_serializing_if = "Option::is_none")]
222    pub origin: Option<String>,
223    /// Switched off.
224    #[serde(default, skip_serializing_if = "is_false")]
225    pub off: bool,
226}

288/// Every highlight, by name, and the categories switched.
289#[derive(Clone, Debug, Default, PartialEq, Deserialize, Serialize)]
290#[serde(deny_unknown_fields)]
291pub struct HighlightFile {
292    /// A category switched off is `false`; one not here is on.
293    #[serde(default, skip_serializing_if = "BTreeMap::is_empty")]
294    pub categories: BTreeMap<String, bool>,
295    /// Each highlight, by its name.
296    #[serde(
297        default,
298        rename = "highlight",
299        skip_serializing_if = "BTreeMap::is_empty"
300    )]
301    pub rules: BTreeMap<String, Rule>,
302}
303
304impl HighlightFile {
305    /// Read from `text`.
306    ///
307    /// # Errors
308    ///
309    /// Why it does not read.
310    pub fn parse(text: &str) -> Result<Self, String> {
311        toml::from_str(text).map_err(|why| why.to_string())
312    }
313
314    /// As the file writes it.
315    #[must_use]
316    pub fn to_toml(&self) -> String {
317        toml::to_string(self).unwrap_or_default()
318    }
319
320    /// The file in the data folder `dir`; none there, no highlights.
321    ///
322    /// # Errors
323    ///
324    /// It is there and does not read.
325    pub fn load(dir: &Path) -> Result<Self, String> {
326        match std::fs::read_to_string(dir.join(FILE)) {
327            Ok(text) => Self::parse(&text).map_err(|why| format!("{FILE}: {why}")),
328            Err(why) if why.kind() == std::io::ErrorKind::NotFound => Ok(Self::default()),
329            Err(why) => Err(format!("{FILE} cannot be read: {why}")),
330        }
331    }
332
333    /// Write it to the data folder `dir`, whole.
334    ///
335    /// # Errors
336    ///
337    /// Why it was not written.
338    pub fn save(&self, dir: &Path) -> Result<(), String> {
339        hydra_session::store::save_bytes(dir, &dir.join(FILE), self.to_toml().as_bytes())
340            .map_err(|why| format!("{FILE} was not written: {why}"))
341    }
342
343    /// Whether `category` is switched on.
344    #[must_use]
345    pub fn category_on(&self, category: &str) -> bool {
346        self.categories.get(category).copied().unwrap_or(true)
347    }
348
349    /// The stream rules its highlights that send their line to another
350    /// stream ([`Rule::redirect`]) are, in their names' order: each finds
351    /// what its highlight finds, capitals counting as it says, in every
352    /// stream, for every character; off while the highlight or its category
353    /// is.
354    #[must_use]
355    pub fn redirects(&self) -> Vec<hydra_session::stream_rules::Rule> {
356        self.rules
357            .iter()
358            .filter_map(|(name, rule)| {
359                let to = rule
360                    .redirect
361                    .as_deref()
362                    .map(str::trim)
363                    .filter(|to| !to.is_empty())?;
364                Some(hydra_session::stream_rules::Rule {
365                    name: name.clone(),
366                    text: None,
367                    regex: Some(super::matcher::pattern_of(rule)?),
368                    from: None,
369                    fighting: false,
370                    looting: false,
371                    marked: false,
372                    creature: Vec::new(),
373                    to: to.to_owned(),
374                    copy: rule.copy,
375                    off: rule.off || !self.category_on(&rule.category),
376                    commands: None,
377                    setup: None,
378                })
379            })
380            .collect()
381    }
382
383    /// Every highlight's [`Rule::redirect`] taken off, once
384    /// [`Self::redirects`] has made the stream rules of them: a highlight
385    /// that did nothing else goes. Whether any was taken.
386    pub fn take_redirects(&mut self) -> bool {
387        let mut taken = false;
388        self.rules.retain(|_, rule| {
389            if rule.redirect.is_none() && !rule.copy {
390                return true;
391            }
392            taken = true;
393            rule.redirect = None;
394            rule.copy = false;
395            does_more(rule)
396        });
397        taken
398    }
399}
400
401/// Whether `rule` does anything to its line or for the player but send it
402/// to another stream.
403fn does_more(rule: &Rule) -> bool {
404    rule.color.is_some()
405        || rule.background.is_some()
406        || rule.squelch
407        || rule.substitute.is_some()
408        || rule.flag.is_some()
409        || rule.status.is_some()
410        || rule.send.is_some()
411        || rule.sound.is_some()
412        || rule.alert.is_some()
413        || rule.notify.is_some()
414        || rule.push.is_some()
415        || rule.speak.is_some()
416        || rule.rumble.is_some()
417        || rule.picture.is_some()
418        || rule.timer.is_some()
419}

421#[cfg(test)]
422mod tests {
423    use super::*;
424
425    /// A highlight that sends its line to another stream is a stream rule:
426    /// its words found as it finds them, capitals and whole words as it
427    /// says, off with its category; taken off, a highlight that did nothing
428    /// else goes, and one that colours stays, coloured.
429    #[test]
430    fn a_redirect_is_a_stream_rule() {
431        let mut file = HighlightFile::default();
432        file.rules.insert(
433            "lnet".to_owned(),
434            Rule {
435                category: "Chat".to_owned(),
436                text: Some("[LNet] | [Merchant]".to_owned()),
437                whole_word: true,
438                redirect: Some(" lnet ".to_owned()),
439                copy: true,
440                ..Rule::default()
441            },
442        );
443        file.rules.insert(
444            "loot".to_owned(),
445            Rule {
446                regex: Some(r"\d+ silvers?".to_owned()),
447                case_sensitive: true,
448                color: Some("#ecc013".to_owned()),
449                redirect: Some("loot".to_owned()),
450                ..Rule::default()
451            },
452        );
453        file.rules.insert(
454            "plain".to_owned(),
455            Rule {
456                text: Some("Bob".to_owned()),
457                color: Some("#ffffff".to_owned()),
458                ..Rule::default()
459            },
460        );
461        file.categories.insert("Chat".to_owned(), false);
462        let made = file.redirects();
463        let regexes: Vec<(&str, Option<&str>, &str, bool, bool)> = made
464            .iter()
465            .map(|rule| {
466                (
467                    rule.name.as_str(),
468                    rule.regex.as_deref(),
469                    rule.to.as_str(),
470                    rule.copy,
471                    rule.off,
472                )
473            })
474            .collect();
475        assert_eq!(
476            regexes,
477            [
478                (
479                    "lnet",
480                    Some(r"(?i)\[LNet\]|\[Merchant\]"),
481                    "lnet",
482                    true,
483                    true
484                ),
485                ("loot", Some(r"\d+ silvers?"), "loot", false, false),
486            ]
487        );
488        assert!(made.iter().all(|rule| rule.check().is_ok()));
489        assert!(file.take_redirects());
490        assert!(!file.rules.contains_key("lnet"), "it did nothing else");
491        let loot = &file.rules["loot"];
492        assert_eq!((loot.redirect.as_deref(), loot.copy), (None, false));
493        assert_eq!(loot.color.as_deref(), Some("#ecc013"));
494        assert!(file.rules.contains_key("plain"));
495        assert!(!file.take_redirects(), "none left");
496        assert!(file.redirects().is_empty());
497    }

499    #[test]
500    fn a_file_reads_and_writes_back_what_it_says() {
501        let text = "[categories]\n\"Wrayth ignores\" = false\n\n\
502                    [highlight.Nisugi]\ncategory = \"Friends\"\ntext = \"Nisugi\"\n\
503                    color = \"#ecc013\"\ncase_sensitive = true\nalert = true\nnotify = \"Nisugi!\"\n\n\
504                    [highlight.lnet]\nregex = '^\\[LNet\\]'\nline = true\n";
505        let file = HighlightFile::parse(text).unwrap();
506        let nisugi = &file.rules["Nisugi"];
507        assert_eq!(nisugi.category, "Friends");
508        assert_eq!(nisugi.alert, Some(Say::Line(true)));
509        assert_eq!(nisugi.notify, Some(Say::Words("Nisugi!".to_owned())));
510        assert!(!nisugi.whole_word && nisugi.case_sensitive);
511        assert_eq!(file.rules["lnet"].category, "General");
512        assert!(!file.category_on("Wrayth ignores") && file.category_on("Friends"));
513        assert_eq!(HighlightFile::parse(&file.to_toml()).unwrap(), file);
514        assert!(HighlightFile::parse("[highlight.x]\nlook = 3\n").is_err());
515        assert_eq!(Say::Line(true).of("  a line "), Some("a line".to_owned()));
516        assert_eq!(Say::Line(false).of("a line"), None);
517    }

1//! How the story and a stream show their lines, with no toolkit in it: a
2//! time on each line, at its start or its end, with seconds or not, on a
3//! 12- or 24-hour clock (`XX:XX`, `XX:XX:XX`, `XX:XX:XX AM/PM`); word wrap;
4//! and the story's prompts and what the player typed, each shown or not.
5//! What a widget of lines draws is decided here, line by line ([`Prompts`]);
6//! how it is drawn is the frontend's.
7//!
8//! Hydra's own answers (`hydra_session::Notice`) are said in a token by
9//! what they are about, and a linked table as runs whose links are Hydra's
10//! commands, the game's, or a web page.
11

22/// Where a line's time goes.
23#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash, Serialize, Deserialize)]
24#[serde(rename_all = "snake_case")]
25pub enum Stamps {
26    /// No time.
27    #[default]
28    None,
29    /// Before the line: `[7:08 PM] You swing ...`.
30    Start,
31    /// After it: `You swing ... [7:08 PM]`.
32    End,
33}
34
35/// How a widget of lines draws them: the story's, or a stream's.
36#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
37#[serde(default)]
38#[expect(
39    clippy::struct_excessive_bools,
40    reason = "a page of switches: each on or off on its own, as the player picks"
41)]
42pub struct Lines {
43    /// Where each line's time goes, if anywhere.
44    pub stamps: Stamps,
45    /// A time says its seconds.
46    pub seconds: bool,
47    /// The clock a time is said on.
48    pub hours: Hours,
49    /// A long line wraps; unwrapped, the widget scrolls sideways.
50    pub wrap: bool,
51    /// The story shows the game's prompts.
52    pub prompts: bool,
53    /// The story shows what the player typed.
54    pub echo: bool,
55    /// The bounty stream's widget says the task in brief
56    /// ([`crate::bounty::brief`]), not as the game words it.
57    pub summary: bool,
58}
59
60impl Default for Lines {
61    /// As the story has always drawn: no times, wrapped, prompts and what
62    /// was typed shown.
63    fn default() -> Self {
64        Self {
65            stamps: Stamps::None,
66            seconds: false,
67            hours: Hours::Twelve,
68            wrap: true,
69            prompts: true,
70            echo: true,
71            summary: false,
72        }
73    }
74}

76impl Lines {
77    /// `text` with the time `at` where this puts it.
78    #[must_use]
79    pub fn stamped(self, at: Stamp, text: &str) -> String {
80        let said = at.said(self.seconds, self.hours);
81        match self.stamps {
82            Stamps::None => text.to_owned(),
83            Stamps::Start => format!("[{said}] {text}"),
84            Stamps::End => format!("{text} [{said}]"),
85        }
86    }
87
88    /// A line of the game's runs with the time `at` where this puts it: a
89    /// plain run of its own before them or after them, or none.
90    #[must_use]
91    pub fn stamped_runs(self, at: Stamp, runs: &[StyledRun]) -> Cow<'_, [StyledRun]> {
92        let run = |text: String| StyledRun {
93            text,
94            ..StyledRun::default()
95        };
96        let said = || at.said(self.seconds, self.hours);
97        match self.stamps {
98            Stamps::None => Cow::Borrowed(runs),
99            Stamps::Start => {
100                let mut all = vec![run(format!("[{}] ", said()))];
101                all.extend_from_slice(runs);
102                Cow::Owned(all)
103            }
104            Stamps::End => {
105                let mut all = runs.to_vec();
106                all.push(run(format!(" [{}]", said())));
107                Cow::Owned(all)
108            }
109        }
110    }
111}

113/// Which of the story's lines are drawn, walking it in order.
114///
115/// A prompt is drawn only when a line was drawn since the last one drawn, or
116/// when it changed (`R>` to `>`). It is decided while walking, where what is
117/// left out is known, because a thought whose Thoughts widget is open is not
118/// drawn, and the prompt after it would be: a lone `>` in the story for
119/// every line another widget showed.
120#[derive(Debug, Default)]
121pub struct Prompts<'a> {
122    /// A line was drawn since the last prompt drawn.
123    since: bool,
124    /// The last prompt drawn; none yet.
125    last: Option<&'a str>,
126}
127
128impl<'a> Prompts<'a> {
129    /// Whether `shown`, the story's next line, is drawn, as `options` say,
130    /// while the streams `open` are shown by widgets of their own and
131    /// Hydra's messages by a Hydra widget when `hydra`. Each line is asked
132    /// in turn, even one out of sight, so the prompts are counted right.
133    pub fn shows(
134        &mut self,
135        shown: &'a Shown,
136        (open, hydra): (&[String], bool),
137        options: Lines,
138    ) -> bool {
139        match shown {
140            Shown::Game(runs) => {
141                self.line(runs);
142                true
143            }
144            Shown::From(stream, runs) => {
145                let draws = !open.contains(stream);
146                if draws {
147                    self.line(runs);
148                }
149                draws
150            }
151            Shown::Typed { .. } => options.echo,
152            Shown::Prompt(prompt) => self.draws(prompt) && options.prompts,
153            Shown::Gap => true,
154            Shown::Hydra(_) => {
155                if !hydra {
156                    self.said();
157                }
158                !hydra
159            }
160            Shown::Said(_) => {
161                self.said();
162                true
163            }
164            // A picture a script placed takes no line of text; a line the
165            // story leaves out, none at all.
166            Shown::Picture(_) | Shown::Left(_) => false,
167        }
168    }
169
170    /// A line was drawn: it earns the next prompt if it has something to
171    /// read ([`visible`]).
172    fn line(&mut self, runs: &[StyledRun]) {
173        self.since |= visible(runs);
174    }
175
176    /// Hydra said something: that earns the next prompt as a line does.
177    fn said(&mut self) {
178        self.since = true;
179    }
180
181    /// Whether `prompt`, next, is drawn: after a line drawn, or changed.
182    fn draws(&mut self, prompt: &'a str) -> bool {
183        let changed = self.last.is_none_or(|last| last.trim() != prompt.trim());
184        let draws = self.since || changed;
185        if draws {
186            self.last = Some(prompt);
187        }
188        self.since = false;
189        draws
190    }
191}

193/// The text of a line of the story that Find looks through: one drawn --
194/// the game's, an echo, Hydra's own while the story shows them -- never a
195/// prompt. The streams `open` and `hydra` are as [`Prompts::shows`] has
196/// them.
197#[must_use]
198pub fn find_text<'a>(
199    shown: &'a Shown,
200    (open, hydra): (&[String], bool),
201    options: Lines,
202) -> Option<Cow<'a, str>> {
203    match shown {
204        Shown::Game(runs) => Some(plain(runs)),
205        Shown::From(stream, runs) if !open.contains(stream) => Some(plain(runs)),
206        Shown::Hydra(text) if !hydra => Some(Cow::Owned(format!("[Hydra] {text}"))),
207        Shown::Said(notice) => Some(Cow::Owned(notice.lines().join("\n"))),
208        Shown::Typed { prompt, line } if options.echo => {
209            Some(Cow::Owned(format!("{prompt}{line}")))
210        }
211        _ => None,
212    }
213}

215/// A line's runs as the text they say.
216#[must_use]
217pub fn plain(runs: &[StyledRun]) -> Cow<'_, str> {
218    match runs {
219        [one] => Cow::Borrowed(one.text.as_str()),
220        _ => Cow::Owned(runs.iter().map(|run| run.text.as_str()).collect()),
221    }
222}

224/// The token Hydra's answer is said in, by what it is about.
225#[must_use]
226pub fn notice_token(kind: NoticeKind) -> Token {
227    match kind {
228        NoticeKind::Error => Token::Wrong,
229        NoticeKind::Warn => Token::Warning,
230        NoticeKind::Info | NoticeKind::Debug => Token::Hydra,
231    }
232}

234/// One line of a linked table of Hydra's, `spans`, as runs in fixed width
235/// and in `color`, `#rrggbb`: Hydra's own command and the game's each a
236/// command link, a page a page's.
237#[must_use]
238pub fn notice_runs(spans: &[Span], color: &str) -> Vec<StyledRun> {
239    spans
240        .iter()
241        .map(|span| StyledRun {
242            text: span.text.clone(),
243            monospace: true,
244            color: Some(color.to_owned()),
245            link: span.link.as_ref().map(|link| match link {
246                NoticeLink::Hydra(command) | NoticeLink::Game(command) => RunLink::Command {
247                    command: command.clone(),
248                },
249                NoticeLink::Url(href) => RunLink::Url { href: href.clone() },
250            }),
251            ..StyledRun::default()
252        })
253        .collect()
254}

256/// Whether `command`, clicked in a line of a linked table, `spans`, is one
257/// of Hydra's own, sent with its symbol before it, rather than the game's.
258#[must_use]
259pub fn hydras_command(spans: &[Span], command: &str) -> bool {
260    spans
261        .iter()
262        .any(|span| matches!(&span.link, Some(NoticeLink::Hydra(own)) if own == command))

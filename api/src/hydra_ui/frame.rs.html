1//! A frame: a picture drawn round a window, a card, or a custom window's
2//! cells, composed from a library of parts by a recipe.
3//!
4//! The frames folder, `global/images/frames/` in the data folder, the tree
5//! the studio and Hydra read alike, holds the player's own library in
6//! `parts/` (`edges/`, `stickers/` and `ornaments/` of PNGs), their recipes
7//! beside it, and installed packs in `packs/<pack>/`, each a library and
8//! recipes of its own. A frame is a recipe, `<name>.toml`,
9//! naming an edge, a sticker for each kind of junction (corner, tee, cross)
10//! and any number of ornament rules, each saying which sides it goes on and
11//! where along them. A recipe finds a part in its own library first, then
12//! every pack's, then the player's own. Every part is authored for the top
13//! run (the edge horizontal, the corner sticker top-left, the tee sticker's
14//! stem pointing down) and turned for the other sides, by rotation or by
15//! mirroring as the recipe says.
16//!
17//! [`Frame::lay`] places the pieces round a set of touching cells: a run of
18//! edge along every line, drawn once where two cells share it and reaching
19//! half a thickness through each junction so the runs meet under the
20//! sticker; the sticker for each junction's kind, centred on it; and the
21//! ornaments where their rules put them. The drawing is a toolkit's: this
22//! module only says where each piece goes and which way it faces.
23

185/// What a slot does to its part.
186#[derive(Clone, Copy, Debug, Default, PartialEq)]
187pub struct Look {
188    /// A colour the part is multiplied by: a grey part recoloured.
189    pub tint: Option<Rgb>,
190    /// A glow behind it.
191    pub glow: Option<Glow>,
192    /// Light from inside it.
193    pub emissive: Option<Emissive>,
194}
195
196impl Look {
197    /// The look `written` in the recipe `name`, each colour read and each
198    /// number within its bounds.
199    fn read(name: &str, (tint, glow, emissive): Written<'_>) -> Result<Self, String> {
200        let colour = |text: &str, what: &str| {
201            parse_hex(text).ok_or_else(|| format!("{name}: {what} {text:?} is not a colour."))
202        };
203        Ok(Self {
204            tint: tint.map(|text| colour(text, "tint")).transpose()?,
205            glow: glow
206                .map(|glow| {
207                    Ok::<Glow, String>(Glow {
208                        color: colour(&glow.color, "glow colour")?,
209                        radius: glow.radius.clamp(0.0, MAX_GLOW),
210                        strength: glow.strength.unwrap_or(1.0).clamp(0.0, 1.0),
211                        pulse: glow.pulse.filter(|period| *period > 0.0),
212                    })
213                })
214                .transpose()?,
215            emissive: emissive
216                .map(|hot| {
217                    Ok::<Emissive, String>(Emissive {
218                        color: colour(&hot.color, "emissive colour")?,
219                        tolerance: hot.tolerance.unwrap_or(0.3).clamp(0.0, 1.0),
220                        radius: hot.radius.clamp(0.0, MAX_GLOW),
221                        strength: hot.strength.unwrap_or(1.0).clamp(0.0, 2.0),
222                        pulse: hot.pulse.filter(|period| *period > 0.0),
223                        cast: hot.cast.unwrap_or(0.0).clamp(0.0, MAX_GLOW),
224                    })
225                })
226                .transpose()?,
227        })
228    }
229}

524/// The arms out of a junction.
525#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
526#[allow(clippy::struct_excessive_bools)]
527struct Arms {
528    up: bool,
529    right: bool,
530    down: bool,
531    left: bool,
532}
533
534impl Arms {
535    fn count(self) -> u8 {
536        u8::from(self.up) + u8::from(self.right) + u8::from(self.down) + u8::from(self.left)
537    }
538
539    /// The junction's kind and the side it faces; none where two arms run
540    /// straight through.
541    fn junction(self) -> Option<(Junction, Side)> {
542        match self.count() {
543            2 => {
544                let side = match (self.right, self.down, self.left, self.up) {
545                    (true, true, false, false) => Side::Top,
546                    (false, true, true, false) => Side::Right,
547                    (false, false, true, true) => Side::Bottom,
548                    (true, false, false, true) => Side::Left,
549                    _ => return None,
550                };
551                Some((Junction::Corner, side))
552            }
553            3 => {
554                let side = if !self.up {
555                    Side::Top
556                } else if !self.right {
557                    Side::Right
558                } else if !self.down {
559                    Side::Bottom
560                } else {
561                    Side::Left
562                };
563                Some((Junction::Tee, side))
564            }
565            4 => Some((Junction::Cross, Side::Top)),
566            _ => None,
567        }
568    }
569}

571/// A line between two junctions, in tenths of a point, with the cells on
572/// its two sides counted.
573#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
574struct Line {
575    /// Across (constant y) or up (constant x).
576    across: bool,
577    /// Its constant coordinate.
578    at: i64,
579    /// Where it starts and ends along the other axis.
580    from: i64,
581    to: i64,
582}
583
584impl Line {
585    /// The side the line is drawn as: the outline's own side where one cell
586    /// touches it, the top or the left between two.
587    fn side(self, cells: &[Rect]) -> (Side, bool) {
588        let (mut before, mut after) = (false, false);
589        for cell in cells {
590            let (x0, x1) = (key(cell.x), key(cell.x + cell.w));
591            let (y0, y1) = (key(cell.y), key(cell.y + cell.h));
592            if self.across {
593                if self.from >= x0 && self.to <= x1 {
594                    before |= self.at == y1;
595                    after |= self.at == y0;
596                }
597            } else if self.from >= y0 && self.to <= y1 {
598                before |= self.at == x1;
599                after |= self.at == x0;
600            }
601        }
602        match (self.across, before, after) {
603            (true, false, true) => (Side::Top, true),
604            (true, true, false) => (Side::Bottom, true),
605            (true, _, _) => (Side::Top, false),
606            (false, false, true) => (Side::Left, true),
607            (false, true, false) => (Side::Right, true),
608            (false, _, _) => (Side::Left, false),
609        }
610    }
611}

613/// A coordinate quantised to a tenth of a point, so that cells whose sides
614/// coincide within rounding meet.
615#[allow(clippy::cast_possible_truncation)]
616fn key(value: f32) -> i64 {
617    (value * 10.0).round() as i64
618}

620/// A quantised coordinate back in points.
621#[allow(clippy::cast_precision_loss)]
622fn pt(key: i64) -> f32 {

626/// The size of the PNG at `path`, from its header.
627fn png_size(path: &Path) -> Result<Size, String> {
628    let bytes =
629        std::fs::read(path).map_err(|err| format!("{} unreadable: {err}", path.display()))?;
630    let signature = [0x89, b'P', b'N', b'G', 0x0d, 0x0a, 0x1a, 0x0a];
631    if bytes.len() < 24 || bytes[..8] != signature || &bytes[12..16] != b"IHDR" {
632        return Err(format!("{} is not a PNG.", path.display()));
633    }
634    let word =
635        |at: usize| u32::from_be_bytes([bytes[at], bytes[at + 1], bytes[at + 2], bytes[at + 3]]);
636    let size = Size {
637        width: word(16),
638        height: word(20),
639    };
640    if size.width == 0 || size.height == 0 {
641        return Err(format!("{} is empty.", path.display()));
642    }
643    Ok(size)
644}

718    fn read_with(path: &Path, libraries: &[PathBuf]) -> Result<Self, String> {
719        let name = path
720            .file_stem()
721            .map(|name| name.to_string_lossy().into_owned())
722            .filter(|name| !name.is_empty())
723            .ok_or_else(|| format!("{} has no name.", path.display()))?;
724        let text = std::fs::read_to_string(path)
725            .map_err(|err| format!("{} unreadable: {err}", path.display()))?;
726        let file: FrameFile =
727            toml::from_str(&text).map_err(|err| format!("{}: {err}", path.display()))?;
728        if !(0.0..=MAX_THICKNESS).contains(&file.thickness) {
729            return Err(format!(
730                "{name}: thickness {} is not within 0 to {MAX_THICKNESS}.",
731                file.thickness
732            ));
733        }
734        let mut images: Vec<Image> = Vec::new();
735        let mut find = |kind: Kind, part: &str| -> Result<usize, String> {
736            let file = format!("{part}.png");
737            let found = libraries
738                .iter()
739                .map(|library| library.join(kind.folder()).join(&file))
740                .find(|path| path.is_file());
741            let Some(path) = found else {
742                return Err(format!(
743                    "{name}: {}/{file} is in no library.",
744                    kind.folder()
745                ));
746            };
747            if let Some(at) = images.iter().position(|image| image.path == path) {
748                return Ok(at);
749            }
750            let size = png_size(&path).map_err(|err| format!("{name}: {err}"))?;
751            images.push(Image {
752                name: part.to_owned(),
753                path,
754                size,
755            });
756            Ok(images.len() - 1)
757        };
758        let look = |written: Written<'_>| Look::read(&name, written);
759        let edge = find(Kind::Edge, &file.edge.image)?;
760        let edge_look = look((
761            file.edge.tint.as_deref(),
762            file.edge.glow.as_ref(),
763            file.edge.emissive.as_ref(),
764        ))?;
765        let stickers_file = file.stickers.unwrap_or_default();
766        let sticker_size = stickers_file.size.unwrap_or(file.thickness * 1.5);
767        if sticker_size < file.thickness + 1.0 {
768            return Err(format!(
769                "{name}: sticker size {sticker_size} is under the thickness plus one."
770            ));
771        }
772        let mut stickers = [None; 3];
773        let mut sticker_looks = [Look::default(); 3];
774        for (slot, part) in [
775            stickers_file.corner.as_ref(),
776            stickers_file.tee.as_ref(),
777            stickers_file.cross.as_ref(),
778        ]
779        .into_iter()
780        .enumerate()
781        {
782            if let Some(part) = part {
783                stickers[slot] = Some(find(Kind::Sticker, part.image())?);
784                sticker_looks[slot] = look(part.look())?;
785            }
786        }
787        let mut ornaments = Vec::with_capacity(file.ornaments.len());
788        for rule in &file.ornaments {
789            ornaments.push(Ornament {
790                image: find(Kind::Ornament, &rule.image)?,
791                sides: rule.sides.clone().unwrap_or_else(|| vec![Side::Top]),
792                place: rule.place.unwrap_or_default(),
793                every: rule.every.filter(|every| *every > 0.0),
794                height: rule.height.unwrap_or(file.thickness).clamp(1.0, 64.0),
795                outer: rule.outer.unwrap_or(false),
796                look: look((
797                    rule.tint.as_deref(),
798                    rule.glow.as_ref(),
799                    rule.emissive.as_ref(),
800                ))?,
801            });
802        }
803        Ok(Self {
804            name,
805            path: path.to_path_buf(),
806            pack: Self::pack_of(path),
807            thickness: file.thickness,
808            turning: file.turn.unwrap_or_default(),
809            images,
810            edge,
811            run: file.edge.run.unwrap_or_default(),
812            edge_look,
813            sticker_size,
814            stickers,
815            sticker_looks,
816            ornaments,
817        })
818    }

917    /// The lines between junctions round `cells`, each once. A junction is
918    /// a cell's corner, so a cell's side splits only where another cell's
919    /// corner lies on it.
920    fn lines(cells: &[Rect]) -> Vec<Line> {
921        let mut corners: Vec<(i64, i64)> = Vec::new();
922        for cell in cells {
923            let (x0, x1) = (key(cell.x), key(cell.x + cell.w));
924            let (y0, y1) = (key(cell.y), key(cell.y + cell.h));
925            corners.extend([(x0, y0), (x1, y0), (x0, y1), (x1, y1)]);
926        }
927        corners.sort_unstable();
928        corners.dedup();
929        // The corners on a line at `at` along `axis`, between `lo` and `hi`.
930        let between = |across: bool, at: i64, lo: i64, hi: i64| -> Vec<i64> {
931            let mut along: Vec<i64> = corners
932                .iter()
933                .filter_map(|&(x, y)| {
934                    let (fixed, free) = if across { (y, x) } else { (x, y) };
935                    (fixed == at && free >= lo && free <= hi).then_some(free)
936                })
937                .collect();
938            along.sort_unstable();
939            along.dedup();
940            along
941        };
942        let mut lines = Vec::new();
943        for cell in cells {
944            let (x0, x1) = (key(cell.x), key(cell.x + cell.w));
945            let (y0, y1) = (key(cell.y), key(cell.y + cell.h));
946            for at in [y0, y1] {
947                for pair in between(true, at, x0, x1).windows(2) {
948                    lines.push(Line {
949                        across: true,
950                        at,
951                        from: pair[0],
952                        to: pair[1],
953                    });
954                }
955            }
956            for at in [x0, x1] {
957                for pair in between(false, at, y0, y1).windows(2) {
958                    lines.push(Line {
959                        across: false,
960                        at,
961                        from: pair[0],
962                        to: pair[1],
963                    });
964                }
965            }
966        }
967        lines.sort_unstable();
968        lines.dedup();
969        lines
970    }

972    /// The arms at every junction of `lines`, by position, top to bottom
973    /// and left to right.
974    fn junctions(lines: &[Line]) -> Vec<((i64, i64), Arms)> {
975        let mut arms: Vec<((i64, i64), Arms)> = Vec::new();
976        let mut arm = |at: (i64, i64), set: fn(&mut Arms)| {
977            if let Some((_, arms)) = arms.iter_mut().find(|(where_, _)| *where_ == at) {
978                set(arms);
979            } else {
980                let mut new = Arms::default();
981                set(&mut new);
982                arms.push((at, new));
983            }
984        };
985        for line in lines {
986            if line.across {
987                arm((line.from, line.at), |a| a.right = true);
988                arm((line.to, line.at), |a| a.left = true);
989            } else {
990                arm((line.at, line.from), |a| a.down = true);
991                arm((line.at, line.to), |a| a.up = true);
992            }
993        }
994        arms.sort_by_key(|(at, _)| (at.1, at.0));
995        arms
996    }

998    /// The ornaments of `rule` along one run of `length` points, as
999    /// distances from the run's start.
1000    fn along(&self, rule: &Ornament, length: f32, width: f32) -> Vec<f32> {
1001        let clear = self.sticker_size / 2.0 + width / 2.0;
1002        let middle = length / 2.0;
1003        if let Some(every) = rule.every {
1004            let mut spots = vec![middle];
1005            let mut step = every;
1006            while middle + step + width / 2.0 <= length - clear {
1007                spots.push(middle - step);
1008                spots.push(middle + step);
1009                step += every;
1010            }
1011            spots
1012        } else {
1013            match rule.place {
1014                Place::Middle => vec![middle],
1015                Place::Ends => {
1016                    if clear * 2.0 <= length {
1017                        vec![clear, length - clear]
1018                    } else {
1019                        Vec::new()
1020                    }
1021                }
1022            }
1023        }
1024    }

1026    /// The pieces round `cells`, touching rectangles: the runs, then the
1027    /// ornaments, then the stickers, in drawing order.
1028    #[must_use]
1029    #[allow(clippy::cast_precision_loss)]
1030    pub fn lay(&self, cells: &[Rect]) -> Vec<Piece> {
1031        let lines = Self::lines(cells);
1032        let t = self.thickness;
1033        let half = t / 2.0;
1034        let mut pieces = Vec::new();
1035        for line in &lines {
1036            let (at, from, to) = (pt(line.at), pt(line.from), pt(line.to));
1037            let (side, _) = line.side(cells);
1038            // Through each junction by half a thickness, so the runs meet.
1039            let rect = if line.across {
1040                Rect {
1041                    x: from - half,
1042                    y: at - half,
1043                    w: to - from + t,
1044                    h: t,
1045                }
1046            } else {
1047                Rect {
1048                    x: at - half,
1049                    y: from - half,
1050                    w: t,
1051                    h: to - from + t,
1052                }
1053            };
1054            pieces.push(Piece {
1055                part: Part::Edge,
1056                image: self.edge,
1057                rect,
1058                turn: self.turn(side),
1059                look: self.edge_look,
1060            });
1061        }
1062        for rule in &self.ornaments {
1063            let picture = self.images[rule.image].size;
1064            let width = rule.height * picture.width as f32 / picture.height as f32;
1065            for line in &lines {
1066                let (side, outer) = line.side(cells);
1067                if !rule.sides.contains(&side) || (rule.outer && !outer) {
1068                    continue;
1069                }
1070                let (at, from, to) = (pt(line.at), pt(line.from), pt(line.to));
1071                for spot in self.along(rule, to - from, width) {
1072                    let rect = if line.across {
1073                        Rect::centred(from + spot, at, width, rule.height)
1074                    } else {
1075                        Rect::centred(at, from + spot, rule.height, width)
1076                    };
1077                    pieces.push(Piece {
1078                        part: Part::Ornament,
1079                        image: rule.image,
1080                        rect,
1081                        turn: self.turn(side),
1082                        look: rule.look,
1083                    });
1084                }
1085            }
1086        }
1087        for ((x, y), arms) in Self::junctions(&lines) {
1088            let Some((junction, side)) = arms.junction() else {
1089                continue;
1090            };
1091            let Some(image) = self.sticker(junction) else {
1092                continue;
1093            };
1094            let turn = match junction {
1095                Junction::Corner => self.corner_turn(side),
1096                Junction::Tee => self.turn(side),
1097                Junction::Cross => Turn::AS_IS,
1098            };
1099            pieces.push(Piece {
1100                part: Part::Sticker,
1101                image,
1102                rect: Rect::centred(pt(x), pt(y), self.sticker_size, self.sticker_size),
1103                turn,
1104                look: self.sticker_looks[Self::slot(junction)],
1105            });
1106        }
1107        pieces
1108    }

1//! Wrayth's settings file, read without an XML library: the few functions
2//! the theme's preset import and the highlights' import share.
3//!
4//! A Wrayth settings file is one `<settings>` element. Real exports have no
5//! CDATA, processing instructions or doctype; every element inside `<strings>`, `<names>`, `<ignores>`,
6//! `<palette>` and `<presets>` closes itself; values are quoted, single or
7//! double. Comments are taken out first: a player who comments an entry
8//! out by hand means it gone.
9
10use std::collections::BTreeMap;
11
12/// One element's attributes, entities decoded.
13pub(crate) type Attributes = BTreeMap<String, String>;
14
15/// `xml` with its comments taken out; one never closed runs to the end.
16pub(crate) fn uncommented(xml: &str) -> String {
17    let mut out = String::with_capacity(xml.len());
18    let mut rest = xml;
19    while let Some(at) = rest.find("<!--") {
20        out.push_str(rest.get(..at).unwrap_or_default());
21        let after = rest.get(at + "<!--".len()..).unwrap_or_default();
22        rest = after
23            .find("-->")
24            .and_then(|end| after.get(end + "-->".len()..))
25            .unwrap_or_default();
26    }
27    out.push_str(rest);
28    out
29}

31/// The element `<name>` anywhere in `xml`: its own attributes, each element
32/// inside it, and how many inside it could not be read. `None` when there
33/// is none.
34pub(crate) fn section(xml: &str, name: &str) -> Option<(Attributes, Vec<Attributes>, usize)> {
35    let opener = format!("<{name}");
36    let at = xml.match_indices(&opener).map(|(at, _)| at).find(|&at| {
37        xml.get(at + opener.len()..)
38            .and_then(|after| after.chars().next())
39            .is_some_and(|c| c == '>' || c == '/' || c.is_whitespace())
40    })?;
41    let (own, closed, body_at) = element(xml, at)?;
42    if closed {
43        return Some((own, Vec::new(), 0));
44    }
45    let end = xml.get(body_at..)?.find(&format!("</{name}>"))? + body_at;
46    let body = xml.get(body_at..end)?;
47    let (mut inside, mut unread) = (Vec::new(), 0);
48    let mut from = 0;
49    while let Some(next) = body.get(from..).and_then(|rest| rest.find('<')) {
50        let at = from + next;
51        if let Some((attributes, _, after)) = element(body, at) {
52            inside.push(attributes);
53            from = after;
54        } else {
55            // One that never ends -- a quote left open -- is passed over, and
56            // the next is read from its own start: the rest are not lost.
57            unread += 1;
58            from = at + 1;
59        }
60    }
61    Some((own, inside, unread))
62}

64/// The element that opens at `at`: its attributes, whether it closes
65/// itself, and where it ends. Quotes are honoured, so a `>` inside a value
66/// does not end it.
67fn element(xml: &str, at: usize) -> Option<(Attributes, bool, usize)> {
68    let tag = xml.get(at + 1..)?;
69    let after_name = tag
70        .char_indices()
71        .skip_while(|(_, c)| !c.is_whitespace() && *c != '>' && *c != '/');
72    let mut attributes = Attributes::new();
73    let (mut key, mut value, mut quote) = (String::new(), String::new(), None);
74    let mut closed = false;
75    for (offset, c) in after_name {
76        match (quote, c) {
77            (Some(q), c) if c == q => {
78                attributes.insert(
79                    std::mem::take(&mut key),
80                    decoded(&std::mem::take(&mut value)),
81                );
82                quote = None;
83            }
84            (Some(_), c) => value.push(c),
85            (None, '"' | '\'') => quote = Some(c),
86            (None, '/') => closed = true,
87            (None, '>') => return Some((attributes, closed, at + 1 + offset + 1)),
88            (None, '=') => {}
89            (None, c) if c.is_whitespace() => closed = false,
90            (None, c) => {
91                closed = false;
92                key.push(c);
93            }
94        }
95    }
96    None
97}

99/// `text` with XML's entities made characters again; one not known is kept
100/// as written.
101fn decoded(text: &str) -> String {
102    let mut out = String::with_capacity(text.len());
103    let mut rest = text;
104    while let Some(at) = rest.find('&') {
105        out.push_str(rest.get(..at).unwrap_or_default());
106        let after = rest.get(at..).unwrap_or_default();
107        let Some(end) = after.find(';') else {
108            out.push_str(after);
109            return out;
110        };
111        let name = after.get(1..end).unwrap_or_default();
112        let character = match name {
113            "amp" => Some('&'),
114            "lt" => Some('<'),
115            "gt" => Some('>'),
116            "quot" => Some('"'),
117            "apos" => Some('\''),
118            _ => name
119                .strip_prefix("#x")
120                .map(|hex| u32::from_str_radix(hex, 16))
121                .or_else(|| name.strip_prefix('#').map(str::parse))
122                .and_then(Result::ok)
123                .and_then(char::from_u32),
124        };
125        match character {
126            Some(character) => out.push(character),
127            None => out.push_str(after.get(..=end).unwrap_or_default()),
128        }
129        rest = after.get(end + 1..).unwrap_or_default();
130    }
131    out.push_str(rest);
132    out
133}

135#[cfg(test)]
136mod tests {
137    use super::*;
138
139    /// One element with a quote left open is not read; those after it are,
140    /// and the one lost is counted.
141    #[test]
142    fn an_element_that_never_ends_does_not_lose_its_section() {
143        let xml = "<strings>
144<h text=\"one\"/>
145<h text=\"two />
146<h text=\"three\"/>
147</strings>";
148        let (_, inside, unread) = section(xml, "strings").expect("the section");
149        let texts: Vec<_> = inside.iter().filter_map(|h| h.get("text")).collect();
150        assert!(texts.contains(&&"one".to_owned()), "{texts:?}");
151        assert!(texts.contains(&&"three".to_owned()), "{texts:?}");
152        assert!(unread > 0, "the one lost is counted");
153    }
154}

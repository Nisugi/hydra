1//! The lookout's word on a line, as a preset the theme colours: a line it
2//! found new (`hydra_lookout`). The story gives it to the line where it came
3//! when the player asks for the mark, and to a line whose resolution stat
4//! is absurd always, in the story and in the `lookout` stream.
5
6use crate::run::StyledRun;
7
8/// The preset of a line the lookout found new.
9pub const PRESET: &str = "lookout";
10
11/// The stream the lookout's lines are shown in.
12pub const STREAM: &str = "lookout";
13
14/// Mark `line` as the lookout's: every run of it without a preset of its
15/// own takes [`PRESET`].
16pub fn mark(line: &mut [StyledRun]) {
17    for run in line.iter_mut().filter(|run| run.preset.is_none()) {
18        run.preset = Some(PRESET.to_owned());
19    }
20}

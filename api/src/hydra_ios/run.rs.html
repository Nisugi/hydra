1//! Hydra running inside the app: started once, on a thread of its own with
2//! its runtime, its table and its page served to the phone alone; told
3//! when the app leaves the screen and comes back to it.
4//!
5//! **Waking.** iOS freezes the app soon after it leaves the screen. The
6//! game's connections often outlive a freeze (the system keeps them open
7//! while nothing reads them), but the page's listener can be gone after a
8//! long one: on waking Hydra dials its own page, and listens again on the
9//! same port when nothing answers. The text the game sent meanwhile comes
10//! at once: it is shown and read into each character's state, but what the
11//! highlights call for is let go of for [`HUSH`], so no trigger sends late
12//! and no sound plays for old text. A walk under way reads all of it before
13//! its next step, and replans when the character is not where it expected.
14//! Each waking is logged: how long the app was away, how each character is
15//! connected, and how many lines each caught up on.
16

30/// How long a start waits for the page to be served.
31const STARTING_FOR: Duration = Duration::from_secs(30);
32
33/// How long the page's server has to say where it listens.
34const SERVED_FOR: Duration = Duration::from_secs(10);
35
36/// How long after waking what the highlights call for is let go of: the
37/// text caught up on comes in the first moments.
38const HUSH: Duration = Duration::from_secs(3);
39
40/// How long after waking the lines caught up on are counted.
41const CAUGHT_UP_AFTER: Duration = Duration::from_secs(2);
42
43/// How long a dial of the page's own port waits for an answer.
44const DIAL_FOR: Duration = Duration::from_secs(1);
45
46/// Hydra, while it runs.
47struct Running {
48    /// The page's address, its pairing in it.
49    url: String,
50    /// The page's port on this device.
51    port: u16,
52    sessions: Sessions,
53    runtime: tokio::runtime::Handle,
54    /// Told to stop: every character quits in order.
55    stop: watch::Sender<bool>,
56    thread: Option<std::thread::JoinHandle<()>>,
57    /// When the app last left the screen, until it comes back.
58    asleep_at: Option<Instant>,
59}
60
61static RUNNING: Mutex<Option<Running>> = Mutex::new(None);
62
63fn running() -> MutexGuard<'static, Option<Running>> {
64    RUNNING.lock().unwrap_or_else(PoisonError::into_inner)
65}

67/// What Hydra's thread says once it has started: the page's address, the
68/// board, and its runtime; or why it could not.
69type Started = Result<(String, Sessions, tokio::runtime::Handle), String>;

71/// Start Hydra with what it keeps under `data` (its `data` and `logs`
72/// folders), and answer the page's address once it is served: as the
73/// running one does, when one runs.
74pub(crate) fn start(data: &Path) -> Result<String, String> {
75    start_on(data, None)
76}

78/// [`start`], the page on `port` when given, else on one no one listens on.
79fn start_on(data: &Path, port: Option<u16>) -> Result<String, String> {
80    let mut running = running();
81    if let Some(now) = running.as_ref() {
82        return Ok(now.url.clone());
83    }
84    std::fs::create_dir_all(data).map_err(|why| format!("{}: {why}", data.display()))?;
85    std::env::set_current_dir(data)
86        .map_err(|why| format!("Hydra cannot run in {}: {why}", data.display()))?;
87    // Its messages kept in its folder from here on.
88    crate::log::keep();
89    let port = match port {
90        Some(port) => port,
91        None => free_port()?,
92    };
93    let (said, heard) = mpsc::channel();
94    let (stop, stopping) = watch::channel(false);
95    let thread = std::thread::Builder::new()
96        .name("hydra".to_owned())
97        .spawn(move || run(port, &said, stopping))
98        .map_err(|why| format!("Hydra's thread cannot start: {why}"))?;
99    match heard.recv_timeout(STARTING_FOR) {
100        Ok(Ok((url, sessions, runtime))) => {
101            eprintln!("[hydra] the page is served on this phone, port {port}");
102            *running = Some(Running {
103                url: url.clone(),
104                port,
105                sessions,
106                runtime,
107                stop,
108                thread: Some(thread),
109                asleep_at: None,
110            });
111            Ok(url)
112        }
113        Ok(Err(why)) => {
114            let _ = stop.send(true);
115            // Let go before the thread is waited for: as it ends it takes
116            // this lock itself, and would wait for ever on it held here.
117            drop(running);
118            let _ = thread.join();
119            Err(why)
120        }
121        Err(_) => {
122            // Told to stop, and not waited for.
123            let _ = stop.send(true);
124            Err("Hydra did not start in time".to_owned())
125        }
126    }
127}

129/// A port no one listens on, on this device.
130fn free_port() -> Result<u16, String> {
131    std::net::TcpListener::bind((Ipv4Addr::LOCALHOST, 0))
132        .and_then(|listener| listener.local_addr())
133        .map(|at| at.port())
134        .map_err(|why| format!("no port is free on this phone: {why}"))
135}

137/// Hydra's thread: the table and the page served on `port`, said on
138/// `said`, until it is stopped or shut down from its hub.
139fn run(port: u16, said: &mpsc::Sender<Started>, mut stopping: watch::Receiver<bool>) {
140    let runtime = match tokio::runtime::Builder::new_multi_thread()
141        .worker_threads(2)
142        .enable_all()
143        .build()
144    {
145        Ok(runtime) => runtime,
146        Err(why) => {
147            let _ = said.send(Err(format!("Hydra's runtime cannot start: {why}")));
148            return;
149        }
150    };
151    runtime.block_on(async {
152        let sessions = Sessions::new(Wake::default());
153        let table = Table::new(sessions.clone(), tokio::runtime::Handle::current());
154        let answering = Arc::clone(&table);
155        let control: HubControl = Arc::new(move |request| {
156            let table = Arc::clone(&answering);
157            Box::pin(async move { table.answer(request).await })
158        });
159        sessions.control(control);
160        table.offer().await;
161        let data = hydra_table::account::data_dir();
162        tokio::spawn(hydra_web::serve_to(
163            sessions.clone(),
164            data,
165            Reach::ThisDevice,
166        ));
167        sessions.want_web(Some(port));
168        match served(&sessions).await {
169            Ok(url) => {
170                let here = tokio::runtime::Handle::current();
171                let _ = said.send(Ok((url, sessions.clone(), here)));
172            }
173            Err(why) => {
174                let _ = said.send(Err(why));
175                return;
176            }
177        }
178        let mut closes = sessions.closes();
179        tokio::select! {
180            _ = stopping.wait_for(|stop| *stop) => {}
181            _ = closes.wait_for(|closing| *closing) => {}
182        }
183        eprintln!("[hydra] stopping; quitting every character");
184        table.stop_everything().await;
185    });
186    // Shut down from its hub: no longer running. Stopped from the app,
187    // the app has taken it already.
188    let mut running = running();
189    if running.as_ref().is_some_and(|now| now.port == port) {
190        *running = None;
191    }
192}

194/// The page's address once its server listens, or why it does not.
195async fn served(sessions: &Sessions) -> Result<String, String> {
196    let until = Instant::now() + SERVED_FOR;
197    while Instant::now() < until {
198        match sessions.glance(Instant::now()).web_at {
199            Some(WebAt::Listening(url)) => return Ok(url),
200            Some(WebAt::Failed(why)) => return Err(why),
201            None => tokio::time::sleep(Duration::from_millis(50)).await,
202        }
203    }
204    Err("the page was not served in time".to_owned())
205}

207/// Quit every character in order and stop, waiting until Hydra has.
208pub(crate) fn stop() {
209    let Some(mut now) = running().take() else {
210        return;
211    };
212    let _ = now.stop.send(true);
213    if let Some(thread) = now.thread.take() {
214        let _ = thread.join();
215    }
216}

218/// Whether Hydra runs, where its page is, and how many characters play.
219pub(crate) fn status() -> serde_json::Value {
220    match running().as_ref() {
221        Some(now) => serde_json::json!({
222            "running": true,
223            "url": now.url,
224            "characters": heard(&now.sessions)
225                .iter()
226                .filter(|(_, state, _)| *state != State::Closed)
227                .count(),
228        }),
229        None => serde_json::json!({ "running": false }),
230    }
231}

233/// The app is leaving the screen.
234pub(crate) fn asleep() {
235    if let Some(now) = running().as_mut() {
236        now.asleep_at = Some(Instant::now());
237        eprintln!("[hydra] asleep");
238    }
239}

241/// The app is on the screen again: the page dialled and listened for again
242/// if it does not answer, the highlights hushed while the text caught up on
243/// comes, and how it went logged.
244pub(crate) fn awake() {
245    let (away, port, sessions, runtime) = {
246        let mut running = running();
247        let Some(now) = running.as_mut() else {
248            return;
249        };
250        (
251            now.asleep_at.take().map(|at| at.elapsed()),
252            now.port,
253            now.sessions.clone(),
254            now.runtime.clone(),
255        )
256    };
257    sessions.highlights().hush_for(HUSH);
258    let page = SocketAddr::from((Ipv4Addr::LOCALHOST, port));
259    let answered = TcpStream::connect_timeout(&page, DIAL_FOR).is_ok();
260    if !answered {
261        sessions.want_web(None);
262        sessions.want_web(Some(port));
263    }
264    let before = heard(&sessions);
265    let connected: Vec<String> = before
266        .iter()
267        .map(|(name, state, _)| {
268            format!(
269                "{name} {}",
270                hydra_board::connection(*state).0.to_lowercase()
271            )
272        })
273        .collect();
274    eprintln!(
275        "[hydra] awake{}: {}; the page {}",
276        away.map_or_else(String::new, |away| format!(" after {}", long(away))),
277        if connected.is_empty() {
278            "no character".to_owned()
279        } else {
280            connected.join(", ")
281        },
282        if answered {
283            "answered"
284        } else {
285            "did not answer, and is listened for again"
286        }
287    );
288    runtime.spawn(async move {
289        tokio::time::sleep(CAUGHT_UP_AFTER).await;
290        let after = heard(&sessions);
291        let caught: Vec<String> = after
292            .iter()
293            .filter_map(|(name, _, lines)| {
294                let was = before
295                    .iter()
296                    .find(|(other, _, _)| other == name)
297                    .map_or(0, |(_, _, lines)| *lines);
298                (*lines > was).then(|| format!("{name} {} lines", lines - was))
299            })
300            .collect();
301        if !caught.is_empty() {
302            eprintln!("[hydra] caught up on waking: {}", caught.join(", "));
303        }
304    });
305}

307/// Each character seated: its name, how it is connected, and how many
308/// lines its story has heard.
309fn heard(sessions: &Sessions) -> Vec<(String, State, u64)> {
310    sessions
311        .glance(Instant::now())
312        .seats
313        .iter()
314        .map(|seat| {
315            let lines = {
316                let story = seat.story();
317                story.dropped + u64::try_from(story.lines.len()).unwrap_or(u64::MAX)
318            };
319            (seat.name.clone(), seat.lifecycle(), lines)
320        })
321        .collect()
322}

324/// `away` in words: `4m 10s`, `52s`.
325fn long(away: Duration) -> String {
326    let seconds = away.as_secs();
327    match seconds / 60 {
328        0 => format!("{seconds}s"),
329        minutes => format!("{minutes}m {}s", seconds % 60),
330    }
331}

333#[cfg(test)]
334pub(crate) mod tests {
335    use super::*;

337    /// Hydra is started by one test at a time: it is one per process.
338    pub(crate) fn one_at_a_time() -> MutexGuard<'static, ()> {
339        static TURN: Mutex<()> = Mutex::new(());
340        TURN.lock().unwrap_or_else(PoisonError::into_inner)
341    }
342
343    /// A page that cannot be served -- its port taken -- is an error said in
344    /// time, not a start that never returns; and Hydra is not left running.
345    #[test]
346    fn a_page_that_cannot_be_served_is_an_error_not_a_hang() {
347        let _turn = one_at_a_time();
348        let taken = std::net::TcpListener::bind((Ipv4Addr::LOCALHOST, 0)).unwrap();
349        let port = taken.local_addr().unwrap().port();
350        let dir = std::env::temp_dir().join(format!("hydra-ios-taken-{}", std::process::id()));
351        let (said, heard) = mpsc::channel();
352        std::thread::spawn(move || {
353            let _ = said.send(start_on(&dir, Some(port)));
354        });
355        let started = heard
356            .recv_timeout(STARTING_FOR + SERVED_FOR)
357            .expect("start returns");
358        assert!(started.is_err(), "{started:?}");
359        assert!(running().is_none(), "not left running");
360    }

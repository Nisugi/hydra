1//! The rows, as the tables hold them.
2//!
3//! Every row names whose it is ([`Who`]) and when ([`Stamp`]): the wall
4//! clock in seconds since the epoch, UTC, and the game's own clock when the
5//! prompt had said it. Days are a report's concern, counted on the game's
6//! Eastern day when asked.
7
8/// Whose row: the instance the character plays on and the character's name.
9#[derive(Debug, Clone, PartialEq, Eq, Hash)]
10pub struct Who {
11    /// The game instance, as the roster names it (`prime`, `platinum`, ...).
12    pub instance: String,
13    /// The character's name.
14    pub character: String,
15}
16
17impl Who {
18    /// A pair.
19    #[must_use]
20    pub fn new(instance: impl Into<String>, character: impl Into<String>) -> Self {
21        Self {
22            instance: instance.into(),
23            character: character.into(),
24        }
25    }
26}

28/// When a row happened.
29#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
30pub struct Stamp {
31    /// Seconds since the Unix epoch, UTC.
32    pub at: i64,
33    /// The game's clock at the prompt that closed the chunk, when known.
34    pub game_time: Option<u32>,
35}
36
37impl Stamp {
38    /// Now, by the wall clock, with the game's clock beside it when known.
39    #[must_use]
40    pub fn now(game_time: Option<u32>) -> Self {
41        let at = std::time::SystemTime::now()
42            .duration_since(std::time::UNIX_EPOCH)
43            .map_or(0, |d| i64::try_from(d.as_secs()).unwrap_or(i64::MAX));
44        Self { at, game_time }
45    }
46}

48/// How a pulse's experience came.
49#[derive(Debug, Clone, Copy, PartialEq, Eq)]
50pub enum Source {
51    /// An absorption pulse, the mind draining into experience.
52    Normal,
53    /// A bounty's reward: a task was finished within the window before it.
54    Bounty,
55    /// A brooch rubbed: the orb's experience, absorbed at once.
56    Brooch,
57    /// Absorbed at once from anything not yet named: books handed in, an
58    /// award.
59    Instant,
60    /// Typed in by hand.
61    Manual,
62}
63
64impl Source {
65    /// As the table stores it.
66    #[must_use]
67    pub const fn as_str(self) -> &'static str {
68        match self {
69            Self::Normal => "normal",
70            Self::Bounty => "bounty",
71            Self::Brooch => "brooch",
72            Self::Instant => "instant",
73            Self::Manual => "manual",
74        }
75    }
76
77    /// From the table's word; an unknown word reads as normal.
78    #[must_use]
79    pub fn parse(word: &str) -> Self {
80        match word {
81            "bounty" => Self::Bounty,
82            "brooch" => Self::Brooch,
83            "instant" => Self::Instant,
84            "manual" => Self::Manual,
85            _ => Self::Normal,
86        }
87    }
88}

242/// What a transaction was: silver banked or taken, paid away, or bounty
243/// points earned.
244#[derive(Debug, Clone, Copy, PartialEq, Eq)]
245pub enum TransactionKind {
246    /// Silver put in the bank: coins or notes handed to the teller.
247    Deposit,
248    /// Silver taken from the bank: coins handed over, or a note written.
249    Withdrawal,
250    /// Silver paid for a service: a note's surcharge, a locksmith's fee, a
251    /// creditor's bill.
252    Fee,
253    /// Silver given on top of a fee.
254    Tip,
255    /// Bounty points earned for a task done.
256    Bounty,
257    /// Silver that came to you without a sale: on a search of your own, by
258    /// the fossil charm, out of a box, with a bounty's reward.
259    Found,
260    /// Silver a shop paid for something sold.
261    Sold,
262    /// Silver a shop took for something bought.
263    Bought,
264    /// Silver handed to someone.
265    Paid,
266    /// A pile another character of the household dropped, picked up: theirs
267    /// by `found_by`, in this character's hands now; not new silver.
268    Picked,
269}
270
271impl TransactionKind {
272    /// As the table stores it.
273    #[must_use]
274    pub const fn as_str(self) -> &'static str {
275        match self {
276            Self::Deposit => "deposit",
277            Self::Withdrawal => "withdrawal",
278            Self::Fee => "fee",
279            Self::Tip => "tip",
280            Self::Bounty => "bounty",
281            Self::Found => "found",
282            Self::Sold => "sold",
283            Self::Bought => "bought",
284            Self::Paid => "paid",
285            Self::Picked => "picked",
286        }
287    }
288
289    /// From the table's word; an unknown word reads as a fee.
290    #[must_use]
291    pub fn parse(word: &str) -> Self {
292        match word {
293            "deposit" => Self::Deposit,
294            "withdrawal" => Self::Withdrawal,
295            "tip" => Self::Tip,
296            "bounty" => Self::Bounty,
297            "found" => Self::Found,
298            "sold" => Self::Sold,
299            "bought" => Self::Bought,
300            "paid" => Self::Paid,
301            "picked" => Self::Picked,
302            _ => Self::Fee,
303        }
304    }
305
306    /// Every kind, in the order a report lists them.
307    pub const ALL: [Self; 10] = [
308        Self::Found,
309        Self::Picked,
310        Self::Sold,
311        Self::Bought,
312        Self::Paid,
313        Self::Fee,
314        Self::Tip,
315        Self::Deposit,
316        Self::Withdrawal,
317        Self::Bounty,
318    ];
319
320    /// Whether the kind is silver coming in.
321    #[must_use]
322    pub const fn is_income(self) -> bool {
323        matches!(self, Self::Found | Self::Sold)
324    }
325
326    /// Whether the kind is silver going out, the bank aside.
327    #[must_use]
328    pub const fn is_outgo(self) -> bool {
329        matches!(self, Self::Bought | Self::Paid | Self::Fee | Self::Tip)
330    }
331}

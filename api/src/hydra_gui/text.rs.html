1//! A styled line as egui lays it out: each run's colour and font, and a
2//! line with links in it drawn so a click on one is the link's.
3//!
4//! A highlight's paint is laid on
5//! the runs before they come here (`hydra_ui::highlight`): its colour as a
6//! run's own, its ground as the run's background.
7
8use egui::text::{LayoutJob, TextFormat};
9use egui::{Color32, Pos2, TextStyle};
10use hydra_ui::theme::{Palette, Token};
11use hydra_ui::{RunLink, StyledRun, run_rgb};
12
13use crate::theme::rgb;

15/// `runs` as one job, in `style`'s fonts and `palette`'s colours: a run's
16/// own colour (a highlight's), else its theme colour ([`run_rgb`]), else
17/// the text colour; monospace runs in the monospace font.
18pub(crate) fn job(runs: &[StyledRun], style: &egui::Style, palette: &Palette) -> LayoutJob {
19    let body = crate::theme::story_font(style);
20    let mono = TextStyle::Monospace.resolve(style);
21    let mut job = LayoutJob::default();
22    for run in runs {
23        job.append(
24            &run.text,
25            0.0,
26            TextFormat {
27                font_id: if run.monospace {
28                    mono.clone()
29                } else {
30                    body.clone()
31                },
32                color: run
33                    .color
34                    .as_deref()
35                    .and_then(|hex| Color32::from_hex(hex).ok())
36                    .or_else(|| run_rgb(run, palette).map(rgb))
37                    .unwrap_or_else(|| style.visuals.text_color()),
38                background: run
39                    .background
40                    .as_deref()
41                    .and_then(|hex| Color32::from_hex(hex).ok())
42                    .unwrap_or(Color32::TRANSPARENT),
43                ..TextFormat::default()
44            },
45        );
46    }
47    job
48}

50/// The links in `job` to the thing a controller's interact mode has in
51/// focus on `context`'s window, marked: the job's sections are its runs.
52fn mark_focus(context: &egui::Context, job: &mut LayoutJob, runs: &[StyledRun]) {
53    let Some(focus) = crate::pad::focus_of(context) else {
54        return;
55    };
56    let accent = crate::theme::color(context, Token::Accent);
57    for (section, run) in job.sections.iter_mut().zip(runs) {
58        if matches!(&run.link, Some(RunLink::Object { exist, .. }) if *exist == focus) {
59            section.format.background = accent.gamma_multiply(0.35);
60            section.format.underline = egui::Stroke::new(1.0, accent);
61        }
62    }
63}

65/// What a line with links in it was asked this frame.
66#[derive(Clone, Debug, PartialEq)]
67pub(crate) enum Acted {
68    /// A link clicked, and where.
69    Clicked(RunLink, Pos2),
70    /// A line to send without an echo: an object carried let go on another,
71    /// `_drag #<item> #<onto>` (`carry.rs`).
72    Quietly(String),
73}

75/// A line whose runs have links in it, `job` drawn as a label is: its words
76/// selected by a drag, as any line's; a link under the pointer shows the
77/// hand, and a click on one is the link's; with the drag key held an
78/// object's link is carried from it instead (`carry.rs`), and one carried let
79/// go on another object's link goes into it. The runs are the job's, in
80/// order, their text its text.
81///
82/// The line is selectable, so a link in it neither stops the line being
83/// selected nor breaks a selection across it. A drag selects; a click,
84/// which is no drag, is still the link's.
85pub(crate) fn linked(ui: &mut egui::Ui, mut job: LayoutJob, runs: &[StyledRun]) -> Option<Acted> {
86    // The link under the pointer last frame is drawn in the `selectedLink`
87    // colour: the job's sections are its runs, one each.
88    let hovered_id = ui.next_auto_id().with("hovered-link");
89    let hovered: Option<usize> = ui.data(|data| data.get_temp(hovered_id)).flatten();
90    if let Some(section) = hovered.and_then(|at| job.sections.get_mut(at)) {
91        section.format.color = crate::theme::color(ui.ctx(), Token::SelectedLink);
92    }
93    mark_focus(ui.ctx(), &mut job, runs);
94    let carrying = crate::carry::held(ui);
95    let sense = if carrying {
96        egui::Sense::click_and_drag()
97    } else {
98        egui::Sense::click()
99    };
100    let (at, galley, response) = egui::Label::new(job)
101        .sense(sense)
102        .selectable(!carrying)
103        .layout_in_ui(ui);
104    let enabled = ui.is_enabled();
105    response
106        .widget_info(|| egui::WidgetInfo::labeled(egui::WidgetType::Label, enabled, galley.text()));
107    if ui.is_rect_visible(response.rect) {
108        let color = ui.visuals().text_color();
109        if carrying {
110            ui.painter()
111                .galley(at, std::sync::Arc::clone(&galley), color);
112        } else {
113            egui::text_selection::LabelSelectionState::label_text_selection(
114                ui,
115                &response,
116                at,
117                std::sync::Arc::clone(&galley),
118                color,
119                egui::Stroke::NONE,
120            );
121        }
122    }
123    let run_at = |pointer: Pos2| {
124        // The nearest boundary between characters, and the character under
125        // the pointer the one before it when the pointer is left of it: on
126        // the right half of a link's last letter, the boundary is past it.
127        let cursor = galley.cursor_from_pos(pointer - at);
128        let mut char_at: usize = cursor.index.into();
129        if char_at > 0 && (pointer - at).x < galley.pos_from_cursor(cursor).min.x {
130            char_at -= 1;
131        }
132        let byte = galley
133            .text()
134            .char_indices()
135            .nth(char_at)
136            .map_or(galley.text().len(), |(byte, _)| byte);
137        let mut start = 0;
138        runs.iter().position(|run| {
139            let span = start..start + run.text.len();
140            start = span.end;
141            span.contains(&byte)
142        })
143    };
144    let run_at = |pointer: Pos2| run_at(pointer).map(|at| (at, &runs[at]));
145    let link_at = |pointer: Pos2| run_at(pointer).and_then(|(_, run)| run.link.clone());
146    let now = response
147        .hover_pos()
148        .and_then(run_at)
149        .filter(|(_, run)| run.link.is_some())
150        .map(|(at, _)| at);
151    if now.is_some() {
152        ui.ctx().set_cursor_icon(egui::CursorIcon::PointingHand);
153    }
154    if now != hovered {
155        ui.data_mut(|data| data.insert_temp(hovered_id, now));
156        ui.ctx().request_repaint();
157    }
158    if carrying
159        && response.drag_started()
160        && let Some(origin) = ui.input(|input| input.pointer.press_origin())
161        && let Some((_, run)) = run_at(origin)
162        && let Some(RunLink::Object { exist, .. }) = &run.link
163    {
164        response.dnd_set_drag_payload(crate::carry::Carried {
165            exist: exist.clone(),
166            name: run.text.clone(),
167        });
168    }
169    if let Some(carried) = response.dnd_release_payload::<crate::carry::Carried>() {
170        let onto = ui
171            .input(|input| input.pointer.interact_pos())
172            .and_then(link_at);
173        if let Some(RunLink::Object { exist, .. }) = onto {
174            // Into another object; onto itself, nothing.
175            return (exist != carried.exist)
176                .then(|| Acted::Quietly(format!("_drag #{} #{exist}", carried.exist)));
177        }
178        // Let go on the line's words: the window's own place takes it, the
179        // story's floor.
180        egui::DragAndDrop::set_payload(ui.ctx(), (*carried).clone());
181    }
182    if !response.clicked() {
183        return None;
184    }
185    let pointer = response.interact_pointer_pos()?;
186    link_at(pointer).map(|link| Acted::Clicked(link, pointer))
187}

189#[cfg(test)]
190mod tests {
191    use super::*;
192
193    fn run(text: &str) -> StyledRun {
194        StyledRun::plain(text)
195    }
196
197    /// A plain run is the text colour, a bold one (a creature) the
198    /// creature colour, and a game preset its own.
199    #[test]
200    fn plain_bold_and_a_preset_each_take_their_colour() {
201        let style = egui::Style::default();
202        let bold = StyledRun {
203            bold: true,
204            ..run("a kobold")
205        };
206        let spoken = StyledRun {
207            preset: Some("speech".to_owned()),
208            ..run("hi")
209        };
210        let job = job(&[run("You see "), bold, spoken], &style, &Palette::bare());
211        assert_eq!(job.text, "You see a koboldhi");
212        let colors: Vec<Color32> = job.sections.iter().map(|s| s.format.color).collect();
213        assert_eq!(
214            colors,
215            [
216                style.visuals.text_color(),
217                rgb(Palette::bare().get(Token::Creature)),
218                Color32::from_rgb(0xf0, 0xee, 0xe8)
219            ]
220        );
221    }
222
223    /// A link takes the link colour; a bold one (a creature) keeps the
224    /// creature colour.
225    #[test]
226    fn a_link_takes_the_link_colour() {
227        let style = egui::Style::default();
228        let link = Some(hydra_ui::RunLink::Command {
229            command: "go north".to_owned(),
230        });
231        let linked = |bold: bool| StyledRun {
232            bold,
233            link: link.clone(),
234            ..run("north")
235        };
236        let job = job(&[linked(false), linked(true)], &style, &Palette::bare());
237        let colors: Vec<Color32> = job.sections.iter().map(|s| s.format.color).collect();
238        assert_eq!(
239            colors,
240            [
241                rgb(Palette::bare().get(Token::Link)),
242                rgb(Palette::bare().get(Token::Creature)),
243            ]
244        );
245    }
246
247    /// A link in words the game gave a preset of their own, a room's
248    /// description or speech, is still drawn as a link; a creature's keeps
249    /// the creature colour.
250    #[test]
251    fn a_link_in_a_preset_is_still_a_link() {
252        let style = egui::Style::default();
253        let palette = Palette::bare();
254        let object = Some(hydra_ui::RunLink::Object {
255            exist: "-483342".to_owned(),
256            noun: "smithy".to_owned(),
257            coord: None,
258        });
259        let in_preset = |preset: &str, bold: bool| StyledRun {
260            preset: Some(preset.to_owned()),
261            bold,
262            link: object.clone(),
263            ..run("smithy")
264        };
265        let colors: Vec<Color32> = [
266            in_preset("roomDesc", false),
267            in_preset("speech", false),
268            in_preset("monsterbold", true),
269            in_preset("roomDesc", true),
270        ]
271        .iter()
272        .map(|run| {
273            job(std::slice::from_ref(run), &style, &palette).sections[0]
274                .format
275                .color
276        })
277        .collect();
278        assert_eq!(
279            colors,
280            [
281                rgb(palette.get(Token::Link)),
282                rgb(palette.get(Token::Link)),
283                rgb(palette.get(Token::Creature)),
284                rgb(palette.get(Token::Creature)),
285            ]
286        );
287    }

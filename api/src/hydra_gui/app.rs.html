1//! The windows: the hub, and each character's play window, on the main
2//! thread.
3//!
4//! eframe owns the main thread for as long as the windows are open, so the
5//! binary runs the sessions and their feeds on its runtime; this draws what the feeds left on the seats, sends
6//! what the player typed, and hands the hub's requests to the binary, which
7//! owns the session table ([`Sessions`]).
8//!
9//! The hub is eframe's own window. Each play window is another native window,
10//! opened when its character is seated; closing it leaves the character
11//! running, and its card brings it back. Closing the hub while other
12//! windows are open hides it, and they play on; once none is left Hydra
13//! rests, its characters playing, until the windows are asked for again
14//! (`App::close_or_hide`). Its Shut down asks first while characters
15//! play, and the binary then stops every character in order.
16//!
17//! A play window keeps its layout and the character's tracked effects in the
18//! data folder's `layouts`, by game and name; the presets every character
19//! adds from are kept there too, and so are the character's own keys. Every
20//! character's keys are the data folder's `keybinds.toml` (`keyed.rs`).
21//!
22//! Every window opens where it was last left: the data folder's
23//! `windows.json` ([`crate::placement`]).

296    /// A play window for each seat new since the last frame, open; none for
297    /// a seat gone. A seat gone has closed its window: the next of its name
298    /// -- the same character, connected again -- opens where that one was
299    /// left, not where it had opened.
300    ///
301    /// A play window opens where the character still plays and its window
302    /// is kept (`Seat::window_kept`): a character new to the board, unless
303    /// it was started headless or for the web; and, as the windows open
304    /// again, one whose window was open as they last closed. None for a
305    /// character that has quit, none the player closed: its card opens it.
306    fn seat(&mut self, seats: &[Arc<Seat>]) {
307        let placements = &mut self.placements;
308        // The windows of the characters gone, by game and name: a character
309        // reconnected is a new seat, and takes its window over as it was.
310        let gone: Vec<u32> = self
311            .characters
312            .keys()
313            .copied()
314            .filter(|id| !seats.iter().any(|seat| seat.id == *id))
315            .collect();
316        let mut left: Vec<(String, String, Play)> = Vec::new();
317        for id in gone {
318            if let Some(character) = self.characters.remove(&id) {
319                placements.closed(&placement::seat_key("play", &character.seat));
320                left.push((
321                    character.seat.game.clone(),
322                    character.seat.name.clone(),
323                    character.play,
324                ));
325            }
326        }
327        for seat in seats {
328            if self.characters.contains_key(&seat.id) {
329                continue;
330            }
331            let carried = left
332                .iter()
333                .position(|(game, name, _)| *game == seat.game && *name == seat.name)
334                .map(|at| left.swap_remove(at).2);
335            let play = match carried {
336                Some(mut play) => {
337                    play.reseat(seat.id);
338                    play
339                }
340                None => Play::new(
341                    seat.id,
342                    &seat.name,
343                    seat.instance.clone(),
344                    Some(self.layouts.clone()),
345                ),
346            };
347            self.characters.insert(
348                seat.id,
349                Character {
350                    play,
351                    seat: Arc::clone(seat),
352                    open: seat.window_kept() && seat.lifecycle() != State::Closed,
353                },
354            );
355        }
356    }

358    fn act(&mut self, context: &egui::Context, action: HubAction) {
359        match action {
360            HubAction::Show(id) => {
361                if let Some(character) = self.characters.get_mut(&id) {
362                    character.open = true;
363                    context.send_viewport_cmd_to(
364                        character.seat.viewport(),
365                        egui::ViewportCommand::Focus,
366                    );
367                }
368            }
369            HubAction::Ask(request) => self.sessions.ask(request, &self.runtime),
370            HubAction::Log(id) => {
371                if let Some(seat) = self.characters.get(&id).map(|c| Arc::clone(&c.seat)) {
372                    self.open_log(&seat);
373                }
374            }
375            HubAction::Menu(choice) => self.menu_chosen(choice, None),
376            HubAction::Update => self.sessions.fetch_update(),
377        }
378    }

393    /// What the Skill trainer widget on `seat` asked, done.
394    fn trainer_asked(&self, seats: &[Arc<Seat>], seat: &Seat, ask: crate::trainer::Ask) {
395        use crate::trainer::{Ask, Asked as For};
396        match ask {
397            Ask::Load | Ask::Web => {
398                seat.ask_goals(if ask == Ask::Load {
399                    For::Trainer
400                } else {
401                    For::Web
402                });
403                seat.send("goals");
404            }
405            Ask::Change(change) => seat.change_goals(change),
406            // Kept by the play window, which never passes one on.
407            Ask::Profile(_) => {}
408            Ask::Apply => {
409                if let Some(seat) = seats.iter().find(|each| each.id == seat.id) {
410                    seat.apply_goals(&self.runtime, self.sessions.wake());
411                }
412            }
413        }

416    /// What a play window asked for, done.
417    fn asked(&mut self, context: &egui::Context, seats: &[Arc<Seat>], seat: &Seat, asked: Asked) {
418        match asked {
419            Asked::Send(line) => {
420                if !self.window_command(seat, &line) {
421                    seat.typed(&line, false);
422                }
423            }
424            Asked::Quietly(line) => seat.send(&line),
425            Asked::FromGame(line) => {
426                let symbols = (self.symbol_of(seat), self.lich_symbol_of(seat));
427                match crate::play::game_line(&line, symbols) {
428                    Ok(line) => seat.send(line),
429                    Err(why) => {
430                        let said = hydra_session::Notice::line(
431                            hydra_session::NoticeKind::Warn,
432                            format!("Not sent: {why}"),
433                        );
434                        seat.story().push(hydra_ui::Shown::Said(said.answering()));
435                    }
436                }
437            }
438            Asked::Command(line) => seat.send(&format!("{}{line}", self.symbol_of(seat))),
439            Asked::Trainer(ask) => self.trainer_asked(seats, seat, ask),
440            Asked::SavePreset(preset) => self.presets.keep(preset),
441            Asked::ForgetPreset(name) => self.presets.forget(&name),
442            Asked::UseSet(chosen) => {
443                let done = self.key_change(seat, &KeyChange::Choose(chosen));
444                if self.settings.open {
445                    self.settings.tell(done.unwrap_or_else(|why| why));
446                }
447            }
448            Asked::Character(n) => self.bring(context, seats, n),
449            Asked::Settings(page) => {
450                let whose = format!("{}:{}", seat.game, seat.name);
451                self.settings.open_at(Some(whose), page.as_deref());
452                context.send_viewport_cmd_to(settings::viewport(), egui::ViewportCommand::Focus);
453            }
454            Asked::ReloadKeys => self.read_keys(),
455            Asked::Log => self.open_log(seat),
456            Asked::Menu(choice) => self.menu_chosen(choice, Some(seat)),
457            Asked::Zoom(by) => {
458                if let Err(why) = self
459                    .own
460                    .set_zoom(context.zoom_factor() + f32::from(by) * crate::own::ZOOM_STEP)
461                {
462                    let said = hydra_session::Notice::line(
463                        hydra_session::NoticeKind::Warn,
464                        format!("The zoom was not kept: {why}"),
465                    );
466                    seat.story().push(hydra_ui::Shown::Said(said.answering()));
467                }
468            }
469            Asked::Room { id, brief } => {
470                let lines = self
471                    .mapped
472                    .described(id, brief)
473                    .unwrap_or_else(|| vec![format!("The map has no room {id}.")]);
474                let said = hydra_session::Notice::prose(hydra_session::NoticeKind::Info, lines);
475                seat.story().push(hydra_ui::Shown::Said(said.answering()));
476            }
477        }
478    }

480    /// The hub hidden, shown again, or every window closed.
481    ///
482    /// Closing the hub while another window is open -- a play window, a
483    /// log, the settings, an editor -- hides it, and the others play on;
484    /// once the last of them closes, or the hub is closed with none open,
485    /// every window closes and Hydra rests. Shutting down closes them all.
486    /// `hydra` launched again shows the hub (`Wake::raise`).
487    ///
488    /// A window is hidden before eframe lets it go: on Windows a window let
489    /// go is destroyed only when its event loop runs again, which it does
490    /// not while Hydra rests, so one left showing would stay on the screen,
491    /// frozen.
492    fn close_or_hide(&mut self, context: &egui::Context) {
493        if self.sessions.wake().raised() {
494            self.hub_shown = HubShown::Showing;
495            context.send_viewport_cmd(egui::ViewportCommand::Visible(true));
496            context.send_viewport_cmd(egui::ViewportCommand::Minimized(false));
497            context.send_viewport_cmd(egui::ViewportCommand::Focus);
498        }
499        // A play window asked to the front, the hub left as it is.
500        for id in self.sessions.wake().raised_seats() {
501            self.act(context, HubAction::Show(id));
502        }
503        if self.hub_shown == HubShown::Hiding {
504            context.send_viewport_cmd(egui::ViewportCommand::Visible(false));
505            self.hub_shown = HubShown::Hidden;
506        }
507        let hidden = self.hub_shown == HubShown::Hidden;
508        // Every window eframe has open, the hub's own among them; and a play
509        // window wanted open, which on the windows' first frame is not yet.
510        let others = context.input(|input| input.raw.viewports.len() > 1)
511            || self.characters.values().any(|character| character.open);
512        let asked = context.input(|input| input.viewport().close_requested());
513        if self.sessions.closing() || (!others && (asked || hidden)) {
514            self.write_waiting();
515            context.send_viewport_cmd(egui::ViewportCommand::Visible(false));
516            // A close the player asked for goes ahead on its own.
517            if !asked {
518                context.send_viewport_cmd(egui::ViewportCommand::Close);
519            }
520        } else if asked {
521            self.write_waiting();
522            context.send_viewport_cmd(egui::ViewportCommand::CancelClose);
523            context.send_viewport_cmd(egui::ViewportCommand::Visible(false));
524            self.hub_shown = HubShown::Hidden;
525        }
526        if self.hub_shown == HubShown::Hidden {
527            // A frame now and then, hidden, to see the last window close.
528            context.request_repaint_after(Duration::from_secs(1));
529        }
530    }

532    /// The windows are closing: a layout change still waiting on a held
533    /// button, every window's place, and whatever else waits to be written,
534    /// written first; and which play windows are open, for the windows
535    /// opened again.
536    fn write_waiting(&mut self) {
537        for character in self.characters.values() {
538            character.seat.keep_window(character.open);
539        }
540        for character in self
541            .characters
542            .values_mut()
543            .filter(|character| character.open)
544        {
545            character.play.keep_waiting_layout();
546        }
547        self.placements.save();
548        crate::store::flush();
549    }

551    /// Each open play window, drawn in its own native window, with the
552    /// keys pressed in it done; then what each asked, done.
553    #[expect(
554        clippy::too_many_lines,
555        reason = "one window's frame in order: its keys, its view, what it asked"
556    )]
557    fn play_windows(&mut self, context: &egui::Context, seats: &[Arc<Seat>], began: Instant) {
558        // Where each is on the map, once a window has a minimap.
559        let wanted = self
560            .characters
561            .values()
562            .any(|character| character.play.maps());
563        let seen: Vec<(u32, Option<Arc<hydra_session::Snapshot>>)> = self
564            .characters
565            .values()
566            .map(|character| (character.seat.id, character.seat.snapshot()))
567            .collect();
568        let mut maps = self.mapped.views(&self.data, wanted, &seen);
569        // The route travel shows each, by the roster's name for it.
570        for character in self.characters.values() {
571            let seat = &character.seat;
572            let name = format!("{}:{}", seat.game, seat.name);
573            match maps.get_mut(&seat.id) {
574                Some(hydra_map::MinimapView::Here { route, .. }) => {
575                    *route = hydra_map::shown_route(&name);
576                }
577                // A course is sailed out: in port again, it comes off.
578                Some(hydra_map::MinimapView::Ocean {
579                    sailing: hydra_map::Sailing::Docked { .. },
580                    ..
581                }) => hydra_map::show_course(&name, None),
582                Some(hydra_map::MinimapView::Ocean { course, .. }) => {
583                    *course = hydra_map::shown_course(&name);
584                }
585                _ => {}
586            }
587        }
588        if maps
589            .values()
590            .any(|view| matches!(view, hydra_map::MinimapView::Waiting(_)))
591            && self.mapped.busy()
592        {
593            // The map loads and its areas are laid out with no event to
594            // say so; laid out, a map still waiting -- a room not on it, a
595            // character not placed yet -- waits for the next event.
596            context.request_repaint_after(Duration::from_millis(500));
597        }
598        let now = Instant::now();
599        let travels: BTreeMap<u32, crate::widget::travel::TravelView> = self
600            .characters
601            .values()
602            .filter(|character| character.play.travels())
603            .filter_map(|character| {
604                let seat = &character.seat;
605                Some((seat.id, travel::Saved::view(&self.data, seat, now)?))
606            })
607            .collect();
608        let trainers: BTreeMap<u32, crate::trainer::TrainerView> = self
609            .characters
610            .values()
611            .map(|character| (character.seat.id, character.seat.trainer()))
612            .collect();
613        let everyone: Vec<(u32, Other)> = self
614            .characters
615            .values()
616            .map(|character| {
617                let seat = &character.seat;
618                let other = Other {
619                    name: seat.name.clone(),
620                    game: seat.game.clone(),
621                    snapshot: seat.snapshot(),
622                    minimap: maps.get(&seat.id).cloned(),
623                };
624                (seat.id, other)
625            })
626            .collect();
627        // The pad, read: its presses are the play window's with the
628        // keyboard, by that character's binds.
629        let (pad_said, pad_connected) = self.pad.poll();
630        // The Controller page waiting for a button has the first pressed.
631        if self.settings.waiting_for_button()
632            && let Some(button) = self.pad.take_press()
633        {
634            self.pad_for_page = Some(button);
635        }
636        if pad_connected {
637            context.request_repaint_after(Duration::from_millis(50));
638        }

640        let pad_binds = pad_session
641            .and_then(|session| self.characters.get(&session))
642            .map(|character| character.seat.clone())

644                let own =
645                    crate::pad::character_path(&self.layouts, seat.instance.as_deref(), &seat.name);
646                (seat, self.pad.binds(&own))

652                    seat.story().push(hydra_ui::Shown::Said(said));
653                }
654                None => self.hub.tell(said),
655            }

659        // moments of the character the pad drives.
660        let highlight_rumbles: Vec<String> = self
661            .characters
662            .values()
663            .flat_map(|character| character.seat.rumbles())
664            .collect();

668            self.pad.rumble(binds, (seat.id, state), highlight_rumbles);

671        // Each character's own keys, read before the windows borrow the app.
672        let files: BTreeMap<u32, PathBuf> = self
673            .characters
674            .values()
675            .map(|character| (character.seat.id, self.mine_path(&character.seat)))
676            .collect();
677        for file in files.values() {
678            self.load_mine(Some(file));
679        }
680        let said: BTreeMap<u32, (Vec<String>, u8)> = self
681            .characters
682            .values()
683            .map(|character| {
684                let mine = files
685                    .get(&character.seat.id)
686                    .and_then(|file| self.mine.get(file));
687                let chosen = mine.map_or(0, |(mine, _)| mine.chosen);
688                let said = self.keys_said(mine, &character.seat.name);
689                (character.seat.id, (said, chosen))
690            })
691            .collect();
692        // Each character's own look, and the fonts it names given.
693        let hydras = self.hydras_theme();
694        let outfits: BTreeMap<u32, Option<Outfit>> = self
695            .characters
696            .values()
697            .map(|character| {
698                let seat = &character.seat;
699                let outfit = seat.instance.as_deref().and_then(|instance| {
700                    self.looks
701                        .outfit(&self.data, (instance, &seat.name), &self.themes, &hydras)
702                });
703                (seat.id, outfit)
704            })
705            .collect();

707            .characters
708            .values()
709            .filter_map(|character| {
710                let seat = &character.seat;
711                let instance = seat.instance.as_deref()?;
712                let style = self.looks.tags(&self.data, (instance, &seat.name))?;
713                Some((seat.id, style))
714            })
715            .collect();
716        // Each character's story sorts its container looks as its file says,
717        // its window open or not.
718        let (looks, data) = (&self.looks, &self.data);
719        for character in self.characters.values_mut() {
720            let seat = &character.seat;
721            let sorting = seat
722                .instance
723                .as_deref()
724                .and_then(|instance| looks.sorting(data, (instance, &seat.name)));
725            // And keeps as many lines as its General page says.
726            let scrollback = seat
727                .instance
728                .as_deref()
729                .map_or_else(hydra_ui::story::Scrollback::default, |instance| {
730                    looks.scrollback(data, (instance, &seat.name))
731                });
732            {
733                let mut story = seat.story();

735                story.scrollback(scrollback);

737            // The history keeps lines as the character's General page says.
738            let kept = seat
739                .instance
740                .as_deref()
741                .map_or_else(hydra_ui::history::Kept::default, |instance| {
742                    looks.history(data, (instance, &seat.name))
743                });
744            character.play.history_kept(&kept);
745        }
746        self.give_fonts(context);
747        let (mut asks, mut sends, mut focused) = (Vec::new(), Vec::new(), Vec::new());
748        asks.extend(self.hotbar_pressed());
749        let (keybinds, mine) = (&self.keys, &self.mine);
750        let (caught, numlock, presets) = (&mut self.caught, self.numlock, &self.presets);
751        let (pad, perf) = (&mut self.pad, self.hub.perf);
752        for character in self.characters.values_mut().filter(|c| c.open) {
753            let seat = Arc::clone(&character.seat);
754            let file = files.get(&seat.id);
755            let own = file.and_then(|file| mine.get(file)).map(|(mine, _)| mine);
756            let keys = keybinds.of(own);
757            let (keys_said, chosen) = said
758                .get(&seat.id)
759                .map_or((&[][..], 0), |(said, chosen)| (said.as_slice(), *chosen));
760            let play = &mut character.play;
761            // Everyone but this window's own: by its seat, not its name,
762            // which another game may share.

764                .iter()
765                .filter(|(id, _)| *id != seat.id)
766                .map(|(_, other)| other.clone())
767                .collect();

769            // With Hydra's own title bar the window has no system frame;
770            // egui sends the change when the setting flips.
771            let builder = self
772                .placements
773                .builder(&place, [980.0, 680.0])
774                .with_title(format!("{} — {TITLE}", seat.name))
775                .with_decorations(!play.own_title_bar());
776            // What every pass asks, kept: a key is in the first pass alone.
777            let (mut asked, mut pressed_sends, mut has) = (Vec::new(), Vec::new(), false);
778            let mut seen = None;
779            let outfit = outfits.get(&seat.id).and_then(Option::as_ref);
780            let closed = context.show_viewport_immediate(seat.viewport(), builder, |ui, _| {
781                let drawing = Instant::now();
782                let worn = crate::theme::wearing(ui.ctx(), outfit);
783                seen = Some(ui.input(|input| input.viewport().clone()));

787                let story = seat.story();
788                // The pad drives the window with the keyboard.
789                let pad_here = pad_binds
790                    .as_ref()
791                    .filter(|(pad_seat, _)| pad_seat.id == seat.id)
792                    .map(|(_, binds)| binds);
793                let mut pad_notes = Vec::new();
794                if let Some(binds) = pad_here {
795                    let window = crate::pad::Window {
796                        session: seat.id,
797                        snapshot: snapshot.as_deref(),
798                        story: &story,
799                        portals: &pad_portals,

802                    pressed.extend(driven.made);

805                }

807                    match made {
808                        Macro::Fill(text) => play.fill(&text),
809                        Macro::Act(action) => asked.extend(keyed::asked(ui.ctx(), play, action)),
810                        send @ Macro::Send(_) => pressed_sends.push(send),
811                        // Sent as a click on it would be: what its state
812                        // chooses now.
813                        Macro::Hotbar { bar, button } => {
814                            let pressed = story
815                                .world()
816                                .map(|world| play.hotbar_press((&bar, &button), &world));

818                                Some(Ok(command)) => asked.push(Asked::Send(command)),
819                                Some(Err(why)) => pad_notes.push(why),

821                            }
822                        }
823                    }
824                }

827                    name: &seat.name,
828                    lifecycle: seat.lifecycle(),

832                    presets,
833                    numlock,
834                    keys: keys_said,
835                    set: chosen,
836                    minimap: maps.get(&seat.id),
837                    travel: travels.get(&seat.id),
838                    trainer: trainers.get(&seat.id),
839                    tags: tags.get(&seat.id).copied(),
840                    symbol: seat.symbol(),
841                    commands: &commands,
842                    perf,
843                };
844                // The theme's frame, when it names one: the content keeps
845                // inside it, and it is painted over the panel after.
846                let framed = crate::frames::worn(ui.ctx());
847                let margin = framed.as_deref().map_or(0.0, crate::frames::margin);
848                asked.extend(
849                    egui::CentralPanel::default()
850                        .frame(egui::Frame::NONE.inner_margin(margin))
851                        .show(ui, |ui| play.show(ui, &view))
852                        .inner,
853                );
854                if let Some(framed) = &framed {
855                    let bound = ui.max_rect().shrink(crate::frames::inset(framed));
856                    crate::frames::paint(ui.ctx(), ui.painter(), framed, &[bound]);
857                }
858                if let Some(binds) = pad_here {
859                    if pad.legend {
860                        crate::pad::legend(ui.ctx(), binds, &pad_held);
861                    }
862                    let state = snapshot.as_deref().map(|snapshot| &snapshot.state);
863                    pad.interact_line(ui.ctx(), seat.id, state);
864                    let window = crate::pad::Window {
865                        session: seat.id,
866                        snapshot: snapshot.as_deref(),
867                        story: &story,
868                        portals: &pad_portals,
869                    };
870                    pad.draw_wheel(ui.ctx(), binds, &window);
871                }
872                drop(story);
873                // Its lines drawn: each shown since the last frame is timed,
874                // and the frame itself (`hydra_ui::perf`).
875                let done = Instant::now();
876                let mut story = seat.story();
877                for note in pad_notes {
878                    let note = hydra_session::Notice::line(hydra_session::NoticeKind::Info, note);
879                    story.push(hydra_ui::Shown::Said(note));
880                }
881                story.painted(began, done);
882                story.perf.paint.push(done - drawing);
883                drop(story);
884                // Counted between events: a clock widget's seconds, drawn
885                // again as each whole second turns, and oftener in
886                // roundtime, whose bar runs.
887                if let Some(snapshot) = snapshot.as_ref() {
888                    let state = &snapshot.state;
889                    if state.in_roundtime() == Some(true) {
890                        ui.ctx().request_repaint_after(Duration::from_millis(200));
891                    } else if let Some(next) = state.until_next_second() {
892                        ui.ctx().request_repaint_after(next);
893                    }
894                }
895                drop(worn);
896                ui.ctx().input(|input| input.viewport().close_requested())
897            });
898            if let Some(seen) = &seen {
899                self.placements.note(&place, seen, Instant::now());
900            }
901            // Closing a play window leaves its character running.
902            if closed {
903                character.open = false;
904                // A layout change waiting on a drag is kept as it closes.
905                character.play.keep_waiting_layout();
906                self.placements.closed(&place);
907            }
908            if has && let Some(file) = file {
909                focused.push((seat.id, file.clone()));
910            }
911            asks.extend(asked.into_iter().map(|asked| (Arc::clone(&seat), asked)));
912            sends.extend(
913                pressed_sends
914                    .into_iter()
915                    .map(|made| (Arc::clone(&seat), made)),
916            );
917        }
918        for (session, file) in focused {
919            self.took_keyboard(session, file);
920        }
921        let now = Instant::now();
922        for (seat, made) in sends {
923            self.send_macro(&seat, &made, now);
924        }
925        for (seat, asked) in asks {
926            self.asked(context, seats, &seat, asked);
927        }
928    }

931impl eframe::App for App {
932    fn ui(&mut self, ui: &mut egui::Ui, frame: &mut eframe::Frame) {
933        let context = ui.ctx().clone();
934        if self.faulted {
935            close_all(&context);
936            return;
937        }
938        let drawn = shielded(|| {
939            self.catch(frame);
940            self.caught_pressed(
941                frame.numpad_keys(),
942                frame.captured_keys(),
943                cfg!(target_os = "macos"),
944            );
945            self.draw(ui);
946        });
947        if !drawn {
948            // What the windows hold may be half changed: drawn no more, and
949            // nothing of it written.
950            self.faulted = true;
951            eprintln!(
952                "[hydra] the windows met a fault and close; every character plays on, \
953                 and `hydra` opens them again"
954            );
955            close_all(&context);
956            return;
957        }
958        crate::opened(crate::FIRST_FRAME);
959    }
960}

962/// Run `frame`, a panic in it caught: `false` when one was. Every window is
963/// drawn on the one thread, so a panic in any of them would otherwise end
964/// Hydra and every character with it; caught, the windows close and the
965/// characters play on.
966fn shielded(frame: impl FnOnce()) -> bool {
967    std::panic::catch_unwind(std::panic::AssertUnwindSafe(frame)).is_ok()
968}
969
970/// Close every window: each hidden first, as [`App::close_or_hide`] says
971/// why, then the hub, whose closing lets them all go.
972fn close_all(context: &egui::Context) {
973    let open: Vec<egui::ViewportId> =
974        context.input(|input| input.raw.viewports.keys().copied().collect());
975    for id in open {
976        context.send_viewport_cmd_to(id, egui::ViewportCommand::Visible(false));
977    }
978    context.send_viewport_cmd_to(egui::ViewportId::ROOT, egui::ViewportCommand::Close);
979}

981impl App {
982    /// The hub, and each play window over it, with the keys pressed in them
983    /// done: all of a frame but the fork's keys, which come through eframe.
984    fn draw(&mut self, ui: &mut egui::Ui) {
985        let began = Instant::now();
986        let context = ui.ctx().clone();
987        // Where every window finds the frames its theme names.
988        crate::frames::keep_frames(&context, &self.data);
989        // What every window opening this frame is made at; the zoom set
990        // below takes from the next.
991        self.placements.drawn_at(context.zoom_factor());
992        self.wear_theme(&context);
993        self.sessions.highlights().set_volume(self.own.volume());
994        self.sessions
995            .highlights()
996            .set_merchants(self.own.merchant_banners(), self.own.merchant_notices());
997        crate::story_picture::set_rules(
998            &context,
999            crate::story_picture::Rules {
1000                shown: self.own.story_pictures(),
1001                folder: self.own.pictures_folder(),
1002            },
1003        );
1004        // Every window drawn at the zoom kept: egui draws them all at one.
1005        if (context.zoom_factor() - self.own.zoom()).abs() > 0.001 {
1006            context.set_zoom_factor(self.own.zoom());
1007        }
1008        if let Some(said) = self.highlights.read_again(
1009            &self.data,
1010            &self.sessions.highlights(),
1011            Some(&crate::repaint(&context)),
1012        ) {
1013            self.hub.tell(said);
1014        }
1015        for why in crate::store::untold() {
1016            self.hub.tell(why);
1017        }
1018        let glance = self.sessions.glance(Instant::now());
1019        self.seat(&glance.seats);
1020        self.close_or_hide(&context);
1021        if glance.said.is_some() {
1022            // A frame when the answer is due to go.
1023            context.request_repaint_after(Duration::from_secs(1));
1024        }
1025        // The merged lines are held while the hub alone is drawn: each feed
1026        // takes the same lock to offer its lines, and would wait out every
1027        // other window's drawing were it held for the frame.
1028        let asked = {
1029            let merged = self.sessions.merged();
1030            let merged = merged
1031                .lock()
1032                .unwrap_or_else(std::sync::PoisonError::into_inner);
1033            let view = HubView {
1034                seats: &glance.seats,
1035                offered: &glance.offered,
1036                roster: &glance.roster,
1037                teams: &glance.teams,
1038                listing: glance.listing.as_ref(),
1039                said: glance.said.as_deref(),
1040                web_at: glance.web_at.as_ref(),
1041                update: glance.update.as_ref(),
1042                card: &self.cards.design,
1043                merged: &merged,
1044            };
1045            egui::CentralPanel::default()
1046                .show(ui, |ui| self.hub.show(ui, &view))
1047                .inner
1048        };
1049        if let Some(action) = asked {
1050            self.act(&context, action);
1051        }
1052        if let Some(next) = self.send_due(&glance.seats, Instant::now()) {
1053            context.request_repaint_after(next);
1054        }
1055        self.keyboard_now = None;
1056        self.play_windows(&context, &glance.seats, began);
1057        self.log_windows(&context, &glance.seats);
1058        self.settings_window(&context, &glance);
1059        self.explorer_window(&context, &glance.seats);
1060        self.survey_window(&context);
1061        self.reports_window(&context, &glance.seats);
1062        self.left_keyboard();
1063        let seen = context.input(|input| input.viewport().clone());
1064        self.placements.note(HUB, &seen, Instant::now());
1065        if let Some(after) = self.placements.save_due(Instant::now()) {
1066            context.request_repaint_after(after);
1067        }
1068        // The whole frame, every window of it, for each character's
1069        // performance (`hydra_ui::perf`), and its log while it is shown.
1070        let took = began.elapsed();
1071        for seat in &glance.seats {
1072            seat.story().perf.frame.push(took);
1073        }
1074        let on = self.hub.perf;
1075        if let Some(next) = self
1076            .perf_log
1077            .tick(on, &self.data, &glance.seats, Instant::now())
1078        {
1079            context.request_repaint_after(next);
1080        }
1081    }

1087/// How the windows are drawn: on Windows by one graphics API alone, Vulkan
1088/// where the computer's graphics card has it and DirectX 12 otherwise.
1089/// wgpu otherwise opens every API it knows to choose among them, and each
1090/// loads its driver for the life of the process, Vulkan's beside
1091/// DirectX's, a hundred megabytes and more of driver for nothing. Vulkan
1092/// first: DirectX 12's device can take half a minute to open on some
1093/// computers, with overlays and drivers in its way, where Vulkan's opens at
1094/// once. `WGPU_BACKEND` still chooses.
1095fn graphics() -> eframe::egui_wgpu::WgpuConfiguration {
1096    let mut config = eframe::egui_wgpu::WgpuConfiguration::default();
1097    if cfg!(windows)
1098        && eframe::wgpu::Backends::from_env().is_none()
1099        && let eframe::egui_wgpu::WgpuSetup::CreateNew(create) = &mut config.wgpu_setup
1100    {
1101        create.instance_descriptor.backends = windows_backend();
1102    }
1103    config
1104}
1105
1106/// Vulkan, when a graphics card here answers to it; DirectX 12 otherwise.
1107/// Asked of a Vulkan instance of its own, before the windows' own is made:
1108/// the driver it loads is the one the windows then use.
1109fn windows_backend() -> eframe::wgpu::Backends {
1110    use eframe::wgpu::{Backends, DeviceType, Instance, InstanceDescriptor};
1111    let mut asked = InstanceDescriptor::new_without_display_handle();
1112    asked.backends = Backends::VULKAN;
1113    let probe = Instance::new(asked);
1114    let card = pollster::block_on(probe.enumerate_adapters(Backends::VULKAN))
1115        .iter()
1116        .any(|adapter| adapter.get_info().device_type != DeviceType::Cpu);
1117    if card {
1118        Backends::VULKAN
1119    } else {
1120        Backends::DX12
1121    }
1122}

1124/// Open the windows over `sessions` -- the hub shown or hidden as `open`
1125/// says, and each character's play window it keeps -- and run until the
1126/// last closes, or
1127/// the binary closes them once the run is over ([`Sessions::close`]). A
1128/// character seated after opens its play window unless it was started
1129/// headless or for the web. The hub's requests run on `runtime`; layouts,
1130/// presets, keys and pictures are kept in `data`, the data folder. Blocks
1131/// the calling thread, which must be the main one.
1132///
1133/// Closed, the windows can be opened again by another call, on the same
1134/// thread: eframe keeps its event loop between them. What drew them is let
1135/// go as they close.
1136///
1137/// # Errors
1138///
1139/// The window could not be opened: no display, or no graphics adapter.
1140pub fn run(
1141    sessions: Sessions,
1142    runtime: tokio::runtime::Handle,
1143    data: &Path,
1144    open: Open,
1145) -> eframe::Result {
1146    let hidden = open != Open::Hub;
1147    let options = eframe::NativeOptions {
1148        viewport: Placements::load(data)
1149            .builder(HUB, [660.0, 520.0])
1150            .with_title(TITLE)
1151            .with_visible(!hidden),
1152        wgpu_options: graphics(),
1153        ..Default::default()
1154    };
1155    let wake = sessions.wake();
1156    let data = data.to_path_buf();
1157    let shown = eframe::run_native(
1158        TITLE,
1159        options,
1160        Box::new(move |creation| {
1161            crate::opened("the window and its graphics are ready");
1162            sessions
1163                .wake()
1164                .opened(std::sync::Arc::new(crate::EguiWindows(
1165                    creation.egui_ctx.clone(),
1166                )));
1167            #[cfg(feature = "doll-infinite")]
1168            hydra_mem::part(hydra_mem::Part::Doll, || {
1169                crate::widget::infinite::open(creation, &data);
1170            });
1171            let mut app = App::new(sessions, runtime, &data);
1172            crate::opened("the app is built");
1173            if hidden {
1174                app.hub_shown = HubShown::Hiding;
1175            }
1176            Ok(Box::new(app))
1177        }),
1178    );
1179    wake.closed();
1180    shown
1181}

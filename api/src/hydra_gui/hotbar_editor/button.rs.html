1//! The editor's button form: the bar's buttons to choose from, and the one
2//! chosen -- what it shows and sends, its key (pressed to be caught), its
3//! slot -- and its states.
4//!
5//! **Its states** are cards, in order, each said as a sentence (`While in
6//! roundtime: dimmed, counts down roundtime.`) and opened to change its
7//! condition (the highlight editor's builder, each test marked as it holds
8//! now) and its look. The state that styles the button now is starred; one
9//! that can never win, below one that holds whenever it does, says so.
10//! Below them, *Otherwise*: the look when none holds, and the countdown
11//! when the state that does has none. States come in one click from the
12//! Add menu, or as the button's command suggests.
13

38    chosen: Option<usize>,

45/// An id for a button labelled `label`, none of `bar`'s already.
46pub(super) fn unique_id(bar: &Bar, label: &str) -> String {
47    let base: String = label
48        .trim()
49        .to_lowercase()
50        .chars()
51        .map(|c| if c.is_alphanumeric() { c } else { '-' })
52        .collect::<String>()
53        .trim_matches('-')
54        .to_owned();
55    let base = if base.is_empty() {
56        "button".to_owned()
57    } else {
58        base
59    };
60    let mut id = base.clone();
61    let mut n = 1;
62    while bar.buttons.iter().any(|button| button.id == id) {
63        n += 1;
64        id = format!("{base}-{n}");
65    }
66    id
67}

69/// The Macro row of the form: the macro the button runs, chosen from
70/// `macros`, and *Lift into a macro* for a button with a chain of its own,
71/// which puts the chain in `lifted`.
72fn macro_row(
73    ui: &mut egui::Ui,
74    button: &mut Button,
75    macros: &[String],
76    lifted: &mut Option<(String, String)>,
77) {
78    ui.label("Macro");
79    ui.horizontal(|ui| {
80        let chosen = button.macro_name.clone().unwrap_or_default();
81        let shown = if chosen.is_empty() {
82            "none: it sends its own chain"
83        } else {
84            chosen.as_str()
85        };
86        egui::ComboBox::from_id_salt("hotbar-button-macro")
87            .selected_text(shown)
88            .show_ui(ui, |ui| {
89                if ui.selectable_label(chosen.is_empty(), "none").clicked() {
90                    button.macro_name = None;
91                }
92                for name in macros {
93                    if ui.selectable_label(chosen == *name, name).clicked() {
94                        button.macro_name = Some(name.clone());
95                    }
96                }
97            })
98            .response
99            .on_hover_text(
100                "The macro this button runs: its chain and its states, kept on \
101                 the Macros page. None, and the button sends what is below.",
102            );
103        if button.macro_name.is_none()
104            && !button.command.trim().is_empty()
105            && ui
106                .small_button("Lift into a macro")
107                .on_hover_text(
108                    "Make a macro of this button's chain, named by its id, and \
109                     have the button run it",
110                )
111                .clicked()
112        {
113            *lifted = Some((button.id.clone(), button.command.clone()));
114            button.macro_name = Some(button.id.clone());
115        }
116    });
117    ui.end_row();
118}

120impl Form {
121    /// The buttons, the one chosen, and its states, in `ui`, changing
122    /// `bar`; tried on `world`, effects' names picked from `names`.
123    pub(super) fn show(
124        &mut self,
125        ui: &mut egui::Ui,
126        bar: &mut Bar,
127        world: Option<&World<'_>>,
128        (names, folder): (&BTreeSet<String>, &Folder),
129        macros: &[String],
130    ) -> Option<(String, String)> {
131        self.chooser(ui, bar);
132        let at = self.chosen.filter(|at| *at < bar.buttons.len())?;
133        ui.add_space(6.0);
134        ui.horizontal(|ui| {
135            if ui.button("Duplicate").clicked() {
136                let mut copy = bar.buttons[at].clone();
137                copy.id = unique_id(bar, &copy.label);
138                // Two buttons on one key: the copy goes without.
139                copy.hotkey = None;
140                copy.nickname = None;
141                copy.slot = None;
142                bar.buttons.push(copy);
143                self.chosen = Some(bar.buttons.len() - 1);
144            }
145            if ui.button("Remove button").clicked() {
146                bar.buttons.remove(at);
147                self.chosen = None;
148                self.open = None;
149            }
150        });
151        let button = self.chosen.and_then(|at| bar.buttons.get_mut(at))?;
152        let lifted = self.fields(ui, button, folder, macros);
153        ui.add_space(6.0);
154        let names: Vec<String> = names.iter().cloned().collect();
155        let seen = Seen {
156            names: &names,
157            folder,
158        };
159        self.states(ui, button, world, seen);
160        lifted
161    }

163    /// The bar's buttons, one chosen by a click; one added.
164    fn chooser(&mut self, ui: &mut egui::Ui, bar: &mut Bar) {
165        ui.horizontal_wrapped(|ui| {
166            for (at, button) in bar.buttons.iter().enumerate() {
167                let shown = if button.label.trim().is_empty() {
168                    button.id.as_str()
169                } else {
170                    button.label.as_str()
171                };
172                if ui
173                    .selectable_label(self.chosen == Some(at), shown)
174                    .clicked()
175                {
176                    self.chosen = Some(at);
177                    self.open = None;
178                    self.capturing = false;
179                }
180            }
181            if ui.button("+ button").clicked() {
182                let id = unique_id(bar, "button");
183                bar.buttons.push(Button {
184                    id,
185                    label: "New".to_owned(),
186                    ..Button::default()
187                });
188                self.chosen = Some(bar.buttons.len() - 1);
189                self.open = None;
190            }
191        });
192    }

194    /// What the button shows and sends, its key and its slot.
195    fn fields(
196        &mut self,
197        ui: &mut egui::Ui,
198        button: &mut Button,
199        folder: &Folder,
200        macros: &[String],
201    ) -> Option<(String, String)> {
202        let mut lifted = None;
203        egui::Grid::new("hotbar-button-fields")
204            .num_columns(2)
205            .spacing([12.0, 6.0])
206            .show(ui, |ui| {
207                ui.label("Label");
208                ui.text_edit_singleline(&mut button.label)
209                    .on_hover_text("What it shows, unless a state says otherwise");
210                ui.end_row();
211                macro_row(ui, button, macros, &mut lifted);
212                ui.label("Sends");
213                ui.add_enabled(
214                    button.macro_name.is_none(),
215                    egui::TextEdit::multiline(&mut button.command)
216                        .desired_rows(2)
217                        .hint_text(
218                            "a chain: commands with a comma between them, s1.5 a wait, rt and \
219                             ct the clock's, stance and the other behaviour words",
220                        ),
221                );
222                ui.end_row();
223                ui.label("Nickname");
224                optional(ui, &mut button.nickname, "your own name for it");
225                ui.end_row();
226                ui.label("Id");
227                ui.add(egui::TextEdit::singleline(&mut button.id).desired_width(120.0))
228                    .on_hover_text(
229                        "Its name in the file; .hotbar presses it by this or its nickname",
230                    );
231                ui.end_row();
232                ui.label("Tooltip");
233                optional(
234                    ui,
235                    &mut button.tooltip,
236                    "said on hover; its label unless given",
237                );
238                ui.end_row();
239                ui.label("Key");
240                self.hotkey(ui, button);
241                ui.end_row();
242                ui.label("Face");
243                ui.horizontal(|ui| {
244                    ui.radio_value(&mut button.icon_mode, IconMode::Text, "its label");
245                    ui.radio_value(&mut button.icon_mode, IconMode::Icon, "its icon")
246                        .on_hover_text("Its label is said on hover");
247                    ui.radio_value(&mut button.icon_mode, IconMode::IconAndLabel, "both");
248                });
249                ui.end_row();
250                if button.icon_mode != IconMode::Text {
251                    ui.label("Icon");
252                    ui.vertical(|ui| icon_ui(ui, &mut button.icon, folder, "button"));
253                    ui.end_row();
254                }
255                ui.label("Slot");
256                ui.horizontal(|ui| {
257                    let mut placed = button.slot.is_some();
258                    ui.checkbox(&mut placed, "its own").on_hover_text(
259                        "Unticked, it takes the first free slot; Ctrl and drag on the bar moves it",
260                    );
261                    if placed {
262                        let mut slot = button.slot.unwrap_or(0);
263                        ui.add(egui::DragValue::new(&mut slot).range(0..=799));
264                        button.slot = Some(slot);
265                    } else {
266                        button.slot = None;
267                    }
268                });
269                ui.end_row();
270            });
271        lifted
272    }

274    /// The button's hotkey: written, or caught from the next key pressed.
275    fn hotkey(&mut self, ui: &mut egui::Ui, button: &mut Button) {
276        ui.horizontal(|ui| {
277            if self.capturing {
278                ui.label("Press a key... (Escape: none)");
279                if let Some(caught) = caught(ui) {
280                    button.hotkey = match caught {
281                        Caught::Key(key) => Some(key),
282                        Caught::Escape => None,
283                    };
284                    self.capturing = false;
285                }
286                if ui.button("Stop").clicked() {
287                    self.capturing = false;
288                }
289                return;
290            }
291            optional(ui, &mut button.hotkey, "none; or alt+h, f5");
292            if ui.button("Capture").clicked() {
293                self.capturing = true;
294            }
295            if let Some(written) = &button.hotkey
296                && let Err(why) = crate::hotbars::hotkey(written)
297            {
298                ui.colored_label(ui.visuals().warn_fg_color, why);
299            }
300        });
301    }

303    /// The button's states, in order, and *Otherwise*.
304    fn states(
305        &mut self,
306        ui: &mut egui::Ui,
307        button: &mut Button,
308        world: Option<&World<'_>>,
309        seen: Seen<'_>,
310    ) {
311        let names = seen.names;
312        crate::form::section_with(ui, "States", "the first that holds styles it");
313        self.offers(ui, button);
314        let winning = world.and_then(|world| button.look(world).state);
315        let shadowed = said::shadowed(&button.states);
316        let live = Live { world, names };
317        let (mut gone, mut up) = (None, None);
318        for (at, state) in button.states.iter_mut().enumerate() {
319            egui::Frame::group(ui.style()).show(ui, |ui| {
320                ui.horizontal(|ui| {
321                    if winning == Some(at) {
322                        ui.colored_label(egui::Color32::from_rgb(240, 200, 60), "\u{2605}")
323                            .on_hover_text("This state styles it now");
324                    } else {
325                        live.mark(ui, &state.when);
326                    }
327                    ui.label(said::state(state));
328                });
329                ui.horizontal(|ui| {
330                    let opened = self.open == Some(at);
331                    if ui
332                        .small_button(if opened { "Close" } else { "Change" })
333                        .clicked()
334                    {
335                        self.open = (!opened).then_some(at);
336                    }
337                    if at > 0
338                        && ui
339                            .small_button("Up")
340                            .on_hover_text("Checked sooner")
341                            .clicked()
342                    {
343                        up = Some(at);
344                    }
345                    if at + 1 < shadowed.len()
346                        && ui
347                            .small_button("Down")
348                            .on_hover_text("Checked later")
349                            .clicked()
350                    {
351                        up = Some(at + 1);
352                    }
353                    if ui.small_button("x").on_hover_text("Take it out").clicked() {
354                        gone = Some(at);
355                    }
356                    if let Some(above) = shadowed[at] {
357                        ui.colored_label(
358                            ui.visuals().warn_fg_color,
359                            format!(
360                                "Never wins: the state {} above holds whenever this does.",
361                                above + 1
362                            ),
363                        );
364                    }
365                });
366                if self.open == Some(at) {
367                    ui.push_id(("hotbar-state", at), |ui| {
368                        let mut when = Some(state.when.clone());
369                        when_live(ui, &mut when, live);
370                        state.when = match when {
371                            // One test alone is that test, as the file had it.
372                            Some(Condition::All { mut conditions }) if conditions.len() == 1 => {
373                                conditions.remove(0)
374                            }
375                            Some(when) => when,
376                            None => Condition::All {
377                                conditions: Vec::new(),
378                            },
379                        };
380                        look_ui(
381                            ui,
382                            (&mut state.style, &mut state.countdown),
383                            Some(&mut state.command),
384                            seen,
385                        );
386                    });
387                }
388            });
389        }
390        if let Some(at) = up {
391            button.states.swap(at - 1, at);
392            self.open = None;
393        }
394        if let Some(at) = gone {
395            button.states.remove(at);
396            self.open = None;
397        }
398        self.otherwise(ui, button, winning.is_none() && world.is_some(), seen);
399    }

401    /// The states the command suggests, and the Add menu's.
402    fn offers(&mut self, ui: &mut egui::Ui, button: &mut Button) {
403        let suggested = said::suggested(&button.command, &button.states);
404        ui.horizontal_wrapped(|ui| {
405            ui.menu_button("+ Add a state", |ui| {
406                for offered in said::offered(&button.command) {
407                    if ui.button(&offered.name).clicked() {
408                        button.states.push(offered.state);
409                        self.open = Some(button.states.len() - 1);
410                        ui.close();
411                    }
412                }
413                ui.separator();
414                if ui.button("One of your own...").clicked() {
415                    button.states.push(ButtonState {
416                        when: Condition::RtActive,
417                        style: Style::default(),
418                        countdown: None,
419                        command: None,
420                    });
421                    self.open = Some(button.states.len() - 1);
422                    ui.close();
423                }
424            });
425            if !suggested.is_empty() {
426                crate::form::hint(ui, "Suggested:");
427            }
428            for offered in suggested {
429                if ui
430                    .small_button(format!("+ {}", offered.name))
431                    .on_hover_text(said::state(&offered.state))
432                    .clicked()
433                {
434                    button.states.push(offered.state);
435                }
436            }
437        });
438    }

440    /// *Otherwise*: the look when no state holds, and the countdown when
441    /// the one that does has none.
442    fn otherwise(&mut self, ui: &mut egui::Ui, button: &mut Button, now: bool, seen: Seen<'_>) {
443        egui::Frame::group(ui.style()).show(ui, |ui| {
444            let base = button.default_style.clone().unwrap_or_default();
445            ui.horizontal(|ui| {
446                if now {
447                    ui.colored_label(egui::Color32::from_rgb(240, 200, 60), "\u{2605}")
448                        .on_hover_text("No state holds now");
449                }
450                ui.label(format!(
451                    "Otherwise: {}.",
452                    said::look(&base, button.countdown.as_ref(), None)
453                ));
454            });
455            let opened = self.open == Some(OTHERWISE);
456            if ui
457                .small_button(if opened { "Close" } else { "Change" })
458                .clicked()
459            {
460                self.open = (!opened).then_some(OTHERWISE);
461            }
462            if opened {
463                ui.push_id("hotbar-otherwise", |ui| {
464                    let mut style = base;
465                    look_ui(ui, (&mut style, &mut button.countdown), None, seen);
466                    button.default_style = (style != Style::default()).then_some(style);
467                });
468            }
469        });
470    }

473/// A key caught for a button.
474pub(crate) enum Caught {
475    /// A key, as a hotbar file writes it.
476    Key(String),
477    /// Escape: no key.
478    Escape,
479}
480
481/// The key pressed this frame, if one was.
482pub(crate) fn caught(ui: &egui::Ui) -> Option<Caught> {
483    ui.input(|input| {
484        input.events.iter().find_map(|event| {
485            let egui::Event::Key {
486                key,
487                physical_key,
488                pressed: true,
489                modifiers,
490                ..
491            } = event
492            else {
493                return None;
494            };
495            if *key == egui::Key::Escape && modifiers.is_none() {
496                return Some(Caught::Escape);
497            }
498            let name = crate::keys::winit_name(physical_key.unwrap_or(*key))?;
499            Some(Caught::Key(
500                crate::keys::Chord::of(&name, *modifiers).written(),
501            ))
502        })
503    })
504}

506/// A field that may be empty, which is none.
507pub(crate) fn optional(ui: &mut egui::Ui, value: &mut Option<String>, hint: &str) {
508    let mut text = value.clone().unwrap_or_default();
509    ui.add(egui::TextEdit::singleline(&mut text).hint_text(hint));
510    *value = (!text.trim().is_empty()).then_some(text);
511}

513/// A look's fields: what it reads, its colours, dimmed, what it counts
514/// down, and what it sends instead when it may.
515fn look_ui(
516    ui: &mut egui::Ui,
517    (style, countdown): (&mut Style, &mut Option<Countdown>),
518    command: Option<&mut Option<String>>,
519    seen: Seen<'_>,
520) {
521    egui::Grid::new("hotbar-look")
522        .num_columns(2)
523        .spacing([12.0, 6.0])
524        .show(ui, |ui| {
525            ui.label("Reads");
526            optional(ui, &mut style.label, "its label");
527            ui.end_row();
528            ui.label("Text colour");
529            colour_ui(ui, &mut style.fg);
530            ui.end_row();
531            ui.label("Ground colour");
532            colour_ui(ui, &mut style.bg);
533            ui.end_row();
534            ui.label("Dimmed");
535            ui.checkbox(&mut style.dim, "");
536            ui.end_row();
537            ui.label("Counts down");
538            countdown_ui(ui, countdown, seen.names);
539            ui.end_row();
540            ui.label("Icon instead");
541            ui.vertical(|ui| icon_ui(ui, &mut style.icon, seen.folder, "look"))
542                .response
543                .on_hover_text("Shown when the button's face has an icon");
544            ui.end_row();
545            if let Some(command) = command {
546                ui.label("Sends instead");
547                optional(ui, command, "the button's command");
548                ui.end_row();
549            }
550        });
551}

553/// A colour that may be none.
554pub(super) fn colour_ui(ui: &mut egui::Ui, colour: &mut Option<String>) {
555    ui.horizontal(|ui| {
556        let mut on = colour.is_some();
557        ui.checkbox(&mut on, "");
558        if !on {
559            *colour = None;
560            crate::form::hint(ui, "as it is otherwise");
561            return;
562        }
563        let mut rgb = colour
564            .as_deref()
565            .and_then(hydra_ui::theme::parse_hex)
566            .unwrap_or([128, 255, 128]);
567        ui.color_edit_button_srgb(&mut rgb);
568        *colour = Some(hydra_ui::theme::hex(rgb));
569    });
570}

572/// What a look counts down: nothing, roundtime, casttime, or an effect's
573/// time, its name picked from those the character has had.
574pub(crate) fn countdown_ui(ui: &mut egui::Ui, countdown: &mut Option<Countdown>, names: &[String]) {
575    ui.horizontal(|ui| {
576        let kind = match countdown {
577            None => "nothing",
578            Some(Countdown::Roundtime) => "roundtime",
579            Some(Countdown::Casttime) => "casttime",
580            Some(Countdown::Effect { .. }) => "an effect's time",
581        };
582        let mut chose = kind;
583        egui::ComboBox::from_id_salt("hotbar-countdown")
584            .selected_text(kind)
585            .show_ui(ui, |ui| {
586                for one in ["nothing", "roundtime", "casttime", "an effect's time"] {
587                    ui.selectable_value(&mut chose, one, one);
588                }
589            });
590        if chose != kind {
591            *countdown = match chose {
592                "roundtime" => Some(Countdown::Roundtime),
593                "casttime" => Some(Countdown::Casttime),
594                "an effect's time" => Some(Countdown::Effect {
595                    category: EffectCategory::ActiveSpells,
596                    name: String::new(),
597                    name_match: NameMatch::Contains,
598                }),
599                _ => None,
600            };
601        }
602        if let Some(Countdown::Effect { category, name, .. }) = countdown {
603            egui::ComboBox::from_id_salt("hotbar-countdown-list")
604                .selected_text(category.dialog())
605                .show_ui(ui, |ui| {
606                    for one in [
607                        EffectCategory::ActiveSpells,
608                        EffectCategory::Buffs,
609                        EffectCategory::Debuffs,
610                        EffectCategory::Cooldowns,
611                    ] {
612                        ui.selectable_value(category, one, one.dialog());
613                    }
614                });
615            ui.add(
616                egui::TextEdit::singleline(name)
617                    .hint_text("its name, or part of it")
618                    .desired_width(140.0),
619            );
620            pick_name(ui, name, names);
621        }
622    });
623}

631    /// A hotkey caught with a modifier is taken with it held: the
632    /// modifier's own press, which comes first, is passed over.
633    #[test]
634    fn a_hotkey_is_caught_with_its_modifier() {
635        let mut harness = Harness::builder().build_ui_state(
636            |ui, keys: &mut Vec<String>| {
637                if let Some(Caught::Key(key)) = caught(ui) {
638                    keys.push(key);
639                }
640            },
641            Vec::new(),
642        );
643        let press = |key| egui::Event::Key {
644            key,
645            physical_key: Some(key),
646            pressed: true,
647            repeat: false,
648            modifiers: egui::Modifiers::ALT,
649        };
650        harness.event(press(egui::Key::AltLeft));
651        harness.run();
652        assert!(harness.state().is_empty(), "Alt alone is no key");
653        harness.event(press(egui::Key::H));
654        harness.run();
655        assert_eq!(
656            harness.state().first().map(String::as_str),
657            Some("Alt+KeyH")
658        );
659    }

12/// A condition, said: `in roundtime`, `health below 50%`, `not hidden`.
13pub(crate) fn condition(when: &Condition) -> String {
14    match when {
15        Condition::All { conditions } if conditions.is_empty() => "always".to_owned(),
16        Condition::Any { conditions } if conditions.is_empty() => "never".to_owned(),
17        Condition::All { conditions } | Condition::Any { conditions } if conditions.len() == 1 => {
18            condition(&conditions[0])
19        }
20        Condition::All { conditions } => joined(conditions, " and "),
21        Condition::Any { conditions } => joined(conditions, " or "),
22        Condition::Not { condition: inner } => not(inner),
23        Condition::EffectActive { name, .. } => format!("{} is up", named(name)),
24        Condition::EffectInactive { name, .. } => format!("{} is not up", named(name)),
25        Condition::EffectTime {
26            name, cmp, seconds, ..
27        } => format!("{} has {} {seconds} s left", named(name), compared(*cmp)),
28        Condition::RtActive => "in roundtime".to_owned(),
29        Condition::CtActive => "in casttime".to_owned(),
30        Condition::Indicator { id, active: true } => id.clone(),
31        Condition::Indicator { id, active: false } => format!("not {id}"),
32        Condition::Vital {
33            vital,
34            cmp,
35            value,
36            unit,
37        } => format!(
38            "{} {} {value}{}",
39            vital_name(*vital),
40            compared(*cmp),
41            match unit {
42                VitalUnit::Percent => "%",
43                VitalUnit::Absolute => "",
44            }
45        ),
46        Condition::Injury { area, cmp, level } => {
47            format!("{area}'s injury {} {level}", compared(*cmp))
48        }
49        Condition::SpellAffordable { number } => format!("{} is affordable", spell(*number)),
50        Condition::TimeOfDay { phase } => format!("it is {}", phase_name(*phase)),
51        Condition::HandEmpty { hand } => format!("{} is empty", hand_name(*hand)),
52        Condition::HandHolds {
53            hand,
54            item_type,
55            name,
56            ..
57        } => {
58            let what = [item_type.as_deref(), name.as_deref()]
59                .into_iter()
60                .flatten()
61                .collect::<Vec<_>>()
62                .join(" ");
63            let what = if what.is_empty() { "something" } else { &what };
64            format!("{} holds {what}", hand_name(*hand))
65        }
66        Condition::SpellPrepared { name: None, .. } => "a spell is prepared".to_owned(),
67        Condition::SpellPrepared {
68            name: Some(name), ..
69        } => format!("{name} is prepared"),
70        Condition::Flag { name, active: true } => format!("flag {name} is set"),
71        Condition::Flag {
72            name,
73            active: false,
74        } => format!("flag {name} is not set"),
75        Condition::Room { name, .. } => format!("in {}", named(name)),
76        Condition::Stance { stance } => format!("stance {stance}"),
77        beyond => beyond_said(beyond),
78    }
79}

81/// A test of Hydra's own as a sentence: the encumbrance, a resource, the
82/// target, the room's creatures.
83fn beyond_said(when: &Condition) -> String {
84    match when {
85        Condition::CrtrStatus { id, active: true } => format!("the target is {id}"),
86        Condition::CrtrStatus { id, active: false } => format!("the target is not {id}"),
87        Condition::Encumbrance { cmp, percent } => {
88            format!("encumbrance {} {percent}%", compared(*cmp))
89        }
90        Condition::Resource {
91            resource,
92            cmp,
93            value,
94        } => format!("{} {} {value}", resource_name(*resource), compared(*cmp)),
95        Condition::TargetFlag { flag, active: true } => format!("the target is {flag}"),
96        Condition::TargetFlag {
97            flag,
98            active: false,
99        } => format!("the target is not {flag}"),
100        Condition::TargetKind { kind } => format!(
101            "the target is {}",
102            match kind {
103                TargetKindName::Undead => "undead",
104                TargetKindName::Living => "living",
105                TargetKindName::Noncorporeal => "noncorporeal",
106            }
107        ),
108        Condition::TargetHealth { cmp, percent } => {
109            format!("the target's health {} {percent}%", compared(*cmp))
110        }
111        Condition::Creatures {
112            cmp,
113            count,
114            hostile,
115        } => format!(
116            "{} {} {count}",
117            if *hostile {
118                "creatures to attack"
119            } else {
120                "creatures in the room"
121            },
122            compared(*cmp)
123        ),
124        Condition::RoomHolds { name, .. } => format!("the room holds {}", named(name)),
125        Condition::RoomSpell { spell } => format!("{spell} is laid here"),
126        _ => String::new(),
127    }
128}

130/// `inner`, turned over.
131fn not(inner: &Condition) -> String {
132    match inner {
133        Condition::RtActive => "not in roundtime".to_owned(),
134        Condition::CtActive => "not in casttime".to_owned(),
135        Condition::SpellAffordable { number } => format!("{} is not affordable", spell(*number)),
136        Condition::EffectActive { name, .. } => format!("{} is not up", named(name)),
137        Condition::Indicator { id, active } => condition(&Condition::Indicator {
138            id: id.clone(),
139            active: !active,
140        }),
141        other => format!("not ({})", condition(other)),
142    }
143}

145fn joined(conditions: &[Condition], with: &str) -> String {

148        .map(condition)

153fn named(name: &str) -> String {
154    if name.trim().is_empty() {
155        "(no name)".to_owned()
156    } else {
157        name.trim().to_owned()
158    }

161fn compared(cmp: Cmp) -> &'static str {
162    match cmp {
163        Cmp::Lt => "below",
164        Cmp::Le => "at most",
165        Cmp::Gt => "above",
166        Cmp::Ge => "at least",
167    }
168}

214/// What a countdown counts, said.
215pub(crate) fn countdown(countdown: &Countdown) -> String {
216    match countdown {
217        Countdown::Roundtime => "roundtime".to_owned(),
218        Countdown::Casttime => "casttime".to_owned(),
219        Countdown::Effect { name, .. } => format!("{}'s time", named(name)),
220        Countdown::RoomSpell { spell } => format!("the room's {spell}"),
221    }
222}

224/// What countdowns count, said: `Barkskin's time, then Barkskin's time`.
225pub(crate) fn countdowns(counts: &Countdowns) -> String {
226    counts
227        .0
228        .iter()
229        .map(|timer| countdown(&timer.countdown))
230        .collect::<Vec<_>>()
231        .join(", then ")
232}

234/// A state's look and what it does, said: `dimmed, counts down roundtime`.
235pub(crate) fn look(style: &Style, counts: &Countdowns, command: Option<&str>) -> String {
236    let mut said = Vec::new();
237    if style.dim {
238        said.push("dimmed".to_owned());
239    }
240    if let Some(label) = &style.label {
241        said.push(format!("reads \u{201c}{label}\u{201d}"));
242    }
243    if let Some(fg) = &style.fg {
244        said.push(format!("text {fg}"));
245    }
246    if let Some(bg) = &style.bg {
247        said.push(format!("ground {bg}"));
248    }
249    if !counts.is_empty() {
250        said.push(format!("counts down {}", countdowns(counts)));
251    }
252    if let Some(command) = command {
253        said.push(format!("sends \u{201c}{command}\u{201d}"));
254    }
255    if said.is_empty() {
256        "plain".to_owned()
257    } else {
258        said.join(", ")
259    }
260}

294/// The states the Add menu offers, for a button that sends `command`.
295pub(crate) fn offered(command: &str) -> Vec<Offered> {
296    let number = spell_number(command).unwrap_or(101);
297    let effect = |category: EffectCategory| Countdown::Effect {
298        category,
299        name: String::new(),
300        name_match: NameMatch::Contains,
301    };
302    let one = |name: &str, when: Condition, style: Style, countdown: Option<Countdown>| Offered {
303        name: name.to_owned(),
304        state: ButtonState {
305            when,
306            style,
307            countdown: countdown.into(),
308            command: None,
309        },
310    };
311    vec![
312        one(
313            "Dim during roundtime",
314            Condition::RtActive,
315            dimmed(),
316            Some(Countdown::Roundtime),
317        ),
318        one(
319            "Dim during casttime",
320            Condition::CtActive,
321            dimmed(),
322            Some(Countdown::Casttime),
323        ),
324        one(
325            "Dim when a spell can't be afforded",
326            not_affordable(number),
327            dimmed(),
328            None,
329        ),
330        one(
331            "Count down an effect while it is up",
332            Condition::EffectActive {
333                category: EffectCategory::ActiveSpells,
334                name: String::new(),
335                name_match: NameMatch::Contains,
336            },
337            lit(LIT),
338            Some(effect(EffectCategory::ActiveSpells)),
339        ),
340        one(
341            "Dim while on cooldown",
342            Condition::EffectActive {
343                category: EffectCategory::Cooldowns,
344                name: String::new(),
345                name_match: NameMatch::Contains,
346            },
347            dimmed(),
348            Some(effect(EffectCategory::Cooldowns)),
349        ),
350        one("Light up while hidden", indicator("hidden"), lit(LIT), None),
351        one("Dim while stunned", indicator("stunned"), dimmed(), None),
352        one(
353            "Light up while a stance",

390/// The spell a command casts or prepares, by number: `incant 909`,
391/// `prep 101`, `cast 909 at kobold`, `channel 1700`.
392fn spell_number(command: &str) -> Option<u16> {
393    let mut words = command.split_whitespace();
394    let verb = words.next()?.to_ascii_lowercase();
395    let casts = ["incant", "prep", "prepare", "cast", "channel", "evoke"];
396    if !casts.iter().any(|cast| verb == *cast) {
397        return None;
398    }
399    words.next()?.parse().ok()
400}

402/// The states a button that sends `command` is suggested, none of them in
403/// `has` already: what its command makes likely.
404pub(crate) fn suggested(command: &str, has: &[ButtonState]) -> Vec<Offered> {
405    let verb = command
406        .split_whitespace()
407        .next()
408        .unwrap_or_default()
409        .to_ascii_lowercase();
410    let second = command
411        .split_whitespace()
412        .nth(1)
413        .unwrap_or_default()
414        .to_ascii_lowercase();
415    let mut offers = Vec::new();
416    let offer =
417        |name: String, when: Condition, style: Style, countdown: Option<Countdown>| Offered {
418            name,
419            state: ButtonState {
420                when,
421                style,
422                countdown: countdown.into(),
423                command: None,
424            },
425        };
426    if let Some(number) = spell_number(command) {
427        offers.push(offer(
428            format!("Dim when {} can't be afforded", spell(number)),
429            not_affordable(number),
430            dimmed(),
431            None,
432        ));
433        offers.push(offer(
434            "Dim during casttime".to_owned(),
435            Condition::CtActive,
436            dimmed(),
437            Some(Countdown::Casttime),
438        ));
439        if let Some(found) = hydra_model::spells::spell(number) {
440            offers.push(offer(
441                format!("Count down {} while it is up", found.name),
442                Condition::EffectActive {
443                    category: EffectCategory::ActiveSpells,
444                    name: found.name.clone(),
445                    name_match: NameMatch::Exact,
446                },
447                lit(LIT),
448                Some(Countdown::Effect {
449                    category: EffectCategory::ActiveSpells,
450                    name: found.name.clone(),
451                    name_match: NameMatch::Exact,
452                }),
453            ));
454        }
455    }
456    let lights = [
457        ("hide", "hidden"),
458        ("kneel", "kneeling"),
459        ("sit", "sitting"),
460        ("stand", "standing"),
461        ("lie", "prone"),
462        ("join", "joined"),
463    ];
464    if let Some((_, id)) = lights.iter().find(|(word, _)| verb == *word) {
465        offers.push(offer(
466            format!("Light up while {id}"),
467            indicator(id),
468            lit(LIT),
469            None,
470        ));
471    }
472    if verb == "stance" && !second.is_empty() {
473        offers.push(offer(
474            format!("Light up while in stance {second}"),
475            Condition::Stance { stance: second },
476            lit(LIT),
477            None,
478        ));
479    }
480    let swings = [
481        "attack", "ambush", "fire", "kill", "feint", "mstrike", "hide", "search", "forage",
482    ];
483    if swings.iter().any(|swing| verb == *swing) {
484        offers.push(offer(
485            "Dim during roundtime".to_owned(),
486            Condition::RtActive,
487            dimmed(),
488            Some(Countdown::Roundtime),
489        ));
490    }
491    offers.retain(|offered| !has.iter().any(|state| state.when == offered.state.when));
492    offers
493}

495/// Of `states`, in order, the state above each that holds whenever it
496/// does, so that it never wins: by place, for each state.
497pub(crate) fn shadowed(states: &[ButtonState]) -> Vec<Option<usize>> {
498    states
499        .iter()
500        .enumerate()
501        .map(|(at, state)| {
502            states[..at]
503                .iter()
504                .position(|above| implies(&state.when, &above.when))
505        })
506        .collect()
507}

509/// Whether `below` holding means `above` holds: the same test; `above`
510/// always true; or a vital or an injury past a nearer mark the same way.
511fn implies(below: &Condition, above: &Condition) -> bool {
512    if below == above {
513        return true;
514    }
515    match (below, above) {
516        (_, Condition::All { conditions }) if conditions.is_empty() => true,
517        (
518            Condition::Vital {
519                vital,
520                cmp,
521                value,
522                unit,
523            },
524            Condition::Vital {
525                vital: vital_above,
526                cmp: cmp_above,
527                value: value_above,
528                unit: unit_above,
529            },
530        ) if vital == vital_above && unit == unit_above => {
531            past(*cmp, i64::from(*value), *cmp_above, i64::from(*value_above))
532        }
533        (
534            Condition::Injury { area, cmp, level },
535            Condition::Injury {
536                area: area_above,
537                cmp: cmp_above,
538                level: level_above,
539            },
540        ) if area == area_above => {
541            past(*cmp, i64::from(*level), *cmp_above, i64::from(*level_above))
542        }
543        _ => false,
544    }
545}

547/// Whether every number `cmp value` holds for also holds `cmp_above
548/// value_above`, on whole numbers.
549fn past(cmp: Cmp, value: i64, cmp_above: Cmp, value_above: i64) -> bool {
550    // As bounds on whole numbers: below `< v` is `<= v-1`.
551    let most = |cmp: Cmp, value: i64| match cmp {
552        Cmp::Lt => Some(value - 1),
553        Cmp::Le => Some(value),
554        Cmp::Gt | Cmp::Ge => None,
555    };
556    let least = |cmp: Cmp, value: i64| match cmp {
557        Cmp::Gt => Some(value + 1),
558        Cmp::Ge => Some(value),
559        Cmp::Lt | Cmp::Le => None,
560    };
561    match (most(cmp, value), most(cmp_above, value_above)) {
562        (Some(most), Some(most_above)) => return most <= most_above,
563        (None, None) => {}
564        _ => return false,
565    }
566    match (least(cmp, value), least(cmp_above, value_above)) {
567        (Some(least), Some(least_above)) => least >= least_above,
568        _ => false,
569    }
570}

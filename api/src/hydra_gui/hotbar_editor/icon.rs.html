10/// The icons folder as the editor offers it: its sheets, and its images.
11#[derive(Clone, Debug, Default)]
12pub(crate) struct Folder {
13    pub(crate) dir: PathBuf,
14    pub(crate) sheets: Sheets,
15    /// The images in it, by file name, the sheets' among them.
16    pub(crate) images: Vec<String>,
17}
18
19impl Folder {
20    /// The icons folder `dir` with `sheets`, its images read.
21    pub(crate) fn read(dir: PathBuf, sheets: Sheets) -> Self {
22        let mut images: Vec<String> = std::fs::read_dir(&dir)
23            .into_iter()
24            .flatten()
25            .flatten()
26            .filter_map(|entry| entry.file_name().into_string().ok())
27            .filter(|name| name.to_ascii_lowercase().ends_with(".png"))
28            .collect();
29        images.sort();
30        Self {
31            dir,
32            sheets,
33            images,
34        }
35    }
36}

57/// `icon` edited in `ui`: what it shows ([`source_ui`]), and how it is
58/// drawn over that ([`look_ui`]). `salt` keeps two apart.
59pub(crate) fn icon_ui(ui: &mut egui::Ui, icon: &mut Option<Icon>, folder: &Folder, salt: &str) {
60    ui.push_id(("hotbar-icon", salt), |ui| {
61        source(ui, icon, folder);
62        if let Some(icon) = icon {
63            look(ui, icon);
64        }
65    });
66}

80/// [`source_ui`]'s fields.
81fn source(ui: &mut egui::Ui, icon: &mut Option<Icon>, folder: &Folder) {
82    let kind = match icon.as_ref().map(|icon| &icon.art) {
83        None | Some(Art::None) => Kind::None,
84        Some(Art::Cell { .. }) => Kind::Cell,
85        Some(Art::Image(_)) => Kind::Image,
86    };
87    let mut chose = kind;
88    ui.horizontal(|ui| {
89        ui.radio_value(&mut chose, Kind::None, "none");
90        ui.radio_value(&mut chose, Kind::Cell, "a sheet's cell");
91        ui.radio_value(&mut chose, Kind::Image, "an image");
92    });
93    if chose != kind {
94        let art = match chose {
95            Kind::None => None,
96            Kind::Cell => Some(Art::Cell {
97                sheet: folder
98                    .sheets
99                    .sheets
100                    .keys()
101                    .next()
102                    .cloned()
103                    .unwrap_or_default(),
104                cell: 1,
105            }),
106            Kind::Image => Some(Art::Image(
107                folder.images.first().cloned().unwrap_or_default(),
108            )),
109        };
110        *icon = art.map(|art| match icon.take() {
111            Some(mut kept) => {
112                kept.art = art;
113                kept
114            }
115            None => Icon::of(art),
116        });
117    }
118    let Some(icon) = icon else {
119        return;
120    };
121    match &mut icon.art {
122        Art::Cell { sheet, cell } => cell_ui(ui, (sheet, cell), folder),
123        Art::Image(path) => image_ui(ui, path, folder),
124        Art::None => {}
125    }
126}

128/// [`look_ui`]'s fields.
129fn look(ui: &mut egui::Ui, icon: &mut Icon) {
130    ui.horizontal(|ui| {
131        ui.checkbox(&mut icon.grayscale, "grey");
132        ui.label("border");
133        super::button::colour_ui(ui, &mut icon.border);
134        if icon.border.is_some() {
135            let mut width = icon.border_width.unwrap_or(2);
136            ui.add(egui::DragValue::new(&mut width).range(1..=10).suffix(" px"));
137            icon.border_width = (width != 2).then_some(width);
138            ui.label("to");
139            super::button::colour_ui(ui, &mut icon.border_end);
140            if icon.border_end.is_some() {
141                egui::ComboBox::from_id_salt("hotbar-border-dir")
142                    .selected_text(dir_name(icon.border_dir))
143                    .show_ui(ui, |ui| {
144                        for dir in [
145                            GradientDir::Horizontal,
146                            GradientDir::Vertical,
147                            GradientDir::DiagonalDown,
148                            GradientDir::DiagonalUp,
149                            GradientDir::Radial,
150                            GradientDir::Square,
151                        ] {
152                            ui.selectable_value(&mut icon.border_dir, dir, dir_name(dir));
153                        }
154                    });
155            }
156        }
157    });
158}

160const fn dir_name(dir: GradientDir) -> &'static str {
161    match dir {
162        GradientDir::Horizontal => "across",
163        GradientDir::Vertical => "down",
164        GradientDir::DiagonalDown => "top left to bottom right",
165        GradientDir::DiagonalUp => "bottom left to top right",
166        GradientDir::Radial => "middle out",
167        GradientDir::Square => "middle out, square",
168    }
169}

171/// A sheet's cell: the sheet, and the cell, picked from the sheet drawn.
172fn cell_ui(ui: &mut egui::Ui, (sheet, cell): (&mut String, &mut u32), folder: &Folder) {
173    ui.horizontal(|ui| {
174        if folder.sheets.sheets.is_empty() {
175            ui.weak("No sheets yet: add one on the Icons page.");
176            if ui.button("Icons page").clicked() {
177                crate::editors_menu::ask_editor(ui.ctx(), crate::editors_menu::Editor::Icons);
178            }
179            return;
180        }
181        egui::ComboBox::from_id_salt("hotbar-sheet")
182            .selected_text(sheet.as_str())
183            .show_ui(ui, |ui| {
184                for name in folder.sheets.sheets.keys() {
185                    ui.selectable_value(sheet, name.clone(), name);
186                }
187            });
188        ui.add(egui::DragValue::new(cell).range(1..=9999).prefix("cell "));
189        let Some(found) = folder.sheets.sheets.get(sheet.as_str()) else {
190            ui.colored_label(ui.visuals().warn_fg_color, "icons.toml names no such sheet");
191            return;
192        };
193        let file = folder.dir.join(&found.path).to_string_lossy().into_owned();
194        let edge = found.cell();
195        ui.menu_button("pick", |ui| {
196            let Some(texture) = crate::pictures::picture(ui.ctx(), &file) else {
197                ui.weak(format!("{file} does not load"));
198                return;
199            };
200            let [width, height] = texture.size();
201            let size = (
202                u32::try_from(width).unwrap_or(0),
203                u32::try_from(height).unwrap_or(0),
204            );
205            let across = (size.0 / edge.max(1)).max(1);
206            egui::ScrollArea::vertical()
207                .max_height(360.0)
208                .show(ui, |ui| {
209                    egui::Grid::new("hotbar-sheet-cells").show(ui, |ui| {
210                        let mut at = 1;
211                        while let Some([left, top, right, bottom]) =
212                            hydra_ui::hotbar::cell_in(at, edge, size)
213                        {
214                            let uv = egui::Rect::from_min_max(
215                                egui::pos2(left, top),
216                                egui::pos2(right, bottom),
217                            );
218                            let picture =
219                                egui::Image::new((texture.id(), egui::vec2(32.0, 32.0))).uv(uv);
220                            if ui
221                                .add(egui::Button::image(picture).selected(at == *cell))
222                                .on_hover_text(format!("cell {at}"))
223                                .clicked()
224                            {
225                                *cell = at;
226                                ui.close();
227                            }
228                            if at % across == 0 {
229                                ui.end_row();
230                            }
231                            at += 1;
232                        }
233                    });
234                });
235        });
236    });
237}

239/// An image: a file in the icons folder, picked, or a path to one anywhere.
240fn image_ui(ui: &mut egui::Ui, path: &mut String, folder: &Folder) {
241    ui.horizontal(|ui| {
242        ui.add(
243            egui::TextEdit::singleline(path)
244                .hint_text("a file in the icons folder, or a path")
245                .desired_width(220.0),
246        );
247        if !folder.images.is_empty() {
248            ui.menu_button("pick", |ui| {
249                egui::ScrollArea::vertical()
250                    .max_height(360.0)
251                    .show(ui, |ui| {
252                        for name in &folder.images {
253                            let file = folder.dir.join(name).to_string_lossy().into_owned();
254                            ui.horizontal(|ui| {
255                                if let Some(texture) = crate::pictures::picture(ui.ctx(), &file) {
256                                    ui.add(egui::Image::new((
257                                        texture.id(),
258                                        egui::vec2(24.0, 24.0),
259                                    )));
260                                }
261                                if ui.button(name).clicked() {
262                                    name.clone_into(path);
263                                    ui.close();
264                                }
265                            });
266                        }
267                    });
268            });
269        }
270    });
271}

273/// The icons folder, the Icons page: its sheets, and a sheet or an image
274/// copied in. A file to copy in, when asked.
275pub(super) fn folder_ui(ui: &mut egui::Ui, folder: &Folder, adding: &mut Adding) -> Option<Adding> {
276    let mut asked = None;
277    ui.vertical(|ui| {
278        ui.label(
279            "Pictures for hotbar buttons and indicators. A sheet is one picture cut into square cells, each an icon; any other PNG here is an icon whole.",
280        );
281        ui.weak(format!(
282            "Kept in {}: icons.toml names the sheets.",
283            folder.dir.display()
284        ));
285        ui.add_space(6.0);
286        ui.strong("Sheets");
287        if folder.sheets.sheets.is_empty() {
288            ui.weak("None yet.");
289        }
290        for (name, sheet) in &folder.sheets.sheets {
291            ui.label(format!(
292                "{name}: {} in cells of {} px",
293                sheet.path,
294                sheet.cell()
295            ));
296        }
297        ui.add_space(6.0);
298        ui.strong("Copy one in");
299            ui.horizontal(|ui| {
300                ui.add(
301                    egui::TextEdit::singleline(&mut adding.file)
302                        .hint_text("a PNG file to copy in")
303                        .desired_width(260.0),
304                );
305                ui.add(
306                    egui::TextEdit::singleline(&mut adding.sheet)
307                        .hint_text("its sheet's name; none for one image")
308                        .desired_width(200.0),
309                );
310                if !adding.sheet.trim().is_empty() {
311                    if adding.cell == 0 {
312                        adding.cell = hydra_ui::hotbar::CELL;
313                    }
314                    ui.add(
315                        egui::DragValue::new(&mut adding.cell)
316                            .range(4..=1024)
317                            .suffix(" px cells"),
318                    );
319                }
320                if ui
321                    .add_enabled(!adding.file.trim().is_empty(), egui::Button::new("Copy in"))
322                    .clicked()
323                {
324                    asked = Some(adding.clone());
325                }
326            });
327        });
328    asked

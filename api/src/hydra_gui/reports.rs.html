1//! The Reports window: every report the ledger answers, for any character
2//! it holds, read without typing.
3//!
4//! One window for the process. Its top strip picks the character, the
5//! scope a report runs over (the latest hunt, every hunt, the last so
6//! many, one by id, or between two) and the name a report may take. Down
7//! the left, a search box over the reports' names and what they show, and
8//! the reports under it by family, each a click; the flares recorded for
9//! the character are listed by name under a fold of their own. The pane
10//! shows the report as the command would say it, its links live: a link
11//! runs that report in the pane, and *Back* and *Forward* walk what has
12//! been run. *Export* writes the report shown to a file under the data
13//! folder's `reports`. A hunt closing runs the report shown again, so a
14//! hunt's numbers are current without a click.
15//!
16//! The attack and creature listings are paged: a row above the pane turns
17//! *Recent* on, with how many, sets the rows a page, and walks the pages.
18//!
19//! The reports are the command families' own (`hydra_exp`,
20//! `hydra_combat_log`): the window composes the command, runs it on a
21//! thread of its own against the ledger file, and draws the answer as
22//! blocks (`hydra_report`): a heading, facts, a table sortable by a click
23//! on a header, a chart painted in the theme's colours, a note, or lines
24//! of text with their links live (`draw`). Only the reports: what writes
25//! the ledger is typed, never clicked. Nothing is sent to the game.
26

41/// The scope picked in the strip.
42#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
43enum Pick {
44    /// The latest hunt.
45    #[default]
46    Latest,
47    /// Every hunt.
48    All,
49    /// The last so many.
50    Last,
51    /// One hunt by id.
52    Hunt,
53    /// Between two: hunt ids for combat, times for the experience report
54    /// over a span.
55    Between,
56}
57
58impl Pick {
59    const ALL: [Pick; 5] = [
60        Pick::Latest,
61        Pick::All,
62        Pick::Last,
63        Pick::Hunt,
64        Pick::Between,
65    ];
66
67    const fn label(self) -> &'static str {
68        match self {
69            Pick::Latest => "Latest hunt",
70            Pick::All => "Every hunt",
71            Pick::Last => "Last n hunts",
72            Pick::Hunt => "One hunt",
73            Pick::Between => "Between",
74        }
75    }
76}

78/// A report listed: a family's catalogue entry, or a flare by name.
79#[derive(Clone, Debug, PartialEq, Eq)]
80enum Listed {
81    Exp(&'static hydra_exp::Entry),
82    Combat(&'static hydra_combat_log::Entry),
83    Rewards(&'static hydra_rewards::Entry),
84    Silver(&'static hydra_silver::Entry),
85    Loot(&'static hydra_loot::Entry),
86    Flare(String),
87}
88
89impl Listed {
90    fn name(&self) -> &str {
91        match self {
92            Listed::Exp(entry) => entry.name,
93            Listed::Combat(entry) => entry.name,
94            Listed::Rewards(entry) => entry.name,
95            Listed::Silver(entry) => entry.name,
96            Listed::Loot(entry) => entry.name,
97            Listed::Flare(name) => name,
98        }
99    }
100
101    fn blurb(&self) -> String {
102        match self {
103            Listed::Exp(entry) => entry.blurb.to_owned(),
104            Listed::Combat(entry) => entry.blurb.to_owned(),
105            Listed::Rewards(entry) => entry.blurb.to_owned(),
106            Listed::Silver(entry) => entry.blurb.to_owned(),
107            Listed::Loot(entry) => entry.blurb.to_owned(),
108            Listed::Flare(name) => format!(
109                "The {name} flare: its proc rate, spread, statuses and damage; a barrier's, \
110                 what it stopped"
111            ),
112        }
113    }
114
115    fn matches(&self, query: &str) -> bool {
116        match self {
117            Listed::Exp(entry) => entry.matches(query),
118            Listed::Combat(entry) => entry.matches(query),
119            Listed::Rewards(entry) => entry.matches(query),
120            Listed::Silver(entry) => entry.matches(query),
121            Listed::Loot(entry) => entry.matches(query),
122            Listed::Flare(name) => {
123                let text = format!("flare {name}").to_lowercase();
124                query
125                    .split_whitespace()
126                    .all(|word| text.contains(&word.to_lowercase()))
127            }
128        }
129    }
130
131    /// A paged listing: the attacks or the creatures.
132    fn paged(&self) -> bool {
133        matches!(self, Listed::Combat(entry) if entry.words == "attacks" || entry.words == "creatures")
134    }
135}

137/// A read the window wants done, never on its thread.
138#[derive(Clone, Debug, PartialEq, Eq)]
139pub(crate) enum Ask {
140    /// The characters the ledger holds.
141    Characters,
142    /// The flares recorded for this character, by name.
143    Flares(Who),
144    /// A report: the command line, for this character.
145    Report(Who, String),
146    /// The report shown written to a file: whose, its command, its text.
147    Export(Who, String, String),
148}
149
150/// What a read answered.
151#[derive(Clone, Debug)]
152pub(crate) enum Reply {
153    /// The characters, or why not.
154    Characters(Result<Vec<Who>, String>),
155    /// The flares recorded for this character.
156    Flares(Who, Vec<String>),
157    /// A report's blocks, or what was wrong, for the command asked.
158    Report(Who, String, Result<Report, String>),
159    /// Where an export went, or what was wrong.
160    Exported(Result<std::path::PathBuf, String>),
161}

163/// Where answers land for the window to take: shared with the read.
164pub(crate) type Inbox = Arc<Mutex<Vec<Reply>>>;

258    /// Open the window, for `who` when one is named; the characters are
259    /// read again each time it opens.
260    pub(crate) fn open_for(&mut self, who: Option<Who>) {
261        self.open = true;
262        if who.is_some() {
263            self.who = who;
264        }
265        self.asked.push(Ask::Characters);
266        if let Some(who) = &self.who {
267            self.asked.push(Ask::Flares(who.clone()));
268        }
269    }

281    /// What it wants read, taken: each is counted as waiting.
282    pub(crate) fn take_asks(&mut self) -> Vec<Ask> {
283        let asked = std::mem::take(&mut self.asked);
284        self.waiting += asked.len();
285        asked

288    /// Take what the reads answered.
289    pub(crate) fn take_replies(&mut self) {
290        let replies: Vec<Reply> = std::mem::take(&mut *crate::logs::lock(&self.inbox));
291        for reply in replies {
292            self.waiting = self.waiting.saturating_sub(1);
293            match reply {
294                Reply::Characters(Ok(characters)) => {
295                    if self.who.is_none() {
296                        self.who = characters.first().cloned();
297                        if let Some(who) = &self.who {
298                            self.asked.push(Ask::Flares(who.clone()));
299                        }
300                    }
301                    self.characters = characters;
302                }
303                Reply::Characters(Err(why)) => {
304                    self.shown = Some(Shown {
305                        command: String::new(),
306                        report: Err(why),
307                        pages: None,
308                        sort: None,
309                    });
310                }
311                Reply::Flares(who, flares) => {
312                    if self.who.as_ref() == Some(&who) {
313                        self.flares = flares;
314                    }
315                }
316                Reply::Exported(Ok(path)) => {
317                    self.said = Some(format!("Written to {}", path.display()));
318                }
319                Reply::Exported(Err(why)) => self.said = Some(why),
320                Reply::Report(who, command, report) => {
321                    // Only the answer to what the pane waits for is shown.
322                    if self.who.as_ref() == Some(&who)
323                        && self.history.get(self.at) == Some(&command)
324                    {
325                        let pages = report
326                            .as_ref()
327                            .ok()
328                            .and_then(|report| hydra_combat_log::paged(&report.text()));
329                        self.shown = Some(Shown {
330                            command,
331                            report,
332                            pages,
333                            sort: None,
334                        });
335                    }
336                }
337            }
338        }
339    }

341    /// Run `command` for the character picked: it goes on the history
342    /// after what is shown, and the pane waits for it.
343    fn run(&mut self, command: String) {
344        let Some(who) = self.who.clone() else {
345            return;
346        };
347        if self.history.get(self.at) != Some(&command) {
348            self.history.truncate(self.at + 1);
349            self.history.push(command.clone());
350            self.at = self.history.len() - 1;
351        }
352        self.asked.push(Ask::Report(who, command));
353    }

355    /// A hunt closed somewhere (`hunts_closed` the count over every seat):
356    /// the report shown is run again.
357    pub(crate) fn hunts_closed(&mut self, hunts_closed: u64) {
358        if self.hunts_closed != hunts_closed {
359            self.hunts_closed = hunts_closed;
360            if self.shown.is_some() {
361                self.run_at(self.at);
362            }
363        }
364    }

366    /// Ask for the report shown written to a file.
367    fn export(&mut self) {
368        let (Some(who), Some(shown)) = (self.who.clone(), self.shown.as_ref()) else {
369            return;
370        };
371        let Ok(report) = &shown.report else {
372            return;
373        };
374        // The text rendering, its links as their words alone.
375        let text: String = hydra_combat_log::linked(&report.text())
376            .iter()
377            .map(|line| {
378                line.iter()
379                    .map(|span| span.text.as_str())
380                    .collect::<String>()
381                    + "\n"
382            })
383            .collect();
384        self.asked
385            .push(Ask::Export(who, shown.command.clone(), text));
386    }

388    /// Run what the history holds at `at` again.
389    fn run_at(&mut self, at: usize) {
390        if let (Some(who), Some(command)) = (self.who.clone(), self.history.get(at).cloned()) {
391            self.at = at;
392            self.asked.push(Ask::Report(who, command));
393        }
394    }

396    /// The scope words the strip holds, for a combat report.
397    fn scope_words(&self) -> String {
398        match self.pick {
399            Pick::Latest => String::new(),
400            Pick::All => "all".to_owned(),
401            Pick::Last => format!("last {}", self.count.trim()),
402            Pick::Hunt => self.hunt.trim().to_owned(),
403            Pick::Between => format!("between {} {}", self.from.trim(), self.to.trim()),
404        }

407    /// The report shown, asked again over the scope the strip holds: for a
408    /// combat report that takes one. `None` for any other, or when the
409    /// scope would not change it.
410    fn rescoped(&self) -> Option<String> {
411        use hydra_combat_log::Takes;
412        let shown = self.history.get(self.at)?;
413        let rest = shown.strip_prefix("cstats")?.trim();
414        // The entry whose words the shown command starts with, the
415        // longest when two would.
416        let entry = hydra_combat_log::CATALOG
417            .iter()
418            .filter(|e| e.takes == Takes::Scope)
419            .filter(|e| rest == e.words || rest.starts_with(&format!("{} ", e.words)))
420            .max_by_key(|e| e.words.len())?;
421        let command = self.command_for(&Listed::Combat(entry));
422        (command != *shown).then_some(command)
423    }

425    /// Ask the report shown again when the scope changed under it.
426    fn rescope(&mut self) {
427        if let Some(command) = self.rescoped() {
428            self.run(command);
429        }
430    }

432    /// The listing words the row holds, at `page`.
433    fn listing_words(&self, page: usize) -> String {
434        Listing {
435            scope: hydra_combat_log::report::Scope::Latest,
436            recent: self
437                .recent
438                .then(|| self.recent_n.trim().parse::<usize>().ok()),
439            page,
440            per: self
441                .per
442                .trim()
443                .parse()
444                .unwrap_or(hydra_combat_log::PER_PAGE),
445        }
446        .words()
447    }

449    /// The command `listed` asks for with what the strip holds.
450    fn command_for(&self, listed: &Listed) -> String {
451        match listed {
452            Listed::Exp(entry) => {
453                let argument = match entry.takes {
454                    hydra_exp::Takes::Nothing => String::new(),
455                    hydra_exp::Takes::Span => {
456                        format!("\"{}\" \"{}\"", self.from.trim(), self.to.trim())
457                    }
458                };
459                entry.command(&argument)
460            }
461            Listed::Combat(entry) => {
462                use hydra_combat_log::Takes;
463                let argument = match entry.takes {
464                    Takes::Nothing => String::new(),
465                    Takes::Scope => self.scope_words(),
466                    Takes::Name => self.name.trim().to_owned(),
467                };
468                let mut command = entry.command(&argument);
469                if listed.paged() {
470                    let words = self.listing_words(1);
471                    if !words.is_empty() {
472                        command.push(' ');
473                        command.push_str(&words);
474                    }
475                }
476                if is_ucs(&command) {
477                    command = with_ucs_show(&command, self.ucs_show);
478                }
479                with_both(&command, self.both)
480            }
481            Listed::Rewards(entry) => entry.command(),
482            Listed::Silver(entry) => entry.command(),
483            Listed::Loot(entry) => entry.command(),
484            Listed::Flare(name) => with_both(&format!("cstats flare {name}"), self.both),
485        }
486    }

580    /// The scope's picker and its count, hunt or range: whether the scope
581    /// changed or one of them finished being typed.

595            Pick::Last => {

598                    .on_hover_text("How many hunts")

604                    .on_hover_text("The hunt's id")
605                    .lost_focus();

607            Pick::Between => {

609                             report over a span, 13:05 today, 07-20 13:05 this year, \

611                ui.label("from");
612                rescope |= ui
613                    .add(egui::TextEdit::singleline(&mut self.from).desired_width(120.0))

627    /// The top strip: the character, the scope and the name, and the way
628    /// back and forward.
629    fn strip(&mut self, ui: &mut egui::Ui) {
630        egui::Panel::top(egui::Id::new("reports-strip")).show(ui, |ui| {
631            ui.horizontal_wrapped(|ui| {
632                let chosen = self
633                    .who
634                    .as_ref()
635                    .map_or_else(|| "No character".to_owned(), label);
636                egui::ComboBox::from_id_salt("reports-who")
637                    .selected_text(chosen)
638                    .show_ui(ui, |ui| {
639                        for who in self.characters.clone() {
640                            let text = label(&who);
641                            if ui
642                                .selectable_label(self.who.as_ref() == Some(&who), text)
643                                .clicked()
644                            {
645                                self.who = Some(who.clone());
646                                self.flares.clear();
647                                self.asked.push(Ask::Flares(who));
648                                self.run_at(self.at);
649                            }
650                        }
651                    });
652                if self.scope_controls(ui) {
653                    self.rescope();
654                }
655                self.both_box(ui);
656                self.ucs_box(ui);
657                ui.label("Name");
658                ui.add(egui::TextEdit::singleline(&mut self.name).desired_width(140.0))
659                    .on_hover_text("A creature's noun, for the reports that take one");
660                if ui
661                    .add_enabled(self.at > 0, egui::Button::new("Back"))
662                    .clicked()
663                {
664                    self.run_at(self.at - 1);
665                }
666                if ui
667                    .add_enabled(
668                        self.at + 1 < self.history.len(),
669                        egui::Button::new("Forward"),
670                    )
671                    .clicked()
672                {

760    /// The row above a listing: *Recent* and how many, the rows a page,
761    /// and the pages walked. Each change runs the listing again from its
762    /// first page; a page walked keeps the rest.
763    fn listing_row(&mut self, ui: &mut egui::Ui, base: &str, pages: Option<(usize, usize)>) {
764        let mut rerun = None;
765        ui.horizontal(|ui| {
766            if ui
767                .checkbox(&mut self.recent, "Recent")
768                .on_hover_text("Newest first; with a number, only the last so many")
769                .changed()

772            }
773            if ui
774                .add_enabled(
775                    self.recent,
776                    egui::TextEdit::singleline(&mut self.recent_n).desired_width(50.0),
777                )
778                .on_hover_text("The last so many; empty for every one, paged")
779                .lost_focus()
780            {
781                rerun = Some(1);
782            }
783            ui.label("Rows a page");
784            if ui
785                .add(egui::TextEdit::singleline(&mut self.per).desired_width(40.0))
786                .lost_focus()
787            {
788                rerun = Some(1);
789            }
790            let (page, last) = pages.unwrap_or((1, 1));
791            if ui
792                .add_enabled(page > 1, egui::Button::new("Prev"))
793                .clicked()
794            {
795                rerun = Some(page - 1);
796            }
797            ui.label(format!("page {page} of {last}"));
798            if ui
799                .add_enabled(page < last, egui::Button::new("Next"))
800                .clicked()
801            {
802                rerun = Some(page + 1);
803            }
804        });
805        if let Some(page) = rerun {
806            let words = self.listing_words(page);
807            let command = if words.is_empty() {
808                base.to_owned()
809            } else {
810                format!("{base} {words}")
811            };
812            self.run(command);
813        }
814    }

820                "No character has anything recorded yet."
821            } else {
822                "Pick a report on the left."

824            return;
825        };

828        if let Some(base) = listing_base(&shown.command) {
829            let pages = shown.pages;
830            self.listing_row(ui, &base, pages);
831        }
832        let Some(shown) = &self.shown else {

834        };
835        ui.separator();
836        let mut sort = shown.sort;

838            Err(why) => {

841            Ok(report) => {
842                clicked = draw::report(ui, report, &mut sort);
843            }
844        });
845        if let Some(shown) = &mut self.shown {
846            shown.sort = sort;
847        }
848        if let Some(command) = clicked {
849            self.run(command);
850        }
851    }

877/// Whether `command` is UCS positioning.
878fn is_ucs(command: &str) -> bool {
879    let mut words = command.split_whitespace();
880    words.next() == Some("cstats")
881        && words
882            .next()
883            .is_some_and(|w| w.eq_ignore_ascii_case("ucs") || w.eq_ignore_ascii_case("positioning"))
884}

886/// UCS positioning's `command` showing `show`: its attack word replaced,
887/// or taken out for each attack; `both` kept last.
888fn with_ucs_show(command: &str, show: Option<&str>) -> String {
889    let mut words: Vec<&str> = command
890        .split_whitespace()
891        .filter(|w| !hydra_combat_log::ucs::SHOWN.contains(&w.to_ascii_lowercase().as_str()))
892        .collect();
893    let both = words.iter().any(|w| w.eq_ignore_ascii_case("both"));
894    if let Some(show) = show {
895        words.push(show);
896    }
897    with_both(&words.join(" "), both)
898}

900/// `command` with `both` said once at its end when `on`, and not at all
901/// when off.
902fn with_both(command: &str, on: bool) -> String {
903    let mut words: Vec<&str> = command
904        .split_whitespace()
905        .filter(|w| !w.eq_ignore_ascii_case("both"))
906        .collect();
907    if on {
908        words.push("both");
909    }
910    words.join(" ")
911}

918/// A listing's command without its listing words, when `command` is one:
919/// `cstats attacks all recent 5 page 2` is `cstats attacks all`.
920fn listing_base(command: &str) -> Option<String> {
921    let mut words = command.split_whitespace();
922    let family = words.next()?;
923    let kind = words.next()?;
924    if !family.eq_ignore_ascii_case("cstats")
925        || !(kind.eq_ignore_ascii_case("attacks") || kind.eq_ignore_ascii_case("creatures"))
926    {
927        return None;
928    }
929    let mut base = vec![family, kind];
930    let mut words = words.peekable();
931    while let Some(word) = words.next() {
932        match word.to_ascii_lowercase().as_str() {
933            "recent" | "page" | "per" => {
934                if words.peek().is_some_and(|w| w.parse::<usize>().is_ok()) {
935                    words.next();
936                }
937            }
938            _ => base.push(word),
939        }
940    }
941    Some(base.join(" "))
942}

944/// Do `ask` against the ledger under `data`. Blocking file I/O: never on
945/// the window's thread.
946pub(crate) fn run(data: &Path, ask: Ask) -> Reply {
947    let ledger = Ledger::open(Ledger::path_in(data)).map_err(|e| e.to_string());
948    match ask {
949        Ask::Characters => Reply::Characters(ledger.and_then(|ledger| {
950            ledger
951                .reader()
952                .and_then(|reader| reader.characters())
953                .map_err(|e| e.to_string())
954        })),
955        Ask::Flares(who) => {
956            let names = ledger
957                .and_then(|ledger| ledger.reader().map_err(|e| e.to_string()))
958                .and_then(|reader| {
959                    hydra_combat_log::flare_names(reader.connection(), &who)
960                        .map_err(|e| e.to_string())
961                })
962                .unwrap_or_default();
963            Reply::Flares(who, names)
964        }
965        Ask::Report(who, command) => {
966            let now = i64::try_from(
967                std::time::SystemTime::now()
968                    .duration_since(std::time::UNIX_EPOCH)
969                    .map_or(0, |d| d.as_secs()),
970            )
971            .unwrap_or(0);
972            let report = ledger.and_then(|ledger| {
973                let word = command.split_whitespace().next().unwrap_or("");
974                if word.eq_ignore_ascii_case("exp") {
975                    hydra_exp::run_report(&ledger, &who, &command, now)
976                } else if word.eq_ignore_ascii_case("rewards") {
977                    hydra_rewards::run_report(&ledger, &who, &command, now)
978                } else if word.eq_ignore_ascii_case("silver") {
979                    hydra_silver::run_report(&ledger, &who, &command, now)
980                } else if word.eq_ignore_ascii_case("loot") {
981                    hydra_loot::run(&ledger, &who, &command, now)
982                        .map(|lines| hydra_combat_log::Out::from_text(&lines))
983                } else {
984                    hydra_combat_log::run_report(&ledger, &who, &command)
985                }
986            });
987            Reply::Report(who, command, report)
988        }
989        Ask::Export(who, command, text) => Reply::Exported(export(data, &who, &command, &text)),
990    }
991}

993/// Write `text` under the data folder's `reports`, named by the character,
994/// the command's words and the moment:
995/// `Nisugi-cstats_intercepts_all-20261014T155012.txt`.
996fn export(data: &Path, who: &Who, command: &str, text: &str) -> Result<std::path::PathBuf, String> {
997    let words: String = command
998        .split_whitespace()
999        .collect::<Vec<_>>()
1000        .join("_")
1001        .chars()
1002        .map(|c| {
1003            if c.is_alphanumeric() || c == '_' || c == '-' {
1004                c
1005            } else {
1006                '-'
1007            }
1008        })
1009        .collect();
1010    let now = hydra_net::eastern::now();
1011    let (y, m, d) = hydra_net::eastern::date(now);
1012    let secs = (now + hydra_net::eastern::offset(now)).rem_euclid(hydra_net::eastern::DAY);
1013    let stamp = format!(
1014        "{y:04}{m:02}{d:02}T{:02}{:02}{:02}",
1015        secs / 3600,
1016        secs % 3600 / 60,
1017        secs % 60
1018    );
1019    let folder = data.join("reports");
1020    std::fs::create_dir_all(&folder).map_err(|e| format!("{}: {e}", folder.display()))?;
1021    let path = folder.join(format!("{}-{words}-{stamp}.txt", who.character));
1022    std::fs::write(&path, text).map_err(|e| format!("{}: {e}", path.display()))?;
1023    Ok(path)

1//! The Reports window: every report the ledger answers, for any character
2//! it holds, read without typing.
3//!
4//! One window for the process. Its top strip picks the character, the
5//! scope a report runs over (the latest hunt, every hunt, the last so
6//! many, one by id, or between two) and the name a report may take. Down
7//! the left, a search box over the reports' names and what they show, and
8//! the reports under it by family, each a click; the flares recorded for
9//! the character are listed by name under a fold of their own. The pane
10//! shows the report as the command would say it, its links live: a link
11//! runs that report in the pane, and *Back* and *Forward* walk what has
12//! been run. *Export* writes the report shown to a file under the data
13//! folder's `reports`. A hunt closing runs the report shown again, so a
14//! hunt's numbers are current without a click.
15//!
16//! The attack and creature listings are paged: a row above the pane turns
17//! *Recent* on, with how many, sets the rows a page, and walks the pages.
18//!
19//! The reports are the command families' own (`hydra_exp`,
20//! `hydra_combat_log`): the window composes the command, runs it on a
21//! thread of its own against the ledger file, and draws the answer. Only
22//! the reports: what writes the ledger is typed, never clicked. Nothing is
23//! sent to the game.
24

40/// The scope picked in the strip.
41#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
42enum Pick {
43    /// The latest hunt.
44    #[default]
45    Latest,
46    /// Every hunt.
47    All,
48    /// The last so many.
49    Last,
50    /// One hunt by id.
51    Hunt,
52    /// Between two: hunt ids for combat, times for the experience report
53    /// over a span.
54    Between,
55}
56
57impl Pick {
58    const ALL: [Pick; 5] = [
59        Pick::Latest,
60        Pick::All,
61        Pick::Last,
62        Pick::Hunt,
63        Pick::Between,
64    ];
65
66    const fn label(self) -> &'static str {
67        match self {
68            Pick::Latest => "Latest hunt",
69            Pick::All => "Every hunt",
70            Pick::Last => "Last n hunts",
71            Pick::Hunt => "One hunt",
72            Pick::Between => "Between",
73        }
74    }
75}

77/// A report listed: a family's catalogue entry, or a flare by name.
78#[derive(Clone, Debug, PartialEq, Eq)]
79enum Listed {
80    Exp(&'static hydra_exp::Entry),
81    Combat(&'static hydra_combat_log::Entry),
82    Flare(String),
83}
84
85impl Listed {
86    fn name(&self) -> &str {
87        match self {
88            Listed::Exp(entry) => entry.name,
89            Listed::Combat(entry) => entry.name,
90            Listed::Flare(name) => name,
91        }
92    }
93
94    fn blurb(&self) -> String {
95        match self {
96            Listed::Exp(entry) => entry.blurb.to_owned(),
97            Listed::Combat(entry) => entry.blurb.to_owned(),
98            Listed::Flare(name) => format!(
99                "The {name} flare: its proc rate, spread, statuses and damage; a barrier's, \
100                 what it stopped"
101            ),
102        }
103    }
104
105    fn matches(&self, query: &str) -> bool {
106        match self {
107            Listed::Exp(entry) => entry.matches(query),
108            Listed::Combat(entry) => entry.matches(query),
109            Listed::Flare(name) => {
110                let text = format!("flare {name}").to_lowercase();
111                query
112                    .split_whitespace()
113                    .all(|word| text.contains(&word.to_lowercase()))
114            }
115        }
116    }
117
118    /// A paged listing: the attacks or the creatures.
119    fn paged(&self) -> bool {
120        matches!(self, Listed::Combat(entry) if entry.words == "attacks" || entry.words == "creatures")
121    }
122}

124/// A read the window wants done, never on its thread.
125#[derive(Clone, Debug, PartialEq, Eq)]
126pub(crate) enum Ask {
127    /// The characters the ledger holds.
128    Characters,
129    /// The flares recorded for this character, by name.
130    Flares(Who),
131    /// A report: the command line, for this character.
132    Report(Who, String),
133    /// The report shown written to a file: whose, its command, its text.
134    Export(Who, String, String),
135}
136
137/// What a read answered.
138#[derive(Clone, Debug)]
139pub(crate) enum Reply {
140    /// The characters, or why not.
141    Characters(Result<Vec<Who>, String>),
142    /// The flares recorded for this character.
143    Flares(Who, Vec<String>),
144    /// A report's lines, or what was wrong, for the command asked.
145    Report(Who, String, Result<Vec<String>, String>),
146    /// Where an export went, or what was wrong.
147    Exported(Result<std::path::PathBuf, String>),
148}
149
150/// Where answers land for the window to take: shared with the read.
151pub(crate) type Inbox = Arc<Mutex<Vec<Reply>>>;

153/// What the pane shows.
154#[derive(Debug)]
155struct Shown {
156    /// The command it answers.
157    command: String,
158    /// Its lines as spans, or what was wrong.
159    lines: Result<Vec<Vec<Span>>, String>,
160    /// The page and the pages, when it is a page of a listing.
161    pages: Option<(usize, usize)>,
162}

237    /// Open the window, for `who` when one is named; the characters are
238    /// read again each time it opens.
239    pub(crate) fn open_for(&mut self, who: Option<Who>) {
240        self.open = true;
241        if who.is_some() {
242            self.who = who;
243        }
244        self.asked.push(Ask::Characters);
245        if let Some(who) = &self.who {
246            self.asked.push(Ask::Flares(who.clone()));
247        }
248    }

250    /// Whose reports the window shows.
251    #[cfg(test)]
252    pub(crate) fn who(&self) -> Option<&Who> {
253        self.who.as_ref()
254    }

256    pub(crate) fn inbox(&self) -> Inbox {
257        Arc::clone(&self.inbox)

260    /// What it wants read, taken: each is counted as waiting.
261    pub(crate) fn take_asks(&mut self) -> Vec<Ask> {
262        let asked = std::mem::take(&mut self.asked);
263        self.waiting += asked.len();
264        asked
265    }

267    /// Take what the reads answered.
268    pub(crate) fn take_replies(&mut self) {
269        let replies: Vec<Reply> = std::mem::take(&mut *crate::logs::lock(&self.inbox));
270        for reply in replies {
271            self.waiting = self.waiting.saturating_sub(1);
272            match reply {
273                Reply::Characters(Ok(characters)) => {
274                    if self.who.is_none() {
275                        self.who = characters.first().cloned();
276                        if let Some(who) = &self.who {
277                            self.asked.push(Ask::Flares(who.clone()));
278                        }
279                    }
280                    self.characters = characters;
281                }
282                Reply::Characters(Err(why)) => {
283                    self.shown = Some(Shown {
284                        command: String::new(),
285                        lines: Err(why),
286                        pages: None,
287                    });
288                }
289                Reply::Flares(who, flares) => {
290                    if self.who.as_ref() == Some(&who) {
291                        self.flares = flares;
292                    }
293                }
294                Reply::Exported(Ok(path)) => {
295                    self.said = Some(format!("Written to {}", path.display()));
296                }
297                Reply::Exported(Err(why)) => self.said = Some(why),
298                Reply::Report(who, command, lines) => {
299                    // Only the answer to what the pane waits for is shown.
300                    if self.who.as_ref() == Some(&who)
301                        && self.history.get(self.at) == Some(&command)
302                    {
303                        let pages = lines.as_deref().ok().and_then(hydra_combat_log::paged);
304                        self.shown = Some(Shown {
305                            command,
306                            lines: lines.map(|lines| hydra_combat_log::linked(&lines)),
307                            pages,
308                        });
309                    }
310                }
311            }
312        }
313    }

315    /// Run `command` for the character picked: it goes on the history
316    /// after what is shown, and the pane waits for it.
317    fn run(&mut self, command: String) {
318        let Some(who) = self.who.clone() else {
319            return;
320        };
321        if self.history.get(self.at) != Some(&command) {
322            self.history.truncate(self.at + 1);
323            self.history.push(command.clone());
324            self.at = self.history.len() - 1;
325        }
326        self.asked.push(Ask::Report(who, command));
327    }
328
329    /// A hunt closed somewhere (`hunts_closed` the count over every seat):
330    /// the report shown is run again.
331    pub(crate) fn hunts_closed(&mut self, hunts_closed: u64) {
332        if self.hunts_closed != hunts_closed {
333            self.hunts_closed = hunts_closed;
334            if self.shown.is_some() {
335                self.run_at(self.at);
336            }
337        }
338    }

340    /// Ask for the report shown written to a file.
341    fn export(&mut self) {
342        let (Some(who), Some(shown)) = (self.who.clone(), self.shown.as_ref()) else {
343            return;
344        };
345        let Ok(lines) = &shown.lines else {
346            return;
347        };
348        let text: String = lines
349            .iter()
350            .map(|line| {
351                line.iter()
352                    .map(|span| span.text.as_str())
353                    .collect::<String>()
354                    + "\n"
355            })
356            .collect();
357        self.asked
358            .push(Ask::Export(who, shown.command.clone(), text));
359    }

361    /// Run what the history holds at `at` again.
362    fn run_at(&mut self, at: usize) {
363        if let (Some(who), Some(command)) = (self.who.clone(), self.history.get(at).cloned()) {
364            self.at = at;
365            self.asked.push(Ask::Report(who, command));
366        }
367    }

369    /// The scope words the strip holds, for a combat report.
370    fn scope_words(&self) -> String {
371        match self.pick {
372            Pick::Latest => String::new(),
373            Pick::All => "all".to_owned(),
374            Pick::Last => format!("last {}", self.count.trim()),
375            Pick::Hunt => self.hunt.trim().to_owned(),
376            Pick::Between => format!("between {} {}", self.from.trim(), self.to.trim()),
377        }
378    }

380    /// The listing words the row holds, at `page`.
381    fn listing_words(&self, page: usize) -> String {
382        Listing {
383            scope: hydra_combat_log::report::Scope::Latest,
384            recent: self
385                .recent
386                .then(|| self.recent_n.trim().parse::<usize>().ok()),
387            page,
388            per: self
389                .per
390                .trim()
391                .parse()
392                .unwrap_or(hydra_combat_log::PER_PAGE),
393        }
394        .words()
395    }

397    /// The command `listed` asks for with what the strip holds.
398    fn command_for(&self, listed: &Listed) -> String {
399        match listed {
400            Listed::Exp(entry) => {
401                let argument = match entry.takes {
402                    hydra_exp::Takes::Nothing => String::new(),
403                    hydra_exp::Takes::Span => {
404                        format!("\"{}\" \"{}\"", self.from.trim(), self.to.trim())
405                    }
406                };
407                entry.command(&argument)
408            }
409            Listed::Combat(entry) => {
410                use hydra_combat_log::Takes;
411                let argument = match entry.takes {
412                    Takes::Nothing => String::new(),
413                    Takes::Scope => self.scope_words(),
414                    Takes::Name => self.name.trim().to_owned(),
415                };
416                let mut command = entry.command(&argument);
417                if listed.paged() {
418                    let words = self.listing_words(1);
419                    if !words.is_empty() {
420                        command.push(' ');
421                        command.push_str(&words);
422                    }
423                }
424                command
425            }
426            Listed::Flare(name) => format!("cstats flare {name}"),
427        }
428    }

462    /// The top strip: the character, the scope and the name, and the way
463    /// back and forward.
464    fn strip(&mut self, ui: &mut egui::Ui) {
465        egui::Panel::top(egui::Id::new("reports-strip")).show(ui, |ui| {
466            ui.horizontal_wrapped(|ui| {
467                let chosen = self
468                    .who
469                    .as_ref()
470                    .map_or_else(|| "No character".to_owned(), label);
471                egui::ComboBox::from_id_salt("reports-who")
472                    .selected_text(chosen)
473                    .show_ui(ui, |ui| {
474                        for who in self.characters.clone() {
475                            let text = label(&who);
476                            if ui
477                                .selectable_label(self.who.as_ref() == Some(&who), text)
478                                .clicked()
479                            {
480                                self.who = Some(who.clone());
481                                self.flares.clear();
482                                self.asked.push(Ask::Flares(who));
483                                self.run_at(self.at);
484                            }
485                        }
486                    });
487                egui::ComboBox::from_id_salt("reports-scope")
488                    .selected_text(self.pick.label())
489                    .show_ui(ui, |ui| {
490                        for pick in Pick::ALL {
491                            ui.selectable_value(&mut self.pick, pick, pick.label());
492                        }
493                    });
494                match self.pick {
495                    Pick::Last => {
496                        ui.add(egui::TextEdit::singleline(&mut self.count).desired_width(40.0))
497                            .on_hover_text("How many hunts");
498                    }
499                    Pick::Hunt => {
500                        ui.add(egui::TextEdit::singleline(&mut self.hunt).desired_width(60.0))
501                            .on_hover_text("The hunt's id");
502                    }
503                    Pick::Between => {
504                        let hover = "A hunt id for the combat reports; for the experience \
505                                     report over a span, 13:05 today, 07-20 13:05 this year, \
506                                     or 2026-07-20 13:05:52";
507                        ui.label("from");
508                        ui.add(egui::TextEdit::singleline(&mut self.from).desired_width(120.0))
509                            .on_hover_text(hover);
510                        ui.label("to");
511                        ui.add(egui::TextEdit::singleline(&mut self.to).desired_width(120.0))
512                            .on_hover_text(hover);
513                    }
514                    Pick::Latest | Pick::All => {}
515                }
516                ui.label("Name");
517                ui.add(egui::TextEdit::singleline(&mut self.name).desired_width(140.0))
518                    .on_hover_text("A creature's noun, for the reports that take one");
519                if ui
520                    .add_enabled(self.at > 0, egui::Button::new("Back"))
521                    .clicked()
522                {
523                    self.run_at(self.at - 1);
524                }
525                if ui
526                    .add_enabled(
527                        self.at + 1 < self.history.len(),
528                        egui::Button::new("Forward"),
529                    )
530                    .clicked()
531                {
532                    self.run_at(self.at + 1);
533                }
534                if ui
535                    .add_enabled(!self.history.is_empty(), egui::Button::new("Refresh"))
536                    .on_hover_text("Run the report shown again")
537                    .clicked()
538                {
539                    self.run_at(self.at);
540                }
541                let exportable = self.shown.as_ref().is_some_and(|shown| shown.lines.is_ok());
542                if ui
543                    .add_enabled(exportable, egui::Button::new("Export"))
544                    .on_hover_text("Write the report shown to a file in the data folder's reports")
545                    .clicked()
546                {
547                    self.export();
548                }
549                if self.waiting > 0 {
550                    ui.spinner();
551                }
552                if let Some(said) = &self.said {
553                    ui.weak(said);
554                }
555            });
556        });
557    }

559    /// The left column: the search box, and the reports that fit it, by
560    /// family. Enter in the box runs the first that fits.
561    fn list(&mut self, ui: &mut egui::Ui) {
562        let search = ui.add(
563            egui::TextEdit::singleline(&mut self.search)
564                .hint_text("Find a report")
565                .desired_width(f32::INFINITY),
566        );
567        let entered = search.lost_focus() && ui.input(|input| input.key_pressed(egui::Key::Enter));
568        let query = self.search.trim().to_owned();
569        let families: Vec<(&'static str, Vec<Listed>)> = self
570            .families()
571            .into_iter()
572            .map(|(family, listed)| {
573                let fits = listed
574                    .into_iter()
575                    .filter(|listed| query.is_empty() || listed.matches(&query))
576                    .collect();
577                (family, fits)
578            })
579            .collect();
580        if entered && let Some(first) = families.iter().flat_map(|(_, fits)| fits).next() {
581            let command = self.command_for(first);
582            self.run(command);
583        }
584        egui::ScrollArea::vertical().show(ui, |ui| {
585            let mut any = false;
586            for (family, fits) in &families {
587                if fits.is_empty() {
588                    continue;
589                }
590                any = true;
591                egui::CollapsingHeader::new(*family)
592                    .default_open(true)
593                    .show(ui, |ui| {
594                        for listed in fits {
595                            let command = self.command_for(listed);
596                            let shown = self
597                                .shown
598                                .as_ref()
599                                .is_some_and(|shown| shown.command == command);
600                            if ui
601                                .selectable_label(shown, listed.name())
602                                .on_hover_text(listed.blurb())
603                                .clicked()
604                            {
605                                self.run(command);
606                            }
607                        }
608                    });
609            }
610            if !any {
611                ui.weak("Nothing fits.");
612            }
613        });
614    }

616    /// The row above a listing: *Recent* and how many, the rows a page,
617    /// and the pages walked. Each change runs the listing again from its
618    /// first page; a page walked keeps the rest.
619    fn listing_row(&mut self, ui: &mut egui::Ui, base: &str, pages: Option<(usize, usize)>) {
620        let mut rerun = None;
621        ui.horizontal(|ui| {
622            if ui
623                .checkbox(&mut self.recent, "Recent")
624                .on_hover_text("Newest first; with a number, only the last so many")
625                .changed()
626            {
627                rerun = Some(1);
628            }
629            if ui
630                .add_enabled(
631                    self.recent,
632                    egui::TextEdit::singleline(&mut self.recent_n).desired_width(50.0),
633                )
634                .on_hover_text("The last so many; empty for every one, paged")
635                .lost_focus()
636            {
637                rerun = Some(1);
638            }
639            ui.label("Rows a page");
640            if ui
641                .add(egui::TextEdit::singleline(&mut self.per).desired_width(40.0))
642                .lost_focus()
643            {
644                rerun = Some(1);
645            }
646            let (page, last) = pages.unwrap_or((1, 1));
647            if ui
648                .add_enabled(page > 1, egui::Button::new("Prev"))
649                .clicked()
650            {
651                rerun = Some(page - 1);
652            }
653            ui.label(format!("page {page} of {last}"));
654            if ui
655                .add_enabled(page < last, egui::Button::new("Next"))
656                .clicked()
657            {
658                rerun = Some(page + 1);
659            }
660        });
661        if let Some(page) = rerun {
662            let words = self.listing_words(page);
663            let command = if words.is_empty() {
664                base.to_owned()
665            } else {
666                format!("{base} {words}")
667            };
668            self.run(command);
669        }
670    }

672    /// The pane: the report shown, its links live.
673    fn pane(&mut self, ui: &mut egui::Ui) {
674        let Some(shown) = &self.shown else {
675            ui.weak(if self.who.is_none() {
676                "No character has anything recorded yet."
677            } else {
678                "Pick a report on the left."
679            });
680            return;
681        };
682        let mut clicked = None;
683        ui.monospace(format!(".{}", shown.command));
684        if let Some(base) = listing_base(&shown.command) {
685            let pages = shown.pages;
686            self.listing_row(ui, &base, pages);
687        }
688        let Some(shown) = &self.shown else {
689            return;
690        };
691        ui.separator();
692        egui::ScrollArea::both().show(ui, |ui| match &shown.lines {
693            Err(why) => {
694                ui.colored_label(theme::color(ui.ctx(), Token::Warning), why);
695            }
696            Ok(lines) => {
697                let palette = theme::palette(ui.ctx());
698                for line in lines {
699                    let runs: Vec<StyledRun> = line
700                        .iter()
701                        .map(|span| StyledRun {
702                            text: span.text.clone(),
703                            monospace: true,
704                            link: span.link.as_ref().map(|link| match link {
705                                NoticeLink::Hydra(command) | NoticeLink::Game(command) => {
706                                    RunLink::Command {
707                                        command: command.clone(),
708                                    }
709                                }
710                                NoticeLink::Url(href) => RunLink::Url { href: href.clone() },
711                            }),
712                            ..StyledRun::default()
713                        })
714                        .collect();
715                    let job = text::job(&runs, ui.style(), &palette);
716                    if let Some(text::Acted::Clicked(RunLink::Command { command }, _)) =
717                        ui.scope(|ui| text::linked(ui, job, &runs)).inner
718                    {
719                        clicked = Some(command);
720                    }
721                }
722            }
723        });
724        if let Some(command) = clicked {
725            self.run(command);
726        }
727    }
728}

735/// A listing's command without its listing words, when `command` is one:
736/// `cstats attacks all recent 5 page 2` is `cstats attacks all`.
737fn listing_base(command: &str) -> Option<String> {
738    let mut words = command.split_whitespace();
739    let family = words.next()?;
740    let kind = words.next()?;
741    if !family.eq_ignore_ascii_case("cstats")
742        || !(kind.eq_ignore_ascii_case("attacks") || kind.eq_ignore_ascii_case("creatures"))
743    {
744        return None;
745    }
746    let mut base = vec![family, kind];
747    let mut words = words.peekable();
748    while let Some(word) = words.next() {
749        match word.to_ascii_lowercase().as_str() {
750            "recent" | "page" | "per" => {
751                if words.peek().is_some_and(|w| w.parse::<usize>().is_ok()) {
752                    words.next();
753                }
754            }
755            _ => base.push(word),
756        }
757    }
758    Some(base.join(" "))
759}

761/// Do `ask` against the ledger under `data`. Blocking file I/O: never on
762/// the window's thread.
763pub(crate) fn run(data: &Path, ask: Ask) -> Reply {
764    let ledger = Ledger::open(Ledger::path_in(data)).map_err(|e| e.to_string());
765    match ask {
766        Ask::Characters => Reply::Characters(ledger.and_then(|ledger| {
767            ledger
768                .reader()
769                .and_then(|reader| reader.characters())
770                .map_err(|e| e.to_string())
771        })),
772        Ask::Flares(who) => {
773            let names = ledger
774                .and_then(|ledger| ledger.reader().map_err(|e| e.to_string()))
775                .and_then(|reader| {
776                    hydra_combat_log::flare_names(reader.connection(), &who)
777                        .map_err(|e| e.to_string())
778                })
779                .unwrap_or_default();
780            Reply::Flares(who, names)
781        }
782        Ask::Report(who, command) => {
783            let now = i64::try_from(
784                std::time::SystemTime::now()
785                    .duration_since(std::time::UNIX_EPOCH)
786                    .map_or(0, |d| d.as_secs()),
787            )
788            .unwrap_or(0);
789            let lines = ledger.and_then(|ledger| {
790                let word = command.split_whitespace().next().unwrap_or("");
791                if word.eq_ignore_ascii_case("exp") {
792                    hydra_exp::run(&ledger, &who, &command, now)
793                } else {
794                    hydra_combat_log::run(&ledger, &who, &command)
795                }
796            });
797            Reply::Report(who, command, lines)
798        }
799        Ask::Export(who, command, text) => Reply::Exported(export(data, &who, &command, &text)),
800    }
801}

803/// Write `text` under the data folder's `reports`, named by the character,
804/// the command's words and the moment:
805/// `Nisugi-cstats_intercepts_all-20261014T155012.txt`.
806fn export(data: &Path, who: &Who, command: &str, text: &str) -> Result<std::path::PathBuf, String> {
807    let words: String = command
808        .split_whitespace()
809        .collect::<Vec<_>>()
810        .join("_")
811        .chars()
812        .map(|c| {
813            if c.is_alphanumeric() || c == '_' || c == '-' {
814                c
815            } else {
816                '-'
817            }
818        })
819        .collect();
820    let now = hydra_net::eastern::now();
821    let (y, m, d) = hydra_net::eastern::date(now);
822    let secs = (now + hydra_net::eastern::offset(now)).rem_euclid(hydra_net::eastern::DAY);
823    let stamp = format!(
824        "{y:04}{m:02}{d:02}T{:02}{:02}{:02}",
825        secs / 3600,
826        secs % 3600 / 60,
827        secs % 60
828    );
829    let folder = data.join("reports");
830    std::fs::create_dir_all(&folder).map_err(|e| format!("{}: {e}", folder.display()))?;
831    let path = folder.join(format!("{}-{words}-{stamp}.txt", who.character));
832    std::fs::write(&path, text).map_err(|e| format!("{}: {e}", path.display()))?;
833    Ok(path)
834}

1//! A Wrayth key set read: the `<macros>` of a Wrayth settings file, each of
2//! its ten sets (`<keys id name>`, its keys `<k key action/>`) into the
3//! Hydra set of the same number. No XML crate: the file is Wrayth's own flat
4//! shape.
5//!
6//! A key is Wrayth's name for it, `Alt-Ctrl-E`, `Keypad 8`, `Page Up`, made
7//! Hydra's. What it does:
8//!
9//! - a `{Token}` is Hydra's action for it (`{HistoryPrev}` is `history_back`,
10//!   `{MacroSet}3` is `macro_set_3`), and a token Hydra has no action for is
11//!   said, not bound;
12//! - text ending `\r` sends, and without it fills the input; `\r` inside it
13//!   breaks one command from the next, `\p` is a second's wait, `\x` (clear
14//!   the line first) is dropped, since a sent command never touches the
15//!   input; `@` and `\?` are said, since Hydra has no cursor mark or prompt.
16

72/// Read `text`, a Wrayth settings file: every key of its `<macros>` Hydra
73/// can bind, and what it could not, in words for the player.
74pub(crate) fn read(text: &str) -> (Vec<Imported>, Vec<String>) {
75    let mut keys = Vec::new();
76    let mut said = Vec::new();
77    let Some(macros) = between(text, "<macros", "</macros>") else {
78        return (
79            keys,
80            vec!["It has no <macros>: no key set to import.".to_owned()],
81        );
82    };
83    for set in macros.split("<keys ").skip(1) {
84        let head = set.split('>').next().unwrap_or_default();
85        let Some(id) = attribute(head, "id").and_then(|id| id.parse::<u8>().ok()) else {
86            said.push("A <keys> with no set number was left out.".to_owned());
87            continue;
88        };
89        if id >= SETS {
90            said.push(format!("Set {id} was left out: Hydra has sets 0 to 9."));
91            continue;
92        }
93        let body = set.split("</keys>").next().unwrap_or_default();
94        for k in body.split("<k ").skip(1) {
95            let (Some(key), Some(action)) = (attribute(k, "key"), attribute(k, "action")) else {
96                continue;
97            };
98            match one(&key, &action) {
99                Ok(Some((chord, does))) => keys.push(Imported {
100                    set: id,
101                    chord,
102                    does,
103                }),
104                Ok(None) => {}
105                Err(why) => said.push(format!("{key} ({}): {why}", in_set(id))),
106            }
107        }
108    }
109    (keys, said)
110}

112/// `set 3`, or `the default set` for Wrayth's set 0.
113fn in_set(set: u8) -> String {
114    if set == 0 {
115        "the default set".to_owned()
116    } else {
117        format!("set {set}")
118    }
119}

121/// One Wrayth key and its action, made Hydra's; `None` for one that does
122/// nothing.
123fn one(key: &str, action: &str) -> Result<Option<(Chord, Macro)>, String> {
124    let chord = chord(key)?;
125    if let Some(why) = chord.refused() {
126        return Err(why);
127    }
128    let Some(does) = macro_of(action)? else {
129        return Ok(None);
130    };
131    does.check()?;
132    Ok(Some((chord, does)))
133}

135/// A Wrayth key's name, `Alt-Ctrl-Keypad Enter`, as a Hydra chord.
136fn chord(key: &str) -> Result<Chord, String> {
137    let mut rest = key.trim();
138    let mut written = String::new();
139    loop {
140        let before = rest;
141        for (wrayth, hydra) in [("Alt-", "Alt+"), ("Ctrl-", "Ctrl+"), ("Shift-", "Shift+")] {
142            if let Some(after) = rest.strip_prefix(wrayth) {
143                written.push_str(hydra);
144                rest = after;
145            }
146        }
147        if rest == before {
148            break;
149        }
150    }
151    let name = KEYS
152        .iter()
153        .find(|(wrayth, _)| wrayth.eq_ignore_ascii_case(rest))
154        .map(|(_, hydra)| (*hydra).to_owned())
155        .or_else(|| {
156            let mut chars = rest.chars();
157            match (chars.next(), chars.next()) {
158                (Some(letter), None) if letter.is_ascii_alphabetic() => {
159                    Some(format!("Key{}", letter.to_ascii_uppercase()))
160                }
161                (Some(digit), None) if digit.is_ascii_digit() => Some(format!("Digit{digit}")),
162                _ => (rest.starts_with(['F', 'f']) && rest[1..].parse::<u8>().is_ok())
163                    .then(|| rest.to_ascii_uppercase()),
164            }
165        })
166        .ok_or_else(|| format!("Hydra has no key called {rest}"))?;
167    written.push_str(&name);
168    Chord::parse(&written)
169}

171/// What a Wrayth action does, as a Hydra macro; `None` for nothing.
172fn macro_of(action: &str) -> Result<Option<Macro>, String> {
173    if let Some(token) = action.strip_prefix('{') {
174        let (name, after) = token.split_once('}').unwrap_or((token, ""));
175        if name == "MacroSet"
176            && let Ok(set) = after.trim().parse::<u8>()
177            && set < SETS
178        {
179            return Ok(Some(Macro::Act(Action::Set(set))));
180        }
181        return TOKENS
182            .iter()
183            .find(|(wrayth, _)| *wrayth == name)
184            .map(|(_, action)| Some(Macro::Act(*action)))
185            .ok_or_else(|| format!("Hydra has no action for {{{name}}}"));
186    }
187    if action.contains('@') || action.contains("\\?") {
188        return Err("Hydra has no cursor mark (@) or prompt (\\?) in a macro".to_owned());
189    }
190    let sends = action.trim_end().ends_with("\\r");
191    let text = action.replace("\\x", "").replace("\\p", "\\rs1\\r");
192    let text = text.replace("\\r", "\r");
193    let text = text.trim_matches('\r');
194    if text.trim().is_empty() {
195        return Ok(None);
196    }
197    Ok(Some(if sends {
198        Macro::Send(text.to_owned())
199    } else {
200        Macro::Fill(text.to_owned())
201    }))
202}

204/// The text between `open`, a tag's start, and `close`.
205fn between<'a>(text: &'a str, open: &str, close: &str) -> Option<&'a str> {
206    let start = text.find(open)?;
207    let end = text[start..].find(close)? + start;
208    Some(&text[start..end])
209}
210
211/// The value of attribute `name` in a tag's text, quoted either way, its
212/// entities read.
213fn attribute(tag: &str, name: &str) -> Option<String> {
214    let mut at = 0;
215    while let Some(found) = tag[at..].find(name) {
216        let start = at + found;
217        at = start + name.len();
218        let before = tag[..start].chars().next_back();
219        if before.is_some_and(|c| !c.is_whitespace()) {
220            continue;
221        }
222        let rest = tag[at..].trim_start().strip_prefix('=')?.trim_start();
223        let quote = rest.chars().next().filter(|c| *c == '\'' || *c == '"')?;
224        let value = rest[1..].split(quote).next()?;
225        return Some(
226            value
227                .replace("&lt;", "<")
228                .replace("&gt;", ">")
229                .replace("&quot;", "\"")
230                .replace("&apos;", "'")
231                .replace("&amp;", "&"),
232        );
233    }
234    None
235}

243    fn find<'a>(keys: &'a [Imported], set: u8, key: &str) -> Option<&'a Macro> {
244        let chord = Chord::parse(key).ok()?;
245        keys.iter()
246            .find(|imported| imported.set == set && imported.chord == chord)
247            .map(|imported| &imported.does)
248    }

250    /// A player's Wrayth set, stock set 0 and a set 1 of their own: its
251    /// keys made Hydra's, sending, filling and acting, and what Hydra has no
252    /// action for said, each by name.
253    #[test]
254    fn a_players_wrayth_set_imports() {
255        let (keys, said) = read(MNSTR);
256        let send = |text: &str| Some(Macro::Send(text.to_owned()));
257        assert_eq!(find(&keys, 0, "Numpad8").cloned(), send("n"), "\\x dropped");
258        assert_eq!(find(&keys, 0, "NumpadAdd").cloned(), send("look"));
259        assert_eq!(
260            find(&keys, 0, "Alt+KeyC").cloned(),
261            send("xml toggle containers")
262        );
263        assert_eq!(
264            find(&keys, 0, "ArrowUp"),
265            Some(&Macro::Act(Action::HistoryBack))
266        );
267        assert_eq!(
268            find(&keys, 0, "Alt+NumpadEnter"),
269            Some(&Macro::Act(Action::RepeatSecondLast))
270        );
271        assert_eq!(
272            find(&keys, 0, "Ctrl+Home"),
273            Some(&Macro::Act(Action::ScrollTop))
274        );
275        assert_eq!(
276            find(&keys, 0, "Alt+Digit3"),
277            Some(&Macro::Act(Action::Set(3)))
278        );
279        assert_eq!(find(&keys, 0, "Tab"), Some(&Macro::Act(Action::NextWindow)));
280        assert_eq!(
281            find(&keys, 1, "F1"),
282            Some(&Macro::Fill("prep 111".to_owned())),
283            "no \\r: it types"
284        );
285        assert_eq!(keys.iter().filter(|key| key.set == 1).count(), 3);
286        for token in [
287            "{Rest}",
288            "{ToggleMusic}",
289            "{Copy}",
290            "{PauseScript}",
291            "{ExportDialog}",
292        ] {
293            assert!(

297        }
298        assert_eq!(keys.len() + said.len(), 63 + 3, "every key bound or said");
299    }

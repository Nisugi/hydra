100/// The winit name `key` is written as, checked: a key this build can see.
101pub(super) fn known(key: &str) -> Result<String, String> {
102    if let Some((winit, _)) = NUMPAD
103        .iter()
104        .find(|(winit, fork)| winit.eq_ignore_ascii_case(key) || fork.eq_ignore_ascii_case(key))
105    {
106        return Ok((*winit).to_owned());
107    }
108    if let Some(name) = Key::ALL
109        .iter()
110        .filter_map(|key| winit_name(*key))
111        .find(|name| name.eq_ignore_ascii_case(key))
112    {
113        return Ok(name);
114    }
115    if let Some((name, _)) = CAPTURED
116        .iter()
117        .find(|(name, _)| name.eq_ignore_ascii_case(key))
118    {
119        return Ok((*name).to_owned());
120    }
121    if key.eq_ignore_ascii_case("NumLock") || key.eq_ignore_ascii_case("Clear") {
122        return Err(format!(
123            "{key} switches the numpad between typing and its keys (NumLock; Clear on a Mac), and is bound to nothing."
124        ));
125    }
126    Err(format!(
127        "{key} is not a key winit names; a key is named by its winit code, such as Numpad8, F13 or KeyA."
128    ))
129}

131/// winit's name for the physical key egui calls `key`: `KeyA` for `A`,
132/// `Digit1` for `Num1`, the same name for the function keys, the arrows and
133/// the rest; `None` for an egui key that is a character and not a key
134/// (`Plus`, `Colon`, `Pipe`...), and for a modifier.
135///
136/// **A modifier is held, not bound.** egui reports Shift, Ctrl, Alt and
137/// Super (Command on a Mac, the Windows key elsewhere), left and right, as
138/// presses of their own, before the key pressed with them. Named, the first
139/// press of Alt+1 was taken for the key, `Alt+AltLeft`; unnamed, it is
140/// passed over, and the key after it is the one bound, with Alt held.
141pub(crate) fn winit_name(key: Key) -> Option<String> {
142    if is_modifier(key) {
143        return None;
144    }
145    let name = format!("{key:?}");
146    if name.len() == 1 {
147        return Some(format!("Key{name}"));
148    }
149    if let Some(digit) = name.strip_prefix("Num")
150        && digit.len() == 1
151    {
152        return Some(format!("Digit{digit}"));
153    }
154    let renamed = match key {
155        Key::Equals => "Equal",
156        Key::Backtick => "Backquote",
157        Key::OpenBracket => "BracketLeft",
158        Key::CloseBracket => "BracketRight",
159        Key::Plus
160        | Key::Colon
161        | Key::Pipe
162        | Key::Questionmark
163        | Key::Exclamationmark
164        | Key::OpenCurlyBracket
165        | Key::CloseCurlyBracket
166        | Key::Copy
167        | Key::Cut
168        | Key::Paste => return None,
169        _ => return Some(name),
170    };
171    Some(renamed.to_owned())
172}

174/// Whether `key` is a modifier: Shift, Ctrl, Alt or Super, either side.
175fn is_modifier(key: Key) -> bool {
176    matches!(
177        key,
178        Key::ShiftLeft
179            | Key::ShiftRight
180            | Key::ControlLeft
181            | Key::ControlRight
182            | Key::AltLeft
183            | Key::AltRight
184            | Key::SuperLeft
185            | Key::SuperRight
186    )
187}

189/// Whether the key of this name types something when pressed. A bound
190/// key that does is taken with what it typed: egui sends a letter's text
191/// beside its press unless Ctrl or Cmd is held, so Alt+C did its macro and
192/// typed a `c` as well.
193pub(crate) fn types(key: &str) -> bool {
194    TYPING.contains(&key)
195}

197/// The chord a numpad press is, its key named as winit names it; `None` for
198/// a release, or a key with no code.
199pub(crate) fn numpad_chord(event: &eframe::NumpadKeyEvent) -> Option<Chord> {
200    if !event.pressed {
201        return None;
202    }
203    let winit::keyboard::PhysicalKey::Code(code) = event.physical_key else {
204        return None;
205    };
206    Some(Chord::of(&format!("{code:?}"), event.modifiers))
207}
208
209/// The chord of a numpad press the fork caught, to be done as the window
210/// with the keyboard binds it; `None` for one let through to be typed.
211pub(crate) fn numpad_caught(event: &eframe::NumpadKeyEvent) -> Option<Chord> {
212    if !event.consumed {
213        return None;
214    }
215    numpad_chord(event)
216}

218/// Every key egui has no name for that a binding may use: what the fork
219/// catches while the Keys page waits for a key, so it can take one.
220pub(crate) fn capturable() -> HashSet<winit::keyboard::KeyCode> {
221    CAPTURED.iter().map(|(_, code)| *code).collect()
222}

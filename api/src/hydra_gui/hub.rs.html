1//! The hub: every character Hydra runs, at a glance, in four tabs --
2//! *Live*, the characters whose connection is open or being retried;
3//! *Offline*, every other: those that stopped by themselves this run, with
4//! why, to reconnect or remove, then the roster's characters not running
5//! (`launch.rs`); *Teams*, roster characters started together (`teams.rs`);
6//! and *New login*, a login by account (`launch.rs`).
7//!
8//! A card is drawn from its seat:
9//! who, how it is connected, its four gauges, and what else a player glances
10//! at -- roundtime, the room, the group. Cards tile across the tab, each as
11//! wide as its four bars until a side is dragged, which sets every card's
12//! width.
13//!
14//! At the header's right, *Shut down*, and beside it the *Menu*
15//! (`crate::main_menu`), as a play window's title bar has it. *Performance*
16//! adds to each live card how quick its window is, stage by stage
17//! (`hydra_ui::perf`), for every live character while it is on; *Web page*
18//! shows the page's address and a code a phone scans.
19//!
20//! Under the header, a newer release of Hydra once one is out: *What's new*
21//! opens its page, *Update* fetches it, and a bar shows the fetch. Fetched,
22//! it runs from Hydra's next start, or at once with *Restart now*, which
23//! names the characters it will disconnect and offers to start them again.
24

32/// Which of the hub's tabs is showing.
33#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
34pub(crate) enum Tab {
35    /// Characters whose connection is open, or being retried.
36    #[default]
37    Live,
38    /// Every other character: those that ended this run, then the roster's
39    /// not on the table.
40    Offline,
41    /// The teams kept, and their editor.
42    Teams,
43    /// A login by account, and the kept passwords.
44    NewLogin,
45}

47/// What the player asked the hub for.
48#[derive(Clone, Debug, PartialEq, Eq)]
49pub(crate) enum HubAction {
50    /// Something for the binary, the owner of the session table.
51    Ask(HubRequest),
52    /// Bring this session's play window up: open it, or to the front.
53    Show(u32),
54    /// Open this session's character's log window.
55    Log(u32),
56    /// What the Menu was asked for, save its *Performance* switch, which
57    /// the hub keeps.
58    Menu(crate::main_menu::Choice),
59    /// Fetch the newer release out.
60    Update,
61}

63/// What the hub shows this frame.
64pub(crate) struct HubView<'a> {
65    /// Every character seated, in the order it was started.
66    pub(crate) seats: &'a [std::sync::Arc<Seat>],
67    /// Characters the binary can start: on the roster, with a kept password,
68    /// not running.
69    pub(crate) offered: &'a [String],
70    /// Every character on the roster.
71    pub(crate) roster: &'a [RosterCard],
72    /// Every team kept.
73    pub(crate) teams: &'a [TeamCard],
74    /// The last account whose characters the login service listed.
75    pub(crate) listing: Option<&'a Listing>,
76    /// What the binary answered the last request, while it is shown.
77    pub(crate) said: Option<&'a str>,
78    /// Where the web page's server is, while it is wanted.
79    pub(crate) web_at: Option<&'a crate::WebAt>,
80    /// A newer release, once one is out, and how far it has come.
81    pub(crate) update: Option<&'a Update>,
82    /// The design every character's card is drawn from (`crate::card`).
83    pub(crate) card: &'a crate::layout::Custom,
84    /// The shared streams merged across every character, for the Chat
85    /// card (`crate::chat`).
86    pub(crate) merged: &'a hydra_ui::MergedHistory,
87}

130impl Hub {
131    /// Draw the hub over `view`, and return what the player asked for, if
132    /// anything.
133    pub(crate) fn show(&mut self, ui: &mut egui::Ui, view: &HubView<'_>) -> Option<HubAction> {
134        let mut asked = None;
135        let (closed, live): (Vec<&Seat>, Vec<&Seat>) = view
136            .seats
137            .iter()
138            .map(AsRef::as_ref)
139            .partition(|seat| seat.lifecycle() == State::Closed);
140        let playing = !live.is_empty();
141        let offline = closed.len() + crate::launch::waiting(view).len();
142        ui.horizontal(|ui| {
143            ui.selectable_value(&mut self.tab, Tab::Live, format!("Live ({})", live.len()));
144            ui.selectable_value(&mut self.tab, Tab::Offline, format!("Offline ({offline})"));
145            ui.selectable_value(
146                &mut self.tab,
147                Tab::Teams,
148                format!("Teams ({})", view.teams.len()),
149            );
150            ui.selectable_value(&mut self.tab, Tab::NewLogin, "New login");
151            ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
152                // With nothing playing there is nothing to lose, so it does
153                // not ask. Closing the window never asks: it only hides it,
154                // or lets Hydra rest.
155                if ui.button("Shut down").clicked() {
156                    if playing {
157                        self.confirming = true;
158                    } else {
159                        asked = Some(HubAction::Ask(HubRequest::Shutdown));
160                    }
161                }
162                let web = crate::main_menu::Web::Shown {
163                    on: &mut self.web,
164                    served: view.web_at.is_some(),
165                };
166                match crate::main_menu::show(ui, self.perf, web) {
167                    Some(crate::main_menu::Choice::Performance(on)) => self.perf = on,
168                    Some(choice) => asked = Some(HubAction::Menu(choice)),
169                    None => {}
170                }
171            });
172        });
173        if let Some(update) = view.update
174            && let Some(action) = update_line(ui, update, &live, &mut self.restarting)
175        {
176            asked = Some(action);
177        }
178        if self.web
179            && let Some(at) = view.web_at
180        {
181            web_page(ui, at);
182        }
183        if self.confirming {
184            ui.horizontal(|ui| {
185                ui.label(SHUT_DOWN_QUESTION);
186                if ui.button("Shut down every character").clicked() {
187                    self.confirming = false;
188                    asked = Some(HubAction::Ask(HubRequest::Shutdown));
189                }
190                if ui.button("Keep playing").clicked() {
191                    self.confirming = false;
192                }
193            });
194        }
195        if let Some(said) = view.said {
196            ui.weak(said);
197        }
198        if let Some(told) = &self.told {
199            ui.colored_label(theme::color(ui.ctx(), T::Wrong), told);
200        }
201        ui.separator();
202        // Below the header, an id of its own: a line appearing above -- the
203        // answer, the shut-down question -- would otherwise renumber every
204        // widget under it, which a debug build draws as red boxes.
205        let body = egui::UiBuilder::new().id(ui.id().with("hub-body"));
206        ui.scope_builder(body, |ui| {
207            let perf = self.perf;
208            match self.tab {
209                Tab::Live => {
210                    self.chat.show(ui, view, view.merged, &mut asked);
211                    list(
212                        ui,
213                        (&live, view.card),
214                        ("No character is running.", perf),
215                        &mut asked,
216                    );
217                }
218                Tab::Offline => {
219                    let launch = &mut self.launch;
220                    egui::ScrollArea::vertical()
221                        .id_salt("hub-offline")
222                        .auto_shrink(false)
223                        .show(ui, |ui| {
224                            if offline == 0 {
225                                ui.label("Every character is playing. New login adds another.");
226                            }
227                            cards(ui, (&closed, view.card), false, &mut asked);
228                            if !closed.is_empty() {
229                                ui.add_space(ui.spacing().item_spacing.y);
230                            }
231                            let width = CardWidth(view.card.inside().x);
232                            launch.cards(ui, view, width, &mut asked);
233                        });
234                }
235                Tab::Teams => self.teams.show(ui, view, &mut asked),
236                Tab::NewLogin => self.launch.show_login(ui, view, &mut asked),
237            }
238        });
239        asked
240    }
241}

243/// A newer release, and how far it has come: *What's new* opens its page,
244/// *Update* asks for it, and *Try again* after a fetch failed. Fetched,
245/// *Restart now* restarts Hydra into it: at once with nothing playing; with
246/// `live` characters playing, once the player says so (`restarting`), with
247/// them started again afterwards unless the box is unticked.
248fn update_line(
249    ui: &mut egui::Ui,
250    update: &Update,
251    live: &[&Seat],
252    restarting: &mut Option<bool>,
253) -> Option<HubAction> {
254    let version = &update.version;
255    let mut asked = None;
256    let restart = |again| Some(HubAction::Ask(HubRequest::Restart { again }));
257    ui.horizontal(|ui| match &update.fetch {
258        Fetch::Out => {
259            ui.label(format!("Hydra {version} is out."));
260            ui.hyperlink_to("What's new", &update.page);
261            if ui.button("Update").clicked() {
262                asked = Some(HubAction::Update);
263            }
264        }
265        Fetch::Fetching(got, total) => {
266            ui.label(format!("Fetching Hydra {version}"));
267            // In thousandths, which a u16 holds and an f32 takes whole.
268            let done = u16::try_from(got.saturating_mul(1000) / (*total).max(1))
269                .unwrap_or(1000)
270                .min(1000);
271            ui.add(
272                egui::ProgressBar::new(f32::from(done) / 1000.0)
273                    .desired_width(160.0)
274                    .show_percentage(),
275            );
276        }
277        Fetch::Ready => {
278            ui.label(format!("Hydra {version} runs from Hydra's next start."));
279            ui.hyperlink_to("What's new", &update.page);
280            if restarting.is_none() && ui.button("Restart now").clicked() {
281                if live.is_empty() {
282                    asked = restart(false);
283                } else {
284                    *restarting = Some(true);
285                }
286            }
287        }
288        Fetch::Failed(why) => {
289            ui.colored_label(
290                theme::color(ui.ctx(), T::Wrong),
291                format!("Hydra {version} could not be fetched: {why}"),
292            );
293            if ui.button("Try again").clicked() {
294                asked = Some(HubAction::Update);
295            }
296        }
297    });
298    if update.fetch == Fetch::Ready
299        && let Some(mut again) = *restarting
300    {
301        let names: Vec<&str> = live.iter().map(|seat| seat.name.as_str()).collect();
302        let (who, them) = match names.as_slice() {
303            [] => (String::new(), "them"),
304            [one] => ((*one).to_owned(), "it"),
305            [rest @ .., last] => (format!("{} and {last}", rest.join(", ")), "them"),
306        };
307        ui.horizontal(|ui| {
308            ui.label(format!("Restart Hydra now? {who} will be disconnected."));
309            ui.checkbox(&mut again, format!("Start {them} again afterwards"));
310            *restarting = Some(again);
311            if ui.button("Restart").clicked() {
312                asked = restart(again);
313                *restarting = None;
314            }
315            if ui.button("Cancel").clicked() {
316                *restarting = None;
317            }
318        });
319    }
320    asked
321}

323/// The web page's address, its pairing in it, and the same as a code a
324/// phone's camera opens; or why it is not served.
325pub(crate) fn web_page(ui: &mut egui::Ui, at: &crate::WebAt) {
326    match at {
327        crate::WebAt::Failed(why) => {
328            ui.colored_label(theme::color(ui.ctx(), T::Wrong), why);
329        }
330        crate::WebAt::Listening(address) => {
331            ui.horizontal(|ui| {
332                qr(ui, address);
333                ui.vertical(|ui| {
334                    ui.label(
335                        "Open this on a phone or tablet on the home network, or scan the code:",
336                    );
337                    ui.add(
338                        egui::Label::new(egui::RichText::new(address).monospace()).selectable(true),
339                    );
340                    if ui.button("Copy").clicked() {
341                        ui.ctx().copy_text(address.clone());
342                    }
343                    ui.weak("The address holds the pairing: give it only to your own devices.");
344                });
345            });
346        }
347    }
348}

350/// `text` as a QR code, dark on light whatever the theme, so a camera
351/// reads it.
352fn qr(ui: &mut egui::Ui, text: &str) {
353    let Ok(code) = qrcode::QrCode::new(text.as_bytes()) else {
354        return;
355    };
356    let width = code.width();
357    let colors = code.to_colors();
358    // A quiet zone of four modules round it, as readers want.
359    let modules = width + 8;
360    let module = 3.0;
361    #[expect(
362        clippy::cast_precision_loss,
363        reason = "a code is under 200 modules wide"
364    )]
365    let side = modules as f32 * module;
366    let (area, response) = ui.allocate_exact_size(egui::vec2(side, side), egui::Sense::hover());
367    response
368        .widget_info(|| egui::WidgetInfo::labeled(egui::WidgetType::Image, true, "Web page code"));
369    let painter = ui.painter();
370    painter.rect_filled(area, 0.0, egui::Color32::WHITE);
371    for (at, color) in colors.iter().enumerate() {
372        if *color != qrcode::Color::Dark {
373            continue;
374        }
375        #[expect(clippy::cast_precision_loss, reason = "as above")]
376        let (x, y) = (((at % width) + 4) as f32, ((at / width) + 4) as f32);
377        painter.rect_filled(
378            egui::Rect::from_min_size(
379                area.min + egui::vec2(x * module, y * module),
380                egui::vec2(module, module),
381            ),
382            0.0,
383            egui::Color32::BLACK,
384        );
385    }
386}

388/// The cards of one tab, drawn from `design` and tiled as many to a row as
389/// fit, or what an empty one says; each with how quick its window is, when
390/// `perf`.
391fn list(
392    ui: &mut egui::Ui,
393    seats: (&[&Seat], &crate::layout::Custom),
394    (none, perf): (&str, bool),
395    asked: &mut Option<HubAction>,
396) {
397    if seats.0.is_empty() {
398        ui.label(none);
399        return;
400    }
401    egui::ScrollArea::vertical()
402        .id_salt("hub-cards")
403        .auto_shrink(false)
404        .show(ui, |ui| cards(ui, seats, perf, asked));
405}

407/// `seats`' cards drawn from `design`, tiled; each with how quick its window
408/// is, when `perf`.
409fn cards(
410    ui: &mut egui::Ui,
411    (seats, design): (&[&Seat], &crate::layout::Custom),
412    perf: bool,
413    asked: &mut Option<HubAction>,
414) {
415    tiled(ui, seats, CardWidth(design.inside().x), |ui, seat| {
416        // Each card keyed by its session, not its place, so a card keeps its
417        // widgets' state as others come and go.
418        let id = egui::Id::new(("hub-card", seat.id));
419        let drawn = card_scope(ui, id, |ui| crate::card::card(ui, seat, design, perf));
420        if let Some(action) = drawn.inner {
421            *asked = Some(action);
422        }
423    });
424}

426/// Lay `items` out as cards `width` wide, as many to a row as fit, each
427/// row's cards aligned at their tops.
428pub(crate) fn tiled<I>(
429    ui: &mut egui::Ui,
430    items: &[I],
431    width: CardWidth,
432    mut card: impl FnMut(&mut egui::Ui, &I),
433) {
434    let outer = width.0 + crate::card::surround(ui).0.total_margin().sum().x;
435    let gap = ui.spacing().item_spacing.x;
436    let room = ui.available_width();
437    let (mut per_row, mut used) = (1, outer);
438    while used + gap + outer <= room {
439        used += gap + outer;
440        per_row += 1;
441    }
442    for row in items.chunks(per_row) {
443        ui.horizontal_top(|ui| {
444            for item in row {
445                card(ui, item);
446            }
447        });
448    }
449}

451/// A card's own area: an id of its own, laid out top to bottom.
452pub(crate) fn card_scope<R>(
453    ui: &mut egui::Ui,
454    id: egui::Id,
455    card: impl FnOnce(&mut egui::Ui) -> R,
456) -> egui::InnerResponse<R> {
457    let own = egui::UiBuilder::new()
458        .id(id)
459        .layout(egui::Layout::top_down(egui::Align::Min));
460    ui.scope_builder(own, card)
461}

463/// How quick a character's window is: each stage's median and slow end
464/// over its last samples, coloured by how they read, and how often it fell
465/// behind. A stage's name says, under the pointer, what it measures.
466pub(crate) fn stages(ui: &mut egui::Ui, id: u32, perf: &hydra_ui::perf::Perf, lichs_text: bool) {
467    use hydra_ui::perf::{Grade, took};
468    let colour = |ui: &egui::Ui, grade: Grade| graded(ui.ctx(), grade);
469    egui::Grid::new(("perf", id))
470        .num_columns(4)
471        .spacing([10.0, 2.0])
472        .show(ui, |ui| {
473            ui.weak("Stage");
474            ui.weak("median");
475            ui.weak("95%");
476            ui.weak("of");
477            ui.end_row();
478            for (stage, samples) in perf.stages(lichs_text) {
479                ui.label(stage.name).on_hover_text(stage.help);
480                if let Some((median, slow)) = samples.spread() {
481                    for time in [median, slow] {
482                        let text = egui::RichText::new(took(time))
483                            .monospace()
484                            .color(colour(ui, stage.grade(time)));
485                        ui.label(text);
486                    }
487                } else {
488                    ui.weak("—");
489                    ui.weak("—");
490                }
491                ui.weak(samples.len().to_string());
492                ui.end_row();
493            }
494        });
495    ui.horizontal(|ui| {
496        ui.label(egui::RichText::new("quick").color(colour(ui, Grade::Quick)));
497        ui.label(egui::RichText::new("noticeable").color(colour(ui, Grade::Noticeable)));
498        ui.label(egui::RichText::new("slow").color(colour(ui, Grade::Slow)));
499        ui.weak("· a stage's name says what it measures");
500    });
501    if perf.gaps > 0 {
502        ui.colored_label(
503            theme::color(ui.ctx(), T::Wrong),
504            format!("Fell behind {} times", perf.gaps),
505        );
506    }
507}

509/// The colour a time is said in, as it reads: the green a connection is
510/// said in, the warning's, the error's.
511pub(crate) fn graded(context: &egui::Context, grade: hydra_ui::perf::Grade) -> egui::Color32 {
512    use hydra_ui::perf::Grade;
513    theme::color(
514        context,
515        match grade {
516            Grade::Quick => T::Stamina,
517            Grade::Noticeable => T::Warning,
518            Grade::Slow => T::Wrong,
519        },
520    )
521}

707        let out = |fetch| Update {

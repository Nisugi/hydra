1//! The hub: every character Hydra runs, at a glance, in four tabs --
2//! *Live*, the characters whose connection is open or being retried;
3//! *Offline*, every other: those that stopped by themselves this run, with
4//! why, to reconnect or remove, then the roster's characters not running
5//! (`launch.rs`); *Teams*, roster characters started together (`teams.rs`);
6//! and *New login*, a login by account (`launch.rs`).
7//!
8//! A card is drawn from its seat:
9//! who, how it is connected, its four gauges, and what else a player glances
10//! at -- roundtime, the room, the group. Cards tile across the tab, each as
11//! wide as its four bars until a side is dragged, which sets every card's
12//! width.
13//!
14//! At the header's right, *Shut down*, and beside it the *Menu*
15//! (`crate::main_menu`), as a play window's title bar has it. *Performance*
16//! adds to each live card how quick its window is, stage by stage
17//! (`hydra_ui::perf`), for every live character while it is on; *Web page*
18//! shows the page's address and a code a phone scans.
19//!
20//! Under the header, a newer release of Hydra once one is out: *What's new*
21//! opens its page, *Update* fetches it, and a bar shows the fetch. Fetched,
22//! it runs from Hydra's next start, or at once with *Restart now*, which
23//! names the characters it will disconnect and offers to start them again.
24

33/// Which of the hub's tabs is showing.
34#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
35pub(crate) enum Tab {
36    /// Characters whose connection is open, or being retried.
37    #[default]
38    Live,
39    /// Every other character: those that ended this run, then the roster's
40    /// not on the table.
41    Offline,
42    /// The teams kept, and their editor.
43    Teams,
44    /// A login by account, and the kept passwords.
45    NewLogin,
46}

48/// What the player asked the hub for.
49#[derive(Clone, Debug, PartialEq, Eq)]
50pub(crate) enum HubAction {
51    /// Something for the binary, the owner of the session table.
52    Ask(HubRequest),
53    /// Bring this session's play window up: open it, or to the front.
54    Show(u32),
55    /// Open this session's character's log window.
56    Log(u32),
57    /// What the Menu was asked for, save its *Performance* switch, which
58    /// the hub keeps.
59    Menu(crate::main_menu::Choice),
60    /// Fetch the newer release out.
61    Update,
62}

64/// What the hub shows this frame.
65pub(crate) struct HubView<'a> {
66    /// Every character seated, in the order it was started.
67    pub(crate) seats: &'a [std::sync::Arc<Seat>],
68    /// Characters the binary can start: on the roster, with a kept password,
69    /// not running.
70    pub(crate) offered: &'a [String],
71    /// Every character on the roster.
72    pub(crate) roster: &'a [RosterCard],
73    /// Every team kept.
74    pub(crate) teams: &'a [TeamCard],
75    /// The last account whose characters the login service listed.
76    pub(crate) listing: Option<&'a Listing>,
77    /// What the binary answered the last request, while it is shown.
78    pub(crate) said: Option<&'a str>,
79    /// A newer release, once one is out, and how far it has come.
80    pub(crate) update: Option<&'a Update>,
81    /// The design every character's card is drawn from (`crate::card`).
82    pub(crate) card: &'a crate::layout::Custom,
83    /// The shared streams merged across every character, for the Chat
84    /// card (`crate::chat`).
85    pub(crate) merged: &'a hydra_ui::MergedHistory,
86}

125pub(crate) const SHUT_DOWN_QUESTION: &str = "Shut Hydra down? Every character will quit.";

127impl Hub {
128    /// The hub's Menu: Performance switched here; anything else chosen,
129    /// asked.
130    fn menu(&mut self, ui: &mut egui::Ui) -> Option<HubAction> {
131        let place = crate::main_menu::Place::Hub { perf: self.perf };
132        match crate::main_menu::show(ui, place)? {
133            crate::main_menu::Choice::Performance(on) => {
134                self.perf = on;
135                None
136            }
137            choice => Some(HubAction::Menu(choice)),
138        }
139    }
140
141    /// Draw the hub over `view`, and return what the player asked for, if
142    /// anything.
143    pub(crate) fn show(&mut self, ui: &mut egui::Ui, view: &HubView<'_>) -> Option<HubAction> {
144        let mut asked = None;
145        let (closed, live): (Vec<&Seat>, Vec<&Seat>) = view
146            .seats
147            .iter()
148            .map(AsRef::as_ref)
149            .partition(|seat| seat.lifecycle() == State::Closed);
150        let playing = !live.is_empty();
151        let offline = closed.len() + crate::launch::waiting(view).len();
152        ui.horizontal(|ui| {
153            ui.selectable_value(&mut self.tab, Tab::Live, format!("Live ({})", live.len()));
154            ui.selectable_value(&mut self.tab, Tab::Offline, format!("Offline ({offline})"));
155            ui.selectable_value(
156                &mut self.tab,
157                Tab::Teams,
158                format!("Teams ({})", view.teams.len()),
159            );
160            ui.selectable_value(&mut self.tab, Tab::NewLogin, "New login");
161            ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
162                // With nothing playing there is nothing to lose, so it does
163                // not ask. Closing the window never asks: it only hides it,
164                // or lets Hydra rest.
165                if ui.button("Shut down").clicked() {
166                    if playing {
167                        self.confirming = true;
168                    } else {
169                        asked = Some(HubAction::Ask(HubRequest::Shutdown));
170                    }
171                }
172                asked = asked.take().or(self.menu(ui));
173            });
174        });
175        if let Some(update) = view.update
176            && let Some(action) = update_line(ui, update, &live, &mut self.restarting)
177        {
178            asked = Some(action);
179        }
180        if self.confirming {
181            ui.horizontal(|ui| {
182                ui.label(SHUT_DOWN_QUESTION);
183                if ui.button("Shut down every character").clicked() {
184                    self.confirming = false;
185                    asked = Some(HubAction::Ask(HubRequest::Shutdown));
186                }
187                if ui.button("Keep playing").clicked() {
188                    self.confirming = false;
189                }
190            });
191        }
192        if let Some(said) = view.said {
193            ui.weak(said);
194        }
195        if let Some(told) = &self.told {
196            ui.colored_label(theme::color(ui.ctx(), T::Wrong), told);
197        }
198        ui.separator();
199        // Below the header, an id of its own: a line appearing above -- the
200        // answer, the shut-down question -- would otherwise renumber every
201        // widget under it, which a debug build draws as red boxes.
202        let body = egui::UiBuilder::new().id(ui.id().with("hub-body"));
203        ui.scope_builder(body, |ui| {
204            let perf = self.perf;
205            match self.tab {
206                Tab::Live => {
207                    self.chat.show(ui, view, view.merged, &mut asked);
208                    list(
209                        ui,
210                        (&live, view.card),
211                        ("No character is running.", perf),
212                        &mut asked,
213                    );
214                }
215                Tab::Offline => {
216                    let launch = &mut self.launch;
217                    egui::ScrollArea::vertical()
218                        .id_salt("hub-offline")
219                        .auto_shrink(false)
220                        .show(ui, |ui| {
221                            if offline == 0 {
222                                ui.label("Every character is playing. New login adds another.");
223                            }
224                            cards(ui, (&closed, view.card), false, &mut asked);
225                            if !closed.is_empty() {
226                                ui.add_space(ui.spacing().item_spacing.y);
227                            }
228                            let width = CardWidth(view.card.inside().x);
229                            launch.cards(ui, view, width, &mut asked);
230                        });
231                }
232                Tab::Teams => self.teams.show(ui, view, &mut asked),
233                Tab::NewLogin => self.launch.show_login(ui, view, &mut asked),
234            }
235        });
236        asked
237    }
238}

240/// A newer release, and how far it has come: *What's new* opens its page,
241/// *Update* asks for it, and *Try again* after a fetch failed. Fetched,
242/// *Restart now* restarts Hydra into it: at once with nothing playing; with
243/// `live` characters playing, once the player says so (`restarting`), with
244/// them started again afterwards unless the box is unticked.
245fn update_line(
246    ui: &mut egui::Ui,
247    update: &Update,
248    live: &[&Seat],
249    restarting: &mut Option<bool>,
250) -> Option<HubAction> {
251    let version = &update.version;
252    let mut asked = None;
253    let restart = |again| Some(HubAction::Ask(HubRequest::Restart { again }));
254    ui.horizontal(|ui| match &update.fetch {
255        Fetch::Out => {
256            ui.label(format!("Hydra {version} is out."));
257            ui.hyperlink_to("What's new", &update.page);
258            if ui.button("Update").clicked() {
259                asked = Some(HubAction::Update);
260            }
261        }
262        Fetch::Fetching(got, total) => {
263            ui.label(format!("Fetching Hydra {version}"));
264            // In thousandths, which a u16 holds and an f32 takes whole.
265            let done = u16::try_from(got.saturating_mul(1000) / (*total).max(1))
266                .unwrap_or(1000)
267                .min(1000);
268            ui.add(
269                egui::ProgressBar::new(f32::from(done) / 1000.0)
270                    .desired_width(160.0)
271                    .show_percentage(),
272            );
273        }
274        Fetch::Ready => {
275            ui.label(format!("Hydra {version} runs from Hydra's next start."));
276            ui.hyperlink_to("What's new", &update.page);
277            if restarting.is_none() && ui.button("Restart now").clicked() {
278                if live.is_empty() {
279                    asked = restart(false);
280                } else {
281                    *restarting = Some(true);
282                }
283            }
284        }
285        Fetch::Failed(why) => {
286            ui.colored_label(
287                theme::color(ui.ctx(), T::Wrong),
288                format!("Hydra {version} could not be fetched: {why}"),
289            );
290            if ui.button("Try again").clicked() {
291                asked = Some(HubAction::Update);
292            }
293        }
294    });
295    if update.fetch == Fetch::Ready
296        && let Some(mut again) = *restarting
297    {
298        let names: Vec<&str> = live.iter().map(|seat| seat.name.as_str()).collect();
299        let (who, them) = match names.as_slice() {
300            [] => (String::new(), "them"),
301            [one] => ((*one).to_owned(), "it"),
302            [rest @ .., last] => (format!("{} and {last}", rest.join(", ")), "them"),
303        };
304        ui.horizontal(|ui| {
305            ui.label(format!("Restart Hydra now? {who} will be disconnected."));
306            ui.checkbox(&mut again, format!("Start {them} again afterwards"));
307            *restarting = Some(again);
308            if ui.button("Restart").clicked() {
309                asked = restart(again);
310                *restarting = None;
311            }
312            if ui.button("Cancel").clicked() {
313                *restarting = None;
314            }
315        });
316    }
317    asked
318}

320/// The web page's address, its pairing in it, and the same as a code a
321/// phone's camera opens; or why it is not served.
322pub(crate) fn web_page(ui: &mut egui::Ui, at: &crate::WebAt) {
323    match at {
324        crate::WebAt::Failed(why) => {
325            ui.colored_label(theme::color(ui.ctx(), T::Wrong), why);
326        }
327        crate::WebAt::Listening(address) => {
328            ui.horizontal(|ui| {
329                qr(ui, address);
330                ui.vertical(|ui| {
331                    ui.label(
332                        "Open this on a phone or tablet on the home network, or scan the code:",
333                    );
334                    ui.add(
335                        egui::Label::new(egui::RichText::new(address).monospace()).selectable(true),
336                    );
337                    if ui.button("Copy").clicked() {
338                        ui.ctx().copy_text(address.clone());
339                    }
340                    crate::form::hint(
341                        ui,
342                        "The address holds the pairing: give it only to your own devices.",
343                    );
344                });
345            });
346        }
347    }
348}

350/// `text` as a QR code, dark on light whatever the theme, so a camera
351/// reads it.
352fn qr(ui: &mut egui::Ui, text: &str) {
353    let Ok(code) = qrcode::QrCode::new(text.as_bytes()) else {
354        return;
355    };
356    let width = code.width();
357    let colors = code.to_colors();
358    // A quiet zone of four modules round it, as readers want.
359    let modules = width + 8;
360    let module = 3.0;
361    #[expect(
362        clippy::cast_precision_loss,
363        reason = "a code is under 200 modules wide"
364    )]
365    let side = modules as f32 * module;
366    let (area, response) = ui.allocate_exact_size(egui::vec2(side, side), egui::Sense::hover());
367    response
368        .widget_info(|| egui::WidgetInfo::labeled(egui::WidgetType::Image, true, "Web page code"));
369    let painter = ui.painter();
370    painter.rect_filled(area, 0.0, egui::Color32::WHITE);
371    for (at, color) in colors.iter().enumerate() {
372        if *color != qrcode::Color::Dark {
373            continue;
374        }
375        #[expect(clippy::cast_precision_loss, reason = "as above")]
376        let (x, y) = (((at % width) + 4) as f32, ((at / width) + 4) as f32);
377        painter.rect_filled(
378            egui::Rect::from_min_size(
379                area.min + egui::vec2(x * module, y * module),
380                egui::vec2(module, module),
381            ),
382            0.0,
383            egui::Color32::BLACK,
384        );
385    }
386}

388/// The cards of one tab, drawn from `design` and tiled as many to a row as
389/// fit, or what an empty one says; each with how quick its window is, when
390/// `perf`.
391fn list(
392    ui: &mut egui::Ui,
393    seats: (&[&Seat], &crate::layout::Custom),
394    (none, perf): (&str, bool),
395    asked: &mut Option<HubAction>,
396) {
397    if seats.0.is_empty() {
398        ui.label(none);
399        return;
400    }
401    egui::ScrollArea::vertical()
402        .id_salt("hub-cards")
403        .auto_shrink(false)
404        .show(ui, |ui| cards(ui, seats, perf, asked));
405}

407/// `seats`' cards drawn from `design`, tiled; each with how quick its window
408/// is, when `perf`.
409fn cards(
410    ui: &mut egui::Ui,
411    (seats, design): (&[&Seat], &crate::layout::Custom),
412    perf: bool,
413    asked: &mut Option<HubAction>,
414) {
415    tiled(ui, seats, CardWidth(design.inside().x), |ui, seat| {
416        // Each card keyed by its session, not its place, so a card keeps its
417        // widgets' state as others come and go.
418        let id = egui::Id::new(("hub-card", seat.id));
419        let drawn = card_scope(ui, id, |ui| crate::card::card(ui, seat, design, perf));
420        if let Some(action) = drawn.inner {
421            *asked = Some(action);
422        }
423    });
424}

426/// Lay `items` out as cards `width` wide, as many to a row as fit, each
427/// row's cards aligned at their tops.
428pub(crate) fn tiled<I>(
429    ui: &mut egui::Ui,
430    items: &[I],
431    width: CardWidth,
432    mut card: impl FnMut(&mut egui::Ui, &I),
433) {
434    let outer = width.0 + crate::card::surround(ui).0.total_margin().sum().x;
435    let gap = ui.spacing().item_spacing.x;
436    let room = ui.available_width();
437    let (mut per_row, mut used) = (1, outer);
438    while used + gap + outer <= room {
439        used += gap + outer;
440        per_row += 1;
441    }
442    for row in items.chunks(per_row) {
443        ui.horizontal_top(|ui| {
444            for item in row {
445                card(ui, item);
446            }
447        });
448    }
449}

451/// A card's own area: an id of its own, laid out top to bottom.
452pub(crate) fn card_scope<R>(
453    ui: &mut egui::Ui,
454    id: egui::Id,
455    card: impl FnOnce(&mut egui::Ui) -> R,
456) -> egui::InnerResponse<R> {
457    let own = egui::UiBuilder::new()
458        .id(id)
459        .layout(egui::Layout::top_down(egui::Align::Min));
460    ui.scope_builder(own, card)
461}

463/// How quick a character's window is: each stage's median and slow end
464/// over its last samples, coloured by how they read, and how often it fell
465/// behind. A stage's name says, under the pointer, what it measures.
466pub(crate) fn stages(ui: &mut egui::Ui, id: u32, perf: &hydra_ui::perf::Perf, lichs_text: bool) {
467    use hydra_ui::perf::{Grade, took};
468    let colour = |ui: &egui::Ui, grade: Grade| graded(ui.ctx(), grade);
469    egui::Grid::new(("perf", id))
470        .num_columns(4)
471        .spacing([10.0, 2.0])
472        .show(ui, |ui| {
473            ui.weak("Stage");
474            ui.weak("median");
475            ui.weak("95%");
476            ui.weak("of");
477            ui.end_row();
478            for (stage, samples) in perf.stages(lichs_text) {
479                ui.label(stage.name).on_hover_text(stage.help);
480                if let Some((median, slow)) = samples.spread() {
481                    for time in [median, slow] {
482                        let text = egui::RichText::new(took(time))
483                            .monospace()
484                            .color(colour(ui, stage.grade(time)));
485                        ui.label(text);
486                    }
487                } else {
488                    ui.weak("—");
489                    ui.weak("—");
490                }
491                ui.weak(samples.len().to_string());
492                ui.end_row();
493            }
494        });
495    ui.horizontal(|ui| {
496        ui.label(egui::RichText::new("quick").color(colour(ui, Grade::Quick)));
497        ui.label(egui::RichText::new("noticeable").color(colour(ui, Grade::Noticeable)));
498        ui.label(egui::RichText::new("slow").color(colour(ui, Grade::Slow)));
499        ui.weak("· a stage's name says what it measures");
500    });
501    if perf.gaps > 0 {
502        ui.colored_label(
503            theme::color(ui.ctx(), T::Wrong),
504            format!("Fell behind {} times", perf.gaps),
505        );
506    }
507}

509/// The colour a time is said in, as it reads: the green a connection is
510/// said in, the warning's, the error's.
511pub(crate) fn graded(context: &egui::Context, grade: hydra_ui::perf::Grade) -> egui::Color32 {
512    use hydra_ui::perf::Grade;
513    theme::color(
514        context,
515        match grade {
516            Grade::Quick => T::Stamina,
517            Grade::Noticeable => T::Warning,
518            Grade::Slow => T::Wrong,
519        },
520    )

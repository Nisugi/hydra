1//! The Field widget: one value of the game state, wherever it is, on a line
2//! of its own -- `level 42`, `silver 1,204`.
3//!
4//! A field is found by its path through the game state's print, the tree
5//! the Game state widget shows (`state.rs`): each step a field's name, or an
6//! item's place in a list, `[2]`. So any value the model holds can be
7//! shown, and a field added to the model can be shown with no change here.
8//! The print has no names of its own to keep a choice by: a field the model
9//! renames is "not found" until it is picked again.
10//!
11//! What is shown is the value alone, as plainly as the print allows:
12//! `Some(42)` is `42`, `None` is `—`, a quoted string its words; a list or a
13//! record shows how many it holds. The field's name is in its hover text,
14//! and names its cell in the cards' editor ([`name`]).
15//!
16//! [`pick`] is the picker the cards' editor offers: the tree to open and
17//! choose from, and a filter that finds any line.
18

24/// The most fields the picker's filter lists.
25const FOUND: usize = 200;

27/// The field at `path` in `state`, drawn as its value alone; the print
28/// kept under `key`, which every field drawn from one character shares.
29pub(crate) fn draw(ui: &mut egui::Ui, state: Option<&GameState>, path: &[String], key: Id) {
30    let value = match state {
31        Some(state) => read(ui, state, path, key).unwrap_or_else(|| "not found".to_owned()),
32        None => "—".to_owned(),
33    };
34    ui.horizontal(|ui| {
35        ui.add(egui::Label::new(value.as_str()).truncate())
36            .on_hover_text(format!("{}: {value}", path.join(" › ")));
37    });
38}

40/// The value at `path` in `state`, said as [`draw`] shows it; none when
41/// nothing is there. The print kept under `key`, and drawn again so what is
42/// read is never older than the print.
43pub(crate) fn read(ui: &egui::Ui, state: &GameState, path: &[String], key: Id) -> Option<String> {
44    let printed = printed(ui, state, key);
45    ui.ctx().request_repaint_after(EVERY);
46    value(top(&printed.tree), path)
47}

49/// What a field at `path` is called: its last name, words apart, and the
50/// place in its list when it is an item of one -- `effects 2`.
51pub(crate) fn name(path: &[String]) -> String {
52    let words = |step: &str| step.trim_matches('"').replace('_', " ");
53    match path {
54        [] => "Field".to_owned(),
55        [.., named, last] if place(last).is_some() && place(named).is_none() => {
56            format!("{} {}", words(named), place(last).unwrap_or_default())
57        }
58        [.., last] => match place(last) {
59            Some(at) => format!("item {at}"),
60            None => words(last),
61        },
62    }
63}

65/// A step that is an item's place in a list: its number.
66fn place(step: &str) -> Option<usize> {
67    step.strip_prefix('[')?.strip_suffix(']')?.parse().ok()
68}

70/// A line's name among its siblings, `at` being its place: what its print
71/// says before `:` -- a field's name, or a map's quoted key -- or its place,
72/// `[2]`, when it says none, as a list's items do.
73fn step(text: &str, at: usize) -> String {
74    named(text).map_or_else(|| format!("[{at}]"), str::to_owned)
75}

77/// The name a line's print gives it before `:`, if it gives one.
78fn named(text: &str) -> Option<&str> {
79    let text = text.trim();
80    let end = if text.starts_with('"') {
81        // A map's key: a quoted string, kept with its quotes.
82        let mut escaped = false;
83        let mut end = None;
84        for (at, c) in text.char_indices().skip(1) {
85            if escaped {
86                escaped = false;
87            } else if c == '\\' {
88                escaped = true;
89            } else if c == '"' {
90                end = Some(at + 1);
91                break;
92            }
93        }
94        end?
95    } else {
96        text.find(|c: char| !(c.is_alphanumeric() || c == '_'))?
97    };
98    (end > 0 && text[end..].starts_with(':')).then(|| &text[..end])
99}

101/// What a line says after its name, if it has one.
102fn said(text: &str) -> &str {
103    let text = text.trim();
104    match named(text) {
105        Some(name) => text[name.len() + 1..].trim(),
106        None => text,
107    }
108}

110/// The value at `path` in `lines`, said plainly; `None` when nothing is
111/// there.
112pub(super) fn value(lines: &[Line], path: &[String]) -> Option<String> {
113    let (first, rest) = path.split_first()?;
114    let line = lines.iter().enumerate().find_map(|(at, line)| {
115        let text = match line {
116            Line::Leaf(text) | Line::Branch(text, _) => text,
117        };
118        (step(text, at) == *first).then_some(line)
119    })?;
120    match (line, rest) {
121        (Line::Leaf(text), []) => Some(plain(said(text))),
122        (Line::Branch(label, inside), []) => {
123            let kind = said(label);
124            Some(if kind.is_empty() {
125                inside.len().to_string()
126            } else {
127                format!("{kind} ({})", inside.len())
128            })
129        }
130        (Line::Branch(_, inside), rest) => value(inside, rest),
131        (Line::Leaf(_), _) => None,
132    }
133}

135/// A value as plainly as the print allows: `Some(42)` is `42`, `None` is
136/// `—`, a quoted string its words.
137fn plain(value: &str) -> String {
138    let mut value = value.trim().trim_end_matches(',').trim();
139    while let Some(inner) = value
140        .strip_prefix("Some(")
141        .and_then(|inner| inner.strip_suffix(')'))
142    {
143        value = inner.trim();
144    }
145    if value == "None" {
146        return "—".to_owned();
147    }
148    if value.len() >= 2 && value.starts_with('"') && value.ends_with('"') {
149        value = &value[1..value.len() - 1];
150    }
151    value.to_owned()
152}

154/// The picker: `state`'s tree to open and choose a value from, or, with a
155/// filter typed, every line it finds with where it is. The path chosen, if
156/// one was. The print kept under `key`, as [`draw`]'s.
157pub(crate) fn pick(ui: &mut egui::Ui, state: Option<&GameState>, key: Id) -> Option<Vec<String>> {
158    let Some(state) = state else {
159        ui.weak("Nobody is playing: a field is picked from a character's game state.");
160        return None;
161    };
162    let printed = printed(ui, state, key);
163    let filter_id = key.with("pick-filter");
164    let mut filter = ui.data(|data| data.get_temp::<String>(filter_id).unwrap_or_default());
165    ui.add(egui::TextEdit::singleline(&mut filter).hint_text("Find a field..."));
166    ui.data_mut(|data| data.insert_temp(filter_id, filter.clone()));
167    let lines = top(&printed.tree);
168    let needle = filter.trim().to_lowercase();
169    if needle.is_empty() {
170        return branch(ui, lines, &mut Vec::new(), key.with("pick-tree"));
171    }
172    let mut found = Vec::new();
173    finding(lines, &needle, &mut Vec::new(), &mut found);
174    if found.is_empty() {
175        ui.weak("Nothing matches.");
176    }
177    let mut chosen = None;
178    for (path, text) in found {
179        let shown = format!(
180            "{} = {}",
181            path.join(" › "),
182            value(lines, &path).unwrap_or(text)
183        );
184        if ui
185            .button(RichText::new(shown).monospace().small())
186            .clicked()
187        {
188            chosen = Some(path);
189        }
190    }
191    chosen
192}

194/// `lines` under `path` to open and choose from: a value chosen with a
195/// click, a branch opened. The path chosen, if one was.
196fn branch(
197    ui: &mut egui::Ui,
198    lines: &[Line],
199    path: &mut Vec<String>,
200    id: Id,
201) -> Option<Vec<String>> {
202    let mut chosen = None;
203    for (at, line) in lines.iter().enumerate() {
204        match line {
205            Line::Leaf(text) => {
206                path.push(step(text, at));
207                if ui
208                    .button(
209                        RichText::new(text.trim_end_matches(','))
210                            .monospace()
211                            .small(),
212                    )
213                    .clicked()
214                {
215                    chosen = Some(path.clone());
216                }
217                path.pop();
218            }
219            Line::Branch(label, inside) => {
220                path.push(step(label, at));
221                let id = id.with(at);
222                let opened = egui::CollapsingHeader::new(RichText::new(label).monospace().small())
223                    .id_salt(id)
224                    .show(ui, |ui| {
225                        if ui
226                            .small_button(format!("Pick this ({} in it)", inside.len()))
227                            .clicked()
228                        {
229                            return Some(path.clone());
230                        }
231                        branch(ui, inside, path, id)
232                    });
233                chosen = chosen.or(opened.body_returned.flatten());
234                path.pop();
235            }
236        }
237    }
238    chosen
239}

241/// Every line under `lines` whose print holds `needle`, with its path, at
242/// most [`FOUND`].
243fn finding(
244    lines: &[Line],
245    needle: &str,
246    path: &mut Vec<String>,
247    found: &mut Vec<(Vec<String>, String)>,
248) {
249    for (at, line) in lines.iter().enumerate() {
250        if found.len() >= FOUND {
251            return;
252        }
253        let (text, inside) = match line {
254            Line::Leaf(text) => (text, None),
255            Line::Branch(label, inside) => (label, Some(inside)),
256        };
257        path.push(step(text, at));
258        if text.to_lowercase().contains(needle) {
259            found.push((path.clone(), plain(said(text))));
260        }
261        if let Some(inside) = inside {
262            finding(inside, needle, path, found);
263        }
264        path.pop();
265    }
266}

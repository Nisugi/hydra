1//! The Containers widget on the window's own character: the containers the
2//! game has opened, arranged as the character chose (`hydra_board::
3//! containers`) -- a section each, folded or open, or a tab each -- each
4//! named by its label, with what the whole inventory said of it.
5//!
6//! A ⚙ menu at the top chooses list or tabs, whether a hidden container is
7//! shown again when it is looked in, and shows a hidden one again. A ⋯ menu
8//! on each container labels it, for the character or for every character,
9//! hides it, or moves it up or down. What is carried and let go on a
10//! container, or on one of its items, goes into it.
11

20/// The window's character's choices for its containers, and the labels kept
21/// for every character, as its Containers widget is given them.
22#[derive(Clone, Debug, Default)]
23pub(crate) struct ContainersShown {
24    /// The character, whose hidden containers are watched apart.
25    pub(crate) who: String,
26    /// Its choices.
27    pub(crate) choices: Containers,
28    /// The labels kept for every character.
29    pub(crate) shared: SharedLabels,
30}
31
32/// A change the widget asks of the containers' choices.
33#[derive(Clone, Debug, PartialEq, Eq)]
34pub(crate) enum Contained {
35    /// One of the character's own.
36    Own(Change),
37    /// A label for every character, by full name; `None` for none.
38    Shared(String, Option<String>),
39}
40
41/// The change `change` to the character's own choices, as the widget asks.
42fn own(change: Change) -> Clicked {
43    Clicked::Containers(Contained::Own(change))
44}

46/// Draw the containers as `shown` arranges them; what was asked: a link, a
47/// put, or a change to the choices.
48pub(super) fn containers(
49    ui: &mut egui::Ui,
50    state: Option<&GameState>,
51    shown: &ContainersShown,
52    id: egui::Id,
53) -> Option<Clicked> {
54    let Some(state) = state else {
55        ui.weak("Containers unknown");
56        return None;
57    };
58    let (arranged, unhide) = shown.choices.arrange(&shown.who, state, &shown.shared);
59    // Looked in since it was hidden: shown, and kept so.
60    if let Some(name) = unhide.into_iter().next() {
61        return Some(own(Change::Unhide(name)));
62    }
63    let mut clicked = settings(ui, &shown.choices);
64    if arranged.is_empty() {
65        ui.weak("No container seen yet");
66        return clicked;
67    }
68    let names: Vec<String> = arranged.iter().map(|each| each.name.clone()).collect();
69    match shown.choices.layout {
70        Layout::List => {
71            for each in &arranged {
72                clicked = section(ui, each, &names, id).or(clicked);
73            }
74        }
75        Layout::Tabs => clicked = tabs(ui, &arranged, &names, id).or(clicked),
76    }
77    clicked
78}

80/// The ⚙ menu: list or tabs, show-on-look, and the hidden containers.
81fn settings(ui: &mut egui::Ui, choices: &Containers) -> Option<Clicked> {
82    let mut clicked = None;
83    ui.horizontal(|ui| {
84        ui.menu_button("⚙", |ui| {
85            for (layout, words) in [(Layout::List, "A list"), (Layout::Tabs, "Tabs")] {
86                if ui.radio(choices.layout == layout, words).clicked() {
87                    clicked = Some(own(Change::Layout(layout)));
88                    ui.close();
89                }
90            }
91            ui.separator();
92            let mut on = choices.show_on_look;
93            if ui
94                .checkbox(&mut on, "Show container window on look")
95                .on_hover_text("A hidden container is shown again when you look in it")
96                .changed()
97            {
98                clicked = Some(own(Change::ShowOnLook(on)));
99            }
100            if !choices.hidden.is_empty() {
101                ui.separator();
102                ui.weak("Hidden");
103                for name in &choices.hidden {
104                    if ui.button(format!("Show {name}")).clicked() {
105                        clicked = Some(own(Change::Unhide(name.clone())));
106                        ui.close();
107                    }
108                }
109            }
110        });
111    });
112    clicked
113}

115/// A container's header words after its name: how many it holds, and what
116/// the whole inventory said of it.
117fn counted(each: &Arranged<'_>) -> String {
118    let mut words = format!("({})", each.items.len());
119    if let Some(detail) = each.detail {
120        words.push_str(&detailed(detail));
121    }
122    words
123}
124
125/// What the whole inventory said of a container: ` 14/25 lb`, ` closed`.
126fn detailed(detail: Detail) -> String {
127    use std::fmt::Write as _;
128    let mut words = String::new();
129    // Writing to a `String` cannot fail.
130    let _ = match (detail.held, detail.capacity) {
131        (Some(held), Some(most)) => write!(words, " {held}/{} lb", most.pounds),
132        (Some(held), None) => write!(words, " {held} lb"),
133        _ => Ok(()),
134    };
135    if detail.closed {
136        words.push_str(" closed");
137    }
138    words
139}

141/// The container's name, a link to it.
142fn named(ui: &mut egui::Ui, each: &Arranged<'_>) -> Option<Clicked> {
143    let object = theme::color(ui.ctx(), T::Object);
144    let runs: [StyledRun; 1] = [hydra_ui::room::object_run(
145        &each.shown,
146        (&each.target, &each.noun),
147        rgb_of(object),
148    )];
149    say(ui, &runs, true)
150}
151
152/// What the container holds, each a link to it; what was asked.
153fn held(ui: &mut egui::Ui, each: &Arranged<'_>) -> Option<Clicked> {
154    if each.items.is_empty() {
155        ui.weak("empty");
156        return None;
157    }
158    let object = theme::color(ui.ctx(), T::Object);
159    let mut clicked = None;
160    for item in each.items {
161        let runs: [StyledRun; 1] = [hydra_ui::room::object_run(
162            &item.text,
163            (&item.id, &item.noun),
164            rgb_of(object),
165        )];
166        // Asked for even after a click, so every item is drawn.
167        clicked = clicked.take().or(say(ui, &runs, true));
168    }
169    clicked
170}

172/// What is carried and let go on `response` put into the container.
173fn put(response: &egui::Response, each: &Arranged<'_>) -> Option<Clicked> {
174    crate::carry::dropped(response, &format!("#{}", each.target), Some(&each.target))
175        .map(Clicked::Quietly)
176}

178/// One container as a section: its arrow, which folds it, its name, its
179/// count and detail, its ⋯ menu, and what it holds when open.
180fn section(
181    ui: &mut egui::Ui,
182    each: &Arranged<'_>,
183    names: &[String],
184    id: egui::Id,
185) -> Option<Clicked> {
186    let mut clicked = None;
187    let header = ui
188        .horizontal(|ui| {
189            let arrow = if each.folded { "▸" } else { "▾" };
190            if ui.add(egui::Button::new(arrow).frame(false)).clicked() {
191                clicked = Some(own(Change::Fold(each.name.clone(), !each.folded)));
192            }
193            clicked = clicked.take().or(named(ui, each));
194            ui.weak(counted(each));
195            clicked = clicked.take().or(menu(ui, each, names, id));
196        })
197        .response;
198    clicked = clicked.or_else(|| put(&header, each));
199    if !each.folded {
200        let body = ui.indent(id.with(("held", &each.name)), |ui| held(ui, each));
201        clicked = clicked.or(body.inner).or_else(|| put(&body.response, each));
202    }
203    clicked
204}

206/// The containers as tabs, the one picked showing what it holds.
207fn tabs(
208    ui: &mut egui::Ui,
209    arranged: &[Arranged<'_>],
210    names: &[String],
211    id: egui::Id,
212) -> Option<Clicked> {
213    let key = id.with("containers-tab");
214    let picked: Option<String> = ui.data(|data| data.get_temp(key));
215    let picked = arranged
216        .iter()
217        .position(|each| Some(&each.name) == picked.as_ref())
218        .unwrap_or(0);
219    let mut clicked = None;
220    ui.horizontal_wrapped(|ui| {
221        for (at, each) in arranged.iter().enumerate() {
222            let tab = ui.selectable_label(at == picked, &each.shown);
223            if tab.clicked() {
224                ui.data_mut(|data| data.insert_temp(key, each.name.clone()));
225            }
226            clicked = clicked.take().or_else(|| put(&tab, each));
227        }
228    });
229    let Some(each) = arranged.get(picked) else {
230        return clicked;
231    };
232    let header = ui
233        .horizontal(|ui| {
234            clicked = clicked.take().or(named(ui, each));
235            ui.weak(counted(each));
236            clicked = clicked.take().or(menu(ui, each, names, id));
237        })
238        .response;
239    clicked = clicked.or_else(|| put(&header, each));
240    let body = ui.vertical(|ui| held(ui, each));
241    clicked.or(body.inner).or_else(|| put(&body.response, each))
242}

244/// The ⋯ menu of one container: its label, for the character or every
245/// character; hide it; move it up or down.
246fn menu(ui: &mut egui::Ui, each: &Arranged<'_>, names: &[String], id: egui::Id) -> Option<Clicked> {
247    let mut clicked = None;
248    ui.menu_button("⋯", |ui| {
249        ui.label(&each.name);
250        let key = id.with(("label", &each.name));
251        let mut words: String = ui.data(|data| data.get_temp(key)).unwrap_or_default();
252        ui.horizontal(|ui| {
253            ui.label("Label");
254            if ui.text_edit_singleline(&mut words).changed() {
255                ui.data_mut(|data| data.insert_temp(key, words.clone()));
256            }
257        });
258        let label = Some(words.trim().to_owned()).filter(|words| !words.is_empty());
259        ui.horizontal(|ui| {
260            if ui.button("For this character").clicked() {
261                clicked = Some(own(Change::Label(each.name.clone(), label.clone())));
262                ui.close();
263            }
264            if ui.button("For every character").clicked() {
265                clicked = Some(Clicked::Containers(Contained::Shared(
266                    each.name.clone(),
267                    label.clone(),
268                )));
269                ui.close();
270            }
271        });
272        ui.separator();
273        if ui.button("Hide").clicked() {
274            clicked = Some(own(Change::Hide(each.name.clone())));
275            ui.close();
276        }
277        for (up, words) in [(true, "Move up"), (false, "Move down")] {
278            if ui.button(words).clicked() {
279                clicked = Some(own(Change::Move {
280                    name: each.name.clone(),
281                    shown: names.to_vec(),
282                    up,
283                }));
284                ui.close();
285            }
286        }
287    });
288    clicked
289}

303    /// Two containers the game has opened, each holding a gem.
304    fn opened() -> GameState {
305        let mut parser = hydra_protocol::Parser::new();
306        let mut state = GameState::default();
307        for (id, title) in [("1", "Satchel"), ("2", "Pouch")] {
308            for line in [
309                format!("<container id='{id}' title=\"{title}\" target='#{id}'/>"),
310                format!("<inv id='{id}'> a <a exist=\"{id}0\" noun=\"gem\">{title} gem</a></inv>"),
311            ] {
312                for frame in parser.parse_line(&line) {
313                    state.apply(&frame);
314                }
315            }
316        }
317        state
318    }

320    /// A container is shown by its label, what it holds is listed, and its
321    /// arrow asks for it folded.
322    #[test]
323    fn a_container_is_labelled_and_folds_by_its_arrow() {
324        let state = opened();
325        let mut shown = ContainersShown {
326            who: "Tester".to_owned(),
327            ..ContainersShown::default()
328        };
329        shown
330            .choices
331            .apply(Change::Label("Pouch".to_owned(), Some("Gems".to_owned())));
332        let asked = Arc::new(Mutex::new(None));
333        let heard = Arc::clone(&asked);
334        let mut harness = Harness::builder()
335            .with_size((300.0, 200.0))
336            .build_ui(move |ui| {
337                if let Some(clicked) = containers(ui, Some(&state), &shown, egui::Id::new("c")) {
338                    *heard.lock().unwrap() = Some(clicked);
339                }
340            });
341        harness.run();
342        harness.get_by_label("Gems");
343        harness.get_by_label("Satchel gem");
344        harness.get_all_by_label("▾").next().unwrap().click();
345        harness.run();
346        let asked = asked.lock().unwrap().take();
347        assert!(
348            matches!(
349                asked,
350                Some(Clicked::Containers(Contained::Own(Change::Fold(_, true))))
351            ),
352            "{asked:?}"
353        );
354    }

356    /// A folded container's items are not drawn.
357    #[test]
358    fn a_folded_container_hides_what_it_holds() {
359        let state = opened();
360        let mut shown = ContainersShown::default();
361        shown.choices.apply(Change::Fold("Pouch".to_owned(), true));
362        let mut harness = Harness::builder()
363            .with_size((300.0, 200.0))
364            .build_ui(move |ui| {
365                let _ = containers(ui, Some(&state), &shown, egui::Id::new("c"));
366            });
367        harness.run();
368        harness.get_by_label("Satchel gem");
369        assert!(harness.query_by_label("Pouch gem").is_none());
370    }

1//! Custom indicators: one the player makes, with a name, lit by a rule of
2//! their own, and drawn as the game's indicators are (`status.rs`): a pill,
3//! or an icon, hidden while off when its look says.
4//!
5//! **Its rule** is one of two, chosen on its page:
6//! - **a condition**, built as a highlight's *When* is: vitals, effects,
7//!   the game's indicators, hands, the stance, the room, the flags triggers
8//!   set and the custom statuses they light;
9//! - **a game value**, any value of the game state, picked by its path as a
10//!   hub card's field is (`field.rs`), and a test of it: it is set, it
11//!   equals some words, it is more or less than a number.
12//!
13//! Each is kept in the layout by the widget's id: its name, its rule and its
14//! look together, a change on its page kept at once.
15//!
16//! **What cannot be told is off**, as a game indicator not yet reported is:
17//! one with no rule yet, with the character's state not known yet, or
18//! whose value is no longer in the game state is drawn off, and its hover
19//! says why.

35/// A custom indicator: its name, its rule and its look.
36#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
37pub(crate) struct CustomIndicator {
38    /// What it is called: drawn on its pill, and its page's and window's
39    /// title. None, [`UNNAMED`].
40    #[serde(default, skip_serializing_if = "String::is_empty")]
41    pub(crate) name: String,
42    /// Which rule lights it: the condition, or the game value.
43    #[serde(default)]
44    pub(crate) by: By,
45    /// The condition it is lit by, once one is built.
46    #[serde(default, skip_serializing_if = "Option::is_none")]
47    pub(crate) when: Option<Condition>,
48    /// The game value it is lit by: its path through the state's print,
49    /// once one is picked.
50    #[serde(default, skip_serializing_if = "Vec::is_empty")]
51    pub(crate) value: Vec<String>,
52    /// The test of the game value.
53    #[serde(default)]
54    pub(crate) test: Test,
55    /// What the game value is compared to: words or a number for
56    /// *equals*, a number for *more than* and *less than*.
57    #[serde(default, skip_serializing_if = "String::is_empty")]
58    pub(crate) than: String,
59    /// Its look: a pill or an icon, hidden while off or not.
60    #[serde(default, skip_serializing_if = "is_plain")]
61    pub(crate) look: IndicatorLook,
62}
63
64impl CustomIndicator {
65    /// What it is called: its own name, or [`UNNAMED`].
66    pub(crate) fn name(&self) -> &str {
67        if self.name.is_empty() {
68            UNNAMED
69        } else {
70            &self.name
71        }
72    }
73}

75/// Whether `look` is the plain one, a pill shown while off.
76fn is_plain(look: &IndicatorLook) -> bool {
77    *look == IndicatorLook::default()
78}

80/// Which of its rules lights a custom indicator.
81#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
82#[serde(rename_all = "snake_case")]
83pub(crate) enum By {
84    /// A condition holds.
85    #[default]
86    When,
87    /// A game value passes its test.
88    Value,
89}

91impl By {
92    /// Each, as its page's row writes it and as a player calls it.
93    const ALL: [(By, &'static str, &'static str); 2] = [
94        (By::When, "when", "a condition"),
95        (By::Value, "value", "a game value"),
96    ];
97}

99/// A test of a game value, as the Field widget says the value.
100#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
101#[serde(rename_all = "snake_case")]
102pub(crate) enum Test {
103    /// It holds something: not none, not empty, not false, not zero, not
104    /// an empty list.
105    #[default]
106    Set,
107    /// It is these words, case aside; or this number, when both are
108    /// numbers.
109    Equals,
110    /// It is a number more than this one.
111    More,
112    /// It is a number less than this one.
113    Less,
114}
115
116impl Test {
117    /// Each, as its page's row writes it and as a player calls it.
118    const ALL: [(Test, &'static str, &'static str); 4] = [
119        (Test::Set, "set", "is set"),
120        (Test::Equals, "equals", "equals"),
121        (Test::More, "more", "is more than"),
122        (Test::Less, "less", "is less than"),
123    ];
124}

139/// How a page's choice row offers `all`: each as it is written and called.
140fn choices<K>(all: &[(K, &str, &str)]) -> RowKind {
141    RowKind::Choice(
142        all.iter()
143            .map(|(_, written, called)| ((*written).to_owned(), (*called).to_owned()))
144            .collect(),
145    )
146}
147
148/// How `kind` is written, from `all`.
149fn written<K: PartialEq>(all: &[(K, &'static str, &str)], kind: &K) -> &'static str {
150    all.iter()
151        .find(|(one, _, _)| one == kind)
152        .map_or("", |(_, written, _)| *written)
153}

155/// A custom indicator's page rows: its name, which rule lights it and that
156/// rule's own rows, then how it is drawn, `every` what every indicator
157/// follows.
158pub(crate) fn rows(custom: Option<&CustomIndicator>, every: &status::EveryIndicator) -> Vec<Row> {
159    let custom = custom.cloned().unwrap_or_default();
160    let row =
161        |key: &str, label: &str, help: &str, kind: RowKind, (value, here): (Value, bool)| Row {
162            key: key.to_owned(),
163            label: label.to_owned(),
164            help: help.to_owned(),
165            kind,
166            value,
167            here,
168            from: None,
169            needs: None,
170        };
171    let mut rows = vec![
172        row(
173            NAME,
174            "Name",
175            "Drawn on its pill, and its window's title.",
176            RowKind::Text,
177            (Value::Text(custom.name.clone()), !custom.name.is_empty()),
178        ),
179        row(
180            BY,
181            "Lit by",
182            "A condition, built as a trigger's When is; or any value of the character's \
183             game state, and a test of it.",
184            choices(&By::ALL),
185            (
186                Value::Text(written(&By::ALL, &custom.by).to_owned()),
187                custom.by != By::default(),
188            ),
189        ),
190    ];
191    match custom.by {
192        By::When => rows.push(row(
193            WHEN,
194            "When",
195            "Lit while this holds: vitals, effects, the game's indicators, hands, the stance, \
196             the room, the flags triggers set and the custom statuses they light. The mark \
197             beside each test shows whether it holds now.",
198            RowKind::Condition,
199            (
200                custom
201                    .when
202                    .as_ref()
203                    .and_then(|when| serde_json::to_string(when).ok())
204                    .map_or(Value::Unset, Value::Text),
205                custom.when.is_some(),
206            ),
207        )),
208        By::Value => {
209            rows.push(row(
210                VALUE,
211                "Game value",
212                "Any value of the character's game state, picked from its print as a hub \
213                 card's field is.",
214                RowKind::GameValue,
215                (
216                    (!custom.value.is_empty())
217                        .then(|| serde_json::to_string(&custom.value).ok())
218                        .flatten()
219                        .map_or(Value::Unset, Value::Text),
220                    !custom.value.is_empty(),
221                ),
222            ));
223            rows.push(row(
224                TEST,
225                "Lit when it",
226                "Is set: it holds something, not none, empty, false or zero. Equals: these \
227                 words, case aside, or this number. More than, less than: a number. A list \
228                 or a record counts as how many it holds.",
229                choices(&Test::ALL),
230                (
231                    Value::Text(written(&Test::ALL, &custom.test).to_owned()),
232                    custom.test != Test::default(),
233                ),
234            ));
235            if custom.test != Test::Set {
236                rows.push(row(
237                    THAN,
238                    "Compared to",
239                    "The words or the number the value is compared to.",
240                    RowKind::Text,
241                    (Value::Text(custom.than.clone()), !custom.than.is_empty()),
242                ));
243            }
244        }
245    }
246    // How it is drawn: never a style's, which has no picture for it.
247    rows.push(Row::heading("How it is drawn"));
248    rows.extend(status::indicator_rows(
249        Some(&custom.look),
250        every,
251        (false, None),
252    ));
253    rows
254}

256/// `key`, one of a custom indicator's page rows, set to `to` in `custom`;
257/// back to its default with none.
258///
259/// # Errors
260///
261/// The key is not a custom indicator's, or `to` does not read as its
262/// row's.
263pub(crate) fn set(custom: &mut CustomIndicator, key: &str, to: Option<&str>) -> Result<(), String> {
264    match key {
265        NAME => to
266            .map(str::trim)
267            .unwrap_or_default()
268            .clone_into(&mut custom.name),
269        BY => {
270            custom.by = match to {
271                None => By::default(),
272                Some(written) => By::ALL
273                    .iter()
274                    .find(|(_, one, _)| *one == written)
275                    .map(|(by, _, _)| *by)
276                    .ok_or_else(|| format!("`{written}` is not a condition or a game value."))?,
277            };
278        }
279        WHEN => {
280            custom.when = match to {
281                None => None,
282                Some(written) => Some(
283                    serde_json::from_str(written)
284                        .map_err(|why| format!("That is not a condition: {why}."))?,
285                ),
286            };
287        }
288        VALUE => {
289            custom.value = match to {
290                None => Vec::new(),
291                Some(written) => serde_json::from_str(written)
292                    .map_err(|why| format!("That is not a game value's path: {why}."))?,
293            };
294        }
295        TEST => {
296            custom.test = match to {
297                None => Test::default(),
298                Some(written) => Test::ALL
299                    .iter()
300                    .find(|(_, one, _)| *one == written)
301                    .map(|(test, _, _)| *test)
302                    .ok_or_else(|| {
303                        format!("`{written}` is not a test: set, equals, more or less.")
304                    })?,
305            };
306        }
307        THAN => {
308            let than = to.map(str::trim).unwrap_or_default();
309            if matches!(custom.test, Test::More | Test::Less)
310                && !than.is_empty()
311                && number(than).is_none()
312            {
313                return Err(format!("`{than}` is not a number."));
314            }
315            than.clone_into(&mut custom.than);
316        }
317        _ => status::indicator_set(&mut custom.look, key, to)?,
318    }
319    Ok(())
320}

322/// Whether `value`, as the Field widget says a value, passes `test`
323/// against `than`.
324pub(crate) fn passes(test: Test, than: &str, value: &str) -> bool {
325    let (than, value) = (than.trim(), value.trim());
326    match test {
327        Test::Set => is_set(value),
328        Test::Equals => match (number(value), number(than)) {
329            (Some(left), Some(right)) => (left - right).abs() < f64::EPSILON,
330            _ => value.to_lowercase() == than.to_lowercase(),
331        },
332        Test::More => number(value)
333            .zip(number(than))
334            .is_some_and(|(left, right)| left > right),
335        Test::Less => number(value)
336            .zip(number(than))
337            .is_some_and(|(left, right)| left < right),
338    }
339}

341/// Whether a value holds something: not none (`—`), not empty, not false,
342/// not zero, not an empty list or record.
343fn is_set(value: &str) -> bool {
344    !matches!(value, "" | "—" | "false" | "[]" | "{}" | "\"\"")
345        && !number(value).is_some_and(|number| number.abs() < f64::EPSILON)
346}
347
348/// The number a value says: itself, or how many a list or record holds as
349/// the Field widget says it, after its kind -- `Effects (3)`.
350fn number(value: &str) -> Option<f64> {
351    let value = value.trim();
352    let number = value.parse::<f64>().ok().or_else(|| {
353        let (_, count) = value.strip_suffix(')')?.rsplit_once('(')?;
354        count.trim().parse::<f64>().ok()
355    })?;
356    number.is_finite().then_some(number)
357}

359/// Why it cannot be told whether a custom indicator is lit: no condition
360/// built yet.
361const NO_CONDITION: &str = "it has no condition yet: build one on its page.";
362/// No game value picked yet.
363const NO_VALUE: &str = "it has no game value yet: pick one on its page.";
364/// Nothing known of the character yet.
365const NOT_KNOWN: &str = "nothing is known of the character yet.";
366/// Its game value is not in the game state.
367const NOT_FOUND: &str = "its game value is not in the game state now: pick it again on its page.";

369/// Draw `custom` for the character in `state`, its flags and custom
370/// statuses `marks` (none for another character's); `name` what it is
371/// shown as and `shown` its look and icons; `key` where the state's print
372/// is kept, for a game value. One with no rule is drawn off.
373pub(super) fn draw(
374    ui: &mut egui::Ui,
375    custom: Option<&CustomIndicator>,
376    (state, marks): (Option<&GameState>, Option<(&Flags, &Flags)>),
377    (name, shown): (&str, Option<&IndicatorShown>),
378    key: Id,
379) {
380    let told = match custom {
381        Some(custom) => told(ui, custom, (state, marks), key),
382        None => Err(NO_CONDITION),
383    };
384    let told = told.map_err(|why| ("not known", format!("{name}: {why} It counts as off.")));
385    status::lamp(ui, name, &told, T::Accent, shown);
386}

388/// Whether `custom` is lit for the character in `state`, its flags and
389/// custom statuses `marks`; why that cannot be told, when it cannot.
390fn told(
391    ui: &egui::Ui,
392    custom: &CustomIndicator,
393    (state, marks): (Option<&GameState>, Option<(&Flags, &Flags)>),
394    key: Id,
395) -> Result<bool, &'static str> {
396    match custom.by {
397        By::When => {
398            let when = custom.when.as_ref().ok_or(NO_CONDITION)?;
399            let state = state.ok_or(NOT_KNOWN)?;
400            let none = Flags::default();
401            let (flags, statuses) = marks.unwrap_or((&none, &none));
402            // A flag, a status or an effect runs out on the clock, with no
403            // word from the game: looked at again each second.
404            ui.ctx().request_repaint_after(Duration::from_secs(1));
405            Ok(when.holds(&World {
406                state,
407                flags,
408                statuses,
409                unix: unix_now(),
410            }))
411        }
412        By::Value => {
413            if custom.value.is_empty() {
414                return Err(NO_VALUE);
415            }
416            let state = state.ok_or(NOT_KNOWN)?;
417            let value = super::field::read(ui, state, &custom.value, key).ok_or(NOT_FOUND)?;
418            Ok(passes(custom.test, &custom.than, &value))
419        }
420    }
421}

423/// The wall clock's seconds, for a condition's time of day.
424fn unix_now() -> i64 {
425    std::time::SystemTime::now()
426        .duration_since(std::time::UNIX_EPOCH)
427        .map_or(0, |since| {
428            i64::try_from(since.as_secs()).unwrap_or(i64::MAX)
429        })
430}

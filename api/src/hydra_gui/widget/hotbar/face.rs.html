1//! A button's icon drawn on its face: a sheet's cell, or an image scaled to
2//! fit keeping its shape; the whole face with `icon`, or a square at the
3//! left with its label beside it with `icon_and_label`; grey when the icon
4//! says, or the button is dimmed; and its border, one colour or two run
5//! one way into the other.

13/// The icons folder and its sheets, and the bar's icons' edge.
14#[derive(Clone, Debug, Default)]
15pub(crate) struct Icons {
16    pub(crate) dir: PathBuf,
17    pub(crate) sheets: Sheets,
18    /// The bar's icons' edge in points; as big as a button allows when
19    /// none.
20    pub(crate) size: Option<u32>,
21}

23/// Where a face's label goes once its icon is drawn.
24pub(super) enum Label {
25    /// Over the whole face: it has no icon, or its icon could not be drawn.
26    Whole,
27    /// Here, beside its icon.
28    In(Rect),
29    /// Nowhere: its icon is its face.
30    Not,
31}
32
33/// The icon `look` shows, drawn in `inner` with `painter` at `opacity`:
34/// where its label goes.
35pub(super) fn icon(
36    painter: &egui::Painter,
37    inner: Rect,
38    look: &Look,
39    icons: &Icons,
40    opacity: f32,
41) -> Label {
42    let Some(icon) = look.icon.as_ref().filter(|_| look.face != IconMode::Text) else {
43        return Label::Whole;
44    };
45    let Some((texture, uv)) = texture(painter.ctx(), icon, icons, look.dim) else {
46        return Label::Whole;
47    };
48    #[expect(
49        clippy::cast_precision_loss,
50        reason = "an icon is a few hundred points"
51    )]
52    let most = icons.size.map_or(f32::INFINITY, |size| size as f32);
53    let (place, label) = if look.face == IconMode::IconAndLabel {
54        {
55            let edge = inner.height().min(inner.width() / 2.0).min(most);
56            let place = Rect::from_min_size(
57                Pos2::new(inner.left(), inner.center().y - edge / 2.0),
58                Vec2::splat(edge),
59            );
60            let rest = Rect::from_min_max(
61                Pos2::new(place.right() + 4.0, inner.top()),
62                inner.right_bottom(),
63            );
64            (place, Label::In(rest))
65        }
66    } else {
67        let edge = inner.width().min(inner.height()).min(most);
68        (
69            Rect::from_center_size(inner.center(), Vec2::splat(edge)),
70            Label::Not,
71        )
72    };
73    // An image keeps its shape inside the square; a cell is square.
74    let size = texture.size_vec2() * uv.size();
75    let scale = (place.width() / size.x).min(place.height() / size.y);
76    let drawn = Rect::from_center_size(place.center(), size * scale);
77    let tint = Color32::WHITE.gamma_multiply(if look.dim { 0.6 } else { 1.0 } * opacity);
78    painter.image(texture.id(), drawn, uv, tint);
79    border(painter, drawn, icon, opacity);
80    label
81}

83/// `icon` drawn in `place` keeping its shape, grey when it says or `grey`,
84/// at `opacity`, with its border: `false` when its picture is not there to
85/// draw yet, or at all.
86pub(crate) fn paint_icon(
87    painter: &egui::Painter,
88    place: Rect,
89    icon: &Icon,
90    icons: &Icons,
91    (grey, opacity): (bool, f32),
92) -> bool {
93    let Some((texture, uv)) = texture(painter.ctx(), icon, icons, grey) else {
94        return false;
95    };
96    let size = texture.size_vec2() * uv.size();
97    let scale = (place.width() / size.x).min(place.height() / size.y);
98    let drawn = Rect::from_center_size(place.center(), size * scale);
99    painter.image(
100        texture.id(),
101        drawn,
102        uv,
103        Color32::WHITE.gamma_multiply(opacity),
104    );
105    border(painter, drawn, icon, opacity);
106    true
107}

109/// The texture `icon` is drawn from, grey when it says or `dim`, and the
110/// part of it that is the icon.
111fn texture(
112    context: &egui::Context,
113    icon: &Icon,
114    icons: &Icons,
115    dim: bool,
116) -> Option<(egui::TextureHandle, Rect)> {
117    let (file, cell) = hydra_ui::hotbar::source(&icon.art, &icons.dir, &icons.sheets)?;
118    let file = file.to_string_lossy();
119    let texture = if icon.grayscale || dim {
120        crate::pictures::grey_picture(context, &file)
121    } else {
122        crate::pictures::picture(context, &file)
123    }?;
124    let uv = match cell {
125        None => Rect::from_min_max(Pos2::ZERO, Pos2::new(1.0, 1.0)),
126        Some((cell, edge)) => {
127            let [width, height] = texture.size();
128            let size = (u32::try_from(width).ok()?, u32::try_from(height).ok()?);
129            let [left, top, right, bottom] = hydra_ui::hotbar::cell_in(cell, edge, size)?;
130            Rect::from_min_max(Pos2::new(left, top), Pos2::new(right, bottom))
131        }
132    };
133    Some((texture, uv))
134}

136fn colour(hex: Option<&String>) -> Option<Color32> {
137    hex.and_then(|hex| hydra_ui::theme::parse_hex(hex))
138        .map(|[red, green, blue]| Color32::from_rgb(red, green, blue))
139}

141/// The icon's border over `rect`, when it has one.
142fn border(painter: &egui::Painter, rect: Rect, icon: &Icon, opacity: f32) {
143    let Some(start) = colour(icon.border.as_ref()) else {
144        return;
145    };
146    let width = f32::from(icon.border_width.unwrap_or(2).clamp(1, 10));
147    let start = start.gamma_multiply(opacity);
148    let Some(end) = colour(icon.border_end.as_ref()).map(|end| end.gamma_multiply(opacity)) else {
149        painter.rect_stroke(
150            rect,
151            0.0,
152            Stroke::new(width, start),
153            egui::StrokeKind::Inside,
154        );
155        return;
156    };
157    painter.add(ring(rect, width, (start, end), icon.border_dir));
158}

160/// A border `width` wide inside `rect`, run from `start` to `end` along
161/// `dir`: a ring of eight pieces, each corner and each side's middle its
162/// own colour.
163fn ring(rect: Rect, width: f32, (start, end): (Color32, Color32), dir: GradientDir) -> Mesh {
164    let around = |rect: Rect| {
165        let (min, max, mid) = (rect.min, rect.max, rect.center());
166        [
167            Pos2::new(min.x, min.y),
168            Pos2::new(mid.x, min.y),
169            Pos2::new(max.x, min.y),
170            Pos2::new(max.x, mid.y),
171            Pos2::new(max.x, max.y),
172            Pos2::new(mid.x, max.y),
173            Pos2::new(min.x, max.y),
174            Pos2::new(min.x, mid.y),
175        ]
176    };
177    let shade = |at: Pos2| {
178        let u = ((at.x - rect.min.x) / rect.width().max(1.0)).clamp(0.0, 1.0);
179        let v = ((at.y - rect.min.y) / rect.height().max(1.0)).clamp(0.0, 1.0);
180        let t = match dir {
181            GradientDir::Horizontal => u,
182            GradientDir::Vertical => v,
183            GradientDir::DiagonalDown => u.midpoint(v),
184            GradientDir::DiagonalUp => u.midpoint(1.0 - v),
185            GradientDir::Radial => ((u - 0.5).hypot(v - 0.5) / 0.5_f32.hypot(0.5)).min(1.0),
186            GradientDir::Square => ((u - 0.5).abs().max((v - 0.5).abs()) * 2.0).min(1.0),
187        };
188        lerp(start, end, t)
189    };
190    let mut mesh = Mesh::default();
191    let (outer, inner) = (around(rect), around(rect.shrink(width)));
192    for (outer, inner) in outer.iter().zip(inner) {
193        for at in [*outer, inner] {
194            mesh.vertices.push(Vertex {
195                pos: at,
196                uv: WHITE_UV,
197                color: shade(at),
198            });
199        }
200    }
201    for piece in 0..8_u32 {
202        let (here, next) = (piece * 2, ((piece + 1) % 8) * 2);
203        mesh.add_triangle(here, here + 1, next);
204        mesh.add_triangle(next, here + 1, next + 1);
205    }
206    mesh
207}

209/// `start` run `t` of the way to `end`.
210fn lerp(start: Color32, end: Color32, t: f32) -> Color32 {
211    let mix = |from: u8, to: u8| {
212        let mixed = f32::from(from) + (f32::from(to) - f32::from(from)) * t;
213        #[expect(
214            clippy::cast_possible_truncation,
215            clippy::cast_sign_loss,
216            reason = "a channel, clamped to a byte"
217        )]
218        let mixed = mixed.round().clamp(0.0, 255.0) as u8;
219        mixed
220    };
221    Color32::from_rgba_premultiplied(
222        mix(start.r(), end.r()),
223        mix(start.g(), end.g()),
224        mix(start.b(), end.b()),
225        mix(start.a(), end.a()),
226    )
227}

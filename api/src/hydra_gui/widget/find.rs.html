1//! Find in a story or a stream: what the play window's
2//! Find bar looks for, asked of the widget by its id as a key's scroll is,
3//! and how many lines it found, said back.
4//!
5//! The widget counts its lines that hold what is looked for, as they are
6//! drawn, whatever their case; each is marked, the current one more; and
7//! when the bar moves to another, the pane the player scrolls is brought to
8//! it, which splits the story as scrolling back does. The current one counts
9//! from the newest, so the first found is the latest said and *next* goes
10//! back through what was.
11
12use egui::Id;
13
14use crate::theme::{self, T};
15
16/// What the Find bar looks for in a widget.
17#[derive(Clone, Debug, Default, PartialEq, Eq)]
18pub(crate) struct Finding {
19    /// What is looked for, as typed.
20    pub(crate) query: String,
21    /// Which line found is current, from the newest: 0 is the latest.
22    pub(crate) current: usize,
23    /// The pane the player scrolls is to be brought to the current one.
24    pub(crate) go: bool,
25}
26
27/// Where a widget's finding is kept, and the count it said back.
28const ASKED: &str = "find";
29const FOUND: &str = "found";
30
31/// Ask the widget whose id is `id` to find `finding`, from this frame on;
32/// `None` stops.
33pub(crate) fn ask(context: &egui::Context, id: Id, finding: Option<Finding>) {
34    context.data_mut(|data| {
35        if let Some(finding) = finding {
36            data.insert_temp(id.with(ASKED), finding);
37        } else {
38            data.remove::<Finding>(id.with(ASKED));
39            data.remove::<usize>(id.with(FOUND));
40        }
41    });
42}
43
44/// How many lines the widget whose id is `id` found when last drawn.
45pub(crate) fn found(context: &egui::Context, id: Id) -> Option<usize> {
46    context.data(|data| data.get_temp(id.with(FOUND)))
47}
48
49/// A widget's lines looked through this frame, as they are drawn.
50pub(super) struct Seek {
51    /// What is looked for, in lower case.
52    query: String,
53    /// The line found that is current, counted from the oldest.
54    current: Option<usize>,
55    go: bool,
56    /// Lines found so far this frame.
57    seen: usize,
58}
59
60impl Seek {
61    /// What the widget `id` is to find this frame among `lines`, the text of
62    /// each it will draw, in order; `None` when it is to find nothing. How
63    /// many hold it is said back. `go` is taken: the pane is brought to the
64    /// current line once.
65    pub(super) fn of<'a>(
66        ui: &egui::Ui,
67        id: Id,
68        lines: impl Iterator<Item = std::borrow::Cow<'a, str>>,
69    ) -> Option<Self> {
70        let finding = ui.data(|data| data.get_temp::<Finding>(id.with(ASKED)))?;
71        let query = finding.query.to_lowercase();
72        let found = if query.trim().is_empty() {
73            0
74        } else {
75            lines.filter(|line| Self::holds(&query, line)).count()
76        };
77        ui.data_mut(|data| data.insert_temp(id.with(FOUND), found));
78        if found == 0 {
79            return None;
80        }
81        if finding.go {
82            let taken = Finding {
83                go: false,
84                ..finding.clone()
85            };
86            ui.data_mut(|data| data.insert_temp(id.with(ASKED), taken));
87        }
88        let current = found
89            .checked_sub(1)
90            .map(|last| last.saturating_sub(finding.current.min(last)));
91        Some(Self {
92            query,
93            current,
94            go: finding.go,
95            seen: 0,
96        })
97    }
98
99    /// Whether `text`, a line's, holds what is looked for.
100    pub(super) fn holds(query: &str, text: &str) -> bool {
101        text.to_lowercase().contains(query)
102    }
103
104    /// A pane begins: its lines are counted afresh, each pane drawing all.
105    pub(super) fn start(&mut self) {
106        self.seen = 0;
107    }
108
109    /// A line drawn by `draw` into `ui`, whose text is `text`: marked when
110    /// it holds what is looked for, the current one more, and the pane
111    /// brought to it when `player` -- the pane the player scrolls -- and the
112    /// bar moved.
113    pub(super) fn line(
114        &mut self,
115        ui: &mut egui::Ui,
116        (text, player): (&str, bool),
117        draw: impl FnOnce(&mut egui::Ui),
118    ) {
119        let top = ui.cursor().top();
120        draw(ui);
121        if !Self::holds(&self.query, text) {
122            return;
123        }
124        let rect = egui::Rect::from_x_y_ranges(ui.max_rect().x_range(), top..=ui.cursor().top());
125        let current = self.current == Some(self.seen);
126        let fill =
127            theme::color(ui.ctx(), T::Accent).gamma_multiply(if current { 0.35 } else { 0.12 });
128        ui.painter()
129            .rect_filled(rect, theme::corner(ui.ctx()), fill);
130        if current && self.go && player {
131            let now = egui::style::ScrollAnimation::none();
132            ui.scroll_to_rect_animation(rect, Some(egui::Align::Center), now);
133        }
134        self.seen += 1;
135    }
136}

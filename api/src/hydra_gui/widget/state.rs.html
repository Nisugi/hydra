1//! The Game state widget: everything in `GameState`, live, for
2//! troubleshooting.
3//!
4//! It shows everything the model holds for the character, as the model
5//! prints itself (`GameState`'s `Debug`). So no field is left off, and a
6//! field added to the model shows here with no change to this file. It is a
7//! tree, closed below its top; a filter finds any line and shows where it
8//! is; and *Copy* takes the whole print for a report.
9//!
10//! The committed fixtures, folded together, print about 600 KB: 15,676
11//! lines, formatted in 6 ms in a debug build. So the print is taken at most
12//! twice a second, not every frame, and kept where whatever reads it finds
13//! it by one key ([`printed`]): a hub card's fields share their character's
14//! (`field.rs`).
15
16use std::sync::Arc;
17use std::time::{Duration, Instant};
18
19use egui::{Id, RichText};
20use hydra_model::GameState;
21
22/// How often the print is taken again: often enough to watch, rarely
23/// enough to cost nothing.
24pub(super) const EVERY: Duration = Duration::from_millis(500);
25
26/// The most lines the filter lists.
27const FOUND: usize = 500;
28
29/// One line of the print: a value, or a branch that opens onto its lines.
30#[derive(Debug, PartialEq, Eq)]
31pub(super) enum Line {
32    /// A value, as printed.
33    Leaf(String),
34    /// A struct, list or wrapper, by the line that opens it.
35    Branch(String, Vec<Line>),
36}
37
38/// The print, as text and as a tree, and when it was taken.
39pub(super) struct Printed {
40    at: Instant,
41    text: String,
42    pub(super) tree: Vec<Line>,
43}
44
45/// `state`'s print, kept under `key` and taken again once it is [`EVERY`]
46/// old.
47pub(super) fn printed(ui: &egui::Ui, state: &GameState, key: Id) -> Arc<Printed> {
48    let now = Instant::now();
49    let kept = ui
50        .data(|data| data.get_temp::<Arc<Printed>>(key))
51        .filter(|printed| now.saturating_duration_since(printed.at) < EVERY);
52    kept.unwrap_or_else(|| {
53        let text = format!("{state:#?}");
54        let printed = Arc::new(Printed {
55            at: now,
56            tree: tree(&text),
57            text,
58        });
59        ui.data_mut(|data| data.insert_temp(key, Arc::clone(&printed)));
60        printed
61    })
62}
63
64/// The fields of the print's one top branch, `GameState { .. }`: the tree's
65/// top as a reader sees it.
66pub(super) fn top(tree: &[Line]) -> &[Line] {
67    match tree {
68        [Line::Branch(_, fields)] => fields.as_slice(),
69        all => all,
70    }
71}

73/// `{:#?}`'s print as a tree. A line ending `{`, `[` or `(` opens a
74/// branch, and one starting `}`, `]` or `)` closes it. A branch holding one
75/// value is that value on one line: `id: Some("8213304")`, not three lines.
76pub(super) fn tree(text: &str) -> Vec<Line> {
77    // The open branches: each's label, its bracket, and its lines so far.
78    let mut open: Vec<(String, char, Vec<Line>)> = vec![(String::new(), ' ', Vec::new())];
79    for line in text.lines().map(str::trim) {
80        if line.starts_with(['}', ']', ')']) {
81            close(&mut open, line.ends_with(','));
82        } else if let Some(bracket) = line.chars().last().filter(|c| matches!(c, '{' | '[' | '(')) {
83            let label = line[..line.len() - 1].trim_end().to_owned();
84            open.push((label, bracket, Vec::new()));
85        } else if let Some((_, _, lines)) = open.last_mut() {
86            lines.push(Line::Leaf(line.to_owned()));
87        }
88    }
89    // A print cut short leaves branches open; they close where it stops.
90    while open.len() > 1 {
91        close(&mut open, false);
92    }
93    open.pop().map(|(_, _, lines)| lines).unwrap_or_default()
94}
95
96/// Close the innermost open branch into its parent.
97fn close(open: &mut Vec<(String, char, Vec<Line>)>, comma: bool) {
98    if open.len() < 2 {
99        return;
100    }
101    let Some((label, bracket, mut lines)) = open.pop() else {
102        return;
103    };
104    let line = match (lines.len(), lines.first()) {
105        (1, Some(Line::Leaf(only))) => {
106            let closing = match bracket {
107                '{' => '}',
108                '[' => ']',
109                _ => ')',
110            };
111            let only = only.trim_end_matches(',');
112            let comma = if comma { "," } else { "" };
113            let text = if bracket == '{' {
114                format!("{label} {{ {only} }}{comma}")
115            } else {
116                format!("{label}{bracket}{only}{closing}{comma}")
117            };
118            lines.clear();
119            Line::Leaf(text)
120        }
121        _ => Line::Branch(label, lines),
122    };
123    if let Some((_, _, parent)) = open.last_mut() {
124        parent.push(line);
125    }
126}

128/// Every line whose text holds `needle` (ignoring case), each with the
129/// branches it is under, at most [`FOUND`].
130pub(super) fn found(lines: &[Line], needle: &str) -> Vec<String> {
131    fn walk<'a>(lines: &'a [Line], needle: &str, under: &mut Vec<&'a str>, out: &mut Vec<String>) {
132        for line in lines {
133            if out.len() >= FOUND {
134                return;
135            }
136            let (text, inside) = match line {
137                Line::Leaf(text) => (text.as_str(), None),
138                Line::Branch(label, lines) => (label.as_str(), Some(lines)),
139            };
140            if text.to_lowercase().contains(needle) {
141                let mut path = under.join(" › ");
142                if !path.is_empty() {
143                    path.push_str(" › ");
144                }
145                out.push(format!("{path}{text}"));
146            }
147            if let Some(inside) = inside {
148                under.push(text);
149                walk(inside, needle, under, out);
150                under.pop();
151            }
152        }
153    }
154    let mut out = Vec::new();
155    walk(lines, &needle.to_lowercase(), &mut Vec::new(), &mut out);
156    out
157}

159/// The widget: the state's print, a filter, and Copy.
160pub(super) fn game_state(ui: &mut egui::Ui, state: Option<&GameState>, who: Option<&str>, id: Id) {
161    if let Some(who) = who {
162        ui.weak(who);
163    }
164    let Some(state) = state else {
165        ui.weak("Nothing seen yet.");
166        return;
167    };
168    let printed = printed(ui, state, id);
169    // Drawn again while open, so what it shows is never older than EVERY.
170    ui.ctx().request_repaint_after(EVERY);
171    let filter_id = id.with("filter");
172    let mut filter = ui.data(|data| data.get_temp::<String>(filter_id).unwrap_or_default());
173    ui.horizontal(|ui| {
174        ui.add(
175            egui::TextEdit::singleline(&mut filter)
176                .hint_text("Find in the game state...")
177                .desired_width((ui.available_width() - 60.0).max(80.0)),
178        );
179        if ui
180            .button("Copy")
181            .on_hover_text("The whole game state, as text, for a report")
182            .clicked()
183        {
184            ui.ctx().copy_text(printed.text.clone());
185        }
186    });
187    ui.data_mut(|data| data.insert_temp(filter_id, filter.clone()));
188    egui::ScrollArea::both()
189        .id_salt(id.with("scroll"))
190        .auto_shrink(false)
191        .show(ui, |ui| {
192            let needle = filter.trim();
193            if needle.is_empty() {
194                draw(ui, top(&printed.tree), id.with("tree"));
195                return;
196            }
197            let found = found(&printed.tree, needle);
198            if found.is_empty() {
199                ui.weak("Nothing matches.");
200            }
201            for line in found {
202                ui.label(RichText::new(line).monospace());
203            }
204        });
205}

207/// `lines`, each branch closed until opened.
208fn draw(ui: &mut egui::Ui, lines: &[Line], id: Id) {
209    for (at, line) in lines.iter().enumerate() {
210        match line {
211            Line::Leaf(text) => {
212                ui.label(RichText::new(text).monospace());
213            }
214            Line::Branch(label, inside) => {
215                let id = id.with(at);
216                egui::CollapsingHeader::new(RichText::new(label).monospace())
217                    .id_salt(id)
218                    .show(ui, |ui| draw(ui, inside, id));
219            }
220        }
221    }
222}

1//! The Targets and Players widgets: who is in the room, one to a line,
2//! with what the game says of each.
3//!
4//! **Targets** stacks the hostile creatures: each a link to it, marked
5//! when it is the one targeted, its statuses after it, and its health as
6//! a thin bar with the percent when the game states it. A dead creature,
7//! an animated decoy and a bare appendage (an arm, a claw, a tentacle) are
8//! left out, as `CreatureInstance::valid_target` leaves them; a line at
9//! the foot can count the appendages instead. Its page chooses how the
10//! name is said (whole, without its article, the noun, or the noun once
11//! the line is full), the order of name, statuses and percent on the
12//! line, the bar and how tall it is, the percent, the bar's colour from
13//! each percent up, the colour of the one targeted, the bestiary book
14//! after the name, and the statuses: how each is said (the game's word,
15//! or whatever the player puts in its place, `STN` or a glyph), in what
16//! colour, in what order of importance, and how many are shown at most.
17//! What is chosen is the board's [`TargetsLook`], which the web page
18//! follows too.
19//!
20//! **Players** stacks the players: each a link, what they are doing after
21//! it (`hiding`, `sitting`, `appears dead`), and `[G]` before a member of
22//! the character's group when asked.
23

38/// What the player chose for one Targets or Players widget on its page.
39#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
40pub(crate) struct RosterLook {
41    /// Targets: everything chosen for the list, as the page follows it.
42    #[serde(flatten)]
43    pub(crate) targets: TargetsLook,
44    /// Players: `[G]` before each member of the character's group.
45    #[serde(default, skip_serializing_if = "std::ops::Not::not")]

47    /// Players: one the room says is dead drawn faint.
48    #[serde(default = "yes", skip_serializing_if = "Clone::clone")]

50}

86    /// What stands for `status`: what the player chose, or its word.

89        self.targets.status_shown(status)

92    /// The bar's colour at `percent` left: the player's thresholds, from
93    /// each percent up; with none, the theme's health colour above half,
94    /// the warning colour to a quarter, the wound colour below.
95    fn health_color(&self, ctx: &egui::Context, percent: u32) -> Color32 {
96        if let Some([r, g, b]) = self.targets.threshold_color(percent) {
97            return Color32::from_rgb(r, g, b);
98        }
99        if !self.targets.thresholds.is_empty() || percent > 50 {
100            theme::color(ctx, T::Health)
101        } else if percent > 25 {
102            theme::color(ctx, T::Warning)
103        } else {
104            theme::color(ctx, T::Wrong)
105        }
106    }
107
108    /// The colour of the one targeted.
109    fn current_color(&self, ctx: &egui::Context) -> Color32 {
110        self.targets.current.map_or_else(
111            || theme::color(ctx, T::Accent),
112            |[r, g, b]| Color32::from_rgb(r, g, b),
113        )
114    }

295/// The Targets widget's rows.
296pub(crate) fn targets_rows(look: Option<&RosterLook>) -> Vec<Row> {
297    let default = TargetsLook::default();
298    let look = look.map(|look| look.targets.clone()).unwrap_or_default();
299    let parts: Vec<(&str, &str)> = PARTS
300        .iter()
301        .map(|(_, written, called)| (*written, *called))
302        .collect();
303    let mut rows = name_rows(&look, &default);
304    rows.extend(health_rows(&look, &default));
305    rows.push(heading("Order on the line"));
306    for (at, place) in PLACES.iter().enumerate() {
307        rows.push(choice_row(
308            place,
309            &format!("{} part", ["First", "Second", "Third"][at]),
310            "Which of the name, the statuses and the health percent comes here on the line; \
311             the part that was here goes where this one was, so each is on the line once.",
312            &parts,
313            look.order[at].as_str(),
314            look.order != default.order,
315        ));
316    }
317    rows.push(heading("Statuses"));
318    rows.push(Row {
319        key: SHOWN.to_owned(),
320        label: "Statuses shown".to_owned(),
321        help: "How many of a creature's statuses are on its line at most, the most \
322               important first as the table orders them; 0, all of them."
323            .to_owned(),
324        kind: RowKind::Whole { min: 0, max: 40 },
325        value: Value::Text(look.shown.to_string()),
326        here: look.shown != default.shown,
327        from: None,
328        needs: None,
329    });
330    rows.push(toggle(
331        WRAP,
332        "Wrap",
333        "A line longer than the widget wraps onto the next; off, it is cut at the edge \
334         with the statuses that matter first, and a wider widget shows more.",
335        look.wrap,
336        default.wrap,
337    ));
338    rows.push(Row {
339        key: STATUSES.to_owned(),
340        label: "Statuses".to_owned(),
341        help: "Every status the game can say, the most important first: move one up or \
342               down to rank it. What stands for it on the line: the game's word as it \
343               comes, or put anything in its place, STN or a glyph. Its colour; none, \
344               faint text."
345            .to_owned(),
346        kind: RowKind::Table(vec![
347            Column::words("Status"),
348            Column::words("Shown as"),
349            Column::colour_or_none("Colour"),
350        ]),
351        value: Value::Table(
352            look.listed()
353                .into_iter()
354                .map(|status| {
355                    vec![
356                        status.status.clone(),
357                        status.shown.unwrap_or(status.status),
358                        status.color.map(crate::menu::hex).unwrap_or_default(),
359                    ]
360                })
361                .collect(),
362        ),
363        here: !look.statuses.is_empty(),
364        from: None,
365        needs: None,
366    });
367    rows
368}

392/// On or off, as the page writes a switch; none, `default`.
393fn switch(to: Option<&str>, default: bool) -> Result<bool, String> {
394    match to.map(str::trim) {
395        None => Ok(default),
396        Some("on") => Ok(true),
397        Some("off") => Ok(false),
398        Some(other) => Err(format!("`{other}` is not on or off.")),
399    }
400}

402/// A whole number from `min` to `max`, as the page writes one; none,
403/// `default`.
404fn whole(to: Option<&str>, default: u32, min: u32, max: u32) -> Result<u32, String> {
405    let Some(written) = to.map(str::trim) else {
406        return Ok(default);
407    };
408    written
409        .parse::<u32>()
410        .ok()
411        .filter(|n| (min..=max).contains(n))
412        .ok_or_else(|| format!("`{written}` is not a whole number from {min} to {max}."))
413}

415/// A colour, as the page writes one; empty or none, none.
416fn color(to: Option<&str>) -> Result<Option<[u8; 3]>, String> {
417    match to.map(str::trim).filter(|to| !to.is_empty()) {
418        None => Ok(None),
419        Some(to) => crate::menu::rgb(to)
420            .map(Some)
421            .ok_or_else(|| format!("`{to}` is not a colour.")),
422    }
423}

425/// The statuses the table `written` says, in its order: each row the
426/// game's word, what stands for it and its colour. Kept as chosen: every
427/// row when the order is not the game's, so the order is; else only the
428/// rows that say something other than the word, or a colour.
429fn statuses(written: &str) -> Result<Vec<StatusLook>, String> {
430    use hydra_board::targets::{status_named, word};
431    let rows: Vec<Vec<String>> =
432        serde_json::from_str(written).map_err(|_| "not rows of cells".to_owned())?;
433    let mut listed: Vec<StatusLook> = Vec::new();
434    for mut cells in rows {
435        cells.resize(3, String::new());
436        let status = cells[0].trim();
437        if status.is_empty() {
438            continue;
439        }
440        let Some(known) = status_named(status) else {
441            return Err(format!("`{status}` is not a status the game says."));
442        };
443        let status = word(known);
444        if listed.iter().any(|look| look.status == status) {
445            continue;
446        }
447        let shown = cells[1].trim();
448        listed.push(StatusLook {
449            shown: (!shown.is_empty() && shown != status).then(|| shown.to_owned()),
450            color: color(Some(&cells[2]))?,
451            status,
452        });
453    }
454    let game_order: Vec<String> = StatusName::ALL
455        .into_iter()
456        .map(word)
457        .filter(|word| listed.iter().any(|look| look.status == *word))
458        .collect();
459    let in_game_order = listed
460        .iter()
461        .map(|look| look.status.clone())
462        .collect::<Vec<_>>()
463        == game_order;
464    if in_game_order {
465        listed.retain(|look| look.shown.is_some() || look.color.is_some());
466    }
467    Ok(listed)
468}

470/// Set the widget's `key` to `to`, or back to its own.
471pub(crate) fn set(look: &mut RosterLook, key: &str, to: Option<&str>) -> Result<(), String> {
472    let default = TargetsLook::default();
473    let targets = &mut look.targets;
474    if let Some(place) = PLACES.iter().position(|place| *place == key) {
475        match to {
476            None => targets.order = default.order,
477            Some(to) => {
478                let part = Part::parse(to)?;
479                if let Some(was) = targets.order.iter().position(|here| *here == part) {
480                    targets.order.swap(place, was);
481                }
482            }
483        }
484        return Ok(());
485    }
486    match key {
487        ARMS => targets.arms = switch(to, default.arms)?,
488        GROUP => look.group = switch(to, false)?,
489        NAME => targets.name = to.map_or(Ok(default.name), NameShown::parse)?,
490        BOOK => targets.book = switch(to, default.book)?,
491        BAR => targets.bar = switch(to, default.bar)?,
492        HEIGHT => targets.height = whole(to, default.height, 1, 40)?,
493        PERCENT => targets.percent = switch(to, default.percent)?,
494        DIM_DEAD => look.dim_dead = switch(to, true)?,
495        BOSS => targets.boss = color(to)?,
496        CURRENT => targets.current = color(to)?,
497        SHOWN => targets.shown = whole(to, default.shown, 0, 40)?,
498        WRAP => targets.wrap = switch(to, default.wrap)?,
499        THRESHOLDS => {
500            targets.thresholds = match to {
501                None => Vec::new(),
502                Some(written) => crate::play::thresholds(written)?,
503            };
504        }
505        STATUSES => {
506            targets.statuses = match to {
507                None => Vec::new(),
508                Some(written) => statuses(written)?,
509            };
510        }
511        other => return Err(format!("The list has no setting {other}.")),
512    }
513    Ok(())
514}

526/// Draw the Targets list for `snapshot`; links live when `own`.
527pub(super) fn draw_targets(
528    ui: &mut egui::Ui,
529    snapshot: Option<&Snapshot>,
530    tags: Option<hydra_model::targetid::Style>,
531    look: &RosterLook,
532    own: bool,
533) -> Option<Clicked> {
534    let Some((listed, arms)) = snapshot.and_then(|snapshot| targets(snapshot, tags, look)) else {
535        ui.weak("Targets unknown");
536        return None;
537    };
538    if listed.is_empty() {
539        ui.weak("No targets");
540    }
541    let mut clicked = None;
542    for target in &listed {
543        clicked = clicked.or(draw_target(ui, target, look, own));
544    }
545    if look.targets.arms && arms > 0 {
546        ui.weak(format!(
547            "{arms} {}",
548            if arms == 1 { "appendage" } else { "appendages" }
549        ));
550    }
551    clicked
552}

554/// The statuses as one run of text: each in its colour, faint unless one
555/// was chosen, a comma between.
556fn statuses_job(ui: &egui::Ui, statuses: &[(String, Option<[u8; 3]>)]) -> LayoutJob {
557    let font = egui::TextStyle::Body.resolve(ui.style());
558    let weak = ui.visuals().weak_text_color();
559    let format = |color: Color32| TextFormat {
560        font_id: font.clone(),
561        color,
562        ..TextFormat::default()
563    };
564    let mut job = LayoutJob::default();
565    for (at, (status, color)) in statuses.iter().enumerate() {
566        if at > 0 {
567            job.append(", ", 0.0, format(weak));
568        }
569        let color = color.map_or(weak, |[r, g, b]| Color32::from_rgb(r, g, b));
570        job.append(status, 0.0, format(color));
571    }
572    job
573}

580/// How wide `words` are, unwrapped, in the body font.
581fn words_width(ui: &egui::Ui, words: &str) -> f32 {
582    let font = egui::TextStyle::Body.resolve(ui.style());
583    ui.painter()
584        .layout_no_wrap(words.to_owned(), font, Color32::WHITE)

586        .x

589/// How wide the parts the look puts after the statuses are on one row,
590/// each with the gap before it: the statuses are cut short of them, so
591/// they stay in sight. `name` is the name as shown.
592fn after_statuses(ui: &egui::Ui, target: &Target, look: &TargetsLook, name: &str) -> f32 {
593    let gap = ui.spacing().item_spacing.x;
594    look.order
595        .iter()
596        .skip_while(|part| **part != Part::Statuses)
597        .skip(1)
598        .map(|part| match part {
599            Part::Name => {
600                let mut words = vec![name.to_owned()];
601                if let Some(tag) = &target.tag {
602                    words.push(format!("({tag})"));
603                }
604                if look.book && target.codex.is_some() {
605                    words.push("\u{1f4d6}".to_owned());
606                }
607                words.iter().map(|word| words_width(ui, word) + gap).sum()
608            }
609            Part::Health => match (target.health, look.percent) {
610                (Some(percent), true) => words_width(ui, &format!("{percent}%")) + gap,
611                _ => 0.0,
612            },
613            Part::Statuses => 0.0,
614        })
615        .sum()
616}

618/// The whole line as words, to see whether the full name has room on it.
619fn line_words(target: &Target, look: &TargetsLook) -> String {
620    let mut words = vec![target.name.clone()];
621    if let Some(tag) = &target.tag {
622        words.push(format!("({tag})"));
623    }
624    if look.book && target.codex.is_some() {
625        words.push("\u{1f4d6}".to_owned());
626    }
627    if !target.statuses.is_empty() {
628        words.push(
629            target
630                .statuses
631                .iter()
632                .map(|(status, _)| status.as_str())
633                .collect::<Vec<_>>()
634                .join(", "),
635        );
636    }
637    if let (Some(percent), true) = (target.health, look.percent) {
638        words.push(format!("{percent}%"));
639    }
640    words.join(" ")
641}

649    look: &RosterLook,
650    own: bool,
651) -> Option<Clicked> {
652    let chosen = &look.targets;
653    let color = if target.current {
654        look.current_color(ui.ctx())
655    } else if let (true, Some([r, g, b])) = (target.boss, chosen.boss) {
656        Color32::from_rgb(r, g, b)
657    } else {
658        theme::color(ui.ctx(), T::Creature)
659    };
660    let name = target.named(chosen.name, || fits(ui, &line_words(target, chosen)));
661    let line = |ui: &mut egui::Ui| {
662        let mut clicked = None;
663        for part in chosen.order {
664            match part {
665                // The targeted one is known by its colour alone: a mark
666                // before it would push its name over.
667                Part::Name => {
668                    let id = target.id.to_string();
669                    let run = object_run(
670                        name,
671                        (&id, target.noun.as_deref().unwrap_or_default()),
672                        color,
673                    );
674                    clicked = clicked.or(say(ui, &[run], own));
675                    if let Some(tag) = &target.tag {
676                        ui.weak(format!("({tag})"));
677                    }
678                    // The book: a hint of its level and hit points, and
679                    // its entry.
680                    if let (true, true, Some((template, _, _)), Some(hint)) =
681                        (chosen.book, own, &target.codex, target.hint())
682                        && ui
683                            .small_button("\u{1f4d6}")
684                            .on_hover_text(format!("{hint}; its bestiary entry"))
685                            .clicked()
686                    {
687                        clicked = Some(Clicked::Command(format!("bestiary {template}")));
688                    }
689                }
690                Part::Statuses => {
691                    if !target.statuses.is_empty() {
692                        let statuses = egui::Label::new(statuses_job(ui, &target.statuses));
693                        if chosen.wrap {
694                            ui.add(statuses);
695                        } else {
696                            // On one row, cut with an ellipsis short of the
697                            // parts after them, which stay in sight.
698                            let room = (ui.available_width()
699                                - after_statuses(ui, target, chosen, name))
700                            .max(0.0);
701                            ui.allocate_ui_with_layout(
702                                egui::vec2(room, ui.available_height()),
703                                egui::Layout::left_to_right(egui::Align::Center),
704                                |ui| ui.add(statuses.truncate()),
705                            );
706                        }
707                    }
708                }
709                Part::Health => {
710                    if let (Some(percent), true) = (target.health, chosen.percent) {
711                        ui.weak(format!("{percent}%"));
712                    }
713                }
714            }
715        }
716        clicked
717    };
718    let clicked = if chosen.wrap {
719        ui.horizontal_wrapped(line).inner
720    } else {
721        // One row, the widget's width and no more, whatever it holds: a long
722        // name or a long row of statuses is cut at the edge and never pushes
723        // the window wider. The statuses are in the order that matters, so
724        // what is cut is what matters least.
725        let height = ui.text_style_height(&egui::TextStyle::Body);
726        let (_, rect) = ui.allocate_space(egui::vec2(ui.available_width(), height));
727        let mut row = ui.new_child(
728            egui::UiBuilder::new()
729                .id_salt(("target", target.id))
730                .max_rect(rect)
731                .layout(egui::Layout::left_to_right(egui::Align::Center)),
732        );
733        row.set_clip_rect(rect.intersect(ui.clip_rect()));
734        line(&mut row)
735    };
736    if let (Some(percent), true) = (target.health, chosen.bar) {
737        #[expect(clippy::cast_precision_loss, reason = "a height in points, at most 40")]
738        let height = chosen.height as f32;
739        let bar = Bar::new(
740            &target.name,
741            Some(Amount {
742                percent,
743                current: None,
744                max: None,
745            }),
746        )
747        .fill(look.health_color(ui.ctx(), percent))
748        .size([ui.available_width(), height])
749        .says(Says {
750            label: false,
751            numbers: false,
752            percent: false,
753            words: false,
754        });
755        ui.add(bar);
756    }
757    clicked
758}

760/// Draw the Players list for `state`; links live when `own`.
761pub(super) fn draw_players(
762    ui: &mut egui::Ui,
763    state: Option<&GameState>,
764    look: &RosterLook,
765    own: bool,
766) -> Option<Clicked> {
767    let Some(state) = state.filter(|state| state.room.saw_players()) else {
768        ui.weak("Players unknown");
769        return None;
770    };
771    if state.room.players.is_empty() {
772        ui.weak("No players");
773    }
774    let color = theme::color(ui.ctx(), T::Player);
775    let mut clicked = None;
776    for player in &state.room.players {
777        let grouped = look.group && state.group.contains(&player.id);
778        // The dead drawn faint: the room says `who appears dead`.
779        let dead = look.dim_dead
780            && player
781                .status
782                .as_ref()
783                .is_some_and(|status| status.as_str().contains("dead"));
784        let color = if dead {
785            color.gamma_multiply(0.45)
786        } else {
787            color
788        };
789        let here = ui
790            .horizontal_wrapped(|ui| {
791                if grouped {
792                    ui.colored_label(theme::color(ui.ctx(), T::Accent), "[G]");
793                }
794                let clicked = say(ui, &player_runs(player, color), own);
795                if let Some(status) = &player.status {
796                    ui.weak(status.as_str());
797                }
798                clicked
799            })
800            .inner;
801        clicked = clicked.or(here);
802    }
803    clicked
804}

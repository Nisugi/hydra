1//! The Targets and Players widgets: who is in the room, one to a line,
2//! with what the game says of each.
3//!
4//! **Targets** stacks the hostile creatures: each a link to it, marked
5//! when it is the one targeted, its statuses after it, and its health as
6//! a thin bar with the percent when the game states it. A dead creature,
7//! an animated decoy and a bare appendage (an arm, a claw, a tentacle) are
8//! left out, as `CreatureInstance::valid_target` leaves them; a line at
9//! the foot can count the appendages instead. Its page chooses the bar,
10//! the percent, the bar's colour from each percent up, the colour of the
11//! one targeted, and how each status is said: the game's word, or
12//! whatever the player puts in its place, `STN` or a glyph.
13//!
14//! **Players** stacks the players: each a link, what they are doing after
15//! it (`hiding`, `sitting`, `appears dead`), and `[G]` before a member of
16//! the character's group when asked.
17

31/// What the player chose for one Targets or Players widget on its page.
32#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
33#[expect(
34    clippy::struct_excessive_bools,
35    reason = "each is a switch on the page, and reads as one"

37pub(crate) struct RosterLook {
38    /// Targets: a line at the foot counting the appendages left out.
39    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
40    pub(crate) arms: bool,
41    /// Players: `[G]` before each member of the character's group.
42    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
43    pub(crate) group: bool,
44    /// Targets: a thin health bar under each, when the game states it.
45    #[serde(default = "yes", skip_serializing_if = "Clone::clone")]
46    pub(crate) bar: bool,
47    /// Targets: the health percent after the statuses.
48    #[serde(default = "yes", skip_serializing_if = "Clone::clone")]
49    pub(crate) percent: bool,
50    /// Targets: the bar's colour from each percent up, lowest first;
51    /// below them all, the health colour. None: the theme's three bands.
52    #[serde(default, skip_serializing_if = "Vec::is_empty")]
53    pub(crate) thresholds: Vec<Threshold>,
54    /// Targets: the colour of the one targeted; none, the accent.
55    #[serde(default, skip_serializing_if = "Option::is_none")]
56    pub(crate) current: Option<[u8; 3]>,
57    /// Targets: the colour of a creature the bestiary calls a boss; none,
58    /// the creature colour.
59    #[serde(default, skip_serializing_if = "Option::is_none")]
60    pub(crate) boss: Option<[u8; 3]>,
61    /// Players: one the room says is dead drawn faint.
62    #[serde(default = "yes", skip_serializing_if = "Clone::clone")]
63    pub(crate) dim_dead: bool,
64    /// Targets: what stands for each status, by the game's word, where
65    /// the player chose something other than the word.
66    #[serde(default, skip_serializing_if = "BTreeMap::is_empty")]
67    pub(crate) statuses: BTreeMap<String, String>,
68}

102/// A status's word as the game says it: `natures_decay` as `natures decay`.
103fn word(status: StatusName) -> String {
104    status.as_str().replace('_', " ")
105}

107impl RosterLook {
108    /// What stands for `status`: what the player chose, or its word.
109    pub(crate) fn status_shown(&self, status: StatusName) -> String {
110        let word = word(status);
111        self.statuses.get(&word).cloned().unwrap_or(word)
112    }
113
114    /// The bar's colour at `percent` left: the player's thresholds, from
115    /// each percent up; with none, the theme's health colour above half,
116    /// the warning colour to a quarter, the wound colour below.
117    fn health_color(&self, ctx: &egui::Context, percent: u32) -> Color32 {
118        let reached = self
119            .thresholds
120            .iter()
121            .rev()
122            .find(|threshold| u32::from(threshold.from) <= percent);
123        if let Some(threshold) = reached {
124            let [r, g, b] = threshold.color;
125            return Color32::from_rgb(r, g, b);
126        }
127        if !self.thresholds.is_empty() || percent > 50 {
128            theme::color(ctx, T::Health)
129        } else if percent > 25 {
130            theme::color(ctx, T::Warning)
131        } else {
132            theme::color(ctx, T::Wrong)
133        }
134    }
135
136    /// The colour of the one targeted.
137    fn current_color(&self, ctx: &egui::Context) -> Color32 {
138        self.current.map_or_else(
139            || theme::color(ctx, T::Accent),
140            |[r, g, b]| Color32::from_rgb(r, g, b),
141        )
142    }
143}

271/// On or off, as the page writes a switch; none, `default`.
272fn switch(to: Option<&str>, default: bool) -> Result<bool, String> {
273    match to.map(str::trim) {
274        None => Ok(default),
275        Some("on") => Ok(true),
276        Some("off") => Ok(false),
277        Some(other) => Err(format!("`{other}` is not on or off.")),
278    }
279}

281/// The statuses the table `written` says: each row the game's word and
282/// what stands for it; a row whose second cell is empty or the word
283/// itself leaves the word as it is.
284fn statuses(written: &str) -> Result<BTreeMap<String, String>, String> {
285    let rows: Vec<Vec<String>> =
286        serde_json::from_str(written).map_err(|_| "not rows of cells".to_owned())?;
287    let mut chosen = BTreeMap::new();
288    for mut cells in rows {
289        cells.resize(2, String::new());
290        let (status, shown) = (cells[0].trim().to_owned(), cells[1].trim().to_owned());
291        if status.is_empty() {
292            continue;
293        }
294        let known = StatusName::ALL
295            .into_iter()
296            .any(|known| word(known) == status.to_ascii_lowercase());
297        if !known {
298            return Err(format!("`{status}` is not a status the game says."));
299        }
300        let status = status.to_ascii_lowercase();
301        if !shown.is_empty() && shown != status {
302            chosen.insert(status, shown);
303        }
304    }
305    Ok(chosen)
306}

308/// Set the widget's `key` to `to`, or back to its own.
309pub(crate) fn set(look: &mut RosterLook, key: &str, to: Option<&str>) -> Result<(), String> {
310    let default = RosterLook::default();
311    match key {
312        ARMS => look.arms = switch(to, default.arms)?,
313        GROUP => look.group = switch(to, default.group)?,
314        BAR => look.bar = switch(to, default.bar)?,
315        PERCENT => look.percent = switch(to, default.percent)?,
316        DIM_DEAD => look.dim_dead = switch(to, default.dim_dead)?,
317        BOSS => {
318            look.boss = match to.map(str::trim).filter(|to| !to.is_empty()) {
319                None => None,
320                Some(to) => {
321                    Some(crate::menu::rgb(to).ok_or_else(|| format!("`{to}` is not a colour."))?)
322                }
323            };
324        }
325        THRESHOLDS => {
326            look.thresholds = match to {
327                None => Vec::new(),
328                Some(written) => crate::play::thresholds(written)?,
329            };
330        }
331        CURRENT => {
332            look.current = match to.map(str::trim).filter(|to| !to.is_empty()) {
333                None => None,
334                Some(to) => {
335                    Some(crate::menu::rgb(to).ok_or_else(|| format!("`{to}` is not a colour."))?)
336                }
337            };
338        }
339        STATUSES => {
340            look.statuses = match to {
341                None => BTreeMap::new(),
342                Some(written) => statuses(written)?,
343            };
344        }
345        other => return Err(format!("The list has no setting {other}.")),
346    }
347    Ok(())
348}

371/// The targets in the room, in the game's order, and how many appendages
372/// were left out; `None` until the game has said what is here.
373pub(crate) fn targets(
374    snapshot: &Snapshot,
375    tags: Option<hydra_model::targetid::Style>,
376    look: &RosterLook,
377) -> Option<(Vec<Target>, usize)> {
378    let state = &snapshot.state;
379    state.room.component("room objs")?;
380    let now = state.game_time_now();
381    let current = state.targeting.current();
382    let mut arms = 0;
383    let mut listed = Vec::new();
384    for creature in state.creatures().in_room() {
385        if creature.hostile_when_first_seen() != Some(true) {
386            continue;
387        }
388        if creature.appendage() {
389            arms += 1;
390            continue;
391        }
392        if !creature.valid_target() {
393            continue;
394        }
395        let tag = tags.and_then(|style| {
396            let (mark, health) = hydra_model::targetid::of(creature)?;
397            Some(hydra_model::targetid::tag(mark, style, Some(health)))
398        });
399        let health = creature.hp_is_stated().then(|| {
400            #[allow(
401                clippy::cast_possible_truncation,
402                clippy::cast_sign_loss,
403                reason = "rounded and clamped to 0-100 in the same expression"
404            )]
405            let percent = creature.hp_percent().clamp(0.0, 100.0).round() as u32;
406            percent
407        });
408        // Named as the room's line names it, `a kobold`, article and all,
409        // where it does; by its link's words alone otherwise.
410        let said = state
411            .room
412            .creatures
413            .iter()
414            .find(|item| item.id.parse::<i64>().ok() == Some(creature.id));
415        listed.push(Target {
416            name: said.map_or_else(|| creature.name.clone(), |item| item.text.clone()),
417            id: creature.id,
418            noun: creature.noun.clone(),
419            tag,
420            current: current == Some(creature.id),
421            boss: creature.is_boss(),
422            codex: creature.template().map(|template| {
423                (
424                    template.name.clone(),
425                    template.level,
426                    template.max_hp.map(|hp| match hp {
427                        hydra_model::state::creature::Stat::Exact(hp) => hp.to_string(),
428                        hydra_model::state::creature::Stat::Range(low, high) => {
429                            format!("{low}-{high}")
430                        }
431                    }),
432                )
433            }),
434            statuses: creature
435                .statuses(now)
436                .into_iter()
437                .map(|status| look.status_shown(status))
438                .collect(),
439            health,
440        });
441    }
442    Some((listed, arms))
443}

445/// Draw the Targets list for `snapshot`; links live when `own`.
446pub(super) fn draw_targets(
447    ui: &mut egui::Ui,
448    snapshot: Option<&Snapshot>,
449    tags: Option<hydra_model::targetid::Style>,
450    look: &RosterLook,
451    own: bool,
452) -> Option<Clicked> {
453    let Some((listed, arms)) = snapshot.and_then(|snapshot| targets(snapshot, tags, look)) else {
454        ui.weak("Targets unknown");
455        return None;
456    };
457    if listed.is_empty() {
458        ui.weak("No targets");
459    }
460    let mut clicked = None;
461    for target in &listed {
462        clicked = clicked.or(draw_target(ui, target, look, own));
463    }
464    if look.arms && arms > 0 {
465        ui.weak(format!(
466            "{arms} {}",
467            if arms == 1 { "appendage" } else { "appendages" }
468        ));
469    }
470    clicked
471}

473/// One target: its name, a link to it, in the targeted colour when
474/// targeted; its tag; its statuses; its health as a percent and a thin
475/// bar when stated, as the look asks.
476fn draw_target(
477    ui: &mut egui::Ui,
478    target: &Target,
479    look: &RosterLook,
480    own: bool,
481) -> Option<Clicked> {
482    let color = if target.current {
483        look.current_color(ui.ctx())
484    } else if let (true, Some([r, g, b])) = (target.boss, look.boss) {
485        Color32::from_rgb(r, g, b)
486    } else {
487        theme::color(ui.ctx(), T::Creature)
488    };
489    let clicked = ui
490        .horizontal_wrapped(|ui| {
491            if target.current {
492                ui.colored_label(color, "\u{25b8}");
493            }
494            let id = target.id.to_string();
495            let name = object_run(
496                &target.name,
497                (&id, target.noun.as_deref().unwrap_or_default()),
498                color,
499            );
500            let mut clicked = say(ui, &[name], own);
501            if let Some(tag) = &target.tag {
502                ui.weak(format!("({tag})"));
503            }
504            // The book: a hint of its level and hit points, and its entry.
505            if let (true, Some((template, level, hp))) = (own, &target.codex) {
506                let hint = match (level, hp) {
507                    (Some(level), Some(hp)) => format!("Level {level}, HP {hp}"),
508                    (Some(level), None) => format!("Level {level}"),
509                    (None, Some(hp)) => format!("HP {hp}"),
510                    (None, None) => template.clone(),
511                };
512                if ui
513                    .small_button("\u{1f4d6}")
514                    .on_hover_text(format!("{hint}; its bestiary entry"))
515                    .clicked()
516                {
517                    clicked = Some(Clicked::Command(format!("bestiary {template}")));
518                }
519            }
520            if !target.statuses.is_empty() {
521                ui.weak(target.statuses.join(", "));
522            }
523            if let (Some(percent), true) = (target.health, look.percent) {
524                ui.weak(format!("{percent}%"));
525            }
526            clicked
527        })
528        .inner;
529    if let (Some(percent), true) = (target.health, look.bar) {
530        let bar = Bar::new(
531            &target.name,
532            Some(Amount {
533                percent,
534                current: None,
535                max: None,
536            }),
537        )
538        .fill(look.health_color(ui.ctx(), percent))
539        .size([ui.available_width(), 6.0])
540        .says(Says {
541            label: false,
542            numbers: false,
543            percent: false,
544            words: false,
545        });
546        ui.add(bar);
547    }
548    clicked
549}

551/// Draw the Players list for `state`; links live when `own`.
552pub(super) fn draw_players(
553    ui: &mut egui::Ui,
554    state: Option<&GameState>,
555    look: &RosterLook,
556    own: bool,
557) -> Option<Clicked> {
558    let Some(state) = state.filter(|state| state.room.saw_players()) else {
559        ui.weak("Players unknown");
560        return None;
561    };
562    if state.room.players.is_empty() {
563        ui.weak("No players");
564    }
565    let color = theme::color(ui.ctx(), T::Player);
566    let mut clicked = None;
567    for player in &state.room.players {
568        let grouped = look.group && state.group.contains(&player.id);
569        // The dead drawn faint: the room says `who appears dead`.
570        let dead = look.dim_dead
571            && player
572                .status
573                .as_ref()
574                .is_some_and(|status| status.as_str().contains("dead"));
575        let color = if dead {
576            color.gamma_multiply(0.45)
577        } else {
578            color
579        };
580        let here = ui
581            .horizontal_wrapped(|ui| {
582                if grouped {
583                    ui.colored_label(theme::color(ui.ctx(), T::Accent), "[G]");
584                }
585                let clicked = say(ui, &player_runs(player, color), own);
586                if let Some(status) = &player.status {
587                    ui.weak(status.as_str());
588                }
589                clicked
590            })
591            .inner;
592        clicked = clicked.or(here);
593    }
594    clicked
595}

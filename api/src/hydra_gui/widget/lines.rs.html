2//! its own page says: a time on each line, at its start or its end, with
3//! seconds or not, on a 12- or 24-hour clock (`XX:XX`, `XX:XX:XX`,

23/// Where a line's time goes.
24#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash, Serialize, Deserialize)]
25#[serde(rename_all = "snake_case")]
26pub(crate) enum Stamps {
27    /// No time.
28    #[default]
29    None,
30    /// Before the line: `[7:08 PM] You swing ...`.
31    Start,
32    /// After it: `You swing ... [7:08 PM]`.
33    End,
34}

36/// How a widget of lines draws them: the story's, or a stream's.
37#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
38#[serde(default)]
39#[expect(
40    clippy::struct_excessive_bools,
41    reason = "a page of switches: each on or off on its own, as the player picks"
42)]
43pub(crate) struct Lines {
44    /// Where each line's time goes, if anywhere.
45    pub(crate) stamps: Stamps,
46    /// A time says its seconds.
47    pub(crate) seconds: bool,
48    /// The clock a time is said on.
49    pub(crate) hours: Hours,
50    /// A long line wraps; unwrapped, the widget scrolls sideways.
51    pub(crate) wrap: bool,
52    /// The story shows the game's prompts.
53    pub(crate) prompts: bool,
54    /// The story shows what the player typed.
55    pub(crate) echo: bool,
56    /// The bounty stream's widget says the task in brief
57    /// (`hydra_ui::bounty::brief`), not as the game words it.
58    pub(crate) summary: bool,
59}

61impl Default for Lines {
62    /// As the story has always drawn: no times, wrapped, prompts and what
63    /// was typed shown.
64    fn default() -> Self {
65        Self {
66            stamps: Stamps::None,
67            seconds: false,
68            hours: Hours::Twelve,
69            wrap: true,
70            prompts: true,
71            echo: true,
72            summary: false,
73        }
74    }
75}

77impl Lines {
78    /// `runs` with the time `at` where this puts it, in the theme's colour
79    /// for a line's time.
80    fn with_stamp(self, at: Stamp, runs: &[StyledRun]) -> Cow<'_, [StyledRun]> {
81        let run = |text: String| StyledRun {
82            text,
83            preset: Some(hydra_ui::TIMESTAMP.to_owned()),
84            ..StyledRun::default()
85        };
86        let said = || at.said(self.seconds, self.hours);
87        match self.stamps {
88            Stamps::None => Cow::Borrowed(runs),
89            Stamps::Start => {
90                let mut all = vec![run(format!("[{}] ", said()))];
91                all.extend_from_slice(runs);
92                Cow::Owned(all)
93            }
94            Stamps::End => {
95                let mut all = runs.to_vec();
96                all.push(run(format!(" [{}]", said())));
97                Cow::Owned(all)
98            }
99        }
100    }
101
102    /// `text` with the time `at` where this puts it.
103    fn stamped(self, at: Stamp, text: &str) -> String {
104        let said = at.said(self.seconds, self.hours);
105        match self.stamps {
106            Stamps::None => text.to_owned(),
107            Stamps::Start => format!("[{said}] {text}"),
108            Stamps::End => format!("{text} [{said}]"),
109        }
110    }
111
112    /// The game's prompt as this draws it, in the theme's prompt colours,
113    /// with what was `typed` at it after it, faint: none for a prompt alone.
114    fn prompt_row(self, ui: &mut egui::Ui, at: Stamp, prompt: &str, typed: &str) {
115        let faint = ui.visuals().weak_text_color();
116        let mut runs = vec![StyledRun::prompt(prompt)];
117        if !typed.is_empty() {
118            runs.push(StyledRun {
119                color: Some(format!(
120                    "#{:02x}{:02x}{:02x}",
121                    faint.r(),
122                    faint.g(),
123                    faint.b()
124                )),
125                ..StyledRun::plain(typed)
126            });
127        }
128        let _ = self.label(ui, at, (None, &runs, Sortable::No));
129    }
130
131    /// A line of the game's as this draws it: its runs, with the time `at`
132    /// where this puts it; kept when the pass's right-click, `clicked`,
133    /// was in it, unless it is `sortable` by no rule; the link clicked in
134    /// it, if one was.
135    fn label(
136        self,
137        ui: &mut egui::Ui,
138        at: Stamp,
139        (clicked, runs, sortable): (Option<egui::Pos2>, &[StyledRun], Sortable<'_>),
140    ) -> Option<Clicked> {
141        let palette = theme::palette(ui.ctx());
142        let stamped = self.with_stamp(at, runs);
143        // A prompt a character at a time, in the theme's prompt colours.
144        let all = hydra_ui::prompt_coloured(&stamped, &palette);
145        let mut job = text::job(&all, ui.style(), &palette);
146        // A blank line the game sent is its formatting -- between the
147        // paragraphs of `exp`, before WEALTH's last line -- and a line tall
148        // in the story's font. Holding no text, its job would hold no font,
149        // and be drawn no height at all.
150        if job.text.is_empty() {
151            job = egui::text::LayoutJob::single_section(
152                String::new(),
153                egui::TextFormat {
154                    font_id: theme::story_font(ui.style()),
155                    ..egui::TextFormat::default()
156                },
157            );
158        }
159        // Every line owns its row, words or none: a selection dragged past
160        // a line's end takes it whole and goes on to the next.
161        let row = ui.scope(|ui| text::linked_row(ui, job, &all).map(Clicked::from));
162        point(ui, row.response.rect, clicked, sortable, runs);
163        row.inner
164    }
165}

206/// Where the pointer right-clicked in this pass, if it did: asked once a
207/// widget, not once a line, since the input is locked to ask.
208fn right_clicked(ctx: &egui::Context) -> Option<egui::Pos2> {
209    ctx.input(|input| {
210        input
211            .pointer
212            .secondary_clicked()
213            .then(|| input.pointer.interact_pos())
214            .flatten()
215    })
216}

218/// `runs`, a line drawn over `rect`, kept as the one right-clicked when
219/// the right-click of this pass, `clicked`, was in it, on the layer the
220/// pointer is over, unless no rule can sort it.
221fn point(
222    ui: &egui::Ui,
223    rect: egui::Rect,
224    clicked: Option<egui::Pos2>,
225    sortable: Sortable<'_>,
226    runs: &[StyledRun],
227) {
228    let main = hydra_session::player_log::MAIN;
229    let (shown, from) = match sortable {
230        Sortable::No => return,
231        Sortable::Main => (main, Some(main)),
232        Sortable::Shown(stream) => (stream, None),
233    };
234    let ctx = ui.ctx();
235    let Some(at) = clicked else {
236        return;
237    };
238    if !rect.contains(at)
239        || !ui.clip_rect().contains(at)
240        || ctx.layer_id_at(at) != Some(ui.layer_id())
241    {
242        return;
243    }
244    let pointed = Pointed {
245        pass: ctx.cumulative_pass_nr(),
246        shown: shown.to_owned(),
247        from: from.map(str::to_owned),
248        runs: runs.to_vec(),
249    };
250    let id = pointed_id(ctx);
251    ctx.data_mut(|data| data.insert_temp(id, pointed));
252}

254/// The line of the game's right-clicked in this pass of the window, if one
255/// was: taken.
256pub(crate) fn pointed(ctx: &egui::Context) -> Option<Pointed> {
257    let id = pointed_id(ctx);
258    let pointed = ctx.data_mut(|data| data.remove_temp::<Pointed>(id))?;
259    (pointed.pass == ctx.cumulative_pass_nr()).then_some(pointed)
260}

262/// The story, a stream's lines left out while a widget of it is `open`
263/// and Hydra's messages while a Hydra widget is (`hydra`), split when the player scrolls back (`split.rs`); the link clicked in it,
264/// if one was.
265///
266/// A prompt is drawn only when a line was drawn since the last one drawn, or
267/// when it changed (`R>` to `>`). It is kept here, where what is left out is
268/// known, because a thought whose Thoughts widget is open is not drawn, and
269/// the prompt after it would be: a lone `>` in the story for every line
270/// another widget showed.
271pub(super) fn story(
272    ui: &mut egui::Ui,
273    story: &Story,
274    (open, hydra): (&[String], bool),
275    (id, options): (Id, Lines),
276) -> Option<Clicked> {
277    let mut clicked = None;
278    let first = (story.dropped, options.wrap);
279    let scroll = super::split::asked(ui, id);
280    let mut finder = Seek::of(
281        ui,
282        id,
283        story
284            .lines
285            .iter()
286            .filter_map(|(_, shown)| said(shown, (open, hydra), options)),
287    );
288    let pictures = crate::story_picture::rules(ui.ctx()).shown;
289    let right_click = right_clicked(ui.ctx());
290    super::split::scrolled(ui, id.with("story"), first, scroll, |ui, tops| {
291        tops.look(&options);
292        let mut prompts = hydra_ui::story::Prompts::default();
293        if let Some(finder) = &mut finder {
294            finder.start();
295        }
296        for (index, (at, shown)) in story.lines.iter().enumerate() {
297            tops.mark(ui);
298            // Whether the line draws anything, decided -- and the prompts'
299            // count kept -- before it is drawn, since a line out of sight is
300            // not; and a line that draws nothing takes no room.
301            let draws = match shown {
302                Shown::Game(runs) => {
303                    prompts.line(runs);
304                    true
305                }
306                Shown::From(stream, runs) => {
307                    let draws = !open.contains(stream);
308                    if draws {
309                        prompts.line(runs);
310                    }
311                    draws
312                }
313                Shown::Typed { .. } => options.echo,
314                Shown::Prompt(prompt) => prompts.draws(prompt) && options.prompts,
315                Shown::Gap => true,
316                Shown::Hydra(_) => {
317                    if !hydra {
318                        prompts.said();
319                    }
320                    !hydra
321                }
322                Shown::Said(_) => {
323                    prompts.said();
324                    true
325                }
326                // Switched off on Hydra's General page, a picture takes no
327                // room.
328                Shown::Picture(_) => pictures,
329            };
330            if !draws {
331                continue;
332            }
333            let text = finder
334                .as_ref()
335                .and_then(|_| said(shown, (open, hydra), options));
336            let line = |ui: &mut egui::Ui| match shown {
337                Shown::Game(runs) => {
338                    clicked = clicked.take().or(options.label(
339                        ui,
340                        *at,
341                        (right_click, runs, Sortable::Main),
342                    ));
343                }
344                Shown::From(stream, runs) => {
345                    let sortable = Sortable::Shown(stream);
346                    clicked =
347                        clicked
348                            .take()
349                            .or(options.label(ui, *at, (right_click, runs, sortable)));
350                }
351                Shown::Typed { prompt, line } => options.prompt_row(ui, *at, prompt, line),
352                Shown::Prompt(prompt) => options.prompt_row(ui, *at, prompt, ""),
353                Shown::Gap => {
354                    let wrong = theme::color(ui.ctx(), T::Wrong);
355                    plain_row(ui, "Some lines were missed here.", wrong, false);
356                }
357                Shown::Hydra(text) => {
358                    weak_row(ui, &options.stamped(*at, &format!("[Hydra] {text}")));
359                }
360                Shown::Said(notice) => {
361                    clicked = clicked.take().or(notice_lines(ui, notice));
362                }
363                Shown::Picture(picture) => crate::story_picture::draw(ui, picture),
364            };
365            // Each line by its own number, never its place: a selection is
366            // kept by the ids of the lines it runs between, and a place's
367            // line changes as old ones go, and text coming in must not change
368            // which lines are selected.
369            let number = story.dropped + u64::try_from(index).unwrap_or(u64::MAX);
370            // Explicit, by the line's number: a child by salt still takes its
371            // parent's count of children before it, which is the line's place
372            // again. The same in the one pane and the top, so a selection goes
373            // on as the widget splits; another in the bottom (`Tops::line_id`).
374            let own = tops.line_id(number);
375            let draw = |ui: &mut egui::Ui| {
376                ui.scope_builder(egui::UiBuilder::new().id(own), line);
377            };
378            // While Find is open every line is drawn: it scrolls to what
379            // it finds, which must be there to scroll to.
380            match (&mut finder, text) {
381                (Some(finder), Some(text)) => finder.line(ui, (&text, tops.player()), draw),
382                _ => tops.line(ui, number, finder.is_none(), draw),
383            }
384        }
385    });
386    // Its blank space is the floor: an object carried and let go there, on
387    // no other object, is dropped.
388    clicked
389        .or_else(|| crate::carry::target(ui, id.with("story"), "drop", None).map(Clicked::Quietly))
390}

392/// The text of a line of the story that Find looks through: one drawn --
393/// the game's, an echo, Hydra's own while the story shows them -- never a
394/// prompt.
395fn said<'a>(
396    shown: &'a Shown,
397    (open, hydra): (&[String], bool),
398    options: Lines,
399) -> Option<Cow<'a, str>> {
400    match shown {
401        Shown::Game(runs) => Some(plain(runs)),
402        Shown::From(stream, runs) if !open.contains(stream) => Some(plain(runs)),
403        Shown::Hydra(text) if !hydra => Some(Cow::Owned(format!("[Hydra] {text}"))),
404        Shown::Said(notice) => Some(Cow::Owned(notice.lines().join("\n"))),
405        Shown::Typed { prompt, line } if options.echo => {
406            Some(Cow::Owned(format!("{prompt}{line}")))
407        }
408        _ => None,
409    }
410}

412/// Hydra's answer, coloured by what it is about, a table in fixed width;
413/// a linked table drawn as a line with links is, so a click on one is the
414/// link's: one of Hydra's own commands, the game's, or a web page.
415pub(super) fn notice_lines(ui: &mut egui::Ui, notice: &hydra_session::Notice) -> Option<Clicked> {
416    use hydra_session::NoticeKind;
417    use hydra_session::notice::{Body, NoticeLink};
418    use hydra_ui::RunLink;
419    let color = match notice.kind {
420        NoticeKind::Error => theme::color(ui.ctx(), T::Wrong),
421        NoticeKind::Warn => theme::color(ui.ctx(), T::Warning),
422        NoticeKind::Info | NoticeKind::Debug => theme::color(ui.ctx(), T::Hydra),
423    };
424    let mut clicked = None;
425    match &notice.body {
426        Body::Help { parts, .. } => clicked = help_drawn(ui, parts),
427        Body::Lines(lines) => {
428            for line in lines {
429                plain_row(ui, line, color, false);
430            }
431        }
432        Body::Mono(lines) => {
433            for line in lines {
434                plain_row(ui, line, color, true);
435            }
436        }
437        Body::Linked { spans, .. } => {
438            let hex = format!("#{:02x}{:02x}{:02x}", color.r(), color.g(), color.b());
439            for line in spans {
440                // Hydra's own command goes with its symbol before it; the
441                // game's as the game's own link; a page as a page.
442                let runs: Vec<StyledRun> = line
443                    .iter()
444                    .map(|span| StyledRun {
445                        text: span.text.clone(),
446                        monospace: true,
447                        color: Some(hex.clone()),
448                        link: span.link.as_ref().map(|link| match link {
449                            NoticeLink::Hydra(command) | NoticeLink::Game(command) => {
450                                RunLink::Command {
451                                    command: command.clone(),
452                                }
453                            }
454                            NoticeLink::Url(href) => RunLink::Url { href: href.clone() },
455                        }),
456                        ..StyledRun::default()
457                    })
458                    .collect();
459                let job = text::job(&runs, ui.style(), &theme::palette(ui.ctx()));
460                let acted = ui.scope(|ui| text::linked_row(ui, job, &runs)).inner;
461                let Some(text::Acted::Clicked(RunLink::Command { command }, _)) = &acted else {
462                    clicked = clicked.or(acted.map(Clicked::from));
463                    continue;
464                };
465                let hydras = line.iter().any(
466                    |span| matches!(&span.link, Some(NoticeLink::Hydra(own)) if own == command),
467                );
468                clicked = Some(if hydras {
469                    Clicked::Command(command.clone())
470                } else {
471                    Clicked::FromGame(command.clone())
472                });
473            }
474        }
475    }
476    clicked
477}

479/// Help, drawn in two columns: each command in the accent colour and the
480/// fixed-width font, a link where a click only shows something; what it
481/// does in the text colour, wrapping within its own column. A heading
482/// stands over its commands, a note runs across both. The command clicked,
483/// if one was.
484fn help_drawn(ui: &mut egui::Ui, parts: &[hydra_session::notice::HelpPart]) -> Option<Clicked> {
485    use hydra_session::notice::HelpPart;
486    let ctx = ui.ctx().clone();
487    let (accent, text) = (theme::color(&ctx, T::Accent), theme::color(&ctx, T::Text));
488    let mono = egui::TextStyle::Monospace.resolve(ui.style());
489    // As wide as the widest command, and no more than half the width.
490    let widest = parts
491        .iter()
492        .filter_map(|part| match part {
493            HelpPart::Entry { command, .. } => Some(
494                ui.painter()
495                    .layout_no_wrap(command.clone(), mono.clone(), accent)
496                    .size()
497                    .x,
498            ),
499            _ => None,
500        })
501        .fold(0.0_f32, f32::max);
502    let column = (widest + 16.0).min(ui.available_width() * 0.5);
503    let mut clicked = None;
504    for (at, part) in parts.iter().enumerate() {
505        match part {
506            HelpPart::Heading(heading) => {
507                if at > 0 {
508                    ui.add_space(6.0);
509                }
510                // Larger and strong, in the text colour: it reads as a heading
511                // whatever the theme's Hydra colour.
512                let size = egui::TextStyle::Body.resolve(ui.style()).size + 2.0;
513                ui.label(egui::RichText::new(heading).strong().size(size).color(text));
514            }
515            HelpPart::Note(note) if note.is_empty() => {
516                ui.add_space(4.0);
517            }
518            HelpPart::Note(note) => {
519                ui.add(egui::Label::new(egui::RichText::new(note).color(text)).wrap());
520            }
521            HelpPart::Entry {
522                command,
523                runs,
524                says,
525            } => {
526                ui.horizontal_top(|ui| {
527                    let mut words = egui::RichText::new(command).monospace().color(accent);
528                    if runs.is_some() {
529                        words = words.underline();
530                    }
531                    let sense = if runs.is_some() {
532                        egui::Sense::click()
533                    } else {
534                        egui::Sense::hover()
535                    };
536                    // Its own column, set in a little, the command at its left.
537                    let height = ui.text_style_height(&egui::TextStyle::Monospace);
538                    let shown = ui
539                        .allocate_ui_with_layout(
540                            egui::vec2(column, height),
541                            egui::Layout::left_to_right(egui::Align::TOP),
542                            |ui| {
543                                // A command longer than its column wraps in it.
544                                ui.set_min_width(column);
545                                ui.set_max_width(column);
546                                ui.add_space(8.0);
547                                ui.add(egui::Label::new(words).sense(sense).wrap())
548                            },
549                        )
550                        .inner;
551                    if let Some(runs) = runs {
552                        let shown = shown.on_hover_cursor(egui::CursorIcon::PointingHand);
553                        if shown.clicked() {
554                            clicked = Some(Clicked::Command(runs.clone()));
555                        }
556                    }
557                    ui.vertical(|ui| {
558                        ui.add(egui::Label::new(egui::RichText::new(says).color(text)).wrap());
559                    });
560                });
561            }
562        }
563    }
564    clicked
565}

567/// A line of Hydra's own, or the game's prompt and what was typed at it,
568/// drawn faint, owning its row like a line of the game's.
569fn weak_row(ui: &mut egui::Ui, text: &str) {

571    plain_row(ui, text, weak, false);

574/// One line of plain words in `color`, owning its row like a line of the
575/// game's: a selection dragged past its end takes it and goes on.
576fn plain_row(ui: &mut egui::Ui, text: &str, color: egui::Color32, monospace: bool) {
577    let run = StyledRun {
578        text: text.to_owned(),
579        monospace,
580        color: Some(format!(
581            "#{:02x}{:02x}{:02x}",
582            color.r(),
583            color.g(),
584            color.b()
585        )),
586        ..StyledRun::default()
587    };
588    let runs = [run];
589    let job = text::job(&runs, ui.style(), &theme::palette(ui.ctx()));
590    ui.scope(|ui| text::linked_row(ui, job, &runs));
591}

593/// A line's runs as the text they say.
594fn plain(runs: &[StyledRun]) -> Cow<'_, str> {
595    match runs {
596        [one] => Cow::Borrowed(one.text.as_str()),
597        _ => Cow::Owned(runs.iter().map(|run| run.text.as_str()).collect()),
598    }
599}

601/// One of the game's streams, split when the player scrolls back; the link
602/// clicked in it, if one was.
603pub(super) fn stream(
604    ui: &mut egui::Ui,
605    seen: &Seen<'_>,
606    stream: &str,
607    (id, options): (Id, Lines),
608) -> Option<Clicked> {
609    let kept = seen.story.streams.get(stream);
610    let right_click = right_clicked(ui.ctx());
611    // The bounty in brief, when its page asks and what the stream says is
612    // a task: read from its own words, so it is never another's.
613    if options.summary
614        && stream == hydra_model::state::bounty::STREAM
615        && let Some(task) = kept.and_then(|kept| {
616            let said: Vec<_> = kept.lines.iter().map(|(_, runs)| plain(runs)).collect();
617            hydra_model::state::bounty::classify(&said.join(" "))
618        })
619    {
620        for line in hydra_ui::bounty::brief(&task) {
621            ui.label(line);
622        }
623        return None;
624    }
625    let mut clicked = None;
626    // A listing stream -- the worn items, the spells, the bounty, the
627    // society, the reserve -- is cleared and written whole each time the
628    // game sends it: a list that scrolls, never a story that splits.
629    if super::kind::PANEL_STREAMS
630        .iter()
631        .any(|(panel, _)| *panel == stream)
632    {
633        let scroll = super::split::asked(ui, id);
634        egui::ScrollArea::vertical()
635            .min_scrolled_height(0.0)
636            .id_salt(id.with("panel"))
637            .auto_shrink(false)
638            .show(ui, |ui| {
639                super::split::keyed(ui, scroll, |ui| {
640                    let Some(kept) = kept else {
641                        ui.weak("Nothing yet");
642                        return;
643                    };
644                    for (at, runs) in &kept.lines {
645                        let line = (right_click, runs.as_slice(), Sortable::No);
646                        clicked = clicked.take().or(options.label(ui, *at, line));
647                    }
648                });
649            });
650        return clicked;
651    }
652    // Its first kept line's number: what it heard, less what it keeps.
653    let first = kept.map_or(0, |kept| {
654        kept.heard
655            .saturating_sub(u64::try_from(kept.lines.len()).unwrap_or(u64::MAX))
656    });
657    let scroll = super::split::asked(ui, id);
658    let lines = kept.into_iter().flat_map(|kept| kept.lines.iter());
659    let mut finder = Seek::of(ui, id, lines.map(|(_, runs)| plain(runs)));
660    super::split::scrolled(
661        ui,
662        id.with("stream"),
663        (first, options.wrap),
664        scroll,
665        |ui, tops| match kept {
666            Some(kept) => {
667                tops.look(&options);
668                if let Some(finder) = &mut finder {
669                    finder.start();
670                }
671                for (index, (at, runs)) in kept.lines.iter().enumerate() {
672                    tops.mark(ui);
673                    let draw = |ui: &mut egui::Ui| {
674                        let line = (right_click, runs.as_slice(), Sortable::Shown(stream));
675                        clicked = clicked.take().or(options.label(ui, *at, line));
676                    };
677                    if let Some(finder) = &mut finder {
678                        finder.line(ui, (&plain(runs), tops.player()), draw);
679                    } else {
680                        let number = first + u64::try_from(index).unwrap_or(u64::MAX);
681                        tops.line(ui, number, true, draw);
682                    }
683                }
684            }
685            None => {
686                ui.weak("Nothing yet.");
687            }
688        },
689    );
690    clicked
691}

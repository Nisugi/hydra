1//! The Uberbar widget: the injury doll with the day's numbers beside it,
2//! the vitals under those, and the bars across the panel under them, each
3//! row as its page chose (`hydra_board::uberbar`).
4//!
5//! The panel is laid out as the Lich script of the same name laid its
6//! dialog out: the doll top left; right of it the labels, a name and its
7//! value a line; under the labels, still right of the doll, the four
8//! vitals saying their numbers alone; the favor under the doll; and across
9//! the whole width the bars for the next ascension point, the next level,
10//! the mind, the stance, the burden, the profession's resource (this week
11//! and in all, side by side), a sorcerer's shadow essence and the way to
12//! the cap. No bar wears a label: each says what the script's said.
13

27/// The doll's column, points across.
28const DOLL_WIDTH: f32 = 110.0;
29/// The least height the doll's box has, points: it grows to what the
30/// doll needs to fill the column's width.
31const DOLL_HEIGHT: f32 = 150.0;
32/// The gap between two bars on one row, points.
33const GAP: f32 = 4.0;
34/// Ascension experience from one point to the next.
35const ASCENSION_POINT: u32 = 50_000;
36/// The profession's resource caps, this week and in all.
37const WEEKLY_CAP: u32 = 50_000;
38/// See [`WEEKLY_CAP`].
39const TOTAL_CAP: u32 = 200_000;

41/// What an Uberbar widget draws from beyond the state: its rows, the
42/// day's numbers when they are shown, and the doll's picture.
43#[derive(Clone, Debug, Default, PartialEq)]
44pub(crate) struct UberbarShown {
45    /// Which rows.
46    pub(crate) look: UberbarLook,
47    /// The day's numbers, read from the ledger; none until read, or when no
48    /// row wants them.
49    pub(crate) facts: Option<Facts>,
50    /// The doll's picture, as an Injuries widget in this layout has it.
51    pub(crate) doll: Option<DollLook>,
52}
53
54/// Draw the panel for `state` as `shown` says.
55pub(super) fn draw(ui: &mut egui::Ui, state: Option<&GameState>, shown: &UberbarShown) {
56    let look = &shown.look;
57    ui.horizontal_top(|ui| {
58        ui.vertical(|ui| {
59            ui.set_width(DOLL_WIDTH);
60            // The box offers the panel's height so the doll scales to the
61            // column's width, not to a fixed height; it takes only what
62            // the doll draws.
63            let tall = ui.available_height().max(DOLL_HEIGHT);
64            ui.allocate_ui(egui::vec2(DOLL_WIDTH, tall), |ui| {
65                let injuries = state.map(|state| &state.character.injuries);
66                super::doll::doll(ui, injuries, shown.doll.as_ref());
67            });
68            if look.favor
69                && let Some(favor) = state.and_then(|state| state.character.currency.voln_favor)
70            {
71                ui.label("Favor");
72                ui.strong(grouped(favor));
73            }
74        });
75        ui.vertical(|ui| {
76            ui.set_min_width(ui.available_width());
77            labels(ui, state, shown);
78            if look.vitals {
79                vitals(ui, state);
80            }
81        });
82    });
83    bars(ui, state, look);
84}

89/// One bar's row: a line across the space, so each bar fits a line of its
90/// own rather than the first taking all the room left.
91fn row(ui: &mut egui::Ui, add: impl FnOnce(&mut egui::Ui)) {
92    ui.allocate_ui_with_layout(
93        egui::vec2(ui.available_width(), LINE),
94        egui::Layout::top_down(egui::Align::Min),
95        add,
96    );
97}

99/// A bar that says `text` and nothing else, `percent` full in `color`.
100fn worded_bar(ui: &mut egui::Ui, text: &str, percent: Option<u32>, color: egui::Color32) {
101    let amount = percent.map(|percent| Amount {
102        percent,
103        current: None,
104        max: None,
105    });
106    let bar = Bar::new(text, amount).fill(color).says(Says {
107        label: true,
108        numbers: false,
109        percent: false,
110        words: false,
111    });
112    ui.add(bar.fitted(ui));
113}

115/// The labels beside the doll: a name and its value a line.
116fn labels(ui: &mut egui::Ui, state: Option<&GameState>, shown: &UberbarShown) {
117    let look = &shown.look;
118    let facts = shown.facts.unwrap_or_default();
119    let mut rows: Vec<(&str, String)> = Vec::new();
120    if look.today {
121        rows.push(("Today:", grouped(facts.today)));
122    }
123    if look.per_hour {
124        rows.push(("AVG/Hr:", grouped(facts.per_hour)));
125    }
126    if look.pulse {
127        rows.push(("Pulse:", grouped(facts.pulse)));
128    }
129    if look.room {
130        let room = state
131            .and_then(|state| state.room.id.clone())
132            .unwrap_or_else(|| "?".to_owned());
133        rows.push(("Room:", room));
134    }
135    if look.silver {
136        rows.push(("Silver:", grouped(facts.silver)));
137    }
138    if look.bounty {
139        rows.push(("Bounty:", grouped(facts.bounty)));
140    }
141    egui::Grid::new(ui.next_auto_id())
142        .num_columns(2)
143        .spacing([8.0, 2.0])
144        .show(ui, |ui| {
145            for (name, value) in rows {
146                ui.label(name);
147                ui.strong(value);
148                ui.end_row();
149            }
150        });
151}

153/// The four vitals under the labels, each saying its numbers alone.
154fn vitals(ui: &mut egui::Ui, state: Option<&GameState>) {
155    let vitals: [(VitalOf, T); 4] = [
156        (GameState::health, T::Health),
157        (GameState::mana, T::Mana),
158        (GameState::stamina, T::Stamina),
159        (GameState::spirit, T::Spirit),
160    ];
161    for (vital, token) in vitals {
162        row(ui, |ui| {
163            let amount = state.and_then(vital).map(|vital| Amount {
164                percent: vital.percent,
165                current: vital.current,
166                max: vital.max,
167            });
168            let bar = Bar::new("", amount)
169                .fill(theme::color(ui.ctx(), token))
170                .says(Says {
171                    label: false,
172                    numbers: true,
173                    percent: false,
174                    words: false,
175                });
176            ui.add(bar.fitted(ui));
177        });
178    }
179}

181/// The bars across the panel, each with the words the script gave it.
182fn bars(ui: &mut egui::Ui, state: Option<&GameState>, look: &UberbarLook) {
183    let experience = state.map(|state| &state.character.experience);
184    let character = state.map(|state| &state.character);
185    if look.ascension {
186        let next = experience.and_then(|e| e.until_ascension_point);
187        let points = experience.and_then(|e| e.ascension_points).unwrap_or(0);
188        let text = next.map_or_else(
189            || "Ascension ?".to_owned(),
190            |next| format!("{} until next ATP [ {} ]", grouped(next.into()), points),
191        );
192        let percent = next.map(|next| 100u32.saturating_sub(next * 100 / ASCENSION_POINT));
193        row(ui, |ui| {
194            worded_bar(ui, &text, percent, theme::color(ui.ctx(), T::Level));
195        });
196    }
197    if look.next_level {
198        let percent = experience.and_then(|e| e.next_level_percent);
199        let text = match (
200            experience.and_then(|e| e.level.as_deref()),
201            experience.and_then(|e| e.next_level.as_deref()),
202            percent,
203        ) {
204            (Some(level), Some(next), Some(percent)) => format!(
205                "{}  ({percent}%)   {} TNL",
206                level.replace("Level ", "Level: "),
207                next.trim_end_matches(" experience")
208                    .trim_end_matches(" until next level")
209            ),
210            _ => "Next level ?".to_owned(),
211        };
212        row(ui, |ui| {
213            worded_bar(ui, &text, percent, theme::color(ui.ctx(), T::Level));
214        });
215    }
216    if look.mind {
217        let words = experience
218            .and_then(|e| e.mind_state.as_deref())
219            .unwrap_or("Mind ?");
220        let field = look
221            .field_experience
222            .then(|| experience.and_then(|e| e.field_experience.zip(e.field_experience_max)))
223            .flatten();
224        let text = match field {
225            Some((held, most)) => format!("{words} ({held}/{most})"),
226            None => words.to_owned(),
227        };
228        let percent = experience.and_then(|e| e.mind_percent);
229        row(ui, |ui| {
230            worded_bar(ui, &text, percent, theme::color(ui.ctx(), T::Mind));
231        });
232    }
233    if look.stance {
234        let text = character
235            .and_then(|c| c.stance.as_deref())
236            .unwrap_or("Stance ?");
237        let percent = character.and_then(|c| c.stance_percent);
238        row(ui, |ui| {
239            worded_bar(ui, text, percent, theme::color(ui.ctx(), T::Stance));
240        });
241    }
242    if look.encumbrance {
243        let text = character
244            .and_then(|c| c.encumbrance.level.as_deref())
245            .unwrap_or("Encumbrance ?");
246        let percent = character.and_then(|c| c.encumbrance.percent);
247        row(ui, |ui| {
248            worded_bar(ui, text, percent, theme::color(ui.ctx(), T::Encumbrance));
249        });
250    }
251    if look.resources {
252        resources(ui, state);
253    }
254    if look.shadow_essence
255        && let Some(essence) = character.and_then(|c| c.standing.shadow_essence)
256    {
257        let text = format!("Essence {essence}/5");
258        row(ui, |ui| {
259            worded_bar(
260                ui,
261                &text,
262                Some(u32::from(essence) * 20),
263                theme::color(ui.ctx(), T::Mana),
264            );
265        });
266    }
267    if look.percent_capped {
268        let total = experience.and_then(|e| e.total_experience);
269        let hundredths = total.map(|total| total * 10_000 / CAP_EXPERIENCE);
270        let text = hundredths.map_or_else(
271            || "Capped ?".to_owned(),
272            |h| format!("{}.{:02} % Capped", h / 100, h % 100),
273        );
274        let percent = hundredths.map(|h| u32::try_from(h / 100).unwrap_or(100));
275        row(ui, |ui| {
276            worded_bar(ui, &text, percent, theme::color(ui.ctx(), T::Level));
277        });
278    }
279}

281/// The profession's resource, this week and in all, side by side.
282fn resources(ui: &mut egui::Ui, state: Option<&GameState>) {
283    let amounts = state.and_then(|state| state.character.standing.resources);
284    let half = (ui.available_width() - GAP) / 2.0;
285    let one = |ui: &mut egui::Ui, held: Option<u32>, cap: u32| {
286        let text = held.map_or_else(
287            || "?".to_owned(),
288            |held| format!("{}/{}", grouped(held.into()), grouped(cap.into())),
289        );
290        let percent = held.map(|held| (held.min(cap) * 100 / cap.max(1)).min(100));
291        ui.allocate_ui_with_layout(
292            egui::vec2(half, LINE),
293            egui::Layout::top_down(egui::Align::Min),
294            |ui| worded_bar(ui, &text, percent, theme::color(ui.ctx(), T::Mind)),
295        );
296    };
297    ui.horizontal(|ui| {
298        ui.spacing_mut().item_spacing.x = GAP;
299        one(ui, amounts.map(|a| a.weekly), WEEKLY_CAP);
300        one(ui, amounts.map(|a| a.total), TOTAL_CAP);
301    });
302}

304/// The widget's page: a switch a row.
305pub(crate) fn rows(look: Option<&UberbarLook>) -> Vec<Row> {
306    let default = UberbarLook::default();
307    let now = look.cloned().unwrap_or_default();
308    hydra_board::uberbar::ROWS
309        .iter()
310        .map(|(key, label, help)| {
311            let on = now.shows(key);
312            Row {
313                key: (*key).to_owned(),
314                label: (*label).to_owned(),
315                help: (*help).to_owned(),
316                kind: RowKind::Toggle,
317                value: Value::On(on),
318                here: on != default.shows(key),
319                from: None,
320                needs: None,
321            }
322        })
323        .collect()
324}

326/// The page's `key` set to `to`, `on` or `off`, or back to its own.
327pub(crate) fn set(look: &mut UberbarLook, key: &str, to: Option<&str>) -> Result<(), String> {
328    let on = match to.map(str::trim) {
329        Some("on") => Some(true),
330        Some("off") => Some(false),
331        Some(other) => return Err(format!("`{other}` is neither on nor off.")),
332        None => None,
333    };
334    look.set(key, on)
335}

337/// The day's numbers for the window's character: read from the ledger on
338/// a thread of their own no oftener than `FACTS_EVERY`, kept here between.
339#[derive(Debug, Default)]
340pub(crate) struct FactsKeeper {
341    facts: Option<Facts>,
342    /// When the ledger was last asked.
343    asked: Option<Instant>,
344    /// The read under way, until it answers.
345    coming: Option<Receiver<Option<Facts>>>,
346}
347
348impl FactsKeeper {
349    /// The numbers as last read; none until the first read answers.
350    pub(crate) fn facts(&self) -> Option<Facts> {
351        self.facts
352    }
353
354    /// Take a read that has answered, and start another when one is due:
355    /// `who`'s day, from the ledger under `data`. The window is asked to
356    /// repaint when the next is due, so the numbers move without a hand on
357    /// the window.
358    pub(crate) fn fresh(&mut self, ctx: &egui::Context, data: &Path, who: hydra_ledger::Who) {
359        if let Some(coming) = &self.coming {
360            match coming.try_recv() {
361                Ok(read) => {
362                    self.facts = read.or(self.facts);
363                    self.coming = None;
364                }
365                Err(TryRecvError::Empty) => {}
366                Err(TryRecvError::Disconnected) => self.coming = None,
367            }
368        }
369        let due = self
370            .asked
371            .is_none_or(|asked| asked.elapsed() >= FACTS_EVERY);
372        if due && self.coming.is_none() {
373            self.asked = Some(Instant::now());
374            let (tx, rx) = std::sync::mpsc::channel();
375            self.coming = Some(rx);
376            let path = hydra_ledger::Ledger::path_in(data);
377            let context = ctx.clone();
378            std::thread::Builder::new()
379                .name("hydra-uberbar".to_owned())
380                .spawn(move || {
381                    let read = hydra_ledger::Ledger::open(path)
382                        .and_then(|ledger| ledger.reader())
383                        .and_then(|reader| Facts::read(&reader, &who, hydra_net::eastern::now()))
384                        .ok();
385                    let _ = tx.send(read);
386                    context.request_repaint();
387                })
388                .ok();
389        }
390        ctx.request_repaint_after(FACTS_EVERY);
391    }

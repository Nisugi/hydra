415    /// The key a theme gives a kind of widget its own face by: its name,
416    /// lowercase, words joined by `_` -- `story`, `room`, `thoughts`,
417    /// `round_time`.
418    pub(crate) fn type_key(&self) -> String {
419        let mut key = String::new();
420        for c in self.name().chars() {
421            if c.is_alphanumeric() {
422                key.extend(c.to_lowercase());
423            } else if !key.is_empty() && !key.ends_with('_') {
424                key.push('_');
425            }
426        }
427        key.trim_end_matches('_').to_owned()
428    }

621    /// How it draws its bar unless the player picks otherwise, when it is
622    /// one bar: a vital says its label, numbers and percent; the pulse its
623    /// words. `None` for a widget that is not a bar.
624    pub(crate) fn bar_look(&self) -> Option<crate::bar::Look> {
625        use crate::bar::{Fills, Look, Place, Says, Threshold};
626        let vital = Says {
627            label: true,
628            numbers: true,
629            percent: true,
630            words: false,
631        };
632        let worded = Says {
633            label: true,
634            numbers: false,
635            percent: true,
636            words: true,
637        };
638        let says = match self {
639            Widget::Health
640            | Widget::Mana
641            | Widget::Stamina
642            | Widget::Spirit
643            | Widget::FieldExperience
644            | Widget::BloodPoints => vital,
645            // The game words these: the label, its word and the percent.
646            Widget::Stance | Widget::Encumbrance | Widget::Mind | Widget::NextLevel => worded,
647            Widget::Pulse => Says {
648                label: true,
649                numbers: false,
650                percent: false,
651                words: false,
652            },
653            // The stack says nothing inside: its page's rows for words are
654            // not drawn.
655            Widget::Vitals => Says {
656                label: false,
657                numbers: false,
658                percent: false,
659                words: false,
660            },
661            _ => return None,
662        };
663        Some(Look {
664            // The stack's rings take its look: fills and tracks of a ring.
665            fills: if *self == Widget::Vitals {
666                Fills::Ring
667            } else {
668                Fills::Right
669            },
670            place: Place::Inside,
671            says,
672            color: None,

864/// A stream's name: the one it is known by, or its id with a capital.
865pub(crate) fn stream_name(id: &str) -> Cow<'static, str> {
866    if let Some((_, name)) = STREAMS
867        .iter()
868        .chain(PANEL_STREAMS.iter())
869        .find(|(known, _)| *known == id)
870    {
871        return Cow::Borrowed(name);
872    }
873    let mut chars = id.chars();
874    Cow::Owned(
875        chars
876            .next()
877            .map(|first| first.to_uppercase().chain(chars).collect())
878            .unwrap_or_default(),
879    )
880}

882/// A dialog's name, from its id: `BetrayerPanel` is "Betrayer panel",
883/// `combat` "Combat".
884pub(crate) fn dialog_name(id: &str) -> Cow<'static, str> {
885    match id {
886        "combat" => return Cow::Borrowed("Combat actions"),
887        "BetrayerPanel" => return Cow::Borrowed("Betrayer panel"),
888        "quick" => return Cow::Borrowed("Quick"),
889        "quick-combat" => return Cow::Borrowed("Quick: combat"),
890        "quick-simu" => return Cow::Borrowed("Quick: simu"),
891        _ => {}
892    }
893    let mut name = String::new();
894    let mut last_lower = false;
895    for c in id.chars() {
896        if c.is_uppercase() && last_lower {
897            name.push(' ');
898            name.extend(c.to_lowercase());
899        } else if name.is_empty() {
900            name.extend(c.to_uppercase());
901        } else {
902            name.push(c);
903        }
904        last_lower = c.is_lowercase();
905    }
906    Cow::Owned(name)
907}

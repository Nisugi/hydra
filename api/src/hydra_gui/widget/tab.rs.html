1//! A tab of a stack, as the player chose it on the widget's page: what it
2//! is called, whether it keeps quiet, how it marks lines unread, and its
3//! colours.
4//!
5//! A tab not showing marks what has come since it last showed: the count
6//! after its name unless the player chose a dot or words of their own. A
7//! quiet tab marks nothing and the next-unread key passes it by. The name
8//! is the widget's unless the player wrote one; an emoji in it is its
9//! glyph. The unread colour paints the tab while it has lines unread, the
10//! showing colour while it is the one shown.
11
12use hydra_ui::settings::{Row, RowKind, Value};
13use serde::{Deserialize, Serialize};
14
15/// What the player chose for one tab.
16#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
17pub(crate) struct TabLook {
18    /// What the tab is called; `None`, the widget's name.
19    #[serde(default, skip_serializing_if = "Option::is_none")]
20    pub(crate) title: Option<String>,
21    /// Marks nothing unread, and the next-unread key passes it by.
22    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
23    pub(crate) quiet: bool,
24    /// How lines unread are marked after the name.
25    #[serde(default, skip_serializing_if = "Mark::is_count")]
26    pub(crate) mark: Mark,
27    /// The tab's colour while it has lines unread; `None`, the theme's.
28    #[serde(default, skip_serializing_if = "Option::is_none")]
29    pub(crate) unread: Option<[u8; 3]>,
30    /// The tab's colour while it is the one showing; `None`, the theme's.
31    #[serde(default, skip_serializing_if = "Option::is_none")]
32    pub(crate) showing: Option<[u8; 3]>,
33}
34
35/// How a tab marks lines unread.
36#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
37#[serde(rename_all = "snake_case")]
38pub(crate) enum Mark {
39    /// `Thoughts 3`: how many.
40    #[default]
41    Count,
42    /// `Thoughts •`.
43    Dot,
44    /// The player's own words or glyph after the name.
45    Words(String),
46}
47
48impl Mark {
49    fn is_count(&self) -> bool {
50        *self == Mark::Count
51    }
52
53    /// What follows the name for `unread` lines.
54    fn said(&self, unread: u64) -> String {
55        match self {
56            Mark::Count => unread.to_string(),
57            Mark::Dot => "\u{2022}".to_owned(),
58            Mark::Words(words) => words.clone(),
59        }
60    }
61
62    /// The choice's value on the page.
63    fn key(&self) -> &str {
64        match self {
65            Mark::Count => "count",
66            Mark::Dot => "dot",
67            Mark::Words(_) => "words",
68        }
69    }
70}

72/// The keys of the tab rows on a widget's page, each beginning `tab_`.
73pub(crate) const PREFIX: &str = "tab_";
74const TITLE: &str = "tab_title";
75const QUIET: &str = "tab_quiet";
76const MARK: &str = "tab_mark";
77const WORDS: &str = "tab_words";
78const UNREAD: &str = "tab_unread";
79const SHOWING: &str = "tab_showing";
80/// The key of the strip's side, which is the cell's, not the tab's.
81pub(crate) const BAR: &str = "tab_bar";

83/// The row for which side the strip of the tab's cell runs along; `bar` as
84/// it is, or `None` for a widget in no cell.
85pub(crate) fn bar_row(bar: Option<crate::layout::Bar>) -> Row {
86    use crate::layout::Bar;
87    let now = bar.unwrap_or_default();
88    Row {
89        key: BAR.to_owned(),
90        label: "Tab bar".to_owned(),
91        help: "Which side of the cell its tabs run along, for every tab in the cell: \
92               across the top or the bottom, or down the left or the right."
93            .to_owned(),
94        kind: RowKind::Choice(
95            Bar::ALL
96                .into_iter()
97                .map(|bar| (bar.key().to_owned(), bar.label().to_owned()))
98                .collect(),
99        ),
100        value: Value::Text(now.key().to_owned()),
101        here: now != Bar::default(),
102        from: None,
103    }
104}

106/// What a tab is called, with its unread mark when it has `unread` lines
107/// and is not quiet: `Thoughts 3`. `:shortcode:`s are emoji.
108pub(crate) fn name(look: Option<&TabLook>, widget_name: &str, unread: Option<u64>) -> String {
109    let title = look
110        .and_then(|look| look.title.clone())
111        .unwrap_or_else(|| widget_name.to_owned());
112    let title = hydra_ui::emoji::expanded(&title).unwrap_or(title);
113    let quiet = look.is_some_and(|look| look.quiet);
114    match unread.filter(|_| !quiet) {
115        Some(unread) => {
116            let mark = look
117                .map_or(Mark::Count, |look| look.mark.clone())
118                .said(unread);
119            if mark.is_empty() {
120                title
121            } else {
122                format!("{title} {mark}")
123            }
124        }
125        None => title,
126    }
127}

129/// The tab's rows on its widget's page.
130pub(crate) fn rows(look: Option<&TabLook>) -> Vec<Row> {
131    let now = look.cloned().unwrap_or_default();
132    let color = |key: &str, label: &str, help: &str, value: Option<[u8; 3]>| Row {
133        key: key.to_owned(),
134        label: label.to_owned(),
135        help: help.to_owned(),
136        kind: RowKind::Color,
137        value: Value::Text(value.map(crate::menu::hex).unwrap_or_default()),
138        here: value.is_some(),
139        from: None,
140    };
141    vec![
142        Row {
143            key: TITLE.to_owned(),
144            label: "Tab title".to_owned(),
145            help: "What its tab is called in a stack; empty, the widget's name. A \
146                   :shortcode: is its emoji."
147                .to_owned(),
148            kind: RowKind::Text,
149            value: Value::Text(now.title.clone().unwrap_or_default()),
150            here: now.title.is_some(),
151            from: None,
152        },
153        Row {
154            key: QUIET.to_owned(),
155            label: "Quiet tab".to_owned(),
156            help: "Its tab marks nothing unread, and the next-unread key passes it by.".to_owned(),
157            kind: RowKind::Toggle,
158            value: Value::On(now.quiet),
159            here: now.quiet,
160            from: None,
161        },
162        Row {
163            key: MARK.to_owned(),
164            label: "Unread mark".to_owned(),
165            help: "After its tab's name while lines are unread: how many, a dot, or the \
166                   words below."
167                .to_owned(),
168            kind: RowKind::Choice(vec![
169                ("count".to_owned(), "How many".to_owned()),
170                ("dot".to_owned(), "A dot".to_owned()),
171                ("words".to_owned(), "These words".to_owned()),
172            ]),
173            value: Value::Text(now.mark.key().to_owned()),
174            here: !now.mark.is_count(),
175            from: None,
176        },
177        Row {
178            key: WORDS.to_owned(),
179            label: "Unread words".to_owned(),
180            help: "The words, or glyph, after the name when the mark is these words; a \
181                   :shortcode: is its emoji."
182                .to_owned(),
183            kind: RowKind::Text,
184            value: Value::Text(match &now.mark {
185                Mark::Words(words) => words.clone(),
186                _ => String::new(),
187            }),
188            here: matches!(now.mark, Mark::Words(_)),
189            from: None,
190        },
191        color(
192            UNREAD,
193            "Unread colour",
194            "Its tab's colour while it has lines unread, #rrggbb; empty, the theme's.",
195            now.unread,
196        ),
197        color(
198            SHOWING,
199            "Showing colour",
200            "Its tab's colour while it is the one showing, #rrggbb; empty, the theme's.",
201            now.showing,
202        ),
203    ]
204}

206/// Set the tab's `key` to `to`, or back to its own.
207pub(crate) fn set(look: &mut TabLook, key: &str, to: Option<&str>) -> Result<(), String> {
208    let named = to.map(str::trim).filter(|to| !to.is_empty());
209    let color = |named: Option<&str>| -> Result<Option<[u8; 3]>, String> {
210        named
211            .map(|to| crate::menu::rgb(to).ok_or_else(|| format!("`{to}` is not a colour.")))
212            .transpose()
213    };
214    match key {
215        TITLE => look.title = named.map(str::to_owned),
216        QUIET => {
217            look.quiet = match named {
218                None | Some("off") => false,
219                Some("on") => true,
220                Some(other) => return Err(format!("`{other}` is not on or off.")),
221            };
222        }
223        MARK => {
224            look.mark = match named {
225                None | Some("count") => Mark::Count,
226                Some("dot") => Mark::Dot,
227                Some("words") => match &look.mark {
228                    Mark::Words(_) => look.mark.clone(),
229                    _ => Mark::Words(String::new()),
230                },
231                Some(other) => return Err(format!("`{other}` is not a mark.")),
232            };
233        }
234        WORDS => {
235            look.mark = match named {
236                Some(words) => Mark::Words(words.to_owned()),
237                None => match &look.mark {
238                    Mark::Words(_) => Mark::Count,
239                    other => other.clone(),
240                },
241            };
242        }
243        UNREAD => look.unread = color(named)?,
244        SHOWING => look.showing = color(named)?,
245        other => return Err(format!("A tab has no setting {other}.")),
246    }
247    Ok(())
248}

250#[cfg(test)]
251mod tests {
252    use super::*;
253
254    /// A tab is named as written, marks its unread as chosen, and keeps
255    /// quiet when told to.
256    #[test]
257    fn a_tab_is_named_and_marked_as_chosen() {
258        assert_eq!(name(None, "Thoughts", Some(3)), "Thoughts 3");
259        assert_eq!(name(None, "Thoughts", None), "Thoughts");
260        let mut look = TabLook::default();
261        set(&mut look, "tab_title", Some(" Chat ")).expect("set");
262        set(&mut look, "tab_mark", Some("dot")).expect("set");
263        assert_eq!(name(Some(&look), "Thoughts", Some(3)), "Chat \u{2022}");
264        set(&mut look, "tab_words", Some("!")).expect("set");
265        assert_eq!(look.mark, Mark::Words("!".to_owned()));
266        assert_eq!(name(Some(&look), "Thoughts", Some(3)), "Chat !");
267        set(&mut look, "tab_quiet", Some("on")).expect("set");
268        assert_eq!(name(Some(&look), "Thoughts", Some(3)), "Chat");
269        set(&mut look, "tab_unread", Some("#ff0000")).expect("set");
270        assert_eq!(look.unread, Some([0xff, 0, 0]));
271        assert!(set(&mut look, "tab_unread", Some("red")).is_err());
272        assert!(set(&mut look, "tab_size", Some("1")).is_err());
273        let back: TabLook =
274            serde_json::from_str(&serde_json::to_string(&look).expect("w")).expect("read");
275        assert_eq!(back, look);
276        assert_eq!(rows(None).len(), 6);
277    }

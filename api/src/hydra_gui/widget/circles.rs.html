1//! The colour an effect's bar wears for the spell circle its spell is in:
2//! a Minor Spirit spell's bar one colour, a Wizard spell's another, so a
3//! glance at a list tells which circles are up.
4//!
5//! The colours are the character's, one for every effect list, kept with
6//! the layout and shown on each list's page as a table of circle and
7//! colour. Hydra starts with a palette over the circles a character casts
8//! from; a row removed leaves its circle in the list's own colour, and the
9//! table emptied leaves every bar so. An effect's spell is found by the
10//! effect's name, as the spell table names it; an effect that is no spell,
11//! or a tracked effect given a colour of its own, is not coloured by its
12//! circle.
13
14use std::collections::BTreeMap;

16use hydra_model::spells;
17use hydra_ui::settings::{Row, RowKind, Value};

19/// The circles' colours, by circle number.
20pub(crate) type Circles = BTreeMap<u16, [u8; 3]>;
21
22/// The key of the table row on an effect list's page.
23pub(crate) const TABLE: &str = "circles";
24
25/// The palette Hydra starts with: the circles a character casts from,
26/// each a hue of its own.
27pub(crate) fn palette() -> Circles {
28    [
29        (1, [0x8f, 0xb8, 0xde]),  // Minor Spirit: pale blue
30        (2, [0x4f, 0x7f, 0xc0]),  // Major Spirit: blue
31        (3, [0xe0, 0xd0, 0x80]),  // Cleric: gold
32        (4, [0x9f, 0xd4, 0x9f]),  // Minor Elemental: pale green
33        (5, [0x4f, 0xa8, 0x5f]),  // Major Elemental: green
34        (6, [0x8f, 0x6f, 0x3f]),  // Ranger: bark
35        (7, [0x9f, 0x5f, 0xbf]),  // Sorcerer: violet
36        (9, [0xd0, 0x70, 0x40]),  // Wizard: flame
37        (10, [0xe0, 0x90, 0xc0]), // Bard: rose
38        (11, [0x60, 0xc0, 0xc0]), // Empath: teal
39        (12, [0xa0, 0xa0, 0xe0]), // Minor Mental: lavender
40        (16, [0xf0, 0xf0, 0xf0]), // Paladin: white
41        (17, [0xb0, 0xb0, 0xb0]), // Arcane: grey
42        (97, [0xc0, 0x40, 0x40]), // Guardians of Sunfist: red
43        (98, [0x70, 0x70, 0x90]), // Order of Voln: slate
44        (99, [0xd0, 0xd0, 0x60]), // Council of Light: light
45    ]
46    .into_iter()
47    .collect()
48}

50/// The colour `effect` wears for its spell's circle, when the effect names
51/// a spell whose circle has one. A percent or count after the name,
52/// `Resist Nature (50%)`, is not part of the name.
53pub(crate) fn color(circles: &Circles, effect: &str) -> Option<[u8; 3]> {
54    let name = effect.split(" (").next().unwrap_or(effect).trim();
55    let spell = spells::spell_named(name)?;
56    circles.get(&spell.circle()).copied()
57}
58
59/// The table row: each circle by name and its colour.
60pub(crate) fn row(circles: Option<&Circles>) -> Row {
61    let palette = palette();
62    let now = circles.unwrap_or(&palette);
63    Row {
64        key: TABLE.to_owned(),
65        label: "Spell circle colours".to_owned(),
66        help: "A bar in its spell's circle's colour, #rrggbb, when the effect is a spell \
67               of a circle listed; the circle by name or number. Shared by every effect \
68               list. A tracked effect's own colour comes first; a circle not listed \
69               leaves the list's colour."
70            .to_owned(),
71        kind: RowKind::Table(vec!["Circle".to_owned(), "Colour".to_owned()]),
72        value: Value::Table(
73            now.iter()
74                .map(|(circle, color)| {
75                    vec![
76                        spells::circle_name(*circle)
77                            .map_or_else(|| circle.to_string(), str::to_owned),
78                        crate::menu::hex(*color),
79                    ]
80                })
81                .collect(),
82        ),
83        here: circles.is_some(),
84        from: None,
85    }
86}

88/// The circles the table `written` says: each row a circle, by name or
89/// number, and its colour.
90pub(crate) fn read(written: &str) -> Result<Circles, String> {
91    let rows: Vec<Vec<String>> =
92        serde_json::from_str(written).map_err(|_| "not rows of cells".to_owned())?;
93    let mut circles = Circles::new();
94    for mut cells in rows {
95        cells.resize(2, String::new());
96        let (circle, color) = (cells[0].trim(), cells[1].trim());
97        if circle.is_empty() && color.is_empty() {
98            continue;
99        }
100        let number = circle.parse::<u16>().ok().or_else(|| {
101            (1..=99).find(|n| {
102                spells::circle_name(*n).is_some_and(|name| name.eq_ignore_ascii_case(circle))
103            })
104        });
105        let number = number.ok_or_else(|| format!("`{circle}` is not a spell circle."))?;
106        let color =
107            crate::menu::rgb(color).ok_or_else(|| format!("`{color}` is not a colour #rrggbb."))?;
108        circles.insert(number, color);
109    }
110    Ok(circles)
111}

113#[cfg(test)]
114mod tests {
115    use super::*;
116
117    /// A spell's bar takes its circle's colour, by the effect's name
118    /// without what follows it; a tracked colour or no spell leaves none.
119    #[test]
120    fn an_effect_is_coloured_by_its_spells_circle() {
121        let circles = palette();
122        assert_eq!(
123            color(&circles, "Spirit Warding I"),
124            circles.get(&1).copied()
125        );
126        assert_eq!(
127            color(&circles, "Elemental Defense I (50%)"),
128            circles.get(&4).copied()
129        );
130        assert_eq!(color(&circles, "Stunned"), None, "not a spell");
131        assert_eq!(color(&Circles::new(), "Spirit Warding I"), None);
132    }
133
134    /// The table reads back by name or number and refuses what is neither;
135    /// the row shows the palette until the player chose.
136    #[test]
137    fn the_table_names_circles_and_reads_them_back() {
138        let read = read(r##"[["Wizard","#d07040"],["1","#8fb8de"],["",""]]"##).expect("read");
139        assert_eq!(read.len(), 2);
140        assert_eq!(read.get(&9), Some(&[0xd0, 0x70, 0x40]));
141        assert!(read_err("[[\"Weather\",\"#000000\"]]").contains("spell circle"));
142        assert!(read_err("[[\"Wizard\",\"red\"]]").contains("#rrggbb"));
143        let row = row(None);
144        assert!(!row.here);
145        assert!(matches!(&row.value, Value::Table(rows) if rows.iter().any(|r| r[0] == "Wizard")));
146    }
147
148    fn read_err(written: &str) -> String {
149        read(written).expect_err("refused")
150    }

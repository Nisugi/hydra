210/// An indicator, filling what it is given: lit in its colour when on, dim
211/// when off, and asking when the game has not said. A screen reader hears
212/// which: *"Stunned: yes"*.
213pub(super) fn indicator(
214    ui: &mut egui::Ui,
215    indicator: Indicator,
216    state: Option<&GameState>,
217    name: &str,
218) {
219    let known = state.and_then(|state| {
220        state
221            .status
222            .is_known(indicator.id())
223            .then(|| state.status.get(indicator.id()))
224    });
225    let size = Vec2::new(
226        ui.available_width(),
227        LINE.min(ui.available_height().max(LINE)),
228    );
229    let (rect, response) = ui.allocate_exact_size(size, Sense::hover());
230    let said = match known {
231        Some(true) => "yes",
232        Some(false) => "no",
233        None => "unknown",
234    };
235    response.widget_info(|| {
236        egui::WidgetInfo::labeled(egui::WidgetType::Label, true, format!("{name}: {said}"))
237    });
238    let lit = theme::color(ui.ctx(), indicator.token());
239    let painter = ui.painter();
240    let rect = rect.shrink(1.0);
241    let (text, color) = match known {
242        Some(true) => {
243            painter.rect_filled(rect, theme::corner(ui.ctx()), lit);
244            (name.to_owned(), readable_on(lit))
245        }
246        Some(false) => {
247            painter.rect_stroke(
248                rect,
249                theme::corner(ui.ctx()),
250                Stroke::new(1.0, ui.visuals().weak_text_color()),
251                StrokeKind::Inside,
252            );
253            (name.to_owned(), ui.visuals().weak_text_color())
254        }
255        None => (format!("{name} ?"), ui.visuals().weak_text_color()),
256    };
257    painter.text(
258        rect.center(),
259        Align2::CENTER_CENTER,
260        text,
261        FontId::proportional(12.0),
262        color,
263    );
264}

266/// The custom statuses `lit` at server second `now`, each lit with its name
267/// and the time it has left; a line saying so when none is.
268pub(super) fn statuses(ui: &mut egui::Ui, lit: &hydra_ui::condition::Flags, now: Option<u32>) {
269    let shown: Vec<(&str, Option<u32>)> = lit.lit(now).collect();
270    if shown.is_empty() {
271        ui.weak("No custom status is lit.");
272        return;
273    }
274    let fill = theme::color(ui.ctx(), T::Accent);
275    ui.horizontal_wrapped(|ui| {
276        for (name, left) in shown {
277            let text = match left {
278                Some(left) => format!("{name} {}", clock(left)),
279                None => name.to_owned(),
280            };
281            egui::Frame::new()
282                .fill(fill)
283                .corner_radius(theme::corner(ui.ctx()))
284                .inner_margin(egui::Margin::symmetric(6, 2))
285                .show(ui, |ui| {
286                    ui.label(egui::RichText::new(text).color(readable_on(fill)));
287                });
288        }
289    });
290}

292/// Seconds as a clock: `1:59`, or `1:02:03` past the hour.
293pub(super) fn clock(seconds: u32) -> String {
294    let (hours, minutes, seconds) = (seconds / 3600, seconds / 60 % 60, seconds % 60);
295    if hours > 0 {
296        format!("{hours}:{minutes:02}:{seconds:02}")
297    } else {
298        format!("{minutes}:{seconds:02}")
299    }
300}

302/// When the next pulse comes, as a bar that fills toward it, drawn as
303/// `look` says.
304pub(super) fn pulse(
305    ui: &mut egui::Ui,
306    state: Option<&GameState>,
307    name: &str,
308    look: Option<&crate::bar::Look>,
309) {
310    let now = state.and_then(GameState::game_time_now);
311    let pulse = state.and_then(|state| state.world.pulse.as_ref());
312    // The next pulse brings mana: drawn in the mana's colour.
313    let colour = theme::color(
314        ui.ctx(),
315        if pulse.is_some_and(|pulse| pulse.mana) {
316            T::Mana
317        } else {
318            T::Pulse
319        },
320    );
321    let (label, percent) = pulse_said(pulse, now, name);
322    if look.is_some_and(|look| look.clock) {
323        super::draw::line(ui, egui::RichText::new(label).color(colour));
324        return;
325    }
326    let amount = percent.map(|percent| Amount {
327        percent,
328        current: None,
329        max: None,
330    });
331    let drawn = super::draw::as_looks(ui, Bar::new(&label, amount).fill(colour), look);
332    ui.add(drawn.fitted(ui));
333}

335/// What the pulse bar says at server second `now`, and how full it is: a
336/// countdown from the pulse to the earliest the next can come, 45 seconds,
337/// which goes on below zero until it does -- *"Pulse 20s"*, *"Pulse 0s"*,
338/// *"Pulse -12s"*, the next coming by -30 -- and a mark, `✦`, when the next
339/// brings mana; the bar filling as it nears and full once it is due;
340/// `name` asking before the first pulse or the game's clock.
341pub(super) fn pulse_said(
342    pulse: Option<&Pulse>,
343    now: Option<u32>,
344    name: &str,
345) -> (String, Option<u32>) {
346    let clock = pulse
347        .zip(now)
348        .and_then(|(pulse, now)| pulse_clock(pulse, now).map(|left| (pulse, left)));
349    match clock {
350        None => (format!("{name} ?"), None),
351        Some((pulse, left)) => {
352            let min = i64::from(pulse.min.max(1));
353            let passed = (min - left).clamp(0, min);
354            let full = u32::try_from(passed * 100 / min).unwrap_or(100);
355            let mana = if pulse.mana { " \u{2726}" } else { "" };
356            (format!("{name} {left}s{mana}"), Some(full))
357        }
358    }
359}

361/// The pulse as a clock at server second `now`: seconds to its earliest,
362/// then below zero, how late it is, until it comes and the clock starts
363/// again: 20 counting down to 0, then -1, -2, -3, until the pulse comes.
364pub(super) fn pulse_clock(pulse: &Pulse, now: u32) -> Option<i64> {
365    let at = pulse.at?;
366    Some(i64::from(pulse.min) - i64::from(now.saturating_sub(at)))
367}

369/// The world events under way, each with where and how long it has left.
370pub(super) fn world_events(ui: &mut egui::Ui, state: Option<&GameState>) {
371    let Some(state) = state else {
372        ui.weak("World events unknown");
373        return;
374    };
375    let now = state.game_time_now();
376    let events: Vec<(Option<&str>, &str, Option<u32>)> = state
377        .world
378        .events(now)
379        .map(|event| {
380            (
381                event.realm.as_deref(),
382                event.text.as_str(),
383                now.zip(event.expires_at)
384                    .map(|(now, at)| at.saturating_sub(now)),
385            )
386        })
387        .collect();
388    events_listed(ui, &events);
389}
390
391/// `events` -- where, what, and seconds left when the game gave an expiry
392/// -- a line each, or that there are none.
393pub(super) fn events_listed(ui: &mut egui::Ui, events: &[(Option<&str>, &str, Option<u32>)]) {
394    if events.is_empty() {
395        ui.weak("No world events.");
396    }
397    for (realm, text, left) in events {
398        let mut line = match realm {
399            Some(realm) => format!("{realm}: {text}"),
400            None => (*text).to_owned(),
401        };
402        if let Some(left) = left {
403            line = format!("{line} ({} left)", clock(*left));
404        }
405        ui.label(line);
406    }
407}

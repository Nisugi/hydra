224/// A game indicator, drawn by [`lamp`]: lit in its colour when on, dim
225/// when off; or its icon, when its look has one, grey and faded while off;
226/// or nothing while off, when its look says. One the game has not reported
227/// since the character came in is drawn off, as every condition reads it,
228/// and its hover says so. A screen reader hears which: *"Stunned: yes"*,
229/// *"Bleeding: not reported yet"*.
230pub(super) fn indicator(
231    ui: &mut egui::Ui,
232    indicator: Indicator,
233    state: Option<&GameState>,
234    name: &str,
235    shown: Option<&IndicatorShown>,
236) {
237    let known = state.and_then(|state| {
238        state
239            .status
240            .is_known(indicator.id())
241            .then(|| state.status.get(indicator.id()))
242    });
243    let told = known.ok_or_else(|| {
244        (
245            "not reported yet",
246            format!(
247                "{name}: the game has not reported it since this character came in, so it counts as off."
248            ),
249        )
250    });
251    lamp(ui, name, &told, indicator.token(), shown);
252}

254/// An indicator drawn, filling what it is given: `told` whether it is lit,
255/// or, when that cannot be told, what a screen reader hears in place of
256/// yes or no and what its hover says, drawn off. Lit, a pill filled with
257/// the palette's `token`; off, an outline. Or its icon, when its look has
258/// one, grey and faded while off; or nothing while off, when its look says.
259pub(super) fn lamp(
260    ui: &mut egui::Ui,
261    name: &str,
262    told: &Result<bool, (&str, String)>,
263    token: T,
264    shown: Option<&IndicatorShown>,
265) {
266    let lit = *told == Ok(true);
267    // Drawn as its look says: its own icon, the style's picture, or a pill.
268    let drawn = shown.map_or(Drawn::Pill, IndicatorShown::drawn);
269    let iconed = shown
270        .filter(|_| drawn == Drawn::Icon)
271        .and_then(|shown| shown.look.icon.as_ref().map(|icon| (icon, shown)));
272    let styled = shown
273        .filter(|_| drawn == Drawn::Style)
274        .and_then(|shown| shown.styled.as_deref());
275    // An icon's largest edge: its look's size, or as big as its room.
276    #[expect(
277        clippy::cast_precision_loss,
278        reason = "an icon is a few hundred points"
279    )]
280    let most = shown
281        .and_then(IndicatorShown::size)
282        .map_or(f32::INFINITY, |size| size as f32);
283    // An icon takes the room it is given, up to its size; a pill, a line.
284    let size = match iconed.map(|_| ()).or(styled.map(|_| ())) {
285        Some(()) => {
286            let room = ui.available_size().max(Vec2::splat(LINE));
287            Vec2::new(room.x, room.y.min(most))
288        }
289        None => Vec2::new(
290            ui.available_width(),
291            LINE.min(ui.available_height().max(LINE)),
292        ),
293    };
294    let (rect, response) = ui.allocate_exact_size(size, Sense::hover());
295    let said = match told {
296        Ok(true) => "yes",
297        Ok(false) => "no",
298        Err((said, _)) => said,
299    };
300    response.widget_info(|| {
301        egui::WidgetInfo::labeled(egui::WidgetType::Label, true, format!("{name}: {said}"))
302    });
303    if let Err((_, why)) = told {
304        response.clone().on_hover_text(why);
305    }
306    // Off, and its look says to hide it while off: nothing.
307    if !lit && shown.is_some_and(IndicatorShown::hides_off) {
308        return;
309    }
310    // An icon, a style's or its own, in a square centred in its room; while
311    // off, as faded and as grey as its look says.
312    let edge = rect.width().min(rect.height()).min(most);
313    let square = egui::Rect::from_center_size(rect.center(), Vec2::splat(edge));
314    let (grey, opacity) = match shown {
315        _ if lit => (false, 1.0),
316        Some(shown) => (shown.grey(), shown.faded()),
317        None => (true, 0.4),
318    };
319    if let Some(path) = styled {
320        let picture = if grey {
321            crate::pictures::grey_picture(ui.ctx(), path)
322        } else {
323            crate::pictures::picture(ui.ctx(), path)
324        };
325        if let Some(picture) = picture {
326            let whole = egui::Rect::from_min_max(egui::pos2(0.0, 0.0), egui::pos2(1.0, 1.0));
327            let tint = egui::Color32::WHITE.gamma_multiply(opacity);
328            ui.painter().image(picture.id(), square, whole, tint);
329            return;
330        }
331    }
332    let painter = ui.painter();
333    if let Some((icon, shown)) = iconed {
334        let drawn =
335            crate::widget::hotbar::paint_icon(painter, square, icon, &shown.icons, (grey, opacity));
336        if drawn {
337            return;
338        }
339    }
340    let fill = theme::color(ui.ctx(), token);
341    let rect = rect.shrink(1.0);
342    let color = if lit {
343        painter.rect_filled(rect, theme::corner(ui.ctx()), fill);
344        readable_on(fill)
345    } else {
346        painter.rect_stroke(
347            rect,
348            theme::corner(ui.ctx()),
349            Stroke::new(1.0, ui.visuals().weak_text_color()),
350            StrokeKind::Inside,
351        );
352        ui.visuals().weak_text_color()
353    };
354    painter.text(
355        rect.center(),
356        Align2::CENTER_CENTER,
357        name,
358        FontId::proportional(12.0),
359        color,
360    );
361}

363/// The custom statuses `lit` at server second `now`, each lit with its name
364/// and the time it has left; a line saying so when none is.
365pub(super) fn statuses(ui: &mut egui::Ui, lit: &hydra_ui::condition::Flags, now: Option<u32>) {
366    let shown: Vec<(&str, Option<u32>)> = lit.lit(now).collect();
367    if shown.is_empty() {
368        ui.weak("No custom status is lit.");
369        return;
370    }
371    let fill = theme::color(ui.ctx(), T::Accent);
372    ui.horizontal_wrapped(|ui| {
373        for (name, left) in shown {
374            let text = match left {
375                Some(left) => format!("{name} {}", clock(left)),
376                None => name.to_owned(),
377            };
378            egui::Frame::new()
379                .fill(fill)
380                .corner_radius(theme::corner(ui.ctx()))
381                .inner_margin(egui::Margin::symmetric(6, 2))
382                .show(ui, |ui| {
383                    ui.label(egui::RichText::new(text).color(readable_on(fill)));
384                });
385        }
386    });
387}

389/// Seconds as a clock: `1:59`, or `1:02:03` past the hour.
390pub(super) fn clock(seconds: u32) -> String {
391    let (hours, minutes, seconds) = (seconds / 3600, seconds / 60 % 60, seconds % 60);
392    if hours > 0 {
393        format!("{hours}:{minutes:02}:{seconds:02}")
394    } else {
395        format!("{minutes}:{seconds:02}")
396    }
397}

399/// When the next pulse comes, as a bar that fills toward it, drawn as
400/// `look` says.
401pub(super) fn pulse(
402    ui: &mut egui::Ui,
403    state: Option<&GameState>,
404    name: &str,
405    look: Option<&crate::bar::Look>,
406) {
407    let now = state.and_then(GameState::game_time_now);
408    let pulse = state.and_then(|state| state.world.pulse.as_ref());
409    // The next pulse brings mana: drawn in the mana's colour.
410    let colour = theme::color(
411        ui.ctx(),
412        if pulse.is_some_and(|pulse| pulse.mana) {
413            T::Mana
414        } else {
415            T::Pulse
416        },
417    );
418    let (label, percent) = pulse_said(pulse, now, name);
419    if look.is_some_and(|look| look.clock) {
420        super::draw::line(ui, egui::RichText::new(label).color(colour));
421        return;
422    }
423    let amount = percent.map(|percent| Amount {
424        percent,
425        current: None,
426        max: None,
427    });
428    let drawn = super::draw::as_looks(ui, Bar::new(&label, amount).fill(colour), look);
429    ui.add(drawn.fitted(ui));
430}

432/// What the pulse bar says at server second `now`, and how full it is: a
433/// countdown from the pulse to the earliest the next can come, 45 seconds,
434/// which goes on below zero until it does -- *"Pulse 20s"*, *"Pulse 0s"*,
435/// *"Pulse -12s"*, the next coming by -30 -- and a mark, `✦`, when the next
436/// brings mana; the bar filling as it nears and full once it is due;
437/// `name` asking before the first pulse or the game's clock.
438pub(super) fn pulse_said(
439    pulse: Option<&Pulse>,
440    now: Option<u32>,
441    name: &str,
442) -> (String, Option<u32>) {
443    let clock = pulse
444        .zip(now)
445        .and_then(|(pulse, now)| pulse_clock(pulse, now).map(|left| (pulse, left)));
446    match clock {
447        None => (format!("{name} ?"), None),
448        Some((pulse, left)) => {
449            let min = i64::from(pulse.min.max(1));
450            let passed = (min - left).clamp(0, min);
451            let full = u32::try_from(passed * 100 / min).unwrap_or(100);
452            let mana = if pulse.mana { " \u{2726}" } else { "" };
453            (format!("{name} {left}s{mana}"), Some(full))
454        }
455    }
456}

458/// The pulse as a clock at server second `now`: seconds to its earliest,
459/// then below zero, how late it is, until it comes and the clock starts
460/// again: 20 counting down to 0, then -1, -2, -3, until the pulse comes.
461pub(super) fn pulse_clock(pulse: &Pulse, now: u32) -> Option<i64> {
462    let at = pulse.at?;
463    Some(i64::from(pulse.min) - i64::from(now.saturating_sub(at)))
464}

466/// The world events under way, each with where and how long it has left.
467pub(super) fn world_events(ui: &mut egui::Ui, state: Option<&GameState>) {
468    let Some(state) = state else {
469        ui.weak("World events unknown");
470        return;
471    };
472    let now = state.game_time_now();
473    let events: Vec<(Option<&str>, &str, Option<u32>)> = state
474        .world
475        .events(now)
476        .map(|event| {
477            (
478                event.realm.as_deref(),
479                event.text.as_str(),
480                now.zip(event.expires_at)
481                    .map(|(now, at)| at.saturating_sub(now)),
482            )
483        })
484        .collect();
485    events_listed(ui, &events);
486}
487
488/// `events` -- where, what, and seconds left when the game gave an expiry
489/// -- a line each, or that there are none.
490pub(super) fn events_listed(ui: &mut egui::Ui, events: &[(Option<&str>, &str, Option<u32>)]) {
491    if events.is_empty() {
492        ui.weak("No world events.");
493    }
494    for (realm, text, left) in events {
495        let mut line = match realm {
496            Some(realm) => format!("{realm}: {text}"),
497            None => (*text).to_owned(),
498        };
499        if let Some(left) = left {
500            line = format!("{line} ({} left)", clock(*left));
501        }
502        ui.label(line);
503    }
504}

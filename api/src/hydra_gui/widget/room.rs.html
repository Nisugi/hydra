11/// A direction on the compass: its word, what the rose calls it, and where
12/// it sits, in cells from the top left of a four-by-three grid.
13const DIRECTIONS: [(&str, &str, (u8, u8)); 11] = [
14    ("northwest", "NW", (0, 0)),
15    ("north", "N", (1, 0)),
16    ("northeast", "NE", (2, 0)),
17    ("west", "W", (0, 1)),
18    ("out", "OUT", (1, 1)),
19    ("east", "E", (2, 1)),
20    ("southwest", "SW", (0, 2)),
21    ("south", "S", (1, 2)),
22    ("southeast", "SE", (2, 2)),
23    ("up", "UP", (3, 0)),
24    ("down", "DN", (3, 2)),
25];

27/// The room's ways out as a compass rose, each lit when the room has it.
28/// A click on a lit one goes that way, when `goes` -- never for a compass
29/// following another character. The direction clicked, if one was.
30pub(super) fn compass(ui: &mut egui::Ui, state: Option<&GameState>, goes: bool) -> Option<String> {
31    let exits = state.and_then(|state| state.room.exits.as_ref());
32    let size = ui.available_size();
33    let cell = (size.x / 4.0).min(size.y / 3.0).max(12.0);
34    let (area, response) = ui.allocate_exact_size(vec2(cell * 4.0, cell * 3.0), Sense::hover());
35    if exits.is_none() {
36        response.widget_info(|| {
37            egui::WidgetInfo::labeled(egui::WidgetType::Label, true, "Exits unknown")
38        });
39    }
40    let mut went = None;
41    for (word, short, (column, row)) in DIRECTIONS {
42        let rect = Rect::from_min_size(
43            pos2(
44                area.min.x + f32::from(column) * cell,
45                area.min.y + f32::from(row) * cell,
46            ),
47            Vec2::splat(cell),
48        )
49        .shrink(2.0);
50        // The game names them short (`n`, `ne`): each said in full first.
51        let open =
52            exits.is_some_and(|exits| exits.iter().any(|exit| hydra_ui::exit_word(exit) == word));
53        let sense = if open && goes {
54            Sense::click()
55        } else {
56            Sense::hover()
57        };
58        let direction = ui.interact(rect, response.id.with(word), sense);
59        direction.widget_info(|| egui::WidgetInfo::labeled(egui::WidgetType::Button, open, word));
60        let painter = ui.painter();
61        let text = if open {
62            let accent = theme::color(ui.ctx(), T::Accent);
63            painter.rect_filled(rect, theme::corner(ui.ctx()), accent);
64            readable_on(accent)
65        } else {
66            painter.rect_stroke(
67                rect,
68                theme::corner(ui.ctx()),
69                Stroke::new(1.0, ui.visuals().weak_text_color()),
70                StrokeKind::Inside,
71            );
72            ui.visuals().weak_text_color()
73        };
74        painter.text(
75            rect.center(),
76            Align2::CENTER_CENTER,
77            short,
78            FontId::proportional((cell * 0.32).clamp(8.0, 14.0)),
79            text,
80        );
81        if direction.clicked() {
82            went = Some(word.to_owned());
83        }
84    }
85    went
86}

88/// Who is fighting here: the character and its friends, then its foes;
89/// each a link to it, live when `own`, the window's own character's.
90pub(super) fn combat(
91    ui: &mut egui::Ui,
92    state: Option<&GameState>,
93    own: bool,
94) -> Option<super::Clicked> {
95    let Some(state) = state else {
96        ui.weak("Combat unknown");
97        return None;
98    };
99    let fighters = hydra_ui::room::fighters(state);
100    fighting(ui, state.character.stance.as_deref(), &fighters, own)
101}

103/// The combat list: *FRIENDLY*, the character with its `stance` and its
104/// friends; *FOES*, the rest, or that there are none. Each a link to it,
105/// live when `own`.
106pub(super) fn fighting(
107    ui: &mut egui::Ui,
108    stance: Option<&str>,
109    fighters: &[Fighter],
110    own: bool,
111) -> Option<super::Clicked> {
112    ui.label(egui::RichText::new("FRIENDLY").small().weak());
113    let player = theme::color(ui.ctx(), T::Player);
114    ui.horizontal(|ui| {
115        ui.colored_label(player, "You");
116        if let Some(stance) = stance {
117            ui.weak(stance);
118        }
119    });
120    let mut clicked = None;
121    for friend in fighters.iter().filter(|fighter| fighter.friend) {
122        clicked = clicked.or(fighter(ui, friend, player, own));
123    }
124    ui.label(egui::RichText::new("FOES").small().weak());
125    let mut foes = fighters.iter().filter(|fighter| !fighter.friend).peekable();
126    if foes.peek().is_none() {
127        ui.weak("No foes.");
128    }
129    for foe in foes {
130        clicked = clicked.or(fighter(ui, foe, theme::color(ui.ctx(), T::Creature), own));
131    }
132    clicked
133}

135/// One fighter: its name, a link to it, live when `own`; whether it is
136/// dead, its statuses, and its health as a thin bar when stated.
137fn fighter(
138    ui: &mut egui::Ui,
139    fighter: &Fighter,
140    color: Color32,
141    own: bool,
142) -> Option<super::Clicked> {
143    let clicked = ui.horizontal_wrapped(|ui| {
144        let id = fighter.id.to_string();
145        let name = hydra_ui::room::object_run(
146            &fighter.name,
147            (&id, fighter.noun.as_deref().unwrap_or_default()),
148            super::described::rgb_of(color),
149        );
150        let clicked = super::described::say(ui, &[name], own);
151        if fighter.dead {
152            ui.weak("dead");
153        }
154        if !fighter.statuses.is_empty() {
155            ui.weak(fighter.statuses.join(", "));
156        }
157        clicked
158    });
159    if let Some(percent) = fighter.health {
160        let bar = Bar::new(
161            &fighter.name,
162            Some(Amount {
163                percent,
164                current: None,
165                max: None,
166            }),
167        )
168        .fill(theme::color(ui.ctx(), T::Health))
169        .size([ui.available_width(), 6.0])
170        .says(Says {
171            label: false,
172            numbers: false,
173            percent: false,
174            words: false,
175        });
176        ui.add(bar);
177    }
178    clicked.inner
179}

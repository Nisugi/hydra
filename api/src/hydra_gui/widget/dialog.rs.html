1//! One of the game's own dialogs, drawn from what the model keeps of it
2//! (`hydra_model::state::dialogs`): the Betrayer panel's blood points and
3//! items, the combat panel's buttons, a panel an event adds.
4//!
5//! Drawn in the order its parts came, one to a line: a label's text, a bar, a
6//! button, a link or a drop-down that sends its command. Not where the game
7//! places each part: its anchors, percent coordinates and `justify` bitfield
8//! are not used yet. A part that sends nothing and says nothing, an edit box or
9//! a skin, is left out.
10//!
11//! A command starting `_` is the client's own and is sent without an echo,
12//! as a dragged item is (`crate::carry`); any other is sent as typed. A
13//! dialog of another character's, followed, sends nothing.
14

20/// Draw the dialog `id` from `state`; the command a part sent, if one did
21/// and the dialog is the window's own character's (`own`).
22pub(super) fn dialog(
23    ui: &mut egui::Ui,
24    state: Option<&GameState>,
25    id: &str,
26    own: bool,
27) -> Option<Clicked> {
28    let Some(dialog) = state.and_then(|state| state.dialogs.get(id)) else {
29        ui.weak("Nothing yet.");
30        return None;
31    };
32    if let Some(title) = &dialog.title {
33        ui.strong(title);
34    }
35    if !dialog.open {
36        ui.weak("Closed by the game.");
37    }
38    let mut sent = None;
39    ui.add_enabled_ui(own, |ui| {
40        for (part_id, part) in &dialog.parts {
41            if let Some(command) = draw_part(ui, part_id, part) {
42                sent = Some(command);
43            }
44        }
45    });
46    // The game's: checked before it goes, whichever way.
47    sent.map(Clicked::FromGame)
48}

50/// Draw one part; the command it sends, when clicked or chosen.
51fn draw_part(ui: &mut egui::Ui, id: &str, part: &Part) -> Option<String> {
52    let attrs = match part {
53        Part::Label(text) => {
54            if !text.trim().is_empty() {
55                ui.label(text);
56            }
57            return None;
58        }
59        Part::Bar { percent, text } => {
60            let fill = f32::from(u16::try_from(*percent).unwrap_or(100).min(100)) / 100.0;
61            ui.add(egui::ProgressBar::new(fill).text(text));
62            return None;
63        }
64        Part::Widget { kind, attrs } => (kind.as_str(), attrs),
65    };
66    let (kind, attrs) = attrs;
67    let get = |name: &str| {
68        attrs
69            .iter()
70            .find(|(key, _)| key == name)
71            .map(|(_, value)| value.as_str())
72    };
73    let command = get("cmd").filter(|cmd| !cmd.trim().is_empty())?;
74    let tip = get("tooltip").unwrap_or_default();
75    let words = get("value")
76        .filter(|words| !words.is_empty())
77        .or_else(|| get("echo"))
78        .or(get("tooltip"))
79        .unwrap_or(id);
80    match kind {
81        "cmdButton" => ui
82            .button(words)
83            .on_hover_text(tip)
84            .clicked()
85            .then(|| command.to_owned()),
86        "link" => ui
87            .link(words)
88            .on_hover_text(tip)
89            .clicked()
90            .then(|| command.to_owned()),
91        // An image with a command is a button: the combat panel's
92        // sheathe and shield. Its picture is a skin's, which Hydra draws
93        // as words.
94        "image" => {
95            let words = get("tooltip").or(get("echo")).unwrap_or(id);
96            ui.button(words).clicked().then(|| command.to_owned())
97        }
98        "dropDownBox" => drop_down(ui, id, command, &get),
99        _ => None,
100    }
101}

103/// A drop-down's choices, `content_text` shown and `content_value` sent in
104/// place of its `%id%` in the command; the choice sent, once made.
105fn drop_down<'a>(
106    ui: &mut egui::Ui,
107    id: &str,
108    command: &str,
109    get: &dyn Fn(&str) -> Option<&'a str>,
110) -> Option<String> {
111    let texts = get("content_text").unwrap_or_default().split(',');
112    let values = get("content_value").unwrap_or_default().split(',');
113    let current = get("value").unwrap_or_default();
114    let mut chosen = None;
115    egui::ComboBox::from_id_salt(("dialog", id))
116        .selected_text(current)
117        .show_ui(ui, |ui| {
118            for (text, value) in texts.zip(values) {
119                if ui.selectable_label(text == current, text).clicked() {
120                    chosen = Some(command.replace(&format!("%{id}%"), value));
121                }
122            }
123        });
124    chosen
125}

127#[cfg(test)]
128mod tests {
129    use std::sync::{Arc, Mutex};
130
131    use egui::Id;
132    use egui_kittest::Harness;
133    use egui_kittest::kittest::Queryable as _;
134
135    use crate::fixture::{snapshot, story};
136    use crate::widget::{Seen, Widget};
137
138    /// A game dialog shows its title and parts in the order they came; its
139    /// button sends its command, and one starting `_` goes without an echo.
140    #[test]
141    fn a_game_dialog_shows_its_parts_and_sends_its_commands() {
142        use hydra_protocol::Frame;
143        let mut ashryn = snapshot();
144        let dialog = Some("BetrayerPanel".to_owned());
145        let button = |id: &str, tooltip: &str, cmd: &str| Frame::InjuryImage {
146            id: id.to_owned(),
147            name: String::new(),
148            dialog: dialog.clone(),
149            attrs: [("id", id), ("tooltip", tooltip), ("cmd", cmd)]
150                .map(|(k, v)| (k.to_owned(), v.to_owned()))
151                .to_vec(),
152        };
153        for frame in [
154            Frame::DialogOpen {
155                id: "BetrayerPanel".to_owned(),
156                title: Some("Betrayer".to_owned()),
157                attrs: Vec::new(),
158            },
159            Frame::Label {
160                id: "lblBPs".to_owned(),
161                value: "Blood Points: 1".to_owned(),
162                dialog: dialog.clone(),
163                attrs: Vec::new(),
164            },
165            button("trade", "trade", "trade points"),
166            button("configure", "configure", "_cmbtpl configure dialog"),
167        ] {
168            ashryn.state.apply(&frame);
169        }
170        let sent = Arc::new(Mutex::new(Vec::new()));
171        let heard = Arc::clone(&sent);
172        let story = story();
173        let widget = Widget::Dialog("BetrayerPanel".to_owned());
174        assert_eq!(widget.name(), "Betrayer panel");
175        let mut harness = Harness::builder()
176            .with_size((300.0, 200.0))
177            .build_ui(move |ui| {
178                let seen = Seen {
179                    snapshot: Some(&ashryn),
180                    story: &story,
181                    who: None,
182                    open: &[],
183                    tags: None,
184                    minimap: None,
185                    travel: None,
186                    trainer: None,
187                    hydra: true,
188                };
189                let clicked = widget.draw(ui, &seen, Id::new("dialog"));
190                let Some(crate::widget::Clicked::FromGame(line)) = clicked else {
191                    return;
192                };
193                heard
194                    .lock()
195                    .unwrap_or_else(std::sync::PoisonError::into_inner)
196                    .push(line);
197            });
198        harness.run();
199        harness.get_by_label("Betrayer");
200        harness.get_by_label("Blood Points: 1");
201        harness.get_by_label("trade").click();
202        harness.run();
203        harness.get_by_label("configure").click();
204        harness.run();
205        assert_eq!(
206            *sent
207                .lock()
208                .unwrap_or_else(std::sync::PoisonError::into_inner),
209            ["trade points", "_cmbtpl configure dialog"]
210        );
211    }

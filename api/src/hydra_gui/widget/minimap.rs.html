1//! The minimap: the area the character is in, laid out by hydra-mapper's
2//! engine (`hydra_map`), drawn in the theme's map colours and following
3//! you, with the maps next door drawn dimmed beside it. Its route is in
4//! [`route`]: what `.route2` shows. Aboard a ship it is the ocean's chart
5//! ([`ocean`]).
6//!
7//! A pure painter over a [`MapScene`] and a camera kept in egui's memory
8//! for the widget. The camera follows you, moving only when you leave a dead
9//! zone in the middle of the view, or, kept centred on its page, with every
10//! step; the wheel zooms about the pointer, a drag pans and stops the
11//! following, and a double-click comes back to you. In a building, the
12//! building is drawn opened on its area's sheet where its door was, the rest
13//! dimmed, and the camera zooms in: two zooms, outside and in, each back to
14//! its default at a door, and a glide between. Resetting to the default at a
15//! door keeps the view predictable. Walking into a map drawn next door, the
16//! view moves by where that map was drawn, so nothing jumps.
17//!
18//! Under the pointer, a room's card: its title, number, building and ways
19//! out. Shift+click says the room in the story, and Ctrl+click its number
20//! and title. A click on a room shows the way there (`.route2`, in a line),
21//! and a click on the goal it rings walks it (`.go2`); a click on nothing
22//! takes the way off (right-click is the window's menu). A way-in dot stands
23//! for the place behind it. The zooms and how many maps out are the widget's
24//! own page's ([`look`]). On the ocean's chart a click on a port shows the
25//! course there (`.sail route`), and a click on the port the course rings
26//! sails it (`.sail`).

34/// Pixels a cell outdoors at first, and in a place, whose rooms are drawn
35/// at half the streets' scale: zoomed in so they are no more crowded than
36/// the streets are; and the least and most either may be zoomed to.
37const ZOOM: f32 = 7.0;
38const ZOOM_INSIDE: f32 = 16.0;
39const ZOOM_RANGE: (f32, f32) = (2.0, 40.0);
40/// Seconds a glide takes.
41const GLIDE: f64 = 0.25;
42/// How much of its colour what is outside the place you are in keeps.
43const FADED: f32 = 0.35;
44/// How far from the middle you may go, as a share of the view each way,
45/// before the camera moves (24 x 16 of an 80 x 56 view).
46const DEAD_ZONE: f32 = 0.3;
47
48/// The camera, kept per widget between frames.
49#[derive(Clone, Copy, Debug)]
50struct Camera {
51    /// The cell in the middle of the view.
52    centre: Vec2,
53    /// Pixels a cell.
54    zoom: f32,
55    /// Whether it follows you; a drag stops it until a double-click.
56    follow: bool,
57    /// The area it was last on: another area recentres.
58    area: u64,
59    /// Whether you were in a place: a door crossed zooms to the other
60    /// side's default.
61    inside: bool,
62    /// A glide under way to `centre` and `zoom`: where it started, and
63    /// when.
64    glide: Option<(Vec2, f32, f64)>,
65}
66
67impl Camera {
68    /// Where the view is this frame: part way along a glide, or where it
69    /// is going.
70    fn drawn(&self, now: f64) -> (Vec2, f32) {
71        let Some((from, zoom, start)) = self.glide else {
72            return (self.centre, self.zoom);
73        };
74        #[allow(clippy::cast_possible_truncation)]
75        let t = (((now - start) / GLIDE).clamp(0.0, 1.0)) as f32;
76        let eased = t * t * 2.0f32.mul_add(-t, 3.0);
77        (
78            from + (self.centre - from) * eased,
79            zoom + (self.zoom - zoom) * eased,
80        )
81    }
82
83    /// Glide from where the view is now to where it is going.
84    fn glide_from(&mut self, drawn: (Vec2, f32), now: f64) {
85        self.glide = Some((drawn.0, drawn.1, now));
86    }
87}

89/// Draw `view` into all the space `ui` gives, and what a click on it asked
90/// for; `own` is false on another character's minimap, which only shows.
91pub(super) fn minimap(
92    ui: &mut egui::Ui,
93    view: Option<&MinimapView>,
94    id: Id,
95    own: bool,
96    look: MinimapLook,
97) -> Option<Clicked> {
98    let size = ui.available_size().max(vec2(80.0, 60.0));
99    let (rect, response) = ui.allocate_exact_size(size, Sense::click_and_drag());
100    let painter = ui.painter_at(rect);
101    let ground = theme::color(ui.ctx(), T::MapBackground);
102    painter.rect_filled(rect, theme::corner(ui.ctx()), ground);
103    if let ocean::Chart::Drawn(asked) = ocean::shown(ui, &response, rect, (id, look), view) {
104        return asked.filter(|_| own);
105    }
106    let Some(MinimapView::Here {
107        scene,
108        room: you,
109        next_door,
110        route,
111    }) = view
112    else {
113        let why = match view {
114            Some(MinimapView::Waiting(why)) => why.as_str(),
115            _ => "No map.",
116        };
117        waiting(&painter, rect, why);
118        return None;
119    };
120    let (scene, you) = (scene.as_ref(), *you);
121    let Some((here_cell, inside)) = whereabouts(scene, you) else {
122        waiting(&painter, rect, "You are not on this area's map.");
123        return None;
124    };
125    // The maps next door as many out as its page says.
126    let next_door: Vec<NextDoor> = next_door
127        .iter()
128        .filter(|n| n.ring <= look.maps_out)
129        .cloned()
130        .collect();
131    let next_door = next_door.as_slice();
132    let default = if inside.is_some() {
133        look.inside
134    } else {
135        look.outside
136    };
137    let camera = aimed(
138        ui,
139        id,
140        &response,
141        rect,
142        (scene, next_door),
143        here_cell,
144        (inside.is_some(), default, look.centred),
145    );
146    let now = ui.input(|i| i.time);
147    let (centre, zoom) = camera.drawn(now);
148    if camera.glide.is_some_and(|(.., start)| now - start < GLIDE) {
149        ui.ctx().request_repaint();
150    }
151
152    let to_screen = |p: Vec2| rect.center() + (p - centre) * zoom;
153    // The maps next door first, dimmed, under this one.
154    let mut layers: Vec<(&MapScene, Vec2)> = vec![(scene, Vec2::ZERO)];
155    layers.extend(
156        next_door
157            .iter()
158            .map(|n| (n.scene.as_ref(), vec2(n.offset.0, n.offset.1))),
159    );
160    let square = (zoom * 0.7).clamp(3.0, 10.0);
161    for &(next, offset) in &layers[1..] {
162        let moved = |p: Vec2| to_screen(p + offset);
163        draw_lines(&painter, next, &moved, zoom, &|_| FADED);
164        draw_rooms(&painter, rect, next, &moved, zoom, &|_| FADED);
165        marks::draw_marks(&painter, next, &moved, zoom, &|_| FADED);
166    }
167    // In a place, the rest of the sheet dimmed; outdoors, everything.
168    let fade = |building: Option<usize>| {
169        if inside.is_none() || building == inside {
170            1.0
171        } else {
172            FADED
173        }
174    };
175    draw_lines(&painter, scene, &to_screen, zoom, &fade);
176    draw_rooms(&painter, rect, scene, &to_screen, zoom, &fade);
177    marks::draw_marks(&painter, scene, &to_screen, zoom, &fade);
178    marks::draw_labels(&painter, scene, &to_screen, zoom, &fade);
179    if let Some(route) = route {
180        route::draw_route(
181            &painter,
182            rect,
183            &layers,
184            route,
185            you,
186            &to_screen,
187            (zoom, look.flow),
188        );
189    }
190    painter.circle_stroke(
191        to_screen(here_cell),
192        square * 0.9 + 3.0,
193        Stroke::new(2.5, theme::color(ui.ctx(), T::MapYou)),
194    );
195    let from_screen = |p: Pos2| centre + (p - rect.center()) / zoom;
196    let reach = (square * 0.9 + 3.0) / zoom;
197    let pointed = |at: Pos2| hit(&layers, from_screen(at), reach);
198    let goal = route::goal(route.as_deref(), you);
199    pointer(ui, &response, &layers, &pointed, (own, goal))
200}

202/// The card for the room under the pointer, on anyone's minimap; and, on
203/// the character's `own`, what a click asked, `goal` the room the way shown
204/// rings.
205fn pointer(
206    ui: &egui::Ui,
207    response: &egui::Response,
208    layers: &[(&MapScene, Vec2)],
209    pointed: &dyn Fn(Pos2) -> Option<u32>,
210    (own, goal): (bool, Option<u32>),
211) -> Option<Clicked> {
212    marks::hover(ui, response, pointed, layers);
213    if !own {
214        return None;
215    }
216    let at = response.interact_pointer_pos().and_then(pointed);
217    clicked(ui, response, at, goal)
218}

220/// One sheet's lines, each as dim as `fade` says for the building it is in.
221pub(crate) fn draw_lines(
222    painter: &egui::Painter,
223    scene: &MapScene,
224    to_screen: &dyn Fn(Vec2) -> Pos2,
225    zoom: f32,
226    fade: &dyn Fn(Option<usize>) -> f32,
227) {
228    for edge in &scene.edges {
229        draw_edge(painter, edge, to_screen, zoom, fade(edge.building));
230    }
231}

233/// One sheet's rooms and way-in dots, over its lines.
234pub(crate) fn draw_rooms(
235    painter: &egui::Painter,
236    rect: Rect,
237    scene: &MapScene,
238    to_screen: &dyn Fn(Vec2) -> Pos2,
239    zoom: f32,
240    fade: &dyn Fn(Option<usize>) -> f32,
241) {
242    let square = (zoom * 0.7).clamp(3.0, 10.0);
243    for room in &scene.rooms {
244        let at = to_screen(cell(room.cell));
245        if rect.expand(square).contains(at) {
246            let f = fade(room.building);
247            painter.rect(
248                Rect::from_center_size(at, Vec2::splat(square)),
249                1.4,
250                theme::color(painter.ctx(), T::MapRoom).gamma_multiply(f),
251                Stroke::new(
252                    1.0,
253                    theme::color(painter.ctx(), T::MapRoomEdge).gamma_multiply(f),
254                ),
255                egui::StrokeKind::Inside,
256            );
257        }
258    }
259    draw_doors(painter, scene, to_screen, zoom, fade(None));
260}

262/// The camera for this frame, kept for the next: a new area recentres at
263/// once; a door crossed, into a place or out, glides to that side's default
264/// zoom; following you glides along, keeping you `centred` or within the
265/// dead zone; the wheel and a drag steer it.
266fn aimed(
267    ui: &egui::Ui,
268    id: Id,
269    response: &egui::Response,
270    rect: Rect,
271    (scene, next_door): (&MapScene, &[NextDoor]),
272    here: Vec2,
273    (inside, default, centred): (bool, f32, bool),
274) -> Camera {
275    let now = ui.input(|i| i.time);
276    // A place is opened on its area's sheet: its sheet is named for the
277    // area and the place (`area@room`), and is the same area.
278    let area_of =
279        |scene: &MapScene| egui::util::hash(scene.area.split('@').next().unwrap_or(&scene.area));
280    let area = area_of(scene);
281    let fresh = Camera {
282        centre: here,
283        zoom: default,
284        follow: true,
285        area,
286        inside,
287        glide: None,
288    };
289    let mut camera = ui.data(|d| d.get_temp::<Camera>(id)).unwrap_or(fresh);
290    // Into a map that was drawn next door: the view moves by where that
291    // map was drawn, so nothing jumps.
292    let crossed = next_door
293        .iter()
294        .find(|n| area_of(&n.scene) == camera.area)
295        .map(|n| vec2(n.offset.0, n.offset.1));
296    if camera.area != area {
297        match crossed {
298            Some(shift) => {
299                camera.area = area;
300                camera.centre += shift;
301                camera.glide = camera
302                    .glide
303                    .map(|(from, zoom, start)| (from + shift, zoom, start));
304            }
305            None => camera = fresh,
306        }
307    }
308    if camera.inside != inside {
309        let was = camera.drawn(now);
310        camera = Camera {
311            glide: None,
312            ..fresh
313        };
314        camera.glide_from(was, now);
315    }
316    let before = (camera.centre, camera.zoom);
317    steer(ui, response, rect, &mut camera);
318    if (camera.centre, camera.zoom) != before {
319        // Steered by hand: no glide to fight it.
320        camera.glide = None;
321    } else if camera.follow {
322        let was = camera.drawn(now);
323        let dead_zone = if centred { 0.0 } else { DEAD_ZONE };
324        keep_in_view(&mut camera, here, rect.size(), dead_zone);
325        if camera.centre != before.0 {
326            camera.glide_from(was, now);
327        }
328    }
329    ui.data_mut(|d| d.insert_temp(id, camera));
330    camera
331}

333/// How long a click waits for a second, which would make it the
334/// double-click that comes back to you: in seconds.
335const DOUBLE_CLICK: f64 = 0.3;
336
337/// A click held until no second one came: what it asks, and when.
338#[derive(Clone)]
339struct Held {
340    at: f64,
341    asks: Option<Clicked>,
342}
343
344/// What a click on the room `pointed` at asks, once [`DOUBLE_CLICK`] has
345/// passed with no second click: a double-click comes back to you, and
346/// neither of its clicks is a click of its own -- or the first, on the
347/// room the way shown rings, would walk there.
348fn clicked(
349    ui: &egui::Ui,
350    response: &egui::Response,
351    pointed: Option<u32>,
352    goal: Option<u32>,
353) -> Option<Clicked> {
354    let id = response.id.with("held click");
355    let now = ui.input(|i| i.time);
356    if response.double_clicked() {
357        ui.data_mut(|d| d.remove::<Held>(id));
358        return None;
359    }
360    let wait = std::time::Duration::from_secs_f64(DOUBLE_CLICK);
361    if response.clicked() {
362        let asks = click_asks(ui, pointed, goal);
363        ui.data_mut(|d| d.insert_temp(id, Held { at: now, asks }));
364        ui.ctx().request_repaint_after(wait);
365        return None;
366    }
367    let held = ui.data(|d| d.get_temp::<Held>(id))?;
368    if now - held.at < DOUBLE_CLICK {
369        ui.ctx().request_repaint_after(wait);
370        return None;
371    }
372    ui.data_mut(|d| d.remove::<Held>(id));
373    held.asks
374}

376/// What a click on the room `pointed` at asks, as the modifiers held say:
377/// Shift, the room in the story; Ctrl, its number and title. A plain one
378/// shows the way there, or walks it on the `goal` the way shown rings; on
379/// nothing, it takes the way shown off.
380fn click_asks(ui: &egui::Ui, pointed: Option<u32>, goal: Option<u32>) -> Option<Clicked> {
381    let modifiers = ui.input(|i| i.modifiers);
382    match pointed {
383        Some(id) if modifiers.shift => Some(Clicked::Room { id, brief: false }),
384        Some(id) if modifiers.command => Some(Clicked::Room { id, brief: true }),
385        _ if modifiers.any() => None,
386        Some(id) if goal == Some(id) => Some(Clicked::Command(format!("go2 {id}"))),
387        Some(id) => Some(Clicked::Command(format!("route2 {id} --brief"))),
388        None if goal.is_some() => Some(Clicked::Command("route2 clear".to_owned())),
389        None => None,
390    }
391}

393/// The room drawn nearest `at` within `reach` cells on any of `layers`
394/// (each sheet and where it is drawn), or the room behind the way-in dot
395/// there.
396pub(crate) fn hit(layers: &[(&MapScene, Vec2)], at: Vec2, reach: f32) -> Option<u32> {
397    layers
398        .iter()
399        .flat_map(|&(scene, offset)| {
400            let rooms = scene
401                .rooms
402                .iter()
403                .map(move |r| (r.id, cell(r.cell) + offset));
404            let doors = scene
405                .doors
406                .iter()
407                .map(move |d| (d.inside, vec2(d.at.0, d.at.1) + offset));
408            rooms.chain(doors)
409        })
410        .map(|(id, p)| (id, (p - at).length()))
411        .filter(|&(_, d)| d <= reach)
412        .min_by(|a, b| a.1.total_cmp(&b.1))
413        .map(|(id, _)| id)
414}

416/// Where you are drawn, and the building you are in: your room, or, for a
417/// room the layout draws only as dots at the rooms it is entered from (the
418/// Issenflow's current), the first of those dots.
419fn whereabouts(scene: &MapScene, you: u32) -> Option<(Vec2, Option<usize>)> {
420    if let Some(here) = scene.room(you) {
421        return Some((cell(here.cell), here.building));
422    }
423    let door = scene.doors.iter().find(|d| d.inside == you)?;
424    Some((vec2(door.at.0, door.at.1), None))
425}

427/// Wheel zoom about the pointer, drag to pan, double-click back to you.
428fn steer(ui: &egui::Ui, response: &egui::Response, rect: Rect, camera: &mut Camera) {
429    if response.dragged() {
430        camera.centre -= response.drag_delta() / camera.zoom;
431        camera.follow = false;
432    }
433    if response.double_clicked() {
434        camera.follow = true;
435    }
436    if response.hovered() {
437        let scroll = ui.input(|i| i.smooth_scroll_delta.y);
438        if scroll != 0.0 {
439            let pointer = ui.input(|i| i.pointer.hover_pos()).unwrap_or(rect.center());
440            let before = camera.centre + (pointer - rect.center()) / camera.zoom;
441            camera.zoom = (camera.zoom * (scroll / 200.0).exp()).clamp(ZOOM_RANGE.0, ZOOM_RANGE.1);
442            camera.centre = before - (pointer - rect.center()) / camera.zoom;
443        }
444    }
445}

447/// Move the camera only as far as keeps `you` within the middle of a view
448/// of `size`.
449fn keep_in_view(camera: &mut Camera, you: Vec2, size: Vec2, dead_zone: f32) {
450    let half = size / camera.zoom * dead_zone;
451    let off = you - camera.centre;
452    if off.x.abs() > half.x {
453        camera.centre.x += off.x - half.x * off.x.signum();
454    }
455    if off.y.abs() > half.y {
456        camera.centre.y += off.y - half.y * off.y.signum();
457    }
458}

460fn draw_edge(
461    painter: &egui::Painter,
462    edge: &SceneEdge,
463    to_screen: &dyn Fn(Vec2) -> Pos2,
464    zoom: f32,
465    fade: f32,
466) {
467    let (line, connector) = (
468        theme::color(painter.ctx(), T::MapLine).gamma_multiply(fade),
469        theme::color(painter.ctx(), T::MapConnector).gamma_multiply(fade),
470    );
471    let points: Vec<Pos2> = edge
472        .path
473        .iter()
474        .map(|&(x, y)| to_screen(vec2(x, y)))
475        .collect();
476    match edge.kind {
477        EdgeKind::Directional => {
478            painter.add(Shape::line(points, Stroke::new(1.35, line)));
479        }
480        EdgeKind::Connector => {
481            painter.extend(Shape::dashed_line(
482                &points,
483                Stroke::new(1.0, line.gamma_multiply(0.7)),
484                5.0,
485                4.0,
486            ));
487        }
488        EdgeKind::Stub => {
489            // A short mark out of each end toward the other.
490            let (Some(&a), Some(&b)) = (points.first(), points.last()) else {
491                return;
492            };
493            let length = (a.distance(b) * 0.2).min(15.0).min(zoom * 2.0);
494            let toward = (b - a).normalized() * length;
495            let stroke = Stroke::new(1.0, connector.gamma_multiply(1.6));
496            painter.extend(Shape::dashed_line(&[a, a + toward], stroke, 3.0, 2.0));
497            painter.extend(Shape::dashed_line(&[b, b - toward], stroke, 3.0, 2.0));
498        }
499    }
500}

502fn draw_doors(
503    painter: &egui::Painter,
504    scene: &MapScene,
505    to_screen: &dyn Fn(Vec2) -> Pos2,
506    zoom: f32,
507    fade: f32,
508) {
509    let colour = theme::color(painter.ctx(), T::MapDoor).gamma_multiply(fade);
510    for door in &scene.doors {
511        let at = to_screen(vec2(door.at.0, door.at.1));
512        let r = if door.named { 3.5 } else { 2.2 } * (zoom / ZOOM).clamp(0.6, 1.6);
513        painter.circle_filled(at, r, colour);
514        if door.named {
515            painter.text(
516                at + vec2(r + 3.0, 0.0),
517                egui::Align2::LEFT_CENTER,
518                &door.place,
519                FontId::proportional(11.0),
520                colour,
521            );
522        }
523    }
524}

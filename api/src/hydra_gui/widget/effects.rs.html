1//! The effect lists, with the character's tracked effects (`crate::tracked`):
2//! the four lists (Active spells, Buffs, Debuffs, Cooldowns), the Missing
3//! widget, and the Timers panel that merges them.
4//!
5//! A tracked row wears its colour and label, and its name runs its command
6//! when clicked. A row that is a spell wears its circle's colour
7//! (`super::circles`) when it has no colour of its own. A tracked effect with missing on that is not up is listed in
8//! an amber Missing box at the top. Right-click a row to track it, untrack
9//! it, or change what it carries; *Track all listed* tracks every row.

25/// One row of a list: an effect that is up, and its time left. A timer of
26/// the player's own is a row too, made an effect of its list.
27struct Up<'a> {
28    category: Category,
29    effect: Cow<'a, Effect>,
30    left: Option<u32>,
31}
32
33/// The player's own timers in `list` that are running at the game's
34/// second `now`: each as an effect of the list `game_id`, filled by how
35/// much of it is left, and its seconds left.
36fn own_timers(state: &GameState, list: List, game_id: &str) -> Vec<(Effect, u32)> {
37    let Some(now) = state.game_time_now() else {
38        return Vec::new();
39    };
40    state
41        .custom_timers
42        .timers()
43        .filter(|(_, timer)| timer.list == list)
44        .filter_map(|(name, timer)| {
45            let left = timer.ends.checked_sub(now).filter(|left| *left > 0)?;
46            let whole = timer
47                .set_at
48                .and_then(|set_at| timer.ends.checked_sub(set_at))
49                .filter(|whole| *whole >= left);
50            let percent = whole.map_or(100, |whole| left * 100 / whole);
51            Some((
52                Effect {
53                    category: game_id.to_owned(),
54                    text: name.to_owned(),
55                    ends_at: Some(timer.ends),
56                    percent,
57                },
58                left,
59            ))
60        })
61        .collect()
62}

64/// The effects of `category` that are up, one per name: the one with the
65/// most time left, in the game's order.
66fn up(state: &GameState, category: Category) -> Vec<Up<'_>> {
67    let now = state.game_time_now();
68    let mut rows: Vec<Up<'_>> = Vec::new();
69    for (id, effect) in state.effects.in_category(category.game_id()) {
70        // Still listed, but run out by the clock: down, as Missing has it.
71        if now.is_some_and(|now| state.effects.active(id, now) == Some(false)) {
72            continue;
73        }
74        let left = now.and_then(|now| state.effects.remaining(id, now));
75        match rows.iter_mut().find(|row| row.effect.text == effect.text) {
76            Some(row) if left > row.left => {
77                row.effect = Cow::Borrowed(effect);
78                row.left = left;
79            }
80            Some(_) => {}
81            None => rows.push(Up {
82                category,
83                effect: Cow::Borrowed(effect),
84                left,
85            }),
86        }
87    }
88    // The player's own timers filed with this list, after the game's.
89    let list = List::named(category.game_id()).unwrap_or_default();
90    for (effect, left) in own_timers(state, list, category.game_id()) {
91        rows.push(Up {
92            category,
93            effect: Cow::Owned(effect),
94            left: Some(left),
95        });
96    }
97    rows
98}

100/// One of the four lists: its Missing box, then its rows, a bar each.
101pub(super) fn list(
102    ui: &mut egui::Ui,
103    category: Category,
104    state: Option<&GameState>,
105    (tracked, circles): (Option<&TrackedEffects>, Option<&Circles>),
106) -> Option<Clicked> {
107    let Some(state) = state else {
108        ui.weak(format!("{} unknown", category.name()));
109        return None;
110    };
111    let mut clicked = None;
112    if let Some(tracked) = tracked {
113        missing_box(ui, state, &tracked.missing(category, state), &mut clicked);
114    }
115    let rows = up(state, category);
116    let names: Vec<String> = rows.iter().map(|row| row.effect.text.clone()).collect();
117    for row in &rows {
118        if let Some(asked) = bar(ui, row, state, (tracked, circles), &names, None) {
119            clicked = Some(asked);
120        }
121    }
122    if rows.is_empty() {
123        ui.weak("None.");
124    }
125    clicked
126}

128/// The Missing widget: every tracked effect with missing on that is not up,
129/// in every list.
130pub(super) fn missing(
131    ui: &mut egui::Ui,
132    state: Option<&GameState>,
133    tracked: Option<&TrackedEffects>,
134) -> Option<Clicked> {
135    let (Some(state), Some(tracked)) = (state, tracked) else {
136        ui.weak("Missing unknown");
137        return None;
138    };
139    let mut clicked = None;
140    let mut any = false;
141    for category in Category::ALL {
142        for entry in tracked.missing(category, state) {
143            any = true;
144            missing_row(ui, state, entry, &mut clicked);
145        }
146    }
147    if !any {
148        ui.weak("Nothing missing.");
149    }
150    clicked
151}

192/// The Timers panel: the lists it merges, each row marked with its list's
193/// glyph, and the Missing box when it is on.
194pub(super) fn timers(
195    ui: &mut egui::Ui,
196    state: Option<&GameState>,
197    (tracked, circles): (Option<&TrackedEffects>, Option<&Circles>),
198    look: &TimersLook,
199) -> Option<Clicked> {
200    let Some(state) = state else {
201        ui.weak("Timers unknown");
202        return None;
203    };
204    let shown: Vec<Category> = Category::ALL
205        .into_iter()
206        .filter(|category| look.shows(*category))
207        .collect();
208    let mut clicked = None;
209    if look.missing
210        && let Some(tracked) = tracked
211    {
212        let gone: Vec<&Tracked> = shown
213            .iter()
214            .flat_map(|category| tracked.missing(*category, state))
215            .collect();
216        missing_box(ui, state, &gone, &mut clicked);
217    }
218    let mut any = false;
219    if look.custom {
220        for (effect, left) in own_timers(state, List::Custom, "") {
221            any = true;
222            own_bar(ui, &effect, left);
223        }
224    }
225    for category in shown {
226        let rows = up(state, category);
227        let names: Vec<String> = rows.iter().map(|row| row.effect.text.clone()).collect();
228        for row in rows.iter().filter(|row| {
229            !look.only_tracked
230                || tracked.is_some_and(|tracked| tracked.find(category, &row.effect.text).is_some())
231        }) {
232            any = true;
233            let mark = Some(glyph(category));
234            if let Some(asked) = bar(ui, row, state, (tracked, circles), &names, mark) {
235                clicked = Some(asked);
236            }
237        }
238    }
239    if !any {
240        ui.weak("None.");
241    }
242    clicked
243}

245/// One of the player's own custom timers in the Timers panel: its mark,
246/// its name and time left, and a bar of how much is left.
247fn own_bar(ui: &mut egui::Ui, effect: &Effect, left: u32) {
248    ui.horizontal(|ui| {
249        ui.label(RichText::new("◆").color(theme::color(ui.ctx(), T::Accent)));
250        ui.add(
251            Bar::new(
252                &format!("{} {}", effect.text, clock(left)),
253                Some(Amount {
254                    percent: effect.percent,
255                    current: None,
256                    max: None,
257                }),
258            )
259            .fill(theme::color(ui.ctx(), T::Accent))
260            .size([ui.available_width(), 16.0])
261            .says(Says {
262                label: true,
263                numbers: false,
264                percent: false,
265                words: false,
266            }),
267        );
268    });
269}

281/// One effect that is up, as a bar: its label or name and time left, in its
282/// colour, its spell's circle's, or its list's; a click runs its command, a
283/// right-click its menu.
284fn bar(
285    ui: &mut egui::Ui,
286    row: &Up<'_>,
287    state: &GameState,
288    (tracked, circles): (Option<&TrackedEffects>, Option<&Circles>),
289    names: &[String],
290    mark: Option<(&str, T)>,
291) -> Option<Clicked> {
292    let entry = tracked.and_then(|tracked| tracked.find(row.category, &row.effect.text));
293    let name = entry
294        .and_then(|entry| entry.label.as_deref())
295        .unwrap_or(&row.effect.text);
296    let label = match row.left {
297        Some(seconds) => format!("{name} {}", clock(seconds)),
298        None => name.to_owned(),
299    };
300    let fill = entry
301        .and_then(|entry| entry.color)
302        .or_else(|| circles.and_then(|circles| circles::color(circles, &row.effect.text)))
303        .map_or_else(
304            || theme::color(ui.ctx(), row.category.token()),
305            |[r, g, b]| Color32::from_rgb(r, g, b),
306        );
307    let amount = Amount {
308        percent: row.effect.percent,
309        current: None,
310        max: None,
311    };
312    let mut clicked = None;
313    let mut row = |ui: &mut egui::Ui| {
314        if let Some((glyph, token)) = mark {
315            ui.label(RichText::new(glyph).color(theme::color(ui.ctx(), token)));
316        }
317        let drawn = ui.add(
318            Bar::new(&label, Some(amount))
319                .fill(fill)
320                .size([ui.available_width(), 16.0])
321                .says(Says {
322                    label: true,
323                    numbers: false,
324                    percent: false,
325                    words: false,
326                }),
327        );
328        if tracked.is_none() {
329            return;
330        }
331        let command = entry.and_then(|entry| tracked::command(entry, state));
332        let click = ui.interact(
333            drawn.rect,
334            drawn.id.with(("effect", row.category, &row.effect.text)),
335            Sense::click(),
336        );
337        let click = match &command {
338            Some(command) => click.on_hover_text(format!("Click: {command}")),
339            None => click,
340        };
341        if click.clicked()
342            && let Some(command) = command
343        {
344            clicked = Some(Clicked::Send(command));
345        }
346        click.context_menu(|ui| {
347            if let Some(change) = row_menu(ui, row.category, &row.effect.text, entry, names) {
348                clicked = Some(Clicked::Track(change));
349            }
350        });
351    };
352    // Only the Timers panel marks its rows: a list's bars stay one under
353    // another, spaced as a list's are.
354    if mark.is_some() {
355        ui.horizontal(|ui| row(ui));
356    } else {
357        row(ui);
358    }
359    clicked
360}

362/// The amber Missing box: each tracked effect not up, its name a button
363/// when it has something to send.
364fn missing_box(
365    ui: &mut egui::Ui,
366    state: &GameState,
367    gone: &[&Tracked],
368    clicked: &mut Option<Clicked>,
369) {
370    if gone.is_empty() {
371        return;
372    }
373    let warning = theme::color(ui.ctx(), T::Warning);
374    egui::Frame::new()
375        .stroke(egui::Stroke::new(1.0, warning))
376        .inner_margin(4.0)
377        .show(ui, |ui| {
378            ui.label(RichText::new("Missing").small().color(warning));
379            for entry in gone {
380                missing_row(ui, state, entry, clicked);
381            }
382        });
383}

385/// One missing effect: its label or name, a link when clicking sends
386/// something, and its menu on a right-click.
387fn missing_row(
388    ui: &mut egui::Ui,
389    state: &GameState,
390    entry: &Tracked,
391    clicked: &mut Option<Clicked>,
392) {
393    let name = entry.label.as_deref().unwrap_or(&entry.name);
394    let color = entry.color.map_or_else(
395        || theme::color(ui.ctx(), T::Warning),
396        |[r, g, b]| Color32::from_rgb(r, g, b),
397    );
398    let response = match tracked::command(entry, state) {
399        Some(command) => {
400            let link = ui
401                .link(RichText::new(name).color(color))
402                .on_hover_text(format!("Click: {command}"));
403            if link.clicked() {
404                *clicked = Some(Clicked::Send(command));
405            }
406            link
407        }
408        None => ui.label(RichText::new(name).color(color)),
409    };
410    response.context_menu(|ui| {
411        if let Some(change) = row_menu(ui, entry.category, &entry.name, Some(entry), &[]) {
412            *clicked = Some(Clicked::Track(change));
413        }
414    });
415}

417/// A row's right-click menu: track or untrack it, and for a tracked one, its
418/// Missing, colour, label and command; and track every row listed.
419fn row_menu(
420    ui: &mut egui::Ui,
421    category: Category,
422    name: &str,
423    entry: Option<&Tracked>,
424    names: &[String],
425) -> Option<Change> {
426    let effect = name.to_owned();
427    let mut change = None;
428    match entry {
429        None => {
430            if ui.button("Track").clicked() {
431                change = Some(Change::Track(category, effect.clone()));
432                ui.close();
433            }
434        }
435        Some(entry) => {
436            if ui.button("Untrack").clicked() {
437                change = Some(Change::Untrack(category, effect.clone()));
438                ui.close();
439            }
440            let mut missing = entry.missing;
441            if ui.checkbox(&mut missing, "List when missing").changed() {
442                change = Some(Change::Missing(category, effect.clone(), missing));
443            }
444            ui.horizontal(|ui| {
445                let mut rgb = entry.color.unwrap_or([0x80, 0x80, 0x80]);
446                ui.label("Colour");
447                if ui.color_edit_button_srgb(&mut rgb).changed() {
448                    change = Some(Change::Color(category, effect.clone(), Some(rgb)));
449                }
450                if entry.color.is_some() && ui.small_button("Its list's").clicked() {
451                    change = Some(Change::Color(category, effect.clone(), None));
452                }
453            });
454            if let Some(typed) = typed(
455                ui,
456                ("label", category, name),
457                "Label",
458                entry.label.as_deref(),
459            ) {
460                change = Some(Change::Label(category, effect.clone(), Some(typed)));
461            }
462            if let Some(typed) = typed(
463                ui,
464                ("command", category, name),
465                "Command",
466                entry.command.as_deref(),
467            ) {
468                change = Some(Change::Command(category, effect.clone(), Some(typed)));
469            }
470        }
471    }
472    if !names.is_empty() {
473        ui.separator();
474        if ui.button("Track all listed").clicked() {
475            change = Some(Change::TrackAll(category, names.to_vec()));
476            ui.close();
477        }
478    }
479    change
480}

482/// A field in a menu, with what it holds now; what was typed, once Enter is
483/// pressed or the field is left changed.
484fn typed(
485    ui: &mut egui::Ui,
486    key: impl std::hash::Hash + std::fmt::Debug,
487    label: &str,
488    now: Option<&str>,
489) -> Option<String> {
490    let id = egui::Id::new(("tracked-field", key));
491    let now = now.unwrap_or_default().to_owned();
492    let mut text = ui
493        .data_mut(|data| data.get_temp::<String>(id))
494        .unwrap_or_else(|| now.clone());
495    let mut done = None;
496    let mut editing = false;
497    ui.horizontal(|ui| {
498        ui.label(label);
499        let field = ui.add(egui::TextEdit::singleline(&mut text).desired_width(140.0));
500        editing = field.has_focus();
501        if field.lost_focus() && text != now {
502            done = Some(text.trim().to_owned());
503        }
504    });
505    if editing {
506        ui.data_mut(|data| data.insert_temp(id, text));
507    } else {
508        ui.data_mut(|data| data.remove::<String>(id));
509    }
510    done
511}

532/// Whether `key` is on in `look`: a list's key is its category's game id.
533fn is_on(look: &TimersLook, key: &str) -> Option<bool> {
534    match key {
535        "only_tracked" => Some(look.only_tracked),
536        "missing" => Some(look.missing),
537        "custom" => Some(look.custom),
538        _ => Category::ALL
539            .into_iter()
540            .find(|category| category.game_id() == key)
541            .map(|category| look.shows(category)),
542    }
543}

545/// The Timers panel's page: a toggle for each list, then the switches.
546pub(crate) fn timers_rows(look: Option<&TimersLook>) -> Vec<hydra_ui::settings::Row> {
547    use hydra_ui::settings::{Row, RowKind, Value};
548    let default = TimersLook::default();
549    let now = look.cloned().unwrap_or_default();
550    let lists =
551        Category::ALL.map(|category| (category.game_id(), category.name(), "Merge this list in."));
552    lists
553        .iter()
554        .chain(SWITCHES.iter())
555        .map(|(key, label, help)| {
556            let on = is_on(&now, key).unwrap_or_default();
557            Row {
558                key: (*key).to_owned(),
559                label: (*label).to_owned(),
560                help: (*help).to_owned(),
561                kind: RowKind::Toggle,
562                value: Value::On(on),
563                here: is_on(&default, key) != Some(on),
564                from: None,
565            }
566        })
567        .collect()
568}

570/// Set the Timers panel's `key` to `to` (`on` or `off`), or its own.
571///
572/// # Errors
573///
574/// A key it does not have, or neither on nor off.
575pub(crate) fn timers_set(look: &mut TimersLook, key: &str, to: Option<&str>) -> Result<(), String> {
576    let on = match to {
577        Some("on") => true,
578        Some("off") => false,
579        Some(other) => return Err(format!("{other:?} is neither on nor off")),
580        None => is_on(&TimersLook::default(), key)
581            .ok_or_else(|| format!("the timers panel has no {key}"))?,
582    };
583    match key {
584        "only_tracked" => look.only_tracked = on,
585        "missing" => look.missing = on,
586        "custom" => look.custom = on,
587        _ => {
588            let category = Category::ALL
589                .into_iter()
590                .find(|category| category.game_id() == key)
591                .ok_or_else(|| format!("the timers panel has no {key}"))?;
592            look.lists.retain(|each| *each != category);
593            if on {
594                look.lists.push(category);
595                look.lists.sort();
596            }
597        }
598    }
599    Ok(())
600}

610    /// A state whose `category` lists `names`, each two minutes from now.
611    fn listing(lists: &[(Category, &[&str])]) -> GameState {
612        let mut state = GameState::default();
613        for (category, names) in lists {
614            state.effects.clear_category(category.game_id());
615            for name in *names {
616                state.effects.insert(
617                    format!("{}:{name}", category.game_id()),
618                    Effect {
619                        category: category.game_id().to_owned(),
620                        text: (*name).to_owned(),
621                        ends_at: None,
622                        percent: 50,
623                    },
624                );
625            }
626        }
627        state
628    }

630    fn tracking(entries: &[(Category, &str)]) -> TrackedEffects {
631        let mut tracked = TrackedEffects::default();
632        for (category, name) in entries {
633            tracked.apply(Change::Track(*category, (*name).to_owned()));
634        }
635        tracked

661    /// The state at the game's second `now`, with the player's own timers
662    /// `timers`: a name, the second it ends, its list.
663    fn timed(now: u32, timers: &[(&str, u32, &str)]) -> GameState {
664        let mut state = listing(&[(Category::Buffs, &["Heroism"])]);
665        let mut parser = hydra_protocol::Parser::new();
666        for frame in parser.parse_line(&format!("<prompt time=\"{now}\">&gt;</prompt>")) {
667            state.apply(&frame);
668        }

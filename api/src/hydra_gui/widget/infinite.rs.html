1//! The injury doll's **Infinite** style: `gs_studio`'s puppet, posed and
2//! moving as the character is.
3//!
4//! `gs_studio` takes the game's words, not Hydra's types (`injury_doll`'s
5//! `DollState::from_game`): each part's wound and scar by the injury
6//! window's id, and the names of what the character is under. So this is an
7//! adapter and nothing more. The feet it folds into the legs itself. The
8//! character is drawn as one of the playable races' figures `gs_studio`
9//! installed in Hydra's data folder (`global/bodies` and `global/puppets`:
10//! `gs-studio --install-doll <data folder>`), the one the player chose
11//! ([`races`]), the first with none chosen: a modelled body, drawn as its
12//! trained splats.
13//!
14//! Built only with the `doll-infinite` feature, on by default. Each doll is its
15//! own view of `gs_studio`'s GPU pass, named for its widget, so two characters'
16//! dolls in one frame do not draw over each other. Where there is no figure
17//! (none installed), or it cannot be drawn, the widget falls back to the Doll.

40/// The figure (its form) drawn for `chosen`: that one where it is one of
41/// `races`', else the first race's first; none where there are none.
42pub(crate) fn figure(races: &[DollRace], chosen: Option<&str>) -> Option<String> {
43    let mut all = races.iter().flat_map(|race| &race.figures);
44    let first = all.clone().next();
45    chosen
46        .and_then(|chosen| all.find(|one| one.form == chosen))
47        .or(first)
48        .map(|one| one.form.clone())
49}

51/// The way the doll's sky is seen: south, where the sun and moons stand
52/// highest.
53const SOUTH: f32 = 180.0;
54
55/// The light and background the doll stands in, as `look` asks: the game's
56/// own sky now, looking south (`now`, the server's clock in unix seconds; a
57/// steady day before it is known), `gs_studio`'s display, or the display's
58/// lights over black or one colour -- and the player's own light over it:
59/// where its key light comes from and how high (not under the sky, whose
60/// sun and moons stand where they are), and how strong. The doll view draws
61/// a flat colour dimmed (its sky at 0.55, its ground at 0.7) with a vignette
62/// toward the corners, so the colour is raised by as much first: the middle
63/// of the window shows the colour as picked.
64fn sky(look: Option<&DollLook>, now: Option<u32>) -> DollSky {
65    let look = look.cloned().unwrap_or_default();
66    let solid = |rgb: [f32; 3]| DollSky {
67        sky_top: rgb.map(|c| c / 0.55),
68        sky_horizon: rgb.map(|c| c / 0.55),
69        ground: rgb.map(|c| c / 0.7),
70        ..display()
71    };
72    let mut sky = match look.backdrop {
73        // (at the player's hour, where one is set)
74        Backdrop::Day => {
75            let hour = look.hour.map(|minutes| f32::from(minutes) / 60.0);
76            match now {
77                Some(now) => game_doll_sky_at(f64::from(now), hour, SOUTH),
78                None => doll_sky(hour, &Sky::default(), true),
79            }
80        }
81        Backdrop::Display => display(),
82        Backdrop::Black => solid([0.0; 3]),
83        Backdrop::Colour => solid(hex_linear(look.colour.as_deref().unwrap_or("#000000"))),
84    };
85    if look.backdrop != Backdrop::Day
86        && let Some(key) = sky.lights.first_mut()
87    {
88        if let Some(from) = look.light_from {
89            key.azimuth = f32::from(from);
90        }
91        if let Some(height) = look.light_height {
92            key.elevation = f32::from(height);
93        }
94    }
95    if let Some(strength) = look.light_strength {
96        let k = f32::from(strength) / 100.0;
97        for light in &mut sky.lights {
98            light.power *= k;
99        }
100    }
101    sky
102}

104/// The light a backdrop has before the player moves it: its key light's
105/// direction and height, degrees ([`display`]'s, for the backdrops that
106/// light the doll as it does).
107pub(crate) fn key_light() -> (f32, f32) {
108    display()
109        .lights
110        .first()
111        .map_or((0.0, 45.0), |key| (key.azimuth, key.elevation))
112}

114/// What the character is, as `gs_studio` reads it, and the statuses the
115/// player asked shown whatever it is under (`shown`, by their ids).
116fn doll_state(state: &GameState, shown: &[String]) -> DollState {
117    let injuries = &state.character.injuries;
118    let wounds = injuries
119        .iter()
120        .filter(|(_, injury)| injury.wound > 0)
121        .map(|(part, injury)| (part.as_str(), injury.wound));
122    let scars = injuries
123        .iter()
124        .filter(|(_, injury)| injury.scar > 0)
125        .map(|(part, injury)| (part.as_str(), injury.scar));
126    let now = state.game_time_now();
127    let statuses = state
128        .status
129        .iter()
130        .filter(|(_, on)| *on)
131        .map(|(name, _)| name)
132        .chain(
133            state
134                .effects
135                .in_category("Debuffs")
136                .filter(|(id, _)| {
137                    now.is_none_or(|now| state.effects.active_in("Debuffs", id, now) != Some(false))
138                })
139                .map(|(_, effect)| effect.text.as_str()),
140        )
141        .chain(shown.iter().map(String::as_str));
142    DollState::from_game(wounds, scars, statuses)
143}

145/// Draw the puppet for `state` in what is left of `ui`, as the doll `id`
146/// looking as `look` says, with the facing slider under it, and its
147/// injuries in words over it while it is held with the left button;
148/// `false` when it could not be, and nothing was. The slider moved = a
149/// `Clicked::Set` of the widget's `facing`.
150pub(super) fn infinite(
151    ui: &mut egui::Ui,
152    state: Option<&GameState>,
153    id: egui::Id,
154    look: Option<&DollLook>,
155) -> (bool, Option<Clicked>) {
156    let chosen = look.and_then(|look| look.figure.as_deref());
157    let Some(state) = state else {
158        return (false, None);
159    };
160    let Some(form) = figure(&races(), chosen) else {
161        return (false, None);
162    };
163    let whole = ui.available_rect_before_wrap();
164    let (rect, strip) = whole.split_top_bottom_at_y(whole.max.y - SLIDER_HEIGHT);
165    let facing = look.and_then(|look| look.facing).unwrap_or(0);
166    let context = ui.ctx().clone();
167    let form: Arc<str> = context.data_mut(|data| {
168        data.get_temp_mut_or_insert_with(egui::Id::new(("infinite-form", &form)), || {
169            Arc::<str>::from(form.as_str())
170        })
171        .clone()
172    });
173    let doll = doll_state(state, look.map_or(&[][..], |look| &look.shown));
174    let sky = sky(look, state.game_time_now());
175    let shared = DollCache::shared(&context);
176    let mut cache = shared
177        .lock()
178        .unwrap_or_else(std::sync::PoisonError::into_inner);
179    // The widget's own name, for its breathing and its turns, and its own
180    // view of the pass.
181    let name = format!("doll:{}", id.value());
182    let key = view::key_for(&doll).with_facing(f32::from(facing));
183    let drawn = view::paint_in_view(
184        ui,
185        rect,
186        &mut cache,
187        &form,
188        key,
189        &sky,
190        Some(&name),
191        gpu::available(&context),
192        id.value(),
193    );
194    drop(cache);
195    if !drawn {
196        return (false, None);
197    }
198    // Held with the left button, the puppet gives way to its injuries in
199    // words, the Text style's, while the button is down.
200    let held = ui.interact(rect, id.with("infinite-hold"), egui::Sense::drag());
201    if held.is_pointer_button_down_on() {
202        in_words(ui, rect, state);
203    }
204    // The slider: the middle faces the viewer, right turns its front to
205    // the right. Dragging sets the widget's own setting, so it is kept.
206    // Centred under the puppet, whatever the strip's width.
207    let mut deg = facing;
208    let mut child = ui.new_child(egui::UiBuilder::new().max_rect(strip));
209    let width = (strip.width() - 8.0).max(20.0);
210    child.spacing_mut().slider_width = width;
211    let slid = child
212        .horizontal(|ui| {
213            // Drawn as a slider is anywhere else, only moved over to sit
214            // in the middle of the strip.
215            ui.add_space((strip.width() - width) / 2.0);
216            ui.add(
217                egui::Slider::new(&mut deg, -180..=180)
218                    .show_value(false)
219                    .step_by(5.0),
220            )
221        })
222        .inner;
223    let slid = slid.on_hover_text(format!(
224        "Turned {deg}\u{b0}: drag to turn the doll, 0 faces you"
225    ));
226    let set = (slid.changed() && deg != facing).then(|| Clicked::Set {
227        key: "facing".to_owned(),
228        to: deg.to_string(),
229    });
230    (true, set)
231}

233/// The injuries in words over `rect`, on a dimmed sheet so they read over
234/// the puppet: what the Text style says, shown while the puppet is held.
235fn in_words(ui: &mut egui::Ui, rect: egui::Rect, state: &GameState) {
236    let sheet = ui.visuals().window_fill.gamma_multiply(0.85);
237    ui.painter().rect_filled(rect, 4.0, sheet);
238    let inner = rect.shrink(6.0);
239    let mut words = ui.new_child(
240        egui::UiBuilder::new()
241            .max_rect(inner)
242            .layout(egui::Layout::top_down(egui::Align::Min)),
243    );
244    words.set_clip_rect(inner);
245    super::doll_text::text(&mut words, Some(&state.character.injuries));
246}

248/// `gs_studio`'s GPU pass, set up once as the window opens: without it the
249/// puppet is drawn unlit on grey. And `gs_studio`'s own files kept in
250/// Hydra's data folder, not the library's default one.
251pub(crate) fn open(creation: &eframe::CreationContext<'_>, data: &std::path::Path) {
252    injury_doll::base::data_dirs::set_base_dir(data.to_path_buf());
253    if let Some(render) = &creation.wgpu_render_state {
254        gpu::init(&creation.egui_ctx, render);
255    }
256}

263    /// The doll stands in the day, the display, or the display's light
264    /// over black or the colour picked.
265    #[test]
266    fn the_doll_stands_before_the_chosen_backdrop() {
267        let with = |backdrop, colour: Option<&str>| DollLook {
268            backdrop,
269            colour: colour.map(str::to_owned),
270            ..DollLook::default()
271        };
272        assert!(!sky(None, None).indoor, "a day outdoors");
273        assert!(sky(Some(&with(Backdrop::Display, None)), None).indoor);
274        let black = sky(Some(&with(Backdrop::Black, None)), None);
275        let dark = |c: [f32; 3]| c.iter().all(|c| c.abs() < f32::EPSILON);
276        assert!(dark(black.sky_top) && dark(black.ground), "{black:?}");
277        assert!(!black.lights.is_empty(), "lit all the same");
278        let red = sky(Some(&with(Backdrop::Colour, Some("#ff0000"))), None);
279        assert!(red.sky_top[0] > 1.0 && red.sky_top[1] == 0.0, "{red:?}");
280        assert!(
281            dark(sky(Some(&with(Backdrop::Colour, None)), None).sky_top),
282            "black with no colour"
283        );
284    }

286    /// The game's sky looking south once its clock is known; the player's
287    /// light turned, raised and dimmed over the display, and only dimmed
288    /// under the sky, whose sun stands where it is.
289    #[test]
290    fn the_players_light_moves_over_the_backdrop() {
291        let noonish = 1_790_000_000;
292        let day = sky(None, Some(noonish));
293        assert!(!day.indoor && day.floor[0] > 0.0);
294        let own = DollLook {
295            backdrop: Backdrop::Display,
296            light_from: Some(90),
297            light_height: Some(10),
298            light_strength: Some(50),
299            ..DollLook::default()
300        };
301        let lit = sky(Some(&own), None);
302        let key = lit.lights.first().expect("a key light");
303        assert!((key.azimuth - 90.0).abs() < 1e-3 && (key.elevation - 10.0).abs() < 1e-3);
304        let plain = sky(
305            Some(&DollLook {
306                backdrop: Backdrop::Display,
307                ..DollLook::default()
308            }),
309            None,
310        );
311        assert!(
312            (key.power - plain.lights[0].power * 0.5).abs() < 1e-4,
313            "half as strong"
314        );
315        let under = DollLook {
316            backdrop: Backdrop::Day,
317            ..own
318        };
319        let sun = sky(Some(&under), Some(noonish));
320        for (a, b) in sun.lights.iter().zip(&day.lights) {
321            assert!(
322                (a.azimuth - b.azimuth).abs() < 1e-3,
323                "the sun stays where it is"
324            );
325            assert!((a.power - b.power * 0.5).abs() < 1e-4, "but dims");
326        }

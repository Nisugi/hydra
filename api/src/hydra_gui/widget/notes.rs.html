1//! The Notes widget: every note the player keeps, by the character that
2//! wrote each, with the one chosen shown beside the list and edited in
3//! place. A note's clipped game text keeps its links, which a click takes
4//! as the story's would; editing is of the words, and a line changed loses
5//! its links while the lines left alone keep theirs.
6//!
7//! What is typed is saved when the field loses the focus, as the settings
8//! pages save; a new note is made by its name, typed under the list.
9

20/// What a window's Notes widget shows.
21#[derive(Clone, Debug)]
22pub(crate) struct NotesView {
23    /// Where the notes file lives.
24    pub(crate) dir: PathBuf,
25    /// The window's character, who new notes are written under.
26    pub(crate) instance: String,
27    /// Its name.
28    pub(crate) name: String,
29    /// Every note, as the disk last read them.
30    pub(crate) notes: Arc<Notes>,
31}
32
33/// What the widget keeps between frames.
34#[derive(Clone, Debug, Default)]
35struct Kept {
36    /// The note shown: its writer's key and its name.
37    chosen: Option<(String, String)>,
38    /// The words being edited, while the note is.
39    draft: Option<String>,
40    /// The name typed for a new note.
41    new_name: String,
42}
43
44/// The width the list of names takes.
45const LIST_WIDTH: f32 = 150.0;
46
47/// Draw `view` into `ui`; `own` is false on another character's window,
48/// which only shows. What a click on a clipped link asked.
49pub(crate) fn notes(
50    ui: &mut egui::Ui,
51    id: Id,
52    view: Option<&NotesView>,
53    own: bool,
54) -> Option<Clicked> {
55    let Some(view) = view else {
56        ui.weak("Notes show on a character's own window.");
57        return None;
58    };
59    let kept_id = id.with("notes-kept");
60    let mut kept: Kept = ui.data(|data| data.get_temp(kept_id)).unwrap_or_default();
61    let mut clicked = None;
62    ui.horizontal_top(|ui| {
63        ui.vertical(|ui| {
64            ui.set_width(LIST_WIDTH);
65            list(ui, id, view, &mut kept, own);
66        });
67        ui.separator();
68        ui.vertical(|ui| {
69            clicked = shown(ui, id, view, &mut kept, own);
70        });
71    });
72    ui.data_mut(|data| data.insert_temp(kept_id, kept));
73    clicked
74}

76/// The names, under the character that wrote each, and the field for a
77/// new one.
78fn list(ui: &mut egui::Ui, id: Id, view: &NotesView, kept: &mut Kept, own: bool) {
79    egui::ScrollArea::vertical()
80        .id_salt(id.with("notes-list"))
81        .auto_shrink([false, true])
82        .show(ui, |ui| {
83            let mut writer: Option<&str> = None;
84            for (who, name, note) in view.notes.all() {
85                if writer != Some(who) {
86                    ui.strong(notes::character_of(who));
87                    writer = Some(who);
88                }
89                let is = kept
90                    .chosen
91                    .as_ref()
92                    .is_some_and(|(w, n)| w == who && n == name);
93                let label = ui
94                    .selectable_label(is, name)
95                    .on_hover_text(note.first_line());
96                if label.clicked() && !is {
97                    kept.chosen = Some((who.to_owned(), name.to_owned()));
98                    kept.draft = None;
99                }
100            }
101            if writer.is_none() {
102                ui.weak("No notes yet.");
103            }
104        });
105    if !own {
106        return;
107    }
108    ui.add_space(4.0);
109    let field = ui.add(
110        egui::TextEdit::singleline(&mut kept.new_name)
111            .hint_text("new note")
112            .desired_width(LIST_WIDTH - 8.0),
113    );
114    let entered = field.lost_focus() && ui.input(|input| input.key_pressed(egui::Key::Enter));
115    let name = kept.new_name.trim().to_owned();
116    if entered && !name.is_empty() && !name.contains(char::is_whitespace) {
117        let who = (view.instance.as_str(), view.name.as_str());
118        if notes::write(&view.dir, who, &name, "", &notes::stamp()).is_ok()
119            && let Some(key) = notes::key_for(who.0, who.1)
120        {
121            kept.chosen = Some((key, name));
122            kept.draft = Some(String::new());
123        }
124        kept.new_name.clear();
125    }
126}

128/// The note chosen: its name, who wrote it and when, Edit or Done, Forget,
129/// and its lines, or the field it is edited in.
130fn shown(
131    ui: &mut egui::Ui,
132    id: Id,
133    view: &NotesView,
134    kept: &mut Kept,
135    own: bool,
136) -> Option<Clicked> {
137    let Some((who, name)) = kept.chosen.clone() else {
138        ui.weak("Choose a note, or type a name for a new one.");
139        return None;
140    };
141    let Some(note) = view.notes.get(&who, &name) else {
142        ui.weak(format!("{name} is gone."));
143        kept.chosen = None;
144        kept.draft = None;
145        return None;
146    };
147    let mut clicked = None;
148    ui.horizontal(|ui| {
149        ui.strong(&name);
150        ui.weak(format!("{}, {}", notes::character_of(&who), note.changed));
151        if !own {
152            return;
153        }
154        match &kept.draft {
155            None => {
156                if ui.small_button("Edit").clicked() {
157                    kept.draft = Some(note.text.clone());
158                }
159            }
160            Some(draft) => {
161                if ui.small_button("Done").clicked() {
162                    save(view, &who, &name, note, draft);
163                    kept.draft = None;
164                }
165            }
166        }
167        if ui
168            .small_button("Forget")
169            .on_hover_text("Take this note out")
170            .clicked()
171        {
172            let _ = notes::remove(&view.dir, &who, &name);
173            kept.chosen = None;
174            kept.draft = None;
175        }
176    });
177    ui.separator();
178    egui::ScrollArea::vertical()
179        .id_salt(id.with("notes-shown"))
180        .auto_shrink([false, true])
181        .show(ui, |ui| match &mut kept.draft {
182            Some(draft) if own => {
183                let field = ui.add(
184                    egui::TextEdit::multiline(draft)
185                        .desired_width(f32::INFINITY)
186                        .desired_rows(8),
187                );
188                if field.lost_focus() {
189                    save(view, &who, &name, note, draft);
190                }
191            }
192            _ => {
193                for line in note.clipped() {
194                    clicked = clicked.take().or(drawn(ui, &line, own));
195                }
196            }
197        });
198    clicked
199}

201/// `draft` kept as the note's words, when it changed them.
202fn save(view: &NotesView, who: &str, name: &str, note: &Note, draft: &str) {
203    if draft != note.text {
204        let _ = notes::rewrite(&view.dir, who, name, draft, &notes::stamp());
205    }
206}

208/// One line of a note: plain words, or linked ones a click takes when the
209/// window is the character's own.
210fn drawn(ui: &mut egui::Ui, line: &Clipped, own: bool) -> Option<Clicked> {
211    if line.links.is_empty() {
212        ui.label(RichText::new(&line.text));
213        return None;
214    }
215    let runs = runs_of(line);
216    let job = crate::text::job(&runs, ui.style(), &crate::theme::palette(ui.ctx()));
217    let acted = ui.scope(|ui| crate::text::linked(ui, job, &runs)).inner;
218    if own { acted.map(Clicked::from) } else { None }
219}

221/// A clipped line as runs: each link its own run, the words between plain.
222fn runs_of(line: &Clipped) -> Vec<StyledRun> {
223    let mut runs = Vec::new();
224    let mut at = 0;
225    let mut links: Vec<_> = line.links.iter().collect();
226    links.sort_by_key(|link| link.from);
227    for link in links {
228        if link.from < at || link.to > line.text.len() || link.from > link.to {
229            continue;
230        }
231        if link.from > at {
232            runs.push(StyledRun::plain(&line.text[at..link.from]));
233        }
234        let run = StyledRun {
235            link: Some(match &link.does {
236                Does::Object { exist, noun, coord } => RunLink::Object {
237                    exist: exist.clone(),
238                    noun: noun.clone(),
239                    coord: coord.clone(),
240                },
241                Does::Command { command } => RunLink::Command {
242                    command: command.clone(),
243                },
244                Does::Url { href } => RunLink::Url { href: href.clone() },
245            }),
246            ..StyledRun::plain(&line.text[link.from..link.to])
247        };
248        runs.push(run);
249        at = link.to;
250    }
251    if at < line.text.len() {
252        runs.push(StyledRun::plain(&line.text[at..]));
253    }
254    runs
255}

257#[cfg(test)]
258mod tests {
259    use super::*;
260    use hydra_session::player_log::links::LogLink;
261
262    /// A clipped line becomes runs with each link its own, the words
263    /// between plain, and a link past the words is passed over.
264    #[test]
265    fn a_clipped_line_is_runs_with_its_links() {
266        let line = Clipped {
267            text: "You also see a large sign and a stool.".to_owned(),
268            links: vec![
269                LogLink {
270                    from: 13,
271                    to: 25,
272                    does: Does::Object {
273                        exist: "7".to_owned(),
274                        noun: "sign".to_owned(),
275                        coord: None,
276                    },
277                },
278                LogLink {
279                    from: 30,
280                    to: 37,
281                    does: Does::Command {
282                        command: "look stool".to_owned(),
283                    },
284                },
285                LogLink {
286                    from: 90,
287                    to: 99,
288                    does: Does::Url {
289                        href: "https://example.invalid".to_owned(),
290                    },
291                },
292            ],
293        };
294        let runs = runs_of(&line);
295        let texts: Vec<&str> = runs.iter().map(|run| run.text.as_str()).collect();
296        assert_eq!(
297            texts,
298            ["You also see ", "a large sign", " and ", "a stool", "."]
299        );
300        assert!(matches!(&runs[1].link, Some(RunLink::Object { noun, .. }) if noun == "sign"));
301        assert!(
302            matches!(&runs[3].link, Some(RunLink::Command { command }) if command == "look stool")
303        );
304        assert!(runs[0].link.is_none() && runs[4].link.is_none());
305    }

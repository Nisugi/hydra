13/// Draw `widget` for `seen` into `ui`. Following another character, a
14/// one-line widget puts its name before what it says, and the rest a line
15/// with its name above. A line to send, when the player asked for one.
16#[allow(
17    clippy::too_many_lines,
18    reason = "the catalog's table: one arm per kind, each a call; split, it would hide which kind draws how"
19)]
20pub(super) fn draw(
21    widget: &Widget,
22    ui: &mut egui::Ui,
23    seen: &Seen<'_>,
24    id: Id,
25    chosen: &super::Chosen,
26) -> Option<super::Clicked> {
27    let state = seen.snapshot.map(|snapshot| &snapshot.state);
28    let look = chosen.look.clone().or_else(|| widget.bar_look());
29    let lines = chosen.lines.unwrap_or_default();

31    let named = |label: &str| {
32        seen.who
33            .map_or_else(|| label.to_owned(), |who| format!("{who} {label}"))
34    };

36        scrolled(ui, id, |ui| {
37            if let Some(who) = seen.who {
38                ui.weak(who);
39            }
40            add(ui);
41        });
42    };
43    match widget {
44        // Never for another character's compass: a click would move this
45        // window's character. A set's pictures once they are read, and words
46        // till then.
47        Widget::Compass => {
48            let goes = seen.who.is_none();

50                .compass

73            }

448/// One line of a one-line widget: never wrapped onto a second, which its
449/// cell has no room for, but cut short with an ellipsis, the whole of it
450/// shown when the pointer rests on it.
451/// How many of the newest answers the Lag widget says the median of.
452const LAG_OF: usize = 10;
453
454/// The game's lag, as its last [`LAG_OF`] answers to what was typed had it,
455/// in the colour its time reads in; under the pointer, the slow end, and
456/// what it measures. A dash before anything is typed.
457fn lag(ui: &mut egui::Ui, story: &hydra_ui::Story, label: &str) {
458    use hydra_ui::perf::{LAG, took};
459    let samples = &story.perf.reply;
460    let Some((median, slow)) = samples.last_spread(LAG_OF) else {
461        ui.add(egui::Label::new(RichText::new(format!("{label} —")).weak()).truncate())
462            .on_hover_text("Lag: how long the game takes to answer what you type. Type something.");
463        return;
464    };
465    let colour = crate::hub::graded(ui.ctx(), LAG.grade(median));
466    let of = samples.len().min(LAG_OF);
467    let said = RichText::new(format!("{label} {}", took(median))).color(colour);
468    ui.add(egui::Label::new(said).truncate())
469        .on_hover_text(format!(
470            "Lag: the median of your last {of} commands, from sent to the game's answer -- \
471         the network and the game, not Hydra. The slowest of them: {}.",
472            took(slow)
473        ));
474}

480/// A widget whose content may be taller than it is given: it scrolls rather
481/// than growing what holds it, and asks for no height of its own, which is
482/// the layout's to say.
483/// Scrolled as a key asked, when it is the window in use.
484fn scrolled(ui: &mut egui::Ui, id: Id, add: impl FnOnce(&mut egui::Ui)) {
485    let scroll = super::split::asked(ui, id);
486    egui::ScrollArea::vertical()
487        .min_scrolled_height(0.0)
488        .id_salt(id.with("scroll"))
489        .auto_shrink(false)
490        .show(ui, |ui| super::split::keyed(ui, scroll, add));
491}

493/// A vital as a bar fitted to what it is given, drawn as `look` says.
494fn vital(ui: &mut egui::Ui, label: &str, vital: Option<Vital>, token: T, look: Option<&bar::Look>) {
495    let amount = vital.map(|vital| Amount {
496        percent: vital.percent,
497        current: vital.current,
498        max: vital.max,
499    });
500    let color = theme::color(ui.ctx(), token);
501    let drawn = as_looks(ui, Bar::new(label, amount).fill(color), look);
502    ui.add(drawn.fitted(ui));
503}

505/// `current` of `max` as a vital is: its percent, and the pair.
506fn out_of(current: u64, max: u64) -> Option<Vital> {
507    let percent = current.saturating_mul(100).checked_div(max)?;
508    Some(Vital {
509        percent: u32::try_from(percent.min(100)).unwrap_or(100),
510        current: i32::try_from(current).ok(),
511        max: i32::try_from(max).ok(),
512    })
513}

515/// Field experience against its most, as the mind bar's `field_exp` and

517fn field_experience(state: &GameState) -> Option<Vital> {
518    let experience = &state.character.experience;
519    out_of(
520        experience.field_experience?.into(),
521        experience.field_experience_max?.into(),

525/// The Betrayer panel's `Blood Points: 50`, out of 100: a label in the
526/// game's `BetrayerPanel` dialog, seen in game logs from 0 to 100.
527fn blood_points(state: &GameState) -> Option<Vital> {
528    let panel = state.dialogs.get("BetrayerPanel")?;
529    let points = panel.parts.iter().find_map(|(id, part)| match part {
530        hydra_model::state::dialogs::Part::Label(text) if id == "lblBPs" => text
531            .strip_prefix("Blood Points:")?
532            .trim()
533            .parse::<u64>()
534            .ok(),
535        _ => None,
536    })?;
537    out_of(points, 100)
538}

540/// `bar` as `look` says, its images among it.
541pub(super) fn as_looks<'a>(ui: &egui::Ui, bar: Bar<'a>, look: Option<&bar::Look>) -> Bar<'a> {
542    let Some(look) = look else {
543        return bar;
544    };
545    let mut drawn = bar.look(look);
546    if let Some(overlay) = look.overlay.as_deref().and_then(|path| overlay(ui, path)) {
547        drawn = drawn.overlay(overlay);
548    }
549    if let Some(image) = look.background.as_deref().and_then(|path| image(ui, path)) {
550        drawn = drawn.background(image.id());
551    }
552    if let Some(image) = look.fill_image.as_deref().and_then(|path| image(ui, path)) {
553        drawn = drawn.fill_image(image.id());
554    }
555    drawn
556}

568/// What a hand holds, after which hand: `?` until the game has said.
569/// An object carried and let go on a hand, put `onto` it (`carry.rs`);
570/// the hand's own item, let go on it, nothing. Another character's widget
571/// asks nothing of this window: the play window draws it and sends nothing
572/// (`play/draw.rs`).
573fn place(ui: &mut egui::Ui, id: Id, onto: &str, hand: Option<&Hand>) -> Option<super::Clicked> {
574    let holding = hand.and_then(Hand::id);
575    crate::carry::target(ui, id, onto, holding).map(super::Clicked::Quietly)
576}

578/// What a hand holds, after which hand. On the window's `own` character's it is
579/// a link, as the game sends it -- left, right and spell alike: a click asks
580/// for the item's menu, as a click on it in the story does, and with the drag
581/// key held the item is carried from it (`carry.rs`). The link clicked, if it
582/// was.
583fn held(ui: &mut egui::Ui, which: &str, hand: Option<&Hand>, own: bool) -> Option<super::Clicked> {
584    let holds = match hand {
585        None | Some(Hand::Unknown) => "?",
586        Some(Hand::Empty) => "empty",
587        Some(Hand::Holding { name, .. }) => name,
588    };
589    let label = egui::Label::new(labels::before(which, None, holds))
590        .truncate()
591        .selectable(false);
592    match hand {
593        Some(Hand::Holding {
594            id: Some(id),
595            noun,
596            name,
597        }) if own => {
598            let carrying = crate::carry::held(ui);
599            let sense = if carrying {
600                crate::carry::sense(ui)
601            } else {
602                egui::Sense::click()
603            };
604            let response = ui.add(label.sense(sense));
605            crate::carry::source(
606                &response,
607                crate::carry::Carried {
608                    exist: id.clone(),
609                    name: name.clone(),
610                },
611            );
612            linked(&response, id, noun.as_deref().unwrap_or_default())
613        }
614        _ => {
615            ui.add(label);
616            None
617        }
618    }
619}

621/// The spell prepared, after `which`: on the window's `own` character's a
622/// link to it, `#spell`, as the game sends it, whose menu a click asks for.
623/// Never carried: a spell is not put anywhere. The link clicked, if it was.
624fn spell_hand(
625    ui: &mut egui::Ui,
626    which: &str,
627    state: Option<&GameState>,
628    own: bool,
629) -> Option<super::Clicked> {
630    let spell = match state.map(|state| state.prepared.as_deref()) {
631        None => "?",
632        Some(None) => "none",
633        Some(Some(spell)) if spell.eq_ignore_ascii_case("none") => "none",
634        Some(Some(spell)) => spell,
635    };
636    let label = egui::Label::new(labels::before(which, None, spell))
637        .truncate()
638        .selectable(false);
639    match state.and_then(|state| state.prepared_id.as_deref()) {
640        Some(id) if own => {
641            let response = ui.add(label.sense(egui::Sense::click()));
642            linked(&response, id, "")
643        }
644        _ => {
645            ui.add(label);
646            None
647        }
648    }
649}

651/// A hand's words as a link to the object `id` (`noun`): the hand shown on
652/// hover, a click its link, where the pointer was.
653fn linked(response: &egui::Response, id: &str, noun: &str) -> Option<super::Clicked> {
654    let response = response
655        .clone()
656        .on_hover_cursor(egui::CursorIcon::PointingHand);
657    response.clicked().then(|| {
658        let at = response
659            .interact_pointer_pos()
660            .unwrap_or(response.rect.center());
661        super::Clicked::Link(
662            hydra_ui::RunLink::Object {
663                exist: id.to_owned(),
664                noun: noun.to_owned(),
665                coord: None,
666            },
667            at,
668        )
669    })
670}

672/// A clock counting down, in whole seconds, or that none runs.
673fn clock(ui: &mut egui::Ui, label: &str, seconds: Option<u32>, token: T) {
674    match seconds.filter(|seconds| *seconds > 0) {
675        Some(seconds) => line(
676            ui,
677            RichText::new(format!("{label} {seconds}s")).color(theme::color(ui.ctx(), token)),
678        ),
679        None => line(ui, RichText::new(format!("{label} —")).weak()),
680    }
681}

683/// The stun left, as a clock: its seconds while the rounds the game gave
684/// run, and `stunned` while the indicator stays lit past them or the game
685/// gave none (`hydra_model`'s `state/stun.rs`).
686fn stun(ui: &mut egui::Ui, label: &str, state: Option<&GameState>) {
687    let left = state.and_then(GameState::stun_remaining).unwrap_or(0);
688    let wrong = theme::color(ui.ctx(), T::Wrong);
689    if left > 0 {
690        line(ui, RichText::new(format!("{label} {left}s")).color(wrong));
691    } else if state.is_some_and(|state| state.status.get("stunned")) {
692        line(ui, RichText::new(format!("{label}: stunned")).color(wrong));
693    } else {
694        line(ui, RichText::new(format!("{label} —")).weak());
695    }
696}

698/// The names a room list shows, each its runs: the creatures that are not
699/// hostile, or the objects, once the game has said what is here. `None`
700/// until the game has said.
701fn room_list(
702    ctx: &egui::Context,
703    widget: &Widget,
704    snapshot: Option<&Snapshot>,
705    tags: Option<hydra_model::targetid::Style>,
706) -> Option<Vec<Vec<StyledRun>>> {
707    let snapshot = snapshot?;
708    let room = &snapshot.state.room;
709    let things = |items: &[RoomItem], look| {
710        room.component("room objs").is_some().then(|| {
711            items
712                .iter()
713                .map(|item| super::described::item_runs(item, look))
714                .collect()
715        })
716    };
717    // Hostile when the game first said, whatever it says now: a sympathied
718    // creature still is, and is the Targets list's.
719    let hostile = |item: &RoomItem| {
720        item.id
721            .parse()
722            .ok()
723            .and_then(|id| snapshot.state.creatures().get(id))
724            .and_then(hydra_model::CreatureInstance::hostile_when_first_seen)
725            == Some(true)
726    };
727    if *widget == Widget::Objects {
728        return things(&room.objects, (theme::color(ctx, T::Object), None));
729    }
730    let npcs: Vec<RoomItem> = room
731        .creatures
732        .iter()
733        .filter(|item| !hostile(item))
734        .cloned()
735        .collect();
736    let creature = theme::color(ctx, T::Creature);
737    things(&npcs, (creature, tags.map(|style| (snapshot, style))))
738}

740/// Hydra's own messages, the newest at the bottom.
741fn hydra(ui: &mut egui::Ui, said: &std::collections::VecDeque<String>, id: Id) {
742    egui::ScrollArea::vertical()
743        .min_scrolled_height(0.0)
744        .id_salt(id.with("said"))
745        .stick_to_bottom(true)
746        .auto_shrink(false)
747        .show(ui, |ui| {
748            if said.is_empty() {
749                ui.weak("Nothing yet.");
750            }
751            for line in said {
752                ui.label(line);
753            }
754        });

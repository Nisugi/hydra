1//! The Skill trainer widget: the character's skills and goals, from the
2//! play.net skill manager `goals` opens (`trainer.rs`), changed a rank at a
3//! time and applied without a browser.
4//!
5//! Each skill under its heading: the ranks it has, the ranks it is to have,
6//! and what the next rank costs; − and + step its goal, ten at a time with
7//! Shift or Ctrl held. The points left and converted are above them, as the
8//! website's arithmetic leaves them. Apply posts the goals; Reset puts them
9//! back. Load sends `goals`; Website sends it for the browser instead.
10//! Profiles saves the goals under a name, and loads or deletes one saved.
11

19/// Ranks a step with Shift or Ctrl held.
20const MANY: u32 = 10;
21
22/// The widget, over `view`, the window's own character's trainer, and its
23/// goal profiles' names: none on another's.
24pub(super) fn trainer(
25    ui: &mut egui::Ui,
26    view: Option<&TrainerView>,
27    profiles: Option<&[String]>,
28) -> Option<Clicked> {
29    let (Some(view), Some(profiles)) = (view, profiles) else {
30        ui.weak("Only the window's own character's trainer shows.");
31        return None;
32    };
33    let mut asked = None;
34    ui.horizontal(|ui| {
35        let busy = matches!(view.status, Status::Loading | Status::Applying);
36        if ui
37            .add_enabled(!busy, egui::Button::new("Load"))
38            .on_hover_text("Send GOALS, and read the skill manager here")
39            .clicked()
40        {
41            asked = Some(Ask::Load);
42        }
43        if ui
44            .button("Website")
45            .on_hover_text("Send GOALS, and open the skill manager in the browser")
46            .clicked()
47        {
48            asked = Some(Ask::Web);
49        }
50    });
51    // On a line of its own, wrapped: a long one never holds the window wide.
52    let said = match &view.status {
53        Status::Idle if view.goals.is_none() => Some("Load reads your skills from play.net."),
54        Status::Idle => None,
55        Status::Loading => Some("Reading the skill manager…"),
56        Status::Applying => Some("Applying…"),
57        Status::Said(said) => {
58            ui.add(egui::Label::new(said.as_str()).wrap());
59            None
60        }
61    };
62    if let Some(said) = said {
63        ui.add(egui::Label::new(RichText::new(said).weak()).wrap());
64    }
65    if let Some(goals) = &view.goals {
66        let applying = view.status == Status::Applying;
67        if let Some(now) = saved(ui, goals, profiles, applying) {
68            asked = Some(now);
69        }
70        if let Some(now) = sheet(ui, goals, applying) {
71            asked = Some(now);
72        }
73    }
74    asked.map(Clicked::Trainer)
75}

77/// The profiles: one loaded or deleted from the menu, the goals saved under
78/// the name written. `applying`, nothing is loaded.
79fn saved(
80    ui: &mut egui::Ui,
81    goals: &SkillGoals,
82    profiles: &[String],
83    applying: bool,
84) -> Option<Ask> {
85    let mut asked = None;
86    ui.horizontal(|ui| {
87        ui.add_enabled_ui(!applying, |ui| {
88            ui.menu_button("Profiles", |ui| {
89                if profiles.is_empty() {
90                    ui.weak("None saved.");
91                }
92                for name in profiles {
93                    ui.horizontal(|ui| {
94                        if ui
95                            .button(name)
96                            .on_hover_text("Load these goals: Apply saves them")
97                            .clicked()
98                        {
99                            asked = Some(Profiled::Load(name.clone()));
100                        }
101                        if ui.small_button("🗑").on_hover_text("Delete it").clicked() {
102                            asked = Some(Profiled::Delete(name.clone()));
103                        }
104                    });
105                }
106            });
107        });
108        let id = ui.id().with("profile-name");
109        let mut name: String = ui.data_mut(|data| data.get_temp(id).unwrap_or_default());
110        ui.add(
111            egui::TextEdit::singleline(&mut name)
112                .hint_text("Profile name")
113                .desired_width(120.0),
114        );
115        let named = name.trim().to_owned();
116        if ui
117            .add_enabled(!named.is_empty(), egui::Button::new("Save"))
118            .on_hover_text("Save the goals now under this name")
119            .clicked()
120        {
121            asked = Some(Profiled::Save {
122                name: named,
123                goals: goals.targets(),
124            });
125            name.clear();
126        }
127        ui.data_mut(|data| data.insert_temp(id, name));
128    });
129    asked.map(Ask::Profile)
130}

132/// The character, its points, its skills under their headings, and Apply
133/// and Reset; `applying`, Apply is held.
134fn sheet(ui: &mut egui::Ui, goals: &SkillGoals, applying: bool) -> Option<Ask> {
135    let mut asked = None;
136    ui.label(
137        RichText::new(format!(
138            "{}, level {} {} {}",
139            goals.name, goals.level, goals.race, goals.profession
140        ))
141        .strong(),
142    );
143    ui.label(format!(
144        "Physical points: {} left, {} converted to mental",
145        goals.physical_left, goals.physical_converted
146    ));
147    ui.label(format!(
148        "Mental points: {} left, {} converted to physical",
149        goals.mental_left, goals.mental_converted
150    ));
151    ui.horizontal(|ui| {
152        let changed = goals.changed();
153        if ui
154            .add_enabled(changed && !applying, egui::Button::new("Apply"))
155            .on_hover_text("Save these goals on play.net, and tell the game")
156            .clicked()
157        {
158            asked = Some(Ask::Apply);
159        }
160        if ui
161            .add_enabled(changed && !applying, egui::Button::new("Reset"))
162            .on_hover_text("Every goal back to the ranks it has")
163            .clicked()
164        {
165            asked = Some(Ask::Change(Change::Reset));
166        }
167    });
168    // How many ranks a press of − or + moves: 1, 10 or 100, chosen on the
169    // bar and kept while the window lives; Shift or Ctrl held multiplies
170    // that again.
171    let step_id = egui::Id::new(("trainer-step", ui.id()));
172    let mut step: u32 = ui.ctx().data(|data| data.get_temp(step_id)).unwrap_or(1);
173    ui.horizontal(|ui| {
174        ui.weak("Step");
175        for size in [1_u32, 10, 100] {
176            if ui
177                .selectable_label(step == size, size.to_string())
178                .on_hover_text(format!("− and + move {size} rank(s) at a time"))
179                .clicked()
180            {
181                step = size;
182            }
183        }
184    });
185    ui.ctx().data_mut(|data| data.insert_temp(step_id, step));
186    let many = ui.input(|input| input.modifiers.shift || input.modifiers.ctrl);
187    let times = if many { step * MANY } else { step };
188    let mut section = None;
189    let changed_color = theme::color(ui.ctx(), T::Accent);
190    egui::Grid::new("trainer-rows")
191        .num_columns(6)
192        .striped(true)
193        .show(ui, |ui| {
194            for row in &goals.rows {
195                if section != Some(&row.section) {
196                    section = Some(&row.section);
197                    ui.label(RichText::new(&row.section).small().weak());
198                    ui.end_row();
199                }
200                let (has, goal) = (goals.has(row.id), goals.goal(row.id));
201                ui.label(&row.name);
202                ui.label(has.to_string());
203                let goal_text = RichText::new(goal.to_string());
204                ui.label(if goal == has {
205                    goal_text
206                } else {
207                    goal_text.color(changed_color).strong()
208                });
209                let step = |up: bool| {
210                    Ask::Change(Change::Step {
211                        id: row.id,
212                        up,
213                        times,
214                    })
215                };
216                if ui
217                    .small_button("−")
218                    .on_hover_text(format!("{times} rank(s) less"))
219                    .clicked()
220                {
221                    asked = Some(step(false));
222                }
223                if ui
224                    .small_button("+")
225                    .on_hover_text(format!("{times} rank(s) more"))
226                    .clicked()
227                {
228                    asked = Some(step(true));
229                }
230                let (physical, mental) = goals.cost_of_next(row.id);
231                ui.weak(format!("{physical}/{mental}"));
232                ui.end_row();
233            }
234        });
235    asked
236}

253    /// What a click on the `at`th widget labelled `label` asks, the widget
254    /// drawn over `view`.
255    fn clicked_on(view: TrainerView, label: &str, at: usize) -> Option<Clicked> {
256        let mut harness = Harness::builder().with_size((500.0, 600.0)).build_ui_state(
257            move |ui, clicked: &mut Option<Clicked>| {
258                if let Some(now) = trainer(ui, Some(&view), Some(&[])) {
259                    *clicked = Some(now);
260                }
261            },
262            None,
263        );
264        harness.run();
265        harness
266            .get_all_by_label(label)
267            .nth(at)
268            .expect(label)
269            .click();
270        harness.run();
271        harness.state().clone()
272    }

288    /// The page read: the character, its points, each skill with its ranks
289    /// and the cost of the next; + steps a goal, and Apply waits for a
290    /// change.
291    #[test]
292    fn the_sheet_shows_the_skills_and_steps_a_goal() {
293        let view = TrainerView {
294            status: Status::Idle,
295            goals: Some(read()),
296        };
297        let shown = view.clone();
298        let harness = Harness::builder()
299            .with_size((500.0, 600.0))
300            .build_ui(move |ui| {
301                let _ = trainer(ui, Some(&shown), Some(&[]));
302            });
303        for words in [
304            "Nisugi, level 100 Half-Elf Ranger",
305            "Physical points: 2817 left, 888 converted to mental",
306            "Armor Use",
307            "Minor Spiritual",
308            "Elemental Lore",
309        ] {
310            assert!(harness.query_by_label(words).is_some(), "{words}");
311        }
312        // Two Weapon Combat is the third row: its +.
313        let row = view
314            .goals
315            .as_ref()
316            .and_then(|goals| goals.rows.iter().position(|row| row.id == 0))
317            .expect("two weapon combat");
318        assert_eq!(
319            clicked_on(view.clone(), "+", row),
320            Some(Clicked::Trainer(Ask::Change(Change::Step {
321                id: 0,
322                up: true,
323                times: 1
324            })))
325        );
326        assert_eq!(
327            clicked_on(view.clone(), "Apply", 0),
328            None,
329            "nothing changed"
330        );
331        let mut changed = view;
332        if let Some(goals) = &mut changed.goals {
333            goals.step(5, 1, true);
334        }
335        assert_eq!(
336            clicked_on(changed, "Apply", 0),
337            Some(Clicked::Trainer(Ask::Apply))
338        );
339    }

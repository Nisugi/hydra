1//! A text widget's scrollback split. Scrolling up splits the widget: the
2//! top is what you are scrolling, and it moves only when you scroll it, not
3//! when new lines come in; the bottom sticks to the newest line as it comes
4//! in. The separator is draggable to set their proportions, and a button on
5//! it scrolls back to the bottom, removing the split.
6//!
7//! One pane, stuck to the newest line, until the player scrolls it back:
8//! then two, the top where the player left it and the bottom following the
9//! newest. The top closes the split too when scrolled back down to the
10//! newest. The wheel scrolls the top wherever
11//! over the widget it is turned, the bottom taking none.
12//!
13//! **The top holds its place when old lines go.** A story keeps its newest
14//! thousand lines, so once full, each new line drops the oldest and the rest
15//! move up by its height. The top is kept on its lines, not its offset: the
16//! widget marks where each line starts ([`Tops`]), and each frame the top
17//! moves down by the height of the lines dropped since the last.
18//!
19//! **The newest line holds the bottom in the frame lines come.** egui
20//! takes a pane stuck to its newest line there only as the pane's frame
21//! ends, after its lines are drawn where the pane was. A line that draws
22//! nothing -- one routed to a stream whose window is open, Hydra's own
23//! beside a Hydra widget -- still drops the oldest from a full story, and
24//! would leave a blank under the newest line until the window next painted;
25//! a new line would show a frame late. The frame is drawn again instead,
26//! before it is shown ([`redrawn_at_newest`]).
27//!
28//! **Keys scroll it too**: a page or a line back opens
29//! the split as the wheel does, and scrolls its top; forward scrolls the top
30//! down, and to the newest line closes it, as its button does. A key's
31//! scroll is asked of a widget by its id ([`ask`]) and done as it is next
32//! drawn, by this or by any widget that scrolls ([`asked`]).
33//!
34//! **A selection dragged past its edge scrolls it**, as text does in an
35//! editor: held above the widget, the pane the player scrolls goes back,
36//! splitting as the wheel does; held below the top -- over the separator or
37//! the bottom -- the top comes forward, to the newest line and one pane
38//! again. The farther past the edge, the faster (`dragged_past`). Lines keep
39//! their ids across the split (`Tops::line_id`), so the selection goes on
40//! in the top; the bottom's lines take no part in it meanwhile, though the
41//! pointer is over them. A selection begun in the bottom scrolls nothing.

52/// How tall the separator is, to take a drag.
53const BAR: f32 = 14.0;

55/// The smallest share of the height either pane is left.
56const LEAST: f32 = 0.1;

58/// A widget's split, kept from frame to frame.
59#[derive(Clone, Debug)]
60#[expect(
61    clippy::struct_excessive_bools,
62    reason = "four things known of the split, each on its own, not one state"
63)]
64struct Split {
65    /// The top pane is open.
66    open: bool,
67    /// The top pane's share of the height.
68    share: f32,
69    /// The top pane's offset when last drawn.
70    offset: f32,
71    /// The top pane is to start at `offset`: it opened this frame, or old
72    /// lines went.
73    place: bool,
74    /// The number of the first line kept when last drawn.
75    first: u64,
76    /// Where each line started when last drawn.
77    ys: Vec<f32>,
78    /// Times the split opened or closed: each pane's id takes it, so each
79    /// begins afresh, and a fresh pane stuck to the newest line starts
80    /// there. An offset set past the end is not pulled back by egui, and
81    /// left the pane empty below its lines.
82    turns: u64,
83    /// The one pane has just come back and is not yet at the newest line:
84    /// it takes no wheel until it is. A fresh pane starts at its top and is
85    /// taken to its end as its first frame closes, which egui skips on a
86    /// frame with a wheel turn in it -- and the rest of the turn that
87    /// scrolled the top down to close the split is such a frame. It left
88    /// the pane far back, and split it again.
89    settling: bool,
90    /// The pane the player scrolls -- the one, or the top -- as last drawn:
91    /// how tall it is, and its offset at the newest line.
92    height: f32,
93    end: f32,
94    /// The button held was pressed in the pane the player scrolls: a
95    /// selection dragged from there scrolls it past its edge.
96    from_player: bool,
97}

99impl Default for Split {
100    fn default() -> Self {
101        Self {
102            open: false,
103            share: 0.6,
104            offset: 0.0,
105            place: false,
106            first: 0,
107            ys: Vec::new(),
108            turns: 0,
109            settling: false,
110            height: 0.0,
111            end: 0.0,
112            from_player: false,
113        }
114    }
115}

117/// A scrolled body of lines, newest at the bottom, split when the player
118/// scrolls back. `first` is the number of the first line kept, which grows
119/// as old lines are dropped; `wrap` whether lines wrap, or the body scrolls
120/// sideways too; `scroll` what a key asked; `add` draws every line, marking
121/// each on its [`Tops`], and may be drawn twice a frame, once a pane.
122pub(super) fn scrolled(
123    ui: &mut egui::Ui,
124    id: Id,
125    (first, wrap): (u64, bool),
126    scroll: Option<Scroll>,
127    mut add: impl FnMut(&mut egui::Ui, &mut Tops),
128) {
129    let state_id = id.with("split");
130    let mut split: Split = ui
131        .data_mut(|data| data.get_temp(state_id))
132        .unwrap_or_default();
133    let mut body = |ui: &mut egui::Ui, player: bool| {
134        if !wrap {
135            ui.style_mut().wrap_mode = Some(egui::TextWrapMode::Extend);
136        }
137        // The theme's room between two lines.
138        ui.spacing_mut().item_spacing.y += crate::theme::line_spacing(ui.ctx());
139        // Where the content began: the pane's offset as it was drawn.
140        let began = ui.max_rect().top();
141        let lines = id.with(if player { "player" } else { "live" });
142        let mut tops = Tops::at(ui, player, lines);
143        add(ui, &mut tops);
144        (tops.keep(ui), began)
145    };
146    if split.open {
147        both(ui, id, &mut split, (first, wrap, scroll), &mut body);
148    } else {
149        let mut area = pane(wrap)
150            .id_salt(id.with(("one", split.turns)))
151            .stick_to_bottom(true);
152        // A key back from the newest line: the pane goes there, and splits.
153        // Scrolled from inside, at once: an offset given the pane is not
154        // the player's scroll to egui, and sticking to the newest line
155        // takes it straight back.
156        let (line, page) = steps(ui, split.height);
157        // A selection held above the pane takes it back, as a key does.
158        let whole = ui.available_rect_before_wrap();
159        pressed_in(ui, whole, &mut split.from_player);
160        let dragged = dragged_past(ui, whole, split.from_player);
161        let back = match scroll {
162            Some(Scroll::PageUp) => Some(page),
163            Some(Scroll::LineUp) => Some(line),
164            Some(Scroll::Top) => Some(split.end),
165            _ => (dragged < 0.0).then_some(-dragged),
166        };
167        let back = back.filter(|_| split.end > 0.0);
168        if back.is_some() {
169            split.settling = false;
170        }
171        if split.settling {
172            area = area.scroll_source(egui::scroll_area::ScrollSource::NONE);
173        }
174        let shown = area.show(ui, |ui| {
175            if let Some(back) = back {
176                let now = egui::style::ScrollAnimation::none();
177                ui.scroll_with_delta_animation(vec2(0.0, back), now);
178            }
179            body(ui, true)
180        });
181        redrawn_at_newest(ui, &shown);
182        let newest = newest(&shown);
183        split.height = shown.inner_rect.height();
184        split.end = newest;
185        let back = shown.state.offset.y < newest - 1.0;
186        if split.settling {
187            split.settling = back;
188        } else if back {
189            // Scrolled back: the top pane starts where the player is.
190            split.open = true;
191            split.offset = shown.state.offset.y;
192            split.place = true;
193            split.first = first;
194            split.ys = shown.inner.0;
195            split.turns += 1;
196        }
197    }
198    ui.data_mut(|data| data.insert_temp(state_id, split));
199}

201/// Both panes and the separator between them.
202fn both(
203    ui: &mut egui::Ui,
204    id: Id,
205    split: &mut Split,
206    (first, wrap, scroll): (u64, bool, Option<Scroll>),
207    body: &mut impl FnMut(&mut egui::Ui, bool) -> (Vec<f32>, f32),
208) {
209    let whole = ui.available_rect_before_wrap();
210    let panes = (whole.height() - BAR).max(0.0);
211    let top_height = (panes * split.share).round();
212    let top_rect = Rect::from_min_size(whole.min, vec2(whole.width(), top_height));
213    let bar_rect = Rect::from_min_size(top_rect.left_bottom(), vec2(whole.width(), BAR));
214    let bottom_rect = Rect::from_min_max(bar_rect.left_bottom(), whole.max);
215
216    // A key scrolls the top, never past the newest line.
217    let (line, page) = steps(ui, split.height);
218    let moved = match scroll {
219        Some(Scroll::PageUp) => Some(split.offset - page),
220        Some(Scroll::PageDown) => Some(split.offset + page),
221        Some(Scroll::LineUp) => Some(split.offset - line),
222        Some(Scroll::LineDown) => Some(split.offset + line),
223        Some(Scroll::Top) => Some(0.0),
224        #[expect(clippy::cast_precision_loss, reason = "a few points a frame")]
225        Some(Scroll::By(points)) => Some(split.offset - points as f32),
226        Some(Scroll::Bottom) | None => None,
227    };
228    if let Some(to) = moved {
229        split.offset = to.clamp(0.0, split.end.max(0.0));
230        split.place = true;
231    }
232    // The wheel over the separator or the bottom, which takes none, scrolls
233    // the top as over the top itself: where the pointer is does not matter.
234    let below = Rect::from_min_max(bar_rect.min, whole.max);
235    let wheel = ui.input(|input| {
236        input
237            .pointer
238            .hover_pos()
239            .filter(|at| below.contains(*at) && ui.clip_rect().contains(*at))
240            .map_or(0.0, |_| input.smooth_scroll_delta.y)
241    });
242    if wheel != 0.0 {
243        split.offset = (split.offset - wheel).clamp(0.0, split.end.max(0.0));
244        split.place = true;
245    }
246    // A selection from the top held above it takes it back; below it, over
247    // the separator or the bottom, on toward the newest line.
248    pressed_in(ui, top_rect, &mut split.from_player);
249    let dragged = dragged_past(ui, top_rect, split.from_player);
250    if dragged != 0.0 {
251        split.offset = (split.offset + dragged).clamp(0.0, split.end.max(0.0));
252        split.place = true;
253    }
254
255    // The top: where the player left it, moved down by the lines dropped.
256    let gone = usize::try_from(first.saturating_sub(split.first)).unwrap_or(usize::MAX);
257    let shift = if gone == 0 {
258        0.0
259    } else {
260        split.ys.get(gone).copied().unwrap_or(split.offset)
261    };
262    let mut top = pane(wrap).id_salt(id.with(("top", split.turns)));
263    if std::mem::take(&mut split.place) || shift > 0.0 {
264        top = top.vertical_scroll_offset((split.offset - shift).max(0.0));
265    }
266    let shown = ui
267        .scope_builder(egui::UiBuilder::new().max_rect(top_rect), |ui| {
268            top.show(ui, |ui| body(ui, true))
269        })
270        .inner;
271    let at_newest = shown.state.offset.y >= newest(&shown) - 1.0;
272    split.height = shown.inner_rect.height();
273    split.end = newest(&shown);
274    split.offset = shown.state.offset.y;
275    split.first = first;
276    split.ys = shown.inner.0;
277
278    // The separator: dragged to share the height, its button back to the
279    // newest line.
280    let bar = ui.interact(bar_rect, id.with("bar"), Sense::drag());
281    let bar = bar.on_hover_and_drag_cursor(CursorIcon::ResizeVertical);
282    if bar.dragged() && panes > 0.0 {
283        split.share = (split.share + bar.drag_delta().y / panes).clamp(LEAST, 1.0 - LEAST);
284    }
285    let line = if bar.hovered() || bar.dragged() {
286        Stroke::new(2.0, theme::color(ui.ctx(), T::Accent))
287    } else {
288        ui.visuals().widgets.noninteractive.bg_stroke
289    };
290    ui.painter()
291        .hline(bar_rect.x_range(), bar_rect.center().y, line);
292    let button = egui::Button::new("⬇ Newest").small();
293    let size = vec2(72.0, BAR);
294    let back = ui
295        .put(Rect::from_center_size(bar_rect.center(), size), button)
296        .on_hover_text("Back to the newest line, and one pane again");
297
298    // The bottom: always the newest, which neither the wheel nor a bar
299    // takes it from.
300    // A selection being dragged from the top: the bottom's lines, the same
301    // words under other ids, take no part in it as the pointer crosses them.
302    let from_top = split.from_player && ui.input(|input| input.pointer.primary_down());
303    ui.scope_builder(egui::UiBuilder::new().max_rect(bottom_rect), |ui| {
304        if from_top {
305            ui.style_mut().interaction.selectable_labels = false;
306        }
307        let bottom = pane(wrap)
308            .id_salt(id.with(("bottom", split.turns)))
309            .stick_to_bottom(true)
310            .scroll_source(egui::scroll_area::ScrollSource::NONE)
311            .show(ui, |ui| body(ui, false));
312        redrawn_at_newest(ui, &bottom);
313    });
314    ui.advance_cursor_after_rect(whole);
315
316    if back.clicked() || at_newest || scroll == Some(Scroll::Bottom) {
317        split.open = false;
318        split.turns += 1;
319        split.settling = true;
320    }
321}

323/// A button pressed this frame: whether it was pressed in `pane`, the one
324/// the player scrolls, kept in `from` until the next press.
325fn pressed_in(ui: &egui::Ui, pane: Rect, from: &mut bool) {
326    let pressed = ui.input(|input| {
327        input
328            .pointer
329            .primary_pressed()
330            .then(|| input.pointer.press_origin())
331            .flatten()
332    });
333    if let Some(at) = pressed {
334        *from = pane.contains(at);
335    }
336}

338/// How far a selection being dragged scrolls the pane at `edge` this frame:
339/// back (less than 0) with the pointer above it, forward below it, faster
340/// the farther past; 0 with the pointer within, or with no selection being
341/// dragged `from` the pane. Asks for the next frame while it scrolls, so a
342/// pointer held still keeps scrolling.
343fn dragged_past(ui: &egui::Ui, edge: Rect, from: bool) -> f32 {
344    /// Points a second at the edge itself, and more for each point past it.
345    const AT_EDGE: f32 = 120.0;
346    const PER_POINT: f32 = 10.0;
347    let (down, at, dt) = ui.input(|input| {
348        (
349            input.pointer.primary_down(),
350            input.pointer.latest_pos(),
351            input.stable_dt.min(0.1),
352        )
353    });
354    // Begun in the pane, on its text: the text takes the drag, so it is
355    // known by a selection being made, not by nothing else being dragged.
356    let (true, true, Some(at)) = (from, down, at) else {
357        return 0.0;
358    };
359    let selecting = ui
360        .ctx()
361        .with_plugin(
362            |selection: &mut egui::text_selection::LabelSelectionState| selection.has_selection(),
363        )
364        .unwrap_or(false);
365    if !selecting {
366        return 0.0;
367    }
368    let past = if at.y < edge.top() {
369        at.y - edge.top()
370    } else if at.y > edge.bottom() {
371        at.y - edge.bottom()
372    } else {
373        return 0.0;
374    };
375    let most = edge.height().max(AT_EDGE) * 2.0;
376    let speed = (AT_EDGE + past.abs() * PER_POINT).min(most);
377    ui.ctx().request_repaint();
378    past.signum() * speed * dt
379}

381/// A pane of lines, as big as it is given.
382fn pane(wrap: bool) -> egui::ScrollArea {
383    let area = if wrap {
384        egui::ScrollArea::vertical()
385    } else {
386        egui::ScrollArea::both()
387    };
388    area.min_scrolled_height(0.0).auto_shrink(false)
389}

391/// A pane stuck to its newest line that egui took there only as its frame
392/// ended, `shown` with where its content began: the frame is drawn again,
393/// before it is shown, so its lines are where they now are.
394fn redrawn_at_newest(ui: &egui::Ui, shown: &egui::scroll_area::ScrollAreaOutput<(Vec<f32>, f32)>) {
395    let drawn = shown.inner_rect.top() - shown.inner.1;
396    let now = shown.state.offset.y;
397    if now >= newest(shown) - 0.5 && (now - drawn).abs() > 0.5 {
398        ui.ctx()
399            .request_discard("a pane stuck to its newest line moved to it");
400    }
401}

403/// The offset a pane is at when its newest line is at its bottom.
404fn newest<R>(shown: &egui::scroll_area::ScrollAreaOutput<R>) -> f32 {
405    (shown.content_size.y - shown.inner_rect.height()).max(0.0)
406}

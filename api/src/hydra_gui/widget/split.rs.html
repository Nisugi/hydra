1//! A text widget's scrollback split. Scrolling up splits the widget: the
2//! top is what you are scrolling, and it moves only when you scroll it, not
3//! when new lines come in; the bottom sticks to the newest line as it comes
4//! in. The separator is draggable to set their proportions, and a button on
5//! it scrolls back to the bottom, removing the split.
6//!
7//! One pane, stuck to the newest line, until the player scrolls it back:
8//! then two, the top where the player left it and the bottom following the
9//! newest. The top closes the split too when scrolled back down to the
10//! newest. The wheel scrolls the top wherever
11//! over the widget it is turned, the bottom taking none.
12//!
13//! **The top holds its place when old lines go.** A story keeps its newest
14//! thousand lines, so once full, each new line drops the oldest and the rest
15//! move up by its height. The top is kept on its lines, not its offset: the
16//! widget marks where each line starts ([`Tops`]), and each frame the top
17//! moves down by the height of the lines dropped since the last.
18//!
19//! **The newest line holds the bottom in the frame lines come.** egui
20//! takes a pane stuck to its newest line there only as the pane's frame
21//! ends, after its lines are drawn where the pane was. A line that draws
22//! nothing -- one routed to a stream whose window is open, Hydra's own
23//! beside a Hydra widget -- still drops the oldest from a full story, and
24//! would leave a blank under the newest line until the window next painted;
25//! a new line would show a frame late. The frame is drawn again instead,
26//! before it is shown ([`redrawn_at_newest`]).
27//!
28//! **Keys scroll it too**: a page or a line back opens
29//! the split as the wheel does, and scrolls its top; forward scrolls the top
30//! down, and to the newest line closes it, as its button does. A key's
31//! scroll is asked of a widget by its id ([`ask`]) and done as it is next
32//! drawn, by this or by any widget that scrolls ([`asked`]).

43/// How tall the separator is, to take a drag.
44const BAR: f32 = 14.0;
45
46/// The smallest share of the height either pane is left.
47const LEAST: f32 = 0.1;

49/// A widget's split, kept from frame to frame.
50#[derive(Clone, Debug)]
51struct Split {
52    /// The top pane is open.
53    open: bool,
54    /// The top pane's share of the height.
55    share: f32,
56    /// The top pane's offset when last drawn.
57    offset: f32,
58    /// The top pane is to start at `offset`: it opened this frame, or old
59    /// lines went.
60    place: bool,
61    /// The number of the first line kept when last drawn.
62    first: u64,
63    /// Where each line started when last drawn.
64    ys: Vec<f32>,
65    /// Times the split opened or closed: each pane's id takes it, so each
66    /// begins afresh, and a fresh pane stuck to the newest line starts
67    /// there. An offset set past the end is not pulled back by egui, and
68    /// left the pane empty below its lines.
69    turns: u64,
70    /// The one pane has just come back and is not yet at the newest line:
71    /// it takes no wheel until it is. A fresh pane starts at its top and is
72    /// taken to its end as its first frame closes, which egui skips on a
73    /// frame with a wheel turn in it -- and the rest of the turn that
74    /// scrolled the top down to close the split is such a frame. It left
75    /// the pane far back, and split it again.
76    settling: bool,
77    /// The pane the player scrolls -- the one, or the top -- as last drawn:
78    /// how tall it is, and its offset at the newest line.
79    height: f32,
80    end: f32,
81}
82
83impl Default for Split {
84    fn default() -> Self {
85        Self {
86            open: false,
87            share: 0.6,
88            offset: 0.0,
89            place: false,
90            first: 0,
91            ys: Vec::new(),
92            turns: 0,
93            settling: false,
94            height: 0.0,
95            end: 0.0,
96        }
97    }
98}

100/// A scrolled body of lines, newest at the bottom, split when the player
101/// scrolls back. `first` is the number of the first line kept, which grows
102/// as old lines are dropped; `wrap` whether lines wrap, or the body scrolls
103/// sideways too; `scroll` what a key asked; `add` draws every line, marking
104/// each on its [`Tops`], and may be drawn twice a frame, once a pane.
105pub(super) fn scrolled(
106    ui: &mut egui::Ui,
107    id: Id,
108    (first, wrap): (u64, bool),
109    scroll: Option<Scroll>,
110    mut add: impl FnMut(&mut egui::Ui, &mut Tops),
111) {
112    let state_id = id.with("split");
113    let mut split: Split = ui
114        .data_mut(|data| data.get_temp(state_id))
115        .unwrap_or_default();
116    let mut body = |ui: &mut egui::Ui, player: bool| {
117        if !wrap {
118            ui.style_mut().wrap_mode = Some(egui::TextWrapMode::Extend);
119        }
120        // The theme's room between two lines.
121        ui.spacing_mut().item_spacing.y += crate::theme::line_spacing(ui.ctx());
122        // Where the content began: the pane's offset as it was drawn.
123        let began = ui.max_rect().top();
124        let mut tops = Tops::at(ui, player);
125        add(ui, &mut tops);
126        (tops.keep(ui), began)
127    };
128    if split.open {
129        both(ui, id, &mut split, (first, wrap, scroll), &mut body);
130    } else {
131        let mut area = pane(wrap)
132            .id_salt(id.with(("one", split.turns)))
133            .stick_to_bottom(true);
134        // A key back from the newest line: the pane goes there, and splits.
135        // Scrolled from inside, at once: an offset given the pane is not
136        // the player's scroll to egui, and sticking to the newest line
137        // takes it straight back.
138        let (line, page) = steps(ui, split.height);
139        let back = match scroll {
140            Some(Scroll::PageUp) => Some(page),
141            Some(Scroll::LineUp) => Some(line),
142            Some(Scroll::Top) => Some(split.end),
143            _ => None,
144        };
145        let back = back.filter(|_| split.end > 0.0);
146        if back.is_some() {
147            split.settling = false;
148        }
149        if split.settling {
150            area = area.scroll_source(egui::scroll_area::ScrollSource::NONE);
151        }
152        let shown = area.show(ui, |ui| {
153            if let Some(back) = back {
154                let now = egui::style::ScrollAnimation::none();
155                ui.scroll_with_delta_animation(vec2(0.0, back), now);
156            }
157            body(ui, true)
158        });
159        redrawn_at_newest(ui, &shown);
160        let newest = newest(&shown);
161        split.height = shown.inner_rect.height();
162        split.end = newest;
163        let back = shown.state.offset.y < newest - 1.0;
164        if split.settling {
165            split.settling = back;
166        } else if back {
167            // Scrolled back: the top pane starts where the player is.
168            split.open = true;
169            split.offset = shown.state.offset.y;
170            split.place = true;
171            split.first = first;
172            split.ys = shown.inner.0;
173            split.turns += 1;
174        }
175    }
176    ui.data_mut(|data| data.insert_temp(state_id, split));
177}

179/// Both panes and the separator between them.
180fn both(
181    ui: &mut egui::Ui,
182    id: Id,
183    split: &mut Split,
184    (first, wrap, scroll): (u64, bool, Option<Scroll>),
185    body: &mut impl FnMut(&mut egui::Ui, bool) -> (Vec<f32>, f32),
186) {
187    let whole = ui.available_rect_before_wrap();
188    let panes = (whole.height() - BAR).max(0.0);
189    let top_height = (panes * split.share).round();
190    let top_rect = Rect::from_min_size(whole.min, vec2(whole.width(), top_height));
191    let bar_rect = Rect::from_min_size(top_rect.left_bottom(), vec2(whole.width(), BAR));
192    let bottom_rect = Rect::from_min_max(bar_rect.left_bottom(), whole.max);
193
194    // A key scrolls the top, never past the newest line.
195    let (line, page) = steps(ui, split.height);
196    let moved = match scroll {
197        Some(Scroll::PageUp) => Some(split.offset - page),
198        Some(Scroll::PageDown) => Some(split.offset + page),
199        Some(Scroll::LineUp) => Some(split.offset - line),
200        Some(Scroll::LineDown) => Some(split.offset + line),
201        Some(Scroll::Top) => Some(0.0),
202        #[expect(clippy::cast_precision_loss, reason = "a few points a frame")]
203        Some(Scroll::By(points)) => Some(split.offset - points as f32),
204        Some(Scroll::Bottom) | None => None,
205    };
206    if let Some(to) = moved {
207        split.offset = to.clamp(0.0, split.end.max(0.0));
208        split.place = true;
209    }
210    // The wheel over the separator or the bottom, which takes none, scrolls
211    // the top as over the top itself: where the pointer is does not matter.
212    let below = Rect::from_min_max(bar_rect.min, whole.max);
213    let wheel = ui.input(|input| {
214        input
215            .pointer
216            .hover_pos()
217            .filter(|at| below.contains(*at) && ui.clip_rect().contains(*at))
218            .map_or(0.0, |_| input.smooth_scroll_delta.y)
219    });
220    if wheel != 0.0 {
221        split.offset = (split.offset - wheel).clamp(0.0, split.end.max(0.0));
222        split.place = true;
223    }
224
225    // The top: where the player left it, moved down by the lines dropped.
226    let gone = usize::try_from(first.saturating_sub(split.first)).unwrap_or(usize::MAX);
227    let shift = if gone == 0 {
228        0.0
229    } else {
230        split.ys.get(gone).copied().unwrap_or(split.offset)
231    };
232    let mut top = pane(wrap).id_salt(id.with(("top", split.turns)));
233    if std::mem::take(&mut split.place) || shift > 0.0 {
234        top = top.vertical_scroll_offset((split.offset - shift).max(0.0));
235    }
236    let shown = ui
237        .scope_builder(egui::UiBuilder::new().max_rect(top_rect), |ui| {
238            top.show(ui, |ui| body(ui, true))
239        })
240        .inner;
241    let at_newest = shown.state.offset.y >= newest(&shown) - 1.0;
242    split.height = shown.inner_rect.height();
243    split.end = newest(&shown);
244    split.offset = shown.state.offset.y;
245    split.first = first;
246    split.ys = shown.inner.0;
247
248    // The separator: dragged to share the height, its button back to the
249    // newest line.
250    let bar = ui.interact(bar_rect, id.with("bar"), Sense::drag());
251    let bar = bar.on_hover_and_drag_cursor(CursorIcon::ResizeVertical);
252    if bar.dragged() && panes > 0.0 {
253        split.share = (split.share + bar.drag_delta().y / panes).clamp(LEAST, 1.0 - LEAST);
254    }
255    let line = if bar.hovered() || bar.dragged() {
256        Stroke::new(2.0, theme::color(ui.ctx(), T::Accent))
257    } else {
258        ui.visuals().widgets.noninteractive.bg_stroke
259    };
260    ui.painter()
261        .hline(bar_rect.x_range(), bar_rect.center().y, line);
262    let button = egui::Button::new("⬇ Newest").small();
263    let size = vec2(72.0, BAR);
264    let back = ui
265        .put(Rect::from_center_size(bar_rect.center(), size), button)
266        .on_hover_text("Back to the newest line, and one pane again");
267
268    // The bottom: always the newest, which neither the wheel nor a bar
269    // takes it from.
270    ui.scope_builder(egui::UiBuilder::new().max_rect(bottom_rect), |ui| {
271        let bottom = pane(wrap)
272            .id_salt(id.with(("bottom", split.turns)))
273            .stick_to_bottom(true)
274            .scroll_source(egui::scroll_area::ScrollSource::NONE)
275            .show(ui, |ui| body(ui, false));
276        redrawn_at_newest(ui, &bottom);
277    });
278    ui.advance_cursor_after_rect(whole);
279
280    if back.clicked() || at_newest || scroll == Some(Scroll::Bottom) {
281        split.open = false;
282        split.turns += 1;
283        split.settling = true;
284    }
285}

287/// A pane of lines, as big as it is given.
288fn pane(wrap: bool) -> egui::ScrollArea {
289    let area = if wrap {
290        egui::ScrollArea::vertical()
291    } else {
292        egui::ScrollArea::both()
293    };
294    area.min_scrolled_height(0.0).auto_shrink(false)
295}

297/// A pane stuck to its newest line that egui took there only as its frame
298/// ended, `shown` with where its content began: the frame is drawn again,
299/// before it is shown, so its lines are where they now are.
300fn redrawn_at_newest(ui: &egui::Ui, shown: &egui::scroll_area::ScrollAreaOutput<(Vec<f32>, f32)>) {
301    let drawn = shown.inner_rect.top() - shown.inner.1;
302    let now = shown.state.offset.y;
303    if now >= newest(shown) - 0.5 && (now - drawn).abs() > 0.5 {
304        ui.ctx()
305            .request_discard("a pane stuck to its newest line moved to it");
306    }
307}
308
309/// The offset a pane is at when its newest line is at its bottom.
310fn newest<R>(shown: &egui::scroll_area::ScrollAreaOutput<R>) -> f32 {
311    (shown.content_size.y - shown.inner_rect.height()).max(0.0)
312}

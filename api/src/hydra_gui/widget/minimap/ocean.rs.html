1//! The minimap aboard a ship: the ocean's chart, the ship on it, and the
2//! deck plan small in a corner.
3//!
4//! The chart is the whole ocean, a room a cell, drawn from
5//! [`hydra_map::ocean`]: each room a dot, a line to each room a heading
6//! leads to when that room is the next cell that way, each port ringed and
7//! named, each sea named over its middle. The ship is ringed where it is
8//! ([`Sailing`]), with a tick the way it last moved; in port the ring is on
9//! the port. A ring drawn broken is a place reckoned from the ship's moves,
10//! not one the game stated. A course travel shows (`.sail`) is drawn from
11//! the ship on, and said at the top: where the wheel is turned, the rooms
12//! left, and the heading to
13//! hold next.
14//!
15//! The camera is the minimap's own kind, kept apart from the land map's:
16//! it follows the ship, the wheel zooms, a drag pans, a double-click comes
17//! back. A click on a port asks for its course to be shown, and a click on
18//! the port the course shown rings asks for it to be sailed.
19

29/// Pixels an ocean cell at first.
30const ZOOM: f32 = 16.0;
31/// The deck plan's corner: its share of the view's width, and the most
32/// pixels it takes each way.
33const DECK_SHARE: f32 = 0.4;
34const DECK_MOST: Vec2 = vec2(150.0, 110.0);
35/// Below this many pixels a cell, the seas are named and the ports are not.
36const NAMES_PORTS_FROM: f32 = 6.0;
37
38/// What the minimap draws aboard.
39struct Aboard<'a> {
40    /// Where the ship is.
41    sailing: Sailing,
42    /// The way it last moved.
43    heading: Option<Heading>,
44    /// Where the wheel is turned, as the game put it.
45    wheel: Option<&'a str>,
46    /// The deck plan, and the room of it the character is in.
47    deck: Option<(&'a Arc<MapScene>, u32)>,
48    /// The course shown.
49    course: Option<&'a ShownCourse>,
50    /// Whether marks run along it.
51    flow: bool,
52}

54/// Each sea's name and the middle of its rooms, worked out once.
55fn seas(ocean: &Ocean) -> &'static [(String, Vec2)] {
56    static SEAS: OnceLock<Vec<(String, Vec2)>> = OnceLock::new();
57    SEAS.get_or_init(|| {
58        let mut sums: std::collections::BTreeMap<&str, (Vec2, f32)> =
59            std::collections::BTreeMap::new();
60        for room in ocean.rooms() {
61            let (sum, count) = sums.entry(room.sea.as_str()).or_default();
62            *sum += cell(room.cell);
63            *count += 1.0;
64        }
65        sums.into_iter()
66            .map(|(sea, (sum, count))| (sea.to_owned(), sum / count.max(1.0)))
67            .collect()
68    })
69}

72pub(super) enum Chart {
73    /// The view is not the one aboard a ship: nothing was drawn.
74    NotAboard,
75    /// It was, and this is what a click on it asked for.
76    Drawn(Option<Clicked>),

79/// Draw `view` when it is the view aboard a ship.
80pub(super) fn shown(
81    ui: &egui::Ui,
82    response: &egui::Response,
83    rect: Rect,
84    (id, look): (Id, super::MinimapLook),
85    view: Option<&MinimapView>,
86) -> Chart {
87    let Some(MinimapView::Ocean {
88        sailing,
89        heading,
90        wheel,
91        deck,
92        course,
93    }) = view
94    else {
95        return Chart::NotAboard;
96    };
97    let aboard = Aboard {
98        sailing: *sailing,
99        heading: *heading,
100        wheel: wheel.as_deref(),
101        deck: deck.as_ref().map(|(scene, room)| (scene, *room)),
102        course: course.as_deref(),
103        flow: look.flow,
104    };
105    Chart::Drawn(draw(ui, response, rect, id, &aboard))
106}

108/// Draw the chart into `rect`, already filled with the map's ground.
109fn draw(
110    ui: &egui::Ui,
111    response: &egui::Response,
112    rect: Rect,
113    id: Id,
114    aboard: &Aboard<'_>,
115) -> Option<Clicked> {
116    let painter = ui.painter_at(rect);
117    let ocean = match hydra_map::ocean() {
118        Ok(ocean) => ocean,
119        Err(why) => {
120            waiting(&painter, rect, &format!("No ocean chart: {why}."));
121            return None;
122        }
123    };
124    let (ship, stated) = match aboard.sailing {
125        Sailing::At { room, stated } => (ocean.room(room), stated),
126        Sailing::Docked { port } => (port.and_then(|port| ocean.room(port)), true),
127        Sailing::Ashore | Sailing::Lost => (None, false),
128    };
129    // Until the ship is found, the first port's waters.
130    let home = ocean
131        .ports()
132        .first()
133        .and_then(|port| ocean.room(port.uid))
134        .map_or(Vec2::ZERO, |room| cell(room.cell));
135    let camera = aimed(
136        ui,
137        id.with("ocean"),
138        response,
139        rect,
140        ship.map(|at| cell(at.cell)),
141        home,
142    );
143    let now = ui.input(|i| i.time);
144    let (centre, zoom) = camera.drawn(now);
145    if camera.glide.is_some_and(|(.., start)| now - start < GLIDE) {
146        ui.ctx().request_repaint();
147    }
148    let to_screen = |p: Vec2| rect.center() + (p - centre) * zoom;
149    draw_water(&painter, rect, ocean, &to_screen, zoom);
150    draw_names(&painter, rect, ocean, &to_screen, zoom);
151    let ahead = aboard.course.map(|course| {
152        let how = (zoom, aboard.flow);
153        draw_course(&painter, (rect, ocean), course, ship, &to_screen, how)
154    });
155    if let Some(at) = ship {
156        draw_ship(&painter, to_screen(cell(at.cell)), zoom, stated, aboard);
157    }
158    let note = match aboard.sailing {
159        Sailing::Lost => Some("The ship's place is not known.".to_owned()),
160        Sailing::Docked { port } => Some(port.and_then(|port| ocean.port_at(port)).map_or_else(
161            || "In port.".to_owned(),
162            |port| format!("In port: {}.", port.name),
163        )),
164        Sailing::At { .. } | Sailing::Ashore => ahead.flatten(),
165    };
166    // The wheel first: it is what the player turns.
167    let wheel = aboard
168        .wheel
169        .filter(|_| !matches!(aboard.sailing, Sailing::Docked { .. }))
170        .map(|wheel| format!("Wheel {wheel}."));
171    let note = match (wheel, note) {
172        (Some(wheel), Some(note)) => Some(format!("{wheel} {note}")),
173        (wheel, note) => wheel.or(note),
174    };
175    if let Some(note) = note {
176        // On the map's ground, so the water under it does not show through.
177        let muted = theme::color(ui.ctx(), T::MapMuted);
178        let words = painter.layout_no_wrap(note, FontId::proportional(11.0), muted);
179        let at = rect.left_top() + vec2(4.0, 4.0);
180        let backing = Rect::from_min_size(at, words.size() + vec2(8.0, 4.0));
181        let ground = theme::color(ui.ctx(), T::MapBackground);
182        painter.rect_filled(backing, theme::corner(ui.ctx()), ground);
183        painter.galley(at + vec2(4.0, 2.0), words, muted);
184    }
185    let from_screen = |p: Pos2| centre + (p - rect.center()) / zoom;
186    let pointed = hover(ui, response, ocean, &from_screen);
187    if let Some((scene, room)) = aboard.deck {
188        draw_deck(ui, rect, scene, room);
189    }
190    // A click on a room: its course shown, or sailed when it is shown. A
191    // port by its name; any other room by its number, where the anchor
192    // is dropped.
193    let room = pointed?;
194    let plain = response.clicked() && !response.double_clicked();
195    if !plain || ui.input(|i| i.modifiers.any()) {
196        return None;
197    }
198    let name = ocean
199        .port_at(room)
200        .map_or_else(|| room.to_string(), |port| port.name.clone());
201    let shown = aboard.course.is_some_and(|course| course.port == name);
202    Some(Clicked::Command(if shown {
203        format!("sail {name}")
204    } else {
205        format!("sail route {name}")
206    }))
207}

209/// The camera for this frame, kept for the next: on the ship, following it.
210fn aimed(
211    ui: &egui::Ui,
212    id: Id,
213    response: &egui::Response,
214    rect: Rect,
215    ship: Option<Vec2>,
216    home: Vec2,
217) -> Camera {
218    let now = ui.input(|i| i.time);
219    let kept = ui.data(|d| d.get_temp::<Camera>(id));
220    let mut camera = kept.unwrap_or(Camera {
221        centre: ship.unwrap_or(home),
222        zoom: ZOOM,
223        follow: true,
224        area: 0,
225        inside: false,
226        glide: None,
227    });
228    let before = (camera.centre, camera.zoom);
229    steer(ui, response, rect, &mut camera);
230    if (camera.centre, camera.zoom) != before {
231        camera.glide = None;
232    } else if camera.follow
233        && let Some(ship) = ship
234    {
235        let was = camera.drawn(now);
236        keep_in_view(&mut camera, ship, rect.size(), DEAD_ZONE);
237        if camera.centre != before.0 {
238            camera.glide_from(was, now);
239        }
240    }
241    ui.data_mut(|d| d.insert_temp(id, camera));
242    camera
243}

245/// The rooms and the open water between them.
246fn draw_water(
247    painter: &egui::Painter,
248    rect: Rect,
249    ocean: &Ocean,
250    to_screen: &dyn Fn(Vec2) -> Pos2,
251    zoom: f32,
252) {
253    let line = Stroke::new(
254        1.0,
255        theme::color(painter.ctx(), T::MapLine).gamma_multiply(0.6),
256    );
257    let room_colour = theme::color(painter.ctx(), T::MapRoom);
258    let edge = theme::color(painter.ctx(), T::MapRoomEdge);
259    let dot = (zoom * 0.22).clamp(1.2, 4.5);
260    let seen = rect.expand(zoom);
261    for room in ocean.rooms() {
262        let at = to_screen(cell(room.cell));
263        if !seen.contains(at) {
264            continue;
265        }
266        // Each link once: the four headings that lead east or south.
267        for heading in [
268            Heading::East,
269            Heading::Southeast,
270            Heading::South,
271            Heading::Southwest,
272        ] {
273            if let Some(there) = neighbour(ocean, room, heading) {
274                painter.line_segment([at, to_screen(cell(there.cell))], line);
275            }
276        }
277    }
278    for room in ocean.rooms() {
279        let at = to_screen(cell(room.cell));
280        if seen.contains(at) {
281            painter.circle(at, dot, room_colour, Stroke::new(0.8, edge));
282        }
283    }
284}

286/// The room `heading` leads to from `room`, when it is the next cell that
287/// way: a heading that leads anywhere else is not a line on the chart.
288fn neighbour<'a>(ocean: &'a Ocean, room: &OceanRoom, heading: Heading) -> Option<&'a OceanRoom> {
289    let (dx, dy) = heading.step();
290    let next = (room.cell.0 + dx, room.cell.1 + dy);
291    let there = room
292        .toward(heading)
293        .and_then(|uid| ocean.room(uid))
294        .filter(|there| there.cell == next);
295    // A link told from the far side only is one too.
296    there.or_else(|| {
297        ocean
298            .at(next)
299            .filter(|there| there.toward(heading.back()) == Some(room.uid))
300    })
301}

303/// The seas' names, and the ports ringed and named.
304fn draw_names(
305    painter: &egui::Painter,
306    rect: Rect,
307    ocean: &Ocean,
308    to_screen: &dyn Fn(Vec2) -> Pos2,
309    zoom: f32,
310) {
311    // Each name on a patch of the map's ground, so the water does not
312    // run through its letters; and none over another: a name that would
313    // land on one placed before it moves down until it is clear.
314    let muted = theme::color(painter.ctx(), T::MapMuted);
315    let ground = theme::color(painter.ctx(), T::MapBackground);
316    let font = FontId::proportional((zoom * 0.9).clamp(10.0, 15.0));
317    let mut placed: Vec<Rect> = Vec::new();
318    for (sea, middle) in seas(ocean) {
319        let at = to_screen(*middle);
320        if !rect.expand(60.0).contains(at) {
321            continue;
322        }
323        let words = painter.layout_no_wrap(sea.clone(), font.clone(), muted);
324        let size = words.size() + vec2(8.0, 2.0);
325        let mut backing = Rect::from_center_size(at, size);
326        for _ in 0..8 {
327            match placed.iter().find(|taken| taken.intersects(backing)) {
328                Some(taken) => {
329                    backing = backing.translate(vec2(0.0, taken.max.y - backing.min.y + 2.0));
330                }
331                None => break,
332            }
333        }
334        placed.push(backing);
335        painter.rect_filled(
336            backing,
337            theme::corner(painter.ctx()),
338            ground.gamma_multiply(0.85),
339        );
340        painter.galley(backing.min + vec2(4.0, 1.0), words, muted);
341    }
342    let door = theme::color(painter.ctx(), T::MapDoor);
343    for port in ocean.ports() {
344        let Some(room) = ocean.room(port.uid) else {
345            continue;
346        };
347        let at = to_screen(cell(room.cell));
348        if !rect.expand(zoom).contains(at) {
349            continue;
350        }
351        let r = (zoom * 0.4).clamp(3.0, 7.0);
352        painter.circle_filled(at, r, door);
353        if zoom >= NAMES_PORTS_FROM {
354            painter.text(
355                at + vec2(r + 3.0, 0.0),
356                egui::Align2::LEFT_CENTER,
357                &port.name,
358                FontId::proportional(11.0),
359                door,
360            );
361        }
362    }
363}

365/// The course from the ship on, in the route's colour: a line room to
366/// room and a ring on the port. What it says of it: the rooms left, and
367/// the heading to hold next and for how many rooms.
368fn draw_course(
369    painter: &egui::Painter,
370    (rect, ocean): (Rect, &Ocean),
371    course: &ShownCourse,
372    ship: Option<&OceanRoom>,
373    to_screen: &dyn Fn(Vec2) -> Pos2,
374    (zoom, flow): (f32, bool),
375) -> Option<String> {
376    let rooms = ship.map_or(course.rooms.as_slice(), |ship| course.from(ship.uid));
377    let cells: Vec<(i32, i32)> = rooms
378        .iter()
379        .filter_map(|uid| ocean.room(*uid))
380        .map(|room| room.cell)
381        .collect();
382    if cells.len() < 2 {
383        return None;
384    }
385    let colour = theme::color(painter.ctx(), T::MapRoute);
386    let width = (zoom * 0.2).clamp(2.0, 4.0);
387    let points: Vec<Pos2> = cells.iter().map(|at| to_screen(cell(*at))).collect();
388    painter.add(Shape::line(points.clone(), Stroke::new(width, colour)));
389    if flow {
390        super::route::draw_flow(painter, rect, &points, colour, width);
391    }
392    if let Some(port) = points.last() {
393        painter.circle_stroke(
394            *port,
395            (zoom * 0.4).clamp(3.0, 7.0) + 4.0,
396            Stroke::new(2.5, colour),
397        );
398    }
399    // The first run: the heading out of the first room, held while it lasts.
400    let step = |pair: &[(i32, i32)]| (pair[1].0 - pair[0].0, pair[1].1 - pair[0].1);
401    let first = cells.windows(2).next().map(step)?;
402    let run = cells
403        .windows(2)
404        .take_while(|pair| step(pair) == first)
405        .count();
406    let left = cells.len() - 1;
407    let rooms_word = |count: usize| if count == 1 { "room" } else { "rooms" };
408    let hold = Heading::ALL
409        .into_iter()
410        .find(|heading| heading.step() == first)
411        .map_or_else(String::new, |heading| {
412            format!(" Hold {} for {run}.", heading.word())
413        });
414    Some(format!(
415        "To {}: {left} {}.{hold}",
416        course.port,
417        rooms_word(left)
418    ))
419}

421/// The ship: a ring, broken when its place is reckoned, and a tick the way
422/// it last moved.
423fn draw_ship(painter: &egui::Painter, at: Pos2, zoom: f32, stated: bool, aboard: &Aboard<'_>) {
424    let you = theme::color(painter.ctx(), T::MapYou);
425    let r = (zoom * 0.45).clamp(5.0, 11.0);
426    let stroke = Stroke::new(2.5, you);
427    if stated {
428        painter.circle_stroke(at, r, stroke);
429    } else {
430        let around: Vec<Pos2> = (0..=24u8)
431            .map(|step| {
432                let angle = f32::from(step) / 24.0 * std::f32::consts::TAU;
433                at + vec2(angle.cos(), angle.sin()) * r
434            })
435            .collect();
436        painter.extend(Shape::dashed_line(&around, stroke, 4.0, 3.0));
437    }
438    let at_sea = matches!(aboard.sailing, Sailing::At { .. });
439    if let Some(heading) = aboard.heading.filter(|_| at_sea) {
440        let (dx, dy) = heading.step();
441        #[allow(clippy::cast_precision_loss)]
442        let way = vec2(dx as f32, dy as f32).normalized();
443        painter.line_segment([at + way * r, at + way * (r + r * 0.9)], stroke);
444    }
445}

447/// The card for the ocean room under the pointer: its sea, its number, and
448/// its port; not while a button is down or let go, or the click would land
449/// on it. The room pointed at, card or none.
450fn hover(
451    ui: &egui::Ui,
452    response: &egui::Response,
453    ocean: &Ocean,
454    from_screen: &dyn Fn(Pos2) -> Vec2,
455) -> Option<i64> {
456    let pressing = ui.input(|i| i.pointer.any_down() || i.pointer.any_released());
457    let room = response
458        .hover_pos()
459        .or_else(|| response.interact_pointer_pos())
460        .map(from_screen)
461        .and_then(|at| {
462            #[allow(clippy::cast_possible_truncation)]
463            let nearest = (at.x.round() as i32, at.y.round() as i32);
464            ocean
465                .at(nearest)
466                .filter(|room| (cell(room.cell) - at).length() <= 0.45)
467        })?;
468    if pressing {
469        return Some(room.uid);
470    }
471    let mut lines = vec![
472        format!("{}, {}", room.sea, room.ocean),
473        room.uid.to_string(),
474    ];
475    if let Some(port) = ocean.port_at(room.uid) {
476        lines.insert(0, format!("Port: {}", port.name));
477    }
478    response.clone().on_hover_ui_at_pointer(|ui| {
479        for line in lines {
480            ui.add(egui::Label::new(line).selectable(false));
481        }
482    });
483    Some(room.uid)
484}

486/// The deck plan, fitted into the view's lower right corner, the room the
487/// character is in ringed.
488fn draw_deck(ui: &egui::Ui, rect: Rect, scene: &MapScene, you: u32) {
489    let Some(first) = scene.rooms.first() else {
490        return;
491    };
492    let size = vec2(rect.width() * DECK_SHARE, rect.height() * DECK_SHARE).min(DECK_MOST);
493    let corner = Rect::from_min_size(rect.right_bottom() - size - vec2(4.0, 4.0), size);
494    let painter = ui.painter_at(corner);
495    let ground = theme::color(ui.ctx(), T::MapBackground);
496    let edge = theme::color(ui.ctx(), T::MapRoomEdge);
497    painter.rect(
498        corner,
499        theme::corner(ui.ctx()),
500        ground,
501        Stroke::new(1.0, edge.gamma_multiply(0.6)),
502        egui::StrokeKind::Inside,
503    );
504    let (mut least, mut most) = (cell(first.cell), cell(first.cell));
505    for room in &scene.rooms {
506        least = least.min(cell(room.cell));
507        most = most.max(cell(room.cell));
508    }
509    let span = (most - least).max(vec2(1.0, 1.0));
510    let room_for = (corner.size() - vec2(28.0, 28.0)).max(vec2(8.0, 8.0));
511    let zoom = (room_for.x / span.x).min(room_for.y / span.y).min(12.0);
512    let middle = (least + most) / 2.0;
513    let to_screen = |p: Vec2| corner.center() + (p - middle) * zoom;
514    draw_lines(&painter, scene, &to_screen, zoom, &|_| 1.0);
515    draw_rooms(&painter, corner, scene, &to_screen, zoom, &|_| 1.0);
516    if let Some(here) = scene.room(you) {
517        let square = (zoom * 0.7).clamp(3.0, 10.0);
518        painter.circle_stroke(
519            to_screen(cell(here.cell)),
520            square * 0.9 + 2.0,
521            Stroke::new(2.0, theme::color(ui.ctx(), T::MapYou)),
522        );
523    }
524}

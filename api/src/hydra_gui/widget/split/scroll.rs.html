1//! A key's scroll: asked of a widget by its id, and done as it is next
2//! drawn, by the split or by any widget that scrolls.
3
4use egui::{Id, vec2};
5
6/// A scroll a key asked of the window in use.
7#[derive(Clone, Copy, Debug, PartialEq, Eq)]
8pub(crate) enum Scroll {
9    /// Back a page, to older lines.
10    PageUp,
11    /// Forward a page.
12    PageDown,
13    /// Back a line.
14    LineUp,
15    /// Forward a line.
16    LineDown,
17    /// To the oldest line kept.
18    Top,
19    /// To the newest line.
20    Bottom,
21    /// Back this many points, or forward for fewer than none: a
22    /// controller's stick, scrolling.
23    By(i32),
24}
25
26/// Where a widget's asked scroll is kept until it is drawn.
27pub(super) const ASKED: &str = "keyed-scroll";
28
29/// Ask the widget whose id is `id` to scroll, as it is next drawn.
30pub(crate) fn ask(context: &egui::Context, id: Id, scroll: Scroll) {
31    context.data_mut(|data| data.insert_temp(id.with(ASKED), scroll));
32}
33
34/// The scroll asked of the widget whose id is `id`, taken.
35pub(crate) fn asked(ui: &egui::Ui, id: Id) -> Option<Scroll> {
36    ui.data_mut(|data| {
37        let asked = data.get_temp::<Scroll>(id.with(ASKED));
38        data.remove::<Scroll>(id.with(ASKED));
39        asked
40    })
41}
42
43/// A line's height and a page's, in `ui`'s text, for a pane `height` tall:
44/// a page keeps a line of the last in sight.
45pub(super) fn steps(ui: &egui::Ui, height: f32) -> (f32, f32) {
46    let line = ui.text_style_height(&egui::TextStyle::Body) + ui.spacing().item_spacing.y;
47    (line, (height - line).max(line))
48}
49
50/// A body of lines, not split, scrolled as a key asked: inside its
51/// scrolling area, before its lines are added (`Top` and the steps), or
52/// after (`Bottom`).
53pub(crate) fn keyed(ui: &mut egui::Ui, scroll: Option<Scroll>, add: impl FnOnce(&mut egui::Ui)) {
54    let (line, page) = steps(ui, ui.clip_rect().height());
55    let by = match scroll {
56        Some(Scroll::PageUp) => page,
57        Some(Scroll::PageDown) => -page,
58        Some(Scroll::LineUp) => line,
59        Some(Scroll::LineDown) => -line,
60        #[expect(clippy::cast_precision_loss, reason = "a few points a frame")]
61        Some(Scroll::By(points)) => points as f32,
62        Some(Scroll::Top | Scroll::Bottom) | None => 0.0,
63    };
64    if scroll == Some(Scroll::Top) {
65        ui.scroll_to_cursor(Some(egui::Align::TOP));
66    } else if by != 0.0 {
67        ui.scroll_with_delta_animation(vec2(0.0, by), egui::style::ScrollAnimation::none());
68    }
69    add(ui);
70    if scroll == Some(Scroll::Bottom) {
71        ui.scroll_to_cursor(Some(egui::Align::BOTTOM));
72    }
73}

75/// Where each line of a pane starts, from the top of its content: what
76/// keeps the top pane on its lines as old ones are dropped.
77///
78/// **Only the lines near the view are drawn** ([`Self::line`]). A story
79/// keeps a thousand lines, and laying every one out every frame is most of
80/// a frame's work; a line drawn once is known by its height, and while it
81/// is well out of sight only that room is kept for it. Every line is drawn
82/// while text is selected, since a copy takes only what is drawn.
83#[derive(Debug)]
84pub(crate) struct Tops {
85    /// The content's top on screen, when the pane began.
86    origin: f32,
87    /// Where each line kept starts, in order.
88    pub(super) ys: Vec<f32>,
89    /// Its pane is the one the player scrolls: the one, or the top.
90    player: bool,
91    /// What it remembers of its lines' heights, and where it keeps them.
92    heights: Heights,
93    heights_id: egui::Id,
94    /// Lines out of sight may be left undrawn: no text is selected.
95    may_skip: bool,
96    /// What each line's id is made under: the same for the one pane and the
97    /// top, so a selection begun in the one goes on in the top as the
98    /// widget splits under a drag.
99    lines: egui::Id,
100}
101
102/// Each line's height by its number, at one width and one size of text: a
103/// change to either lays every line out anew.
104#[derive(Clone, Debug, Default)]
105struct Heights {
106    width: f32,
107    row: f32,
108    /// What else lays a line out: the line spacing and the wrap, hashed.
109    look: u64,
110    /// The widget's own look (times, wrapping), hashed.
111    widget: u64,
112    of: std::collections::HashMap<u64, f32>,
113    /// The lowest number drawn or kept this frame: lines below it are gone.
114    lowest: Option<u64>,
115}
116
117/// What else lays a line out, hashed: the line spacing, the wrap, and the
118/// story's font and the monospace one as well as the body's -- the game's
119/// lines are drawn in the story's, its tables in the monospace one, each
120/// sized on its own.
121fn look_of(ui: &egui::Ui) -> u64 {
122    let story = crate::theme::story_font(ui.style());
123    let mono = egui::TextStyle::Monospace.resolve(ui.style());
124    egui::util::hash((
125        ui.spacing().item_spacing.y.to_bits(),
126        format!("{:?}", ui.wrap_mode()),
127        format!("{story:?}"),
128        format!("{mono:?}"),
129    ))
130}
131
132impl Tops {
133    /// Marks from the top of `ui`'s content, in the pane the player
134    /// scrolls (`player`) or the other.
135    pub(super) fn at(ui: &egui::Ui, player: bool, lines: egui::Id) -> Self {
136        let heights_id = ui.id().with("heights");
137        let width = ui.available_width();
138        let row = ui.text_style_height(&egui::TextStyle::Body);
139        let mut heights: Heights = ui
140            .data_mut(|data| data.get_temp(heights_id))
141            .unwrap_or_default();
142        let look = look_of(ui);
143        if (heights.width - width).abs() > 0.5
144            || (heights.row - row).abs() > 0.01
145            || heights.look != look
146        {
147            heights = Heights {
148                width,
149                row,
150                look,
151                ..Heights::default()
152            };
153        }
154        heights.lowest = None;
155        let selecting = ui
156            .ctx()
157            .with_plugin(
158                |selection: &mut egui::text_selection::LabelSelectionState| {
159                    selection.has_selection()
160                },
161            )
162            .unwrap_or(false);
163        Self {
164            origin: ui.cursor().top(),
165            ys: Vec::new(),
166            player,
167            heights,
168            heights_id,
169            may_skip: !selecting,
170            lines,
171        }
172    }
173
174    /// The id line `number` is drawn under: the line's own, whichever pane
175    /// the player scrolls draws it, so a selection is kept as the widget
176    /// splits or comes back to one pane.
177    pub(crate) fn line_id(&self, number: u64) -> egui::Id {
178        self.lines.with(number)
179    }
180
181    /// The widget's own look, `look`: a change to it lays every line out
182    /// anew, since a time before each line, or wrapping, changes its height.
183    pub(crate) fn look(&mut self, look: &impl std::hash::Hash) {
184        let widget = egui::util::hash(look);
185        if self.heights.widget != widget {
186            self.heights.of.clear();
187            self.heights.widget = widget;
188        }
189    }
190
191    /// Draw line `number` with `draw`; or, when it has been drawn before at
192    /// this width, `skip` allows it, and it is more than a view's height out
193    /// of sight, keep only its room.
194    pub(crate) fn line(
195        &mut self,
196        ui: &mut egui::Ui,
197        number: u64,
198        skip: bool,
199        draw: impl FnOnce(&mut egui::Ui),
200    ) {
201        self.heights.lowest = Some(self.heights.lowest.map_or(number, |low| low.min(number)));
202        let top = ui.cursor().top();
203        let view = ui.clip_rect();
204        if self.may_skip
205            && skip
206            && let Some(&height) = self.heights.of.get(&number)
207            && (top + height < view.top() - view.height() || top > view.bottom() + view.height())
208        {
209            ui.add_space(height);
210            return;
211        }
212        draw(ui);
213        self.heights.of.insert(number, ui.cursor().top() - top);
214    }
215
216    /// Keep what it learned of its lines' heights for the next frame, less
217    /// the lines that are gone.
218    pub(super) fn keep(mut self, ui: &egui::Ui) -> Vec<f32> {
219        if let Some(lowest) = self.heights.lowest {
220            self.heights.of.retain(|number, _| *number >= lowest);
221        }
222        let (id, heights) = (self.heights_id, self.heights);
223        ui.data_mut(|data| data.insert_temp(id, heights));
224        self.ys
225    }
226
227    /// Whether its pane is the one the player scrolls.
228    pub(crate) fn player(&self) -> bool {
229        self.player
230    }
231
232    /// The next line kept starts here, drawn or not.
233    pub(crate) fn mark(&mut self, ui: &egui::Ui) {
234        self.ys.push(ui.cursor().top() - self.origin);
235    }
236}

238#[cfg(test)]
239mod look_tests {
240    /// The story's font sizes the game's lines: a change to it alone lays
241    /// every line out anew.
242    #[test]
243    fn the_monospace_font_is_part_of_the_look() {
244        let looks: Vec<u64> = [12.0, 24.0]
245            .into_iter()
246            .map(|size| {
247                let look = std::cell::Cell::new(0);
248                let mut harness = egui_kittest::Harness::new_ui(|ui| {
249                    ui.style_mut()
250                        .text_styles
251                        .insert(egui::TextStyle::Monospace, egui::FontId::monospace(size));
252                    look.set(super::look_of(ui));
253                });
254                harness.run();
255                drop(harness);
256                look.get()
257            })
258            .collect();
259        assert_ne!(looks[0], looks[1]);
260    }
261

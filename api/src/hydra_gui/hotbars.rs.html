1//! A character's hotbars: every character's `hotbars.toml` in the data
2//! folder, with the character's own beside its layout over it
3//! (`hydra_ui::hotbar`). Read off the window's thread (`crate::disk`),
4//! again within a second of either changing.
5//!
6//! **A button moved** on its widget is written to the file its bar came
7//! from, the character's own when it has that bar: each of the bar's
8//! buttons given the slot it now has, the file's own text kept --
9//! comments, order, what Hydra does not read.
10//!
11//! **A button's hotkey** presses it in its character's window, as a click
12//! would: Vellum's way of writing one (`alt+h`, `f5`) or the keybinds
13//! file's (`Alt+KeyH`). A key a keybind holds is the keybind's, and the
14//! button says so; of two buttons on one key, the first in the bars' order
15//! has it.
16//!
17//! **`.hotbar <bar> <button>`** presses a button by its id or nickname,
18//! whatever slot it is in: what it sends is what a click would, the
19//! command its state now chooses.
20

37/// What the disk keeps a hotbars file, or the icons folder's sheets, under.
38fn key(path: &Path) -> String {
39    format!("hotbars:{}", path.display())

42/// One character's hotbars, as its files have them.
43#[derive(Debug, Default)]
44pub(crate) struct Hotbars {
45    /// Every character's file, and the character's own; none, and nothing
46    /// is read.
47    paths: Option<(PathBuf, PathBuf)>,
48    /// The bars, the character's over every character's.
49    pub(crate) file: HotbarFile,
50    /// The character's own bars alone: which file a bar is written to.
51    own: HotbarFile,
52    /// Why a file does not read, when one does not.
53    problem: Option<String>,
54    /// The icons folder's sheets (`hydra_ui::hotbar::Sheets`).
55    pub(crate) sheets: hydra_ui::hotbar::Sheets,
56    /// Each file's text as Hydra has it -- as last read, with the player's
57    /// changes made -- by [`Text`]; none before the first read.
58    texts: [Option<Arc<Result<String, String>>>; 3],
59    /// The disk's reads last taken, by their generations.
60    seen: [Option<u64>; 3],
61    /// Each button's hotkey, in the bars' order.
62    hotkeys: Vec<Hotkey>,
63    /// The buttons' keys a keybind holds, which are the keybind's.
64    taken: BTreeSet<Chord>,
65}

67/// Which file's text.
68#[derive(Clone, Copy, Debug, PartialEq, Eq)]
69enum Text {
70    /// Every character's hotbars.
71    Every = 0,
72    /// The character's own.
73    Own = 1,
74    /// The icons folder's sheets.
75    Sheets = 2,
76}

78/// A button's hotkey: the button, the key as written, and the key or why it
79/// is none.
80#[derive(Clone, Debug)]
81struct Hotkey {
82    bar: String,
83    id: String,
84    written: String,
85    chord: Result<Chord, String>,

89    /// The character `name` on `instance`'s hotbars, its layouts kept in
90    /// `layouts`, inside the data folder: read off the window's thread, so
91    /// there are none until they have been.
92    pub(crate) fn load(layouts: &Path, instance: Option<&str>, name: &str) -> Self {
93        let every = layouts.parent().unwrap_or(layouts).join(FILE);
94        let own = character_path(layouts, instance, name);
95        let mut hotbars = Self {
96            paths: Some((every, own)),
97            ..Self::default()
98        };
99        // A window opening reads them afresh, not as another left them.
100        hotbars.reload();
101        hotbars
102    }

104    /// Each file's path, by [`Text`].
105    fn path(&self, text: Text) -> Option<PathBuf> {
106        let (every, own) = self.paths.as_ref()?;
107        Some(match text {
108            Text::Every => every.clone(),
109            Text::Own => own.clone(),
110            Text::Sheets => self.icons_dir().join(SHEETS),
111        })
112    }

114    /// The files as last read off the window's thread (`crate::disk`),
115    /// taken when any of them changed since, unless a change of the
116    /// player's own waits to be written to it; `repaint` drawn again when a
117    /// read is made.
118    pub(crate) fn fresh(&mut self, repaint: Option<&egui::Context>) {
119        let disk = crate::disk::Disk::hydras();
120        let repaint = repaint.map(crate::repaint);
121        let mut changed = false;
122        for text in [Text::Every, Text::Own, Text::Sheets] {
123            let Some(path) = self.path(text) else {
124                return;
125            };
126            // Shared, not copied: three files' texts would otherwise be
127            // copied out for every open window every frame.
128            let read = disk.kept(&key(&path), &path, repaint.as_ref(), |path| {
129                Some(Arc::new(read_or_empty(path)))
130            });
131            let Some((generation, read)) = read else {
132                continue;
133            };
134            let at = text as usize;
135            if self.seen[at] != Some(generation) && !crate::store::pending(&path) {
136                self.seen[at] = Some(generation);
137                self.texts[at] = Some(read);
138                changed = true;
139            }
140        }
141        if changed && self.texts.iter().all(Option::is_some) {
142            self.parse();
143        }
144    }

146    /// The icons folder: `icons` in the data folder.
147    pub(crate) fn icons_dir(&self) -> PathBuf {
148        self.paths
149            .as_ref()
150            .and_then(|(every, _)| every.parent())
151            .map_or_else(|| PathBuf::from(ICONS), |data| data.join(ICONS))

154    /// The bars and the sheets, as the texts have them.
155    fn parse(&mut self) {
156        let named = |text: Text| {
157            self.path(text)
158                .and_then(|path| {
159                    path.file_name()
160                        .map(|name| name.to_string_lossy().into_owned())
161                })
162                .unwrap_or_default()
163        };
164        let text = |at: Text| match self.texts[at as usize].as_deref() {
165            Some(Ok(text)) => Ok(text.as_str()),
166            Some(Err(why)) => Err(why.clone()),
167            None => Ok(""),
168        };
169        let bars = |at: Text| {
170            text(at).and_then(|text| {
171                HotbarFile::parse(text).map_err(|why| format!("{}: {why}", named(at)))
172            })
173        };
174        let (every, own) = (bars(Text::Every), bars(Text::Own));
175        let sheets = text(Text::Sheets).and_then(|text| {
176            hydra_ui::hotbar::Sheets::parse(text).map_err(|why| format!("{SHEETS}: {why}"))
177        });
178        self.problem = [
179            every.as_ref().err(),
180            own.as_ref().err(),
181            sheets.as_ref().err(),
182        ]
183        .into_iter()
184        .flatten()
185        .cloned()
186        .reduce(|one, other| format!("{one}; {other}"));
187        self.sheets = sheets.unwrap_or_default();
188        self.own = own.unwrap_or_default();
189        self.file = every.unwrap_or_default().merged(self.own.clone());
190        self.hotkeys = self
191            .file
192            .bars
193            .iter()
194            .flat_map(|bar| {
195                bar.buttons.iter().filter_map(|button| {
196                    let written = button.hotkey.as_deref()?.trim();
197                    (!written.is_empty()).then(|| Hotkey {
198                        bar: bar.name.clone(),
199                        id: button.id.clone(),
200                        written: written.to_owned(),
201                        chord: hotkey(written),
202                    })
203                })
204            })
205            .collect();
206    }

208    /// Why a file does not read, or a change to one was not written, when
209    /// that is so.
210    pub(crate) fn problem(&self) -> Option<String> {
211        let unwritten = [Text::Every, Text::Own, Text::Sheets]
212            .into_iter()
213            .filter_map(|text| self.path(text))
214            .filter_map(|path| crate::store::failed(&path))
215            .map(|why| format!("not saved: {why}"));
216        self.problem
217            .clone()
218            .into_iter()
219            .chain(unwritten)
220            .reduce(|one, other| format!("{one}; {other}"))
221    }

223    /// `edit` made to a file's text at once, and to the file with the next
224    /// batch of saves (`crate::store`).
225    ///
226    /// # Errors
227    ///
228    /// Nothing is kept here, the file does not read, or the edit cannot be
229    /// made to it.
230    fn change(
231        &mut self,
232        text: Text,
233        edit: impl Fn(&str) -> Result<String, String> + Clone + Send + 'static,
234    ) -> Result<(), String> {
235        let path = self.path(text).ok_or("Nothing is saved here.")?;
236        let now = match self.texts[text as usize].as_deref() {
237            Some(Ok(now)) => now.clone(),
238            Some(Err(why)) => return Err(why.clone()),
239            None => return Err("the hotbars are not read yet".to_owned()),
240        };
241        let changed = edit(&now)?;
242        self.texts[text as usize] = Some(Arc::new(Ok(changed)));
243        self.parse();
244        let dir = path.parent().unwrap_or(Path::new(".")).to_owned();
245        crate::store::edit(&dir, &path, Box::new(move |now: &str| edit(now)));
246        Ok(())
247    }

249    /// Which of the buttons' keys a keybind holds: `held`.
250    pub(crate) fn keybinds_hold(&mut self, held: impl Fn(&Chord) -> bool) {
251        self.taken = self
252            .hotkeys
253            .iter()
254            .filter_map(|hotkey| hotkey.chord.as_ref().ok())
255            .filter(|chord| held(chord))
256            .cloned()
257            .collect();
258    }

260    /// The button `chord` presses, by its bar and id: the first with that
261    /// key, when no keybind holds it.
262    pub(crate) fn pressed_by(&self, chord: &Chord) -> Option<(&str, &str)> {
263        if self.taken.contains(chord) {
264            return None;
265        }
266        self.hotkeys
267            .iter()
268            .find(|hotkey| hotkey.chord.as_ref() == Ok(chord))
269            .map(|hotkey| (hotkey.bar.as_str(), hotkey.id.as_str()))
270    }

272    /// The bar `bar`'s button `id`'s hotkey, as the keybinds file writes a
273    /// key; or why it presses nothing. None when it has none.
274    pub(crate) fn hotkey_of(&self, bar: &str, id: &str) -> Option<Result<String, String>> {
275        let hotkey = self
276            .hotkeys
277            .iter()
278            .find(|hotkey| hotkey.bar == bar && hotkey.id == id)?;
279        let written = &hotkey.written;
280        Some(match &hotkey.chord {
281            Err(why) => Err(format!("its key {written} presses nothing: {why}")),
282            Ok(chord) if self.taken.contains(chord) => Err(format!(
283                "its key {} presses nothing: a keybind holds it, and wins",
284                chord.written()
285            )),
286            Ok(chord) => match self.pressed_by(chord) {
287                Some((first_bar, first)) if (first_bar, first) != (bar, id) => Err(format!(
288                    "its key {} presses nothing: {first_bar}'s {first} has it first",
289                    chord.written()
290                )),
291                _ => Ok(chord.written()),
292            },
293        })
294    }

296    /// Take from `input` every key press a button's hotkey has, when
297    /// `bound` (a keybind) does not hold it: the buttons pressed, in order.
298    pub(crate) fn take(
299        &self,
300        input: &mut egui::InputState,
301        bound: impl Fn(&Chord) -> bool,
302    ) -> Vec<Macro> {
303        if self.hotkeys.is_empty() {
304            return Vec::new();
305        }
306        let mut pressed = Vec::new();
307        input.events.retain(|event| {
308            let egui::Event::Key {
309                key,
310                physical_key,
311                pressed: true,
312                repeat,
313                modifiers,
314            } = event
315            else {
316                return true;
317            };
318            let Some(name) = crate::keys::winit_name(physical_key.unwrap_or(*key)) else {
319                return true;
320            };
321            let chord = Chord::of(&name, *modifiers);
322            if bound(&chord) {
323                return true;
324            }
325            let Some((bar, button)) = self.pressed_by(&chord) else {
326                return true;
327            };
328            if !*repeat {
329                pressed.push(Macro::Hotbar {
330                    bar: bar.to_owned(),
331                    button: button.to_owned(),
332                });
333            }
334            false
335        });
336        pressed
337    }

344    /// The bar `name`'s button in slot `from` moved to slot `to`, at once
345    /// and with the next batch of saves to the file the bar came from; why
346    /// not, as the files' problem, until they are read again.
347    pub(crate) fn move_button(&mut self, name: &str, from: u16, to: u16) {
348        if let Err(why) = self.moved(name, from, to) {
349            self.problem = Some(format!("The button was not moved: {why}"));
350        }
351    }

353    fn moved(&mut self, name: &str, from: u16, to: u16) -> Result<(), String> {
354        let text = if self.own.bar(name).is_some() {
355            Text::Own
356        } else {
357            Text::Every
358        };
359        let Some(bar) = self.file.bar(name) else {
360            return Err(format!("There is no hotbar {name}."));
361        };
362        let mut moved = bar.clone();
363        moved.move_button(from, to);
364        let slots: Vec<Option<u16>> = moved.buttons.iter().map(|button| button.slot).collect();
365        let name = moved.name;
366        self.change(text, move |now| slotted(now, &name, &slots))
367    }

369    /// The files read again now, off the window's thread.
370    pub(crate) fn reload(&mut self) {
371        let disk = crate::disk::Disk::hydras();
372        for text in [Text::Every, Text::Own, Text::Sheets] {
373            if let Some(path) = self.path(text) {
374                disk.look_again(&key(&path));
375            }
376        }
377        self.fresh(None);
378    }

386    /// `bar` kept in the character's own file (`own`) or every
387    /// character's, in place of the bar `was` (its name, and whether it was
388    /// the character's own), the rest of each file kept: at once, and with
389    /// the next batch of saves.
390    ///
391    /// # Errors
392    ///
393    /// Nothing is saved here, a file does not read or would not, or another
394    /// bar has the name.
395    pub(crate) fn save_bar(
396        &mut self,
397        bar: &Bar,
398        own: bool,
399        was: Option<(&str, bool)>,
400    ) -> Result<(), String> {
401        let file = |own: bool| if own { Text::Own } else { Text::Every };
402        let replacing = was
403            .filter(|(_, was_own)| *was_own == own)
404            .map(|(name, _)| name.to_owned());
405        let kept = bar.clone();
406        self.change(file(own), move |now| {
407            with_bar(now, &kept, replacing.as_deref())
408        })?;
409        // Moved to the other file: taken out of the one it was in.
410        if let Some((name, was_own)) = was.filter(|(_, was_own)| *was_own != own) {
411            let name = name.to_owned();
412            self.change(file(was_own), move |now| without_bar(now, &name))?;
413        }
414        Ok(())
415    }

417    /// The bar `name` taken out of the character's own file (`own`) or
418    /// every character's: at once, and with the next batch of saves.
419    ///
420    /// # Errors
421    ///
422    /// Nothing is saved here, or the file does not read.
423    pub(crate) fn remove_bar(&mut self, name: &str, own: bool) -> Result<(), String> {
424        let file = if own { Text::Own } else { Text::Every };

429    /// The image at `file` copied into the icons folder, and named a sheet
430    /// of cells `cell` pixels on edge in its `icons.toml` when `sheet`
431    /// names one, with the next batch of saves: what was done.
432    ///
433    /// # Errors
434    ///
435    /// The file is not a PNG, or the manifest does not read or would not.
436    pub(crate) fn copy_in(
437        &mut self,
438        file: &Path,
439        sheet: Option<(&str, u32)>,
440    ) -> Result<String, String> {
441        let name = file
442            .file_name()
443            .map(|name| name.to_string_lossy().into_owned())
444            .filter(|name| name.to_ascii_lowercase().ends_with(".png"))
445            .ok_or("an icon is a PNG file")?;
446        let dir = self.icons_dir();
447        let into = dir.join(&name);
448        if file != into {
449            crate::store::copy(&dir, &into, file.to_owned());
450        }
451        Ok(match sheet {
452            Some((sheet, cell)) => {
453                let (named, path) = (sheet.to_owned(), name.clone());
454                self.change(Text::Sheets, move |now| {
455                    with_sheet(now, &named, (&path, cell))
456                })?;
457                format!("{name} is the sheet {sheet}, in cells of {cell} px.")
458            }
459            None => format!("{name} is in the icons folder."),
460        })
461    }
462}

464/// The icons manifest `old` with the sheet `name` its picture `path` in
465/// cells of `cell` pixels, in place of one of that name; the rest as it
466/// was.
467///
468/// # Errors
469///
470/// The manifest does not read, or would not.
471pub(crate) fn with_sheet(
472    old: &str,
473    name: &str,
474    (path, cell): (&str, u32),
475) -> Result<String, String> {
476    let mut document: DocumentMut = old.parse().map_err(|why| format!("{why}"))?;
477    let sheets = document
478        .entry("sheets")
479        .or_insert_with(|| {
480            let mut table = toml_edit::Table::new();
481            table.set_implicit(true);
482            Item::Table(table)
483        })
484        .as_table_mut()
485        .ok_or("the manifest's sheets are not a [sheets] table")?;
486    let mut sheet = toml_edit::Table::new();
487    sheet["path"] = toml_edit::value(path);
488    sheet["cell"] = toml_edit::value(i64::from(cell));
489    sheets[name] = Item::Table(sheet);
490    let text = document.to_string();
491    hydra_ui::hotbar::Sheets::parse(&text).map_err(|why| format!("it would not read: {why}"))?;
492    Ok(text)
493}

495/// The file at `path`, or nothing when there is none yet.
496fn read_or_empty(path: &Path) -> Result<String, String> {
497    match std::fs::read_to_string(path) {
498        Ok(text) => Ok(text),
499        Err(why) if why.kind() == std::io::ErrorKind::NotFound => Ok(String::new()),
500        Err(why) => Err(why.to_string()),
501    }
502}

513/// `old` with `bar` in place of the bar `replacing`, or of its own name,
514/// or added after the rest; the rest of the file as it was.
515///
516/// # Errors
517///
518/// The file does not read or would not, or another bar has the name.
519pub(crate) fn with_bar(old: &str, bar: &Bar, replacing: Option<&str>) -> Result<String, String> {
520    let mut document: DocumentMut = old.parse().map_err(|why| format!("{why}"))?;
521    let written = toml::to_string(&HotbarFile {
522        bars: vec![bar.clone()],
523    })
524    .map_err(|why| why.to_string())?;
525    let mut fresh: DocumentMut = written.parse().map_err(|why| format!("{why}"))?;
526    let table = bars_of(&mut fresh)?
527        .get(0)
528        .cloned()
529        .ok_or("the bar was not written")?;
530    let bars = bars_of(&mut document)?;
531    let named = |name: &str| {
532        bars.iter()
533            .position(|one| one.get("name").and_then(Item::as_str) == Some(name))
534    };
535    let at = named(replacing.unwrap_or(&bar.name));
536    if let Some(other) = named(&bar.name)
537        && Some(other) != at
538    {
539        return Err(format!("There is a hotbar {} already.", bar.name));
540    }
541    let mut table = table;
542    // Where it was, with what was written above it: the file's own words
543    // before its first bar among them.
544    if let Some(place) = at.and_then(|at| bars.get_mut(at)) {
545        table.decor_mut().clone_from(place.decor());
546        table.set_position(place.position());
547        *place = table;
548    } else {
549        table.set_position(None);
550        bars.push(table);
551    }
552    let text = document.to_string();
553    HotbarFile::parse(&text).map_err(|why| format!("it would not read: {why}"))?;
554    Ok(text)
555}

557/// `old` without the bar `name`; the rest of the file as it was.
558///
559/// # Errors
560///
561/// The file does not read.
562pub(crate) fn without_bar(old: &str, name: &str) -> Result<String, String> {
563    let mut document: DocumentMut = old.parse().map_err(|why| format!("{why}"))?;
564    if document.get("bars").is_none() {
565        return Ok(old.to_owned());
566    }
567    let bars = bars_of(&mut document)?;
568    let Some(at) = bars
569        .iter()
570        .position(|one| one.get("name").and_then(Item::as_str) == Some(name))
571    else {
572        return Ok(old.to_owned());
573    };
574    // What was written above it stays, above the bar after it.
575    let above = bars
576        .get(at)
577        .and_then(|gone| gone.decor().prefix())
578        .and_then(|prefix| prefix.as_str())
579        .unwrap_or_default()
580        .to_owned();
581    bars.remove(at);
582    let mut text = match bars.get_mut(at) {
583        Some(next) => {
584            let own = next
585                .decor()
586                .prefix()
587                .and_then(|prefix| prefix.as_str())
588                .unwrap_or_default()
589                .to_owned();
590            next.decor_mut().set_prefix(format!("{above}{own}"));
591            document.to_string()
592        }
593        None => format!("{document}{above}"),
594    };
595    if !text.ends_with('\n') && !text.is_empty() {
596        text.push('\n');
597    }
598    Ok(text)
599}

601/// `old` with the bar `name`'s buttons given `slots`, in their order; the
602/// rest of the file as it was.
603///
604/// # Errors
605///
606/// The file does not read, has no such bar, or a different count of
607/// buttons.
608pub(crate) fn slotted(old: &str, name: &str, slots: &[Option<u16>]) -> Result<String, String> {
609    let mut document: DocumentMut = old.parse().map_err(|why| format!("{why}"))?;
610    let bars = document
611        .get_mut("bars")
612        .and_then(Item::as_array_of_tables_mut)
613        .ok_or_else(|| "the file has no [[bars]]".to_owned())?;
614    let bar = bars
615        .iter_mut()
616        .find(|bar| bar.get("name").and_then(Item::as_str) == Some(name))
617        .ok_or_else(|| format!("the file has no bar {name}"))?;
618    let buttons = bar
619        .get_mut("buttons")
620        .ok_or_else(|| format!("the bar {name} has no buttons"))?;
621    let put = |slot: Option<u16>| slot.map(|slot| Value::from(i64::from(slot)));
622    match buttons {
623        Item::ArrayOfTables(tables) if tables.len() == slots.len() => {
624            for (table, slot) in tables.iter_mut().zip(slots) {
625                match put(*slot) {
626                    Some(slot) => table["slot"] = Item::Value(slot),
627                    None => _ = table.remove("slot"),
628                }
629            }
630        }
631        Item::Value(Value::Array(array)) if array.len() == slots.len() => {
632            for (button, slot) in array.iter_mut().zip(slots) {
633                let Some(button) = button.as_inline_table_mut() else {
634                    return Err(format!("the bar {name}'s buttons are not tables"));
635                };
636                match put(*slot) {
637                    Some(slot) => _ = button.insert("slot", slot),
638                    None => _ = button.remove("slot"),
639                }
640            }
641        }
642        _ => return Err(format!("the bar {name}'s buttons are not as read")),
643    }
644    let text = document.to_string();
645    HotbarFile::parse(&text).map_err(|why| format!("it would not read: {why}"))?;
646    Ok(text)
647}

649/// A hotkey as a hotbar file writes it -- `alt+h`, `f5`, `ctrl+up`, or as
650/// the keybinds file does, `Alt+KeyH` -- as a key with its modifiers.
651///
652/// # Errors
653///
654/// A key Hydra does not know, or one that types rather than commands.
655pub(crate) fn hotkey(written: &str) -> Result<Chord, String> {
656    let mut parts: Vec<String> = written
657        .split('+')
658        .map(|part| part.trim().to_owned())
659        .collect();
660    if let Some(key) = parts.last_mut() {
661        let lower = key.to_ascii_lowercase();
662        let named = match lower.as_str() {
663            one if one.len() == 1 && one.chars().all(|c| c.is_ascii_alphabetic()) => {
664                Some(format!("Key{}", one.to_ascii_uppercase()))
665            }
666            one if one.len() == 1 && one.chars().all(|c| c.is_ascii_digit()) => {
667                Some(format!("Digit{one}"))
668            }
669            "up" | "down" | "left" | "right" => Some(format!("Arrow{}", capital(&lower))),
670            "pageup" | "page_up" | "pgup" => Some("PageUp".to_owned()),
671            "pagedown" | "page_down" | "pgdn" => Some("PageDown".to_owned()),
672            "esc" => Some("Escape".to_owned()),
673            "return" => Some("Enter".to_owned()),
674            "del" => Some("Delete".to_owned()),
675            "ins" => Some("Insert".to_owned()),
676            _ => None,
677        };
678        if let Some(named) = named {
679            *key = named;
680        }
681    }
682    let chord = Chord::parse(&parts.join("+"))?;
683    match chord.refused() {
684        Some(why) => Err(why),
685        None => Ok(chord),
686    }
687}

689/// `word` with its first letter a capital.
690fn capital(word: &str) -> String {
691    let mut chars = word.chars();
692    chars.next().map_or_else(String::new, |first| {
693        first.to_ascii_uppercase().to_string() + chars.as_str()
694    })
695}

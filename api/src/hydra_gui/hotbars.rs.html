1//! A character's hotbars: every character's `hotbars.toml` in the data
2//! folder, with the character's own beside its layout over it
3//! (`hydra_ui::hotbar`). Read off the window's thread (`crate::disk`),
4//! again within a second of either changing.
5//!
6//! **A button moved** on its widget is written to the file its bar came
7//! from, the character's own when it has that bar: each of the bar's
8//! buttons given the slot it now has, the file's own text kept --
9//! comments, order, what Hydra does not read.
10//!
11//! **A button's hotkey** presses it in its character's window, as a click
12//! would: Vellum's way of writing one (`alt+h`, `f5`) or the keybinds
13//! file's (`Alt+KeyH`). A key a keybind holds is the keybind's, and the
14//! button says so; of two buttons on one key, the first in the bars' order
15//! has it.
16//!
17//! **`.hotbar <bar> <button>`** presses a button by its id, macro or nickname,
18//! whatever slot it is in: what it sends is what a click would, the
19//! command its state now chooses.
20

42/// One character's hotbars, as its files have them.
43#[derive(Debug, Default)]
44pub(crate) struct Hotbars {
45    /// Every character's file, and the character's own; none, and nothing
46    /// is read.
47    paths: Option<(PathBuf, PathBuf)>,
48    /// The bars, the character's over every character's.
49    pub(crate) file: HotbarFile,
50    /// The character's own bars alone: which file a bar is written to.
51    own: HotbarFile,
52    /// Why a file does not read, when one does not.
53    problem: Option<String>,
54    /// The icons folder's sheets (`hydra_ui::hotbar::Sheets`).
55    pub(crate) sheets: hydra_ui::hotbar::Sheets,
56    /// Each file's text as Hydra has it -- as last read, with the player's
57    /// changes made -- by [`Text`]; none before the first read.
58    texts: [Option<Arc<Result<String, String>>>; 5],
59    /// The disk's reads last taken, by their generations.
60    seen: [Option<u64>; 5],
61    /// Everyone's macros file and the character's own.
62    macro_paths: Option<(PathBuf, PathBuf)>,
63    /// The macros, everyone's with the character's own over them.
64    pub(crate) macros: hydra_ui::macros::MacroFile,
65    /// The character's own macros alone, as its file has them.
66    pub(crate) macros_own: hydra_ui::macros::MacroFile,
67    /// When each macro last ran, and where.
68    pub(crate) runs: hydra_ui::macros::Runs,
69    /// Each button's hotkey, in the bars' order.
70    hotkeys: Vec<Hotkey>,
71    /// The buttons' keys a keybind holds, which are the keybind's.
72    taken: BTreeSet<Chord>,

75/// Which file's text.
76#[derive(Clone, Copy, Debug, PartialEq, Eq)]
77enum Text {
78    /// Every character's hotbars.
79    Every = 0,
80    /// The character's own.
81    Own = 1,
82    /// The icons folder's sheets.
83    Sheets = 2,
84    /// Every character's macros.
85    MacrosEvery = 3,
86    /// The character's own macros.
87    MacrosOwn = 4,

123    /// The character `name` on `instance`'s hotbars, its layouts kept in
124    /// `layouts`, inside the data folder: read off the window's thread, so
125    /// there are none until they have been.
126    pub(crate) fn load(layouts: &Path, instance: Option<&str>, name: &str) -> Self {
127        let every = layouts.parent().unwrap_or(layouts).join(FILE);
128        let own = character_path(layouts, instance, name);
129        let macros_every = layouts
130            .parent()
131            .unwrap_or(layouts)
132            .join(hydra_ui::macros::FILE);
133        let macros_own = hydra_board::macros::character_path(layouts, instance, name);
134        let mut hotbars = Self {
135            paths: Some((every, own)),
136            macro_paths: Some((macros_every, macros_own)),
137            ..Self::default()
138        };
139        // A window opening reads them afresh, not as another left them.
140        hotbars.reload();
141        hotbars
142    }

144    /// Each file's path, by [`Text`].
145    fn path(&self, text: Text) -> Option<PathBuf> {
146        let (every, own) = self.paths.as_ref()?;
147        Some(match text {
148            Text::Every => every.clone(),
149            Text::Own => own.clone(),
150            Text::Sheets => self.icons_dir().join(SHEETS),
151            Text::MacrosEvery => self.macro_paths.as_ref()?.0.clone(),
152            Text::MacrosOwn => self.macro_paths.as_ref()?.1.clone(),
153        })
154    }

156    /// The files as last read off the window's thread (`crate::disk`),
157    /// taken when any of them changed since, unless a change of the
158    /// player's own waits to be written to it; `repaint` drawn again when a
159    /// read is made.
160    pub(crate) fn fresh(&mut self, repaint: Option<&egui::Context>) {
161        let disk = crate::disk::Disk::hydras();
162        let repaint = repaint.map(crate::repaint);
163        let mut changed = false;
164        for text in TEXTS {
165            let Some(path) = self.path(text) else {
166                return;
167            };
168            // Shared, not copied: three files' texts would otherwise be
169            // copied out for every open window every frame.
170            let read = disk.kept(&key(&path), &path, repaint.as_ref(), |path| {
171                Some(Arc::new(read_or_empty(path)))
172            });
173            let Some((generation, read)) = read else {
174                continue;
175            };
176            let at = text as usize;
177            if self.seen[at] != Some(generation) && !crate::store::pending(&path) {
178                self.seen[at] = Some(generation);
179                self.texts[at] = Some(read);
180                changed = true;
181            }
182        }
183        if changed && self.texts.iter().all(Option::is_some) {
184            self.parse();
185        }
186    }

189    pub(crate) fn icons_dir(&self) -> PathBuf {
190        self.paths
191            .as_ref()
192            .and_then(|(every, _)| every.parent())
193            .map_or_else(|| PathBuf::from(ICONS), |data| data.join(ICONS))

196    /// The bars and the sheets, as the texts have them.
197    fn parse(&mut self) {
198        let named = |text: Text| {
199            self.path(text)
200                .and_then(|path| {
201                    path.file_name()
202                        .map(|name| name.to_string_lossy().into_owned())
203                })
204                .unwrap_or_default()
205        };
206        let text = |at: Text| match self.texts[at as usize].as_deref() {
207            Some(Ok(text)) => Ok(text.as_str()),
208            Some(Err(why)) => Err(why.clone()),
209            None => Ok(""),
210        };
211        let bars = |at: Text| {
212            text(at).and_then(|text| {
213                HotbarFile::parse(text).map_err(|why| format!("{}: {why}", named(at)))
214            })
215        };
216        let (every, own) = (bars(Text::Every), bars(Text::Own));
217        let sheets = text(Text::Sheets).and_then(|text| {
218            hydra_ui::hotbar::Sheets::parse(text).map_err(|why| format!("{SHEETS}: {why}"))
219        });
220        let macros = |at: Text| {
221            text(at).and_then(|text| {
222                hydra_ui::macros::MacroFile::parse(text)
223                    .map_err(|why| format!("{}: {why}", named(at)))
224            })
225        };
226        let (macros_every, macros_own) = (macros(Text::MacrosEvery), macros(Text::MacrosOwn));
227        self.problem = [
228            every.as_ref().err(),
229            own.as_ref().err(),
230            sheets.as_ref().err(),
231            macros_every.as_ref().err(),
232            macros_own.as_ref().err(),
233        ]
234        .into_iter()
235        .flatten()
236        .cloned()
237        .reduce(|one, other| format!("{one}; {other}"));
238        self.sheets = sheets.unwrap_or_default();
239        self.own = own.unwrap_or_default();
240        self.file = every.unwrap_or_default().merged(self.own.clone());
241        self.macros_own = macros_own.unwrap_or_default();
242        self.macros = macros_every
243            .unwrap_or_default()
244            .merged(self.macros_own.clone());
245        let buttons = self.file.bars.iter().flat_map(|bar| {
246            bar.buttons.iter().filter_map(|button| {
247                let written = button.hotkey.as_deref()?.trim();
248                (!written.is_empty()).then(|| Hotkey {
249                    target: Pressed::Button {
250                        bar: bar.name.clone(),
251                        id: button.id.clone(),
252                    },
253                    written: written.to_owned(),
254                    chord: hotkey(written),
255                })
256            })
257        });
258        // A macro's key is its own: in every set, or the set it names
259        // (`set`), which the window keeps apart (`keys`).
260        let macros = self.macros.macros.iter().filter_map(|made| {
261            let written = made.key.as_deref()?.trim();
262            (!written.is_empty()).then(|| Hotkey {
263                target: Pressed::Macro(made.name.clone()),
264                written: written.to_owned(),
265                chord: hotkey(written),
266            })
267        });
268        self.hotkeys = buttons.chain(macros).collect();
269    }

271    /// Why a file does not read, or a change to one was not written, when
272    /// that is so.
273    pub(crate) fn problem(&self) -> Option<String> {
274        let unwritten = TEXTS
275            .into_iter()
276            .filter_map(|text| self.path(text))
277            .filter_map(|path| crate::store::failed(&path))
278            .map(|why| format!("not saved: {why}"));
279        self.problem
280            .clone()
281            .into_iter()
282            .chain(unwritten)
283            .reduce(|one, other| format!("{one}; {other}"))
284    }

286    /// `edit` made to a file's text at once, and to the file with the next
287    /// batch of saves (`crate::store`).
288    ///
289    /// # Errors
290    ///
291    /// Nothing is kept here, the file does not read, or the edit cannot be
292    /// made to it.
293    fn change(
294        &mut self,
295        text: Text,
296        edit: impl Fn(&str) -> Result<String, String> + Clone + Send + 'static,
297    ) -> Result<(), String> {
298        let path = self.path(text).ok_or("Nothing is saved here.")?;
299        let now = match self.texts[text as usize].as_deref() {
300            Some(Ok(now)) => now.clone(),
301            Some(Err(why)) => return Err(why.clone()),
302            None => return Err("the hotbars are not read yet".to_owned()),
303        };
304        let changed = edit(&now)?;
305        self.texts[text as usize] = Some(Arc::new(Ok(changed)));
306        self.parse();
307        let dir = path.parent().unwrap_or(Path::new(".")).to_owned();
308        crate::store::edit(&dir, &path, Box::new(move |now: &str| edit(now)));
309        Ok(())
310    }

312    /// Which of the buttons' keys a keybind holds: `held`.
313    pub(crate) fn keybinds_hold(&mut self, held: impl Fn(&Chord) -> bool) {
314        self.taken = self
315            .hotkeys
316            .iter()
317            .filter_map(|hotkey| hotkey.chord.as_ref().ok())
318            .filter(|chord| held(chord))
319            .cloned()
320            .collect();
321    }

323    /// The button `chord` presses, by its bar and id: the first with that
324    /// key, when no keybind holds it.
325    pub(crate) fn pressed_by(&self, chord: &Chord) -> Option<&Pressed> {
326        if self.taken.contains(chord) {
327            return None;
328        }
329        self.hotkeys
330            .iter()
331            .find(|hotkey| hotkey.chord.as_ref() == Ok(chord))
332            .map(|hotkey| &hotkey.target)
333    }

335    /// The bar `bar`'s button `id`'s hotkey, as the keybinds file writes a
336    /// key; or why it presses nothing. None when it has none.
337    pub(crate) fn hotkey_of(&self, bar: &str, id: &str) -> Option<Result<String, String>> {
338        let target = Pressed::Button {
339            bar: bar.to_owned(),
340            id: id.to_owned(),
341        };
342        let hotkey = self.hotkeys.iter().find(|hotkey| hotkey.target == target)?;
343        let written = &hotkey.written;
344        Some(match &hotkey.chord {
345            Err(why) => Err(format!("its key {written} presses nothing: {why}")),
346            Ok(chord) if self.taken.contains(chord) => Err(format!(
347                "its key {} presses nothing: a keybind holds it, and wins",
348                chord.written()
349            )),
350            Ok(chord) => match self.pressed_by(chord) {
351                Some(Pressed::Button {
352                    bar: first_bar,
353                    id: first,
354                }) if (first_bar.as_str(), first.as_str()) != (bar, id) => Err(format!(
355                    "its key {} presses nothing: {first_bar}'s {first} has it first",
356                    chord.written()
357                )),
358                Some(Pressed::Macro(first)) => Err(format!(
359                    "its key {} presses nothing: the macro {first} has it first",
360                    chord.written()
361                )),
362                _ => Ok(chord.written()),
363            },
364        })
365    }

367    /// Take from `input` every key press a button's hotkey has, when
368    /// `bound` (a keybind) does not hold it: the buttons pressed, in order.
369    pub(crate) fn take(
370        &self,
371        input: &mut egui::InputState,
372        bound: impl Fn(&Chord) -> bool,
373    ) -> Vec<Macro> {
374        if self.hotkeys.is_empty() {
375            return Vec::new();
376        }
377        let mut pressed = Vec::new();
378        input.events.retain(|event| {
379            let egui::Event::Key {
380                key,
381                physical_key,
382                pressed: true,
383                repeat,
384                modifiers,
385            } = event
386            else {
387                return true;
388            };
389            let Some(name) = crate::keys::winit_name(physical_key.unwrap_or(*key)) else {
390                return true;
391            };
392            let chord = Chord::of(&name, *modifiers);
393            if bound(&chord) {
394                return true;
395            }
396            let Some(target) = self.pressed_by(&chord) else {
397                return true;
398            };
399            if !*repeat {
400                pressed.push(match target {
401                    Pressed::Button { bar, id } => Macro::Hotbar {
402                        bar: bar.clone(),
403                        button: id.clone(),
404                    },
405                    Pressed::Macro(name) => Macro::Run(name.clone()),
406                });
407            }
408            false
409        });
410        pressed
411    }

418    /// The bar `name`'s button in slot `from` moved to slot `to`, at once
419    /// and with the next batch of saves to the file the bar came from; why
420    /// not, as the files' problem, until they are read again.
421    pub(crate) fn move_button(&mut self, name: &str, from: u16, to: u16) {
422        if let Err(why) = self.moved(name, from, to) {
423            self.problem = Some(format!("The button was not moved: {why}"));
424        }
425    }

427    fn moved(&mut self, name: &str, from: u16, to: u16) -> Result<(), String> {
428        let text = if self.own.bar(name).is_some() {
429            Text::Own
430        } else {
431            Text::Every
432        };
433        let Some(bar) = self.file.bar(name) else {
434            return Err(format!("There is no hotbar {name}."));
435        };
436        let mut moved = bar.clone();
437        moved.move_button(from, to);
438        let slots: Vec<Option<u16>> = moved.buttons.iter().map(|button| button.slot).collect();
439        let name = moved.name;
440        self.change(text, move |now| slotted(now, &name, &slots))
441    }

443    /// The files read again now, off the window's thread.
444    pub(crate) fn reload(&mut self) {
445        let disk = crate::disk::Disk::hydras();
446        for text in TEXTS {
447            if let Some(path) = self.path(text) {
448                disk.look_again(&key(&path));
449            }
450        }
451        self.fresh(None);
452    }

460    /// `bar` kept in the character's own file (`own`) or every
461    /// character's, in place of the bar `was` (its name, and whether it was
462    /// the character's own), the rest of each file kept: at once, and with
463    /// the next batch of saves.
464    ///
465    /// # Errors
466    ///
467    /// Nothing is saved here, a file does not read or would not, or another
468    /// bar has the name.
469    pub(crate) fn save_bar(
470        &mut self,
471        bar: &Bar,
472        own: bool,
473        was: Option<(&str, bool)>,
474    ) -> Result<(), String> {
475        if bar.name.trim().is_empty() {
476            return Err("a bar needs a name.".to_owned());
477        }
478        let file = |own: bool| if own { Text::Own } else { Text::Every };
479        let replacing = was
480            .filter(|(_, was_own)| *was_own == own)
481            .map(|(name, _)| name.to_owned());
482        let kept = bar.clone();
483        self.change(file(own), move |now| {
484            with_bar(now, &kept, replacing.as_deref())
485        })?;
486        // Moved to the other file: taken out of the one it was in.
487        if let Some((name, was_own)) = was.filter(|(_, was_own)| *was_own != own) {
488            let name = name.to_owned();
489            self.change(file(was_own), move |now| without_bar(now, &name))?;
490        }
491        Ok(())
492    }

506    /// The image at `file` copied into the icons folder, and named a sheet
507    /// of cells `cell` pixels on edge in its `icons.toml` when `sheet`
508    /// names one, with the next batch of saves: what was done.
509    ///
510    /// # Errors
511    ///
512    /// The file is not a PNG, or the manifest does not read or would not.
513    pub(crate) fn copy_in(
514        &mut self,
515        file: &Path,
516        sheet: Option<(&str, u32)>,
517    ) -> Result<String, String> {
518        let name = file
519            .file_name()
520            .map(|name| name.to_string_lossy().into_owned())
521            .filter(|name| name.to_ascii_lowercase().ends_with(".png"))
522            .ok_or("an icon is a PNG file")?;
523        // The manifest's change made, or refused, before the image is
524        // copied: a refusal copies nothing, so an icon of that name is
525        // never replaced by an import said not to be done.
526        let said = match sheet {
527            Some((sheet, cell)) => {
528                let (named, path) = (sheet.to_owned(), name.clone());
529                self.change(Text::Sheets, move |now| {
530                    with_sheet(now, &named, (&path, cell))
531                })?;
532                format!("{name} is the sheet {sheet}, in cells of {cell} px.")
533            }
534            None => format!("{name} is in the icons folder."),
535        };
536        let dir = self.icons_dir();
537        let into = dir.join(&name);
538        if file != into {
539            crate::store::copy(&dir, &into, file.to_owned());
540        }
541        Ok(said)
542    }
543}

545/// The icons manifest `old` with the sheet `name` its picture `path` in
546/// cells of `cell` pixels, in place of one of that name; the rest as it
547/// was.
548///
549/// # Errors
550///
551/// The manifest does not read, or would not.
552pub(crate) fn with_sheet(
553    old: &str,
554    name: &str,
555    (path, cell): (&str, u32),
556) -> Result<String, String> {
557    let mut document: DocumentMut = old.parse().map_err(|why| format!("{why}"))?;
558    let sheets = document
559        .entry("sheets")
560        .or_insert_with(|| {
561            let mut table = toml_edit::Table::new();
562            table.set_implicit(true);
563            Item::Table(table)
564        })
565        .as_table_mut()
566        .ok_or("the manifest's sheets are not a [sheets] table")?;
567    let mut sheet = toml_edit::Table::new();
568    sheet["path"] = toml_edit::value(path);
569    sheet["cell"] = toml_edit::value(i64::from(cell));
570    sheets[name] = Item::Table(sheet);
571    let text = document.to_string();
572    hydra_ui::hotbar::Sheets::parse(&text).map_err(|why| format!("it would not read: {why}"))?;
573    Ok(text)
574}

576/// The file at `path`, or nothing when there is none yet.
577fn read_or_empty(path: &Path) -> Result<String, String> {
578    match std::fs::read_to_string(path) {
579        Ok(text) => Ok(text),
580        Err(why) if why.kind() == std::io::ErrorKind::NotFound => Ok(String::new()),
581        Err(why) => Err(why.to_string()),
582    }
583}

594/// `old` with `bar` in place of the bar `replacing`, or of its own name,
595/// or added after the rest; the rest of the file as it was.
596///
597/// # Errors
598///
599/// The file does not read or would not, or another bar has the name.
600pub(crate) fn with_bar(old: &str, bar: &Bar, replacing: Option<&str>) -> Result<String, String> {
601    let mut document: DocumentMut = old.parse().map_err(|why| format!("{why}"))?;
602    let written = toml::to_string(&HotbarFile {
603        bars: vec![bar.clone()],
604    })
605    .map_err(|why| why.to_string())?;
606    let mut fresh: DocumentMut = written.parse().map_err(|why| format!("{why}"))?;
607    let table = bars_of(&mut fresh)?
608        .get(0)
609        .cloned()
610        .ok_or("the bar was not written")?;
611    let bars = bars_of(&mut document)?;
612    let named = |name: &str| {
613        bars.iter()
614            .position(|one| one.get("name").and_then(Item::as_str) == Some(name))
615    };
616    // Only the bar it was takes its place: a new one (`replacing` none)
617    // never takes the place of a bar that has its name.
618    let at = replacing.and_then(named);
619    if let Some(other) = named(&bar.name)
620        && Some(other) != at
621    {
622        return Err(format!("There is a hotbar {} already.", bar.name));
623    }
624    let mut table = table;
625    // Where it was, with what was written above it: the file's own words
626    // before its first bar among them.
627    if let Some(place) = at.and_then(|at| bars.get_mut(at)) {
628        table.decor_mut().clone_from(place.decor());
629        table.set_position(place.position());
630        *place = table;
631    } else {
632        table.set_position(None);
633        bars.push(table);
634    }
635    let text = document.to_string();
636    HotbarFile::parse(&text).map_err(|why| format!("it would not read: {why}"))?;
637    Ok(text)
638}

640/// `old` without the bar `name`; the rest of the file as it was.
641///
642/// # Errors
643///
644/// The file does not read.
645pub(crate) fn without_bar(old: &str, name: &str) -> Result<String, String> {
646    let mut document: DocumentMut = old.parse().map_err(|why| format!("{why}"))?;
647    if document.get("bars").is_none() {
648        return Ok(old.to_owned());
649    }
650    let bars = bars_of(&mut document)?;
651    let Some(at) = bars
652        .iter()
653        .position(|one| one.get("name").and_then(Item::as_str) == Some(name))
654    else {
655        return Ok(old.to_owned());
656    };
657    // What was written above it stays, above the bar after it.
658    let above = bars
659        .get(at)
660        .and_then(|gone| gone.decor().prefix())
661        .and_then(|prefix| prefix.as_str())
662        .unwrap_or_default()
663        .to_owned();
664    bars.remove(at);
665    let mut text = match bars.get_mut(at) {
666        Some(next) => {
667            let own = next
668                .decor()
669                .prefix()
670                .and_then(|prefix| prefix.as_str())
671                .unwrap_or_default()
672                .to_owned();
673            next.decor_mut().set_prefix(format!("{above}{own}"));
674            document.to_string()
675        }
676        None => format!("{document}{above}"),
677    };
678    if !text.ends_with('\n') && !text.is_empty() {
679        text.push('\n');
680    }
681    Ok(text)
682}

684/// `old` with the bar `name`'s buttons given `slots`, in their order; the
685/// rest of the file as it was.
686///
687/// # Errors
688///
689/// The file does not read, has no such bar, or a different count of
690/// buttons.
691pub(crate) fn slotted(old: &str, name: &str, slots: &[Option<u16>]) -> Result<String, String> {
692    let mut document: DocumentMut = old.parse().map_err(|why| format!("{why}"))?;
693    let bars = document
694        .get_mut("bars")
695        .and_then(Item::as_array_of_tables_mut)
696        .ok_or_else(|| "the file has no [[bars]]".to_owned())?;
697    let bar = bars
698        .iter_mut()
699        .find(|bar| bar.get("name").and_then(Item::as_str) == Some(name))
700        .ok_or_else(|| format!("the file has no bar {name}"))?;
701    let buttons = bar
702        .get_mut("buttons")
703        .ok_or_else(|| format!("the bar {name} has no buttons"))?;
704    let put = |slot: Option<u16>| slot.map(|slot| Value::from(i64::from(slot)));
705    match buttons {
706        Item::ArrayOfTables(tables) if tables.len() == slots.len() => {
707            for (table, slot) in tables.iter_mut().zip(slots) {
708                match put(*slot) {
709                    Some(slot) => table["slot"] = Item::Value(slot),
710                    None => _ = table.remove("slot"),
711                }
712            }
713        }
714        Item::Value(Value::Array(array)) if array.len() == slots.len() => {
715            for (button, slot) in array.iter_mut().zip(slots) {
716                let Some(button) = button.as_inline_table_mut() else {
717                    return Err(format!("the bar {name}'s buttons are not tables"));
718                };
719                match put(*slot) {
720                    Some(slot) => _ = button.insert("slot", slot),
721                    None => _ = button.remove("slot"),
722                }
723            }
724        }
725        _ => return Err(format!("the bar {name}'s buttons are not as read")),
726    }
727    let text = document.to_string();
728    HotbarFile::parse(&text).map_err(|why| format!("it would not read: {why}"))?;
729    Ok(text)
730}

732/// A hotkey as a hotbar file writes it -- `alt+h`, `f5`, `ctrl+up`, or as
733/// the keybinds file does, `Alt+KeyH` -- as a key with its modifiers.
734///
735/// # Errors
736///
737/// A key Hydra does not know, or one that types rather than commands.
738pub(crate) fn hotkey(written: &str) -> Result<Chord, String> {
739    let mut parts: Vec<String> = written
740        .split('+')
741        .map(|part| part.trim().to_owned())
742        .collect();
743    if let Some(key) = parts.last_mut() {
744        let lower = key.to_ascii_lowercase();
745        let named = match lower.as_str() {
746            one if one.len() == 1 && one.chars().all(|c| c.is_ascii_alphabetic()) => {
747                Some(format!("Key{}", one.to_ascii_uppercase()))
748            }
749            one if one.len() == 1 && one.chars().all(|c| c.is_ascii_digit()) => {
750                Some(format!("Digit{one}"))
751            }
752            "up" | "down" | "left" | "right" => Some(format!("Arrow{}", capital(&lower))),
753            "pageup" | "page_up" | "pgup" => Some("PageUp".to_owned()),
754            "pagedown" | "page_down" | "pgdn" => Some("PageDown".to_owned()),
755            "esc" => Some("Escape".to_owned()),
756            "return" => Some("Enter".to_owned()),
757            "del" => Some("Delete".to_owned()),
758            "ins" => Some("Insert".to_owned()),
759            _ => None,
760        };
761        if let Some(named) = named {
762            *key = named;
763        }
764    }
765    let chord = Chord::parse(&parts.join("+"))?;
766    match chord.refused() {
767        Some(why) => Err(why),
768        None => Ok(chord),
769    }
770}

772/// `word` with its first letter a capital.
773fn capital(word: &str) -> String {
774    let mut chars = word.chars();
775    chars.next().map_or_else(String::new, |first| {
776        first.to_ascii_uppercase().to_string() + chars.as_str()
777    })
778}

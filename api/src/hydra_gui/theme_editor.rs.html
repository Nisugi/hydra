1//! The theme editor: Settings' *Themes* page, where a theme is made.
2//! Generate first: a seed and a background, a
3//! scheme, the dials, and the whole palette shown as it comes out with a
4//! sample of story text; then each token with its pin; then the shape and
5//! the type.
6//! *Wear while editing* puts the draft on every window as it changes,
7//! while the page shows.
8//! *Save* writes the theme to the `themes` folder, holding only what
9//! differs from its base; a built-in is never written over.
10//!
11//! A theme's shape and type are tried here rather than by editing a TOML
12//! file.
13//!
14//! The editor holds the draft whole, resolved: the recipe with its pins,
15//! the shape and the type, over the base it names. What is saved is the difference
16//! ([`Editor::draft`]), so a file says only what it changes.
17

34/// What the editor asks of the app.
35#[derive(Clone, Debug, PartialEq)]
36pub(crate) enum Asked {
37    /// Write this theme to the themes folder.
38    Save(Theme),
39    /// Wear this outfit on every window while editing; `None`, Hydra's own
40    /// again.
41    Preview(Option<Outfit>),
42    /// Import the preset colours of this Wrayth settings file or skin as a
43    /// theme, called this unless it is empty.
44    Import {
45        /// The file.
46        file: std::path::PathBuf,
47        /// What to call it; empty, for the file.
48        name: String,
49    },
50}

52/// The theme editor.
53#[derive(Debug)]
54pub(crate) struct Editor {
55    /// Whether it is showing.
56    pub(crate) open: bool,
57    /// The theme the draft began as, for the *Start from* choice.
58    from: String,
59    /// The draft's name.
60    name: String,
61    /// The theme the draft takes what it does not say from.
62    base: Option<String>,
63    /// The draft's recipe, whole, its own pins in it.
64    recipe: Recipe,
65    /// The draft's shape, whole.
66    shape: Shape,
67    /// The draft's type, whole.
68    kind: Type,
69    /// The frame the draft names, as its file says; kept through a save.
70    frame: Option<String>,
71    /// Whether the draft is worn on every window as it changes.
72    wearing: bool,
73    /// What the editor has to say: why a save did nothing, or that it did.
74    notice: Option<String>,
75    /// The outfit last sent to be worn, so it is sent again only when the
76    /// draft changed.
77    sent: Option<Outfit>,
78    /// The Wrayth file to import, as typed or pasted.
79    import_file: String,
80    /// What to call the imported theme; empty, for the file.
81    import_name: String,
82}

84impl Default for Editor {
85    /// A draft over Despana, named for it.
86    fn default() -> Self {
87        let mut editor = Self {
88            open: false,
89            from: Theme::DEFAULT.to_owned(),
90            name: String::new(),
91            base: None,
92            recipe: Recipe::default(),
93            shape: Shape::default(),
94            kind: Type::default(),
95            frame: None,
96            wearing: false,
97            notice: None,
98            sent: None,
99            import_file: String::new(),
100            import_name: String::new(),
101        };
102        editor.start_from(&Themes::built_in(), Theme::DEFAULT);
103        editor
104    }
105}

107impl Editor {
108    /// Open it; whether it was closed before.
109    pub(crate) fn open(&mut self) -> bool {
110        let was_closed = !self.open;
111        self.open = true;
112        was_closed
113    }
114
115    /// Begin a draft from the theme named `name`: a built-in becomes the
116    /// base of a new theme named after it; a file's theme is edited as it
117    /// is, over its own base.
118    pub(crate) fn start_from(&mut self, themes: &Themes, name: &str) {
119        let Some(theme) = themes.get(name) else {
120            return;
121        };
122        let built_in = Themes::built_in().get(name).is_some();
123        let (base, own_name) = if built_in {
124            (Some(theme.name.clone()), format!("My {}", theme.name))
125        } else {
126            (theme.base.clone(), theme.name.clone())
127        };
128        let Ok((mut recipe, shape, kind)) = themes.resolve(name) else {
129            return;
130        };
131        if built_in {
132            // Its pins are the base's; the draft starts with none of its own.
133            recipe.pins.clear();
134            recipe.plain.clear();
135            recipe.names.clear();
136        } else {
137            recipe.pins = theme.pins.clone();
138            recipe.plain.clone_from(&theme.plain);
139            recipe.names.clone_from(&theme.names);
140        }
141        self.from.clone_from(&theme.name);
142        self.name = own_name;
143        self.base = base;
144        self.recipe = recipe;
145        self.shape = shape;
146        self.kind = kind;
147        self.frame.clone_from(&theme.frame);
148        self.notice = None;
149        self.sent = None;
150    }
151
152    /// What is said under the top bar.
153    pub(crate) fn said(&mut self, notice: String) {
154        self.notice = Some(notice);
155    }
156
157    /// The draft as a theme: over its base, saying only what differs.
158    pub(crate) fn draft(&self, themes: &Themes) -> Theme {
159        let (base_recipe, base_shape, base_kind) = self
160            .base
161            .as_deref()
162            .and_then(|base| themes.resolve(base).ok())
163            .unwrap_or_default();
164        Theme {
165            name: self.name.trim().to_owned(),
166            base: self.base.clone(),
167            recipe: RecipeFile::differing(&self.recipe, &base_recipe),
168            pins: self.recipe.pins.clone(),
169            plain: self.recipe.plain.clone(),
170            names: self.recipe.names.clone(),
171            shape: ShapeFile::differing(&self.shape, &base_shape),
172            kind: TypeFile::differing(&self.kind, &base_kind),
173            frame: self.frame.clone(),
174        }
175    }
176
177    /// The draft worn: its palette, shape and type, over its base.
178    ///
179    /// # Errors
180    ///
181    /// The base cannot be resolved.
182    pub(crate) fn outfit(&self, themes: &Themes) -> Result<Outfit, String> {
183        let mut draft = self.draft(themes);
184        if draft.name.is_empty() {
185            "\u{1}draft".clone_into(&mut draft.name);
186        }
187        let name = draft.name.clone();
188        let mut themes = themes.clone();
189        themes.add(draft);
190        themes.outfit(&name)
191    }

193    /// Draw the editor into `ui`; `names` are the fonts a theme may name.
194    /// What it asks of the app.
195    #[expect(
196        clippy::too_many_lines,
197        reason = "the editor in order: its top row, its sections, its preview, what it asks"
198    )]
199    pub(crate) fn show(&mut self, ui: &mut egui::Ui, themes: &Themes, names: &Names) -> Vec<Asked> {
200        let mut asked = Vec::new();
201        let theme_names = themes.names();
202        ui.horizontal(|ui| {
203            ui.label("Start from");
204            let mut from = self.from.clone();
205            egui::ComboBox::from_id_salt("theme-from")
206                .selected_text(&from)
207                .show_ui(ui, |ui| {
208                    for name in &theme_names {
209                        ui.selectable_value(&mut from, name.clone(), name);
210                    }
211                });
212            if from != self.from {
213                self.start_from(themes, &from);
214            }
215            ui.separator();
216            ui.label("Name");
217            ui.add(egui::TextEdit::singleline(&mut self.name).desired_width(160.0));
218            ui.label("over");
219            let shown = self.base.clone().unwrap_or_else(|| "nothing".to_owned());
220            egui::ComboBox::from_id_salt("theme-base")
221                .selected_text(shown)
222                .show_ui(ui, |ui| {
223                    ui.selectable_value(&mut self.base, None, "nothing");
224                    for name in theme_names
225                        .iter()
226                        .filter(|n| !n.eq_ignore_ascii_case(&self.name))
227                    {
228                        ui.selectable_value(&mut self.base, Some(name.clone()), name);
229                    }
230                });
231            ui.separator();
232            if ui
233                .checkbox(&mut self.wearing, "Wear while editing")
234                .on_hover_text("Every window in the draft as it changes")
235                .changed()
236                && !self.wearing
237            {
238                self.sent = None;
239                asked.push(Asked::Preview(None));
240            }
241            if ui.button("Save").clicked() {
242                match self.savable(themes) {
243                    Ok(theme) => asked.push(Asked::Save(theme)),
244                    Err(why) => self.notice = Some(why),
245                }
246            }
247        });
248        if let Some(notice) = &self.notice {
249            ui.label(notice.as_str());
250        }
251        ui.separator();
252        let outfit = self.outfit(themes);
253        let height = (ui.available_height() - 8.0).max(200.0);
254        ui.horizontal_top(|ui| {
255            ui.vertical(|ui| {
256                ui.set_width(430.0);
257                egui::ScrollArea::vertical()
258                    .id_salt("theme-controls")
259                    .max_height(height)
260                    .auto_shrink(false)
261                    .show(ui, |ui| {
262                        egui::CollapsingHeader::new("How to")
263                            .default_open(false)
264                            .show(ui, how_to);
265                        egui::CollapsingHeader::new("Generate")
266                            .default_open(true)
267                            .show(ui, |ui| {
268                                generate::show(ui, &mut self.recipe, themes, &self.from);
269                            });
270                        if let Ok(outfit) = &outfit {
271                            egui::CollapsingHeader::new("Pins")
272                                .show(ui, |ui| pins::show(ui, &mut self.recipe, &outfit.palette));
273                        }
274                        egui::CollapsingHeader::new("Shape")
275                            .show(ui, |ui| shape::show(ui, &mut self.shape));
276                        egui::CollapsingHeader::new("Frame")
277                            .show(ui, |ui| frame::show(ui, &mut self.frame, &mut self.shape));
278                        egui::CollapsingHeader::new("Type")
279                            .show(ui, |ui| typeface::show(ui, &mut self.kind, names));
280                        egui::CollapsingHeader::new("Import from Wrayth")
281                            .show(ui, |ui| asked.extend(self.import(ui)));
282                    });
283            });
284            ui.separator();
285            ui.vertical(|ui| {
286                egui::ScrollArea::vertical()
287                    .id_salt("theme-preview")
288                    .max_height(height)
289                    .auto_shrink(false)
290                    .show(ui, |ui| match &outfit {
291                        Ok(outfit) => preview::show(ui, outfit),
292                        Err(why) => {
293                            ui.colored_label(
294                                ui.visuals().warn_fg_color,
295                                format!("Not worn: {why}"),
296                            );
297                        }
298                    });
299            });
300        });
301        if self.wearing
302            && let Ok(outfit) = outfit
303            && self.sent.as_ref() != Some(&outfit)
304        {
305            self.sent = Some(outfit.clone());
306            asked.push(Asked::Preview(Some(outfit)));
307        }
308        asked
309    }

311    /// The *Import from Wrayth* section: a file and a name, and the button.
312    fn import(&mut self, ui: &mut egui::Ui) -> Option<Asked> {
313        ui.label(
314            "A Wrayth skin (.skn) or settings file (.xml): its preset colours -- room names, \
315             creatures, speech, whispers, thoughts, watching, links -- become a theme over Despana.",
316        );
317        egui::Grid::new("theme-import")
318            .num_columns(2)
319            .show(ui, |ui| {
320                ui.label("File");
321                ui.add(
322                    egui::TextEdit::singleline(&mut self.import_file)
323                        .hint_text(r"E:\Wrayth Skins\storm.skn")
324                        .desired_width(300.0),
325                );
326                ui.end_row();
327                ui.label("Name");
328                ui.add(
329                    egui::TextEdit::singleline(&mut self.import_name)
330                        .hint_text("the file's")
331                        .desired_width(160.0),
332                );
333                ui.end_row();
334            });
335        let file = self.import_file.trim().trim_matches('"');
336        let clicked = ui
337            .add_enabled(!file.is_empty(), egui::Button::new("Import"))
338            .clicked();
339        clicked.then(|| Asked::Import {
340            file: std::path::PathBuf::from(file),
341            name: self.import_name.trim().to_owned(),
342        })
343    }

345    /// The draft as it would be saved, or why it cannot be: no name, a
346    /// built-in's name, or a base that is itself.
347    fn savable(&self, themes: &Themes) -> Result<Theme, String> {
348        let name = self.name.trim();
349        if name.is_empty() {
350            return Err("Give the theme a name first.".to_owned());
351        }
352        if Themes::built_in().get(name).is_some() {
353            return Err(format!("{name} is built in; save it under another name."));
354        }
355        if self
356            .base
357            .as_deref()
358            .is_some_and(|base| base.eq_ignore_ascii_case(name))
359        {
360            return Err("A theme cannot be its own base.".to_owned());
361        }
362        self.outfit(themes)?;
363        Ok(self.draft(themes))
364    }

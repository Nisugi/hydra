1//! A frame drawn round a window or a card: the kit the theme names
2//! (`hydra_ui::frame`), read from the frames folder, its pieces painted.
3//!
4//! The kit's measures and every picture come through the disk cache, so a
5//! pack updated is drawn afresh within a second. Each piece is a textured
6//! quad where `hydra_ui::frame::lay` puts it, untinted.
7
8use std::path::{Path, PathBuf};
9
10use egui::{Color32, Rect, TextureId, pos2};
11use hydra_ui::frame::{Measures, Piece};
12
13pub(crate) mod packs;

15/// Where the data folder is kept in egui's data.
16fn key() -> egui::Id {
17    egui::Id::new("frames-pool")
18}
19
20/// Keep the data folder for every window to find frames in.
21pub(crate) fn keep_frames(ctx: &egui::Context, data: &Path) {
22    let known = ctx.data(|data| data.get_temp::<PathBuf>(key()));
23    if known.as_deref() != Some(data) {
24        ctx.data_mut(|data_| data_.insert_temp(key(), data.to_path_buf()));
25    }
26}
27
28/// The data folder kept: the pool every pack of art goes into.
29pub(crate) fn pool(ctx: &egui::Context) -> Option<PathBuf> {
30    ctx.data(|data| data.get_temp::<PathBuf>(key()))
31}
32
33/// The frames folder of the data folder `data`: the studio's tree, which
34/// the studio and Hydra read alike.
35pub(crate) fn frames_in(data: &Path) -> PathBuf {
36    data.join("global")
37        .join("images")
38        .join(hydra_ui::frame::FOLDER)
39}
40
41/// The frames folder kept.
42pub(crate) fn folder(ctx: &egui::Context) -> Option<PathBuf> {
43    pool(ctx).map(|data| frames_in(&data))
44}

46/// A kit read: its measures and its eight pictures, in
47/// [`Piece::ALL`]'s order.
48#[derive(Clone, Copy, Debug, PartialEq)]
49pub(crate) struct Kit {
50    /// Its measures, as its `kit.toml` gives them.
51    pub(crate) measures: Measures,
52    pictures: [TextureId; 8],
53}
54
55impl Kit {
56    /// The texture of `piece`.
57    fn picture(&self, piece: Piece) -> TextureId {
58        let at = Piece::ALL
59            .iter()
60            .position(|each| *each == piece)
61            .unwrap_or(0);
62        self.pictures[at]
63    }
64}
65
66/// The kit named `name`, `pack/style`; none while it is being read, or when
67/// its measures or any of its pictures do not read.
68pub(crate) fn kit(ctx: &egui::Context, name: &str) -> Option<Kit> {
69    let folder = hydra_ui::frame::folder(&folder(ctx)?, name)?;
70    let list = folder.join("kit.toml").to_string_lossy().into_owned();
71    let measures = crate::pictures::kept_watching(ctx, "frame-kit", &list, &list, |list| {
72        let text = std::fs::read_to_string(list).ok()?;
73        Measures::parse(&text).ok()
74    })?;
75    let mut pictures = [TextureId::default(); 8];
76    for (at, piece) in Piece::ALL.iter().enumerate() {
77        let path = folder.join(piece.file());
78        pictures[at] = crate::pictures::picture(ctx, &path.to_string_lossy())?.id();
79    }
80    Some(Kit { measures, pictures })
81}

83/// Every kit in the frames folder, by name: read again when the kits
84/// folder changes, and when a pack lands.
85pub(crate) fn listed(ctx: &egui::Context) -> Vec<hydra_ui::frame::Listed> {
86    let Some(frames) = folder(ctx) else {
87        return Vec::new();
88    };
89    let watched = frames
90        .join(hydra_ui::frame::KITS)
91        .to_string_lossy()
92        .into_owned();
93    let key = format!("{watched}#{}", packs::generation());
94    crate::pictures::kept_watching(ctx, "frame-kits", &key, &watched, move |_| {
95        Some(hydra_ui::frame::list(&frames))
96    })
97    .unwrap_or_default()
98}

100/// The frame a theme wears: its kit, and how thick it is drawn.
101#[derive(Clone, Copy, Debug, PartialEq)]
102pub(crate) struct Worn {
103    /// The kit.
104    pub(crate) kit: Kit,
105    /// How thick, in points: what a window's content keeps in from its
106    /// edge.
107    pub(crate) thickness: f32,
108}
109
110impl Worn {
111    /// The room a window or a card leaves inside its bound for the frame:
112    /// its thickness. The kit's overhang lies over the content.
113    pub(crate) fn margin(&self) -> f32 {
114        self.thickness
115    }
116
117    /// Paint it round `window` with `painter`, flush with its edge.
118    pub(crate) fn paint(&self, painter: &egui::Painter, window: Rect) {
119        self.paint_seen(painter, window, 1.0);
120    }
121
122    /// [`Self::paint`], `opacity` of the way solid: a window drawn see-through
123    /// wears its frame as see-through.
124    pub(crate) fn paint_seen(&self, painter: &egui::Painter, window: Rect, opacity: f32) {
125        let tint = Color32::WHITE.gamma_multiply(opacity.clamp(0.0, 1.0));
126        let laid =
127            hydra_ui::frame::Rect::new(window.min.x, window.min.y, window.width(), window.height());
128        let painter = painter.with_clip_rect(painter.clip_rect().intersect(window));
129        for quad in hydra_ui::frame::lay(laid, self.thickness, &self.kit.measures) {
130            let at =
131                Rect::from_min_size(pos2(quad.at.x, quad.at.y), egui::vec2(quad.at.w, quad.at.h));
132            let uv =
133                Rect::from_min_size(pos2(quad.uv.x, quad.uv.y), egui::vec2(quad.uv.w, quad.uv.h));
134            painter.image(self.kit.picture(quad.piece), at, uv, tint);
135        }
136    }
137}

139/// The frame the theme worn by `ctx` names, at the theme's thickness, if
140/// it names one that reads.
141pub(crate) fn worn(ctx: &egui::Context) -> Option<Worn> {
142    let outfit = crate::theme::outfit(ctx);
143    let name = outfit.frame.as_deref()?;
144    Some(Worn {
145        kit: kit(ctx, name)?,
146        thickness: outfit.shape.frame_thickness,
147    })
148}

154    /// A kit made here at 8 pixels thick: runs 24 long and 12 deep, a dark
155    /// rail on the outer 8 with a light tooth every repeat reaching 4
156    /// further in; corners 16 a side, gold, with a dark notch on their
157    /// inner corner so a cut one shows.
158    fn made(ctx: &egui::Context) -> (Kit, Vec<egui::TextureHandle>) {
159        let rail = Color32::from_rgb(0x40, 0x30, 0x20);
160        let tooth = Color32::from_rgb(0xe0, 0xd0, 0xa0);
161        let run = |across: bool, outer_first: bool| {
162            let (long, deep) = (24, 12);
163            let size = if across { [long, deep] } else { [deep, long] };
164            let mut image = egui::ColorImage::filled(size, Color32::TRANSPARENT);
165            for along in 0..long {
166                for depth in 0..deep {
167                    let inward = if outer_first { depth } else { deep - 1 - depth };
168                    let colour = if inward < 8 {
169                        Some(rail)
170                    } else if along < 4 {
171                        Some(tooth)
172                    } else {
173                        None
174                    };
175                    if let Some(colour) = colour {
176                        let at = if across {
177                            (along, depth)
178                        } else {
179                            (depth, along)
180                        };
181                        image[at] = colour;
182                    }
183                }
184            }
185            image
186        };
187        let corner = |notch: (usize, usize)| {
188            let mut image = egui::ColorImage::filled([16, 16], Color32::from_rgb(0xd4, 0xaf, 0x37));
189            for y in 0..4 {
190                for x in 0..4 {
191                    image[(notch.0 + x, notch.1 + y)] = rail;
192                }
193            }
194            image
195        };
196        let images = [
197            run(true, true),
198            run(true, false),
199            run(false, true),
200            run(false, false),
201            corner((12, 12)),
202            corner((0, 12)),
203            corner((12, 0)),
204            corner((0, 0)),
205        ];
206        let textures: Vec<egui::TextureHandle> = images
207            .into_iter()
208            .enumerate()
209            .map(|(at, image)| {
210                ctx.load_texture(format!("kit-{at}"), image, egui::TextureOptions::NEAREST)
211            })
212            .collect();
213        let mut pictures = [TextureId::default(); 8];
214        for (at, texture) in textures.iter().enumerate() {
215            pictures[at] = texture.id();
216        }
217        let measures = Measures {
218            bar: 8.0,
219            overhang: 4.0,
220            corner: 16.0,
221            run: 24.0,
222        };
223        (Kit { measures, pictures }, textures)
224    }

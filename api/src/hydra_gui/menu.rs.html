1//! A settings page, as a play window draws a widget's own: its rows, each
2//! edited as its kind is -- a toggle, a choice, a colour, a number, words, a
3//! table of rows.
4//!
5//! The menu around the pages is `settings_menu.rs`.
6//!
7//! A text setting is typed and committed on Enter, or when the field is
8//! left; a value that is not of its kind is refused here, before anything is
9//! sent. *Use default* is offered where the page sets a value.
10
11use std::collections::HashMap;
12
13use hydra_ui::settings::{Change, Page, Row, RowKind, Value};
14
15/// What a row asks for.
16enum Wanted {
17    /// This value, as the command would type it.
18    Set(String),
19    /// The setting's default: taken out of the file.
20    Default,
21}
22
23/// A table's rows, each its cells.
24type Rows = Vec<Vec<String>>;
25
26/// A table being edited.
27#[derive(Debug, Default)]
28struct TableEdit {
29    /// Its rows as last read from the page.
30    read: Rows,
31    /// The rows last sent, until the page shows them.
32    sent: Option<Rows>,
33    /// The rows as they are now, with any new row not yet filled in.
34    draft: Rows,
35}
36
37/// A page's state, which outlives a frame.
38#[derive(Debug, Default)]
39pub(crate) struct Menu {
40    /// What is typed into a field, by page and key, while it has the focus.
41    typed: HashMap<(String, String), String>,
42    /// A table being edited, by page and key.
43    tables: HashMap<(String, String), TableEdit>,
44    /// What the page last said: why a value was not sent.
45    note: Option<String>,
46}
47
48/// What a row that picks from what there is offers: the fonts, the icons
49/// folder's sheets and pictures, and the shown character as a condition's
50/// row and a game value's try them.
51#[derive(Clone, Copy)]
52pub(crate) struct Pickers<'a> {
53    /// The fonts a font's row offers.
54    pub(crate) fonts: &'a crate::fonts::Names,
55    /// The icons an icon's row offers.
56    pub(crate) icons: &'a crate::hotbar_editor::icon::Folder,
57    /// The shown character, whose state a condition's row marks each test
58    /// against and a game value's row picks from; its effects' names.
59    pub(crate) live: crate::highlight_editor::when::Live<'a, 'a>,
60}

62impl Menu {
63    /// Draw `page`, and what it last said; the change asked, if one was.
64    pub(crate) fn page(
65        &mut self,
66        ui: &mut egui::Ui,
67        page: &Page,
68        pickers: Pickers<'_>,
69    ) -> Option<Change> {
70        let change = self.page_drawn(ui, "", page, pickers);
71        if let Some(note) = &self.note {
72            ui.colored_label(ui.visuals().error_fg_color, note);
73        }
74        change
75    }
76
77    /// One page: where it is kept, when a change takes effect, and its rows.
78    fn page_drawn(
79        &mut self,
80        ui: &mut egui::Ui,
81        character: &str,
82        page: &Page,
83        pickers: Pickers<'_>,
84    ) -> Option<Change> {
85        ui.strong(&page.title);
86        ui.weak(format!(
87            "Saved in {}; a change takes effect {}.",
88            page.file, page.takes
89        ));
90        if let Some(problem) = &page.problem {
91            ui.colored_label(ui.visuals().error_fg_color, problem);
92            return None;
93        }
94        let mut wanted = None;
95        let tables = page
96            .rows
97            .iter()
98            .filter_map(|row| row.key.split_once('.').map(|(table, _)| table))
99            .collect::<std::collections::BTreeSet<_>>()
100            .len();
101        egui::ScrollArea::vertical()
102            .id_salt(("settings-page", &page.id))
103            .auto_shrink(false)
104            .show(ui, |ui| {
105                egui::Grid::new(("settings-rows", &page.id))
106                    .num_columns(3)
107                    .striped(true)
108                    .show(ui, |ui| {
109                        let mut table = None;
110                        for row in &page.rows {
111                            // (a row of buttons is drawn under the rest,
112                            // across the page)
113                            if matches!(row.kind, RowKind::Buttons(_)) {
114                                continue;
115                            }
116                            if row.kind == RowKind::Heading {
117                                ui.strong(&row.label);
118                                ui.end_row();
119                                continue;
120                            }
121                            // A heading where a page's settings are in
122                            // several tables: a hunt profile's.
123                            let this = row.key.split_once('.').map(|(table, _)| table);
124                            if tables > 1 && this.is_some() && this != table {
125                                ui.strong(this.unwrap_or_default());
126                                ui.end_row();
127                            }
128                            table = this;
129                            if let Some(asked) = self.row(ui, page, row, pickers) {
130                                wanted = Some(Change {
131                                    character: character.to_owned(),
132                                    page: page.id.clone(),
133                                    key: row.key.clone(),
134                                    to: match asked {
135                                        Wanted::Set(value) => Some(value),
136                                        Wanted::Default => None,
137                                    },
138                                });
139                            }
140                            ui.end_row();
141                        }
142                    });
143                // Its rows of buttons, under the rest and across the page.
144                let mut first = true;
145                for row in &page.rows {
146                    let RowKind::Buttons(words) = &row.kind else {
147                        continue;
148                    };
149                    if first {
150                        ui.separator();
151                        first = false;
152                    }
153                    if let Some(asked) = buttons_row(ui, page, row, words) {
154                        wanted = Some(Change {
155                            character: character.to_owned(),
156                            page: page.id.clone(),
157                            key: row.key.clone(),
158                            to: match asked {
159                                Wanted::Set(value) => Some(value),
160                                Wanted::Default => None,
161                            },
162                        });
163                    }
164                }
165            });
166        wanted
167    }

169    /// One row: its name, its value to change, and *Use default* when the
170    /// file sets it. What it asks, when anything.
171    #[expect(
172        clippy::too_many_lines,
173        reason = "one row drawn in order: its name, its value by kind, its default"
174    )]
175    fn row(
176        &mut self,
177        ui: &mut egui::Ui,
178        page: &Page,
179        row: &Row,
180        pickers: Pickers<'_>,
181    ) -> Option<Wanted> {
182        let label = ui.label(&row.label).on_hover_text(&row.help);
183        let mut wanted = None;
184        match (&row.kind, &row.value) {
185            (RowKind::Toggle, value) => {
186                let mut on = matches!(value, Value::On(true));
187                let toggled = ui.add(egui::Checkbox::without_text(&mut on));
188                // Its own label is empty; a screen reader hears the row's.
189                toggled.widget_info(|| {
190                    egui::WidgetInfo::selected(egui::WidgetType::Checkbox, true, on, &row.label)
191                });
192                if toggled.changed() {
193                    wanted = Some(Wanted::Set(if on { "on" } else { "off" }.to_owned()));
194                }
195            }
196            (RowKind::Choice(choices), value) => {
197                // A list that drops down: the choices side by side would wrap
198                // in the value's narrow column, a letter to a line.
199                let now = shown(value);
200                let called = choices
201                    .iter()
202                    .find(|(choice, _)| *choice == now)
203                    .map_or(now.as_str(), |(_, called)| called.as_str());
204                let dropped = egui::ComboBox::from_id_salt(("settings-choice", &page.id, &row.key))
205                    .selected_text(called)
206                    .show_ui(ui, |ui| {
207                        for (choice, called) in choices {
208                            if ui.selectable_label(now == *choice, called).clicked()
209                                && now != *choice
210                            {
211                                wanted = Some(Wanted::Set(choice.clone()));
212                            }
213                        }
214                    });
215                // It shows its choice; a screen reader hears the row's name.
216                dropped.response.widget_info(|| {
217                    egui::WidgetInfo::labeled(egui::WidgetType::ComboBox, true, &row.label)
218                });
219            }
220            (RowKind::Color, value) => {
221                let mut rgb = rgb(&shown(value)).unwrap_or([0x80, 0x80, 0x80]);
222                let picked = ui.color_edit_button_srgb(&mut rgb);
223                picked.widget_info(|| {
224                    egui::WidgetInfo::labeled(egui::WidgetType::ColorButton, true, &row.label)
225                });
226                if picked.changed() {
227                    wanted = Some(Wanted::Set(hex(rgb)));
228                }
229            }
230            (RowKind::Slider { min, max, step }, value) => {
231                wanted = slider_row(ui, row, value, (*min, *max, *step));
232            }
233            (RowKind::TimeOfDay, value) => wanted = clock_row(ui, row, value),
234            // (drawn by the page: `page_drawn`)
235            (RowKind::Buttons(_) | RowKind::Heading, _) => {}
236            (RowKind::Font, value) => wanted = font_row(ui, page, row, value, pickers.fonts),
237            (RowKind::Icon, value) => wanted = icon_row(ui, page, row, value, pickers.icons),
238            (RowKind::Condition, value) => {
239                wanted = condition_row(ui, page, row, value, pickers.live);
240            }
241            (RowKind::GameValue, value) => {
242                let state = pickers.live.world.map(|world| world.state);
243                wanted = game_value_row(ui, page, row, value, state);
244            }
245            (RowKind::Map, value) => {
246                // Read-only, whatever the value: a hunt page shows a
247                // character's own setting, or a table of targets, as words.
248                // Only nothing is "none": a non-map value read as none would
249                // hide the setting in effect.
250                let words = shown(value);
251                let words = if words.is_empty() {
252                    "none".to_owned()
253                } else {
254                    words
255                };
256                ui.label(words).on_hover_text(&row.help);
257            }
258            (RowKind::Table(headings), value) => {
259                wanted = self.table(ui, &(page.id.clone(), row.key.clone()), headings, value);
260            }
261            (kind, value) => {
262                let shown = shown(value);
263                let key = (page.id.clone(), row.key.clone());
264                let mut text = self
265                    .typed
266                    .get(&key)
267                    .cloned()
268                    .unwrap_or_else(|| shown.clone());
269                // Sized outright: a grid gives a field only its column's
270                // width from the last frame, which would never grow.
271                let size = egui::vec2(220.0, ui.spacing().interact_size.y);
272                let field = ui
273                    .add_sized(size, egui::TextEdit::singleline(&mut text))
274                    .labelled_by(label.id);
275                if field.has_focus() {
276                    self.typed.insert(key, text);
277                } else {
278                    self.typed.remove(&key);
279                    if field.lost_focus() && text != shown {
280                        match typed(kind, &text) {
281                            Ok(to) => {
282                                self.note = None;
283                                wanted = Some(Wanted::Set(to));
284                            }
285                            Err(why) => self.note = Some(format!("{}: {why}", row.label)),
286                        }
287                    }
288                }
289            }
290        }
291        if row.here {
292            if ui
293                .small_button("Use default")
294                .on_hover_text("Take this setting out of the file")
295                .clicked()
296            {
297                wanted = Some(Wanted::Default);
298            }
299        } else {
300            ui.weak(row.from.as_deref().unwrap_or("default"));
301        }
302        wanted
303    }

305    /// A table's rows to edit, a field to a cell, with a button to take a row
306    /// out and one to add a row. The rows to send when a cell is left changed
307    /// or a row taken out; a row with nothing in it is not sent.
308    fn table(
309        &mut self,
310        ui: &mut egui::Ui,
311        key: &(String, String),
312        headings: &[String],
313        value: &Value,
314    ) -> Option<Wanted> {
315        let rows = match value {
316            Value::Table(rows) => rows.clone(),
317            _ => Vec::new(),
318        };
319        let edit = self.tables.entry(key.clone()).or_insert_with(|| TableEdit {
320            read: rows.clone(),
321            sent: None,
322            draft: rows.clone(),
323        });
324        // The page changed: by this table's own edit coming back, which
325        // keeps the draft (and any row not yet filled in), or from elsewhere,
326        // which wins.
327        if edit.read != rows {
328            if edit.sent.as_ref() != Some(&rows) {
329                edit.draft.clone_from(&rows);
330            }
331            edit.sent = None;
332            edit.read = rows;
333        }
334        let draft = &mut edit.draft;
335        let mut edited = false;
336        let mut gone = None;
337        ui.vertical(|ui| {
338            egui::Grid::new(("settings-table", &key.0, &key.1))
339                .striped(true)
340                .show(ui, |ui| {
341                    for heading in headings {
342                        ui.strong(heading);
343                    }
344                    ui.end_row();
345                    for (at, cells) in draft.iter_mut().enumerate() {
346                        cells.resize(headings.len(), String::new());
347                        for (column, cell) in cells.iter_mut().enumerate() {
348                            let width = if column + 1 == headings.len() {
349                                180.0
350                            } else {
351                                100.0
352                            };
353                            let size = egui::vec2(width, ui.spacing().interact_size.y);
354                            let field = ui.add_sized(size, egui::TextEdit::singleline(cell));
355                            if field.lost_focus() {
356                                edited = true;
357                            }
358                        }
359                        if ui
360                            .small_button("✕")
361                            .on_hover_text("Take this row out")
362                            .clicked()
363                        {
364                            gone = Some(at);
365                        }
366                        ui.end_row();
367                    }
368                });
369            if ui.small_button("Add").clicked() {
370                draft.push(vec![String::new(); headings.len()]);
371            }
372        });
373        if let Some(at) = gone {
374            draft.remove(at);
375            edited = true;
376        }
377        let filled: Vec<Vec<String>> = draft
378            .iter()
379            .filter(|cells| cells.iter().any(|cell| !cell.trim().is_empty()))
380            .map(|cells| cells.iter().map(|cell| cell.trim().to_owned()).collect())
381            .collect();
382        let last = edit.sent.as_ref().unwrap_or(&edit.read);
383        if !edited || filled == *last {
384            return None;
385        }
386        edit.sent = Some(filled.clone());
387        serde_json::to_string(&filled).ok().map(Wanted::Set)
388    }
389}

391/// A value as a field shows it: a list's items joined by commas.
392fn shown(value: &Value) -> String {
393    match value {
394        Value::Unset => String::new(),
395        Value::On(on) => on.to_string(),
396        Value::Text(text) => text.clone(),
397        Value::List(items) => items.join(", "),
398        Value::Map(pairs) => pairs
399            .iter()
400            .map(|(name, value)| format!("{name} = {value}"))
401            .collect::<Vec<_>>()
402            .join(", "),
403        Value::Table(rows) => rows
404            .iter()
405            .map(|cells| cells.join(" | "))
406            .collect::<Vec<_>>()
407            .join(", "),
408    }
409}

411/// What was typed for a setting of `kind`, written as it would be typed after
412/// `;heal set <key>`: a number as it is, words quoted, a list in brackets.
413///
414/// # Errors
415///
416/// What was typed is not of that kind, in words for the player.
417pub(crate) fn typed(kind: &RowKind, text: &str) -> Result<String, String> {
418    let text = text.trim();
419    let items = || {
420        text.split([',', ' '])
421            .map(str::trim)
422            .filter(|item| !item.is_empty())
423    };
424    match kind {
425        RowKind::Whole { min, max } => match text.parse::<u32>() {
426            Ok(n) if (*min..=*max).contains(&n) => Ok(n.to_string()),
427            _ => Err(format!("a whole number from {min} to {max}")),
428        },
429        RowKind::Number { min, max } | RowKind::Slider { min, max, .. } => {
430            match text.parse::<f64>() {
431                Ok(n) if (*min..=*max).contains(&n) => Ok(text.to_owned()),
432                _ => Err(format!("a number from {min} to {max}")),
433            }
434        }
435        RowKind::Text => Ok(quoted(text)),
436        RowKind::Numbers => {
437            let numbers: Result<Vec<u32>, _> = items().map(str::parse::<u32>).collect();
438            numbers
439                .map(|numbers| {
440                    let listed: Vec<String> = numbers.iter().map(u32::to_string).collect();
441                    format!("[{}]", listed.join(", "))
442                })
443                .map_err(|_| "whole numbers, separated by commas".to_owned())
444        }
445        RowKind::Words => {
446            let words: Vec<String> = text
447                .split(',')
448                .map(str::trim)
449                .filter(|word| !word.is_empty())
450                .map(quoted)
451                .collect();
452            Ok(format!("[{}]", words.join(", ")))
453        }
454        RowKind::Toggle
455        | RowKind::Map
456        | RowKind::Choice(_)
457        | RowKind::Color
458        | RowKind::Font
459        | RowKind::Table(_)
460        | RowKind::TimeOfDay
461        | RowKind::Buttons(_)
462        | RowKind::Heading
463        | RowKind::Icon
464        | RowKind::Condition
465        | RowKind::GameValue => Err("changed another way".to_owned()),
466    }
467}

469/// A slider's row: its value from `min` to `max` in steps of `step`,
470/// changed as it moves, so what it sets is seen at once. What it asks.
471fn slider_row(
472    ui: &mut egui::Ui,
473    row: &Row,
474    value: &Value,
475    (min, max, step): (f64, f64, f64),
476) -> Option<Wanted> {
477    let mut n = shown(value).parse::<f64>().unwrap_or(min).clamp(min, max);
478    let slid = ui.add(egui::Slider::new(&mut n, min..=max).step_by(step));
479    slid.widget_info(|| egui::WidgetInfo::slider(true, n, &row.label));
480    slid.changed().then(|| Wanted::Set(n.to_string()))
481}

483/// How wide a button of a row of them is at least, points: four to six
484/// fit across a page.
485const BUTTON_WIDTH: f32 = 96.0;
486
487/// A row of buttons to toggle ([`RowKind::Buttons`]): its label a heading,
488/// its buttons in columns across the page, as many as fit (four to six),
489/// each as wide; a button on is lit, as the hub's *Performance* is. The
490/// words on after a click, when one was.
491fn buttons_row(
492    ui: &mut egui::Ui,
493    page: &Page,
494    row: &Row,
495    words: &[(String, String)],
496) -> Option<Wanted> {
497    let on: &[String] = match &row.value {
498        Value::List(on) => on,
499        _ => &[],
500    };
501    ui.add_space(ui.spacing().item_spacing.y);
502    ui.strong(&row.label).on_hover_text(&row.help);
503    let gap = ui.spacing().item_spacing.x;
504    let width = ui.available_width();
505    let mut columns: u8 = 6;
506    while columns > 4 && (BUTTON_WIDTH + gap) * f32::from(columns) - gap > width {
507        columns -= 1;
508    }
509    let each =
510        ((width - gap * f32::from(columns - 1)) / f32::from(columns)).max(BUTTON_WIDTH * 0.5);
511    let size = egui::vec2(each, ui.spacing().interact_size.y);
512    let mut clicked = None;
513    egui::Grid::new(("settings-buttons", &page.id, &row.key))
514        .num_columns(usize::from(columns))
515        .show(ui, |ui| {
516            for (at, (word, called)) in words.iter().enumerate() {
517                let lit = on.contains(word);
518                let button = ui
519                    .add_sized(size, egui::Button::selectable(lit, called.as_str()))
520                    .on_hover_text(&row.help);
521                if button.clicked() {
522                    clicked = Some(word.as_str());
523                }
524                if (at + 1) % usize::from(columns) == 0 {
525                    ui.end_row();
526                }
527            }
528        });
529    let clicked = clicked?;
530    // each as it was, the one clicked turned over
531    let now: Vec<&str> = words
532        .iter()
533        .map(|(word, _)| word.as_str())
534        .filter(|word| on.iter().any(|one| one.as_str() == *word) != (*word == clicked))
535        .collect();
536    Some(if now.is_empty() {
537        Wanted::Default
538    } else {
539        Wanted::Set(now.join(","))
540    })
541}

543/// A time of day's row ([`RowKind::TimeOfDay`]): a slider over the day, a
544/// quarter hour a step, its value shown as a clock reads it. What it asks.
545fn clock_row(ui: &mut egui::Ui, row: &Row, value: &Value) -> Option<Wanted> {
546    let mut hours = shown(value).parse::<f64>().unwrap_or(12.0).clamp(0.0, 24.0);
547    let slid = ui.add(
548        egui::Slider::new(&mut hours, 0.0..=24.0)
549            .step_by(0.25)
550            .custom_formatter(|hours, _| clock(hours))
551            .custom_parser(clock_read),
552    );
553    slid.widget_info(|| egui::WidgetInfo::slider(true, hours, &row.label));
554    slid.changed().then(|| Wanted::Set(hours.to_string()))
555}

557/// `hours` from midnight as a clock reads them: `14:15`.
558fn clock(hours: f64) -> String {
559    let whole = hours.floor();
560    let minutes = ((hours - whole) * 60.0).round();
561    format!("{whole:02.0}:{minutes:02.0}")
562}
563
564/// A time of day typed as a clock reads it (`14:15`) or as hours
565/// (`14.25`): hours from midnight.
566fn clock_read(text: &str) -> Option<f64> {
567    let text = text.trim();
568    match text.split_once(':') {
569        Some((hours, minutes)) => {
570            Some(hours.trim().parse::<f64>().ok()? + minutes.trim().parse::<f64>().ok()? / 60.0)
571        }
572        None => text.parse().ok(),
573    }
574    .filter(|hours| hours.is_finite())
575}

577/// A font's row: its value chosen from the fonts there are
578/// ([`crate::fonts::picker`]); none, the default. What it asks.
579fn font_row(
580    ui: &mut egui::Ui,
581    page: &Page,
582    row: &Row,
583    value: &Value,
584    fonts: &crate::fonts::Names,
585) -> Option<Wanted> {
586    let mut chosen = match value {
587        Value::Text(name) if !name.is_empty() => Some(name.clone()),
588        _ => None,
589    };
590    let id = ui.make_persistent_id(("settings-font", &page.id, &row.key));
591    crate::fonts::picker(ui, id, &mut chosen, fonts, "the theme's")
592        .then(|| chosen.map_or(Wanted::Default, Wanted::Set))
593}

595/// An icon's row: picked as a hotbar button's icon is, from the icons
596/// folder's sheets and pictures (`hotbar_editor::icon::icon_ui`); none,
597/// the default. What it asks: the icon in JSON, as the row holds it.
598fn icon_row(
599    ui: &mut egui::Ui,
600    page: &Page,
601    row: &Row,
602    value: &Value,
603    icons: &crate::hotbar_editor::icon::Folder,
604) -> Option<Wanted> {
605    let before: Option<hydra_ui::hotbar::Icon> = match value {
606        Value::Text(written) => serde_json::from_str(written).ok(),
607        _ => None,
608    };
609    let mut icon = before.clone();
610    ui.vertical(|ui| {
611        let salt = format!("{}/{}", page.id, row.key);
612        crate::hotbar_editor::icon::icon_ui(ui, &mut icon, icons, &salt);
613    });
614    if icon == before {
615        return None;
616    }
617    match icon.filter(|icon| icon.art != hydra_ui::hotbar::Art::None) {
618        Some(icon) => serde_json::to_string(&icon).ok().map(Wanted::Set),
619        None => Some(Wanted::Default),
620    }
621}

623/// A condition's row: the highlight editor's *When* builder over the
624/// condition it holds, each test marked as it holds now for the character
625/// `live` tries it on. What it asks: the condition in JSON, as the row
626/// holds it; none built, the default.
627fn condition_row(
628    ui: &mut egui::Ui,
629    page: &Page,
630    row: &Row,
631    value: &Value,
632    live: crate::highlight_editor::when::Live<'_, '_>,
633) -> Option<Wanted> {
634    let before: Option<hydra_ui::condition::Condition> = match value {
635        Value::Text(written) => serde_json::from_str(written).ok(),
636        _ => None,
637    };
638    let mut when = before.clone();
639    ui.push_id(("settings-condition", &page.id, &row.key), |ui| {
640        ui.vertical(|ui| crate::highlight_editor::when::when_live(ui, &mut when, live));
641    });
642    if when == before {
643        return None;
644    }
645    match when {
646        Some(when) => serde_json::to_string(&when).ok().map(Wanted::Set),
647        None => Some(Wanted::Default),
648    }
649}

651/// A game value's row: the value picked and what it is now in `state`, and
652/// the Field widget's picker, closed until opened, to pick another. What it
653/// asks: the path picked in JSON, as the row holds it.
654fn game_value_row(
655    ui: &mut egui::Ui,
656    page: &Page,
657    row: &Row,
658    value: &Value,
659    state: Option<&hydra_model::GameState>,
660) -> Option<Wanted> {
661    let path: Vec<String> = match value {
662        Value::Text(written) => serde_json::from_str(written).unwrap_or_default(),
663        _ => Vec::new(),
664    };
665    let key = egui::Id::new(("settings-game-value", &page.id, &row.key));
666    let mut picked = None;
667    ui.vertical(|ui| {
668        if path.is_empty() {
669            ui.weak("None picked yet.");
670        } else {
671            let now = state.and_then(|state| crate::widget::field::read(ui, state, &path, key));
672            ui.label(format!(
673                "{} = {}",
674                path.join(" › "),
675                now.as_deref().unwrap_or("not found")
676            ));
677        }
678        egui::CollapsingHeader::new("Pick a value")
679            .id_salt(key.with("open"))
680            .show(ui, |ui| {
681                egui::ScrollArea::vertical()
682                    .id_salt(key.with("scroll"))
683                    .max_height(280.0)
684                    .show(ui, |ui| picked = crate::widget::field::pick(ui, state, key));
685            });
686    });
687    picked
688        .filter(|picked| *picked != path)
689        .and_then(|picked| serde_json::to_string(&picked).ok())
690        .map(Wanted::Set)
691}

693/// A colour written `#rrggbb`, as a colour row holds it.
694#[must_use]
695pub(crate) fn rgb(written: &str) -> Option<[u8; 3]> {
696    let digits = written.trim().strip_prefix('#')?;
697    if digits.len() != 6 {
698        return None;
699    }
700    let byte = |at: usize| u8::from_str_radix(digits.get(at..at + 2)?, 16).ok();
701    Some([byte(0)?, byte(2)?, byte(4)?])
702}
703
704/// A colour as a colour row writes it: `#rrggbb`.
705#[must_use]
706pub(crate) fn hex([red, green, blue]: [u8; 3]) -> String {
707    format!("#{red:02x}{green:02x}{blue:02x}")
708}

710/// `text` as a TOML string: in quotes, a quote or a backslash in it escaped.
711fn quoted(text: &str) -> String {
712    format!("\"{}\"", text.replace('\\', "\\\\").replace('"', "\\\""))
713}

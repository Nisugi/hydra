1//! A custom window's inside: its widgets held bare, each in a cell -- a
2//! free rect, kept from the inside's top left -- the play window's own model
3//! one level down.
4//!
5//! When the window is resized its cells follow: across, each scales with the
6//! inside's width, so a row of bars stays as wide as the window and two
7//! clocks side by side stay halves; down, a cell on the bottom edge keeps to
8//! it, so the last list takes the new room and a line stays a line. That is
9//! the one anchor a custom window needs; no graph of anchors is solved per
10//! frame.
11

18/// How near the bottom edge a cell's bottom must be to keep to it.
19const TOUCHING: f32 = 0.5;

21/// The least a tab that shares the strip is given, when the widths given
22/// the others leave it less: every tab is then narrowed alike.
23const SHARING: f32 = 40.0;

25/// The smallest a cell becomes, when its window shrinks or it is resized.
26pub(crate) const SMALLEST: Vec2 = Vec2::new(20.0, LINE);

129    /// Take widget `placed` out: its cell goes, or, when the cell is a tab
130    /// stack of others too, its tab.
131    pub(crate) fn take(&mut self, placed: u32) -> Option<Placed> {
132        let at = self
133            .cells
134            .iter()
135            .position(|cell| cell.tabs.iter().any(|tab| tab.id == placed))?;
136        let cell = &mut self.cells[at];
137        let tab = cell.tabs.iter().position(|tab| tab.id == placed)?;
138        let taken = cell.tabs.remove(tab);
139        if cell.tabs.is_empty() {
140            self.cells.remove(at);
141        } else {
142            cell.showing = cell.showing.min(cell.tabs.len() - 1);
143        }
144        Some(taken)
145    }

147    /// A custom window titled `title` of one cell filling its inside, `inside`
148    /// across: a tab stack of `tabs`, `showing` the one shown.
149    pub(crate) fn stack(title: &str, tabs: Vec<Placed>, showing: usize, inside: Vec2) -> Self {
150        Self {
151            title: title.to_owned(),
152            inside: [inside.x, inside.y],
153            cells: vec![Cell {
154                rect: kept(Rect::from_min_size(Pos2::ZERO, inside)),
155                tabs,
156                showing,
157                bar: Bar::default(),
158                look: None,
159                tab_height: None,

164    /// Take out the whole cell showing or stacking widget `placed`.
165    pub(crate) fn take_cell(&mut self, placed: u32) -> Option<Cell> {
166        let at = self
167            .cells
168            .iter()
169            .position(|cell| cell.tabs.iter().any(|tab| tab.id == placed))?;
170        Some(self.cells.remove(at))
171    }

173    /// The cell whose stacking middle is at `at`, from the inside's top left
174    /// (`stacks_at`): the one drawn last, on top, where cells overlap.
175    pub(crate) fn stack_at(&self, at: Pos2) -> Option<usize> {
176        self.cells
177            .iter()
178            .rposition(|cell| stacks_at(cell.rect(), at))
179    }

181    /// `tabs`, `showing` the one shown, let go at `at` inside: onto the
182    /// middle of a cell, whose tab stack they join, or else in a cell of
183    /// their own with its corner there, as big as the one shown asks and the
184    /// inside allows, and kept inside.
185    pub(crate) fn land(&mut self, tabs: Vec<Placed>, showing: usize, at: Pos2) {
186        if let Some(onto) = self.stack_at(at)
187            && stack_together(&tabs)
188            && self.cells[onto].stacks()
189        {
190            self.cells[onto].tabs.extend(tabs);
191            return;
192        }
193        let Some(shown) = tabs.get(showing).or_else(|| tabs.first()) else {
194            return;
195        };
196        let inside = Vec2::new(self.inside[0], self.inside[1]);
197        let strip = if tabs.len() > 1 { LINE } else { 0.0 };
198        let size = (shown.widget.size() + Vec2::new(0.0, strip))
199            .min(inside)
200            .max(SMALLEST);
201        let furthest = (inside - size).max(Vec2::ZERO).to_pos2();
202        self.cells.push(Cell {
203            rect: kept(Rect::from_min_size(at.clamp(Pos2::ZERO, furthest), size)),
204            tabs,
205            showing,
206            bar: Bar::default(),
207            look: None,
208            tab_height: None,
209        });
210    }

212    /// `from` -- a tab, or a whole cell -- taken out and joined to the tab
213    /// stack of the cell holding widget `into`. Nothing moves when that is
214    /// where it already is.
215    pub(crate) fn stack_onto(&mut self, from: super::Taking, into: u32) {
216        let holding = |cell: &Cell, placed: u32| cell.tabs.iter().any(|tab| tab.id == placed);
217        // Only streams stack: neither side may hold anything else.
218        let (super::Taking::Tab(moving) | super::Taking::Cell(moving)) = from;
219        let stacks = self.cells.iter().all(|cell| {
220            !(holding(cell, moving) || holding(cell, into))
221                || match from {
222                    super::Taking::Tab(_) if holding(cell, moving) && !holding(cell, into) => cell
223                        .tabs
224                        .iter()
225                        .find(|tab| tab.id == moving)
226                        .is_some_and(|tab| tab.widget.stacks()),
227                    _ => cell.stacks(),
228                }
229        });
230        if !stacks {
231            return;
232        }
233        let already = match from {
234            super::Taking::Tab(placed) | super::Taking::Cell(placed) => self
235                .cells
236                .iter()
237                .any(|cell| holding(cell, placed) && holding(cell, into)),
238        };
239        if already {
240            return;
241        }
242        let tabs = match from {
243            super::Taking::Tab(placed) => self.take(placed).map(|tab| vec![tab]),
244            super::Taking::Cell(placed) => self.take_cell(placed).map(|cell| cell.tabs),
245        };
246        if let (Some(tabs), Some(cell)) =
247            (tabs, self.cells.iter_mut().find(|cell| holding(cell, into)))
248        {
249            cell.tabs.extend(tabs);
250        }
251    }

253    /// A custom window titled `title` whose inside is `inside` across, its
254    /// widgets in `rows`: a row's widgets side by side, sharing its width, as
255    /// tall as its tallest asks, and the last row down to the bottom. When
256    /// the rows ask more than the inside has, those taller than a line share
257    /// what the one-line rows leave.
258    pub(crate) fn rows(title: &str, rows: Vec<Vec<Placed>>, inside: Vec2) -> Self {
259        let asked: Vec<f32> = rows
260            .iter()
261            .map(|row| {
262                row.iter()
263                    .map(|placed| placed.widget.size().y)
264                    .fold(LINE, f32::max)
265            })
266            .collect();
267        let total: f32 = asked.iter().sum();
268        let heights: Vec<f32> = if total <= inside.y {
269            asked
270        } else {
271            let lines: f32 = asked.iter().filter(|height| **height <= LINE).sum();
272            let tall = asked.iter().filter(|height| **height > LINE).count();
273            let share = (inside.y - lines) / f32::from(u16::try_from(tall.max(1)).unwrap_or(1));
274            asked
275                .iter()
276                .map(|height| {
277                    if *height <= LINE {
278                        *height
279                    } else {
280                        share.max(LINE)
281                    }
282                })
283                .collect()
284        };
285        let count = rows.len();
286        let mut cells = Vec::new();
287        let mut y = 0.0;
288        for (at, (row, height)) in rows.into_iter().zip(heights).enumerate() {
289            let height = if at + 1 == count {
290                (inside.y - y).max(height)
291            } else {
292                height
293            };
294            let width = inside.x / f32::from(u16::try_from(row.len().max(1)).unwrap_or(1));
295            let mut x = 0.0;
296            for placed in row {
297                cells.push(Cell {
298                    rect: kept(Rect::from_min_size(pos2(x, y), Vec2::new(width, height))),
299                    tabs: vec![placed],
300                    showing: 0,
301                    bar: Bar::default(),
302                    look: None,
303                    tab_height: None,
304                });
305                x += width;
306            }
307            y += height;
308        }
309        Self {
310            title: title.to_owned(),
311            inside: [inside.x, inside.y],
312            cells,
313        }
314    }

327    /// Keep the cells to an inside now `inside` across, as the module says:
328    /// scaled across, the bottom row kept to the bottom. Whether anything
329    /// moved.
330    pub(crate) fn fit(&mut self, inside: Vec2) -> bool {
331        let [width, height] = self.inside;
332        if (inside.x - width).abs() < TOUCHING && (inside.y - height).abs() < TOUCHING {
333            return false;
334        }
335        let across = if width > 0.0 { inside.x / width } else { 1.0 };
336        for cell in &mut self.cells {
337            let mut at = cell.rect();
338            at.min.x *= across;
339            at.max.x = (at.max.x * across).max(at.min.x + SMALLEST.x);
340            if at.max.y >= height - TOUCHING {
341                at.max.y = inside.y.max(at.min.y + SMALLEST.y);
342            }
343            cell.rect = kept(at);
344        }
345        self.inside = [inside.x, inside.y];
346        true
347    }

361    /// Tab `at` moved to `to` along the strip; the tab showing stays the
362    /// one showing. Whether anything moved.
363    pub(crate) fn move_tab(&mut self, at: usize, to: usize) -> bool {
364        if at == to || at >= self.tabs.len() || to >= self.tabs.len() {
365            return false;
366        }
367        let showing = self.tabs.get(self.showing).map(|tab| tab.id);
368        let tab = self.tabs.remove(at);
369        self.tabs.insert(to, tab);
370        if let Some(showing) = showing {
371            self.showing = self
372                .tabs
373                .iter()
374                .position(|tab| tab.id == showing)
375                .unwrap_or(0);
376        }
377        true
378    }

404/// Which of the tab rects `tabs` (of `tabs_and_body`) the pointer at `at`
405/// is over, along the strip alone (`vertical`, down a side): a tab dragged
406/// along the strip is still over it a little off it.
407pub(crate) fn tab_at(tabs: &[Rect], at: Pos2, vertical: bool) -> Option<usize> {
408    let along = |rect: &Rect| {
409        if vertical {
410            rect.y_range().contains(at.y)
411        } else {
412            rect.x_range().contains(at.x)
413        }
414    };
415    let past = |rect: &Rect| {
416        if vertical {
417            at.y >= rect.min.y
418        } else {
419            at.x >= rect.min.x
420        }
421    };
422    tabs.iter()
423        .position(along)
424        .or_else(|| (!tabs.is_empty() && past(&tabs[0])).then_some(tabs.len() - 1))
425}

427/// Whether something let go at `at` stacks onto a cell at `rect`: over the
428/// middle half of its width. Anywhere else over it, a cell is moved or
429/// placed there instead, so a full custom window can still be rearranged.
430pub(crate) fn stacks_at(rect: Rect, at: Pos2) -> bool {
431    rect.shrink2(Vec2::new(rect.width() / 4.0, 0.0))
432        .contains(at)
433}

435/// A cell drawn at `at` holding a tab of each of `widths`: a rect for each
436/// tab, when it is a tab stack, in a strip along the side `bar` says -- one
437/// line tall across the top or bottom, the tabs side by side, each as wide
438/// as its width or sharing what is left ([`across`]); or down the left or
439/// right, the tabs one under another, each a line tall -- and the rest,
440/// which the widget shown fills.
441pub(crate) fn tabs_and_body(
442    at: Rect,
443    widths: &[Option<f32>],
444    height: f32,
445    bar: Bar,
446) -> (Vec<Rect>, Rect) {
447    let tabs = widths.len();
448    if tabs < 2 {
449        return (Vec::new(), at);
450    }
451    let (strip, body) = match bar {
452        Bar::Top => at.split_top_bottom_at_y(at.min.y + height),
453        Bar::Bottom => {
454            let (body, strip) = at.split_top_bottom_at_y(at.max.y - height);
455            (strip, body)
456        }
457        Bar::Left => at.split_left_right_at_x(at.min.x + SIDE_STRIP.min(at.width() / 3.0)),
458        Bar::Right => {
459            let (body, strip) =
460                at.split_left_right_at_x(at.max.x - SIDE_STRIP.min(at.width() / 3.0));
461            (strip, body)
462        }
463    };
464    let rects = if bar.vertical() {
465        (0..tabs)
466            .map(|n| {
467                let y = strip.min.y + height * f32::from(u16::try_from(n).unwrap_or(u16::MAX));
468                Rect::from_min_size(pos2(strip.min.x, y), Vec2::new(strip.width(), height))
469            })
470            .filter(|tab| tab.max.y <= strip.max.y + 0.5)
471            .collect()
472    } else {
473        across(strip, widths, height)
474    };
475    (rects, body)
476}

478/// Tabs side by side along `strip`, `height` tall: each given a width
479/// takes it, and those given none share what is left alike. Where that
480/// would leave a sharing tab less than [`SHARING`], every tab is narrowed
481/// alike to fit, a sharing one as if [`SHARING`] wide.
482fn across(strip: Rect, widths: &[Option<f32>], height: f32) -> Vec<Rect> {
483    let given: f32 = widths.iter().flatten().sum();
484    let sharing = widths.iter().filter(|width| width.is_none()).count();
485    let sharing = f32::from(u16::try_from(sharing).unwrap_or(u16::MAX));
486    let room = strip.width();
487    let (narrowed, share) = if given + sharing * SHARING > room {
488        let wanted = given + sharing * SHARING;
489        (room / wanted, SHARING * room / wanted)
490    } else if sharing > 0.0 {
491        (1.0, (room - given) / sharing)
492    } else {
493        (1.0, 0.0)
494    };
495    let mut x = strip.min.x;
496    widths
497        .iter()
498        .map(|width| {
499            let wide = width.map_or(share, |width| width * narrowed);
500            let tab = Rect::from_min_size(pos2(x, strip.min.y), Vec2::new(wide, height));
501            x += wide;
502            tab
503        })
504        .collect()
505}
506/// Every widget placed in a layout, and its tabs turned.

508    /// Every widget showing, in the order the windows are drawn: a
509    /// standalone one, and the tab showing in each cell.
510    pub(crate) fn showing(&self) -> Vec<u32> {
511        self.holders
512            .iter()
513            .flat_map(|holder| match &holder.holds {
514                Holds::One(placed) => vec![placed.id],
515                Holds::Custom(custom) => custom
516                    .cells
517                    .iter()
518                    .filter_map(|cell| cell.shown().map(|placed| placed.id))
519                    .collect(),
520            })
521            .collect()
522    }

524    /// The cell whose tabs hold widget `placed`, and where in them it is.
525    fn cell_of(&mut self, placed: u32) -> Option<(&mut Cell, usize)> {
526        self.holders
527            .iter_mut()
528            .filter_map(|holder| match &mut holder.holds {
529                Holds::Custom(custom) => Some(custom),
530                Holds::One(_) => None,
531            })
532            .flat_map(|custom| custom.cells.iter_mut())
533            .find_map(|cell| {
534                let at = cell.tabs.iter().position(|tab| tab.id == placed)?;
535                Some((cell, at))
536            })
537    }

539    /// The tab after widget `placed` in its stack shown, or the one before
540    /// (`forward` or not), round from the last to the first: which shows
541    /// now. `None` for a widget in no custom window.
542    pub(crate) fn turn_tab(&mut self, placed: u32, forward: bool) -> Option<u32> {
543        let (cell, at) = self.cell_of(placed)?;
544        let tabs = cell.tabs.len();
545        cell.showing = if forward {
546            (at + 1) % tabs
547        } else {
548            (at + tabs - 1) % tabs
549        };
550        cell.shown().map(|shown| shown.id)
551    }

553    /// Show the tab that is widget `placed`; whether it is one.
554    pub(crate) fn show_tab(&mut self, placed: u32) -> bool {
555        let Some((cell, at)) = self.cell_of(placed) else {
556            return false;
557        };
558        cell.showing = at;
559        true
560    }

562    /// Tab `placed` moved along its strip: one left, one right, or to the
563    /// front. The tab showing stays the one showing. Whether it moved.
564    pub(crate) fn move_tab(&mut self, placed: u32, shift: Shift) -> bool {
565        let Some((cell, at)) = self.cell_of(placed) else {
566            return false;
567        };
568        let to = match shift {
569            Shift::Left => at.checked_sub(1),
570            Shift::Right => (at + 1 < cell.tabs.len()).then_some(at + 1),
571            Shift::Front => (at != 0).then_some(0),
572        };
573        to.is_some_and(|to| cell.move_tab(at, to))
574    }

607    /// The tabs of the cell holding tab `placed` `height` tall, or the

613                true
614            }
615            None => false,
616        }
617    }
618
619    /// The tabs of the cell holding tab `placed` drawn as `look`, or as the
620    /// theme says for none. Whether there was such a cell.
621    pub(crate) fn set_tab_look(
622        &mut self,
623        placed: u32,
624        look: Option<hydra_ui::theme::TabStyle>,
625    ) -> bool {
626        match self.cell_of(placed) {
627            Some((cell, _)) => {
628                cell.look = look;
629                true
630            }
631            None => false,
632        }
633    }
634
635    /// The strip of the cell holding tab `placed` along `bar`. Whether
636    /// there was such a cell.
637    pub(crate) fn set_tab_bar(&mut self, placed: u32, bar: Bar) -> bool {
638        match self.cell_of(placed) {
639            Some((cell, _)) => {
640                cell.bar = bar;
641                true
642            }
643            None => false,
644        }
645    }
646
647    /// Every widget placed here, standalone or in a custom window, showing
648    /// or a tab behind another.
649    pub(crate) fn placed(&self) -> Vec<&Placed> {
650        self.holders
651            .iter()
652            .flat_map(|holder| holder.holds.placed())
653            .collect()
654    }
655}

657impl Holds {
658    /// Every widget this holds: the one, or each cell's tabs in turn.
659    pub(crate) fn placed(&self) -> Box<dyn Iterator<Item = &Placed> + '_> {
660        match self {
661            Self::One(placed) => Box::new(std::iter::once(placed)),
662            Self::Custom(custom) => Box::new(custom.cells.iter().flat_map(|cell| cell.tabs.iter())),
663        }
664    }
665
666    /// A custom window's title; a standalone window has its widget's.
667    pub(crate) fn title(&self) -> Option<&str> {
668        match self {
669            Self::One(_) => None,
670            Self::Custom(custom) => Some(custom.title.as_str()),
671        }
672    }

1//! A play window's drawers: a main area and four drawers, top, bottom,
2//! left and right, each holding windows as the main area does, their edges
3//! on the same grid and snapping as the main area's do.
4//!
5//! A drawer can clip or push the main area, can go fully translucent, and
6//! where it is translucent and has no window a click goes through it.
7//!
8//! - **Push** takes the drawer's strip from the main area, whose windows
9//!   are drawn inside what is left. **Clip** lays the drawer over the main
10//!   area, which does not move.
11//! - **Opacity** is the backdrop's, and a clip drawer's only: a push drawer
12//!   has nothing behind it to show.
13//! - **Click-through is not a setting.** It follows from opacity, so the
14//!   player can always see what they are clicking. Below [`CLEAR`], the
15//!   backdrop hides nothing, so a press on it reaches what is under it; a
16//!   press on a window in the drawer is the window's. A drop is another
17//!   question with another answer: it lands in the drawer under the pointer
18//!   however clear it is ([`Zones::at`]).
19//!
20//! A window's rect is kept from its zone's top left, so the windows in a
21//! drawer travel with it. **A push drawer squeezes the main area**, as a
22//! play window sized away from the area the layout was arranged for does:
23//! its windows are drawn in what is left, in proportion, so windows that
24//! tiled it still tile it; moving each over instead would push the
25//! right-hand column back across the story. What is kept is not changed, so
26//! shutting the drawer, or sizing the window back, puts every window back:
27//! a window made narrow for a moment must not lose its layout. A window is
28//! always drawn inside its zone ([`Zones::fit`]).
29

35/// A backdrop less opaque than this hides nothing: a press on it passes
36/// through. A tuning value, not a design question.
37pub(crate) const CLEAR: f32 = 0.2;
38
39/// The narrowest a drawer is drawn, whatever it was dragged to.
40pub(crate) const THINNEST: f32 = 40.0;
41
42/// How far, in points, a play area may be from the one its layout was
43/// arranged for and still be it.
44const SETTLED: f32 = 4.0;

46/// Where a window lives.
47#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash, Serialize, Deserialize)]
48#[serde(rename_all = "snake_case")]
49pub(crate) enum Zone {
50    /// The main area.
51    #[default]
52    Main,
53    /// The drawer along the top.
54    Top,
55    /// Along the bottom.
56    Bottom,
57    /// Down the left.
58    Left,
59    /// Down the right.
60    Right,
61}
62
63impl Zone {
64    /// The four drawers, in the order the menu lists them.
65    pub(crate) const DRAWERS: [Self; 4] = [Self::Top, Self::Bottom, Self::Left, Self::Right];
66
67    /// Its name, for a player.
68    pub(crate) fn name(self) -> &'static str {
69        match self {
70            Self::Main => "Main area",
71            Self::Top => "Top drawer",
72            Self::Bottom => "Bottom drawer",
73            Self::Left => "Left drawer",
74            Self::Right => "Right drawer",
75        }
76    }
77
78    /// Whether it is the main area, which a kept window does not name.
79    #[expect(
80        clippy::trivially_copy_pass_by_ref,
81        reason = "serde's skip_serializing_if hands a reference"
82    )]
83    pub(crate) fn is_main(&self) -> bool {
84        *self == Self::Main
85    }
86
87    /// Whether it lies across the play area, its size a height.
88    fn across(self) -> bool {
89        matches!(self, Self::Top | Self::Bottom)
90    }
91}

129impl Drawer {
130    /// Its backdrop's opacity as drawn: solid when it pushes, since nothing
131    /// is behind it, and solid for a value no slider gives -- a hand-edited
132    /// layout must not make a drawer unseen and unrecoverable.
133    pub(crate) fn shown_opacity(&self) -> f32 {
134        match self.mode {
135            Mode::Clip if self.opacity.is_finite() => self.opacity.clamp(0.0, 1.0),
136            Mode::Clip | Mode::Push => 1.0,
137        }
138    }
139
140    /// Whether a press on its bare backdrop reaches what is under it: a
141    /// clip drawer that hides nothing.
142    pub(crate) fn clear(&self) -> bool {
143        self.shown_opacity() < CLEAR
144    }
145}

205/// Where the main area and each open drawer lie in a play area this frame,
206/// from its top left.
207#[derive(Clone, Debug, PartialEq)]
208pub(crate) struct Zones {
209    /// The main area.
210    main: Rect,
211    /// How much of the play area the layout was arranged for the main area
212    /// is, across and down: its windows are drawn at this share of their
213    /// kept size.
214    squeeze: Vec2,
215    /// The part of the squeeze that is the drawers', not the window's size.
216    by_drawers: Vec2,
217    /// Each open drawer.
218    open: Vec<(Zone, Rect)>,
219}
220
221impl Default for Zones {
222    /// No drawer open, and a main area without end: a layout's rects as
223    /// they are kept.
224    fn default() -> Self {
225        Self {
226            main: Rect::from_min_max(Pos2::ZERO, pos2(f32::INFINITY, f32::INFINITY)),
227            squeeze: Vec2::splat(1.0),
228            by_drawers: Vec2::splat(1.0),
229            open: Vec::new(),
230        }
231    }
232}
233
234impl Zones {
235    /// The zones of a play area `area` across with `drawers`, for a layout
236    /// arranged for an area `arranged` across (`None`, this one). The
237    /// drawers across it run its width; those down its sides, between
238    /// them. Open drawers are drawn narrower when they would leave the main
239    /// area less than the smallest window, keeping their proportions, and
240    /// never narrower than [`THINNEST`]; what is kept is not changed.
241    pub(crate) fn of(drawers: &Drawers, area: Vec2, arranged: Option<Vec2>) -> Self {
242        let shown = |drawer: &Drawer| {
243            if drawer.open && drawer.size.is_finite() {
244                drawer.size.max(THINNEST)
245            } else {
246                0.0
247            }
248        };
249        let (top, bottom) = squeezed(
250            area.y,
251            SMALLEST.y,
252            shown(&drawers.top),
253            shown(&drawers.bottom),
254        );
255        let (left, right) = squeezed(
256            area.x,
257            SMALLEST.x,
258            shown(&drawers.left),
259            shown(&drawers.right),
260        );
261        let (width, height) = (area.x, area.y);
262        let sides = (top, height - bottom);
263        let mut open = Vec::new();
264        let mut main = Rect::from_min_size(Pos2::ZERO, area);
265        for (zone, drawer, rect) in [
266            (
267                Zone::Top,
268                &drawers.top,
269                Rect::from_min_max(Pos2::ZERO, pos2(width, top)),
270            ),
271            (
272                Zone::Bottom,
273                &drawers.bottom,
274                Rect::from_min_max(pos2(0.0, height - bottom), pos2(width, height)),
275            ),
276            (
277                Zone::Left,
278                &drawers.left,
279                Rect::from_min_max(pos2(0.0, sides.0), pos2(left, sides.1)),
280            ),
281            (
282                Zone::Right,
283                &drawers.right,
284                Rect::from_min_max(pos2(width - right, sides.0), pos2(width, sides.1)),
285            ),
286        ] {
287            if !drawer.open || !rect.is_positive() {
288                continue;
289            }
290            if drawer.mode == Mode::Push {
291                match zone {
292                    Zone::Top => main.min.y = rect.max.y,
293                    Zone::Bottom => main.max.y = rect.min.y,
294                    Zone::Left => main.min.x = rect.max.x,
295                    Zone::Right => main.max.x = rect.min.x,
296                    Zone::Main => {}
297                }
298            }
299            open.push((zone, rect));
300        }
301        let share = |part: f32, whole: f32| if whole > 0.0 { part / whole } else { 1.0 };
302        // An area a pixel or two off the arranged one is the arranged one:
303        // a panel settling as the window opens is no reason to scale.
304        let settled = |now: f32, arranged: f32| {
305            if arranged > 0.0 && (now - arranged).abs() > SETTLED {
306                arranged
307            } else {
308                now
309            }
310        };
311        let whole = arranged.map_or(area, |arranged| {
312            Vec2::new(settled(area.x, arranged.x), settled(area.y, arranged.y))
313        });
314        let squeeze = Vec2::new(share(main.width(), whole.x), share(main.height(), whole.y));
315        let by_drawers = Vec2::new(share(main.width(), area.x), share(main.height(), area.y));
316        Self {
317            main,
318            squeeze,
319            by_drawers,
320            open,
321        }
322    }

324    /// Where zone `zone` lies; `None` for a drawer that is shut.
325    pub(crate) fn rect(&self, zone: Zone) -> Option<Rect> {
326        if zone == Zone::Main {
327            return Some(self.main);
328        }
329        self.open
330            .iter()
331            .find(|(open, _)| *open == zone)
332            .map(|(_, rect)| *rect)
333    }
334
335    /// Each open drawer and where it lies.
336    pub(crate) fn drawers(&self) -> &[(Zone, Rect)] {
337        &self.open
338    }
339
340    /// The zone a window let go at `at` lands in: an open drawer there,
341    /// however clear, over the main area it lies on: a drop targets what
342    /// the pointer visually sits on.
343    pub(crate) fn at(&self, at: Pos2) -> Zone {
344        self.open
345            .iter()
346            .find(|(_, rect)| rect.contains(at))
347            .map_or(Zone::Main, |(zone, _)| *zone)
348    }
349
350    /// Where a window kept at `kept` in zone `zone` is drawn: from the
351    /// zone's corner, squeezed with the main area, and inside the zone
352    /// ([`Self::inside`]). `None` when the zone is a drawer that is shut.
353    pub(crate) fn fit(&self, kept: Rect, zone: Zone) -> Option<Rect> {
354        let within = self.rect(zone)?;
355        let share = self.share(zone);
356        let drawn =
357            Rect::from_min_size(within.min + kept.min.to_vec2() * share, kept.size() * share);
358        self.inside(drawn, zone)
359    }
360
361    /// `at`, from the play area's top left, made no bigger than zone `zone`
362    /// and moved inside it; `None` when the zone is a drawer that is shut.
363    pub(crate) fn inside(&self, at: Rect, zone: Zone) -> Option<Rect> {
364        let within = self.rect(zone)?;
365        let size = at.size().min(within.size());
366        let furthest = (within.max - size).max(within.min);
367        Some(Rect::from_min_size(
368            at.min.clamp(within.min, furthest),
369            size,
370        ))
371    }
372
373    /// A rect `at` from the play area's top left, as kept in zone `zone`:
374    /// from the zone's corner, unsqueezed.
375    pub(crate) fn keep(&self, at: Rect, zone: Zone) -> Rect {
376        let corner = self.rect(zone).map_or(Pos2::ZERO, |rect| rect.min);
377        let share = self.share(zone);
378        Rect::from_min_size(((at.min - corner) / share).to_pos2(), at.size() / share)
379    }
380
381    /// Whether the layout is drawn scaled for a play area not the one it
382    /// was arranged for.
383    pub(crate) fn scaled(&self) -> bool {
384        self.squeeze != self.by_drawers
385    }
386
387    /// A main-area rect `at`, drawn, as it would be kept for a layout
388    /// arranged for this very area: only the drawers' squeeze undone.
389    pub(crate) fn unsqueezed_by_drawers(&self, at: Rect) -> Rect {
390        let corner = self.main.min;
391        Rect::from_min_size(
392            ((at.min - corner) / self.by_drawers).to_pos2(),
393            at.size() / self.by_drawers,
394        )
395    }
396
397    /// The share of its kept size a window in zone `zone` is drawn at.
398    fn share(&self, zone: Zone) -> Vec2 {
399        if zone == Zone::Main {
400            self.squeeze
401        } else {
402            Vec2::splat(1.0)
403        }
404    }

406    /// The edge a drawer is resized by, as a strip `width` thick on its
407    /// side toward the main area.
408    pub(crate) fn edge(zone: Zone, rect: Rect, width: f32) -> Rect {
409        let half = width / 2.0;
410        match zone {
411            Zone::Top => Rect::from_min_max(
412                pos2(rect.min.x, rect.max.y - half),
413                pos2(rect.max.x, rect.max.y + half),
414            ),
415            Zone::Bottom => Rect::from_min_max(
416                pos2(rect.min.x, rect.min.y - half),
417                pos2(rect.max.x, rect.min.y + half),
418            ),
419            Zone::Left => Rect::from_min_max(
420                pos2(rect.max.x - half, rect.min.y),
421                pos2(rect.max.x + half, rect.max.y),
422            ),
423            Zone::Right => Rect::from_min_max(
424                pos2(rect.min.x - half, rect.min.y),
425                pos2(rect.min.x + half, rect.max.y),
426            ),
427            Zone::Main => Rect::NOTHING,
428        }
429    }

431    /// How a drag of `by` on drawer `zone`'s edge changes its size: toward
432    /// the main area, larger.
433    pub(crate) fn grown(zone: Zone, by: Vec2) -> f32 {
434        match zone {
435            Zone::Top => by.y,
436            Zone::Bottom => -by.y,
437            Zone::Left => by.x,
438            Zone::Right => -by.x,
439            Zone::Main => 0.0,
440        }
441    }

443    /// Whether drawer `zone` is resized along x.
444    pub(crate) fn sideways(zone: Zone) -> bool {
445        !zone.across()
446    }
447}

449/// Two opposite drawers `a` and `b` thick, in a play area `whole` long,
450/// drawn so the main area keeps `least`: each made thinner by the same
451/// share when they would not fit. A shut drawer, 0, stays 0.
452fn squeezed(whole: f32, least: f32, a: f32, b: f32) -> (f32, f32) {
453    let room = (whole - least).max(0.0);
454    let total = a + b;
455    if total <= room || total <= 0.0 {
456        return (a, b);
457    }
458    let share = room / total;
459    (a * share, b * share)
460}

462/// A window's place among the zones: where it is drawn, moved and let go.
463impl super::Layout {
464    /// Where window `id` is drawn in `zones`, from the play area's top left:
465    /// inside its zone ([`Zones::fit`]); `None` when it is not here or its
466    /// drawer is shut.
467    pub(crate) fn shown(&self, id: u32, zones: &Zones) -> Option<Rect> {
468        let holder = self.holder(id)?;
469        zones.fit(holder.rect(), holder.zone)
470    }
471
472    /// Put window `id` at `at`, from the play area's top left, in the zone
473    /// it lives in: while a gesture lasts, which may carry it anywhere.
474    pub(crate) fn set_shown(&mut self, id: u32, at: Rect, zones: &Zones) {
475        if let Some(holder) = self.holders.iter_mut().find(|holder| holder.id == id) {
476            let kept = zones.keep(at, holder.zone);
477            holder.set(kept);
478        }
479    }
480
481    /// Window `id` let go at `at`, from the play area's top left, into zone
482    /// `zone`: made to fit there, and kept from its corner.
483    pub(crate) fn put(&mut self, id: u32, zone: Zone, at: Rect, zones: &Zones) {
484        let Some(holder) = self.holders.iter_mut().find(|holder| holder.id == id) else {
485            return;
486        };
487        let Some(inside) = zones.inside(at, zone) else {
488            return;
489        };
490        holder.zone = zone;
491        holder.set(zones.keep(inside, zone));
492    }
493}

629    fn a_push_drawer_squeezes_the_main_areas_windows() {
630        let drawers = Drawers {
631            left: open(300.0, Mode::Push),
632            ..Drawers::default()
633        };
634        let zones = Zones::of(&drawers, AREA, None);

636        let side = Rect::from_min_size(pos2(660.0, 0.0), vec2(340.0, 200.0));
637        let (drawn_story, drawn_side) = (zones.fit(story, Zone::Main), zones.fit(side, Zone::Main));

643            same(
644                drawn_side,

646            ),
647            "still beside the story, and to the edge: {drawn_side:?}"
648        );
649        let back = drawn_side.map(|drawn| zones.keep(drawn, Zone::Main));
650        assert!(same(back, side), "{back:?}");

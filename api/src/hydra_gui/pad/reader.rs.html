1//! The pad, read through gilrs: the buttons pressed since the last read,
2//! and what is held now, on any pad connected.
3//!
4//! On Windows gilrs reads through Windows.Gaming.Input, which tells a
5//! program of a pad's buttons only while one of its windows has the focus.
6
7use std::collections::BTreeSet;
8use std::time::Instant;
9
10use gilrs::{Axis, Button, EventType, Gilrs};
11
12use super::Press;

14/// A button's name, as a bind's key writes it; none for one not bound.
15fn name(button: Button) -> Option<&'static str> {
16    Some(match button {
17        Button::South => "south",
18        Button::East => "east",
19        Button::North => "north",
20        Button::West => "west",
21        Button::DPadUp => "dpad_up",
22        Button::DPadDown => "dpad_down",
23        Button::DPadLeft => "dpad_left",
24        Button::DPadRight => "dpad_right",
25        Button::LeftTrigger => "l1",
26        Button::LeftTrigger2 => "l2",
27        Button::RightTrigger => "r1",
28        Button::RightTrigger2 => "r2",
29        Button::LeftThumb => "l3",
30        Button::RightThumb => "r3",
31        Button::Select => "select",
32        Button::Start => "start",
33        Button::Mode => "guide",
34        _ => return None,
35    })
36}

38/// Every button with a name.
39const EVERY: [Button; 17] = [
40    Button::South,
41    Button::East,
42    Button::North,
43    Button::West,
44    Button::DPadUp,
45    Button::DPadDown,
46    Button::DPadLeft,

59/// What one read found.
60pub(super) struct Polled {
61    /// The buttons pressed, in order, each with what was held then.
62    pub(super) pressed: Vec<Press>,
63    /// What is held now.
64    pub(super) held: BTreeSet<&'static str>,
65    /// A pad connected or gone, said.
66    pub(super) said: Vec<String>,
67    /// Whether any pad is connected.
68    pub(super) connected: bool,
69    /// The first pad's sticks.
70    pub(super) sticks: Option<super::Sticks>,
71    /// How far the analog triggers `l2` and `r2` are pulled, 0 to 1, the
72    /// furthest of any pad's.
73    pub(super) triggers: (f32, f32),
74}
75
76/// The pads.
77pub(super) struct Reader {
78    gilrs: Gilrs,
79    /// The rumbles playing, each until it is over: gilrs stops one let go.
80    playing: Vec<(gilrs::ff::Effect, Instant)>,
81}
82
83impl Reader {
84    /// Ready to read.
85    ///
86    /// # Errors
87    ///
88    /// Why no pad can be read on this computer.
89    pub(super) fn new() -> Result<Self, String> {
90        Gilrs::new()
91            .map(|gilrs| Self {
92                gilrs,
93                playing: Vec::new(),
94            })
95            .map_err(|why| why.to_string())
96    }
97
98    /// `pulses` played at `now` on every pad that can rumble.
99    pub(super) fn play(&mut self, pulses: super::rumble::Buzz, now: Instant) {
100        use gilrs::ff::{BaseEffect, BaseEffectType, EffectBuilder, Replay, Ticks};
101        self.playing.retain(|(_, until)| *until > now);
102        let pads: Vec<gilrs::GamepadId> = self
103            .gilrs
104            .gamepads()
105            .filter(|(_, pad)| pad.is_ff_supported())
106            .map(|(id, _)| id)
107            .collect();
108        if pads.is_empty() {
109            return;
110        }
111        #[expect(
112            clippy::cast_possible_truncation,
113            clippy::cast_sign_loss,
114            reason = "a strength 0 to 1, of the most a pad gives"
115        )]
116        let magnitude = (pulses.strength.clamp(0.0, 1.0) * f32::from(u16::MAX)) as u16;
117        let mut builder = EffectBuilder::new();
118        for pulse in 0..pulses.pulses {
119            builder.add_effect(BaseEffect {
120                kind: BaseEffectType::Strong { magnitude },
121                scheduling: Replay {
122                    after: Ticks::from_ms(pulse * (pulses.pulse_ms + pulses.gap_ms)),
123                    play_for: Ticks::from_ms(pulses.pulse_ms),
124                    with_delay: Ticks::from_ms(0),
125                },
126                envelope: gilrs::ff::Envelope::default(),
127            });
128        }
129        for pad in &pads {
130            builder.add_gamepad(&self.gilrs.gamepad(*pad));
131        }
132        if let Ok(effect) = builder.finish(&mut self.gilrs)
133            && effect.play().is_ok()
134        {
135            let until = now + pulses.length() + std::time::Duration::from_millis(100);
136            self.playing.push((effect, until));
137        }
138    }
139
140    /// What is held now, on any pad.
141    fn held(&self) -> BTreeSet<&'static str> {
142        self.gilrs
143            .gamepads()
144            .flat_map(|(_, pad)| {
145                EVERY
146                    .into_iter()
147                    .filter(move |button| pad.is_pressed(*button))
148            })
149            .filter_map(name)
150            .collect()
151    }
152
153    /// The events since the last read.
154    pub(super) fn poll(&mut self) -> Polled {
155        let mut pressed = Vec::new();
156        let mut said = Vec::new();
157        while let Some(event) = self.gilrs.next_event() {
158            match event.event {
159                EventType::ButtonPressed(button, _) => {
160                    if let Some(button) = name(button) {
161                        pressed.push(button);
162                    }
163                }
164                EventType::Connected => said.push(format!(
165                    "Controller connected: {}.",
166                    self.gilrs.gamepad(event.id).name()
167                )),
168                EventType::Disconnected => said.push(format!(
169                    "Controller disconnected: {}.",
170                    self.gilrs.gamepad(event.id).name()
171                )),
172                _ => {}
173            }
174        }
175        let held = self.held();
176        // Held as each was pressed: what is held now, and the button itself
177        // pressed, though let go again before the read.
178        let pressed = pressed
179            .into_iter()
180            .map(|button| Press {
181                button,
182                held: held.clone(),
183            })
184            .collect();
185        let sticks = self.gilrs.gamepads().next().map(|(_, pad)| {
186            (
187                (pad.value(Axis::LeftStickX), pad.value(Axis::LeftStickY)),
188                (pad.value(Axis::RightStickX), pad.value(Axis::RightStickY)),
189            )
190        });
191        let pulled = |button| {
192            self.gilrs
193                .gamepads()
194                .map(|(_, pad)| {
195                    pad.button_data(button).map_or_else(
196                        || if pad.is_pressed(button) { 1.0 } else { 0.0 },
197                        gilrs::ev::state::ButtonData::value,
198                    )
199                })
200                .fold(0.0_f32, f32::max)
201        };
202        let triggers = (pulled(Button::LeftTrigger2), pulled(Button::RightTrigger2));
203        Polled {
204            pressed,
205            held,
206            said,
207            connected: sticks.is_some(),
208            sticks,
209            triggers,
210        }
211    }

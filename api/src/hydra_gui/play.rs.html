1//! A character's play window: one command input that sends on this
2//! character, and its widgets -- the story, the room, the vitals, the
3//! clocks, the injuries and the rest -- held by windows inside it that drag,
4//! resize and snap, their layout kept by the character's name.
5//!
6//! The keys' actions are in `keyed.rs` and the Find bar in `find.rs`.
7//!
8//! A play window has one command input, and a command sent in it is sent on
9//! that character. One window per character; closing it leaves the character
10//! running, and the hub opens it again.
11//!
12//! What it shows are widgets, each one thing (`crate::widget`), held by
13//! windows inside it -- a standalone window around one, or a custom window
14//! of several, bare (`holders.rs`, `crate::layout`).
15//!
16//! **The focus rule**: a click nothing else took returns the
17//! keyboard to the command input, so the player can type without clicking it.

50/// What a play window asks for.
51#[derive(Clone, Debug, PartialEq)]
52pub(crate) enum Asked {
53    /// Send this line on the character, as typed.
54    Send(String),
55    /// Run this chain on the character (`hydra_session::steps`): a hotbar
56    /// button's, a macro's.
57    Chain(String),
58    /// Press the hotbar `bar`'s button `id`, as a click on it does: its
59    /// chain run as the button's state chooses now.
60    Press {
61        /// The bar.
62        bar: String,
63        /// The button's id.
64        id: String,
65    },
66    /// Send this line the game gave -- a link's command, a dialog's button,
67    /// an object menu's entry -- once checked ([`game_line`]): one line, and
68    /// never Hydra's or Lich's.
69    FromGame(String),
70    /// Send this line as `Send` does: a menu asked for on a click, which the
71    /// player did not type, and the story does not echo (`_menu`).
72    Quietly(String),
73    /// Keep this custom window in the presets every character adds from.
74    SavePreset(crate::layout::Preset),
75    /// Forget the preset of this name.
76    ForgetPreset(String),
77    /// Use this macro set over set 0, kept for the character: a key's
78    /// action.
79    UseSet(u8),
80    /// Bring up the play window of the hub's first to ninth character: a
81    /// key's action.
82    Character(u8),
83    /// Open the settings menu on this character, at this page when one is
84    /// named: its keys, or a widget's own page.
85    Settings(Option<String>),
86    /// Read the keybinds files again.
87    ReloadKeys,
88    /// Open the character's log window.
89    Log,
90    /// What the title bar's Menu was asked for, from this character.
91    Menu(crate::main_menu::Choice),
92    /// Step the zoom of every window up (`1`) or down (`-1`): the title
93    /// bar's − or +.
94    Zoom(i8),
95    /// Say a room of the map in the character's story, from its minimap:
96    /// its number and title alone when `brief`.
97    Room {
98        /// The map's room number.
99        id: u32,
100        /// Only its number and title.
101        brief: bool,
102    },
103    /// Run this on the character's own command line, its command symbol
104    /// put before it: a travel command from its minimap.
105    Command(String),
106    /// Do this with the character's skill trainer.
107    Trainer(crate::trainer::Ask),
108    /// Bring this rule, made from a line, to the Streams page to be looked
109    /// over and saved: *Send lines like this to*.
110    StreamRule(Box<crate::stream_editor::Brought>),
111    /// Fights kept in the story as well as combat (`true`), or moved out
112    /// of it: the stream rule that takes them changed.
113    KeepCombat(bool),
114    /// The lines like this one are part of a fight (`true`), or not: the
115    /// player's word, kept for the chunk kinds.
116    FightPart(bool, Vec<hydra_ui::StyledRun>),
117}

297        session: u32,
298        name: &str,
299        instance: Option<String>,
300        layouts: Option<PathBuf>,
301    ) -> Self {
302        // A change to one of its files still waiting to be written is
303        // written first, so none is read as it was: a character started
304        // again within the minute a save waits.
305        if let Some(dir) = layouts.as_deref()
306            && crate::store::pending_for(&crate::layout::character_file(
307                dir,
308                instance.as_deref(),
309                name,
310            ))
311        {
312            crate::store::flush();
313        }
314        let layout = layouts
315            .as_deref()
316            .and_then(|dir| Layout::load(dir, instance.as_deref(), name));
317        let history = layouts.as_deref().map_or_else(History::default, |dir| {
318            history::load(dir, instance.as_deref(), name)
319        });
320        let tracked = layouts.as_deref().map_or_else(Default::default, |dir| {
321            crate::tracked::TrackedEffects::load(dir, instance.as_deref(), name)
322        });
323        let profiles = layouts.as_deref().map_or_else(Default::default, |dir| {
324            crate::trainer::GoalProfiles::load(dir, instance.as_deref(), name)
325        });
326        let containers = crate::widget::containers::ContainersShown {
327            who: name.to_owned(),
328            choices: layouts.as_deref().map_or_else(Default::default, |dir| {
329                hydra_board::containers::Containers::load(dir, instance.as_deref(), name)
330            }),
331            shared: layouts.as_deref().map_or_else(
332                Default::default,
333                hydra_board::containers::SharedLabels::load,
334            ),
335        };
336        let hotbars = layouts.as_deref().map_or_else(Default::default, |dir| {
337            crate::hotbars::Hotbars::load(dir, instance.as_deref(), name)
338        });
339        let remembered = layouts.as_deref().map_or_else(Default::default, |dir| {
340            crate::widget::aim::Remembered::load(dir, instance.as_deref(), name)
341        });
342        Self {
343            session,
344            name: name.to_owned(),
345            instance,
346            layouts,
347            layout,
348            tracked,
349            profiles,
350            containers,

361            title_bands: std::collections::HashMap::new(),

363            zones: Zones::default(),
364            read: std::collections::HashMap::new(),

366            menu: None,

368            asking: None,
369            menus_asked: Arc::default(),
370            out: None,

389    /// `.savelayout <name>`: this window's whole layout, every window
390    /// sized and placed as it is, saved under `name` for any character to
391    /// take. What was done.
392    ///
393    /// # Errors
394    ///
395    /// No layout yet, nowhere to keep it, or it could not be written.
396    pub(crate) fn save_layout_as(&self, name: &str) -> Result<String, String> {
397        let dir = self
398            .layouts
399            .as_deref()
400            .ok_or_else(|| "Layouts are not kept anywhere.".to_owned())?;
401        let layout = self
402            .layout
403            .as_ref()
404            .ok_or_else(|| "The window has no layout yet.".to_owned())?;
405        let path = layout.save_named(dir, name)?;
406        Ok(format!("Layout saved as {}.", path.display()))
407    }
408
409    /// `.loadlayout <name>`: the layout saved as `name` becomes this
410    /// window's, kept under the character's name from now. What was done.
411    ///
412    /// # Errors
413    ///
414    /// Nowhere layouts are kept, or no layout of that name reads.
415    pub(crate) fn load_layout_named(&mut self, name: &str) -> Result<String, String> {
416        let dir = self
417            .layouts
418            .as_deref()
419            .ok_or_else(|| "Layouts are not kept anywhere.".to_owned())?;
420        let loaded = Layout::load_named(dir, name).ok_or_else(|| {
421            let names = crate::layout::named_layouts(dir);
422            if names.is_empty() {
423                format!("There is no layout saved as {name}; .savelayout <name> saves this one.")
424            } else {
425                format!(
426                    "There is no layout saved as {name}. Saved: {}.",
427                    names.join(", ")
428                )
429            }
430        })?;
431        self.layout = Some(loaded);
432        self.arranging = false;
433        self.save_now();
434        Ok(format!("Layout {name} loaded."))
435    }

469    /// What the top bar asked, done: whether the layout changed, and what
470    /// the window is to do.
471    fn top_done(&mut self, top: Option<draw::Top>, drawers: Drawers) -> (bool, Option<Asked>) {
472        let asked = match top {
473            Some(draw::Top::Keys) => Asked::Menu(crate::main_menu::Choice::Editor(
474                crate::editors_menu::Editor::Keybinds,
475            )),
476            Some(draw::Top::Settings) => Asked::Settings(None),
477            Some(draw::Top::LayoutSettings) => Asked::Settings(Some(titlebar::PAGE.to_owned())),
478            Some(draw::Top::ReloadKeys) => Asked::ReloadKeys,
479            Some(draw::Top::Log) => Asked::Log,
480            Some(draw::Top::Menu(choice)) => Asked::Menu(choice),
481            Some(draw::Top::Zoom(by)) => Asked::Zoom(by),
482            Some(draw::Top::AddWidget) => {
483                self.adding = Some(menu::Adding::default());
484                return (false, None);
485            }
486            Some(draw::Top::Fit) => {
487                self.layout = None;
488                return (true, None);
489            }
490            Some(draw::Top::Drawers) => {
491                let layout = self.layout.as_mut();
492                return (
493                    layout.map(|layout| layout.drawers = drawers).is_some(),
494                    None,
495                );
496            }
497            Some(draw::Top::NewCustom) => {
498                let Some(layout) = &mut self.layout else {
499                    return (false, None);
500                };
501                layout.new_custom();
502                self.arranging = !layout.locked;
503                return (true, None);
504            }
505            None => return (false, None),
506        };
507        (false, Some(asked))
508    }

510    /// Draw the window into `ui` -- a viewport's whole area -- and return
511    /// what the player asked for, if anything.
512    pub(crate) fn show(&mut self, ui: &mut egui::Ui, view: &PlayView<'_>) -> Option<Asked> {
513        // Held here, or in the settings menu: a widget's page there changes
514        // this window's layout too.
515        let held = ui.input(|input| input.pointer.any_down())
516            || ui
517                .ctx()
518                .input_for(crate::settings_menu::viewport(), |input| {
519                    input.pointer.any_down()
520                });
521        if held {
522            self.save_held.get_or_insert(false);
523        } else if self.save_held.take() == Some(true) {
524            self.save_now();
525        }
526        let mut asked = None;
527        self.hotbars.fresh(Some(ui.ctx()));
528        // An Uberbar placed wants the day's numbers: read when due.
529        if let Some(layout) = &self.layout
530            && layout.wants_uberbar_facts()
531            && let Some(data) = self.layouts.as_deref().and_then(std::path::Path::parent)
532            && let Some(instance) = self.instance.as_deref()
533        {
534            self.uberbar_facts.fresh(
535                ui.ctx(),
536                data,
537                hydra_ledger::Who::new(instance, self.name.as_str()),
538            );
539        }
540        self.completing.symbol = view.symbol;
541        if self.completing.words != view.commands {
542            self.completing.words = view.commands.to_vec();
543        }
544        let session = self.session;
545        let (mut changed, top_asked) = self.top_bar(ui, view);
546        asked = asked.or(top_asked);
547        draw::alerts(ui, session, view.story, &mut self.banners_gone);
548        // The command line: at the foot of the window, unless a Command line
549        // widget is placed, when that widget draws it.
550        let placed_input = self.layout.as_ref().is_some_and(Layout::has_input);
551        if !placed_input {
552            egui::Panel::bottom(Id::new(("play-input-panel", session))).show(ui, |ui| {
553                if let Some(line) = self.input(ui) {
554                    asked = Some(Asked::Send(line));
555                }
556            });
557        }
558        let area = ui.available_rect_before_wrap();
559        if let Some(snapshot) = view.snapshot {
560            let targeting = &snapshot.state.targeting;
561            self.targets = (targeting.ids().to_vec(), targeting.current());
562        }
563        self.ask_find(ui.ctx());
564        // A Command line widget's line, entered, is sent with what the
565        // widgets asked (`holders.rs`).
566        changed |= self.arrange(ui, view);
567        self.find_bar(ui.ctx(), area);
568        self.trainer_popup(ui.ctx(), area, view.trainer);
569        let heard: Vec<&str> = view.story.streams.ids().collect();
570        let made = (view.streams.iter().map(String::as_str)).filter(|id| !heard.contains(id));
571        let received: Vec<Widget> = (heard.iter().copied().chain(made))
572            .map(|id| Widget::Stream(id.to_owned()))
573            .chain(view.snapshot.into_iter().flat_map(|snapshot| {
574                (snapshot.state.dialogs.iter()).map(|(id, _)| Widget::Dialog(id.to_owned()))
575            }))
576            .collect();
577        let state = view.snapshot.map(|snapshot| &snapshot.state);
578        changed |= self.add_list(
579            ui.ctx(),
580            area,
581            (view.others, state),
582            view.presets,
583            &received,
584        );
585        changed |= self.right_click(
586            ui.ctx(),
587            area,
588            (view.others, &received, view.streams, view.combat_kept),
589        );
590        self.object_menu(ui.ctx(), view);
591        self.calibrator(ui.ctx());
592        crate::carry::show(ui.ctx(), session);
593        asked = asked.or(self.out.take());
594        if changed {
595            self.save();
596        }
597        // The focus rule: nothing holds the keyboard, so the input takes it.
598        if ui.ctx().memory(|memory| memory.focused().is_none()) {
599            let id = self.input_id();
600            ui.ctx().memory_mut(|memory| memory.request_focus(id));
601        }
602        asked
603    }

642    /// Change the character's containers' choices, or a label kept for every
643    /// character, and keep them, when the change changed anything.
644    pub(crate) fn contain(&mut self, change: crate::widget::containers::Contained) {
645        use crate::widget::containers::Contained;
646        let saved = match change {
647            Contained::Own(change) => {
648                if !self.containers.choices.apply(change) {
649                    return;
650                }
651                self.layouts.as_ref().map(|dir| {
652                    self.containers
653                        .choices
654                        .save(dir, self.instance.as_deref(), &self.name)
655                })
656            }
657            Contained::Shared(name, label) => {
658                if !self.containers.shared.set(name, label) {
659                    return;
660                }
661                self.layouts
662                    .as_ref()
663                    .map(|dir| self.containers.shared.save(dir))
664            }
665        };
666        if let Some(saved) = saved {
667            self.unsaved = saved.err().map(|why| why.to_string());
668        }
669    }

758    /// A goal profile saved or deleted, and kept; or loaded, which is the
759    /// change to ask of the character's trainer.
760    pub(crate) fn profiled(
761        &mut self,
762        profiled: crate::trainer::Profiled,
763    ) -> Option<crate::trainer::Ask> {
764        use crate::trainer::{Ask, Change, Profiled};
765        match profiled {
766            Profiled::Load(name) => {
767                let goals = self.profiles.profiles.get(&name)?.clone();
768                return Some(Ask::Change(Change::Aim { name, goals }));
769            }
770            Profiled::Save { name, goals } => {
771                self.profiles.profiles.insert(name.clone(), goals.clone());
772                self.change_profile(&name, Some(goals));
773            }
774            Profiled::Delete(name) => {
775                self.profiles.profiles.remove(&name)?;
776                self.change_profile(&name, None);
777            }
778        }
779        None
780    }

825    /// How large its window draws everything: its layout's own zoom, held
826    /// in Hydra's bounds; none, Hydra's zoom.
827    pub(crate) fn zoom(&self) -> Option<f32> {
828        self.layout
829            .as_ref()
830            .and_then(|layout| layout.zoom)
831            .filter(|zoom| zoom.is_finite())
832            .map(|zoom| zoom.clamp(hydra_board::own::ZOOM_LEAST, hydra_board::own::ZOOM_MOST))
833    }

835    /// Its window drawn `by` steps larger, smaller below 0, from `from`, the
836    /// zoom it is drawn at now: held in Hydra's bounds, rounded to a
837    /// hundredth, and kept with its layout. What it is; none with no layout
838    /// yet.
839    pub(crate) fn zoom_by(&mut self, by: i8, from: f32) -> Option<f32> {
840        use hydra_board::own::{ZOOM_LEAST, ZOOM_MOST, ZOOM_STEP};
841        let layout = self.layout.as_mut()?;
842        let zoom = from + f32::from(by) * ZOOM_STEP;
843        let zoom = (zoom.clamp(ZOOM_LEAST, ZOOM_MOST) * 100.0).round() / 100.0;
844        layout.zoom = Some(zoom);
845        self.save();
846        Some(zoom)
847    }

849    /// The top bar, drawn and its asks done: whether the layout changed,
850    /// and what the window is to do.
851    fn top_bar(&mut self, ui: &mut egui::Ui, view: &PlayView<'_>) -> (bool, Option<Asked>) {
852        let session = self.session;
853        let unsaved = self
854            .unsaved
855            .clone()
856            .or_else(|| self.layout_saves.failed())
857            .map(|why| format!("Layout not saved: {why}"))
858            .or_else(|| {
859                (self.history_saves.failed()).map(|why| format!("History not saved: {why}"))
860            });
861        let mut arranging = self.arranging;
862        let mut locked = self.layout.as_ref().is_some_and(|layout| layout.locked);
863        let set_guides = self
864            .layout
865            .as_ref()
866            .map(crate::layout::Layout::guides)
867            .unwrap_or_default();
868        let mut guides = set_guides;
869        let mut drawers = self
870            .layout
871            .as_ref()
872            .map_or_else(Drawers::default, |layout| layout.drawers);
873        let title = self
874            .layout
875            .as_ref()
876            .map(|layout| layout.title.clone())
877            .unwrap_or_default();
878        let top = egui::Panel::top(Id::new(("play-top", session)))
879            .show(ui, |ui| {
880                draw::top(
881                    ui,
882                    view,
883                    &title,
884                    &mut drawers,
885                    (&mut arranging, &mut locked, &mut guides),
886                    unsaved.as_deref(),
887                )
888            })
889            .inner;
890        if title.own {
891            draw::resize_edges(ui.ctx());
892        }
893        self.arranging = arranging && !locked;
894        if let Some(layout) = &mut self.layout
895            && layout.locked != locked
896        {
897            layout.locked = locked;
898            self.save();
899        }
900        if let Some(layout) = &mut self.layout
901            && guides != set_guides
902        {
903            layout.set_guides(guides);
904            self.save();
905        }
906        self.top_done(top, drawers)
907    }

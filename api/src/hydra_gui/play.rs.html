1//! A character's play window: one command input that sends on this
2//! character, and its widgets -- the story, the room, the vitals, the
3//! clocks, the injuries and the rest -- held by windows inside it that drag,
4//! resize and snap, their layout kept by the character's name.
5//!
6//! The keys' actions are in `keyed.rs` and the Find bar in `find.rs`.
7//!
8//! A play window has one command input, and a command sent in it is sent on
9//! that character. One window per character; closing it leaves the character
10//! running, and the hub opens it again.
11//!
12//! What it shows are widgets, each one thing (`crate::widget`), held by
13//! windows inside it -- a standalone window around one, or a custom window
14//! of several, bare (`holders.rs`, `crate::layout`).
15//!
16//! **The focus rule**: a click nothing else took returns the
17//! keyboard to the command input, so the player can type without clicking it.

50/// What a play window asks for.
51#[derive(Clone, Debug, PartialEq)]
52pub(crate) enum Asked {
53    /// Send this line on the character, as typed.
54    Send(String),
55    /// Run this chain on the character (`hydra_session::steps`): a hotbar
56    /// button's, a macro's.
57    Chain(String),
58    /// Press the hotbar `bar`'s button `id`, as a click on it does: its
59    /// chain run as the button's state chooses now.
60    Press {
61        /// The bar.
62        bar: String,
63        /// The button's id.
64        id: String,
65    },
66    /// Send this line the game gave -- a link's command, a dialog's button,
67    /// an object menu's entry -- once checked ([`game_line`]): one line, and
68    /// never Hydra's or Lich's.
69    FromGame(String),
70    /// Send this line as `Send` does: a menu asked for on a click, which the
71    /// player did not type, and the story does not echo (`_menu`).
72    Quietly(String),
73    /// Keep this custom window in the presets every character adds from.
74    SavePreset(crate::layout::Preset),
75    /// Forget the preset of this name.
76    ForgetPreset(String),
77    /// Use this macro set over set 0, kept for the character: a key's
78    /// action.
79    UseSet(u8),
80    /// Bring up the play window of the hub's first to ninth character: a
81    /// key's action.
82    Character(u8),
83    /// Open the settings menu on this character, at this page when one is
84    /// named: its keys, or a widget's own page.
85    Settings(Option<String>),
86    /// Read the keybinds files again.
87    ReloadKeys,
88    /// Open the character's log window.
89    Log,
90    /// What the title bar's Menu was asked for, from this character.
91    Menu(crate::main_menu::Choice),
92    /// Step the zoom of every window up (`1`) or down (`-1`): the title
93    /// bar's − or +.
94    Zoom(i8),
95    /// Say a room of the map in the character's story, from its minimap:
96    /// its number and title alone when `brief`.
97    Room {
98        /// The map's room number.
99        id: u32,
100        /// Only its number and title.
101        brief: bool,
102    },
103    /// Run this on the character's own command line, its command symbol
104    /// put before it: a travel command from its minimap.
105    Command(String),
106    /// Do this with the character's skill trainer.
107    Trainer(crate::trainer::Ask),
108    /// Bring this rule, made from a line, to the Streams page to be looked
109    /// over and saved: *Send lines like this to*.
110    StreamRule(Box<crate::stream_editor::Brought>),
111}

277    /// A play window for session `session`, the character `name` on the
278    /// game `instance` names, with its layout from `layouts` when one was
279    /// saved there.
280    pub(crate) fn new(
281        session: u32,
282        name: &str,
283        instance: Option<String>,
284        layouts: Option<PathBuf>,
285    ) -> Self {
286        // A change to one of its files still waiting to be written is
287        // written first, so none is read as it was: a character started
288        // again within the minute a save waits.
289        if let Some(dir) = layouts.as_deref()
290            && crate::store::pending_for(&crate::layout::character_file(
291                dir,
292                instance.as_deref(),
293                name,
294            ))
295        {
296            crate::store::flush();
297        }
298        let layout = layouts
299            .as_deref()
300            .and_then(|dir| Layout::load(dir, instance.as_deref(), name));
301        let history = layouts.as_deref().map_or_else(History::default, |dir| {
302            history::load(dir, instance.as_deref(), name)
303        });
304        let tracked = layouts.as_deref().map_or_else(Default::default, |dir| {
305            crate::tracked::TrackedEffects::load(dir, instance.as_deref(), name)
306        });
307        let profiles = layouts.as_deref().map_or_else(Default::default, |dir| {
308            crate::trainer::GoalProfiles::load(dir, instance.as_deref(), name)
309        });
310        let hotbars = layouts.as_deref().map_or_else(Default::default, |dir| {
311            crate::hotbars::Hotbars::load(dir, instance.as_deref(), name)
312        });
313        let remembered = layouts.as_deref().map_or_else(Default::default, |dir| {
314            crate::widget::aim::Remembered::load(dir, instance.as_deref(), name)
315        });
316        Self {
317            session,
318            name: name.to_owned(),
319            instance,
320            layouts,
321            layout,
322            tracked,
323            profiles,
324            remembered,
325            hotbars,
326            engaged: Vec::new(),
327            arranging: false,
328            moving: holders::Moving::Not,
329            #[cfg(test)]
330            room_parts: false,
331            cell: None,
332            insides: Vec::new(),
333            title_bands: std::collections::HashMap::new(),
334            retitled: Vec::new(),
335            zones: Zones::default(),
336            read: std::collections::HashMap::new(),
337            adding: None,
338            menu: None,
339            calibrating: None,
340            asking: None,
341            menus_asked: 0,
342            out: None,
343            unsaved: None,
344            save_held: None,
345            banners_gone: None,
346            input: String::new(),
347            history,
348            walked: false,
349            completing: complete::Completing::default(),
350            in_use: None,
351            unread_tabs: Vec::new(),
352            targets: (Vec::new(), None),
353            find: None,
354            trainer_popup: trainer_popup::TrainerPopup::default(),
355            shown_rects: Vec::new(),
356            history_saves: crate::store::Later::default(),
357            layout_saves: crate::store::Later::default(),
358        }

361    /// `.savelayout <name>`: this window's whole layout, every window
362    /// sized and placed as it is, saved under `name` for any character to
363    /// take. What was done.
364    ///
365    /// # Errors
366    ///
367    /// No layout yet, nowhere to keep it, or it could not be written.
368    pub(crate) fn save_layout_as(&self, name: &str) -> Result<String, String> {
369        let dir = self
370            .layouts
371            .as_deref()
372            .ok_or_else(|| "Layouts are not kept anywhere.".to_owned())?;
373        let layout = self
374            .layout
375            .as_ref()
376            .ok_or_else(|| "The window has no layout yet.".to_owned())?;
377        let path = layout.save_named(dir, name)?;
378        Ok(format!("Layout saved as {}.", path.display()))
379    }
380
381    /// `.loadlayout <name>`: the layout saved as `name` becomes this
382    /// window's, kept under the character's name from now. What was done.
383    ///
384    /// # Errors
385    ///
386    /// Nowhere layouts are kept, or no layout of that name reads.
387    pub(crate) fn load_layout_named(&mut self, name: &str) -> Result<String, String> {
388        let dir = self
389            .layouts
390            .as_deref()
391            .ok_or_else(|| "Layouts are not kept anywhere.".to_owned())?;
392        let loaded = Layout::load_named(dir, name).ok_or_else(|| {
393            let names = crate::layout::named_layouts(dir);
394            if names.is_empty() {
395                format!("There is no layout saved as {name}; .savelayout <name> saves this one.")
396            } else {
397                format!(
398                    "There is no layout saved as {name}. Saved: {}.",
399                    names.join(", ")
400                )
401            }
402        })?;
403        self.layout = Some(loaded);
404        self.arranging = false;
405        self.save_now();
406        Ok(format!("Layout {name} loaded."))
407    }

441    /// What the top bar asked, done: whether the layout changed, and what
442    /// the window is to do.
443    fn top_done(&mut self, top: Option<draw::Top>, drawers: Drawers) -> (bool, Option<Asked>) {
444        let asked = match top {
445            Some(draw::Top::Keys) => Asked::Menu(crate::main_menu::Choice::Editor(
446                crate::editors_menu::Editor::Keybinds,
447            )),
448            Some(draw::Top::Settings) => Asked::Settings(None),
449            Some(draw::Top::LayoutSettings) => Asked::Settings(Some(titlebar::PAGE.to_owned())),
450            Some(draw::Top::ReloadKeys) => Asked::ReloadKeys,
451            Some(draw::Top::Log) => Asked::Log,
452            Some(draw::Top::Menu(choice)) => Asked::Menu(choice),
453            Some(draw::Top::Zoom(by)) => Asked::Zoom(by),
454            Some(draw::Top::AddWidget) => {
455                self.adding = Some(menu::Adding::default());
456                return (false, None);
457            }
458            Some(draw::Top::Fit) => {
459                self.layout = None;
460                return (true, None);
461            }
462            Some(draw::Top::Drawers) => {
463                let layout = self.layout.as_mut();
464                return (
465                    layout.map(|layout| layout.drawers = drawers).is_some(),
466                    None,
467                );
468            }
469            Some(draw::Top::NewCustom) => {
470                let Some(layout) = &mut self.layout else {
471                    return (false, None);
472                };
473                layout.new_custom();
474                self.arranging = !layout.locked;
475                return (true, None);
476            }
477            None => return (false, None),
478        };
479        (false, Some(asked))
480    }

482    /// Draw the window into `ui` -- a viewport's whole area -- and return
483    /// what the player asked for, if anything.
484    pub(crate) fn show(&mut self, ui: &mut egui::Ui, view: &PlayView<'_>) -> Option<Asked> {
485        // Held here, or in the settings menu: a widget's page there changes
486        // this window's layout too.
487        let held = ui.input(|input| input.pointer.any_down())
488            || ui
489                .ctx()
490                .input_for(crate::settings_menu::viewport(), |input| {
491                    input.pointer.any_down()
492                });
493        if held {
494            self.save_held.get_or_insert(false);
495        } else if self.save_held.take() == Some(true) {
496            self.save_now();
497        }
498        let mut asked = None;
499        self.hotbars.fresh(Some(ui.ctx()));
500        self.completing.symbol = view.symbol;
501        if self.completing.words != view.commands {
502            self.completing.words = view.commands.to_vec();
503        }
504        let session = self.session;
505        let (mut changed, top_asked) = self.top_bar(ui, view);
506        asked = asked.or(top_asked);
507        draw::alerts(ui, session, view.story, &mut self.banners_gone);
508        // The command line: at the foot of the window, unless a Command line
509        // widget is placed, when that widget draws it.
510        let placed_input = self.layout.as_ref().is_some_and(Layout::has_input);
511        if !placed_input {
512            egui::Panel::bottom(Id::new(("play-input-panel", session))).show(ui, |ui| {
513                if let Some(line) = self.input(ui) {
514                    asked = Some(Asked::Send(line));
515                }
516            });
517        }
518        let area = ui.available_rect_before_wrap();
519        if let Some(snapshot) = view.snapshot {
520            let targeting = &snapshot.state.targeting;
521            self.targets = (targeting.ids().to_vec(), targeting.current());
522        }
523        self.ask_find(ui.ctx());
524        // A Command line widget's line, entered, is sent with what the
525        // widgets asked (`holders.rs`).
526        changed |= self.arrange(ui, view);
527        self.find_bar(ui.ctx(), area);
528        self.trainer_popup(ui.ctx(), area, view.trainer);
529        let heard: Vec<&str> = view.story.streams.ids().collect();
530        let made = (view.streams.iter().map(String::as_str)).filter(|id| !heard.contains(id));
531        let received: Vec<Widget> = (heard.iter().copied().chain(made))
532            .map(|id| Widget::Stream(id.to_owned()))
533            .chain(view.snapshot.into_iter().flat_map(|snapshot| {
534                (snapshot.state.dialogs.iter()).map(|(id, _)| Widget::Dialog(id.to_owned()))
535            }))
536            .collect();
537        let state = view.snapshot.map(|snapshot| &snapshot.state);
538        changed |= self.add_list(
539            ui.ctx(),
540            area,
541            (view.others, state),
542            view.presets,
543            &received,
544        );
545        changed |= self.right_click(ui.ctx(), area, (view.others, &received, view.streams));
546        self.object_menu(ui.ctx(), view);
547        self.calibrator(ui.ctx());
548        crate::carry::show(ui.ctx(), session);
549        asked = asked.or(self.out.take());
550        if changed {
551            self.save();
552        }
553        // The focus rule: nothing holds the keyboard, so the input takes it.
554        if ui.ctx().memory(|memory| memory.focused().is_none()) {
555            let id = self.input_id();
556            ui.ctx().memory_mut(|memory| memory.request_focus(id));
557        }
558        asked
559    }

582    /// Change the tracked effects, and keep them, when the change changed
583    /// anything.
584    pub(crate) fn track(&mut self, change: crate::tracked::Change) {
585        if !self.tracked.apply(change) {
586            return;
587        }
588        let Some(dir) = &self.layouts else {
589            return;
590        };
591        self.unsaved = self
592            .tracked
593            .save(dir, self.instance.as_deref(), &self.name)
594            .err()
595            .map(|why| why.to_string());
596    }

679    /// A goal profile saved or deleted, and kept; or loaded, which is the
680    /// change to ask of the character's trainer.
681    pub(crate) fn profiled(
682        &mut self,
683        profiled: crate::trainer::Profiled,
684    ) -> Option<crate::trainer::Ask> {
685        use crate::trainer::{Ask, Change, Profiled};
686        match profiled {
687            Profiled::Load(name) => {
688                let goals = self.profiles.profiles.get(&name)?.clone();
689                return Some(Ask::Change(Change::Aim { name, goals }));
690            }
691            Profiled::Save { name, goals } => {
692                self.profiles.profiles.insert(name.clone(), goals.clone());
693                self.change_profile(&name, Some(goals));
694            }
695            Profiled::Delete(name) => {
696                self.profiles.profiles.remove(&name)?;
697                self.change_profile(&name, None);
698            }
699        }
700        None
701    }

716    /// Keep the layout under the character's name, or say why it could not
717    /// be: at once, unless a pointer button is held, when it is kept once
718    /// the button is let go.
719    fn save(&mut self) {

721            *due = true;
722            return;
723        }
724        self.save_now();
725    }

728    /// closing, and will not see the button let go.
729    pub(crate) fn keep_waiting_layout(&mut self) {
730        if self.save_held.take() == Some(true) {

746    /// How large its window draws everything: its layout's own zoom, held
747    /// in Hydra's bounds; none, Hydra's zoom.
748    pub(crate) fn zoom(&self) -> Option<f32> {
749        self.layout
750            .as_ref()
751            .and_then(|layout| layout.zoom)
752            .filter(|zoom| zoom.is_finite())
753            .map(|zoom| zoom.clamp(hydra_board::own::ZOOM_LEAST, hydra_board::own::ZOOM_MOST))
754    }

756    /// Its window drawn `by` steps larger, smaller below 0, from `from`, the
757    /// zoom it is drawn at now: held in Hydra's bounds, rounded to a
758    /// hundredth, and kept with its layout. What it is; none with no layout
759    /// yet.
760    pub(crate) fn zoom_by(&mut self, by: i8, from: f32) -> Option<f32> {
761        use hydra_board::own::{ZOOM_LEAST, ZOOM_MOST, ZOOM_STEP};
762        let layout = self.layout.as_mut()?;
763        let zoom = from + f32::from(by) * ZOOM_STEP;
764        let zoom = (zoom.clamp(ZOOM_LEAST, ZOOM_MOST) * 100.0).round() / 100.0;
765        layout.zoom = Some(zoom);
766        self.save();
767        Some(zoom)
768    }

770    /// The top bar, drawn and its asks done: whether the layout changed,
771    /// and what the window is to do.
772    fn top_bar(&mut self, ui: &mut egui::Ui, view: &PlayView<'_>) -> (bool, Option<Asked>) {
773        let session = self.session;
774        let unsaved = self
775            .unsaved
776            .clone()
777            .or_else(|| self.layout_saves.failed())
778            .map(|why| format!("Layout not saved: {why}"))
779            .or_else(|| {
780                (self.history_saves.failed()).map(|why| format!("History not saved: {why}"))
781            });
782        let mut arranging = self.arranging;
783        let mut locked = self.layout.as_ref().is_some_and(|layout| layout.locked);
784        let set_guides = self
785            .layout
786            .as_ref()
787            .map(crate::layout::Layout::guides)
788            .unwrap_or_default();
789        let mut guides = set_guides;
790        let mut drawers = self
791            .layout
792            .as_ref()
793            .map_or_else(Drawers::default, |layout| layout.drawers);
794        let title = self
795            .layout
796            .as_ref()
797            .map(|layout| layout.title.clone())
798            .unwrap_or_default();
799        let top = egui::Panel::top(Id::new(("play-top", session)))
800            .show(ui, |ui| {
801                draw::top(
802                    ui,
803                    view,
804                    &title,
805                    &mut drawers,
806                    (&mut arranging, &mut locked, &mut guides),
807                    unsaved.as_deref(),
808                )
809            })
810            .inner;
811        if title.own {
812            draw::resize_edges(ui.ctx());
813        }
814        self.arranging = arranging && !locked;
815        if let Some(layout) = &mut self.layout
816            && layout.locked != locked
817        {
818            layout.locked = locked;
819            self.save();
820        }
821        if let Some(layout) = &mut self.layout
822            && guides != set_guides
823        {
824            layout.set_guides(guides);
825            self.save();
826        }
827        self.top_done(top, drawers)
828    }

830    /// The command input at the window's foot: Enter sends. Up and down
831    /// walk what was sent as keys bind them (`keyed.rs`).
832    fn input(&mut self, ui: &mut egui::Ui) -> Option<String> {
833        let id = self.input_id();
834        let mut typing = command_line::Typing {
835            input: &mut self.input,
836            walked: &mut self.walked,
837            history: &self.history,
838            id,
839        };
840        if command_line::command_line(ui, &mut typing) {
841            self.enter()
842        } else {
843            None
844        }
845    }

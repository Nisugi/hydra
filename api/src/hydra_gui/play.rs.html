1//! A character's play window: one command input that sends on this
2//! character, and its widgets -- the story, the room, the vitals, the
3//! clocks, the injuries and the rest -- held by windows inside it that drag,
4//! resize and snap, their layout kept by the character's name.
5//!
6//! The keys' actions are in `keyed.rs` and the Find bar in `find.rs`.
7//!
8//! A play window has one command input, and a command sent in it is sent on
9//! that character. One window per character; closing it leaves the character
10//! running, and the hub opens it again.
11//!
12//! What it shows are widgets, each one thing (`crate::widget`), held by
13//! windows inside it -- a standalone window around one, or a custom window
14//! of several, bare (`holders.rs`, `crate::layout`).
15//!
16//! **The focus rule**: a click nothing else took returns the
17//! keyboard to the command input, so the player can type without clicking it.

47/// What a play window asks for.
48#[derive(Clone, Debug, PartialEq)]
49pub(crate) enum Asked {
50    /// Send this line on the character, as typed.
51    Send(String),
52    /// Send this line the game gave -- a link's command, a dialog's button,
53    /// an object menu's entry -- once checked ([`game_line`]): one line, and
54    /// never Hydra's or Lich's.
55    FromGame(String),
56    /// Send this line as `Send` does: a menu asked for on a click, which the
57    /// player did not type, and the story does not echo (`_menu`).
58    Quietly(String),
59    /// Keep this custom window in the presets every character adds from.
60    SavePreset(crate::layout::Preset),
61    /// Forget the preset of this name.
62    ForgetPreset(String),
63    /// Use this macro set over set 0, kept for the character: a key's
64    /// action.
65    UseSet(u8),
66    /// Bring up the play window of the hub's first to ninth character: a
67    /// key's action.
68    Character(u8),
69    /// Open the settings menu on this character, at this page when one is
70    /// named: its keys, or a widget's own page.
71    Settings(Option<String>),
72    /// Read the keybinds files again.
73    ReloadKeys,
74    /// Open the character's log window.
75    Log,
76    /// Step the zoom of every window up (`1`) or down (`-1`): the title
77    /// bar's − or +.
78    Zoom(i8),
79    /// Say a room of the map in the character's story, from its minimap:
80    /// its number and title alone when `brief`.
81    Room {
82        /// The map's room number.
83        id: u32,
84        /// Only its number and title.
85        brief: bool,
86    },
87    /// Run this on the character's own command line, its command symbol
88    /// put before it: a travel command from its minimap.
89    Command(String),
90    /// Do this with the character's skill trainer.
91    Trainer(crate::trainer::Ask),
92}

293    /// `.savelayout <name>`: this window's whole layout, every window
294    /// sized and placed as it is, saved under `name` for any character to
295    /// take. What was done.
296    ///
297    /// # Errors
298    ///
299    /// No layout yet, nowhere to keep it, or it could not be written.
300    pub(crate) fn save_layout_as(&self, name: &str) -> Result<String, String> {
301        let dir = self
302            .layouts
303            .as_deref()
304            .ok_or_else(|| "Layouts are not kept anywhere.".to_owned())?;
305        let layout = self
306            .layout
307            .as_ref()
308            .ok_or_else(|| "The window has no layout yet.".to_owned())?;
309        let path = layout.save_named(dir, name)?;
310        Ok(format!("Layout saved as {}.", path.display()))
311    }

313    /// `.loadlayout <name>`: the layout saved as `name` becomes this
314    /// window's, kept under the character's name from now. What was done.
315    ///
316    /// # Errors
317    ///
318    /// Nowhere layouts are kept, or no layout of that name reads.
319    pub(crate) fn load_layout_named(&mut self, name: &str) -> Result<String, String> {
320        let dir = self
321            .layouts
322            .as_deref()
323            .ok_or_else(|| "Layouts are not kept anywhere.".to_owned())?;
324        let loaded = Layout::load_named(dir, name).ok_or_else(|| {
325            let names = crate::layout::named_layouts(dir);
326            if names.is_empty() {
327                format!("There is no layout saved as {name}; .savelayout <name> saves this one.")
328            } else {
329                format!(
330                    "There is no layout saved as {name}. Saved: {}.",
331                    names.join(", ")
332                )
333            }
334        })?;
335        self.layout = Some(loaded);
336        self.arranging = false;
337        self.save_now();
338        Ok(format!("Layout {name} loaded."))
339    }

373    /// What the top bar asked, done: whether the layout changed, and what
374    /// the window is to do.
375    fn top_done(
376        &mut self,
377        top: Option<draw::Top>,
378        grid: f32,
379        drawers: Drawers,
380    ) -> (bool, Option<Asked>) {
381        let asked = match top {
382            Some(draw::Top::Keys) => Asked::Settings(Some(crate::settings_menu::KEYS.to_owned())),
383            Some(draw::Top::Settings) => Asked::Settings(None),
384            Some(draw::Top::ReloadKeys) => Asked::ReloadKeys,
385            Some(draw::Top::Log) => Asked::Log,
386            Some(draw::Top::Zoom(by)) => Asked::Zoom(by),
387            Some(draw::Top::AddWidget) => {
388                self.adding = Some(menu::Adding::default());
389                return (false, None);
390            }
391            Some(draw::Top::Fit) => {
392                self.layout = None;
393                return (true, None);
394            }
395            Some(draw::Top::FitHere) => {
396                // Through the zones the windows were last drawn by: the
397                // frame before, in the same area.
398                let area = self.zones.area();
399                let done = area.is_finite()
400                    && self.layout.as_mut().is_some_and(|layout| {
401                        layout.fit_to(area, &self.zones);
402                        true
403                    });
404                return (done, None);
405            }
406            Some(draw::Top::Grid) => {
407                let layout = self.layout.as_mut();
408                return (layout.map(|layout| layout.grid = grid).is_some(), None);
409            }
410            Some(draw::Top::Drawers) => {
411                let layout = self.layout.as_mut();
412                return (
413                    layout.map(|layout| layout.drawers = drawers).is_some(),
414                    None,
415                );
416            }
417            Some(draw::Top::NewCustom) => {
418                let Some(layout) = &mut self.layout else {
419                    return (false, None);
420                };
421                layout.new_custom();
422                self.arranging = !layout.locked;
423                return (true, None);
424            }
425            None => return (false, None),
426        };
427        (false, Some(asked))
428    }

430    /// Draw the window into `ui` -- a viewport's whole area -- and return
431    /// what the player asked for, if anything.
432    pub(crate) fn show(&mut self, ui: &mut egui::Ui, view: &PlayView<'_>) -> Option<Asked> {
433        // Held here, or in the settings menu: a widget's page there changes
434        // this window's layout too.
435        let held = ui.input(|input| input.pointer.any_down())
436            || ui
437                .ctx()
438                .input_for(crate::settings_menu::viewport(), |input| {
439                    input.pointer.any_down()
440                });
441        if held {
442            self.save_held.get_or_insert(false);
443        } else if self.save_held.take() == Some(true) {
444            self.save_now();
445        }
446        let mut asked = None;
447        self.hotbars.fresh(Some(ui.ctx()));
448        self.completing.symbol = view.symbol;
449        if self.completing.words != view.commands {
450            self.completing.words = view.commands.to_vec();
451        }
452        let session = self.session;
453        let (mut changed, top_asked) = self.top_bar(ui, view);
454        asked = asked.or(top_asked);
455        draw::alerts(ui, session, view.story);
456        // The command line: at the foot of the window, unless a Command line
457        // widget is placed, when it is drawn where that widget is, once the
458        // widgets are (below).
459        let placed_input = self.layout.as_ref().is_some_and(Layout::has_input);
460        if !placed_input {
461            egui::Panel::bottom(Id::new(("play-input-panel", session))).show(ui, |ui| {
462                if let Some(line) = self.input(ui) {
463                    asked = Some(Asked::Send(line));
464                }
465            });
466        }
467        let area = ui.available_rect_before_wrap();
468        if let Some(snapshot) = view.snapshot {
469            let targeting = &snapshot.state.targeting;
470            self.targets = (targeting.ids().to_vec(), targeting.current());
471        }
472        self.ask_find(ui.ctx());
473        changed |= self.arrange(ui, view);
474        if placed_input && let Some(line) = self.placed_input(ui.ctx()) {
475            asked = Some(Asked::Send(line));
476        }
477        self.find_bar(ui.ctx(), area);
478        let received: Vec<Widget> = (view.story.streams.ids())
479            .map(|id| Widget::Stream(id.to_owned()))
480            .chain(view.snapshot.into_iter().flat_map(|snapshot| {
481                (snapshot.state.dialogs.iter()).map(|(id, _)| Widget::Dialog(id.to_owned()))
482            }))
483            .collect();
484        changed |= self.add_list(ui.ctx(), area, view.others, view.presets, &received);
485        changed |= self.right_click(ui.ctx(), area, view.others, &received);
486        self.object_menu(ui.ctx(), view);
487        self.calibrator(ui.ctx());
488        crate::carry::show(ui.ctx(), session);
489        asked = asked.or(self.out.take());
490        if changed {
491            self.save();
492        }
493        // The focus rule: nothing holds the keyboard, so the input takes it.
494        if ui.ctx().memory(|memory| memory.focused().is_none()) {
495            let id = self.input_id();
496            ui.ctx().memory_mut(|memory| memory.request_focus(id));
497        }
498        asked
499    }

564    /// A goal profile saved or deleted, and kept; or loaded, which is the
565    /// change to ask of the character's trainer.
566    pub(crate) fn profiled(
567        &mut self,
568        profiled: crate::trainer::Profiled,
569    ) -> Option<crate::trainer::Ask> {
570        use crate::trainer::{Ask, Change, Profiled};
571        match profiled {
572            Profiled::Load(name) => {
573                let goals = self.profiles.profiles.get(&name)?.clone();
574                return Some(Ask::Change(Change::Aim { name, goals }));
575            }
576            Profiled::Save { name, goals } => {
577                self.profiles.profiles.insert(name, goals);
578            }
579            Profiled::Delete(name) => {
580                self.profiles.profiles.remove(&name)?;
581            }
582        }
583        if let Some(dir) = &self.layouts {
584            self.unsaved = self
585                .profiles
586                .save(dir, self.instance.as_deref(), &self.name)
587                .err()
588                .map(|why| why.to_string());
589        }
590        None
591    }

604    /// Write a layout change still waiting on a held button: the window is
605    /// closing, and will not see the button let go.
606    pub(crate) fn keep_waiting_layout(&mut self) {
607        if self.save_held.take() == Some(true) {
608            self.save_now();
609        }
610    }
611
612    fn save_now(&mut self) {
613        let (Some(dir), Some(layout)) = (&self.layouts, &self.layout) else {
614            return;
615        };
616        layout.save_later(
617            &mut self.layout_saves,
618            dir,
619            (self.instance.as_deref(), &self.name),
620        );
621    }
622
623    /// The top bar, drawn and its asks done: whether the layout changed,
624    /// and what the window is to do.
625    fn top_bar(&mut self, ui: &mut egui::Ui, view: &PlayView<'_>) -> (bool, Option<Asked>) {
626        let session = self.session;
627        let mut grid = self.layout.as_ref().map_or(GRID, |layout| layout.grid);
628        let unsaved = self
629            .unsaved
630            .clone()
631            .or_else(|| self.layout_saves.failed())
632            .map(|why| format!("Layout not saved: {why}"))
633            .or_else(|| {
634                (self.history_saves.failed()).map(|why| format!("History not saved: {why}"))
635            });
636        let mut arranging = self.arranging;
637        let mut locked = self.layout.as_ref().is_some_and(|layout| layout.locked);
638        let mut drawers = self
639            .layout
640            .as_ref()
641            .map_or_else(Drawers::default, |layout| layout.drawers);
642        let title = self
643            .layout
644            .as_ref()
645            .map(|layout| layout.title.clone())
646            .unwrap_or_default();
647        let top = egui::Panel::top(Id::new(("play-top", session)))
648            .show(ui, |ui| {
649                draw::top(
650                    ui,
651                    view,
652                    &title,
653                    (&mut grid, &mut drawers),
654                    (&mut arranging, &mut locked),
655                    unsaved.as_deref(),
656                )
657            })
658            .inner;
659        if title.own {
660            draw::resize_edges(ui.ctx());
661        }
662        self.arranging = arranging && !locked;
663        if let Some(layout) = &mut self.layout
664            && layout.locked != locked
665        {
666            layout.locked = locked;
667            self.save();
668        }
669        self.top_done(top, grid, drawers)
670    }

672    /// The command line drawn where the Command line widget is, over its
673    /// place, as the widget was drawn this frame; nothing while the widget
674    /// is a tab behind another.
675    fn placed_input(&mut self, ctx: &egui::Context) -> Option<String> {
676        let rect = self
677            .layout
678            .as_ref()?
679            .placed()
680            .into_iter()
681            .find(|placed| placed.widget == Widget::Input)
682            .and_then(|placed| {
683                self.shown_rects
684                    .iter()
685                    .find(|(id, _)| *id == placed.id)
686                    .map(|(_, rect)| *rect)
687            })?;
688        egui::Area::new(Id::new(("play-input-area", self.session)))
689            .order(egui::Order::Middle)
690            .fixed_pos(rect.min)
691            .show(ctx, |ui| {
692                ui.set_clip_rect(rect);
693                ui.set_width(rect.width());
694                ui.set_height(rect.height());
695                self.input(ui)
696            })
697            .inner
698    }

700    /// The command input: Enter sends. Up and down walk what was sent as
701    /// keys bind them (`keyed.rs`).
702    fn input(&mut self, ui: &mut egui::Ui) -> Option<String> {
703        let id = self.input_id();
704        let output = egui::TextEdit::singleline(&mut self.input)
705            .id(id)
706            // Tab is the keys', never egui's to move the keyboard on.
707            .lock_focus(true)
708            .hint_text("Type a command")
709            .desired_width(f32::INFINITY)
710            .show(ui);
711        let response = output.response.response.clone();
712        if std::mem::take(&mut self.walked) {
713            let mut state = egui::TextEdit::load_state(ui.ctx(), id).unwrap_or_default();
714            let end = egui::text::CCursor::new(self.input.chars().count());
715            state
716                .cursor
717                .set_char_range(Some(egui::text::CCursorRange::one(end)));
718            state.store(ui.ctx(), id);
719        }
720        self.ghost(ui, &output);
721        let entered =
722            response.lost_focus() && ui.input(|input| input.key_pressed(egui::Key::Enter));
723        if !entered {
724            return None;
725        }
726        response.request_focus();
727        self.enter()
728    }
729
730    /// The ghost after what is typed (`complete.rs`), in the weak colour,
731    /// while the cursor stands at the end with nothing selected.
732    fn ghost(&self, ui: &egui::Ui, output: &egui::text_edit::TextEditOutput) {
733        let Some(range) = output.cursor_range else {
734            return;
735        };
736        let end = self.input.chars().count();
737        if range.primary.index.0 != end || range.secondary.index.0 != end {
738            return;
739        }
740        let Some(rest) = complete::ghost(&self.input, &self.history) else {
741            return;
742        };
743        let font = egui::TextStyle::Body.resolve(ui.style());
744        let galley = ui
745            .painter()
746            .layout_no_wrap(rest, font, ui.visuals().weak_text_color());
747        let at = egui::pos2(
748            output.galley_pos.x + output.galley.rect.width(),
749            output.galley_pos.y,
750        );
751        // Inside the field, not the text's own clip, which hugs what is
752        // typed and would cut the ghost away.
753        let field = output.response.rect.shrink2(egui::vec2(0.0, 1.0));
754        ui.painter()
755            .with_clip_rect(field)
756            .galley(at, galley, ui.visuals().weak_text_color());
757    }

1//! The editor's *Packs*: the studio catalogue's packs of the kinds Hydra
2//! reads -- dolls, compass sets, the gauge art and frame kits -- fetched
3//! and installed on a click.
4//!
5//! **Grouped to stay short.** A section for each kind, closed until opened,
6//! says how many it has; a kind with more than [`GROUP`] is in groups of
7//! that many, *Sets 1–10*, *Sets 11–20*, in the catalogue's order, so a set
8//! published later joins the end and no group's sets shift. A group's
9//! previews are fetched as it is first opened, not the whole catalogue's.
10//! What is installed shows the same way before the catalogue is looked at,
11//! each pack by its kept preview or its first picture.
12

17/// How many packs a group of a kind holds.
18const GROUP: usize = 10;

20/// The packs the studio's catalogue offers of the kinds Hydra reads: each
21/// with its kind, version and size, installed or not, and a button to get,
22/// update or remove it.
23pub(crate) fn show(ui: &mut egui::Ui) {
24    let Some(pool) = crate::frames::pool(ui.ctx()) else {
25        return;
26    };
27    let repaint = {
28        let ctx = ui.ctx().clone();
29        move || ctx.request_repaint()
30    };
31    let (listing, fetches) = packs::snapshot();
32    ui.horizontal(|ui| {
33        let label = match listing {
34            Listing::Idle => "Look for packs",
35            Listing::Fetching => "Looking…",
36            Listing::Ready(_) | Listing::Failed(_) => "Look again",
37        };
38        if ui
39            .add_enabled(listing != Listing::Fetching, egui::Button::new(label))
40            .on_hover_text(packs::catalogue_url())
41            .clicked()
42        {
43            packs::list(repaint.clone());
44        }
45    });
46    // The pool's ledger, read once a frame: what is installed, and of it
47    // what is shown under its kind when the catalogue does not offer it.
48    let kept = packs::kept(&pool);
49    let have = packs::installed(&kept);
50    let mut offered: Vec<&str> = Vec::new();
51    match &listing {
52        Listing::Idle => {
53            if have.is_empty() {
54                ui.weak(
55                    "Packs of art -- dolls, compasses, gauges and frames -- fetched from the studio's catalogue on a click.",
56                );
57            }
58        }
59        Listing::Fetching => {}
60        Listing::Failed(why) => {
61            ui.colored_label(ui.visuals().warn_fg_color, why);
62        }
63        Listing::Ready(catalogue) => {
64            offered = kinds(ui, catalogue, (&fetches, &have), &pool, &repaint);
65        }
66    }
67    // What is installed and not offered any more, or all of it before the
68    // catalogue is looked at.
69    installed_only(ui, &kept, &offered, &pool);
70}

72/// The catalogue's packs of Hydra's kinds, a section for each kind and,
73/// in a kind of more than [`GROUP`], a group for each [`GROUP`] of them: a
74/// group's previews asked for as it is opened. The ids offered, opened or
75/// not.
76pub(super) fn kinds<'a>(
77    ui: &mut egui::Ui,
78    catalogue: &'a Catalogue,
79    (fetches, have): (&BTreeMap<String, Fetch>, &BTreeMap<String, String>),
80    pool: &std::path::Path,
81    repaint: &(impl Fn() + Clone + Send + 'static),
82) -> Vec<&'a str> {
83    if catalogue.hydras().next().is_none() {
84        ui.weak("No packs offered yet.");
85    }
86    let rows = |ui: &mut egui::Ui, group: &[&Offered]| {
87        packs::previews(group, pool, repaint.clone());
88        for pack in group {
89            let state = (
90                fetches.get(&pack.id),
91                have.get(&pack.id).map(String::as_str),
92            );
93            ui.horizontal(|ui| row(ui, pack, state, pool, repaint));
94        }
95    };
96    let mut offered = Vec::new();
97    for kind in packs::KINDS {
98        let of_kind: Vec<&Offered> = catalogue
99            .hydras()
100            .filter(|pack| pack.kind == kind)
101            .collect();
102        if of_kind.is_empty() {
103            continue;
104        }
105        offered.extend(of_kind.iter().map(|pack| pack.id.as_str()));
106        let title = format!("{} ({})", packs::kind_title(kind), of_kind.len());
107        egui::CollapsingHeader::new(title)
108            .id_salt(("packs-kind", kind))
109            .show(ui, |ui| {
110                if of_kind.len() <= GROUP {
111                    rows(ui, &of_kind);
112                    return;
113                }
114                for (at, group) in of_kind.chunks(GROUP).enumerate() {
115                    egui::CollapsingHeader::new(group_title(at, group.len()))
116                        .id_salt(("packs-group", kind, at))
117                        .show(ui, |ui| rows(ui, group));
118                }
119            });
120    }
121    offered
122}

124/// The title of the `at`th group of a kind, of `len` packs: *Sets 11–20*.
125pub(super) fn group_title(at: usize, len: usize) -> String {
126    let first = at * GROUP + 1;
127    format!("Sets {first}–{}", first + len.max(1) - 1)
128}

130/// One offered pack's row: its title, version and size, where its
131/// fetch stands or whether it is installed, and the button that fits.
132fn row(
133    ui: &mut egui::Ui,
134    pack: &packs::Offered,
135    (fetch, installed): (Option<&Fetch>, Option<&str>),
136    pool: &std::path::Path,
137    repaint: &(impl Fn() + Clone + Send + 'static),
138) {
139    preview(ui, pack, pool);
140    ui.label(pack.shown()).on_hover_text(&pack.id);
141    ui.weak(format!("{} · {}", pack.version, packs::sized(pack.size)));
142    match (fetch, installed) {
143        (Some(Fetch::Fetching), _) => {
144            ui.weak("fetching…");
145        }
146        (Some(Fetch::Failed(why)), _) => {
147            if ui.small_button("Try again").clicked() {
148                packs::get(pack, pool, repaint.clone());
149            }
150            ui.colored_label(ui.visuals().warn_fg_color, why);
151        }
152        (Some(Fetch::Landed(summary)), _) => {
153            ui.weak("installed").on_hover_text(summary);
154        }
155        (None, Some(version)) if version == pack.version => {
156            ui.weak("installed");
157        }
158        (None, Some(version)) => {
159            if ui
160                .small_button("Update")
161                .on_hover_text(format!("{version} installed"))
162                .clicked()
163            {
164                packs::get(pack, pool, repaint.clone());
165            }
166        }
167        (None, None) => {
168            if ui.small_button("Get").clicked() {
169                packs::get(pack, pool, repaint.clone());
170            }
171        }
172    }
173    if installed.is_some() && ui.small_button("Remove").clicked() {
174        let said = packs::remove(pool, &pack.id).unwrap_or_else(|why| why);
175        ui.weak(said);
176    }
177}

179/// How tall a preview is drawn in the list.
180const PREVIEW_TALL: f32 = 60.0;
181
182/// `pack`'s preview, once fetched ([`picture`]).
183fn preview(ui: &mut egui::Ui, pack: &packs::Offered, pool: &std::path::Path) {
184    picture(
185        ui,
186        (&pack.id, pack.shown()),
187        &packs::preview_path(pool, &pack.id),
188    );
189}

191/// Pack `id`'s picture at `path`, called `title`, where there is one:
192/// drawn [`PREVIEW_TALL`] in the row, no wider than half the row, so a
193/// gauge pack's contact sheet (every piece in seven colours, over a
194/// thousand pixels wide) stays a glance; clicked, it opens full size in a
195/// window of its own, and closes on another click or its close button.
196fn picture(ui: &mut egui::Ui, (id, title): (&str, &str), path: &std::path::Path) {
197    if !path.is_file() {
198        return;
199    }
200    let Some(texture) = crate::pictures::picture(ui.ctx(), &path.to_string_lossy()) else {
201        return;
202    };
203    let [width, height] = texture.size();
204    #[allow(clippy::cast_precision_loss)]
205    let natural = egui::vec2(width as f32, height as f32);
206    let scale =
207        (PREVIEW_TALL / natural.y.max(1.0)).min(ui.available_width() * 0.5 / natural.x.max(1.0));
208    let small = natural * scale;
209    let open_key = egui::Id::new(("pack-preview-open", id));
210    let mut open = ui.data(|data| data.get_temp(open_key)).unwrap_or(false);
211    if ui
212        .add(egui::Button::image((texture.id(), small)).frame(false))
213        .on_hover_text("Full size")
214        .clicked()
215    {
216        open = !open;
217    }
218    if open {
219        let shown = natural.min(ui.ctx().content_rect().size() * 0.9);
220        egui::Window::new(title)
221            .id(open_key.with("window"))
222            .open(&mut open)
223            .default_size(shown)
224            .show(ui.ctx(), |ui| {
225                egui::ScrollArea::both().show(ui, |ui| {
226                    ui.image((texture.id(), natural));
227                });
228            });
229    }
230    ui.data_mut(|data| data.insert_temp(open_key, open));
231}

233/// The packs installed that the catalogue does not offer, or every one
234/// before it is looked at: a section for each kind, by the kind its ledger
235/// says, closed, with its count; a kind of more than [`GROUP`] in groups of
236/// that many. Each pack with its kept preview, or its first picture where
237/// none is kept, its name under its kind, its version and Remove.
238pub(super) fn installed_only(
239    ui: &mut egui::Ui,
240    kept: &BTreeMap<String, packs::Kept>,
241    offered: &[&str],
242    pool: &std::path::Path,
243) {
244    let shown: Vec<(&String, &packs::Kept)> = kept
245        .iter()
246        .filter(|(id, _)| !offered.contains(&id.as_str()))
247        .collect();
248    if shown.is_empty() {
249        return;
250    }
251    crate::form::section(
252        ui,
253        if offered.is_empty() {
254            "Installed"
255        } else {
256            "Installed, no longer offered"
257        },
258    );
259    let rows = |ui: &mut egui::Ui, group: &[&(&String, &packs::Kept)]| {
260        for (id, pack) in group {
261            ui.horizontal(|ui| kept_row(ui, (id, pack), pool));
262        }
263    };
264    for kind in packs::KINDS {
265        let of_kind: Vec<&(&String, &packs::Kept)> =
266            shown.iter().filter(|(_, pack)| pack.kind == kind).collect();
267        if of_kind.is_empty() {
268            continue;
269        }
270        let title = format!("{} ({})", packs::kind_title(kind), of_kind.len());
271        egui::CollapsingHeader::new(title)
272            .id_salt(("packs-installed", kind))
273            .show(ui, |ui| {
274                if of_kind.len() <= GROUP {
275                    rows(ui, &of_kind);
276                    return;
277                }
278                for (at, group) in of_kind.chunks(GROUP).enumerate() {
279                    egui::CollapsingHeader::new(group_title(at, group.len()))
280                        .id_salt(("packs-installed-group", kind, at))
281                        .show(ui, |ui| rows(ui, group));
282                }
283            });
284    }
285}

287/// One installed pack's row: its picture, its name under its kind, its
288/// version, and Remove.
289fn kept_row(ui: &mut egui::Ui, (id, pack): (&str, &packs::Kept), pool: &std::path::Path) {
290    let name = packs::kept_name(id, &pack.kind);
291    let preview = packs::preview_path(pool, id);
292    match &pack.picture {
293        Some(own) if !preview.is_file() => picture(ui, (id, &name), own),
294        _ => picture(ui, (id, &name), &preview),
295    }
296    ui.label(&name).on_hover_text(id);
297    crate::form::hint(ui, format!("{} · installed", pack.version));
298    if ui
299        .small_button("Remove")
300        .on_hover_text("Take the pack out of the data folder")
301        .clicked()
302    {
303        let said = packs::remove(pool, id).unwrap_or_else(|why| why);
304        crate::form::hint(ui, said);
305    }
306}

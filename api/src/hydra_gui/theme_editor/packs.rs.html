6/// The packs the studio's catalogue offers of the kinds Hydra reads,
7/// frames and dolls: each with its kind, version and size, installed or
8/// not, and a button to get, update or remove it.
9pub(super) fn show(ui: &mut egui::Ui) {
10    let Some(pool) = crate::frames::pool(ui.ctx()) else {
11        return;
12    };
13    let repaint = {
14        let ctx = ui.ctx().clone();
15        move || ctx.request_repaint()
16    };
17    let (listing, fetches) = packs::snapshot();
18    ui.horizontal(|ui| {
19        let label = match listing {
20            Listing::Idle => "Look for packs",
21            Listing::Fetching => "Looking…",
22            Listing::Ready(_) | Listing::Failed(_) => "Look again",
23        };
24        if ui
25            .add_enabled(listing != Listing::Fetching, egui::Button::new(label))
26            .on_hover_text(packs::catalogue_url())
27            .clicked()
28        {
29            packs::list(&pool, repaint.clone());
30        }
31    });
32    let have = packs::installed(&pool);
33    let mut offered: Vec<&str> = Vec::new();
34    match &listing {
35        Listing::Idle => {
36            if have.is_empty() {
37                ui.weak(
38                    "Packs of parts and recipes, fetched from the studio's catalogue on a click.",
39                );
40            }
41        }
42        Listing::Fetching => {}
43        Listing::Failed(why) => {
44            ui.colored_label(ui.visuals().warn_fg_color, why);
45        }
46        Listing::Ready(catalogue) => {
47            if catalogue.hydras().next().is_none() {
48                ui.weak("No packs offered yet.");
49            }
50            for pack in catalogue.hydras() {
51                offered.push(&pack.id);
52                let state = (
53                    fetches.get(&pack.id),
54                    have.get(&pack.id).map(String::as_str),
55                );
56                ui.horizontal(|ui| row(ui, pack, state, &pool, &repaint));
57            }
58        }
59    }
60    // What is installed and not offered any more.
61    installed_only(ui, &have, &offered, &pool);
62}

64/// One offered pack's row: its title, kind, version and size, where its
65/// fetch stands or whether it is installed, and the button that fits.
66fn row(
67    ui: &mut egui::Ui,
68    pack: &packs::Offered,
69    (fetch, installed): (Option<&Fetch>, Option<&str>),
70    pool: &std::path::Path,
71    repaint: &(impl Fn() + Clone + Send + 'static),
72) {
73    // Its preview, once fetched, about three lines tall.
74    let preview = packs::preview_path(pool, &pack.id);
75    if preview.is_file()
76        && let Some(texture) = crate::pictures::picture(ui.ctx(), &preview.to_string_lossy())
77    {
78        let [width, height] = texture.size();
79        #[allow(clippy::cast_precision_loss)]
80        let wide = 60.0 * width as f32 / (height.max(1)) as f32;
81        ui.image((texture.id(), egui::vec2(wide, 60.0)));
82    }
83    ui.label(pack.shown()).on_hover_text(&pack.id);
84    ui.weak(format!(
85        "{} · {} · {}",
86        pack.kind,
87        pack.version,
88        packs::sized(pack.size)
89    ));
90    match (fetch, installed) {
91        (Some(Fetch::Fetching), _) => {
92            ui.weak("fetching…");
93        }
94        (Some(Fetch::Failed(why)), _) => {
95            if ui.small_button("Try again").clicked() {
96                packs::get(pack, pool, repaint.clone());
97            }
98            ui.colored_label(ui.visuals().warn_fg_color, why);
99        }
100        (Some(Fetch::Landed(summary)), _) => {
101            ui.weak("installed").on_hover_text(summary);
102        }
103        (None, Some(version)) if version == pack.version => {
104            ui.weak("installed");
105        }
106        (None, Some(version)) => {
107            if ui
108                .small_button("Update")
109                .on_hover_text(format!("{version} installed"))
110                .clicked()
111            {
112                packs::get(pack, pool, repaint.clone());
113            }
114        }
115        (None, None) => {
116            if ui.small_button("Get").clicked() {
117                packs::get(pack, pool, repaint.clone());
118            }
119        }
120    }
121    if installed.is_some() && ui.small_button("Remove").clicked() {
122        let said = packs::remove(pool, &pack.id).unwrap_or_else(|why| why);
123        ui.weak(said);
124    }
125}

127/// The packs installed that the catalogue no longer offers, or offered
128/// before it was looked at, each with Remove.
129fn installed_only(
130    ui: &mut egui::Ui,
131    have: &std::collections::BTreeMap<String, String>,
132    offered: &[&str],
133    pool: &std::path::Path,
134) {
135    for (id, version) in have {
136        if offered.contains(&id.as_str()) {
137            continue;
138        }
139        ui.horizontal(|ui| {
140            ui.label(id);
141            ui.weak(format!("{version} · installed"));
142            if ui.small_button("Remove").clicked() {
143                let said = packs::remove(pool, id).unwrap_or_else(|why| why);
144                ui.weak(said);
145            }
146        });
147    }

1//! The editor's *Shape*: each kind of thing's corners, the edges, the
2//! density, the scrollbar, the shadows, the panes' gap and padding, the
3//! bars' thickness, the title bars' height and the buttons.
4
5use hydra_ui::theme::{Buttons, Density, Shape, TabStyle, TitleAlign};
6
7/// Draw the shape's controls into `ui`, changing `shape` as the player
8/// does.
9pub(super) fn show(ui: &mut egui::Ui, shape: &mut Shape) {
10    ui.strong("Corners");
11    let corners = &mut shape.corners;
12    for (value, label, help) in [
13        (
14            &mut corners.controls,
15            "Controls",
16            "Buttons, fields, checkboxes, dropdowns, tabs",
17        ),
18        (&mut corners.windows, "Windows", "Windows, menus and popups"),
19        (&mut corners.bars, "Bars", "The vitals and every other bar"),
20        (
21            &mut corners.panes,
22            "In panes",
23            "Indicators, compass cells, a find's hits, the window in use's outline",
24        ),
25    ] {
26        ui.add(egui::Slider::new(value, 0..=32).text(label))
27            .on_hover_text(help);
28    }
29    ui.separator();
30    ui.add(egui::Slider::new(&mut shape.stroke, 0.0..=8.0).text("Edges"))
31        .on_hover_text("An edge's width, on controls and windows");
32    ui.horizontal(|ui| {
33        ui.label("Density");
34        for density in Density::ALL {
35            ui.selectable_value(&mut shape.density, density, density.name());
36        }
37    })
38    .response
39    .on_hover_text("The gaps between controls and the padding in them");
40    ui.horizontal(|ui| {
41        ui.label("Buttons");
42        for buttons in Buttons::ALL {
43            ui.selectable_value(&mut shape.buttons, buttons, buttons.name());
44        }
45    })
46    .response
47    .on_hover_text("How a button is drawn at rest; every style fills under the pointer");
48    ui.add(egui::Slider::new(&mut shape.scrollbar, 2.0..=32.0).text("Scrollbar"));
49    ui.checkbox(&mut shape.shadows, "Windows and menus cast shadows");
50    ui.separator();
51    ui.add(egui::Slider::new(&mut shape.pane_gap, 0.0..=32.0).text("Gap between cells"))
52        .on_hover_text("Between the cells of a custom window");
53    ui.add(egui::Slider::new(&mut shape.pane_padding, 0.0..=32.0).text("Room inside a pane"))
54        .on_hover_text("Round each widget, inside its pane");
55    ui.add(egui::Slider::new(&mut shape.bar_thickness, 0.0..=200.0).text("Bar thickness"))
56        .on_hover_text("The most a bar is thick, centred in its pane; 0 fills the pane");
57    ui.add(egui::Slider::new(&mut shape.title_height, 0.0..=64.0).text("Title bars"))
58        .on_hover_text("A window's title bar and a cell's tabs; 0 is the usual height");
59    ui.horizontal(|ui| {
60        ui.label("Title text");
61        for align in TitleAlign::ALL {
62            ui.selectable_value(&mut shape.title_align, align, align.name());
63        }
64    })
65    .response
66    .on_hover_text("Where a window's title sits in its title bar");
67    ui.horizontal(|ui| {
68        ui.label("Tabs");
69        for style in TabStyle::ALL {
70            ui.selectable_value(&mut shape.tabs, style, style.name());
71        }
72    })
73    .response
74    .on_hover_text(
75        "How a cell's tabs are drawn: their words alone, a line between them, an edge \
76         round each, or each on the tab ground; a cell may choose its own",
77    );
78}

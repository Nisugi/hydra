1//! The editor's *Pins*: every token by its group, each as it comes out of
2//! the harmony, pinned to a colour of the player's own with a click, and
3//! for a free token the other hues it could take.
4
5use hydra_ui::theme::{Group, Palette, Recipe, Role, Token, hue_variants};
6
7/// The groups in the order shown, each with what to call it.
8const GROUPS: [(Group, &str); 8] = [
9    (Group::Surfaces, "Surfaces and text"),
10    (Group::Text, "The story and the room"),
11    (Group::Vitals, "Vitals"),
12    (Group::Status, "Status"),
13    (Group::Injuries, "Injuries"),
14    (Group::Map, "The map"),
15    (Group::Marks, "The map's marks"),
16    (Group::Chrome, "Chrome"),
17];
18
19/// Draw every token's row into `ui`: its colour in `palette` now, its pin
20/// in `recipe` made, moved or taken off.
21pub(super) fn show(ui: &mut egui::Ui, recipe: &mut Recipe, palette: &Palette) {
22    for (group, called) in GROUPS {
23        egui::CollapsingHeader::new(called)
24            .id_salt(("theme-group", called))
25            .show(ui, |ui| {
26                for token in Token::ALL.into_iter().filter(|t| t.group() == group) {
27                    row(ui, recipe, palette, token);
28                }
29            });
30    }
31}
32
33/// One token's row: its name, its pin's switch, its colour, and its hues.
34fn row(ui: &mut egui::Ui, recipe: &mut Recipe, palette: &Palette, token: Token) {
35    ui.horizontal(|ui| {
36        let mut pinned = recipe.pins.contains_key(&token);
37        if ui
38            .checkbox(&mut pinned, "")
39            .on_hover_text("Pinned: kept as written; the rest harmonise round it")
40            .changed()
41        {
42            if pinned {
43                recipe.pins.insert(token, palette.get(token));
44            } else {
45                recipe.pins.remove(&token);
46            }
47        }
48        let mut rgb = recipe
49            .pins
50            .get(&token)
51            .copied()
52            .unwrap_or(palette.get(token));
53        if ui.color_edit_button_srgb(&mut rgb).changed() {
54            recipe.pins.insert(token, rgb);
55        }
56        ui.label(token.name());
57        if let Role::Free { .. } = token.role() {
58            for hue in hue_variants(token, recipe).into_iter().skip(1).take(6) {
59                let (rect, response) =
60                    ui.allocate_exact_size(egui::vec2(12.0, 12.0), egui::Sense::click());
61                ui.painter().rect_filled(rect, 2.0, crate::theme::rgb(hue));
62                if response
63                    .on_hover_text(format!(
64                        "Pin {} to {}",
65                        token.name(),
66                        hydra_ui::theme::hex(hue)
67                    ))
68                    .clicked()
69                {
70                    recipe.pins.insert(token, hue);
71                }
72            }
73        }
74    });
75}

1//! The editor's *Pins*: every token by its group, each as it comes out of
2//! the harmony, pinned to a colour of the player's own with a click, and
3//! for a free token the other hues it could take. A kind of line -- a
4//! creature's death, its spell coming -- can also be *None*: drawn as the
5//! story's own text, as if it had no colour.
6//!
7//! Under a kind of line named one by one -- a flare, a status on the
8//! character -- each of its names can take a colour of its own over the
9//! kind's, or *None*: the names the combat reader's tables have, and any
10//! the theme colours that they no longer have.
11
12use hydra_ui::combat_line;
13use hydra_ui::theme::{Group, Palette, Recipe, Role, Token, hue_variants};

15/// The groups in the order shown, each with what to call it.
16const GROUPS: [(Group, &str); 9] = [
17    (Group::Surfaces, "Surfaces and text"),
18    (Group::Text, "The story and the room"),
19    (Group::Channels, "Channels"),
20    (Group::Vitals, "Vitals"),
21    (Group::Status, "Status"),
22    (Group::Injuries, "Injuries"),
23    (Group::Map, "The map"),
24    (Group::Marks, "The map's marks"),
25    (Group::Chrome, "Chrome"),
26];
27
28/// Draw every token's row into `ui`: its colour in `palette` now, its pin
29/// in `recipe` made, moved or taken off.
30pub(super) fn show(ui: &mut egui::Ui, recipe: &mut Recipe, palette: &Palette) {
31    for (group, called) in GROUPS {
32        egui::CollapsingHeader::new(called)
33            .id_salt(("theme-group", called))
34            .show(ui, |ui| {
35                for token in Token::ALL.into_iter().filter(|t| t.group() == group) {
36                    row(ui, recipe, palette, token);
37                    if let Some((kind, _)) = combat_line::NAMED.iter().find(|(_, t)| *t == token) {
38                        by_name(ui, recipe, palette, kind, token);
39                    }
40                }
41            });
42    }
43}

45/// One token's row: its name, its pin's switch, its colour, and its hues;
46/// for a kind of line, *None* before them.
47fn row(ui: &mut egui::Ui, recipe: &mut Recipe, palette: &Palette, token: Token) {
48    ui.horizontal(|ui| {
49        if token.may_be_plain() {
50            let mut none = recipe.plain.contains(&token);
51            if ui
52                .checkbox(&mut none, "None")
53                .on_hover_text("Drawn as the story's own text, with no colour of its own")
54                .changed()
55            {
56                if none {
57                    recipe.plain.insert(token);
58                    recipe.pins.remove(&token);
59                } else {
60                    recipe.plain.remove(&token);
61                }
62            }
63            if none {
64                ui.label(token.name());
65                return;
66            }
67        }
68        let mut pinned = recipe.pins.contains_key(&token);
69        if ui
70            .checkbox(&mut pinned, "")
71            .on_hover_text("Pinned: kept as written; the rest harmonise round it")
72            .changed()
73        {
74            if pinned {
75                recipe.pins.insert(token, palette.get(token));
76            } else {
77                recipe.pins.remove(&token);
78            }
79        }
80        let mut rgb = recipe
81            .pins
82            .get(&token)
83            .copied()
84            .unwrap_or(palette.get(token));
85        if ui.color_edit_button_srgb(&mut rgb).changed() {
86            recipe.pins.insert(token, rgb);
87        }
88        ui.label(token.name());
89        if let Role::Free { .. } = token.role() {
90            for hue in hue_variants(token, recipe).into_iter().skip(1).take(6) {
91                let (rect, response) =
92                    ui.allocate_exact_size(egui::vec2(12.0, 12.0), egui::Sense::click());
93                ui.painter().rect_filled(rect, 2.0, crate::theme::rgb(hue));
94                if response
95                    .on_hover_text(format!(
96                        "Pin {} to {}",
97                        token.name(),
98                        hydra_ui::theme::hex(hue)
99                    ))
100                    .clicked()
101                {
102                    recipe.pins.insert(token, hue);
103                }
104            }
105        }
106    });
107}

109/// The names of `kind`, a kind of line named one by one and coloured as
110/// `token`: those the theme colours first, then the rest, narrowed by a
111/// filter.
112fn by_name(ui: &mut egui::Ui, recipe: &mut Recipe, palette: &Palette, kind: &str, token: Token) {
113    let prefix = format!("{kind}:");
114    let own: Vec<String> = recipe
115        .names
116        .keys()
117        .filter_map(|preset| preset.strip_prefix(&prefix))
118        .map(str::to_owned)
119        .collect();
120    let called = if kind == combat_line::FLARE {
121        "flare"
122    } else {
123        "status on you"
124    };
125    egui::CollapsingHeader::new(format!("Each {called} by name ({} coloured)", own.len()))
126        .id_salt(("theme-names", kind))
127        .show(ui, |ui| {
128            let id = egui::Id::new(("theme-names-filter", kind));
129            let mut filter: String = ui.data(|data| data.get_temp(id)).unwrap_or_default();
130            ui.add(egui::TextEdit::singleline(&mut filter).hint_text("Filter"));
131            ui.data_mut(|data| data.insert_temp(id, filter.clone()));
132            let filter = filter.trim().to_lowercase();
133            let known = combat_line::names(kind);
134            let rest = known.iter().filter(|name| !own.iter().any(|o| o == *name));
135            let listed: Vec<&str> = own
136                .iter()
137                .map(String::as_str)
138                .chain(rest.copied())
139                .filter(|name| name.contains(&filter))
140                .collect();
141            egui::ScrollArea::vertical()
142                .id_salt(("theme-names-list", kind))
143                .max_height(240.0)
144                .show(ui, |ui| {
145                    for name in listed {
146                        name_row(
147                            ui,
148                            recipe,
149                            palette,
150                            (kind, name),
151                            token,
152                            known.contains(&name),
153                        );
154                    }
155                });
156        });
157}

159/// One name's row: *None*, its own colour's switch, its colour -- its own,
160/// else its kind's -- and the name; one the tables do not have is said so.
161fn name_row(
162    ui: &mut egui::Ui,
163    recipe: &mut Recipe,
164    palette: &Palette,
165    (kind, name): (&str, &str),
166    token: Token,
167    known: bool,
168) {
169    let preset = format!("{kind}:{name}");
170    ui.horizontal(|ui| {
171        let own = recipe.names.get(&preset).copied();
172        let mut none = own == Some(None);
173        if ui
174            .checkbox(&mut none, "None")
175            .on_hover_text("Drawn as the story's own text, with no colour of its own")
176            .changed()
177        {
178            if none {
179                recipe.names.insert(preset.clone(), None);
180            } else {
181                recipe.names.remove(&preset);
182            }
183        }
184        let label = |ui: &mut egui::Ui| {
185            let label = ui.label(name);
186            if !known {
187                label.on_hover_text("Not a name the combat reader has now: kept as the theme says");
188            }
189        };
190        if none {
191            label(ui);
192            return;
193        }
194        let shown = palette.named(&preset).unwrap_or(palette.get(token));
195        let mut set = matches!(own, Some(Some(_)));
196        if ui
197            .checkbox(&mut set, "")
198            .on_hover_text("A colour of its own; unticked, its kind's")
199            .changed()
200        {
201            if set {
202                recipe.names.insert(preset.clone(), Some(shown));
203            } else {
204                recipe.names.remove(&preset);
205            }
206        }
207        let mut rgb = shown;
208        if ui.color_edit_button_srgb(&mut rgb).changed() {
209            recipe.names.insert(preset.clone(), Some(rgb));
210        }
211        label(ui);
212    });
213}

215#[cfg(test)]
216mod tests {
217    use egui::accesskit::Role as Access;
218    use egui_kittest::Harness;
219    use egui_kittest::kittest::Queryable;
220    use hydra_ui::theme::generate;
221
222    use super::*;
223
224    /// The flares by name: a filter narrows them to the names holding it,
225    /// *None* leaves one plain, and *None* again gives it back its kind's
226    /// colour.
227    #[test]
228    fn a_flare_is_coloured_by_its_name() {
229        let mut harness = Harness::new_ui_state(
230            |ui, recipe: &mut Recipe| {
231                let palette = generate(recipe);
232                by_name(ui, recipe, &palette, combat_line::FLARE, Token::Flare);
233            },
234            Recipe::default(),
235        );
236        harness
237            .get_by_label("Each flare by name (0 coloured)")
238            .click();
239        harness.run();
240        harness.get_by_role(Access::TextInput).click();
241        harness.run();
242        harness.get_by_role(Access::TextInput).type_text("acid");
243        harness.run();
244        assert!(harness.query_by_label("acid").is_some());
245        assert!(harness.query_by_label("minoracid904_melt").is_some());
246        assert!(harness.query_by_label("fire").is_none(), "filtered out");
247
248        let first_none = |harness: &Harness<'_, Recipe>| {
249            harness
250                .get_all_by_label("None")
251                .next()
252                .expect("a None")
253                .click();
254        };
255        first_none(&harness);
256        harness.run();
257        assert_eq!(harness.state().names.get("flare:acid"), Some(&None));
258        assert!(
259            harness
260                .query_by_label("Each flare by name (1 coloured)")
261                .is_some()
262        );
263        first_none(&harness);
264        harness.run();
265        assert!(harness.state().names.is_empty());
266    }

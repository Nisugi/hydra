1//! The hotbar editor (`.hotbars`): every bar on the left, marked as every
2//! character's or this character's own; the bar chosen on the right -- its
3//! name, where it is kept, its grid and its buttons' size, drawn as its
4//! widget draws it against the character now -- and its slots, each filled
5//! with a macro. A bar is only a layout: what a button shows and sends is
6//! its macro's, made on the Macros page. The look pieces in `button.rs`
7//! are the macro editor's.
8//!
9//! Nothing is written here: the editor asks the app ([`Asked`]), which
10//! writes through the character's hotbars (`crate::hotbars`), the rest of
11//! each file kept.
12

25/// What the editor asks of the app.
26#[derive(Clone, Debug, PartialEq)]
27pub(crate) enum Asked {
28    /// Keep `bar` in the character's own file (`own`) or every
29    /// character's, in place of the bar `was`: its name, and whether it was
30    /// the character's own.
31    Save {
32        bar: Box<Bar>,
33        own: bool,
34        was: Option<(String, bool)>,
35    },
36    /// Take the bar called this out of the character's own file or every
37    /// character's.
38    Remove { name: String, own: bool },
39    /// Copy a file into the icons folder: a sheet, named so, or an image.
40    CopyIn(Adding),
41}
42
43/// What the editor is drawn over: the character's bars, each with whether
44/// it is the character's own; the character, and its state now.
45pub(crate) struct View<'a, 'w> {
46    pub(crate) bars: Vec<(Bar, bool)>,
47    /// The macros a slot may hold: what each looks like, and their names.
48    pub(crate) macros: hydra_ui::macros::MacroFile,
49    /// The character, by name; none while no character is in Hydra.
50    pub(crate) character: Option<&'a str>,
51    pub(crate) world: Option<&'a World<'w>>,
52    /// The icons folder: its sheets and images.
53    pub(crate) folder: Folder,
54}

56/// The hotbar editor.
57#[derive(Debug, Default)]
58pub(crate) struct Editor {
59    /// Whether it is showing.
60    pub(crate) open: bool,
61    /// The bar in the form as it is kept: its name, and whether it is the
62    /// character's own; none for a new one.
63    chosen: Option<(String, bool)>,
64    /// The bar as the form has it.
65    draft: Bar,
66    /// The bar as it was when chosen or last kept: the form's changes are
67    /// what differs.
68    kept: Bar,
69    /// Where the form's bar is kept: the character's own file.
70    own: bool,
71    /// A new bar is in the form: it stays there with its name cleared,
72    /// until it is saved or another is chosen.
73    begun: bool,
74    /// The slot picked on the grid, to fill.
75    slot: Option<u16>,
76    /// The effects' names the character has had while the editor was open.
77    names: BTreeSet<String>,
78    /// What the editor has to say: why a save did nothing, or that it did.
79    notice: Option<String>,
80    /// A sheet or an image to copy into the icons folder.
81    adding: Adding,
82    /// The bars picked in the list, as name and whether own, to take away.
83    picks: crate::picks::Picks<(String, bool)>,
84}
85
86impl Editor {
87    /// What is said under the form.
88    pub(crate) fn said(&mut self, notice: String) {
89        self.notice = Some(notice);
90    }
91
92    /// The form's bar was kept, where the form said.
93    pub(crate) fn kept(&mut self) {
94        self.chosen = Some((self.draft.name.clone(), self.own));
95        self.kept = self.draft.clone();
96    }
97
98    /// The bar in the form was taken away: the form starts a new one.
99    pub(crate) fn removed(&mut self) {
100        *self = Self {
101            open: self.open,
102            names: std::mem::take(&mut self.names),
103            notice: self.notice.take(),
104            ..Self::default()
105        };
106    }
107
108    /// The bar `bar` in the form, kept in the character's own file when
109    /// `own`.
110    fn choose(&mut self, bar: &Bar, own: bool) {
111        self.chosen = Some((bar.name.clone(), own));
112        self.picks.only(&(bar.name.clone(), own));
113        self.draft = bar.clone();
114        self.kept = bar.clone();
115        self.own = own;
116        self.slot = None;
117        self.notice = None;
118    }
119
120    /// Whether the form holds changes not kept.
121    fn changed(&self) -> bool {
122        self.draft != self.kept
123            || self
124                .chosen
125                .as_ref()
126                .is_some_and(|(_, own)| *own != self.own)
127    }

129    /// Draw the editor into `ui` over `view`. What it asks of the app.
130    pub(crate) fn show(&mut self, ui: &mut egui::Ui, view: &View<'_, '_>) -> Vec<Asked> {
131        if let Some(world) = view.world {
132            self.names.extend(
133                world
134                    .state
135                    .effects
136                    .iter()
137                    .map(|(_, effect)| effect.text.trim().to_owned())
138                    .filter(|name| !name.is_empty()),
139            );
140        }
141        let mut asked = Vec::new();
142        egui::Panel::left(egui::Id::new("hotbars-list"))
143            .resizable(true)
144            .default_size(200.0)
145            .show(ui, |ui| asked.extend(self.list(ui, view)));
146        egui::CentralPanel::default().show(ui, |ui| {
147            egui::ScrollArea::vertical()
148                .id_salt("hotbar-form")
149                .show(ui, |ui| asked.extend(self.form(ui, view)));
150        });
151        asked
152    }
153
154    /// The bars, each chosen by a click, several picked to take away; a
155    /// new one begun.
156    fn list(&mut self, ui: &mut egui::Ui, view: &View<'_, '_>) -> Vec<Asked> {
157        ui.heading("Hotbars");
158        let order: Vec<(String, bool)> = view
159            .bars
160            .iter()
161            .map(|(bar, own)| (bar.name.clone(), *own))
162            .collect();
163        self.picks.keep(&order);
164        let asked: Vec<Asked> = self
165            .picks
166            .remove_bar(ui, "bars")
167            .into_iter()
168            .map(|(name, own)| Asked::Remove { name, own })
169            .collect();
170        if view.bars.is_empty() {
171            ui.weak("None yet.");
172        }
173        for (bar, own) in &view.bars {
174            let key = (bar.name.clone(), *own);
175            let chosen = self.chosen.as_ref() == Some(&key);
176            let badge = if *own { "own" } else { "every" };
177            let response = ui
178                .selectable_label(
179                    chosen || self.picks.has(&key),
180                    format!("{}  [{badge}]", bar.name),
181                )
182                .on_hover_text(if *own {
183                    "This character's own: it replaces every character's bar of its name"
184                } else {
185                    "Every character's"
186                });
187            if response.clicked() && self.picks.click(ui, &key, &order) {
188                self.choose(bar, *own);
189            }
190        }
191        ui.separator();
192        if ui.button("+ New bar").clicked() {
193            let mut name = "bar".to_owned();
194            let mut n = 1;
195            while view.bars.iter().any(|(bar, _)| bar.name == name) {
196                n += 1;
197                name = format!("bar{n}");
198            }
199            *self = Self {
200                open: true,
201                draft: Bar {
202                    name,
203                    ..Bar::default()
204                },
205                begun: true,
206                names: std::mem::take(&mut self.names),
207                ..Self::default()
208            };
209            self.kept = Bar::default();
210        }
211        asked
212    }

214    /// The bar's form, its preview, its buttons, and saving.
215    fn form(&mut self, ui: &mut egui::Ui, view: &View<'_, '_>) -> Vec<Asked> {
216        let mut asked = Vec::new();
217        if self.chosen.is_none() && !self.begun {
218            ui.weak("Choose a bar on the left, or begin a new one.");
219            return asked;
220        }
221        self.bar_fields(ui, view);
222        crate::form::section(ui, "As it looks now");
223        let shown = crate::widget::hotbar::Shown {
224            bar: Some(self.draft.clone()),
225            macros: view.macros.clone(),
226            ..crate::widget::hotbar::Shown::default()
227        };
228        let state = view.world.map(|world| world.state);
229        let marks = view.world.map(|world| (world.flags, world.statuses));
230        let _ = crate::widget::hotbar::hotbar(
231            ui,
232            &shown,
233            (state, marks),
234            false,
235            egui::Id::new("hotbar-editor-preview"),
236        );
237        self.slots(ui, view);
238        ui.separator();
239        ui.horizontal(|ui| {
240            let changed = self.changed();
241            if ui
242                .add_enabled(changed, egui::Button::new("Save bar"))
243                .clicked()
244            {
245                asked.push(Asked::Save {
246                    bar: Box::new(self.draft.clone()),
247                    own: self.own,
248                    was: self.chosen.clone(),
249                });
250            }
251            if changed {
252                crate::form::hint(ui, "unsaved changes");
253            }
254            if let Some((name, own)) = &self.chosen
255                && ui.button("Remove bar").clicked()
256            {
257                asked.push(Asked::Remove {
258                    name: name.clone(),
259                    own: *own,
260                });
261            }
262        });
263        if let Some(notice) = &self.notice {
264            ui.label(notice);
265        }
266        ui.separator();
267        // The icons folder is shared with the indicators: kept on a page of
268        // its own.
269        ui.horizontal(|ui| {
270            ui.weak("Icon sheets and pictures are kept on the Icons page.");
271            if ui.button("Icons page").clicked() {
272                crate::editors_menu::ask_editor(ui.ctx(), crate::editors_menu::Editor::Icons);
273            }
274        });
275        asked
276    }

278    /// The Icons page: the icons folder `folder`, its sheets, a sheet or an
279    /// image copied in, and what the last copy did. What it asked.
280    pub(crate) fn icons(&mut self, ui: &mut egui::Ui, folder: &Folder) -> Vec<Asked> {
281        let mut asked = Vec::new();
282        if let Some(adding) = icon::folder_ui(ui, folder, &mut self.adding) {
283            asked.push(Asked::CopyIn(adding));
284        }
285        if let Some(notice) = &self.notice {
286            ui.label(notice);
287        }
288        asked
289    }

291    /// The file asked to be copied into the icons folder was: the form is
292    /// ready for another.
293    pub(crate) fn copied(&mut self) {
294        self.adding = Adding::default();
295    }

297    /// The bar's slots as a grid, each showing its macro: a click picks one
298    /// to fill or empty, a drag onto another moves it there.
299    fn slots(&mut self, ui: &mut egui::Ui, view: &View<'_, '_>) {
300        crate::form::section_with(
301            ui,
302            "Slots",
303            "click one to put a macro in it; drag one onto another to move it",
304        );
305        let grid = self.draft.grid();
306        let columns = usize::from(grid.columns.max(1));
307        let mut moved = None;
308        egui::Grid::new("hotbar-slots")
309            .spacing([4.0, 4.0])
310            .show(ui, |ui| {
311                for (at, place) in grid.slots.iter().enumerate() {
312                    let slot = u16::try_from(at).unwrap_or(u16::MAX);
313                    let text = place.map_or_else(
314                        || egui::RichText::new("empty").weak(),
315                        |place| {
316                            egui::RichText::new(slot_label(
317                                &self.draft.buttons[place],
318                                &view.macros,
319                            ))
320                        },
321                    );
322                    let picked = self.slot == Some(slot);
323                    // One button takes the click and, filled, the drag.
324                    let response = ui.add_sized(
325                        [96.0, 24.0],
326                        egui::Button::new(text)
327                            .selected(picked)
328                            .sense(egui::Sense::click_and_drag()),
329                    );
330                    if place.is_some() {
331                        response.dnd_set_drag_payload(slot);
332                    }
333                    if response.clicked() {
334                        self.slot = Some(slot);
335                    }
336                    if let Some(from) = response.dnd_release_payload::<u16>() {
337                        moved = Some((*from, slot));
338                    }
339                    if (at + 1) % columns == 0 {
340                        ui.end_row();
341                    }
342                }
343            });
344        if let Some((from, to)) = moved
345            && from != to
346        {
347            self.draft.move_button(from, to);
348            self.slot = Some(to);
349        }
350        let Some(slot) = self
351            .slot
352            .filter(|slot| usize::from(*slot) < grid.slots.len())
353        else {
354            crate::form::hint(ui, "Pick a slot to fill it.");
355            return;
356        };
357        let place = grid.slots[usize::from(slot)];
358        let now = place
359            .and_then(|place| hydra_ui::macros::macro_of(&self.draft.buttons[place]))
360            .map(str::to_owned);
361        let mut chosen = now.clone();
362        ui.horizontal(|ui| {
363            ui.label(format!("Slot {}", slot + 1));
364            egui::ComboBox::from_id_salt("hotbar-slot-macro")
365                .selected_text(chosen.as_deref().unwrap_or("empty"))
366                .show_ui(ui, |ui| {
367                    ui.selectable_value(&mut chosen, None, "empty");
368                    for made in &view.macros.macros {
369                        ui.selectable_value(&mut chosen, Some(made.name.clone()), &made.name);
370                    }
371                });
372            if ui
373                .button("Macros page")
374                .on_hover_text("Make or change a macro: what a slot shows and sends is its")
375                .clicked()
376            {
377                crate::editors_menu::ask_editor(ui.ctx(), crate::editors_menu::Editor::Macros);
378            }
379        });
380        if chosen != now {
381            fill(&mut self.draft, slot, chosen);
382        }
383    }

385    /// The bar's name, where it is kept, its grid and its buttons' size.
386    fn bar_fields(&mut self, ui: &mut egui::Ui, view: &View<'_, '_>) {
387        let grid = self.draft.grid();
388        egui::Grid::new("hotbar-bar-fields")
389            .num_columns(2)
390            .spacing([12.0, 6.0])
391            .show(ui, |ui| {
392                ui.label("Name");
393                ui.add(egui::TextEdit::singleline(&mut self.draft.name).desired_width(160.0))
394                    .on_hover_text("What a Hotbar widget's page and .hotbar call it");
395                ui.end_row();
396                ui.label("Kept for");
397                ui.horizontal(|ui| {
398                    ui.radio_value(&mut self.own, false, "every character");
399                    match view.character {
400                        Some(name) => {
401                            ui.radio_value(&mut self.own, true, format!("{name} only"));
402                        }
403                        None => {
404                            ui.add_enabled(false, egui::RadioButton::new(false, "one character"));
405                        }
406                    }
407                });
408                ui.end_row();
409                ui.label("Grid");
410                ui.horizontal(|ui| {
411                    let mut rows = self.draft.rows.unwrap_or(grid.rows);
412                    let mut columns = self.draft.columns.unwrap_or(grid.columns);
413                    ui.add(
414                        egui::DragValue::new(&mut rows)
415                            .range(1..=hydra_ui::hotbar::MOST_ROWS)
416                            .suffix(" rows"),
417                    );
418                    ui.add(
419                        egui::DragValue::new(&mut columns)
420                            .range(1..=hydra_ui::hotbar::MOST_COLUMNS)
421                            .suffix(" columns"),
422                    );
423                    if Some(rows) != self.draft.rows.or(Some(grid.rows)) {
424                        self.draft.rows = Some(rows);
425                    }
426                    if Some(columns) != self.draft.columns.or(Some(grid.columns)) {
427                        self.draft.columns = Some(columns);
428                    }
429                    crate::form::hint(ui, "grows rows when its buttons need more slots");
430                });
431                ui.end_row();
432                ui.label("Each button");
433                ui.horizontal(|ui| {
434                    let size =
435                        |ui: &mut egui::Ui, value: &mut Option<u16>, default: u16, suffix: &str| {
436                            let mut now = value.unwrap_or(default);
437                            ui.add(
438                                egui::DragValue::new(&mut now)
439                                    .range(12..=400)
440                                    .suffix(suffix),
441                            );
442                            if now != value.unwrap_or(default) {
443                                *value = Some(now);
444                            }
445                        };
446                    size(ui, &mut self.draft.button_width, 72, " wide");
447                    size(ui, &mut self.draft.button_height, 28, " high");
448                    crate::form::hint(ui, "points, whatever a button shows");
449                });
450                ui.end_row();
451                ui.label("Icons");
452                ui.horizontal(|ui| {
453                    let mut set = self.draft.icon_size.is_some();
454                    ui.checkbox(&mut set, "at most");
455                    if set {
456                        let mut edge = self.draft.icon_size.unwrap_or(24);
457                        ui.add(
458                            egui::DragValue::new(&mut edge)
459                                .range(8..=400)
460                                .suffix(" points"),
461                        );
462                        self.draft.icon_size = Some(edge);
463                    } else {
464                        self.draft.icon_size = None;
465                        crate::form::hint(ui, "as big as a button allows, scaled to fit");
466                    }
467                });
468                ui.end_row();
469            });
470    }

473/// What the grid shows for `button`: its macro's label, else its name.
474fn slot_label(button: &hydra_ui::hotbar::Button, macros: &hydra_ui::macros::MacroFile) -> String {
475    match hydra_ui::macros::macro_of(button) {
476        Some(name) => macros
477            .named(name)
478            .and_then(|made| made.label.clone())
479            .unwrap_or_else(|| name.to_owned()),
480        None => button.label.clone(),
481    }
482}

484/// `bar`'s slot `slot` holding the macro `chosen`, or emptied for none;
485/// every other button kept in the slot it is in.
486fn fill(bar: &mut Bar, slot: u16, chosen: Option<String>) {
487    let grid = bar.grid();
488    for (at, place) in grid.slots.iter().enumerate() {
489        if let Some(place) = place {
490            bar.buttons[*place].slot = u16::try_from(at).ok();
491        }
492    }
493    let place = grid.slots.get(usize::from(slot)).copied().flatten();
494    match (place, chosen) {
495        (Some(place), None) => {
496            bar.buttons.remove(place);
497        }
498        (Some(place), Some(name)) => {
499            let button = &mut bar.buttons[place];
500            button.id.clone_from(&name);
501            button.macro_name = Some(name);
502        }
503        (None, Some(name)) => bar.buttons.push(hydra_ui::hotbar::Button {
504            id: name.clone(),
505            slot: Some(slot),
506            macro_name: Some(name),
507            ..hydra_ui::hotbar::Button::default()
508        }),
509        (None, None) => {}
510    }

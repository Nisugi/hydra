1//! A progress bar: it fills across or upright (or as an orb, a ring, or a
2//! panel under its stat's emblem), its
3//! height and width adjustable, with an overlay laid over it, and text
4//! inside it, outside it on any of the four sides, or none -- the text
5//! numbers, a percent, a label, or any mix of them.
6//!
7//! An overlay may be a skin's frame painted over the bar, a nine-slice
8//! ([`Overlay::framed`]). Unlike egui's own bar, an empty bar paints no

266/// Where a bar's words are drawn again, in the outline's colour, for an
267/// outline `width` points wide: none for 0; a point each way for 1; and for
268/// a wider one, each way and each corner at every point out to its width.
269fn outline_offsets(width: u8) -> Vec<Vec2> {
270    let ways = |out: f32| {
271        [
272            Vec2::new(-out, 0.0),
273            Vec2::new(out, 0.0),
274            Vec2::new(0.0, -out),
275            Vec2::new(0.0, out),
276        ]
277    };
278    match width {
279        0 => Vec::new(),
280        1 => ways(1.0).to_vec(),
281        _ => (1..=width)
282            .map(f32::from)
283            .flat_map(|out| {
284                ways(out).into_iter().chain([
285                    Vec2::new(-out, -out),
286                    Vec2::new(out, -out),
287                    Vec2::new(-out, out),
288                    Vec2::new(out, out),
289                ])
290            })
291            .collect(),
292    }
293}

339/// The gap between a bar and text outside it.
340const GAP: f32 = 4.0;
341
342/// `bar` no thicker than `thickness` across the way it fills, centred where
343/// it was; a `thickness` of 0 leaves it as it is, and so does an orb's or a
344/// ring's, which are as thick as they are wide.
345fn thinned(bar: egui::Rect, fills: Fills, thickness: f32) -> egui::Rect {
346    if thickness <= 0.0 {
347        return bar;
348    }
349    let size = match fills {
350        Fills::Right | Fills::Left => Vec2::new(bar.width(), bar.height().min(thickness)),
351        Fills::Up | Fills::Down => Vec2::new(bar.width().min(thickness), bar.height()),
352        Fills::Orb | Fills::Ring | Fills::Panel => return bar,
353    };
354    egui::Rect::from_center_size(bar.center(), size)
355}

432    /// Drawn as `look` says: which way it fills, where its text goes, what
433    /// the text says, its colour where the player chose one, and a ring's
434    /// thickness. Its images are the caller's to load.
435    pub fn look(self, look: &Look) -> Self {
436        let mut drawn = self.fills(look.fills).text(look.place).says(look.says);
437        // The highest threshold reached, else its own colour.
438        let reached = drawn.value.and_then(|value| {
439            look.thresholds
440                .iter()
441                .filter(|threshold| u32::from(threshold.from) <= value.percent)
442                .max_by_key(|threshold| threshold.from)
443        });
444        if let Some(color) = reached.map(|threshold| threshold.color).or(look.color) {
445            drawn = drawn.fill(crate::theme::rgb(color));
446        }
447        drawn.ring = look.ring;
448        drawn.track = look.track;
449        drawn.text_color = look.text_color.map(crate::theme::rgb);
450        drawn.outline_color = look.outline_color.map(crate::theme::rgb);
451        drawn.outline = look.outline.min(Look::OUTLINE_MOST);
452        drawn
453    }

474    /// Its shape painted in `bar_rect`, its trough and fill the colours
475    /// given, and what lies over it: as a vessel, when it is an orb that is
476    /// one; else its track, fill, overlay and kit. Whether the fill covers
477    /// the middle.
478    fn painted(
479        &self,
480        painter: &egui::Painter,
481        bar_rect: egui::Rect,
482        (trough, fill): (Color32, Color32),
483    ) -> bool {
484        let share = self
485            .value
486            .map_or(0.0, |amount| shape::share(amount.percent));
487        if let (Some(vessel), Fills::Orb) = (&self.vessel, self.fills) {
488            let frame = shape::frame(self.fills, bar_rect);
489            return vessel::paint(painter, frame, share, vessel, fill);
490        }
491        let paint = shape::Paint {
492            trough,
493            fill,
494            background: self.glassed(painter.ctx()),
495            track: self.track,
496            kit: self.kit,
497            fill_image: self.fill_image,
498            ring: f32::from(self.ring) / 100.0,
499        };
500        let covered = shape::paint(painter, self.fills, bar_rect, share, &paint);
501        self.dressed(painter, bar_rect, paint.ring);
502        covered
503    }

505    /// Its overlay and its kit over the fill in `bar_rect`, those it has.
506    fn dressed(&self, painter: &egui::Painter, bar_rect: egui::Rect, ring: f32) {
507        let frame = shape::frame(self.fills, bar_rect);
508        if let Some(overlay) = self.overlay {
509            overlay.paint(painter, frame);
510        }
511        if let Some(kit) = &self.kit {
512            shape::kit(painter, self.fills, frame, kit, ring);
513        }
514        if let (Some(emblem), Fills::Panel) = (self.emblem, self.fills) {
515            let side = frame.width().min(frame.height());
516            let square = Rect::from_center_size(frame.center(), Vec2::splat(side));
517            let whole = Rect::from_min_max(egui::pos2(0.0, 0.0), egui::pos2(1.0, 1.0));
518            painter.image(emblem, square, whole, Color32::WHITE);
519        }
520    }

537    /// Sized to what `ui` has left, its text outside it counted: as wide and
538    /// as tall as the room, across or upright, so the space it is given is its
539    /// shape: a bar filling across and made taller grows taller.
540    pub fn fitted(mut self, ui: &egui::Ui) -> Self {
541        let room = ui.available_size();
542        // Beside it, all its words; above a panel, those that name it.
543        let words = if self.place == Place::Corner {
544            self.parts().0
545        } else {
546            self.words()
547        };
548        let text = if self.place == Place::Inside || self.place == Place::Hidden || words.is_empty()
549        {
550            Vec2::ZERO
551        } else {
552            ui.painter()
553                .layout_no_wrap(
554                    words,
555                    egui::TextStyle::Body.resolve(ui.style()),
556                    Color32::PLACEHOLDER,
557                )
558                .size()
559                + Vec2::splat(GAP)
560        };
561        let left = match self.place {
562            Place::Above | Place::Below | Place::Corner => Vec2::new(room.x, room.y - text.y),
563            Place::Left | Place::Right => Vec2::new(room.x - text.x, room.y),
564            Place::Inside | Place::Hidden => room,
565        };
566        self.size = Some(Vec2::new(left.x.max(LEAST), left.y.max(LEAST)));
567        self
568    }

597    /// The words, in order, joined by spaces; empty for none.
598    #[must_use]
599    pub fn words(&self) -> String {
600        let said = self.said();
601        match self.end.filter(|end| !end.is_empty()) {
602            Some(end) if said.is_empty() => end.to_owned(),
603            Some(end) => format!("{said} {end}"),
604            None => said,
605        }
606    }

608    /// Its words but its [`Self::end`]: what names it and what measures it.
609    fn said(&self) -> String {
610        let (named, measured) = self.parts();
611        [named, measured]
612            .into_iter()
613            .filter(|part| !part.is_empty())
614            .collect::<Vec<_>>()
615            .join(" ")
616    }

618    /// The words in two: what names it, its label and the game's word for
619    /// its state; and what measures it, its numbers and its percent.
620    fn parts(&self) -> (String, String) {
621        let mut parts = Vec::new();
622        let state = self
623            .state
624            .filter(|words| self.says.words && !words.is_empty());
625        if self.says.label && !self.label.is_empty() {
626            // `Stance: offensive`, and `Stance 80%` with no word after it.
627            parts.push(match state {
628                Some(_) => format!("{}:", self.label),
629                None => self.label.to_owned(),
630            });
631        }
632        if let Some(state) = state {
633            parts.push(state.to_owned());
634        }
635        let named = parts.join(" ");
636        let mut parts = Vec::new();
637        match self.value {
638            Some(amount) => {
639                if self.says.numbers
640                    && let (Some(current), Some(max)) = (amount.current, amount.max)
641                {
642                    parts.push(format!("{current}/{max}"));
643                }
644                if self.says.percent {
645                    parts.push(format!("{}%", amount.percent));
646                }
647            }
648            None if self.says.numbers || self.says.percent => parts.push("?".to_owned()),
649            None => {}
650        }
651        (named, parts.join(" "))
652    }
653
654    fn bar_size(&self) -> Vec2 {
655        self.size.unwrap_or(match self.fills {
656            Fills::Right | Fills::Left => Vec2::new(72.0, 18.0),
657            Fills::Up | Fills::Down => Vec2::new(18.0, 72.0),
658            Fills::Orb | Fills::Ring => Vec2::splat(48.0),
659            Fills::Panel => Vec2::new(56.0, 72.0),
660        })
661    }
662}

664impl egui::Widget for Bar<'_> {
665    fn ui(self, ui: &mut egui::Ui) -> egui::Response {
666        let words = self.words();
667        let (beside, corner) = self.texts(ui);
668        let bar = self.bar_size();
669        let text = beside.as_ref().map_or(Vec2::ZERO, |galley| galley.size());
670        let outside = beside.is_some() && self.place != Place::Inside;
671        let total = match self.place {
672            Place::Above | Place::Below | Place::Corner if outside => {
673                Vec2::new(bar.x.max(text.x), bar.y + GAP + text.y)
674            }
675            Place::Left | Place::Right if outside => {
676                Vec2::new(text.x + GAP + bar.x, bar.y.max(text.y))
677            }
678            _ => bar,
679        };
680        let (rect, response) = ui.allocate_exact_size(total, Sense::hover());
681        let bar_rect = match self.place {
682            Place::Above | Place::Corner if outside => {
683                Align2::CENTER_BOTTOM.align_size_within_rect(bar, rect)
684            }
685            Place::Below if outside => Align2::CENTER_TOP.align_size_within_rect(bar, rect),
686            Place::Left if outside => Align2::RIGHT_CENTER.align_size_within_rect(bar, rect),
687            Place::Right if outside => Align2::LEFT_CENTER.align_size_within_rect(bar, rect),
688            _ => rect,
689        };
690        let bar_rect = thinned(
691            bar_rect,
692            self.fills,
693            crate::theme::shape(ui.ctx()).bar_thickness,
694        );
695        let enabled = ui.is_enabled();
696        response.widget_info(|| {
697            WidgetInfo::labeled(WidgetType::ProgressIndicator, enabled, words.clone())
698        });
699        if !ui.is_rect_visible(rect) {
700            return response;
701        }
702        let visuals = ui.visuals();
703        let (trough, ground) = (visuals.extreme_bg_color, visuals.text_color());
704        let fill = self.fill.unwrap_or(visuals.selection.bg_fill);
705        let painter = ui.painter();
706        let covered = self.painted(painter, bar_rect, (trough, fill));
707        if let Some(galley) = beside {
708            if outside {
709                let anchor = match self.place {
710                    Place::Above | Place::Corner => Align2::CENTER_TOP,
711                    Place::Below => Align2::CENTER_BOTTOM,
712                    Place::Left => Align2::LEFT_CENTER,
713                    _ => Align2::RIGHT_CENTER,
714                };
715                let at = anchor.align_size_within_rect(galley.size(), rect).min;
716                self.write(painter, galley, at, (None, ground));
717            } else {
718                let at = Align2::CENTER_CENTER
719                    .align_size_within_rect(galley.size(), bar_rect)
720                    .min;
721                let behind = if covered { fill } else { trough };
722                self.write(painter, galley, at, (Some(behind), ground));
723            }
724        }
725        if let Some(galley) = corner {
726            let inner = bar_rect.shrink(CORNER_INSET);
727            let at = Align2::RIGHT_BOTTOM
728                .align_size_within_rect(galley.size(), inner)
729                .min;
730            let behind = self.behind(bar_rect, inner.right_bottom(), (trough, fill));
731            self.write(painter, galley, at, (Some(behind), ground));
732        }
733        if let Some(end) = self.end.filter(|_| self.place != Place::Hidden) {
734            self.ends(ui, bar_rect, end, (trough, fill, ground));
735        }
736        response
737    }
738}

740/// How far in from a bar's edge its corner words sit.
741const CORNER_INSET: f32 = 3.0;
742
743impl Bar<'_> {
744    /// Its words laid out: those beside it or over it, as its place says;
745    /// and, for [`Place::Corner`], its measures apart, for its corner.
746    fn texts(&self, ui: &egui::Ui) -> (Option<Arc<Galley>>, Option<Arc<Galley>>) {
747        // Words at its ends are laid out by `ends`, over it.
748        if self.end.is_some() {
749            return (None, None);
750        }
751        let lay = |text: String| {
752            (!text.is_empty()).then(|| {
753                ui.painter().layout_no_wrap(
754                    text,
755                    egui::TextStyle::Body.resolve(ui.style()),
756                    Color32::PLACEHOLDER,
757                )
758            })
759        };
760        match self.place {
761            Place::Hidden => (None, None),
762            Place::Corner => {
763                let (named, measured) = self.parts();
764                (lay(named), lay(measured))
765            }
766            _ => (lay(self.words()), None),
767        }
768    }
769
770    /// Its words over its left end, cut short with an ellipsis before they
771    /// reach `end`, and `end` over its right in figures of one width; each
772    /// readable on what lies behind it, the bar's `trough` or `fill`.
773    fn ends(
774        &self,
775        ui: &egui::Ui,
776        bar_rect: Rect,
777        end: &str,
778        (trough, fill, ground): (Color32, Color32, Color32),
779    ) {
780        let painter = ui.painter();
781        let body = egui::TextStyle::Body.resolve(ui.style());
782        let figures = egui::FontId::monospace(body.size);
783        let inner = bar_rect.shrink2(Vec2::new(CORNER_INSET, 0.0));
784        let end = painter.layout_no_wrap(end.to_owned(), figures, Color32::PLACEHOLDER);
785        let end_at = Align2::RIGHT_CENTER.align_size_within_rect(end.size(), inner);
786        let mut job =
787            egui::text::LayoutJob::simple_singleline(self.said(), body, Color32::PLACEHOLDER);
788        job.wrap = egui::text::TextWrapping::truncate_at_width(
789            (end_at.min.x - GAP - inner.min.x).max(0.0),
790        );
791        let named = painter.layout_job(job);
792        let named_at = Align2::LEFT_CENTER.align_size_within_rect(named.size(), inner);
793        let behind = |at: Rect| self.behind(bar_rect, at.center(), (trough, fill));
794        let over = behind(named_at);
795        self.write(painter, named, named_at.min, (Some(over), ground));
796        let over = behind(end_at);
797        self.write(painter, end, end_at.min, (Some(over), ground));
798    }
799
800    /// What lies behind the point `at` of the bar drawn in `bar_rect`: its
801    /// fill where the fill reaches, else its trough.
802    fn behind(
803        &self,
804        bar_rect: Rect,
805        at: egui::Pos2,
806        (trough, fill): (Color32, Color32),
807    ) -> Color32 {
808        let share = self
809            .value
810            .map_or(0.0, |amount| shape::share(amount.percent));
811        let frame = shape::frame(self.fills, bar_rect);
812        if share > 0.0 && shape::filled(frame, self.fills, share).contains(at) {
813            fill
814        } else {
815            trough
816        }
817    }
818
819    /// `galley` painted at `at`: beside the bar (`over` none) in its words'
820    /// colour or the window's `ground` text colour, outlined only when an
821    /// outline's colour was chosen; or over the bar, readable on `over`,
822    /// outlined in the other colour so it reads where the fill's edge runs
823    /// through it.
824    fn write(
825        &self,
826        painter: &egui::Painter,
827        galley: Arc<Galley>,
828        at: egui::Pos2,
829        (over, ground): (Option<Color32>, Color32),
830    ) {
831        let (color, outline) = match over {
832            None => (self.text_color.unwrap_or(ground), self.outline_color),
833            Some(behind) => {
834                let color = self.text_color.unwrap_or_else(|| readable_on(behind));
835                let outline = self.outline_color.unwrap_or_else(|| readable_on(color));
836                (color, Some(outline))
837            }
838        };
839        if let Some(outline) = outline {
840            for offset in outline_offsets(self.outline) {
841                painter.galley(at + offset, galley.clone(), outline);
842            }
843        }
844        painter.galley(at, galley, color);
845    }
846}

1//! The reading view: a line's class, the presets, and *Dedup*. None of it
2//! changes what is read or written.
3
4use std::collections::BTreeMap;
5
6use hydra_session::player_log::reader::Entry;

8/// The class of a tag: its last part. `main/combat` is `combat`.
9#[must_use]
10pub(crate) fn class(tag: &str) -> &str {
11    tag.rsplit('/').next().unwrap_or(tag)
12}
13
14/// A preset: which classes it ticks.
15#[derive(Clone, Copy, Debug, PartialEq, Eq)]
16pub(crate) enum Preset {
17    /// Every class.
18    Everything,
19    /// Only what our definitions called combat.
20    Combat,
21    /// What people said.
22    Social,
23    /// Everything but combat and the room's comings and goings.
24    Quiet,
25}
26
27impl Preset {
28    /// Every preset, in the order the window offers them.
29    pub(crate) const ALL: [Preset; 4] = [
30        Preset::Everything,
31        Preset::Combat,
32        Preset::Social,
33        Preset::Quiet,
34    ];
35
36    /// What the button says.
37    pub(crate) const fn label(self) -> &'static str {
38        match self {
39            Preset::Everything => "Everything",
40            Preset::Combat => "Combat",
41            Preset::Social => "Social",
42            Preset::Quiet => "Quiet",
43        }
44    }
45
46    /// Whether this preset ticks `class`.
47    pub(crate) fn ticks(self, class: &str) -> bool {
48        /// Where people talk: the game's streams for it.
49        const SOCIAL: &[&str] = &["thoughts", "speech", "whispers", "talk", "ooc"];
50        /// The noise a quiet read leaves out, beside combat.
51        const NOISE: &[&str] = &[
52            "combat",
53            "atmospherics",
54            "arrivals",
55            "logons",
56            "death",
57            "deaths",
58        ];
59        match self {
60            Preset::Everything => true,
61            Preset::Combat => class == "combat",
62            Preset::Social => SOCIAL.contains(&class),
63            Preset::Quiet => !NOISE.contains(&class),
64        }
65    }
66}

68/// One line as the window shows it: the line, and how many times it came in
69/// a row when *Dedup* folded it.
70pub(crate) type Shown<'a> = (&'a Entry, usize);
71
72/// The lines of `entries` whose class `ticked` ticks (a class it has not
73/// heard of is shown), folded when `dedup` says.
74/// As the places of those lines in `entries`, and their counts.
75fn rows(entries: &[Entry], ticked: &BTreeMap<String, bool>, dedup: bool) -> Vec<(usize, usize)> {
76    let mut rows: Vec<(usize, usize)> = Vec::new();
77    for (at, entry) in entries.iter().enumerate() {
78        if !ticked.get(class(&entry.stream)).copied().unwrap_or(true) {
79            continue;
80        }
81        if dedup
82            && let Some((last, count)) = rows.last_mut()
83            && entries[*last].stream == entry.stream
84            && entries[*last].text == entry.text
85        {
86            *count += 1;
87            continue;
88        }
89        rows.push((at, 1));
90    }
91    rows
92}

94/// What [`rows`] keeps of some lines, kept from frame to frame until the
95/// lines or the filters change: a whole day is thousands of lines, and
96/// sifting them every frame is most of the window's work.
97#[derive(Debug, Default)]
98pub(crate) struct Filtered {
99    /// The lines' generation, the classes ticked and *Dedup*, as sifted.
100    key: Option<(u64, BTreeMap<String, bool>, bool)>,
101    rows: Vec<(usize, usize)>,
102}
103
104impl Filtered {
105    /// The lines of `entries` shown, the `generation`th set of them: sifted again
106    /// only when that or the filters changed. A view over the rows kept, so
107    /// a frame that draws a few of a day's lines copies none of the rest.
108    pub(crate) fn of<'a>(
109        &'a mut self,
110        entries: &'a [Entry],
111        generation: u64,
112        ticked: &BTreeMap<String, bool>,
113        dedup: bool,
114    ) -> Rows<'a> {
115        let fresh = self
116            .key
117            .as_ref()
118            .is_some_and(|(at, was, then)| *at == generation && was == ticked && *then == dedup);
119        if !fresh {
120            self.rows = rows(entries, ticked, dedup);
121            self.key = Some((generation, ticked.clone(), dedup));
122        }
123        Rows {
124            entries,
125            rows: &self.rows,
126        }
127    }
128}
129
130/// The lines shown, as [`Filtered::of`] kept them: each an entry and the
131/// times it came in a row.
132#[derive(Clone, Copy, Debug)]
133pub(crate) struct Rows<'a> {
134    entries: &'a [Entry],
135    rows: &'a [(usize, usize)],
136}
137
138impl<'a> Rows<'a> {
139    /// How many lines are shown.
140    pub(crate) const fn len(&self) -> usize {
141        self.rows.len()
142    }
143
144    /// The `index`th line shown.
145    pub(crate) fn get(&self, index: usize) -> Option<Shown<'a>> {
146        let &(at, count) = self.rows.get(index)?;
147        Some((self.entries.get(at)?, count))
148    }
149}

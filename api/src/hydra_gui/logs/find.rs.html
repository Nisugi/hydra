1//! Find in *Recent*: Ctrl+F opens a bar over the day's lines, as the play
2//! window's does, and looks through the lines the filters show as it is
3//! typed, whatever the case. Each line holding it is marked, the current one
4//! more, and what it found in the line too. The count is from the newest,
5//! so the first found is the latest: F3, and Enter in the bar, go back to
6//! the one before, Shift+F3 and Shift+Enter forward, the list brought to
7//! it. A line the filters leave out is never gone to; the bar says how many
8//! of those hold it. Escape in the bar, or its ×, closes it.
9//!
10//! On a line *Show in day* went to, what is typed is found from there: the
11//! current is the latest at or before it.
12

20/// How wide the bar's field is.
21const FIELD: f32 = 220.0;

23/// The Find bar, while it is open.
24#[derive(Debug, Default)]
25pub(crate) struct Finding {
26    /// What is looked for, as typed.
27    pub(crate) query: String,
28    /// Which line found is current, from the newest: 0 is the latest.
29    pub(crate) current: usize,
30    /// The bar's field is to take the keyboard.
31    pub(crate) focus: bool,
32    /// The list is to be brought to the current line.
33    go: bool,
34    /// What is looked for changed: the current is found again, from the
35    /// line marked when there is one.
36    anchored: bool,
37}
38
39impl Finding {
40    /// A bar just opened, its field taking the keyboard.
41    pub(crate) fn opened() -> Self {
42        Self {
43            focus: true,
44            ..Self::default()
45        }
46    }
47}

49/// What Find marks in the lines drawn this frame.
50#[derive(Clone, Debug)]
51pub(crate) struct Marks {
52    /// What is looked for, in lower case.
53    pub(crate) query: String,
54    /// The row of the current line found.
55    pub(crate) current: Option<usize>,
56    /// The list is to be brought to it.
57    pub(crate) go: bool,
58}
59
60/// The rows shown that hold what is looked for, and how many lines the
61/// filters leave out do: found again only when the query, the lines or the
62/// view change, a day being thousands of lines.
63#[derive(Debug, Default)]
64pub(crate) struct Matches {
65    /// What they were found for: the query, the lines' generation, the view.
66    key: Option<(String, u64, View)>,
67    rows: Vec<usize>,
68    hidden: usize,
69}
70
71impl Matches {
72    /// The rows of `shown`, among `entries`, the `generation`th lines, that
73    /// hold `query`, in lower case, and how many lines `view` leaves out do.
74    fn of(
75        &mut self,
76        query: &str,
77        (entries, shown): (&[Entry], Rows<'_>),
78        generation: u64,
79        view: &View,
80    ) -> (&[usize], usize) {
81        let fresh = self
82            .key
83            .as_ref()
84            .is_some_and(|(was, at, then)| was == query && *at == generation && then == view);
85        if !fresh {
86            self.rows.clear();
87            self.hidden = 0;
88            if !query.is_empty() {
89                self.rows = (0..shown.len())
90                    .filter(|row| {
91                        shown
92                            .get(*row)
93                            .is_some_and(|(entry, _)| holds(query, &entry.text))
94                    })
95                    .collect();
96                self.hidden = entries
97                    .iter()
98                    .filter(|entry| !view.shows(entry) && holds(query, &entry.text))
99                    .count();
100            }
101            self.key = Some((query.to_owned(), generation, view.clone()));
102        }
103        (&self.rows, self.hidden)
104    }
105}

107/// The bar, above the lines, while `finding` is open: what is looked for,
108/// which found is current and how many, how many more the filters leave
109/// out, the steps and ×. What it marks in the lines `shown` this frame,
110/// among `entries`, the `generation`th; `None` when it marks nothing.
111/// `anchor` is the row of the line marked, which a new query is found from.
112pub(crate) fn bar(
113    ui: &mut egui::Ui,
114    (finding, matches): (&mut Option<Finding>, &mut Matches),
115    (entries, shown, generation): (&[Entry], Rows<'_>, u64),
116    view: &View,
117    anchor: Option<usize>,
118) -> Option<Marks> {
119    let bar = finding.as_mut()?;
120    let query = bar.query.trim().to_lowercase();
121    let (rows, hidden) = matches.of(&query, (entries, shown), generation, view);
122    let found = rows.len();
123    // Counted from the newest: as many as are newer than the line marked.
124    if std::mem::take(&mut bar.anchored) {
125        bar.current = anchor.map_or(0, |anchor| rows.iter().filter(|row| **row > anchor).count());
126        bar.go = true;
127    }
128    // F3 and Shift+F3, wherever the keyboard is; Shift first, as F3 alone
129    // would take it too.
130    let mut stepped = ui.input_mut(|input| {
131        if input.consume_key(Modifiers::SHIFT, Key::F3) {
132            Some(false)
133        } else if input.consume_key(Modifiers::NONE, Key::F3) {
134            Some(true)
135        } else {
136            None
137        }
138    });
139    let mut closed = false;
140    ui.horizontal(|ui| {
141        let label = ui.label("Find");
142        let field = ui
143            .add(egui::TextEdit::singleline(&mut bar.query).desired_width(FIELD))
144            .labelled_by(label.id);
145        if std::mem::take(&mut bar.focus) {
146            field.request_focus();
147        }
148        if field.changed() {
149            bar.current = 0;
150            bar.anchored = true;
151            bar.go = true;
152        }
153        let (enter, escape, shift) = ui.input(|input| {
154            (
155                input.key_pressed(Key::Enter),
156                input.key_pressed(Key::Escape),
157                input.modifiers.shift,
158            )
159        });
160        if field.lost_focus() && enter {
161            stepped = Some(!shift);
162            field.request_focus();
163        }
164        closed |= field.lost_focus() && escape;
165        if !query.is_empty() {
166            if found == 0 {
167                ui.weak("none");
168            } else {
169                ui.weak(format!("{} of {found}", bar.current.min(found - 1) + 1));
170            }
171        }
172        if hidden > 0 {
173            ui.weak(format!("{hidden} more not shown"))
174                .on_hover_text("Lines holding it in the classes or senders not ticked.");
175        }
176        if ui
177            .small_button("⬆")
178            .on_hover_text("Found before (F3)")
179            .clicked()
180        {
181            stepped = Some(true);
182        }
183        if ui
184            .small_button("⬇")
185            .on_hover_text("Found after (Shift+F3)")
186            .clicked()
187        {
188            stepped = Some(false);
189        }
190        closed |= ui
191            .small_button("×")
192            .on_hover_text("Close (Escape)")
193            .clicked();
194    });
195    if let Some(older) = stepped {
196        bar.current = if older {
197            bar.current + 1
198        } else {
199            bar.current.saturating_sub(1)
200        };
201        bar.go = true;
202    }
203    bar.current = bar.current.min(found.saturating_sub(1));
204    let go = std::mem::take(&mut bar.go);
205    let current = found
206        .checked_sub(1 + bar.current)
207        .and_then(|at| rows.get(at).copied());
208    if closed {
209        *finding = None;
210        return None;
211    }
212    (found > 0).then_some(Marks { query, current, go })
213}

215/// `runs`, a line's, with each place it holds `query` (in lower case) on
216/// `ground`: a run is split where a find starts or ends. A line whose lower
217/// case is not as long as it is, which no game line is, is left as it was:
218/// its finds could not be placed.
219pub(crate) fn marked(runs: Vec<StyledRun>, query: &str, ground: &str) -> Vec<StyledRun> {
220    let text: String = runs.iter().map(|run| run.text.as_str()).collect();
221    let lower = text.to_lowercase();
222    if query.is_empty() || lower.len() != text.len() {
223        return runs;
224    }
225    let found: Vec<(usize, usize)> = lower
226        .match_indices(query)
227        .map(|(at, _)| (at, at + query.len()))
228        .filter(|(from, to)| text.is_char_boundary(*from) && text.is_char_boundary(*to))
229        .collect();
230    if found.is_empty() {
231        return runs;
232    }
233    let mut out = Vec::with_capacity(runs.len() + 2 * found.len());
234    let mut start = 0;
235    for run in runs {
236        let end = start + run.text.len();
237        let mut cuts: Vec<usize> = found
238            .iter()
239            .flat_map(|(from, to)| [*from, *to])
240            .filter(|cut| *cut > start && *cut < end)
241            .collect();
242        cuts.sort_unstable();
243        cuts.dedup();
244        let mut from = start;
245        for to in cuts.into_iter().chain(std::iter::once(end)) {
246            let inside = found
247                .iter()
248                .any(|(a, b)| *a <= from && to <= *b && from < to);
249            out.push(StyledRun {
250                text: run.text[from - start..to - start].to_owned(),
251                background: if inside {
252                    Some(ground.to_owned())
253                } else {
254                    run.background.clone()
255                },
256                ..run.clone()
257            });
258            from = to;
259        }
260        start = end;
261    }
262    out

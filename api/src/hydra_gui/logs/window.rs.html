14/// The window's tabs.
15#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
16pub(crate) enum Tab {
17    /// A day, whole, newest at the bottom.
18    #[default]
19    Recent,
20    /// Lines holding a text, newest first.
21    Search,
22    /// Days written to a file.
23    Export,
24}

26/// One character's log window.
27#[derive(Debug)]
28pub(crate) struct Logs {
29    /// Whose log, as the table named the character.
30    pub(crate) character: String,
31    /// The key its window's place is kept by, as its character's seat gave
32    /// it on opening: by the time the window goes with its character, the
33    /// seat that knew its team is gone. Empty until it opens.
34    pub(crate) place: String,
35    /// Whether it is showing.
36    pub(crate) open: bool,
37    pub(super) tab: Tab,
38    pub(super) days: Days,
39    pub(super) usage: Option<Usage>,
40    /// The day *Recent* shows, once one is chosen.
41    pub(super) day: Option<String>,
42    pub(super) lines: Vec<Entry>,
43    pub(super) query: String,
44    pub(super) regex: bool,
45    pub(super) found: Option<Found>,
46    /// The search last asked: searches run at once, and only its answer is
47    /// shown.
48    pub(super) searched: Option<(String, bool)>,
49    /// Bumped whenever the day's lines or what was found change: what the
50    /// filtered rows are kept against.
51    pub(super) generation: u64,
52    /// The day's lines and what was found, as the filters leave them.
53    pub(super) day_rows: super::view::Filtered,
54    pub(super) found_rows: super::view::Filtered,
55    /// The classes the window has seen, each ticked or not.
56    pub(super) ticked: BTreeMap<String, bool>,
57    pub(super) dedup: bool,
58    pub(super) from: String,
59    pub(super) to: String,
60    /// What the window last had to say: an error, where an export went.
61    pub(super) said: Option<String>,
62    /// Lines the character's log could not keep up with and did not record.
63    pub(crate) dropped: u64,
64    /// Reads asked and not yet answered.
65    pub(super) waiting: usize,
66    /// Asks made before the window was first drawn: its days.
67    pub(super) asked: Vec<Ask>,
68    pub(super) inbox: Inbox,
69    /// The links clicked in its lines, for the character's play window to
70    /// act on.
71    pub(super) clicked: Vec<hydra_ui::RunLink>,
72}

74impl Logs {
75    /// A window for `character`, open, asking for its days.
76    pub(crate) fn new(character: &str) -> Self {
77        Self {
78            character: character.to_owned(),
79            place: String::new(),
80            open: true,
81            tab: Tab::Recent,
82            days: Vec::new(),
83            usage: None,
84            day: None,
85            lines: Vec::new(),
86            query: String::new(),
87            regex: false,
88            found: None,
89            searched: None,
90            generation: 0,
91            day_rows: super::view::Filtered::default(),
92            found_rows: super::view::Filtered::default(),
93            ticked: BTreeMap::new(),
94            dedup: false,
95            from: String::new(),
96            to: String::new(),
97            said: None,
98            dropped: 0,
99            waiting: 0,
100            asked: vec![Ask::Days],
101            inbox: Inbox::default(),
102            clicked: Vec::new(),
103        }
104    }
105
106    /// Where its reads answer.
107    pub(crate) fn inbox(&self) -> Inbox {
108        Arc::clone(&self.inbox)
109    }
110
111    /// What it wants read, taken: each is counted as waiting.
112    pub(crate) fn take_clicks(&mut self) -> Vec<hydra_ui::RunLink> {
113        std::mem::take(&mut self.clicked)
114    }
115
116    pub(crate) fn take_asks(&mut self) -> Vec<Ask> {
117        let asked = std::mem::take(&mut self.asked);
118        self.waiting += asked.len();
119        asked
120    }
121
122    /// Ask for a search for `query`, a regular expression when `regex`: the
123    /// one whose answer is shown, whatever answers before or after it.
124    pub(crate) fn ask_search(&mut self, query: &str, regex: bool) {
125        let search = (query.to_owned(), regex);
126        self.searched = Some(search.clone());
127        self.asked.push(Ask::Search(search.0, search.1));
128    }
129
130    /// Take what the reads answered.
131    pub(crate) fn take_replies(&mut self) {
132        let replies = std::mem::take(&mut *crate::logs::lock(&self.inbox));
133        for reply in replies {
134            self.waiting = self.waiting.saturating_sub(1);
135            self.answer(reply);
136        }
137    }
138
139    pub(super) fn answer(&mut self, reply: Reply) {
140        match reply {
141            Reply::Days(Ok((days, usage))) => {
142                // Opened on the newest day, and kept on the one chosen.
143                let newest = days.first().map(|(day, _)| day.clone());
144                if self.day.is_none()
145                    && let Some(day) = newest.clone()
146                {
147                    self.asked.push(Ask::Day(day.clone()));
148                    self.day = Some(day);
149                }
150                if self.from.is_empty() {
151                    self.from = newest.clone().unwrap_or_default();
152                    self.to = newest.unwrap_or_default();
153                }
154                self.days = days;
155                self.usage = Some(usage);
156            }
157            Reply::Day(day, Ok(lines)) => {
158                if self.day.as_deref() == Some(day.as_str()) {
159                    self.learn(&lines);
160                    self.lines = lines;
161                    self.generation += 1;
162                }
163            }
164            // A search asked before the last one, answering late.
165            Reply::Found(search, _) if self.searched.as_ref() != Some(&search) => {}
166            Reply::Found(_, Ok(found)) => {
167                self.learn(&found.entries);
168                self.found = Some(found);
169                self.generation += 1;
170            }
171            Reply::Exported(Ok(done)) => {
172                self.said = Some(format!(
173                    "{} lines from {} days written to {}",
174                    done.lines,
175                    done.days,
176                    done.path.display()
177                ));
178            }
179            Reply::Days(Err(why))
180            | Reply::Day(_, Err(why))
181            | Reply::Found(_, Err(why))
182            | Reply::Exported(Err(why)) => self.said = Some(why),
183        }
184    }
185
186    /// Closed: what it read is let go, a day of lines can be tens of
187    /// megabytes, and its days and day read again when it opens.
188    pub(crate) fn close(&mut self) {
189        self.open = false;
190        self.lines = Vec::new();
191        self.found = None;
192        self.searched = None;
193        self.day_rows = super::view::Filtered::default();
194        self.found_rows = super::view::Filtered::default();
195        self.generation += 1;
196        self.asked.push(Ask::Days);
197        if let Some(day) = self.day.clone() {
198            self.asked.push(Ask::Day(day));
199        }
200    }
201
202    /// Tick each class these lines hold that the window had not seen.
203    pub(super) fn learn(&mut self, lines: &[Entry]) {
204        for entry in lines {
205            self.ticked
206                .entry(class(&entry.stream).to_owned())
207                .or_insert(true);
208        }
209    }
210
211    /// Tick what `preset` ticks, and nothing else.
212    pub(crate) fn preset(&mut self, preset: Preset) {
213        for (class, on) in &mut self.ticked {
214            *on = preset.ticks(class);
215        }
216    }
217
218    /// The tags an export writes: the classes ticked, when any is not.
219    pub(super) fn export_streams(&self) -> Streams {
220        if self.ticked.values().all(|on| *on) {
221            return Streams::all();
222        }
223        Streams::classes(
224            self.ticked
225                .iter()
226                .filter(|(_, on)| **on)
227                .map(|(class, _)| class.clone()),
228        )
229    }

231    /// The folder this character's log is in.
232    pub(crate) fn folder(&self) -> PathBuf {
233        writer::dir(&writer::root(), &self.character)
234    }

237/// Open `folder` in the system's file browser. What it cannot is said.
238pub(crate) fn open_folder(folder: &Path) -> Result<(), String> {
239    let program = if cfg!(target_os = "windows") {
240        "explorer"
241    } else if cfg!(target_os = "macos") {
242        "open"
243    } else {
244        "xdg-open"
245    };
246    std::fs::create_dir_all(folder).map_err(|e| e.to_string())?;
247    std::process::Command::new(program)
248        .arg(folder)
249        .spawn()
250        .map(drop)
251        .map_err(|e| format!("{program} could not open {}: {e}", folder.display()))
252}

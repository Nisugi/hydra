14/// The window's tabs.

193    pub(crate) fn take_asks(&mut self) -> Vec<Ask> {
194        let asked = std::mem::take(&mut self.asked);
195        // Today read on is no wait to show: it is asked every second.
196        self.waiting += asked
197            .iter()
198            .filter(|ask| !matches!(ask, Ask::More(..)))
199            .count();
200        asked
201    }

203    /// Today's lines read on as they are written, a second apart, while
204    /// *Recent* shows `today`'s: when to look again, `now` being the
205    /// window's seconds.
206    pub(crate) fn follow_today(&mut self, now: f64, today: &str) -> Option<std::time::Duration> {
207        if self.tab != Tab::Recent || self.day.as_deref() != Some(today) || self.following {
208            return None;
209        }
210        let tail = self.tail.clone()?;
211        let since = now - self.polled;
212        if since < POLL {
213            return Some(std::time::Duration::from_secs_f64(POLL - since));
214        }
215        self.polled = now;
216        self.following = true;
217        self.asked.push(Ask::More(today.to_owned(), tail));
218        Some(std::time::Duration::from_secs_f64(POLL))
219    }

221    /// Ask for a search for `query`, a regular expression when `regex`, of
222    /// the days chosen and the classes and senders ticked: the one whose
223    /// answer is shown, whatever answers before or after it.
224    pub(crate) fn ask_search(&mut self, query: &str, regex: bool) {
225        let (mut first, mut last) = (
226            self.range.0.clone().unwrap_or_default(),
227            self.range
228                .1
229                .clone()
230                .unwrap_or_else(|| "9999-12-31".to_owned()),
231        );
232        if first > last {
233            std::mem::swap(&mut first, &mut last);
234        }
235        let search = Search {
236            query: query.to_owned(),
237            regex,
238            days: (first, last),
239            streams: self.view.export_streams(),
240        };
241        self.searched = Some(search.clone());
242        self.asked.push(Ask::Search(search, None));
243    }

245    /// Ask for the search's next page, when there is one: once.
246    pub(crate) fn load_more(&mut self) {
247        let (Some(search), Some(found)) = (&self.searched, &mut self.found) else {
248            return;
249        };
250        if let Some(resume) = found.resume.take() {
251            self.asked.push(Ask::Search(search.clone(), Some(resume)));
252        }
253    }

263            return;

265        self.asked.push(Ask::Around(hit.clone(), around::READ));

272            && context.widen()
273        {

299    /// Take what the reads answered.
300    pub(crate) fn take_replies(&mut self) {
301        let replies = std::mem::take(&mut *crate::logs::lock(&self.inbox));
302        for reply in replies {
303            if !matches!(reply, Reply::More(..)) {
304                self.waiting = self.waiting.saturating_sub(1);
305            }
306            self.answer(reply);
307        }
308    }

310    pub(super) fn answer(&mut self, reply: Reply) {
311        match reply {
312            Reply::Days(Ok((days, usage))) => {
313                // Opened on the newest day, and kept on the one chosen.
314                let newest = days.first().map(|(day, _)| day.clone());
315                if self.day.is_none()
316                    && let Some(day) = newest.clone()
317                {
318                    self.asked.push(Ask::Day(day.clone()));
319                    self.day = Some(day);
320                }
321                if self.from.is_empty() {
322                    self.from = newest.clone().unwrap_or_default();
323                    self.to = newest.unwrap_or_default();
324                }
325                self.days = days;
326                self.usage = Some(usage);
327            }
328            Reply::Day(day, Ok((lines, tail))) => {
329                if self.day.as_deref() == Some(day.as_str()) {
330                    self.view.learn(&lines);
331                    self.lines = lines;
332                    self.tail = Some(tail);
333                    self.day_selection = None;
334                    self.generation += 1;
335                }
336            }
337            Reply::More(day, from, read) => {
338                self.following = false;
339                let ours =
340                    self.day.as_deref() == Some(day.as_str()) && self.tail.as_ref() == Some(&from);
341                match read {
342                    Ok((lines, tail)) if ours => {
343                        if !lines.is_empty() {
344                            self.view.learn(&lines);
345                            self.lines.extend(lines);
346                            self.generation += 1;
347                        }
348                        self.tail = Some(tail);
349                    }
350                    Err(why) if ours => self.said = Some(why),
351                    // Read from before the day was read again, or another day.
352                    _ => {}
353                }
354            }
355            // A search asked before the last one, answering late.
356            Reply::Found(search, _, _) if self.searched.as_ref() != Some(&search) => {}
357            Reply::Found(_, further, Ok(page)) => {
358                self.view.learn(&page.entries);
359                match &mut self.found {
360                    Some(found) if further => {
361                        found.entries.extend(page.entries);
362                        found.resume = page.resume;
363                    }
364                    _ => {
365                        self.found = Some(page);
366                        self.context = None;
367                        self.found_selection = None;
368                    }
369                }
370                self.generation += 1;
371            }
372            // Lines around a result no longer picked.
373            Reply::Around(hit, _)
374                if self
375                    .context
376                    .as_ref()
377                    .is_none_or(|context| context.hit != hit) => {}
378            Reply::Around(_, Ok(Some((lines, at)))) => {
379                if let Some(context) = &mut self.context {
380                    context.heard(lines, at);
381                }
382            }
383            Reply::Around(hit, Ok(None)) => {
384                self.said = Some(format!("That line is no longer in {}'s log.", hit.day));
385            }
386            Reply::Exported(Ok(done)) => {
387                self.said = Some(format!(
388                    "{} lines from {} days written to {}",
389                    done.lines,
390                    done.days,
391                    done.path.display()
392                ));
393            }
394            Reply::Days(Err(why))
395            | Reply::Day(_, Err(why))
396            | Reply::Found(_, _, Err(why))
397            | Reply::Around(_, Err(why))
398            | Reply::Exported(Err(why)) => self.said = Some(why),
399        }
400    }

402    /// Closed: what it read is let go, a day of lines can be tens of
403    /// megabytes, and nothing is asked until it opens again ([`Self::reopen`]):
404    /// a read answered while it is closed would wait in its inbox, unseen,
405    /// as big as the day it read.
406    pub(crate) fn close(&mut self) {
407        self.open = false;
408        self.lines = Vec::new();
409        self.found = None;
410        self.searched = None;
411        self.context = None;
412        self.tail = None;
413        self.following = false;
414        self.day_selection = None;
415        self.found_selection = None;
416        self.day_rows = super::view::Filtered::default();
417        self.found_rows = super::view::Filtered::default();
418        self.matches = super::find::Matches::default();
419        self.generation += 1;
420    }
421
422    /// Opened again: its days and the day it showed, read again.
423    pub(crate) fn reopen(&mut self) {
424        if self.open {
425            return;
426        }
427        self.open = true;
428        self.asked.push(Ask::Days);
429        if let Some(day) = self.day.clone() {
430            self.asked.push(Ask::Day(day));
431        }
432    }

443/// Open `folder` in the system's file browser. What it cannot is said.
444pub(crate) fn open_folder(folder: &Path) -> Result<(), String> {
445    let program = if cfg!(target_os = "windows") {
446        "explorer"
447    } else if cfg!(target_os = "macos") {
448        "open"
449    } else {
450        "xdg-open"
451    };
452    std::fs::create_dir_all(folder).map_err(|e| e.to_string())?;
453    std::process::Command::new(program)
454        .arg(folder)
455        .spawn()
456        .map(drop)
457        .map_err(|e| format!("{program} could not open {}: {e}", folder.display()))
458}

460/// The menu of the line right-clicked in a list, `pointed` among `rows`,
461/// when it is a line of the game's a rule can sort.
462pub(super) fn menu_for(
463    rows: super::view::Rows<'_>,
464    pointed: Option<(usize, egui::Pos2)>,
465) -> Option<LineMenu> {
466    let (row, at) = pointed?;
467    let (entry, _) = rows.get(row)?;
468    sortable(&entry.stream)?;
469    Some(LineMenu {
470        entry: entry.clone(),
471        at,
472    })
473}

475/// The stream the game sent a line tagged `tag` to, when it is one a rule
476/// can sort: not a command, Hydra's own word or a readout.
477pub(super) fn sortable(tag: &str) -> Option<&str> {
478    use hydra_session::player_log::{COMMANDS, NOTICES, READOUTS};
479    let source = tag.split('/').next().unwrap_or(tag);
480    (source != COMMANDS && source != NOTICES && !READOUTS.contains(&source)).then_some(source)

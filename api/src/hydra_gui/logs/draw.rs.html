13    /// Draw the window into `ui`.
14    pub(crate) fn show(&mut self, ui: &mut egui::Ui) {
15        self.take_replies();
16        // Today, read on as it is played.
17        let now = ui.input(|input| input.time);
18        if let Some(after) = self.follow_today(now, &hydra_net::date_dir()) {
19            ui.ctx().request_repaint_after(after);
20        }
21        // Ctrl+F: Find over *Recent*'s lines; in *Search*, its field.
22        if ui.input_mut(|input| input.consume_key(egui::Modifiers::COMMAND, egui::Key::F)) {
23            match self.tab {
24                Tab::Recent => match &mut self.finding {
25                    Some(bar) => bar.focus = true,
26                    None => self.finding = Some(Finding::opened()),
27                },
28                Tab::Search => self.focus_query = true,
29                Tab::Export => {}
30            }
31        }
32        ui.horizontal(|ui| {
33            for (tab, label) in [
34                (Tab::Recent, "Recent"),
35                (Tab::Search, "Search"),
36                (Tab::Export, "Export"),
37            ] {
38                ui.selectable_value(&mut self.tab, tab, label);
39            }
40            if self.waiting > 0 {
41                ui.spinner();
42            }
43        });
44        ui.separator();
45        match self.tab {
46            Tab::Recent => self.recent(ui),
47            Tab::Search => self.search(ui),
48            Tab::Export => self.export(ui),
49        }
50        self.line_menu(ui.ctx());
51    }

53    /// The menu of the line right-clicked, while it is open: the streams
54    /// the lines like it can be sent to, a rule made for them brought to
55    /// the Streams page on a click. A click outside it, or Escape, closes
56    /// it.
57    fn line_menu(&mut self, context: &egui::Context) {
58        let Some(menu) = &self.line_menu else {
59            return;
60        };
61        let (pressed, pointer, escape) = context.input(|input| {
62            (
63                input.pointer.primary_pressed(),
64                input.pointer.interact_pos(),
65                input.key_pressed(egui::Key::Escape),
66            )
67        });
68        let from = super::window::sortable(&menu.entry.stream);
69        let mut picked = None;
70        let shown = egui::Area::new(egui::Id::new(("log-line-menu", &self.character)))
71            .order(egui::Order::Foreground)
72            .fixed_pos(menu.at)
73            .show(context, |ui| {
74                egui::Frame::menu(ui.style()).show(ui, |ui| {
75                    ui.set_min_width(170.0);
76                    ui.label("Send lines like this to:");
77                    picked = crate::stream_editor::send_to(ui, from, &self.own_streams);
78                });
79            });
80        if let Some(to) = &picked {
81            let runs = super::lines::runs(&menu.entry);
82            self.brought = Some(crate::stream_editor::Brought::of(from, &runs, to));
83        }
84        let outside = pressed && pointer.is_some_and(|at| !shown.response.rect.contains(at));
85        if picked.is_some() || outside || escape {
86            self.line_menu = None;
87        }
88    }

90    /// A day, whole: the day picker, *Go to*, the filters, the lines;
91    /// today's read on as it is played.
92    fn recent(&mut self, ui: &mut egui::Ui) {
93        let goto = self.day_row(ui);
94        self.filters(ui);
95        let lines = self.day_rows.of(&self.lines, self.generation, &self.view);
96        // Go to a time: the first line shown at or after it, marked.
97        if let Some(typed) = goto {
98            match time_of(&typed) {
99                None => {
100                    self.said = Some(format!(
101                        "`{typed}` is not a time of day: say 12:12, or 12:12:30."
102                    ));
103                }
104                Some(time) => {
105                    let first = (0..lines.len())
106                        .filter_map(|row| lines.get(row))
107                        .find(|(entry, _)| entry.at.as_str() >= time.as_str());
108                    match first {
109                        Some((entry, _)) => self.spot = Some(Spot::new(Hit::of(entry))),
110                        None => self.said = Some(format!("No line shown is at or after {time}.")),
111                    }
112                }
113            }
114        }
115        let spot = self
116            .spot
117            .as_mut()
118            .and_then(|spot| spot.row(&self.lines, lines, self.generation));
119        let mut unmark = false;
120        if let Some(marked) = &self.spot {
121            ui.horizontal(|ui| {
122                let at = marked.hit.at.get(..8).unwrap_or(&marked.hit.at);
123                ui.weak(format!("Marked: the line at {at}"));
124                if marked.left_out(spot) {
125                    ui.weak("· its class or sender is not ticked, so it is not shown");
126                }
127                unmark = ui.small_button("×").on_hover_text("Unmark it").clicked();
128            });
129        }
130        let marks = find::bar(
131            ui,
132            (&mut self.finding, &mut self.matches),
133            (&self.lines, lines, self.generation),
134            &self.view,
135            spot.map(|(row, _)| row),
136        );
137        selection_bar(ui, lines, &mut self.day_selection, self.view.show);
138        let drawn = Drawn {
139            show: self.view.show,
140            // The newest kept in view, unless Find is going through them or
141            // a line is marked.
142            follow: self.finding.is_none() && self.spot.is_none(),
143            tall: ui.available_height() - 64.0,
144            marks: marks.as_ref(),
145            spot,
146            groups: &[],
147            pick: true,
148            selected: self.day_selection,
149            salt: "log-day",
150        };
151        let acted = lines_area(ui, lines, drawn);
152        self.clicked.extend(acted.link);
153        if let Some(menu) = super::window::menu_for(lines, acted.pointed) {
154            self.line_menu = Some(menu);
155        }
156        if let Some(picked) = acted.picked
157            && let Some(index) = lines.index(picked.row)
158        {
159            select(&mut self.day_selection, index, picked.shift);
160        }
161        if unmark {
162            self.spot = None;
163        }
164        self.footer(ui);
165    }

167    /// The day picker, *Refresh*, *Go to*, and whether the day is today's,
168    /// read on as it is played: the time typed to go to, once Enter is
169    /// pressed in it.
170    fn day_row(&mut self, ui: &mut egui::Ui) -> Option<String> {
171        let chosen = self.day.clone().unwrap_or_default();
172        let shown = self.day_label(&chosen);
173        let labels: Vec<(String, String)> = self
174            .days
175            .iter()
176            .map(|(day, _)| (day.clone(), self.day_label(day)))
177            .collect();
178        let live = self.day.as_deref() == Some(hydra_net::date_dir().as_str());
179        let (mut picked, mut refresh, mut goto) = (None, false, None);
180        ui.horizontal(|ui| {
181            ui.label("Day");
182            egui::ComboBox::from_id_salt(("log-day", &self.character))
183                .selected_text(shown)
184                .show_ui(ui, |ui| {
185                    for (day, label) in labels {
186                        if ui.selectable_label(day == chosen, label).clicked() {
187                            picked = Some(day);
188                        }
189                    }
190                });
191            refresh = ui
192                .button("Refresh")
193                .on_hover_text("Read the days and this day again")
194                .clicked();
195            if live {
196                ui.weak("● live")
197                    .on_hover_text("Today: its lines come in as they are written");
198            }
199            ui.separator();
200            let label = ui.label("Go to");
201            let field = ui
202                .add(
203                    egui::TextEdit::singleline(&mut self.goto)
204                        .hint_text("12:12")
205                        .desired_width(70.0),
206                )
207                .labelled_by(label.id)
208                .on_hover_text("The first line shown at or after a time of day, marked");
209            if field.lost_focus()
210                && ui.input(|input| input.key_pressed(egui::Key::Enter))
211                && !self.goto.trim().is_empty()
212            {
213                goto = Some(self.goto.trim().to_owned());
214            }
215        });
216        if let Some(day) = picked.filter(|day| *day != chosen) {
217            self.choose_day(day);
218        }
219        if refresh {
220            self.asked.push(Ask::Days);
221            if let Some(day) = self.day.clone() {
222                self.asked.push(Ask::Day(day));
223            }
224        }
225        goto
226    }

228    /// The presets, *Dedup*, *Show*, a box per class the window has seen,
229    /// and one per sender.
230    fn filters(&mut self, ui: &mut egui::Ui) {
231        let view = &mut self.view;
232        ui.horizontal(|ui| {
233            for preset in Preset::ALL {
234                if ui.button(preset.label()).clicked() {
235                    view.preset(preset);
236                }
237            }
238            ui.checkbox(&mut view.dedup, "Dedup")
239                .on_hover_text("Show a run of the same line once, with its count. Nothing is left out of an export.");
240            show_menu(ui, &mut view.show);
241        });
242        ui.horizontal_wrapped(|ui| {
243            for (class, on) in &mut view.ticked {
244                ui.checkbox(on, class.as_str());
245            }
246        });
247        if !view.senders.is_empty() {
248            ui.horizontal_wrapped(|ui| {
249                ui.weak("Sent by")
250                    .on_hover_text("Commands are shown when who sent them is ticked.");
251                for (sender, on) in &mut view.senders {
252                    ui.checkbox(on, sender.as_str());
253                }
254            });
255        }
256        ui.separator();
257    }

259    /// Lines holding a text, in the days chosen and the classes and senders
260    /// ticked, newest first, a day at a time; a page more on asking; the
261    /// lines around the one picked under them.
262    fn search(&mut self, ui: &mut egui::Ui) {
263        ui.horizontal(|ui| {
264            let typed = ui.add(
265                egui::TextEdit::singleline(&mut self.query)
266                    .hint_text("Text to find, any case")
267                    .desired_width(320.0),
268            );
269            if std::mem::take(&mut self.focus_query) {
270                typed.request_focus();
271            }
272            ui.checkbox(&mut self.regex, "Regex")
273                .on_hover_text("A regular expression, where case matters unless it says (?i)");
274            let entered = typed.lost_focus() && ui.input(|i| i.key_pressed(egui::Key::Enter));
275            if (ui.button("Search").clicked() || entered) && !self.query.trim().is_empty() {
276                let query = self.query.trim().to_owned();
277                self.ask_search(&query, self.regex);
278            }
279        });
280        ui.horizontal(|ui| {
281            ui.label("Days");
282            day_choice(
283                ui,
284                ("log-search-from", &self.character),
285                &mut self.range.0,
286                ("first kept", &self.days),
287            );
288            ui.label("to");
289            day_choice(
290                ui,
291                ("log-search-to", &self.character),
292                &mut self.range.1,
293                ("last kept", &self.days),
294            );
295        });
296        self.filters(ui);
297        let asked = self.results(ui);
298        match asked {
299            Asked::None => {}
300            Asked::More => self.load_more(),
301            Asked::Hit(hit, true) => self.show_in_day(hit),
302            Asked::Hit(hit, false) => self.pick(hit),
303        }
304        self.footer(ui);
305    }

307    /// What the search found, a heading above each day's, with *Load more*
308    /// while there is more; the lines around the one picked under them; a
309    /// word when the ticks changed since it was asked. What a click there
310    /// asked.
311    fn results(&mut self, ui: &mut egui::Ui) -> Asked {
312        let again = self.searched.as_ref().is_some_and(|search| {
313            self.found.is_some() && search.streams != self.view.export_streams()
314        }) && ui
315            .horizontal(|ui| {
316                ui.weak("The ticks changed since this search, which looked through the others.");
317                ui.small_button("Search again").clicked()
318            })
319            .inner;
320        if again && let Some(search) = self.searched.clone() {
321            self.ask_search(&search.query, search.regex);
322        }
323        let mut asked = Asked::None;
324        if let Some(found) = &self.found {
325            let lines = self
326                .found_rows
327                .of(&found.entries, self.generation, &self.view);
328            let groups = around::groups(lines);
329            ui.horizontal(|ui| {
330                ui.label(format!(
331                    "{} found in {} {}, newest first",
332                    lines.len(),
333                    groups.len(),
334                    if groups.len() == 1 { "day" } else { "days" }
335                ));
336                if found.resume.is_some()
337                    && ui
338                        .button("Load more")
339                        .on_hover_text("The next thousand, further back")
340                        .clicked()
341                {
342                    asked = Asked::More;
343                }
344            });
345            let room = ui.available_height() - 64.0;
346            let tall = if self.context.is_some() {
347                room * 0.5
348            } else {
349                room
350            };
351            let picked = self.context.as_ref().and_then(|context| {
352                (0..lines.len()).find(|row| {
353                    lines
354                        .get(*row)
355                        .is_some_and(|(entry, _)| context.hit.is(entry))
356                })
357            });
358            selection_bar(ui, lines, &mut self.found_selection, self.view.show);
359            let drawn = Drawn {
360                show: self.view.show,
361                follow: false,
362                tall,
363                marks: None,
364                spot: picked.map(|row| (row, false)),
365                groups: &groups,
366                pick: true,
367                selected: self.found_selection,
368                salt: "log-results",
369            };
370            let acted = lines_area(ui, lines, drawn);
371            self.clicked.extend(acted.link);
372            if let Some(menu) = super::window::menu_for(lines, acted.pointed) {
373                self.line_menu = Some(menu);
374            }
375            if let Some(picked) = acted.picked
376                && let (Some((entry, _)), Some(index)) =
377                    (lines.get(picked.row), lines.index(picked.row))
378            {
379                select(&mut self.found_selection, index, picked.shift);
380                // Shift+click only selects.
381                if !picked.shift {
382                    asked = Asked::Hit(Hit::of(entry), picked.twice);
383                }
384            }
385            self.context_panel(ui, room - tall);
386        }
387        asked
388    }

390    /// The lines around the result picked, as they were written, every
391    /// class: *More* lines, *Show in day*, and × to close.
392    fn context_panel(&mut self, ui: &mut egui::Ui, tall: f32) {
393        let Some(context) = &mut self.context else {
394            return;
395        };
396        ui.separator();
397        let mut asked = Panel::None;
398        ui.horizontal(|ui| {
399            let at = context.hit.at.get(..8).unwrap_or(&context.hit.at);
400            ui.strong(format!("Around {} {at}", context.hit.day));
401            if ui
402                .button("More")
403                .on_hover_text(format!("{} more lines each side", around::STEP))
404                .clicked()
405            {
406                asked = Panel::More;
407            }
408            if ui
409                .button("Show in day")
410                .on_hover_text(
411                    "Recent, at this line in its day; a double-click on a result does too",
412                )
413                .clicked()
414            {
415                asked = Panel::ShowInDay;
416            }
417            if ui.small_button("×").on_hover_text("Close").clicked() {
418                asked = Panel::Close;
419            }
420        });
421        // Brought to it once its lines are read.
422        let go = context.shown().is_some() && std::mem::take(&mut context.go);
423        let hit = context.hit.clone();
424        if let Some((rows, at)) = context.shown() {
425            let drawn = Drawn {
426                show: self.view.show,
427                follow: false,
428                tall: (tall - 28.0).max(60.0),
429                marks: None,
430                spot: Some((at, go)),
431                groups: &[],
432                pick: false,
433                selected: None,
434                salt: "log-around",
435            };
436            let acted = lines_area(ui, rows, drawn);
437            self.clicked.extend(acted.link);
438            if let Some(menu) = super::window::menu_for(rows, acted.pointed) {
439                self.line_menu = Some(menu);
440            }
441        } else {
442            ui.weak("Reading the lines around it...");
443        }
444        match asked {
445            Panel::None => {}
446            Panel::More => self.widen(),
447            Panel::ShowInDay => self.show_in_day(hit),
448            Panel::Close => self.context = None,
449        }
450    }

454        ui.label(
455            "Write every line of the days chosen, of the ticked kinds and senders, to one text \

463                    .show_ui(ui, |ui| {

465                            ui.selectable_value(value, day.clone(), day);
466                        }
467                    });
468                ui.end_row();
469            }
470        });
471        self.filters(ui);
472        if ui.button("Create log file").clicked() && !self.from.is_empty() {
473            self.asked.push(Ask::Export(
474                self.from.clone(),
475                self.to.clone(),
476                self.view.export_streams(),
477                self.view.parts(),
478            ));
479        }
480        if ui.button("Open exports folder").clicked() {
481            self.said = open_folder(&self.folder().join("exports")).err();
482        }
483        self.footer(ui);

486    /// What was said, the disk used, and the folder.
487    fn footer(&mut self, ui: &mut egui::Ui) {
488        ui.separator();
489        ui.horizontal(|ui| {
490            if ui.button("Open logs folder").clicked() {
491                self.said = open_folder(&self.folder()).err();
492            }
493            if let Some(usage) = &self.usage {
494                ui.label(format!(
495                    "{} days · {} in all, {} plain, {} archived",
496                    usage.days,
497                    size(usage.total()),
498                    size(usage.plain),
499                    size(usage.archived)
500                ));
501            }
502        });
503        // Lines the log fell behind on: the days read are not whole.
504        if self.dropped > 0 {
505            ui.colored_label(
506                crate::theme::color(ui.ctx(), crate::theme::T::Wrong),
507                format!(
508                    "{} lines this session came faster than the log could keep, and were not \
509                     recorded.",
510                    self.dropped
511                ),
512            );
513        }
514        // An archive that cannot be read: its days are not listed.
515        for why in self.usage.iter().flat_map(|usage| &usage.unreadable) {
516            ui.colored_label(
517                crate::theme::color(ui.ctx(), crate::theme::T::Wrong),
518                format!("Could not read an archive, so its days are not listed: {why}"),
519            );
520        }
521        if let Some(said) = &self.said {
522            ui.label(said.as_str());
523        }
524    }

526    /// A day as the picker shows it: with its size, or that it is archived.
527    pub(super) fn day_label(&self, day: &str) -> String {
528        match self.days.iter().find(|(d, _)| d == day) {
529            Some((_, Some(bytes))) => format!("{day} · {}", size(*bytes)),
530            Some((_, None)) => format!("{day} · archived"),
531            None if day.is_empty() => "none kept yet".to_owned(),
532            None => day.to_owned(),
533        }
534    }

569/// A day to choose, among `days`, or none: `none` says what none means.
570fn day_choice(
571    ui: &mut egui::Ui,
572    salt: impl std::hash::Hash + std::fmt::Debug,
573    chosen: &mut Option<String>,
574    (none, days): (&str, &Days),
575) {
576    egui::ComboBox::from_id_salt(salt)
577        .selected_text(chosen.as_deref().unwrap_or(none))
578        .show_ui(ui, |ui| {
579            ui.selectable_value(chosen, None, none);
580            for (day, _) in days {
581                ui.selectable_value(chosen, Some(day.clone()), day);
582            }
583        });
584}

586/// `typed`, a time of day as a player types it -- `12`, `12:12`, `9:05:30`
587/// -- as the log writes one, `HH:MM:SS`; `None` when it is none.
588pub(super) fn time_of(typed: &str) -> Option<String> {
589    let mut parts = typed.trim().split(':');
590    let mut next = |most: u32| -> Option<u32> {
591        match parts.next() {
592            None => Some(0),
593            Some(part) => part.trim().parse().ok().filter(|n| *n <= most),
594        }
595    };
596    let (hours, minutes, seconds) = (next(23)?, next(59)?, next(59)?);
597    if parts.next().is_some() {
598        return None;
599    }
600    Some(format!("{hours:02}:{minutes:02}:{seconds:02}"))
601}

14/// A search as asked: the text, whether it is a regular expression, the
15/// days it looks through, first and last (`YYYY-MM-DD`), and the tags it
16/// keeps.
17#[derive(Clone, Debug, PartialEq, Eq)]
18pub(crate) struct Search {
19    pub(crate) query: String,
20    pub(crate) regex: bool,
21    pub(crate) days: (String, String),
22    pub(crate) streams: Streams,
23}

25/// One line of a day, known by its day, its stamp, its tag and its text: a
26/// result picked, or the line *Show in day* went to.
27#[derive(Clone, Debug, PartialEq, Eq)]
28pub(crate) struct Hit {
29    pub(crate) day: String,
30    pub(crate) at: String,
31    pub(crate) stream: String,
32    pub(crate) text: String,
33}
34
35impl Hit {
36    /// `entry`, as a hit.
37    pub(crate) fn of(entry: &Entry) -> Self {
38        Self {
39            day: entry.day.clone(),
40            at: entry.at.clone(),
41            stream: entry.stream.clone(),
42            text: entry.text.clone(),
43        }
44    }
45
46    /// Whether `entry` is this line.
47    pub(crate) fn is(&self, entry: &Entry) -> bool {
48        entry.at == self.at
49            && entry.stream == self.stream
50            && entry.text == self.text
51            && entry.day == self.day
52    }
53}

55/// A read the window wants done.
56#[derive(Clone, Debug, PartialEq, Eq)]
57pub(crate) enum Ask {
58    /// The days kept, and the disk they take.
59    Days,
60    /// One day's lines, and how far its files were read.
61    Day(String),
62    /// The lines a day being played added since it was read this far.
63    More(String, Tail),
64    /// A page of a search: the first, or the one after where the last
65    /// stopped.
66    Search(Search, Option<Resume>),
67    /// The lines around a line, this many before it and after.
68    Around(Hit, usize),
69    /// An export of these days, `from` to `to`, of these tags, with these
70    /// parts of each line.
71    Export(String, String, Streams, Parts),
72}
73
74/// What a read answered.
75#[derive(Clone, Debug)]
76pub(crate) enum Reply {
77    /// The days kept, and the disk used.
78    Days(Result<(Days, Usage), String>),
79    /// A day's lines, and how far its files were read.
80    Day(String, Result<(Vec<Entry>, Tail), String>),
81    /// The lines a day added since the read it was asked from, and how far
82    /// its files are read now.
83    More(String, Tail, Result<(Vec<Entry>, Tail), String>),
84    /// A page of a search, with the search it answers, and whether it is a
85    /// page after the first.
86    Found(Search, bool, Result<Page, String>),
87    /// The lines around a line, and where it is among them; `None` when its
88    /// day no longer holds it.
89    Around(Hit, Result<Option<(Vec<Entry>, usize)>, String>),
90    /// What an export wrote.
91    Exported(Result<Exported, String>),
92}

94/// The days kept, newest first, each with its plain size (`None`: archived).
95pub(crate) type Days = Vec<(String, Option<u64>)>;
96
97/// Where answers land for the window to take: shared with the read.
98pub(crate) type Inbox = Arc<Mutex<Vec<Reply>>>;

100/// Do `ask` against `character`'s log under `root`, a line written with no
101/// class read back as `rules`, the stream rules now, sort it
102/// (`reader::sort_again`). Blocking file I/O: never on the window's thread.
103pub(crate) fn run(root: &Path, (character, rules): (&str, &Compiled), ask: Ask) -> Reply {
104    let text = |e: std::io::Error| e.to_string();
105    match ask {
106        Ask::Days => Reply::Days(days(root, character).map_err(text)),
107        Ask::Day(day) => {
108            let lines = reader::read_day_from(root, character, &day, None).map_err(text);
109            Reply::Day(day, sorted(lines, rules))
110        }
111        Ask::More(day, tail) => {
112            let lines = reader::read_day_from(root, character, &day, Some(&tail)).map_err(text);
113            Reply::More(day, tail, sorted(lines, rules))
114        }
115        Ask::Search(search, resume) => {
116            let further = resume.is_some();
117            let page = if search.regex {
118                Pattern::regex(&search.query)
119            } else {
120                Pattern::literal(&search.query)
121            }
122            .and_then(|pattern| {
123                reader::search_days(
124                    root,
125                    character,
126                    (&pattern, &search.streams),
127                    (&search.days.0, &search.days.1),
128                    resume.as_ref(),
129                    Some(rules),
130                )
131                .map_err(text)
132            });
133            Reply::Found(search, further, page)
134        }
135        Ask::Around(hit, radius) => {
136            let read = reader::around(
137                root,
138                character,
139                &hit.day,
140                (&hit.at, &hit.stream, &hit.text),
141                (radius, Some(rules)),
142            )
143            .map_err(text);
144            Reply::Around(hit, read)
145        }
146        Ask::Export(from, to, streams, parts) => {
147            let (from, to) = if from <= to { (from, to) } else { (to, from) };
148            let out = reader::export_path(root, character, (&from, &to));
149            Reply::Exported(
150                reader::export(
151                    root,
152                    character,
153                    (&from, &to),
154                    (&streams, Some(rules)),
155                    parts,
156                    &out,
157                )
158                .map_err(text),
159            )
160        }
161    }
162}

164/// A day's `lines` as `rules` sort them now, once read.
165fn sorted<T>(
166    lines: Result<(Vec<Entry>, T), String>,
167    rules: &Compiled,
168) -> Result<(Vec<Entry>, T), String> {
169    lines.map(|(mut lines, tail)| {
170        reader::sort_again(&mut lines, rules);
171        (lines, tail)
172    })
173}

175/// The days kept with their plain sizes, and the disk used.
176fn days(root: &Path, character: &str) -> std::io::Result<(Days, Usage)> {
177    let plain = writer::days(root, character)?;
178    let days = reader::days(root, character)?
179        .into_iter()
180        .map(|day| {
181            let sizes: Vec<u64> = plain
182                .iter()
183                .filter(|path| writer::day_of(path).as_deref() == Some(day.as_str()))
184                .map(|path| std::fs::metadata(path).map_or(0, |m| m.len()))
185                .collect();
186            let size = (!sizes.is_empty()).then(|| sizes.iter().sum());
187            (day, size)
188        })
189        .collect();
190    Ok((days, archive::usage(root, character)?))
191}

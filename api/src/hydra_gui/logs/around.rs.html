1//! A result looked at closer. Picked, the lines around it are shown in a
2//! panel under the results as they were written, every class: what was
3//! around it is what happened, whatever the filters show. *More* shows more
4//! each side. *Show in day* takes *Recent* to its day and to the line
5//! itself, marked, and Find there carries on from it.
6//!
7//! The results are grouped by day, a heading above each day's.
8
9use hydra_session::player_log::reader::Entry;

11use super::read::Hit;
12use super::view::Rows;

14/// Lines a click on *More* shows, each side; and how many show at first.
15pub(crate) const STEP: usize = 5;
16
17/// Lines read each side of a result at a time: *More* reads again only
18/// past them.
19pub(crate) const READ: usize = 50;
20
21/// The panel under the results: the lines around the one picked.
22#[derive(Debug)]
23pub(crate) struct Context {
24    pub(crate) hit: Hit,
25    /// Lines shown each side of it.
26    radius: usize,
27    /// Lines asked for each side of it.
28    pub(crate) asked: usize,
29    /// The lines read around it, and where it is among them; `None` until
30    /// read.
31    read: Option<(Vec<Entry>, usize)>,
32    /// Each line read, one row each: the panel filters none.
33    rows: Vec<(usize, usize)>,
34    /// The panel is to be brought to it.
35    pub(crate) go: bool,
36}
37
38impl Context {
39    /// The panel for `hit`, its lines asked for.
40    pub(crate) const fn new(hit: Hit) -> Self {
41        Self {
42            hit,
43            radius: STEP,
44            asked: READ,
45            read: None,
46            rows: Vec::new(),
47            go: false,
48        }
49    }
50
51    /// The lines read around it, and where it is among them.
52    pub(crate) fn heard(&mut self, lines: Vec<Entry>, at: usize) {
53        self.rows = (0..lines.len()).map(|row| (row, 1)).collect();
54        self.read = Some((lines, at));
55        self.go = true;
56    }
57
58    /// The lines shown, as many each side of it as asked and its day
59    /// holds, and its row among them; `None` until they are read.
60    pub(crate) fn shown(&self) -> Option<(Rows<'_>, usize)> {
61        let (lines, at) = self.read.as_ref()?;
62        let first = at.saturating_sub(self.radius);
63        let last = at
64            .saturating_add(self.radius)
65            .min(lines.len().checked_sub(1)?);
66        let window = lines.get(first..=last)?;
67        let rows = self.rows.get(..window.len())?;
68        Some((Rows::new(window, rows), at - first))
69    }
70
71    /// [`STEP`] more lines each side: whether they must be read first, its
72    /// day holding more than was read.
73    pub(crate) fn widen(&mut self) -> bool {
74        self.radius += STEP;
75        let whole_day = self
76            .read
77            .as_ref()
78            .is_some_and(|(lines, at)| *at < self.asked && lines.len() - at <= self.asked);
79        if self.radius > self.asked && !whole_day {
80            self.asked = self.radius + READ;
81            return true;
82        }
83        false
84    }
85}

87/// The line *Show in day* went to, marked in *Recent*.
88#[derive(Debug)]
89pub(crate) struct Spot {
90    pub(crate) hit: Hit,
91    /// Where it is among the day's lines, as found in their `generation`th
92    /// set: `None` when they do not hold it.
93    place: Option<(u64, Option<usize>)>,
94    /// The list is to be brought to it.
95    go: bool,
96}
97
98impl Spot {
99    /// The line `hit`, the list to be brought to it.
100    pub(crate) const fn new(hit: Hit) -> Self {
101        Self {
102            hit,
103            place: None,
104            go: true,
105        }
106    }
107
108    /// Its row among `rows`, those shown of `entries`, the `generation`th
109    /// lines; and whether the list is to be brought to it, said once.
110    /// `None` until its day is read, or when the filters leave it out.
111    pub(crate) fn row(
112        &mut self,
113        entries: &[Entry],
114        rows: Rows<'_>,
115        generation: u64,
116    ) -> Option<(usize, bool)> {
117        if self.place.is_none_or(|(at, _)| at != generation) {
118            let index = entries.iter().position(|entry| self.hit.is(entry));
119            self.place = Some((generation, index));
120        }
121        let index = self.place.and_then(|(_, index)| index)?;
122        let row = rows.row_of(index)?;
123        Some((row, std::mem::take(&mut self.go)))
124    }
125
126    /// Whether its day holds it and the filters leave it out: with no row.
127    pub(crate) fn left_out(&self, row: Option<(usize, bool)>) -> bool {
128        row.is_none() && self.place.is_some_and(|(_, index)| index.is_some())
129    }
130}

132/// One day's results: a heading above them.
133#[derive(Debug)]
134pub(crate) struct Group {
135    pub(crate) day: String,
136    /// Its heading's place in the list.
137    start: usize,
138    /// Its first row.
139    row: usize,
140    /// How many rows it has.
141    pub(crate) count: usize,
142}
143
144/// The days `rows` span, in their order, each with its rows.
145pub(crate) fn groups(rows: Rows<'_>) -> Vec<Group> {
146    let mut groups: Vec<Group> = Vec::new();
147    for row in 0..rows.len() {
148        let Some((entry, _)) = rows.get(row) else {
149            continue;
150        };
151        match groups.last_mut() {
152            Some(group) if group.day == entry.day => group.count += 1,
153            _ => {
154                let start = row + groups.len();
155                groups.push(Group {
156                    day: entry.day.clone(),
157                    start,
158                    row,
159                    count: 1,
160                });
161            }
162        }
163    }
164    groups
165}
166
167/// What a place in a list holds: a day's heading, or a row.
168pub(crate) enum Item<'a> {
169    Day(&'a Group),
170    Row(usize),
171}
172
173/// What the list's `place`th place holds, a heading above each of `groups`
174/// (none, and every place is a row).
175pub(crate) fn item(groups: &[Group], place: usize) -> Item<'_> {
176    let Some(at) = groups
177        .partition_point(|group| group.start <= place)
178        .checked_sub(1)
179    else {
180        return Item::Row(place);
181    };
182    let group = &groups[at];
183    if place == group.start {
184        Item::Day(group)
185    } else {
186        Item::Row(group.row + place - group.start - 1)
187    }
188}
189
190/// The place in the list of `row`, a heading above each of `groups`.
191pub(crate) fn place_of(groups: &[Group], row: usize) -> usize {
192    row + groups.partition_point(|group| group.row <= row)
193}

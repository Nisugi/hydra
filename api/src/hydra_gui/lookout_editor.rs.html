1//! The Editors window's Lookout page: the lines the lookout leaves out, one
2//! pattern a line, kept in `lookout.toml` (`hydra_lookout::settings`).
3//!
4//! The list as written is the whole list: the built-in one is there to
5//! edit until a save, and *The built-in list* brings it back. Each pattern
6//! that does not read is named under the list and left out when it is used.
7
8use std::path::Path;
9
10use hydra_lookout::Settings;
11
12/// The page's state, which outlives a frame.
13#[derive(Debug, Default)]
14pub(crate) struct LookoutEditor {
15    /// The list as typed, one pattern a line; read from the file the first
16    /// time the page shows.
17    text: Option<String>,
18    /// What the last save said.
19    said: Option<String>,
20}
21
22impl LookoutEditor {
23    /// The page drawn in `ui`, the file kept in `data`.
24    pub(crate) fn show(&mut self, ui: &mut egui::Ui, data: &Path) {
25        crate::form::hint(
26            ui,
27            "Lines the lookout leaves out, one pattern a line (Rust's regex: no look-around). \
28             The commonest first: a line is left at the first pattern that takes it.",
29        );
30        let text = self.text.get_or_insert_with(|| {
31            let settings = Settings::read(data).unwrap_or_default();
32            settings.ignore_list().join("\n")
33        });
34        egui::ScrollArea::vertical()
35            .id_salt("lookout-ignore")
36            .max_height(ui.available_height() - 90.0)
37            .show(ui, |ui| {
38                ui.add(
39                    egui::TextEdit::multiline(text)
40                        .code_editor()
41                        .desired_rows(20)
42                        .desired_width(f32::INFINITY),
43                );
44            });
45        let unread: Vec<String> = patterns(text)
46            .into_iter()
47            .filter_map(|pattern| {
48                regex::Regex::new(&hydra_model::pattern::ascii(&pattern))
49                    .err()
50                    .map(|why| format!("{pattern}: {why}"))
51            })
52            .collect();
53        for why in &unread {
54            ui.colored_label(ui.visuals().warn_fg_color, why);
55        }
56        let (mut saved, mut built_in) = (false, false);
57        ui.horizontal(|ui| {
58            saved = ui.button("Save").clicked();
59            built_in = ui
60                .button("The built-in list")
61                .on_hover_text("The list Hydra comes with, saved in place of this one")
62                .clicked();
63        });
64        if saved {
65            self.said = Some(save(data, Some(patterns(text))));
66        }
67        if built_in {
68            self.said = Some(save(data, None));
69            self.text = None;
70        }
71        if let Some(said) = &self.said {
72            ui.weak(said);
73        }
74    }
75}
76
77/// The patterns `text` holds, a line each, blank lines left out.
78fn patterns(text: &str) -> Vec<String> {
79    text.lines()
80        .map(str::trim)
81        .filter(|line| !line.is_empty())
82        .map(str::to_owned)
83        .collect()
84}
85
86/// `ignore` saved as the list, `None` the built-in one, and every lookout
87/// told: what was done.
88fn save(data: &Path, ignore: Option<Vec<String>>) -> String {
89    let mut settings = match Settings::read(data) {
90        Ok(settings) => settings,
91        Err(why) => return format!("Not saved: {why}"),
92    };
93    let built_in: Vec<String> = hydra_lookout::settings::IGNORE
94        .iter()
95        .map(|pattern| (*pattern).to_owned())
96        .collect();
97    settings.ignore = ignore.filter(|ignore| *ignore != built_in);
98    match settings.write(data) {
99        Ok(()) => {
100            hydra_lookout::reload(data);
101            match &settings.ignore {
102                Some(ignore) => format!("Saved: {} patterns.", ignore.len()),
103                None => "Saved: the built-in list.".to_owned(),
104            }
105        }
106        Err(why) => format!("Not saved: {why}"),
107    }
108}

110#[cfg(test)]
111mod tests {
112    use super::*;
113
114    #[test]
115    fn a_list_is_saved_and_the_built_in_one_is_none() {
116        let dir = std::env::temp_dir().join(format!("hydra-gui-lookout-{}", std::process::id()));
117        let _ = std::fs::remove_dir_all(&dir);
118        std::fs::create_dir_all(&dir).unwrap();
119        assert_eq!(
120            save(&dir, Some(vec!["^Hello".to_owned()])),
121            "Saved: 1 patterns."
122        );
123        assert_eq!(
124            Settings::read(&dir).unwrap().ignore,
125            Some(vec!["^Hello".to_owned()])
126        );
127        assert_eq!(save(&dir, None), "Saved: the built-in list.");
128        assert_eq!(Settings::read(&dir).unwrap().ignore, None);
129        assert_eq!(patterns("a\n\n  b  \n"), ["a", "b"]);
130        let _ = std::fs::remove_dir_all(&dir);
131    }

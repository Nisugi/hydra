1//! What a trigger does, said in a sentence from the form as it stands: what
2//! it looks for and whose it is, then each thing it does, then what its
3//! cooldown holds back.
4
5use std::fmt::Write as _;
6
7use super::{Draft, Says};
8
9impl Draft {
10    /// What the trigger in the form does, in words a player reads.
11    pub(crate) fn sentence(&self) -> String {
12        let Some(when) = self.when_said() else {
13            return "It looks for nothing yet: give it words or a pattern to look for, or a \
14                    condition under When."
15                .to_owned();
16        };
17        let does = self.does_said();
18        let mut said = if does.is_empty() {
19            format!("{when}: nothing yet.")
20        } else {
21            format!("{when}: {}.", listed(&does))
22        };
23        let held = self.held_back();
24        if !held.is_empty() {
25            let one = matches!(held.as_slice(), [only] if only != "commands");
26            let verb = if one { "comes" } else { "come" };
27            let _ = write!(
28                said,
29                " Its {} {verb} at most once every {}.",
30                listed(&held),
31                seconds(self.cooldown)
32            );
33        }
34        said
35    }
36
37    /// When it acts, and on whom: `When the game says "Bob" (whole words),
38    /// on Ashryn`. `None` when it looks for nothing and has no condition.
39    fn when_said(&self) -> Option<String> {
40        let pattern = self.pattern.trim();
41        let mut when = if pattern.is_empty() {
42            self.when.as_ref()?;
43            "Each time its condition comes to hold".to_owned()
44        } else if self.regex {
45            format!("When a line matches the pattern \"{pattern}\"")
46        } else {
47            let words: Vec<String> = pattern
48                .split('|')
49                .map(str::trim)
50                .filter(|word| !word.is_empty())
51                .map(|word| format!("\"{word}\""))
52                .collect();
53            format!("When the game says {}", words.join(" or "))
54        };
55        let mut how = Vec::new();
56        if !pattern.is_empty() && !self.regex && self.whole_word {
57            how.push("whole words");
58        }
59        if !pattern.is_empty() && self.case_sensitive {
60            how.push("capitals as written");
61        }
62        if !how.is_empty() {
63            let _ = write!(when, " ({})", how.join(", "));
64        }
65        if !pattern.is_empty() && self.when.is_some() {
66            when.push_str(", while its condition holds");
67        }
68        let characters: Vec<&str> = self
69            .characters
70            .split(',')
71            .map(str::trim)
72            .filter(|name| !name.is_empty())
73            .collect();
74        if !characters.is_empty() {
75            let _ = write!(when, ", on {}", characters.join(" or "));
76        }
77        Some(when)
78    }

80    /// Each thing it does, in the form's order.
81    fn does_said(&self) -> Vec<String> {
82        let mut does = Vec::new();
83        if self.color.is_some() || self.background.is_some() {
84            does.push(if self.line {
85                "colours the whole line".to_owned()
86            } else {
87                "colours the words".to_owned()
88            });
89        }
90        if self.squelch {
91            does.push("hides the line".to_owned());
92        }
93        if !self.substitute.is_empty() {
94            does.push(format!(
95                "writes \"{}\" in place of what it found",
96                self.substitute
97            ));
98        }
99        let sound = self.sound.trim();
100        if !sound.is_empty() {
101            does.push(if self.volume == 100 {
102                format!("plays {sound}")
103            } else {
104                format!("plays {sound} at {}%", self.volume)
105            });
106        }
107        let (says, words) = &self.alert;
108        does.extend(said(*says, words, "shows the line in a banner", |words| {
109            format!("shows \"{words}\" in a banner")
110        }));
111        let (says, words) = &self.notify;
112        does.extend(said(
113            *says,
114            words,
115            "sends the line as a notification",
116            |words| format!("sends the notification \"{words}\""),
117        ));
118        let (says, words) = &self.push;
119        does.extend(said(
120            *says,
121            words,
122            "pushes the line to your phone",
123            |words| format!("pushes \"{words}\" to your phone"),
124        ));
125        let (says, words) = &self.speak;
126        does.extend(said(*says, words, "reads the line aloud", |words| {
127            format!("says \"{words}\" aloud")
128        }));
129        let rumble = self.rumble.trim();
130        if !rumble.is_empty() {
131            does.push(format!("rumbles the controller ({rumble})"));
132        }
133        if !self.picture.trim().is_empty() {
134            does.push("places a picture after the line".to_owned());
135        }
136        does.extend(self.timer_said());
137        for (kind, mark) in [("flag", &self.flag), ("status", &self.status)] {
138            let name = mark.name.trim();
139            if name.is_empty() {
140                continue;
141            }
142            does.push(if mark.clear {
143                format!("clears the {kind} {name}")
144            } else if mark.seconds > 0 {
145                format!("sets the {kind} {name} for {} s", mark.seconds)
146            } else {
147                format!("sets the {kind} {name}")
148            });
149        }
150        let commands: Vec<String> = self
151            .send
152            .lines()
153            .map(str::trim)
154            .filter(|command| !command.is_empty())
155            .map(|command| format!("\"{command}\""))
156            .collect();
157        match commands.len() {
158            0 => {}
159            1..=3 => does.push(format!("sends {}", listed(&commands))),
160            many => does.push(format!("sends {many} commands")),
161        }
162        does
163    }

165    /// The timer it starts, how long for, and what ends it.
166    fn timer_said(&self) -> Option<String> {
167        let timer = self.timer.trim();
168        if timer.is_empty() {
169            return None;
170        }
171        let caught = self.caught.trim();
172        let mut said = format!("starts the timer \"{timer}\"");
173        if !caught.is_empty() {
174            let multiplied = if self.times == 1 {
175                String::new()
176            } else {
177                format!(" \u{d7} {}", self.times)
178            };
179            let _ = write!(said, " for {caught}{multiplied} s");
180            if self.seconds > 0 {
181                let _ = write!(said, " ({} s when the line says none)", self.seconds);
182            }
183        } else if self.seconds > 0 {
184            let _ = write!(said, " for {} s", self.seconds);
185        }
186        let down = self.down.trim();
187        if !down.is_empty() {
188            let _ = write!(said, ", which \"{down}\" ends");
189        }
190        Some(said)
191    }
192
193    /// What its cooldown holds back, of what it does.
194    fn held_back(&self) -> Vec<String> {
195        let mut held = Vec::new();
196        for (on, name) in [
197            (!self.sound.trim().is_empty(), "sound"),
198            (self.alert.0 != Says::Nothing, "banner"),
199            (self.notify.0 != Says::Nothing, "notification"),
200            (self.push.0 != Says::Nothing, "push"),
201            (self.speak.0 != Says::Nothing, "speech"),
202            (!self.rumble.trim().is_empty(), "rumble"),
203            (!self.picture.trim().is_empty(), "picture"),
204            (!self.timer.trim().is_empty(), "timer"),
205            (self.sends().is_some(), "commands"),
206        ] {
207            if on {
208                held.push(name.to_owned());
209            }
210        }
211        held
212    }
213}

215/// What a banner, a notification or speech says, as `line` for the line or
216/// `words` for words of its own; nothing for nothing.
217fn said(says: Says, words: &str, line: &str, own: impl Fn(&str) -> String) -> Option<String> {
218    match says {
219        Says::Nothing => None,
220        Says::Words if !words.trim().is_empty() => Some(own(words.trim())),
221        Says::Line | Says::Words => Some(line.to_owned()),
222    }
223}
224
225/// `a`, `a and b`, `a, b and c`.
226fn listed(items: &[String]) -> String {
227    match items {
228        [] => String::new(),
229        [one] => one.clone(),
230        [rest @ .., last] => format!("{} and {last}", rest.join(", ")),
231    }
232}
233
234/// Seconds as a player reads them: `10 s`, `2.5 s`.
235fn seconds(seconds: f64) -> String {
236    if seconds.fract() == 0.0 {
237        format!("{seconds:.0} s")
238    } else {
239        format!("{seconds:.1} s")
240    }

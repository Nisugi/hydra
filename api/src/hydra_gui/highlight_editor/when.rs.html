1//! The form's *When*: a highlight's condition, built from drop-downs.
2//!
3//! The top is a group, every one of its conditions or any one; a condition
4//! in it may be a group of its own, one level down, as the editor builds
5//! them (a file may nest deeper, and is kept so, shown here as a group it
6//! does not open). Each test may be turned over by *not*. None at all, and
7//! the highlight has no condition.
8//!
9//! **Live**, as the hotbar editor draws it: beside each test and group a
10//! mark, lit when it holds for the character now; and an effect's name
11//! picked from those the character has had.
12

20/// What the builder shows beside the tests: whether each holds now, and
21/// the effects' names to pick from. None of it, for a highlight.
22#[derive(Clone, Copy, Debug, Default)]
23pub(crate) struct Live<'a, 'w> {
24    /// The character it is tried on.
25    pub(crate) world: Option<&'a World<'w>>,
26    /// The effects' names the character has had.
27    pub(crate) names: &'a [String],
28}
29
30impl Live<'_, '_> {
31    /// A mark of whether `condition` holds now, when it is tried on one.
32    pub(crate) fn mark(self, ui: &mut egui::Ui, condition: &Condition) {
33        let Some(world) = self.world else {
34            return;
35        };
36        let (colour, said) = if condition.holds(world) {
37            (egui::Color32::from_rgb(90, 200, 90), "Holds now")
38        } else {
39            (egui::Color32::from_rgb(200, 80, 80), "Does not hold now")
40        };
41        let (rect, response) = ui.allocate_exact_size(egui::vec2(12.0, 12.0), egui::Sense::hover());
42        ui.painter().circle_filled(rect.center(), 5.0, colour);
43        response.on_hover_text(said);
44    }
45}

249/// A group's conditions, and whether every one must hold.
250fn group_of(condition: &Condition) -> Option<(&Vec<Condition>, bool)> {
251    match condition {
252        Condition::All { conditions } => Some((conditions, true)),
253        Condition::Any { conditions } => Some((conditions, false)),
254        _ => None,
255    }
256}
257
258/// `conditions` as a group: every one, or any one.
259fn group(conditions: Vec<Condition>, every: bool) -> Condition {
260    if every {
261        Condition::All { conditions }
262    } else {
263        Condition::Any { conditions }
264    }
265}

272/// [`when_ui`], each test marked as it holds now and an effect's name
273/// picked, by `live`.
274pub(crate) fn when_live(ui: &mut egui::Ui, when: &mut Option<Condition>, live: Live<'_, '_>) {
275    let (mut conditions, mut every) = match when.take() {
276        None => (Vec::new(), true),
277        Some(top) => match group_of(&top) {
278            Some((conditions, every)) => (conditions.clone(), every),
279            None => (vec![top], true),
280        },
281    };
282    rows(ui, &mut conditions, &mut every, (0, 0), live);
283    *when = (!conditions.is_empty()).then(|| group(conditions, every));
284}

286/// A group's rows, `depth` 0 at the top, `salt` keeping its widgets apart:
287/// its conditions, each changed or taken out, and one added; a group added
288/// at the top alone.
289fn rows(
290    ui: &mut egui::Ui,
291    conditions: &mut Vec<Condition>,
292    every: &mut bool,
293    (depth, salt): (usize, usize),
294    live: Live<'_, '_>,
295) {
296    let id = (depth, salt);
297    ui.horizontal(|ui| {
298        live.mark(ui, &group(conditions.clone(), *every));
299        egui::ComboBox::from_id_salt(("highlight-group", id))
300            .selected_text(if *every { "Every one of" } else { "Any one of" })
301            .show_ui(ui, |ui| {
302                ui.selectable_value(every, true, "Every one of");
303                ui.selectable_value(every, false, "Any one of");
304            });
305        ui.weak(if conditions.is_empty() {
306            "nothing: no condition"
307        } else {
308            "these:"
309        });
310    });
311    let mut gone = None;
312    for (at, condition) in conditions.iter_mut().enumerate() {
313        ui.horizontal(|ui| {
314            ui.add_space(16.0);
315            if ui.small_button("x").on_hover_text("Take it out").clicked() {
316                gone = Some(at);
317            }
318            if group_of(condition).is_none() {
319                live.mark(ui, condition);
320            }
321            let inner = group_of(condition).map(|(inner, every)| (inner.clone(), every));
322            match inner {
323                Some((mut inner, mut inner_every)) if depth == 0 => {
324                    ui.vertical(|ui| rows(ui, &mut inner, &mut inner_every, (1, at), live));
325                    *condition = group(inner, inner_every);
326                }
327                Some(_) => {
328                    ui.weak("a group, as the file writes it");
329                }
330                None => test_ui(ui, condition, (depth, salt * 100 + at), live.names),
331            }
332        });
333    }
334    if let Some(at) = gone {
335        conditions.remove(at);
336    }
337    ui.horizontal(|ui| {
338        ui.add_space(16.0);
339        if ui.small_button("+ condition").clicked() {
340            conditions.push(fresh("vital"));
341        }
342        if depth == 0 && ui.small_button("+ group").clicked() {
343            conditions.push(Condition::Any {
344                conditions: vec![fresh("indicator")],
345            });
346        }
347    });
348}

350/// One test: turned over or not, its kind, and its fields.
351fn test_ui(ui: &mut egui::Ui, condition: &mut Condition, id: (usize, usize), names: &[String]) {
352    let mut not = matches!(condition, Condition::Not { .. });
353    let mut test = match &*condition {
354        Condition::Not { condition } => (**condition).clone(),
355        other => other.clone(),
356    };
357    ui.checkbox(&mut not, "not");
358    let mut kind = kind_of(&test);
359    egui::ComboBox::from_id_salt(("highlight-kind", id))
360        .selected_text(kind)
361        .show_ui(ui, |ui| {
362            for one in KINDS.iter().chain(&["stance"]) {
363                ui.selectable_value(&mut kind, one, *one);
364            }
365        });
366    if kind != kind_of(&test) {
367        test = fresh(kind);
368    }
369    fields(ui, &mut test, id, names);
370    *condition = if not {
371        Condition::Not {
372            condition: Box::new(test),
373        }
374    } else {
375        test
376    };
377}

379/// The fields of `test`.
380fn fields(ui: &mut egui::Ui, test: &mut Condition, id: (usize, usize), names: &[String]) {
381    let salt = |what: &str| egui::Id::new(("highlight-field", id, what.to_owned()));
382    match test {
383        Condition::EffectActive {
384            category,
385            name,
386            name_match,
387        }
388        | Condition::EffectInactive {
389            category,
390            name,
391            name_match,
392        } => {
393            category_ui(ui, category, salt("category"));
394            named(ui, name, name_match, names);
395        }
396        Condition::EffectTime {
397            category,
398            name,
399            name_match,
400            cmp,
401            seconds,
402        } => {
403            category_ui(ui, category, salt("category"));
404            named(ui, name, name_match, names);
405            cmp_ui(ui, cmp, salt("cmp"));
406            ui.add(egui::DragValue::new(seconds).range(0..=86_400).suffix(" s"));
407        }
408        Condition::RtActive | Condition::CtActive => {
409            ui.weak("is under way");
410        }
411        Condition::Indicator {
412            id: indicator,
413            active,
414        } => {
415            pick(
416                ui,
417                indicator,
418                &crate::widget::status::Indicator::ALL.map(crate::widget::status::Indicator::id),
419                salt("indicator"),
420            );
421            ui.checkbox(active, "lit");
422        }
423        Condition::Vital {
424            vital,
425            cmp,
426            value,
427            unit,
428        } => vital_fields(ui, (vital, cmp, value, unit), id),
429        Condition::Injury { area, cmp, level } => {
430            pick(ui, area, &AREAS, salt("area"));
431            cmp_ui(ui, cmp, salt("cmp"));
432            ui.add(egui::DragValue::new(level).range(0..=6))
433                .on_hover_text("1-3 a wound, 4-6 a scar, 0 none");
434        }
435        Condition::SpellAffordable { number } => {
436            ui.add(egui::DragValue::new(number).range(100..=2000));
437        }
438        Condition::TimeOfDay { phase } => phase_ui(ui, phase, salt("phase")),
439        Condition::HandEmpty { hand } => hand_ui(ui, hand, salt("hand")),
440        Condition::HandHolds {
441            hand,
442            item_type,
443            name,
444            name_match,
445        } => {
446            hand_ui(ui, hand, salt("hand"));
447            optional(ui, item_type, "a kind: weapon, gem...");
448            optional(ui, name, "a name");
449            contains(ui, name_match);
450        }
451        Condition::SpellPrepared { name, name_match } => {
452            optional(ui, name, "any spell; or its name");
453            contains(ui, name_match);
454        }
455        Condition::Flag { name, active } => {
456            ui.add(
457                egui::TextEdit::singleline(name)
458                    .hint_text("its name")
459                    .desired_width(120.0),
460            );
461            ui.checkbox(active, "set");
462        }
463        Condition::Room { name, name_match } => named(ui, name, name_match, &[]),
464        Condition::Stance { stance } => pick(ui, stance, &STANCES, salt("stance")),
465        Condition::All { .. } | Condition::Any { .. } | Condition::Not { .. } => {}
466        beyond => fields_beyond(ui, beyond, id),
467    }
468}

470/// The fields of a test of Hydra's own: the encumbrance, a resource, the
471/// target, the room's creatures.
472fn fields_beyond(ui: &mut egui::Ui, test: &mut Condition, id: (usize, usize)) {
473    let salt = |what: &str| egui::Id::new(("highlight-field", id, what.to_owned()));
474    match test {
475        Condition::CrtrStatus { id, active } => {
476            let statuses: Vec<&str> = StatusName::ALL.iter().map(|s| s.as_str()).collect();
477            pick(ui, id, &statuses, salt("status"));
478            ui.checkbox(active, "on");
479            ui.weak("on the target").on_hover_text(TARGET);
480        }
481        Condition::Encumbrance { cmp, percent } => {
482            cmp_ui(ui, cmp, salt("cmp"));
483            ui.add(egui::DragValue::new(percent).range(0..=100).suffix("%"));
484        }
485        Condition::Resource {
486            resource,
487            cmp,
488            value,
489        } => {
490            choose(
491                ui,
492                resource,
493                &[
494                    (ResourceKind::ShadowEssence, "shadow essence"),
495                    (ResourceKind::ResourceWeekly, "society resource this week"),
496                    (ResourceKind::ResourceTotal, "society resource in all"),
497                    (ResourceKind::CovertArts, "covert arts charges"),
498                ],
499                salt("resource"),
500            );
501            cmp_ui(ui, cmp, salt("cmp"));
502            ui.add(egui::DragValue::new(value).range(0..=1_000_000));
503        }
504        Condition::TargetFlag { flag, active } => {
505            let flags: Vec<&str> = Classification::ALL.iter().map(|c| c.wire_name()).collect();
506            pick(ui, flag, &flags, salt("flag"));
507            ui.checkbox(active, "set");
508            ui.weak("on the target").on_hover_text(TARGET);
509        }
510        Condition::TargetKind { kind } => {
511            choose(
512                ui,
513                kind,
514                &[
515                    (TargetKindName::Undead, "undead"),
516                    (TargetKindName::Living, "living"),
517                    (TargetKindName::Noncorporeal, "noncorporeal"),
518                ],
519                salt("kind"),
520            );
521            ui.weak("the target, by the bestiary").on_hover_text(TARGET);
522        }
523        Condition::TargetHealth { cmp, percent } => {
524            ui.weak("the target's health").on_hover_text(TARGET);
525            cmp_ui(ui, cmp, salt("cmp"));
526            ui.add(egui::DragValue::new(percent).range(0..=100).suffix("%"));
527        }
528        Condition::Creatures {
529            cmp,
530            count,
531            hostile,
532        } => {
533            cmp_ui(ui, cmp, salt("cmp"));
534            ui.add(egui::DragValue::new(count).range(0..=100));
535            ui.checkbox(hostile, "that can be attacked").on_hover_text(
536                "Off: every creature in the room, the friendly and the dead among them",
537            );
538        }
539        _ => {}
540    }
541}

658/// A menu of the names the character has had, those holding what is typed
659/// in `name` alone; one picked is `name`.
660pub(crate) fn pick_name(ui: &mut egui::Ui, name: &mut String, names: &[String]) {
661    if names.is_empty() {
662        return;
663    }
664    ui.menu_button("seen", |ui| {
665        let typed = name.trim().to_lowercase();
666        let mut shown = 0;
667        egui::ScrollArea::vertical()
668            .max_height(260.0)
669            .show(ui, |ui| {
670                for one in names {
671                    if !typed.is_empty() && !one.to_lowercase().contains(&typed) {
672                        continue;
673                    }
674                    shown += 1;
675                    if ui.button(one).clicked() {
676                        one.clone_into(name);
677                        ui.close();
678                    }
679                }
680                if shown == 0 {
681                    ui.weak("None seen holds that.");
682                }
683            });
684    })
685    .response
686    .on_hover_text("The names the character has had");
687}

689/// A drop-down of `choices` into `value`.
690fn choose<T: Copy + PartialEq>(
691    ui: &mut egui::Ui,
692    value: &mut T,
693    choices: &[(T, &str)],
694    id: egui::Id,
695) {
696    let shown = choices
697        .iter()
698        .find(|(choice, _)| choice == value)
699        .map_or("", |(_, name)| name);
700    let longest = choices
701        .iter()
702        .map(|(_, name)| name.len())
703        .max()
704        .unwrap_or(1);
705    #[expect(clippy::cast_precision_loss, reason = "a word's length")]
706    let width = 24.0 + 8.0 * longest as f32;
707    egui::ComboBox::from_id_salt(id)
708        .width(width)
709        .selected_text(shown)
710        .show_ui(ui, |ui| {
711            for (choice, name) in choices {
712                ui.selectable_value(value, *choice, *name);
713            }
714        });
715}
716
717/// A drop-down of `choices` into `value`, a word.
718fn pick(ui: &mut egui::Ui, value: &mut String, choices: &[&str], id: egui::Id) {
719    egui::ComboBox::from_id_salt(id)
720        .selected_text(value.clone())
721        .show_ui(ui, |ui| {
722            for choice in choices {
723                ui.selectable_value(value, (*choice).to_owned(), *choice);
724            }
725        });
726}

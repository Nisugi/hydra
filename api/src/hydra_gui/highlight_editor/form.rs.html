15/// What a banner or a notification says, as the form chooses it.
16#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
17enum Says {
18    #[default]
19    Nothing,
20    Line,
21    Words,
22}
23
24impl Says {
25    const ALL: [Self; 3] = [Self::Nothing, Self::Line, Self::Words];
26
27    fn name(self) -> &'static str {
28        match self {
29            Self::Nothing => "Nothing",
30            Self::Line => "The line",
31            Self::Words => "These words:",
32        }
33    }
34
35    /// `say` as the form shows it, and its words.
36    fn of(say: Option<&Say>) -> (Self, String) {
37        match say {
38            None | Some(Say::Line(false)) => (Self::Nothing, String::new()),
39            Some(Say::Line(true)) => (Self::Line, String::new()),
40            Some(Say::Words(words)) => (Self::Words, words.clone()),
41        }
42    }
43
44    /// As the file writes it, with `words`.
45    fn say(self, words: &str) -> Option<Say> {
46        match self {
47            Self::Nothing => None,
48            Self::Line => Some(Say::Line(true)),
49            Self::Words if words.trim().is_empty() => Some(Say::Line(true)),
50            Self::Words => Some(Say::Words(words.trim().to_owned())),
51        }
52    }
53}

61/// A flag or a status as the form edits it: its name, its seconds (none
62/// until cleared), or that it clears.
63#[derive(Clone, Debug, Default, PartialEq, Eq)]
64struct MarkDraft {
65    name: String,
66    seconds: u32,
67    clear: bool,
68}
69
70impl MarkDraft {
71    fn of(mark: Option<&Mark>) -> Self {
72        mark.map_or_else(Self::default, |mark| Self {
73            name: mark.name().to_owned(),
74            seconds: mark.seconds().unwrap_or(0),
75            clear: mark.clears(),
76        })
77    }
78
79    fn mark(&self) -> Option<Mark> {
80        let name = self.name.trim();
81        if name.is_empty() {
82            return None;
83        }
84        Some(if self.clear {
85            Mark::Full {
86                name: name.to_owned(),
87                seconds: None,
88                clear: true,
89            }
90        } else if self.seconds > 0 {
91            Mark::Full {
92                name: name.to_owned(),
93                seconds: Some(self.seconds),
94                clear: false,
95            }
96        } else {
97            Mark::Named(name.to_owned())
98        })
99    }
100
101    /// Its row: the name, then set for so long or until cleared, or cleared.
102    fn show(&mut self, ui: &mut egui::Ui, hint: &str) {
103        ui.horizontal(|ui| {
104            ui.add(
105                egui::TextEdit::singleline(&mut self.name)
106                    .hint_text(hint)
107                    .desired_width(140.0),
108            );
109            if self.name.trim().is_empty() {
110                return;
111            }
112            ui.checkbox(&mut self.clear, "clears it");
113            if !self.clear {
114                ui.label("for");
115                ui.add(
116                    egui::DragValue::new(&mut self.seconds)
117                        .range(0..=86_400)
118                        .suffix(" s"),
119                )
120                .on_hover_text("0: until cleared");
121            }
122        });
123    }
124}

154/// The width of a section's first column, its labels, so the sections'
155/// fields line up one under another.
156const LABELS: f32 = 104.0;
157
158/// A grid of the form's labels and fields, `id` its own. One outside the
159/// sections, `outside`, has its labels' column wider by a section's
160/// indent, so its fields line up with theirs.
161fn grid(ui: &mut egui::Ui, (id, outside): (&str, bool), rows: impl FnOnce(&mut egui::Ui)) {
162    let indent = if outside { ui.spacing().indent } else { 0.0 };
163    egui::Grid::new(("highlight-form", id))
164        .num_columns(2)
165        .min_col_width(LABELS + indent)
166        .spacing([12.0, 6.0])
167        .show(ui, rows);
168}

170/// `words` cut to `most` characters, `...` after one cut.
171fn clipped(words: &str, most: usize) -> String {
172    if words.chars().count() <= most {
173        return words.to_owned();
174    }
175    let cut: String = words.chars().take(most).collect();
176    format!("{}...", cut.trim_end())
177}

321    /// The form as a highlight.
322    pub(crate) fn rule(&self) -> Rule {
323        let pattern = (!self.pattern.is_empty()).then(|| self.pattern.clone());
324        let category = self.category.trim();
325        Rule {
326            category: if category.is_empty() {
327                "General"
328            } else {
329                category
330            }
331            .to_owned(),
332            text: if self.regex { None } else { pattern.clone() },
333            regex: if self.regex { pattern } else { None },
334            case_sensitive: self.case_sensitive,
335            whole_word: self.whole_word && !self.regex,
336            color: self.color.map(hex),
337            background: self.background.map(hex),
338            line: self.line,
339            squelch: self.squelch,
340            sound: (!self.sound.trim().is_empty()).then(|| self.sound.trim().to_owned()),
341            volume: (self.volume != 100).then_some(self.volume),
342            rumble: (!self.rumble.trim().is_empty()).then(|| self.rumble.trim().to_owned()),
343            picture: written(&self.picture),
344            alert: self.alert.0.say(&self.alert.1),
345            notify: self.notify.0.say(&self.notify.1),
346            push: self.push.0.say(&self.push.1),
347            speak: self.speak.0.say(&self.speak.1),
348            cooldown: ((self.cooldown - COOLDOWN).abs() > f64::EPSILON).then_some(self.cooldown),
349            characters: self
350                .characters
351                .split(',')
352                .map(str::trim)
353                .filter(|name| !name.is_empty())
354                .map(str::to_owned)
355                .collect(),
356            stream: written(&self.stream),
357            timer: written(&self.timer),
358            seconds: (self.seconds > 0).then_some(self.seconds),
359            caught: written(&self.caught),
360            times: (self.times != 1).then_some(self.times),
361            down: written(&self.down),
362            list: (self.list != List::Custom).then(|| self.list.name().to_owned()),
363            origin: self.origin.clone(),
364            when: self.when.clone(),
365            rearm: ((self.rearm - REARM).abs() > f64::EPSILON).then_some(self.rearm),
366            substitute: (!self.substitute.is_empty()).then(|| self.substitute.clone()),
367            // A line is moved by a stream rule alone.
368            redirect: None,
369            copy: false,
370            flag: self.flag.mark(),
371            status: self.status.mark(),
372            send: self.sends(),
373            approved: self.approved.clone(),
374            off: self.off,
375        }
376    }

378    /// Draw the form into `ui`; `chosen` is the name the highlight is kept
379    /// by, `None` for a new one, and `categories` those there are. What it
380    /// asks.
381    ///
382    /// Its name and category, then its parts in four [`Section`]s, each
383    /// folded with a line saying what is in it. A section opens as a
384    /// trigger is chosen when the trigger uses it, Find it always; each
385    /// trigger keeps its own open or folded, so another chosen opens as
386    /// its own parts say.
387    pub(crate) fn show(
388        &mut self,
389        ui: &mut egui::Ui,
390        chosen: Option<&str>,
391        categories: &[String],
392    ) -> Vec<Asked> {
393        let mut asked = Vec::new();
394        ui.heading(chosen.map_or_else(|| "A new trigger".to_owned(), str::to_owned));
395        if let Some(origin) = &self.origin {
396            crate::form::hint(ui, format!("Brought by an import: {origin}"));
397        }
398        ui.label(egui::RichText::new(self.sentence()).italics());
399        ui.add_space(4.0);
400        grid(ui, ("what", true), |ui| self.what(ui, categories));
401        let whose = chosen.unwrap_or_default();
402        for section in Section::ALL {
403            let header = self.header(ui, section);
404            let open = section == Section::Find
405                || self.uses(section)
406                || self.kind.section() == Some(section);
407            egui::CollapsingHeader::new(header)
408                .id_salt(("trigger-section", section.name(), whose))
409                .default_open(open)
410                .open(self.fresh.then_some(open))
411                .show(ui, |ui| {
412                    grid(ui, (section.name(), false), |ui| match section {
413                        Section::Find => self.find(ui),
414                        Section::Show => self.show_it(ui),
415                        Section::Tell => self.tell(ui),
416                        Section::Act => self.act(ui),
417                    });
418                    if section == Section::Find {
419                        self.when_ui(ui, whose);
420                    }
421                });
422        }
423        if self.limited() {
424            grid(ui, ("cooldown", true), |ui| {
425                ui.label("Cooldown");
426                ui.add(
427                    egui::DragValue::new(&mut self.cooldown)
428                        .range(0.0..=3600.0)
429                        .speed(0.5)
430                        .suffix(" s"),
431                )
432                .on_hover_text(
433                    "How long before its sound, banner, notification, speech, rumble, \
434                     picture, timer and sends come again",
435                );
436                ui.end_row();
437            });
438        }
439        ui.add_space(8.0);
440        ui.separator();
441        ui.horizontal(|ui| {
442            if ui.button("Save").clicked() {
443                asked.push(Asked::Save {
444                    was: chosen.map(str::to_owned),
445                    name: self.name.trim().to_owned(),
446                    rule: Box::new(self.rule()),
447                });
448            }
449            if let Some(chosen) = chosen
450                && ui.button("Remove").clicked()
451            {
452                asked.push(Asked::Remove(chosen.to_owned()));
453            }
454        });
455        self.fresh = false;
456        asked
457    }

459    /// A section's header: its name, then what is in it, the colours worn
460    /// shown in them. Read aloud, the same words with a space between.
461    fn header(&self, ui: &egui::Ui, section: Section) -> egui::text::LayoutJob {
462        let font = egui::TextStyle::Body.resolve(ui.style());
463        let small = egui::TextStyle::Small.resolve(ui.style());
464        let strong = egui::TextFormat::simple(font.clone(), ui.visuals().strong_text_color());
465        // The summary beside the name in the small print, so the name
466        // stands and the summary is read as a note.
467        let plain = |color| egui::TextFormat::simple(small.clone(), color);
468        let weak = ui.visuals().weak_text_color();
469        let mut job = egui::text::LayoutJob::default();
470        job.append(section.name(), 0.0, strong);
471        job.append(" ", 12.0, plain(weak));
472        let parts = self.parts(section);
473        if parts.is_empty() {
474            job.append("nothing", 0.0, plain(weak));
475            return job;
476        }
477        if section == Section::Show && (self.color.is_some() || self.background.is_some()) {
478            let sample = if self.regex || self.pattern.trim().is_empty() {
479                "Aa".to_owned()
480            } else {
481                clipped(
482                    self.pattern.split('|').next().unwrap_or_default().trim(),
483                    16,
484                )
485            };
486            let rgb = |rgb: Option<Rgb>| rgb.map(|[r, g, b]| egui::Color32::from_rgb(r, g, b));
487            job.append(
488                &sample,
489                0.0,
490                egui::TextFormat {
491                    color: rgb(self.color).unwrap_or_else(|| ui.visuals().text_color()),
492                    background: rgb(self.background).unwrap_or_default(),
493                    ..plain(weak)
494                },
495            );
496            job.append(" ", 0.0, plain(weak));
497        }
498        job.append(&parts.join(" · "), 0.0, plain(weak));
499        job
500    }

502    /// What `section` holds, in a few words each; none for nothing.
503    fn parts(&self, section: Section) -> Vec<String> {
504        let mut parts = Vec::new();
505        let mut put = |on: bool, words: String| {
506            if on {
507                parts.push(words);
508            }
509        };
510        let pattern = self.pattern.trim();
511        match section {
512            Section::Find => {
513                let kind = if self.regex { "pattern" } else { "words" };
514                put(
515                    !pattern.is_empty(),
516                    format!("{kind} {}", clipped(pattern, 40)),
517                );
518                put(
519                    !pattern.is_empty() && self.case_sensitive,
520                    "capitals count".to_owned(),
521                );
522                put(
523                    !pattern.is_empty() && self.whole_word && !self.regex,
524                    "whole words".to_owned(),
525                );
526                put(
527                    pattern.is_empty() && self.when.is_some(),
528                    "its condition alone".to_owned(),
529                );
530                put(
531                    !pattern.is_empty() && self.when.is_some(),
532                    "while its condition holds".to_owned(),
533                );
534                let characters = self.characters.trim();
535                put(!characters.is_empty(), format!("for {characters}"));
536                let stream = self.stream.trim();
537                put(!stream.is_empty(), format!("in the {stream} stream"));
538            }
539            Section::Show => {
540                let coloured = self.color.is_some() || self.background.is_some();
541                let what = if self.line {
542                    "whole line coloured"
543                } else {
544                    "coloured"
545                };
546                put(coloured, what.to_owned());
547                put(self.squelch, "hidden".to_owned());
548                put(
549                    !self.substitute.is_empty(),
550                    format!("written as {}", clipped(&self.substitute, 30)),
551                );
552            }
553            Section::Tell => {
554                let sound = self.sound.trim();
555                let volume = if self.volume == 100 {
556                    String::new()
557                } else {
558                    format!(" at {}%", self.volume)
559                };
560                put(!sound.is_empty(), format!("{sound}{volume}"));
561                put(self.alert.0 != Says::Nothing, "banner".to_owned());
562                put(self.notify.0 != Says::Nothing, "notification".to_owned());
563                put(self.push.0 != Says::Nothing, "push to phone".to_owned());
564                put(self.speak.0 != Says::Nothing, "speech".to_owned());
565                let rumble = self.rumble.trim();
566                put(!rumble.is_empty(), format!("rumble {rumble}"));
567                put(!self.picture.trim().is_empty(), "picture".to_owned());
568            }
569            Section::Act => {
570                let commands: Vec<&str> = self
571                    .send
572                    .lines()
573                    .map(str::trim)
574                    .filter(|command| !command.is_empty())
575                    .collect();
576                match commands.as_slice() {
577                    [] => {}
578                    [one] => put(true, format!("sends {}", clipped(one, 30))),
579                    many => put(true, format!("sends {} commands", many.len())),
580                }
581                let timer = self.timer.trim();
582                put(!timer.is_empty(), format!("timer {timer}"));
583                for (kind, mark) in [("flag", &self.flag), ("status", &self.status)] {
584                    let name = mark.name.trim();
585                    let verb = if mark.clear { "clears " } else { "" };
586                    put(!name.is_empty(), format!("{verb}{kind} {name}"));
587                }
588            }
589        }
590        parts
591    }

887    /// Its send as the file writes it: a command a line, `\r` between.
888    fn sends(&self) -> Option<String> {

890            .send
891            .lines()

898    /// A line to try the trigger on, and what it makes of it as it is
899    /// typed: the line as it would show, what its pattern caught, and what
900    /// it would do.
901    fn try_it(&mut self, ui: &mut egui::Ui) {
902        ui.add(
903            egui::TextEdit::singleline(&mut self.test)
904                .hint_text("Paste a line from the game")
905                .desired_width(f32::INFINITY),
906        );
907        if self.test.is_empty() {
908            return;
909        }
910        let mut rule = self.rule();
911        rule.characters.clear();
912        // Tried on the line as if it reached its stream.
913        rule.stream = None;
914        rule.off = false;
915        if let Err(why) = Matcher::check_rule("this", &rule) {
916            ui.colored_label(ui.visuals().warn_fg_color, why);
917            return;
918        }
919        if self.pattern.trim().is_empty() {
920            crate::form::hint(
921                ui,
922                "A condition alone acts as its condition comes to hold, not on a line.",
923            );
924            return;
925        }
926        // Tried as if its condition held: the window has no character here.
927        if rule.when.take().is_some() {
928            crate::form::hint(
929                ui,
930                "It acts only while its condition holds; tried here as if it did.",
931            );
932        }
933        let caught = hydra_ui::highlight::caught(&rule, &self.test)
934            .ok()
935            .flatten();
936        // Tried as if approved, so what it would send shows.
937        rule.approved.clone_from(&rule.send);
938        let mut file = HighlightFile::default();
939        file.rules.insert("this".to_owned(), rule);
940        let (matcher, _) = Matcher::new(&file, "");
941        let line = [StyledRun::plain(&self.test)];
942        let response = matcher.respond(&line);
943        let job = crate::text::job(&response.runs, ui.style(), &crate::theme::palette(ui.ctx()));
944        ui.label(job);
945        let mut said = Vec::new();
946        match &caught {
947            None => said.push("It does not match this line.".to_owned()),
948            Some(groups) if !groups.is_empty() => said.push(format!(
949                "Caught: {}",
950                groups
951                    .iter()
952                    .enumerate()
953                    .map(|(at, group)| match group {
954                        Some(words) => format!("${} \"{words}\"", at + 1),
955                        None => format!("${} nothing", at + 1),
956                    })
957                    .collect::<Vec<_>>()
958                    .join(" \u{b7} ")
959            )),
960            Some(_) => {}
961        }
962        said.extend(done(&response));
963        for line in said {
964            ui.label(line);
965        }
966    }
967}

969/// What `response` says a trigger does to a line and beyond, a line each:
970/// hidden, its alerts, its timer, flag and status, its sends, where the
971/// line goes.
972fn done(response: &hydra_ui::highlight::Response) -> Vec<String> {
973    let mut said = Vec::new();
974    if response.squelched {
975        said.push("The line is hidden.".to_owned());
976    }
977    for fired in &response.fired {
978        if let Some(alert) = &fired.alert {
979            said.push(format!("Banner: {alert}"));
980        }
981        if let Some(sound) = &fired.sound {
982            said.push(format!("Sound: {sound}"));
983        }
984        if let Some(notify) = &fired.notify {
985            said.push(format!("Notification: {notify}"));
986        }
987        if let Some(push) = &fired.push {
988            said.push(format!("Push to phone: {push}"));
989        }
990        if let Some(speak) = &fired.speak {
991            said.push(format!("Speech: {speak}"));
992        }
993        if let Some(rumble) = &fired.rumble {
994            said.push(format!("Rumble: {rumble}"));
995        }
996        if let Some(picture) = &fired.picture {
997            said.push(format!("Picture: {picture}"));
998        }
999        if let Some(timing) = &fired.timer {
1000            said.push(match timing.seconds {
1001                Some(seconds) => format!("Timer {}: {seconds} s", timing.name),
1002                None => format!("Timer {}: ended", timing.name),
1003            });
1004        }
1005        for (what, mark) in [("Flag", &fired.flag), ("Status", &fired.status)] {
1006            let Some(mark) = mark else {
1007                continue;
1008            };
1009            said.push(match (mark.clears(), mark.seconds()) {
1010                (true, _) => format!("{what} {}: cleared", mark.name()),
1011                (false, Some(seconds)) => {
1012                    format!("{what} {}: set for {seconds} s", mark.name())
1013                }
1014                (false, None) => format!("{what} {}: set", mark.name()),
1015            });
1016        }
1017        for command in &fired.send {
1018            said.push(format!("Sends: {command}"));
1019        }
1020    }
1021    said
1022}

1027    (!words.is_empty()).then(|| words.to_owned())

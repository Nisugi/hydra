15/// What a banner or a notification says, as the form chooses it.
16#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
17enum Says {
18    #[default]
19    Nothing,
20    Line,
21    Words,
22}
23
24impl Says {
25    const ALL: [Self; 3] = [Self::Nothing, Self::Line, Self::Words];
26
27    fn name(self) -> &'static str {
28        match self {
29            Self::Nothing => "Nothing",
30            Self::Line => "The line",
31            Self::Words => "These words:",
32        }
33    }
34
35    /// `say` as the form shows it, and its words.
36    fn of(say: Option<&Say>) -> (Self, String) {
37        match say {
38            None | Some(Say::Line(false)) => (Self::Nothing, String::new()),
39            Some(Say::Line(true)) => (Self::Line, String::new()),
40            Some(Say::Words(words)) => (Self::Words, words.clone()),
41        }
42    }
43
44    /// As the file writes it, with `words`.
45    fn say(self, words: &str) -> Option<Say> {
46        match self {
47            Self::Nothing => None,
48            Self::Line => Some(Say::Line(true)),
49            Self::Words if words.trim().is_empty() => Some(Say::Line(true)),
50            Self::Words => Some(Say::Words(words.trim().to_owned())),
51        }
52    }
53}

61/// A flag or a status as the form edits it: its name, its seconds (none
62/// until cleared), or that it clears.
63#[derive(Clone, Debug, Default, PartialEq, Eq)]
64struct MarkDraft {
65    name: String,
66    seconds: u32,
67    clear: bool,
68}
69
70impl MarkDraft {
71    fn of(mark: Option<&Mark>) -> Self {
72        mark.map_or_else(Self::default, |mark| Self {
73            name: mark.name().to_owned(),
74            seconds: mark.seconds().unwrap_or(0),
75            clear: mark.clears(),
76        })
77    }
78
79    fn mark(&self) -> Option<Mark> {
80        let name = self.name.trim();
81        if name.is_empty() {
82            return None;
83        }
84        Some(if self.clear {
85            Mark::Full {
86                name: name.to_owned(),
87                seconds: None,
88                clear: true,
89            }
90        } else if self.seconds > 0 {
91            Mark::Full {
92                name: name.to_owned(),
93                seconds: Some(self.seconds),
94                clear: false,
95            }
96        } else {
97            Mark::Named(name.to_owned())
98        })
99    }
100
101    /// Its row: the name, then set for so long or until cleared, or cleared.
102    fn show(&mut self, ui: &mut egui::Ui, hint: &str) {
103        ui.horizontal(|ui| {
104            ui.add(
105                egui::TextEdit::singleline(&mut self.name)
106                    .hint_text(hint)
107                    .desired_width(140.0),
108            );
109            if self.name.trim().is_empty() {
110                return;
111            }
112            ui.checkbox(&mut self.clear, "clears it");
113            if !self.clear {
114                ui.label("for");
115                ui.add(
116                    egui::DragValue::new(&mut self.seconds)
117                        .range(0..=86_400)
118                        .suffix(" s"),
119                )
120                .on_hover_text("0: until cleared");
121            }
122        });
123    }
124}

154/// The width of a section's first column, its labels, so the sections'
155/// fields line up one under another.
156const LABELS: f32 = 104.0;
157
158/// A grid of the form's labels and fields, `id` its own. One outside the
159/// sections, `outside`, has its labels' column wider by a section's
160/// indent, so its fields line up with theirs.
161fn grid(ui: &mut egui::Ui, (id, outside): (&str, bool), rows: impl FnOnce(&mut egui::Ui)) {
162    let indent = if outside { ui.spacing().indent } else { 0.0 };
163    egui::Grid::new(("highlight-form", id))
164        .num_columns(2)
165        .min_col_width(LABELS + indent)
166        .spacing([12.0, 6.0])
167        .show(ui, rows);
168}

170/// `words` cut to `most` characters, `...` after one cut.
171fn clipped(words: &str, most: usize) -> String {
172    if words.chars().count() <= most {
173        return words.to_owned();
174    }
175    let cut: String = words.chars().take(most).collect();
176    format!("{}...", cut.trim_end())
177}

319    /// The form as a highlight.
320    pub(crate) fn rule(&self) -> Rule {
321        let pattern = (!self.pattern.is_empty()).then(|| self.pattern.clone());
322        let category = self.category.trim();
323        Rule {
324            category: if category.is_empty() {
325                "General"
326            } else {
327                category
328            }
329            .to_owned(),
330            text: if self.regex { None } else { pattern.clone() },
331            regex: if self.regex { pattern } else { None },
332            case_sensitive: self.case_sensitive,
333            whole_word: self.whole_word && !self.regex,
334            color: self.color.map(hex),
335            background: self.background.map(hex),
336            line: self.line,
337            squelch: self.squelch,
338            sound: (!self.sound.trim().is_empty()).then(|| self.sound.trim().to_owned()),
339            volume: (self.volume != 100).then_some(self.volume),
340            rumble: (!self.rumble.trim().is_empty()).then(|| self.rumble.trim().to_owned()),
341            picture: written(&self.picture),
342            alert: self.alert.0.say(&self.alert.1),
343            notify: self.notify.0.say(&self.notify.1),
344            speak: self.speak.0.say(&self.speak.1),
345            cooldown: ((self.cooldown - COOLDOWN).abs() > f64::EPSILON).then_some(self.cooldown),
346            characters: self
347                .characters
348                .split(',')
349                .map(str::trim)
350                .filter(|name| !name.is_empty())
351                .map(str::to_owned)
352                .collect(),
353            timer: written(&self.timer),
354            seconds: (self.seconds > 0).then_some(self.seconds),
355            caught: written(&self.caught),
356            times: (self.times != 1).then_some(self.times),
357            down: written(&self.down),
358            list: (self.list != List::Custom).then(|| self.list.name().to_owned()),
359            origin: self.origin.clone(),
360            when: self.when.clone(),
361            rearm: ((self.rearm - REARM).abs() > f64::EPSILON).then_some(self.rearm),
362            substitute: (!self.substitute.is_empty()).then(|| self.substitute.clone()),
363            redirect: written(&self.redirect),
364            copy: self.copy && !self.redirect.trim().is_empty(),
365            flag: self.flag.mark(),
366            status: self.status.mark(),
367            send: self.sends(),
368            approved: self.approved.clone(),
369            off: self.off,
370        }
371    }

373    /// Draw the form into `ui`; `chosen` is the name the highlight is kept
374    /// by, `None` for a new one, and `categories` those there are. What it
375    /// asks.
376    ///
377    /// Its name and category, then its parts in four [`Section`]s, each
378    /// folded with a line saying what is in it. A section opens as a
379    /// trigger is chosen when the trigger uses it, Find it always; each
380    /// trigger keeps its own open or folded, so another chosen opens as
381    /// its own parts say.
382    pub(crate) fn show(
383        &mut self,
384        ui: &mut egui::Ui,
385        chosen: Option<&str>,
386        categories: &[String],
387    ) -> Vec<Asked> {
388        let mut asked = Vec::new();
389        ui.heading(chosen.map_or_else(|| "A new trigger".to_owned(), str::to_owned));
390        if let Some(origin) = &self.origin {
391            ui.weak(format!("Brought by an import: {origin}"));
392        }
393        ui.label(self.sentence());
394        ui.add_space(4.0);
395        grid(ui, ("what", true), |ui| self.what(ui, categories));
396        let whose = chosen.unwrap_or_default();
397        for section in Section::ALL {
398            let header = self.header(ui, section);
399            let open = section == Section::Find
400                || self.uses(section)
401                || self.kind.section() == Some(section);
402            egui::CollapsingHeader::new(header)
403                .id_salt(("trigger-section", section.name(), whose))
404                .default_open(open)
405                .open(self.fresh.then_some(open))
406                .show(ui, |ui| {
407                    grid(ui, (section.name(), false), |ui| match section {
408                        Section::Find => self.find(ui),
409                        Section::Show => self.show_it(ui),
410                        Section::Tell => self.tell(ui),
411                        Section::Act => self.act(ui),
412                    });
413                    if section == Section::Find {
414                        self.when_ui(ui, whose);
415                    }
416                });
417        }
418        if self.limited() {
419            grid(ui, ("cooldown", true), |ui| {
420                ui.label("Cooldown");
421                ui.add(
422                    egui::DragValue::new(&mut self.cooldown)
423                        .range(0.0..=3600.0)
424                        .speed(0.5)
425                        .suffix(" s"),
426                )
427                .on_hover_text(
428                    "How long before its sound, banner, notification, speech, rumble, \
429                     picture, timer and sends come again",
430                );
431                ui.end_row();
432            });
433        }
434        ui.add_space(6.0);
435        ui.horizontal(|ui| {
436            if ui.button("Save").clicked() {
437                asked.push(Asked::Save {
438                    was: chosen.map(str::to_owned),
439                    name: self.name.trim().to_owned(),
440                    rule: Box::new(self.rule()),
441                });
442            }
443            if let Some(chosen) = chosen
444                && ui.button("Remove").clicked()
445            {
446                asked.push(Asked::Remove(chosen.to_owned()));
447            }
448        });
449        self.fresh = false;
450        asked
451    }

453    /// A section's header: its name, then what is in it, the colours worn
454    /// shown in them. Read aloud, the same words with a space between.
455    fn header(&self, ui: &egui::Ui, section: Section) -> egui::text::LayoutJob {
456        let font = egui::TextStyle::Body.resolve(ui.style());
457        let plain = |color| egui::TextFormat::simple(font.clone(), color);
458        let weak = ui.visuals().weak_text_color();
459        let mut job = egui::text::LayoutJob::default();
460        job.append(section.name(), 0.0, plain(ui.visuals().strong_text_color()));
461        job.append(" ", 12.0, plain(weak));
462        let parts = self.parts(section);
463        if parts.is_empty() {
464            job.append("nothing", 0.0, plain(weak));
465            return job;
466        }
467        if section == Section::Show && (self.color.is_some() || self.background.is_some()) {
468            let sample = if self.regex || self.pattern.trim().is_empty() {
469                "Aa".to_owned()
470            } else {
471                clipped(
472                    self.pattern.split('|').next().unwrap_or_default().trim(),
473                    16,
474                )
475            };
476            let rgb = |rgb: Option<Rgb>| rgb.map(|[r, g, b]| egui::Color32::from_rgb(r, g, b));
477            job.append(
478                &sample,
479                0.0,
480                egui::TextFormat {
481                    color: rgb(self.color).unwrap_or_else(|| ui.visuals().text_color()),
482                    background: rgb(self.background).unwrap_or_default(),
483                    ..plain(weak)
484                },
485            );
486            job.append(" ", 0.0, plain(weak));
487        }
488        job.append(&parts.join(" · "), 0.0, plain(weak));
489        job
490    }

492    /// What `section` holds, in a few words each; none for nothing.
493    fn parts(&self, section: Section) -> Vec<String> {
494        let mut parts = Vec::new();
495        let mut put = |on: bool, words: String| {
496            if on {
497                parts.push(words);
498            }
499        };
500        let pattern = self.pattern.trim();
501        match section {
502            Section::Find => {
503                let kind = if self.regex { "pattern" } else { "words" };
504                put(
505                    !pattern.is_empty(),
506                    format!("{kind} {}", clipped(pattern, 40)),
507                );
508                put(
509                    !pattern.is_empty() && self.case_sensitive,
510                    "capitals count".to_owned(),
511                );
512                put(
513                    !pattern.is_empty() && self.whole_word && !self.regex,
514                    "whole words".to_owned(),
515                );
516                put(
517                    pattern.is_empty() && self.when.is_some(),
518                    "its condition alone".to_owned(),
519                );
520                put(
521                    !pattern.is_empty() && self.when.is_some(),
522                    "while its condition holds".to_owned(),
523                );
524                let characters = self.characters.trim();
525                put(!characters.is_empty(), format!("for {characters}"));
526            }
527            Section::Show => {
528                let coloured = self.color.is_some() || self.background.is_some();
529                let what = if self.line {
530                    "whole line coloured"
531                } else {
532                    "coloured"
533                };
534                put(coloured, what.to_owned());
535                put(self.squelch, "hidden".to_owned());
536                put(
537                    !self.substitute.is_empty(),
538                    format!("written as {}", clipped(&self.substitute, 30)),
539                );
540                let redirect = self.redirect.trim();
541                let kept = if self.copy { ", kept here too" } else { "" };
542                put(!redirect.is_empty(), format!("moved to {redirect}{kept}"));
543            }
544            Section::Tell => {
545                let sound = self.sound.trim();
546                let volume = if self.volume == 100 {
547                    String::new()
548                } else {
549                    format!(" at {}%", self.volume)
550                };
551                put(!sound.is_empty(), format!("{sound}{volume}"));
552                put(self.alert.0 != Says::Nothing, "banner".to_owned());
553                put(self.notify.0 != Says::Nothing, "notification".to_owned());
554                put(self.speak.0 != Says::Nothing, "speech".to_owned());
555                let rumble = self.rumble.trim();
556                put(!rumble.is_empty(), format!("rumble {rumble}"));
557                put(!self.picture.trim().is_empty(), "picture".to_owned());
558            }
559            Section::Act => {
560                let commands: Vec<&str> = self
561                    .send
562                    .lines()
563                    .map(str::trim)
564                    .filter(|command| !command.is_empty())
565                    .collect();
566                match commands.as_slice() {
567                    [] => {}
568                    [one] => put(true, format!("sends {}", clipped(one, 30))),
569                    many => put(true, format!("sends {} commands", many.len())),
570                }
571                let timer = self.timer.trim();
572                put(!timer.is_empty(), format!("timer {timer}"));
573                for (kind, mark) in [("flag", &self.flag), ("status", &self.status)] {
574                    let name = mark.name.trim();
575                    let verb = if mark.clear { "clears " } else { "" };
576                    put(!name.is_empty(), format!("{verb}{kind} {name}"));
577                }
578            }
579        }
580        parts
581    }

614    /// What it looks for, a line to try it on, and whose it is.

620            ui.radio_value(&mut self.regex, true, "A pattern")

622        });
623        ui.end_row();
624        ui.label("");
625        ui.add(
626            egui::TextEdit::singleline(&mut self.pattern)
627                .hint_text(self.kind.hint(self.regex))
628                .desired_width(f32::INFINITY),
629        );
630        ui.end_row();
631        ui.label("");
632        ui.horizontal(|ui| {
633            ui.checkbox(&mut self.case_sensitive, "Capitals count");
634            ui.add_enabled(
635                !self.regex,
636                egui::Checkbox::new(&mut self.whole_word, "Whole words only"),
637            );
638        });
639        ui.end_row();
640        // Level with its field, not halfway down what it says.
641        ui.with_layout(egui::Layout::top_down(egui::Align::Min), |ui| {
642            ui.label("Try it");
643        });
644        ui.vertical(|ui| self.try_it(ui));
645        ui.end_row();
646        ui.label("Characters");
647        ui.add(
648            egui::TextEdit::singleline(&mut self.characters)
649                .hint_text("everyone's; or names, with commas"),
650        );
651        ui.end_row();

864    /// Its send as the file writes it: a command a line, `\r` between.

868            .lines()

875    /// A line to try the trigger on, and what it makes of it as it is
876    /// typed: the line as it would show, what its pattern caught, and what
877    /// it would do.
878    fn try_it(&mut self, ui: &mut egui::Ui) {
879        ui.add(
880            egui::TextEdit::singleline(&mut self.test)
881                .hint_text("Paste a line from the game")
882                .desired_width(f32::INFINITY),
883        );
884        if self.test.is_empty() {
885            return;
886        }
887        let mut rule = self.rule();
888        rule.characters.clear();
889        rule.off = false;
890        if let Err(why) = Matcher::check_rule("this", &rule) {
891            ui.colored_label(ui.visuals().warn_fg_color, why);
892            return;
893        }
894        if self.pattern.trim().is_empty() {
895            ui.weak("A condition alone acts as its condition comes to hold, not on a line.");
896            return;
897        }
898        // Tried as if its condition held: the window has no character here.
899        if rule.when.take().is_some() {
900            ui.weak("It acts only while its condition holds; tried here as if it did.");
901        }
902        let caught = hydra_ui::highlight::caught(&rule, &self.test)
903            .ok()
904            .flatten();
905        // Tried as if approved, so what it would send shows.
906        rule.approved.clone_from(&rule.send);
907        let mut file = HighlightFile::default();
908        file.rules.insert("this".to_owned(), rule);
909        let (matcher, _) = Matcher::new(&file, "");
910        let line = [StyledRun::plain(&self.test)];
911        let response = matcher.respond(&line);
912        let job = crate::text::job(&response.runs, ui.style(), &crate::theme::palette(ui.ctx()));
913        ui.label(job);
914        let mut said = Vec::new();
915        match &caught {
916            None => said.push("It does not match this line.".to_owned()),
917            Some(groups) if !groups.is_empty() => said.push(format!(
918                "Caught: {}",
919                groups
920                    .iter()
921                    .enumerate()
922                    .map(|(at, group)| match group {
923                        Some(words) => format!("${} \"{words}\"", at + 1),
924                        None => format!("${} nothing", at + 1),
925                    })
926                    .collect::<Vec<_>>()
927                    .join(" \u{b7} ")
928            )),
929            Some(_) => {}
930        }
931        said.extend(done(&response));
932        for line in said {
933            ui.label(line);
934        }
935    }
936}

938/// What `response` says a trigger does to a line and beyond, a line each:
939/// hidden, its alerts, its timer, flag and status, its sends, where the
940/// line goes.
941fn done(response: &hydra_ui::highlight::Response) -> Vec<String> {
942    let mut said = Vec::new();
943    if response.squelched {
944        said.push("The line is hidden.".to_owned());
945    }
946    for fired in &response.fired {
947        if let Some(alert) = &fired.alert {
948            said.push(format!("Banner: {alert}"));
949        }
950        if let Some(sound) = &fired.sound {
951            said.push(format!("Sound: {sound}"));
952        }
953        if let Some(notify) = &fired.notify {
954            said.push(format!("Notification: {notify}"));
955        }
956        if let Some(speak) = &fired.speak {
957            said.push(format!("Speech: {speak}"));
958        }
959        if let Some(rumble) = &fired.rumble {
960            said.push(format!("Rumble: {rumble}"));
961        }
962        if let Some(picture) = &fired.picture {
963            said.push(format!("Picture: {picture}"));
964        }
965        if let Some(timing) = &fired.timer {
966            said.push(match timing.seconds {
967                Some(seconds) => format!("Timer {}: {seconds} s", timing.name),
968                None => format!("Timer {}: ended", timing.name),
969            });
970        }
971        for (what, mark) in [("Flag", &fired.flag), ("Status", &fired.status)] {
972            let Some(mark) = mark else {
973                continue;
974            };
975            said.push(match (mark.clears(), mark.seconds()) {
976                (true, _) => format!("{what} {}: cleared", mark.name()),
977                (false, Some(seconds)) => {
978                    format!("{what} {}: set for {seconds} s", mark.name())
979                }
980                (false, None) => format!("{what} {}: set", mark.name()),
981            });
982        }
983        for command in &fired.send {
984            said.push(format!("Sends: {command}"));
985        }
986    }
987    if let Some((stream, copy)) = &response.redirect {
988        said.push(if *copy {
989            format!("The line goes to the {stream} stream too.")
990        } else {
991            format!("The line goes to the {stream} stream instead.")
992        });
993    }
994    said
995}

1000    (!words.is_empty()).then(|| words.to_owned())

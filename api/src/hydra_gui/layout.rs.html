1//! A play window's layout: the windows it holds -- a
2//! standalone window around one widget, or a custom window of several held
3//! bare -- where each sits, and the grid their edges snap to. It is kept by
4//! the character's name, since a `SessionId` starts again at 0 every run
5//! and a name does not.
6//!
7//! The model is free rects, with the grid as a snap target and its pitch
8//! adjustable rather than fixed. A window's rect is kept from the play
9//! area's top left, so a
10//! play window moved across the screen keeps its layout; a custom window's
11//! cells are kept from its inside's top left (`custom.rs`).
12//!
13//! The layout remembers the play area it was arranged for. Drawn in a play
14//! window of another size, its windows are drawn at the share of their kept
15//! size that the area is of the arranged one, so windows that tiled it
16//! still tile it and none overlaps; what is kept is not changed, and the
17//! window sized back shows the layout as it was. *Fit to this window* in
18//! the Layout menu makes the size now the arranged one.
19//!
20//! Every window and every placed widget has an id of its own, given when it
21//! is placed and never derived from what it shows, so two of
22//! one kind can live side by side.

163/// One window in a play window: a standalone window or a custom window.
164#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
165pub(crate) struct Holder {
166    /// Its own id.
167    pub(crate) id: u32,
168    /// The main area or the drawer it lives in.
169    #[serde(default, skip_serializing_if = "Zone::is_main")]
170    pub(crate) zone: Zone,
171    /// Its rect from its zone's top left: x, y, width, height.
172    rect: [f32; 4],
173    /// What it holds.
174    pub(crate) holds: Holds,
175    /// Its title is hidden, except while the play
176    /// window is arranged, when every window shows what it is. The
177    /// window's own choice, not its widget's: its widgets draw the same in
178    /// it either way.
179    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
180    pub(crate) untitled: bool,
181    /// How solid it is, 0 to 1: at 0 it has no frame, background or title,
182    /// only its widgets drawn, and a click goes through it to what is
183    /// under it. `None`, solid. While the play window is arranged every
184    /// window is drawn whole, so a clear one can be found and moved.
185    #[serde(default, skip_serializing_if = "Option::is_none")]
186    pub(crate) background: Option<f32>,
187    /// Kept above every other window of the play window, the drawers'
188    /// among them.
189    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
190    pub(crate) on_top: bool,
191    /// Its title's colour, red, green and blue; none, the theme's.
192    #[serde(default, skip_serializing_if = "Option::is_none")]
193    pub(crate) title_color: Option<[u8; 3]>,
194    /// Its title's size, in points; none, the theme's.
195    #[serde(default, skip_serializing_if = "Option::is_none")]
196    pub(crate) title_size: Option<f32>,
197}

199impl Holder {
200    /// How solid it is drawn: `background`, held to 0 to 1; whole while
201    /// the play window is `arranging`.
202    pub(crate) fn solidity(&self, arranging: bool) -> f32 {
203        match self.background {
204            Some(background) if !arranging && background.is_finite() => background.clamp(0.0, 1.0),
205            _ => 1.0,
206        }
207    }
208}

210/// What a window holds: chrome follows from it, never from
211/// a setting of the widget's.
212#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
213#[serde(rename_all = "snake_case")]
214pub(crate) enum Holds {
215    /// One widget, framed and titled: a standalone window.
216    One(Placed),
217    /// Widgets held bare, with one frame for all: a custom window.
218    Custom(Custom),
219}

221/// A widget placed somewhere, with the id it was given there.
222#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
223pub(crate) struct Placed {
224    /// Its own id.
225    pub(crate) id: u32,
226    /// What it shows.
227    pub(crate) widget: Widget,
228}

230impl Holder {
231    /// Its title bar's words: its widget's name, or the custom window's own:
232    /// for a test, which finds a window as a player does. The play window
233    /// adds whose it is to a widget following another character
234    /// (`play/draw.rs`, `title`).
235    #[cfg(test)]
236    pub(crate) fn title(&self) -> std::borrow::Cow<'_, str> {
237        match &self.holds {
238            Holds::One(placed) => placed.widget.name(),
239            Holds::Custom(custom) => std::borrow::Cow::Borrowed(&custom.title),
240        }
241    }

243    /// Where it sits, from its zone's top left: the play area's, for a
244    /// window in the main area while no drawer pushes it.
245    pub(crate) fn rect(&self) -> Rect {
246        rect(self.rect)
247    }
248
249    /// Put it at `at`.
250    pub(crate) fn set(&mut self, at: Rect) {
251        self.rect = kept(at);
252    }

255impl Layout {
256    /// The first layout, for a play area `area` across: the story on the
257    /// left, and down the right the vitals, what the hands hold and the
258    /// clocks, the injuries, the room and Hydra's messages, their edges on the
259    /// grid. The vitals, the hands and the room are custom windows of single
260    /// widgets. The first three
261    /// are as tall as what they hold; in a short area they give way, so the
262    /// room and Hydra's messages keep the smallest a window can be.
263    pub(crate) fn fitted(area: Vec2) -> Self {
264        Self::fitted_as(area, false)
265    }
266
267    /// The same, but for its Room window: a custom window of the room's
268    /// parts, each its own widget, as the default was before the Room
269    /// widget; the tests of arranging cells arrange those.
270    #[cfg(test)]
271    pub(crate) fn with_room_parts(area: Vec2) -> Self {
272        Self::fitted_as(area, true)
273    }
274
275    /// [`Self::fitted`], its Room window the one Room widget, or with
276    /// `parts` the room's parts in a custom window.
277    fn fitted_as(area: Vec2, parts: bool) -> Self {
278        let on_grid = |value: f32| (value / GRID).round() * GRID;
279        let split = on_grid(area.x * 0.66).max(SMALLEST.x);
280        let side = (area.x - split).max(SMALLEST.x);
281        let fixed = [130.0, 110.0, 110.0];
282        let left = area.y - 2.0 * SMALLEST.y;
283        let scale = (left / fixed.iter().sum::<f32>()).clamp(0.0, 1.0);
284        let [vitals, loadout, injuries] = fixed.map(|height| (height * scale).floor());
285        let rest = area.y - vitals - loadout - injuries;
286        let room = on_grid(rest * 0.6).clamp(SMALLEST.y, (rest - SMALLEST.y).max(SMALLEST.y));
287        let hydra = (rest - room).max(SMALLEST.y);
288        let mut layout = Self {
289            version: VERSION,
290            grid: GRID,
291            next: 1,
292            holders: Vec::new(),
293            follows: BTreeMap::new(),
294            looks: BTreeMap::new(),
295            locked: false,
296            title: crate::play::titlebar::TitleBar::default(),
297            rooms: BTreeMap::new(),
298            lists: BTreeMap::new(),
299            lines: BTreeMap::new(),
300            dolls: BTreeMap::new(),
301            circles: None,
302            aims: BTreeMap::new(),
303            labels: BTreeMap::new(),
304            tabs: BTreeMap::new(),
305            rosters: BTreeMap::new(),
306            quicks: BTreeMap::new(),
307            hotbars: BTreeMap::new(),
308            timers: BTreeMap::new(),
309            minimaps: BTreeMap::new(),
310            faces: BTreeMap::new(),
311            indicators: BTreeMap::new(),
312            customs: BTreeMap::new(),
313            compasses: BTreeMap::new(),
314            drawers: Drawers::default(),
315            arranged: Some([area.x, area.y]),
316        };
317        let story = layout.place(Widget::Story);
318        layout.add(
319            Rect::from_min_size(pos2(0.0, 0.0), Vec2::new(split, area.y.max(SMALLEST.y))),
320            Holds::One(story),
321        );
322        let mut y = 0.0;
323        let mut next = |height: f32| {
324            let at = Rect::from_min_size(pos2(split, y), Vec2::new(side, height));
325            y += height;
326            at
327        };
328        let vitals_at = next(vitals);
329        layout.custom(
330            vitals_at,
331            "Vitals",
332            &[
333                &[Widget::Health],
334                &[Widget::Mana],
335                &[Widget::Stamina],
336                &[Widget::Spirit],
337            ],
338        );
339        let loadout_at = next(loadout);
340        layout.custom(
341            loadout_at,
342            "Loadout",
343            &[
344                &[Widget::RightHand],
345                &[Widget::LeftHand],
346                &[Widget::Roundtime, Widget::CastTime],
347            ],
348        );
349        let injuries_at = next(injuries);
350        let injuries = layout.place(Widget::Injuries);
351        layout.add(injuries_at, Holds::One(injuries));
352        let room_at = next(room);
353        if parts {
354            layout.custom(
355                room_at,
356                "Room",
357                &[
358                    &[Widget::RoomTitle],
359                    &[Widget::Creatures],
360                    &[Widget::Objects],
361                    &[Widget::Players],
362                    &[Widget::Exits],
363                ],
364            );
365        } else {
366            let room = layout.place(Widget::Room);
367            layout.add(room_at, Holds::One(room));
368        }
369        let hydra_at = next(hydra);
370        let hydra = layout.place(Widget::Hydra);
371        layout.add(hydra_at, Holds::One(hydra));
372        layout
373    }
374
375    /// `widget`, given an id of its own.
376    fn place(&mut self, widget: Widget) -> Placed {
377        let id = self.next;
378        self.next += 1;
379        Placed { id, widget }
380    }
381
382    /// A window holding `holds` at `at`, on top of the rest; its id.
383    fn add(&mut self, at: Rect, holds: Holds) -> u32 {
384        let id = self.next;
385        self.next += 1;
386        self.holders.push(Holder {
387            id,
388            zone: Zone::Main,
389            rect: kept(at),
390            holds,
391            untitled: false,
392            background: None,
393            on_top: false,
394            title_color: None,
395            title_size: None,
396        });
397        id
398    }
399
400    /// A custom window titled `title` at `at`, its widgets in `rows`.
401    fn custom(&mut self, at: Rect, title: &str, rows: &[&[Widget]]) -> u32 {
402        let rows: Vec<Vec<Placed>> = rows
403            .iter()
404            .map(|row| {
405                row.iter()
406                    .map(|widget| self.place(widget.clone()))
407                    .collect()
408            })
409            .collect();
410        let inside = (at.size() - CHROME).max(Vec2::ZERO);
411        self.add(at, Holds::Custom(Custom::rows(title, rows, inside)))
412    }
413
414    /// The window `id`, if it is here.
415    pub(crate) fn holder(&self, id: u32) -> Option<&Holder> {
416        self.holders.iter().find(|holder| holder.id == id)
417    }
418}

420/// A kept rect as a rect.
421fn rect([x, y, width, height]: [f32; 4]) -> Rect {
422    Rect::from_min_size(pos2(x, y), Vec2::new(width, height))
423}
424
425/// A rect as it is kept: x, y, width, height.
426fn kept(at: Rect) -> [f32; 4] {
427    [at.min.x, at.min.y, at.width(), at.height()]
428}

433impl Layout {
434    /// The play area it was arranged for, once known.
435    pub(crate) fn arranged(&self) -> Option<Vec2> {
436        self.arranged.map(|[x, y]| Vec2::new(x, y))
437    }
438
439    /// It is arranged for a play area `area` across, its windows kept as
440    /// they are: for a layout from a file that kept no area.
441    pub(crate) fn arranged_for(&mut self, area: Vec2) {
442        self.arranged = Some([area.x, area.y]);
443    }
444
445    /// Make the play area `area` across, which it is drawn in through
446    /// `zones`, the one it is arranged for: each main-area window kept
447    /// where it is drawn now, so nothing moves, and from now on it is drawn
448    /// at its kept size there.
449    pub(crate) fn fit_to(&mut self, area: Vec2, zones: &Zones) {
450        for holder in &mut self.holders {
451            if holder.zone != Zone::Main {
452                continue;
453            }
454            if let Some(drawn) = zones.fit(holder.rect(), Zone::Main) {
455                holder.set(zones.unsqueezed_by_drawers(drawn));
456            }
457        }
458        self.arranged = Some([area.x, area.y]);
459    }

1//! The performance log: while the hub's *Performance* is on, each live
2//! character's stages (`hydra_ui::perf`) every [`EVERY`], a row each, in
3//! `perf/<date>.csv` in the data folder, for diagnosing a slow window.
4//!
5//! A row says what happened since the character's last: how many game and
6//! Lich lines and typed lines, and each stage's count, median and 95th
7//! percentile over its samples since. So a burst -- a key held on `look` --
8//! is its own rows, not folded into all that came before.
9
10use std::io::Write as _;
11use std::path::Path;
12use std::sync::Arc;
13use std::time::{Duration, Instant};
14
15use hydra_session::State;
16use hydra_session::lich::View;
17
18use crate::seat::Seat;
19
20/// How often a row is written for each character.
21pub(crate) const EVERY: Duration = Duration::from_secs(10);
22
23/// When the log last wrote, while it writes.
24#[derive(Debug, Default)]
25pub(crate) struct PerfLog {
26    since: Option<Instant>,
27}
28
29impl PerfLog {
30    /// The log, on or off, at `now`: with it on, each of `seats`' rows once
31    /// [`EVERY`] has passed, in `data`. When the next is due, while it is on.
32    pub(crate) fn tick(
33        &mut self,
34        on: bool,
35        data: &Path,
36        seats: &[Arc<Seat>],
37        now: Instant,
38    ) -> Option<Duration> {
39        if !on {
40            if self.since.take().is_some() {
41                // Switched off: nothing more is kept for it.
42                for seat in seats {
43                    seat.story().perf.collect(false);
44                }
45            }
46            return None;
47        }
48        let Some(since) = self.since else {
49            // Switched on: what came before is not this log's, and from now
50            // every sample is kept for its rows.
51            for seat in seats {
52                let mut story = seat.story();
53                let (lich, view) = (story.lich_running(), story.view());
54                let _ = story.perf.log_row(lich, view_name(view));
55                story.perf.collect(true);
56            }
57            self.since = Some(now);
58            return Some(EVERY);
59        };
60        // A character seated since the log went on collects too.
61        for seat in seats {
62            seat.story().perf.collect(true);
63        }
64        let waited = now.saturating_duration_since(since);
65        if waited < EVERY {
66            return Some(EVERY.saturating_sub(waited));
67        }
68        self.since = Some(now);
69        let time = hydra_net::line_time();
70        let rows: Vec<String> = seats
71            .iter()
72            .filter(|seat| seat.lifecycle() != State::Closed)
73            .map(|seat| {
74                let mut story = seat.story();
75                let (lich, view) = (story.lich_running(), story.view());
76                let row = story.perf.log_row(lich, view_name(view));
77                format!("{time},{}:{},{row}", seat.game, seat.name)
78            })
79            .collect();
80        let (folder, day) = (data.join("perf"), hydra_net::date_dir());
81        let written = move || {
82            if let Err(why) = write(&folder, &day, &rows) {
83                eprintln!("[perf] not written: {why}");
84            }
85        };
86        // Off the window's thread; under test, at once.
87        if cfg!(test) {
88            written();
89        } else {
90            std::thread::spawn(move || hydra_mem::part(hydra_mem::Part::WindowThreads, written));
91        }
92        Some(EVERY)
93    }
94}

96/// A view as the log names it.
97const fn view_name(view: View) -> &'static str {
98    match view {
99        View::Lich => "lich",
100        View::Game => "game",
101    }
102}
103
104/// Append `rows` to `<folder>/<day>.csv`, its header first when it is new.
105fn write(folder: &Path, day: &str, rows: &[String]) -> std::io::Result<()> {
106    if rows.is_empty() {
107        return Ok(());
108    }
109    std::fs::create_dir_all(folder)?;
110    let path = folder.join(format!("{day}.csv"));
111    let new = !path.exists();
112    let mut file = std::fs::OpenOptions::new()
113        .create(true)
114        .append(true)
115        .open(path)?;
116    let mut text = String::new();
117    if new {
118        text.push_str("time,character,");
119        text.push_str(&hydra_ui::perf::Perf::log_header());
120        text.push('\n');
121    }
122    for row in rows {
123        text.push_str(row);
124        text.push('\n');
125    }
126    file.write_all(text.as_bytes())
127}

129#[cfg(test)]
130mod tests {
131    use super::*;
132
133    /// Switched on, nothing before is logged; a row each live character
134    /// once its time has come, under a header written once.
135    #[test]
136    fn a_row_each_character_every_ten_seconds() {
137        let data = std::env::temp_dir().join(format!("hydra-perf-{}", std::process::id()));
138        let _ = std::fs::remove_dir_all(&data);
139        let (seat, _lines) = Seat::idle(1, "Ashryn");
140        seat.story().perf.game_lines = 99;
141        let seats = vec![seat];
142        let mut log = PerfLog::default();
143        let start = Instant::now();
144        assert_eq!(log.tick(true, &data, &seats, start), Some(EVERY));
145        seats[0].story().perf.game_lines = 104;
146        assert_eq!(
147            log.tick(true, &data, &seats, start + Duration::from_secs(4)),
148            Some(Duration::from_secs(6))
149        );
150        log.tick(true, &data, &seats, start + EVERY);
151        log.tick(true, &data, &seats, start + EVERY * 2);
152        let written = std::fs::read_to_string(
153            data.join("perf")
154                .join(format!("{}.csv", hydra_net::date_dir())),
155        )
156        .unwrap();
157        let lines: Vec<&str> = written.lines().collect();
158        assert_eq!(lines.len(), 3, "{written}");
159        assert!(lines[0].starts_with("time,character,lich,view,game_lines"));
160        assert!(lines[1].contains(",GS3:Ashryn,off,lich,5,"), "{}", lines[1]);
161        assert!(lines[2].contains(",GS3:Ashryn,off,lich,0,"), "{}", lines[2]);
162        assert_eq!(log.tick(false, &data, &seats, start + EVERY * 3), None);
163        let _ = std::fs::remove_dir_all(&data);
164    }
165}

68/// A widget's own font and size, as its page shows them: none, the theme's
69/// for its kind; an effects window's, with `every` what every effects
70/// window wears, none of its own is every one's.
71fn face_rows(
72    face: Option<&hydra_ui::theme::Face>,
73    every: Option<&hydra_ui::theme::Face>,
74) -> Vec<Row> {
75    let from = |own: bool, theirs: bool| {
76        (!own && theirs).then(|| crate::widget::effects::FROM_EVERY.to_owned())
77    };
78    let font = face.and_then(|face| face.font.clone());
79    let size = face.and_then(|face| face.size);
80    let every_font = every.and_then(|every| every.font.clone());
81    let every_size = every.and_then(|every| every.size);
82    vec![
83        Row {
84            key: FONT.to_owned(),
85            label: "Font".to_owned(),
86            help: "This widget's own font, from the fonts folder or the computer's; none, its kind's in the theme.".to_owned(),
87            kind: RowKind::Font,
88            value: font
89                .clone()
90                .or_else(|| every_font.clone())
91                .map_or(Value::Unset, Value::Text),
92            here: font.is_some(),
93            from: from(font.is_some(), every_font.is_some()),
94            needs: None,
95        },
96        Row {
97            key: TEXT_SIZE.to_owned(),
98            label: "Text size".to_owned(),
99            help: "This widget's own text size, in points; 0, its kind's in the theme.".to_owned(),
100            kind: RowKind::Number {
101                min: 0.0,
102                max: 48.0,
103            },
104            value: size
105                .or(every_size)
106                .map_or(Value::Unset, |size| Value::Text(format!("{size}"))),
107            here: size.is_some(),
108            from: from(size.is_some(), every_size.is_some()),
109            needs: None,
110        },
111    ]
112}

114/// The *Every effects window* rows on an effects window's page, the
115/// character's: how every one draws its rows and its text, the effects it
116/// tracks, and the spell circles' colours, folded.
117fn every_effects_rows(
118    layout: &crate::layout::Layout,
119    tracked: &crate::tracked::TrackedEffects,
120) -> Vec<Row> {
121    let every = &layout.every_effects;
122    let mut rows = vec![heading("Every effects window")];
123    rows.extend(crate::widget::effects::every_rows(&every.rows));
124    rows.extend(
125        face_rows(Some(&every.face), None)
126            .into_iter()
127            .map(|mut row| {
128                let key = if row.key == FONT {
129                    EVERY_FONT
130                } else {
131                    EVERY_TEXT_SIZE
132                };
133                key.clone_into(&mut row.key);
134                row.help = format!(
135                    "Every effects window's {}, unless its own page says; none, the theme's.",
136                    row.label.to_lowercase()
137                );
138                row
139            }),
140    );
141    rows.push(crate::tracked::table(tracked));
142    rows.push(Row::fold("Spell circle colours"));
143    rows.push(hydra_ui::effects::circles_row(layout.circles.as_ref()));
144    rows
145}

147/// `face` with the page's `key` set `to`, as the page writes it, or back to
148/// the theme's (`None`).
149fn face_set(face: &mut hydra_ui::theme::Face, key: &str, to: Option<&str>) -> Result<(), String> {
150    match key {
151        FONT => {
152            face.font = to
153                .map(str::trim)
154                .filter(|name| !name.is_empty())
155                .map(str::to_owned);
156        }
157        _ => {
158            face.size = match to.map(str::trim) {
159                None => None,
160                Some(written) => {
161                    let size: f32 = written
162                        .parse()
163                        .map_err(|_| format!("`{written}` is not a size."))?;
164                    (size >= 6.0).then(|| size.min(48.0))
165                }
166            };
167        }
168    }
169    Ok(())
170}

230/// The styles under `folder`, each a pack's folder deep, by path.
231fn styles_in(folder: &std::path::Path) -> Vec<PathBuf> {
232    let folders = |of: &std::path::Path| -> Vec<PathBuf> {
233        std::fs::read_dir(of)
234            .map(|entries| {
235                entries
236                    .filter_map(Result::ok)
237                    .map(|entry| entry.path())
238                    .filter(|path| path.is_dir())
239                    .collect()
240            })
241            .unwrap_or_default()
242    };
243    let mut found: Vec<PathBuf> = folders(folder)
244        .iter()
245        .flat_map(|pack| folders(pack))
246        .collect();
247    found.sort();
248    found
249}

251/// The vessels under `folder`, a pack's folder and a style's deep: every
252/// PNG but the masks, by path.
253fn vessels_in(folder: &std::path::Path) -> Vec<PathBuf> {
254    let mut found = Vec::new();
255    let folders = |of: &std::path::Path| -> Vec<PathBuf> {
256        std::fs::read_dir(of)
257            .map(|entries| {
258                entries
259                    .filter_map(Result::ok)
260                    .map(|entry| entry.path())
261                    .filter(|path| path.is_dir())
262                    .collect()
263            })
264            .unwrap_or_default()
265    };
266    for pack in folders(folder) {
267        for style in folders(&pack) {
268            found.extend(pngs(&style).into_iter().filter(|path| {
269                !path
270                    .file_stem()
271                    .is_some_and(|stem| stem.to_string_lossy().ends_with("_mask"))
272            }));
273        }
274    }
275    found.sort();
276    found
277}

333/// The kit folders in `folder`'s folders, a pack's each: every folder
334/// holding a `kit.toml`, by path; none when it cannot be read. A kit loose
335/// in `folder` itself counts too.
336fn kits_within(folder: &std::path::Path) -> Vec<PathBuf> {
337    let kits_in = |pack: &std::path::Path| -> Vec<PathBuf> {
338        std::fs::read_dir(pack)
339            .map(|entries| {
340                entries
341                    .filter_map(Result::ok)
342                    .map(|entry| entry.path())
343                    .filter(|path| path.join("kit.toml").is_file())
344                    .collect()
345            })
346            .unwrap_or_default()
347    };
348    let mut found = kits_in(folder);
349    if let Ok(entries) = std::fs::read_dir(folder) {
350        for entry in entries.filter_map(Result::ok) {
351            let path = entry.path();
352            if path.is_dir() {
353                found.extend(kits_in(&path));
354            }
355        }
356    }
357    found.sort();
358    found
359}

378fn pngs(folder: &std::path::Path) -> Vec<PathBuf> {

380        .map(|entries| {

516    /// [`Play::widget_pages`], `now` the game's clock (unix seconds) where it
517    /// is known: the time of day a page shows as the game's own.
518    pub(crate) fn widget_pages_at(&self, pictures: &Pictures, now: Option<u32>) -> Vec<Page> {
519        let Some(layout) = &self.layout else {
520            return Vec::new();
521        };
522        let file = self
523            .layouts
524            .as_ref()
525            .map_or_else(String::new, |dir| dir.display().to_string());
526        let mut pages = Vec::new();
527        for holder in &layout.holders {
528            let window = holder.holds.title();
529            // A custom window's own page, before its widgets': its window's
530            // settings are its own, not theirs.
531            if let Some(title) = window {
532                pages.push(custom_window_page(holder, title, &file));
533            }
534            for one in holder.holds.placed() {
535                let mut rows = self.own_rows(layout, one, pictures, now);
536                let indicator =
537                    matches!(one.widget, Widget::Indicator(_) | Widget::CustomIndicator);
538                let effects = one.widget.is_effects_window();
539                // (the Injuries page's text under a heading of its own, over
540                // its statuses' buttons; an effects window's under its bars)
541                if one.widget == Widget::Injuries || effects {
542                    rows.push(heading("Text"));
543                }
544                // An indicator's text is its pill's alone.
545                if indicator {
546                    rows.push(heading("Pill text"));
547                }
548                let every_face = effects.then_some(&layout.every_effects.face);
549                rows.extend(face_rows(layout.faces.get(&one.id), every_face));
550                // Its tab's own, for what can be a tab of a stack.
551                if one.widget.stacks() {
552                    rows.extend(crate::widget::tab::rows(layout.tabs.get(&one.id)));
553                    rows.push(crate::widget::tab::bar_row(layout.tab_bar(one.id)));
554                    rows.push(crate::widget::tab::look_row(layout.tab_look(one.id)));
555                    rows.push(crate::widget::tab::height_row(layout.tab_height(one.id)));
556                }
557                // A standalone window's, set once and seldom again, folded
558                // away under its widget's rows.
559                if window.is_none() {
560                    rows.push(Row::fold("Window"));
561                    rows.extend(window_rows(holder));
562                }
563                // Last, what is the character's, not this widget's.
564                if indicator {
565                    rows.extend(every_indicator_rows(layout, &one.widget, pictures));
566                }
567                if effects {
568                    rows.extend(every_effects_rows(layout, &self.tracked));
569                }
570                // A custom indicator's page is called by its own name.
571                let name = shelves::called(layout, one);
572                pages.push(Page {
573                    id: page_id(one.id),
574                    title: window
575                        .map_or_else(|| name.to_string(), |window| format!("{name} ({window})")),
576                    file: file.clone(),
577                    takes: "at once".to_owned(),
578                    problem: None,
579                    rows,
580                });
581            }
582        }
583        // The window's own pages, after its widgets': its title bar, and its
584        // drawers.
585        pages.push(layout.title.page(file.clone()));
586        pages.push(window::drawers_page(&layout.drawers, file));
587        pages
588    }

590    /// The rows of widget `one`'s page that are its kind's own, before its
591    /// face's, its tab's and its window's; `now` the game's clock, as
592    /// [`Play::widget_pages_at`] has it.
593    fn own_rows(
594        &self,
595        layout: &crate::layout::Layout,
596        one: &crate::layout::Placed,
597        pictures: &Pictures,
598        now: Option<u32>,
599    ) -> Vec<Row> {
600        match (&one.widget, one.widget.bar_look()) {
601            (_, Some(default)) => {
602                let pulse = one.widget == Widget::Pulse;
603                bar_rows(
604                    layout
605                        .looks
606                        .get(&one.id)
607                        .cloned()
608                        .zip(one.widget.bar_token())
609                        .map(|(look, token)| look.themed(token))
610                        .as_ref(),
611                    &default,
612                    pictures,
613                    (pulse, one.widget.emblems()),
614                )
615            }

784    pub(crate) fn widget_change(
785        &mut self,
786        page: &str,
787        key: &str,
788        to: Option<&str>,
789    ) -> Result<String, String> {
790        if page == super::titlebar::PAGE || page == DRAWERS_PAGE {
791            return self.window_change(page, key, to);
792        }
793        if let Some(holder) = page.strip_prefix(WINDOW_PREFIX) {
794            let holder = holder
795                .parse::<u32>()
796                .map_err(|_| format!("There is no {page} page."))?;
797            return self.custom_window_change(holder, key, to);
798        }
799        let placed = page
800            .strip_prefix(PREFIX)
801            .and_then(|id| id.parse::<u32>().ok())
802            .ok_or_else(|| format!("There is no {page} page."))?;
803        // The character's, not the widget's: every effect list shares them.
804        if key == crate::tracked::TABLE {
805            let effects = match to {
806                Some(written) => crate::tracked::from_table(written)?,
807                None => Vec::new(),
808            };
809            self.track(crate::tracked::Change::All(effects));
810            return Ok("Tracked effects: changed.".to_owned());
811        }
812        let layout = self
813            .layout
814            .as_mut()
815            .ok_or_else(|| "The window has no layout yet.".to_owned())?;
816        // The character's too, not the widget's.
817        if let Some(said) = character_change(layout, key, to) {
818            let said = said?;
819            self.save();
820            return Ok(said);
821        }
822        let widget = layout
823            .placed()
824            .into_iter()
825            .find(|one| one.id == placed)
826            .map(|one| one.widget.clone())
827            .ok_or_else(|| "That widget is no longer in the window.".to_owned())?;
828        let holder = layout
829            .holders
830            .iter()
831            .find(|holder| holder.holds.placed().any(|one| one.id == placed))
832            .map(|holder| holder.id)
833            .ok_or_else(|| "That widget is no longer in the window.".to_owned())?;
834        if [BACKGROUND, LAYER, TITLE_COLOR, TITLE_SIZE].contains(&key) {
835            let said = holder_change(layout, holder, key, to)?;
836            self.save();
837            return Ok(said);
838        }
839        if key == TITLED {
840            return self.title_change(holder, to);
841        }
842        // The cell's, not the tab's: every tab of the cell shares its strip.
843        if [
844            crate::widget::tab::BAR,
845            crate::widget::tab::LOOK,
846            crate::widget::tab::HEIGHT,
847        ]
848        .contains(&key)
849        {
850            let said = strip_change(layout, placed, (key, to))?;
851            self.save();
852            return Ok(said);
853        }
854        if key.starts_with(crate::widget::tab::PREFIX) {
855            keep(
856                &mut layout.tabs,
857                placed,
858                &crate::widget::tab::TabLook::default(),
859                |look| crate::widget::tab::set(look, key, to),
860            )?;
861            self.save();
862            return Ok(format!("{}: its tab changed.", widget.name()));
863        }
864        if key == FONT || key == TEXT_SIZE {
865            keep(
866                &mut layout.faces,
867                placed,
868                &hydra_ui::theme::Face::default(),
869                |face| face_set(face, key, to),
870            )?;
871            self.save();
872            return Ok(format!("{}: changed.", widget.name()));
873        }
874        layout_change(layout, &widget, placed, (key, to))?;
875        self.save();
876        Ok(format!("{}: changed.", widget.name()))
877    }
878}

880/// The strip of the cell holding widget `placed` -- the side it runs along,
881/// how its tabs are drawn, or how tall -- set as `key` and `to` say. What
882/// was done.
883fn strip_change(
884    layout: &mut crate::layout::Layout,
885    placed: u32,
886    (key, to): (&str, Option<&str>),
887) -> Result<String, String> {
888    if key == crate::widget::tab::HEIGHT {
889        let (least, most) = crate::widget::tab::HEIGHTS;
890        let height = match to.map(str::trim) {
891            None => None,
892            Some(written) => Some(
893                written
894                    .parse::<u8>()
895                    .ok()
896                    .filter(|height| (least..=most).contains(height))
897                    .ok_or_else(|| format!("`{written}` is not a height, {least} to {most}."))?,
898            ),
899        };
900        if !layout.set_tab_height(placed, height) {
901            return Err("That widget is in no cell.".to_owned());
902        }
903        return Ok(match height {
904            Some(height) => format!("Its tabs are {height} points tall."),
905            None => "Its tabs are as tall as the theme's title bars.".to_owned(),
906        });
907    }
908    if key == crate::widget::tab::LOOK {
909        let look = match to.map(str::trim) {
910            None | Some(crate::widget::tab::THEMES) => None,
911            Some(to) => Some(
912                hydra_ui::theme::TabStyle::named(to)
913                    .ok_or_else(|| format!("`{to}` is not a tab look."))?,
914            ),
915        };
916        if !layout.set_tab_look(placed, look) {
917            return Err("That widget is in no cell.".to_owned());
918        }
919        return Ok(match look {
920            Some(look) => format!("Its tabs are drawn {}.", look.name()),
921            None => "Its tabs are drawn as the theme says.".to_owned(),
922        });
923    }
924    let bar = match to.map(str::trim) {
925        None => crate::layout::Bar::default(),
926        Some(to) => crate::layout::Bar::of_key(to)
927            .ok_or_else(|| format!("`{to}` is not a side of the cell."))?,
928    };
929    if !layout.set_tab_bar(placed, bar) {
930        return Err("That widget is in no cell.".to_owned());
931    }
932    Ok(format!(
933        "Its tabs run along the {}.",
934        bar.label().to_lowercase()
935    ))
936}

1069/// Widget `placed`'s entry in `kept` changed by `change`: kept only while it
1070/// differs from its kind's own, `default`.
1071fn keep<T: Clone + PartialEq>(
1072    kept: &mut BTreeMap<u32, T>,
1073    placed: u32,
1074    default: &T,
1075    change: impl FnOnce(&mut T) -> Result<(), String>,
1076) -> Result<(), String> {
1077    let mut now = kept
1078        .get(&placed)
1079        .cloned()
1080        .unwrap_or_else(|| default.clone());
1081    change(&mut now)?;
1082    if now == *default {
1083        kept.remove(&placed);
1084    } else {
1085        kept.insert(placed, now);
1086    }
1087    Ok(())
1088}

1470fn lines_set(
1471    lines: &mut Lines,
1472    key: &str,
1473    to: Option<&str>,
1474    kind: LinesKind,
1475) -> Result<(), String> {
1476    let story = kind == LinesKind::Story;
1477    let default = Lines::default();
1478    let switch = |to: Option<&str>, default: bool| to.map_or(Ok(default), on_off);
1479    match (key, to) {
1480        ("stamps", Some(to)) => {
1481            lines.stamps = STAMPS
1482                .iter()
1483                .find(|(_, value, _)| *value == to)
1484                .map(|(stamps, ..)| *stamps)
1485                .ok_or_else(|| format!("A line's time does not go `{to}`."))?;
1486        }
1487        ("stamps", None) => lines.stamps = default.stamps,
1488        ("hours", Some(to)) => {
1489            lines.hours = HOURS
1490                .iter()
1491                .find(|(_, value, _)| *value == to)
1492                .map(|(hours, ..)| *hours)
1493                .ok_or_else(|| format!("There is no `{to}`-hour clock."))?;
1494        }
1495        ("hours", None) => lines.hours = default.hours,
1496        ("seconds", to) => lines.seconds = switch(to, default.seconds)?,
1497        ("wrap", to) => lines.wrap = switch(to, default.wrap)?,
1498        ("prompts", to) if story => lines.prompts = switch(to, default.prompts)?,
1499        ("echo", to) if story => lines.echo = switch(to, default.echo)?,
1500        ("summary", to) if kind == LinesKind::Bounty => {
1501            lines.summary = switch(to, default.summary)?;
1502        }
1503        (key, _) => return Err(format!("It has no setting {key}.")),
1504    }
1505    Ok(())
1506}

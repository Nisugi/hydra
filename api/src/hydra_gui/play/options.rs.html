1//! A widget's own page: a health bar's right-click, *Settings...*, takes
2//! the player to that bar's settings -- horizontal or vertical, its text
3//! inside or outside it, a percent, current/max, a label with numbers or
4//! any mix, its colour, its overlay. Every way in opens the one settings
5//! menu at its place (`settings_menu.rs`): the right-click's *Settings...*
6//! asks for this widget's page there.
7//!
8//! The pages are the play window's, built from its layout, and a change is
9//! made to the layout and saved with it: no behavior is behind them. A bar
10//! widget's page holds its [`Look`]; the Room widget's, which of its parts
11//! it shows ([`RoomParts`]): which streams show in the Room window.

49/// A widget's own font and size, as its page shows them: none, the theme's
50/// for its kind.
51fn face_rows(face: Option<&hydra_ui::theme::Face>) -> Vec<Row> {
52    vec![
53        Row {
54            key: FONT.to_owned(),
55            label: "Font".to_owned(),
56            help: "This widget's own font, from the fonts folder or the computer's; none, its kind's in the theme.".to_owned(),
57            kind: RowKind::Font,
58            value: face
59                .and_then(|face| face.font.clone())
60                .map_or(Value::Unset, Value::Text),
61            here: face.is_some_and(|face| face.font.is_some()),
62            from: None,
63        },
64        Row {
65            key: TEXT_SIZE.to_owned(),
66            label: "Text size".to_owned(),
67            help: "This widget's own text size, in points; 0, its kind's in the theme.".to_owned(),
68            kind: RowKind::Number {
69                min: 0.0,
70                max: 48.0,
71            },
72            value: face
73                .and_then(|face| face.size)
74                .map_or(Value::Unset, |size| Value::Text(format!("{size}"))),
75            here: face.is_some_and(|face| face.size.is_some()),
76            from: None,
77        },
78    ]
79}

81/// `face` with the page's `key` set `to`, as the page writes it, or back to
82/// the theme's (`None`).
83fn face_set(face: &mut hydra_ui::theme::Face, key: &str, to: Option<&str>) -> Result<(), String> {
84    match key {
85        FONT => {
86            face.font = to
87                .map(str::trim)
88                .filter(|name| !name.is_empty())
89                .map(str::to_owned);
90        }
91        _ => {
92            face.size = match to.map(str::trim) {
93                None => None,
94                Some(written) => {
95                    let size: f32 = written
96                        .parse()
97                        .map_err(|_| format!("`{written}` is not a size."))?;
98                    (size >= 6.0).then(|| size.min(48.0))
99                }
100            };
101        }
102    }
103    Ok(())
104}

142/// The PNGs in `folder`, by name; none when it cannot be read.
143fn pngs(folder: &std::path::Path) -> Vec<PathBuf> {
144    let mut found: Vec<PathBuf> = std::fs::read_dir(folder)
145        .map(|entries| {
146            entries
147                .filter_map(Result::ok)
148                .map(|entry| entry.path())
149                .filter(|path| {
150                    path.extension()
151                        .is_some_and(|x| x.eq_ignore_ascii_case("png"))
152                })
153                .collect()
154        })
155        .unwrap_or_default();
156    found.sort();
157    found
158}

261    /// [`Play::widget_pages`], `now` the game's clock (unix seconds) where it
262    /// is known: the time of day a page shows as the game's own.
263    pub(crate) fn widget_pages_at(&self, pictures: &Pictures, now: Option<u32>) -> Vec<Page> {
264        let Some(layout) = &self.layout else {
265            return Vec::new();
266        };
267        let file = self
268            .layouts
269            .as_ref()
270            .map_or_else(String::new, |dir| dir.display().to_string());
271        let mut pages = Vec::new();
272        for holder in &layout.holders {
273            let window = holder.holds.title();
274            for one in holder.holds.placed() {
275                let mut rows = self.own_rows(layout, one, pictures, now);
276                // (the Injuries page's text and window under a heading of
277                // their own, over its statuses' buttons)
278                if one.widget == Widget::Injuries {
279                    rows.push(heading("Text and window"));
280                }
281                rows.extend(face_rows(layout.faces.get(&one.id)));
282                // Its tab's own, for what can be a tab of a stack.
283                if one.widget.tab_group() == crate::widget::Group::Streams {
284                    rows.extend(crate::widget::tab::rows(layout.tabs.get(&one.id)));
285                    rows.push(crate::widget::tab::bar_row(layout.tab_bar(one.id)));
286                }
287                rows.push(title_row(holder.untitled));
288                rows.extend(title_look_rows(holder));
289                rows.push(background_row(holder.background));
290                rows.push(on_top_row(holder.on_top));
291                // A custom indicator's page is called by its own name.
292                let name = layout.customs.get(&one.id).map_or_else(
293                    || one.widget.name(),
294                    |custom| std::borrow::Cow::Borrowed(custom.name()),
295                );
296                pages.push(Page {
297                    id: page_id(one.id),
298                    title: window
299                        .map_or_else(|| name.to_string(), |window| format!("{name} ({window})")),
300                    file: file.clone(),
301                    takes: "at once".to_owned(),
302                    problem: None,
303                    rows,
304                });
305            }
306        }
307        // The window's own page, after its widgets'.
308        pages.push(layout.title.page(file));
309        pages
310    }

312    /// The rows of widget `one`'s page that are its kind's own, before its
313    /// face's, its tab's and its window's; `now` the game's clock, as
314    /// [`Play::widget_pages_at`] has it.
315    fn own_rows(
316        &self,
317        layout: &crate::layout::Layout,
318        one: &crate::layout::Placed,
319        pictures: &Pictures,
320        now: Option<u32>,
321    ) -> Vec<Row> {
322        match (&one.widget, one.widget.bar_look()) {
323            (_, Some(default)) => {
324                let pulse = one.widget == Widget::Pulse;
325                bar_rows(
326                    layout
327                        .looks
328                        .get(&one.id)
329                        .cloned()
330                        .zip(one.widget.bar_token())
331                        .map(|(look, token)| look.themed(token))
332                        .as_ref(),
333                    &default,
334                    &pictures.overlays,
335                    pulse,
336                )
337            }
338            (Widget::Injuries, None) => doll_rows(layout.dolls.get(&one.id), &pictures.dolls, now),
339            (Widget::Room, None) => room_rows(layout.rooms.get(&one.id)),
340            (Widget::Aim, None) => crate::widget::aim::rows(layout.aims.get(&one.id)),
341            (Widget::RightHand | Widget::LeftHand | Widget::Prepared | Widget::Reserve, None) => {
342                crate::widget::labels::rows(&one.widget, layout.labels.get(&one.id))
343            }
344            (Widget::Timers, None) => {
345                let mut rows = crate::widget::effects::timers_rows(layout.timers.get(&one.id));
346                rows.push(crate::tracked::table(&self.tracked));
347                rows.push(crate::widget::circles::row(layout.circles.as_ref()));
348                rows
349            }
350            (Widget::Effects(_) | Widget::Missing, None) => {
351                vec![
352                    crate::tracked::table(&self.tracked),
353                    crate::widget::circles::row(layout.circles.as_ref()),
354                ]
355            }
356            (Widget::Quick, None) => crate::widget::quick::rows(layout.quicks.get(&one.id)),
357            (Widget::Hotbar, None) => {
358                crate::widget::hotbar::rows(layout.hotbars.get(&one.id), &self.hotbars.names())
359            }
360            (Widget::Minimap, None) => {
361                crate::widget::minimap::look::rows(layout.minimaps.get(&one.id))
362            }
363            (Widget::Creatures, None) => {
364                crate::widget::roster::targets_rows(layout.rosters.get(&one.id))
365            }
366            (Widget::Players, None) => {
367                crate::widget::roster::players_rows(layout.rosters.get(&one.id))
368            }
369            (Widget::Npcs | Widget::Objects, None) => list_rows(layout.lists.get(&one.id).copied()),
370            (Widget::Indicator(_), _) => {
371                crate::widget::status::indicator_rows(layout.indicators.get(&one.id))
372            }
373            (Widget::CustomIndicator, _) => {
374                crate::widget::custom_indicator::rows(layout.customs.get(&one.id))
375            }
376            (Widget::Compass, None) => {
377                crate::compass_art::rows(layout.compasses.get(&one.id), &pictures.compasses)
378            }
379            (Widget::Story | Widget::Stream(_), None) => {
380                lines_rows(layout.lines.get(&one.id), lines_kind(&one.widget))
381            }
382            _ => Vec::new(),
383        }
384    }

397    /// Set `key` on the widget page `page` to `to`, as the page writes it,
398    /// or back to the kind's own (`None`); the layout saved. What was done.
399    ///
400    /// # Errors
401    ///
402    /// Why nothing was: no such widget, or not a value it takes.
403    /// Turn every Injuries widget in the window to `deg` (`.doll face`).
404    /// How many there were.
405    pub(crate) fn turn_dolls(&mut self, deg: i16) -> usize {
406        let dolls: Vec<u32> = self
407            .layout
408            .as_ref()
409            .map(|layout| {
410                layout
411                    .placed()
412                    .into_iter()
413                    .filter(|one| one.widget == Widget::Injuries)
414                    .map(|one| one.id)
415                    .collect()
416            })
417            .unwrap_or_default();
418        for &placed in &dolls {
419            let _ = self.widget_change(&page_id(placed), "facing", Some(&deg.to_string()));
420        }
421        dolls.len()
422    }

424    /// The title of the window widget `placed` is in shown, or hidden for
425    /// `to` off; its strip taken off or given back when next drawn, so what
426    /// it holds stays where it was (`holders.rs`).
427    fn title_change(&mut self, placed: u32, to: Option<&str>) -> Result<String, String> {
428        let shown = match to {
429            None => true,
430            Some(to) => on_off(to)?,
431        };
432        let holder = self
433            .layout
434            .as_mut()
435            .and_then(|layout| {
436                layout
437                    .holders
438                    .iter_mut()
439                    .find(|holder| holder.holds.placed().any(|one| one.id == placed))
440            })
441            .ok_or_else(|| "That widget is no longer in the window.".to_owned())?;
442        let was = holder.shows_title(false);
443        holder.untitled = !shown;
444        if holder.shows_title(false) != was {
445            let id = holder.id;
446            self.retitled.retain(|(one, _)| *one != id);
447            self.retitled.push((id, shown));
448        }
449        self.save();
450        Ok(if shown {
451            "Its window shows its title.".to_owned()
452        } else {
453            "Its window's title is hidden, except while arranging.".to_owned()
454        })
455    }

457    pub(crate) fn widget_change(
458        &mut self,
459        page: &str,
460        key: &str,
461        to: Option<&str>,
462    ) -> Result<String, String> {
463        if page == super::titlebar::PAGE {
464            let layout = self
465                .layout
466                .as_mut()
467                .ok_or_else(|| "The window has no layout yet.".to_owned())?;
468            let said = layout.title.set(key, to)?;
469            self.save();
470            return Ok(said);
471        }
472        let placed = page
473            .strip_prefix(PREFIX)
474            .and_then(|id| id.parse::<u32>().ok())
475            .ok_or_else(|| format!("There is no {page} page."))?;
476        // The character's, not the widget's: every effect list shares them.
477        if key == crate::tracked::TABLE {
478            let effects = match to {
479                Some(written) => crate::tracked::from_table(written)?,
480                None => Vec::new(),
481            };
482            self.track(crate::tracked::Change::All(effects));
483            return Ok("Tracked effects: changed.".to_owned());
484        }
485        let layout = self
486            .layout
487            .as_mut()
488            .ok_or_else(|| "The window has no layout yet.".to_owned())?;
489        // The character's too: the circles' colours every effect list wears.
490        if key == crate::widget::circles::TABLE {
491            layout.circles = match to {
492                Some(written) => Some(crate::widget::circles::read(written)?),
493                None => None,
494            };
495            self.save();
496            return Ok("Spell circle colours: changed.".to_owned());
497        }
498        let widget = layout
499            .placed()
500            .into_iter()
501            .find(|one| one.id == placed)
502            .map(|one| one.widget.clone())
503            .ok_or_else(|| "That widget is no longer in the window.".to_owned())?;
504        if [BACKGROUND, ON_TOP, TITLE_COLOR, TITLE_SIZE].contains(&key) {
505            let said = window_change(layout, placed, key, to)?;
506            self.save();
507            return Ok(said);
508        }
509        if key == TITLED {
510            return self.title_change(placed, to);
511        }
512        // The cell's, not the tab's: every tab of the cell shares its strip.
513        if key == crate::widget::tab::BAR {
514            let bar = match to.map(str::trim) {
515                None => crate::layout::Bar::default(),
516                Some(to) => crate::layout::Bar::of_key(to)
517                    .ok_or_else(|| format!("`{to}` is not a side of the cell."))?,
518            };
519            if !layout.set_tab_bar(placed, bar) {
520                return Err("That widget is in no cell.".to_owned());
521            }
522            self.save();
523            return Ok(format!(
524                "Its tabs run along the {}.",
525                bar.label().to_lowercase()
526            ));
527        }
528        if key.starts_with(crate::widget::tab::PREFIX) {
529            keep(
530                &mut layout.tabs,
531                placed,
532                &crate::widget::tab::TabLook::default(),
533                |look| crate::widget::tab::set(look, key, to),
534            )?;
535            self.save();
536            return Ok(format!("{}: its tab changed.", widget.name()));
537        }
538        if key == FONT || key == TEXT_SIZE {
539            keep(
540                &mut layout.faces,
541                placed,
542                &hydra_ui::theme::Face::default(),
543                |face| face_set(face, key, to),
544            )?;
545            self.save();
546            return Ok(format!("{}: changed.", widget.name()));
547        }
548        layout_change(layout, &widget, placed, (key, to))?;
549        self.save();
550        Ok(format!("{}: changed.", widget.name()))
551    }
552}

554/// `key` set to `to` on the widget `widget` placed as `placed`, kept in
555/// `layout` by its kind.
556fn layout_change(
557    layout: &mut crate::layout::Layout,
558    widget: &Widget,
559    placed: u32,
560    (key, to): (&str, Option<&str>),
561) -> Result<(), String> {
562    match widget {
563        Widget::Room => keep(&mut layout.rooms, placed, &RoomParts::default(), |parts| {
564            room_set(parts, key, to)
565        })?,
566        Widget::Npcs | Widget::Objects => {
567            keep(&mut layout.lists, placed, &Listing::default(), |listing| {
568                list_set(listing, key, to)
569            })?;
570        }
571        Widget::Creatures | Widget::Players => keep(
572            &mut layout.rosters,
573            placed,
574            &crate::widget::roster::RosterLook::default(),
575            |look| crate::widget::roster::set(look, key, to),
576        )?,
577        Widget::Injuries => keep(&mut layout.dolls, placed, &DollLook::default(), |look| {
578            doll_set(look, key, to)
579        })?,
580        Widget::RightHand | Widget::LeftHand | Widget::Prepared | Widget::Reserve => keep(
581            &mut layout.labels,
582            placed,
583            &crate::widget::labels::LabelLook::default(),
584            |look| crate::widget::labels::set(look, key, to),
585        )?,
586        Widget::Indicator(_) => keep(
587            &mut layout.indicators,
588            placed,
589            &crate::widget::IndicatorLook::default(),
590            |look| crate::widget::status::indicator_set(look, key, to),
591        )?,
592        Widget::CustomIndicator => keep(
593            &mut layout.customs,
594            placed,
595            &crate::widget::custom_indicator::CustomIndicator::default(),
596            |custom| crate::widget::custom_indicator::set(custom, key, to),
597        )?,
598        Widget::Compass => keep(
599            &mut layout.compasses,
600            placed,
601            &crate::compass_art::CompassLook::default(),
602            |look| crate::compass_art::set(look, key, to),
603        )?,
604        Widget::Aim => keep(
605            &mut layout.aims,
606            placed,
607            &crate::widget::aim::AimLook::default(),
608            |look| crate::widget::aim::set(look, key, to),
609        )?,
610        Widget::Timers => keep(
611            &mut layout.timers,
612            placed,
613            &crate::widget::effects::TimersLook::default(),
614            |look| crate::widget::effects::timers_set(look, key, to),
615        )?,
616        Widget::Quick => keep(
617            &mut layout.quicks,
618            placed,
619            &crate::widget::quick::QuickLook::default(),
620            |look| crate::widget::quick::set(look, key, to),
621        )?,
622        Widget::Hotbar => keep(
623            &mut layout.hotbars,
624            placed,
625            &crate::widget::hotbar::HotbarLook::default(),
626            |look| crate::widget::hotbar::set(look, key, to),
627        )?,
628        Widget::Minimap => keep(
629            &mut layout.minimaps,
630            placed,
631            &crate::widget::minimap::MinimapLook::default(),
632            |look| crate::widget::minimap::look::set(look, key, to),
633        )?,
634        Widget::Story | Widget::Stream(_) => {
635            let kind = lines_kind(widget);
636            keep(&mut layout.lines, placed, &Lines::default(), |lines| {
637                lines_set(lines, key, to, kind)
638            })?;
639        }
640        _ => {
641            let default = widget
642                .bar_look()
643                .ok_or_else(|| format!("{} has no settings of its own.", widget.name()))?;
644            keep(&mut layout.looks, placed, &default, |look| {
645                set(look, &default, key, to)
646            })?;
647        }
648    }
649    Ok(())
650}

652/// Widget `placed`'s entry in `kept` changed by `change`: kept only while it
653/// differs from its kind's own, `default`.
654fn keep<T: Clone + PartialEq>(
655    kept: &mut BTreeMap<u32, T>,
656    placed: u32,
657    default: &T,
658    change: impl FnOnce(&mut T) -> Result<(), String>,
659) -> Result<(), String> {
660    let mut now = kept
661        .get(&placed)
662        .cloned()
663        .unwrap_or_else(|| default.clone());
664    change(&mut now)?;
665    if now == *default {
666        kept.remove(&placed);
667    } else {
668        kept.insert(placed, now);
669    }
670    Ok(())
671}

673/// `on` or `off`, as the menu writes a switch.
674/// The key of a widget page's row for its window's title.
675const TITLED: &str = "titled";

677/// The window that holds the widget `placed`, its row `key` set `to`, a
678/// percent for its background, on or off for on top, a colour or a size
679/// for its title: what was done.
680///
681/// # Errors
682///
683/// The widget is no longer in the window, or `to` does not read.
684fn window_change(
685    layout: &mut crate::layout::Layout,
686    placed: u32,
687    key: &str,
688    to: Option<&str>,
689) -> Result<String, String> {
690    let holder = layout
691        .holders
692        .iter_mut()
693        .find(|holder| holder.holds.placed().any(|one| one.id == placed))
694        .ok_or_else(|| "That widget is no longer in the window.".to_owned())?;
695    if key == TITLE_COLOR || key == TITLE_SIZE {
696        return title_look_change(holder, key, to);
697    }
698    if key == ON_TOP {
699        holder.on_top = match to {
700            None => false,
701            Some(to) => on_off(to)?,
702        };
703        return Ok(if holder.on_top {
704            "Its window is kept above the others.".to_owned()
705        } else {
706            "Its window takes its turn with the others.".to_owned()
707        });
708    }
709    let percent = match to.map(str::trim) {
710        None => None,
711        Some(to) => Some(
712            to.parse::<f32>()
713                .ok()
714                .filter(|percent| (0.0..=100.0).contains(percent))
715                .ok_or_else(|| format!("`{to}` is not a percent, 0 to 100."))?,
716        ),
717    };
718    holder.background = percent
719        .filter(|percent| *percent < 100.0)
720        .map(|percent| percent / 100.0);
721    Ok(match holder.background {
722        None => "Its window is solid.".to_owned(),
723        Some(clear) if clear <= 0.0 => "Its window is clear: its widgets alone, and a click \
724                                        goes through it. Arrange shows it whole."
725            .to_owned(),
726        Some(_) => "Its window is see-through.".to_owned(),
727    })
728}

739/// `holder`'s title's colour or size, `key`, set `to`; the theme's again
740/// with none: what was done.
741fn title_look_change(
742    holder: &mut crate::layout::Holder,
743    key: &str,
744    to: Option<&str>,
745) -> Result<String, String> {
746    let to = to.map(str::trim).filter(|to| !to.is_empty());
747    if key == TITLE_COLOR {
748        holder.title_color = match to {
749            None => None,
750            Some(to) => {
751                Some(crate::menu::rgb(to).ok_or_else(|| format!("`{to}` is not a colour."))?)
752            }
753        };
754        return Ok(if holder.title_color.is_some() {
755            "Its window's title has a colour of its own.".to_owned()
756        } else {
757            "Its window's title is the theme's colour.".to_owned()
758        });
759    }
760    let (least, most) = TITLE_SIZES;
761    holder.title_size = match to {
762        None => None,
763        Some(to) => Some(
764            to.parse::<f32>()
765                .ok()
766                .filter(|size| (least..=most).contains(size))
767                .ok_or_else(|| format!("`{to}` is not a size, {least} to {most}."))?,
768        ),
769    };
770    Ok(match holder.title_size {
771        Some(size) => format!("Its window's title is {size} points."),
772        None => "Its window's title is the theme's size.".to_owned(),
773    })
774}

883/// How lines are drawn, `key` set to `to` or to the kind's own; the
884/// prompts and what was typed only on the `story`.
885/// Which widget of lines it is, for the rows only some have: the story's
886/// prompts and echo, the bounty's summary.
887#[derive(Clone, Copy, PartialEq, Eq)]
888enum LinesKind {
889    Story,
890    Bounty,
891    Stream,
892}
893
894/// The kind of `widget`, a widget of lines.
895fn lines_kind(widget: &Widget) -> LinesKind {
896    match widget {
897        Widget::Story => LinesKind::Story,
898        Widget::Stream(id) if id == hydra_model::state::bounty::STREAM => LinesKind::Bounty,
899        _ => LinesKind::Stream,
900    }
901}
902
903fn lines_set(
904    lines: &mut Lines,
905    key: &str,
906    to: Option<&str>,
907    kind: LinesKind,
908) -> Result<(), String> {
909    let story = kind == LinesKind::Story;
910    let default = Lines::default();
911    let switch = |to: Option<&str>, default: bool| to.map_or(Ok(default), on_off);
912    match (key, to) {
913        ("stamps", Some(to)) => {
914            lines.stamps = STAMPS
915                .iter()
916                .find(|(_, value, _)| *value == to)
917                .map(|(stamps, ..)| *stamps)
918                .ok_or_else(|| format!("A line's time does not go `{to}`."))?;
919        }
920        ("stamps", None) => lines.stamps = default.stamps,
921        ("hours", Some(to)) => {
922            lines.hours = HOURS
923                .iter()
924                .find(|(_, value, _)| *value == to)
925                .map(|(hours, ..)| *hours)
926                .ok_or_else(|| format!("There is no `{to}`-hour clock."))?;
927        }
928        ("hours", None) => lines.hours = default.hours,
929        ("seconds", to) => lines.seconds = switch(to, default.seconds)?,
930        ("wrap", to) => lines.wrap = switch(to, default.wrap)?,
931        ("prompts", to) if story => lines.prompts = switch(to, default.prompts)?,
932        ("echo", to) if story => lines.echo = switch(to, default.echo)?,
933        ("summary", to) if kind == LinesKind::Bounty => {
934            lines.summary = switch(to, default.summary)?;
935        }
936        (key, _) => return Err(format!("It has no setting {key}.")),
937    }
938    Ok(())
939}

941/// The Room widget's rows: each part, on or off, as `parts` says or all.
942fn room_rows(parts: Option<&RoomParts>) -> Vec<Row> {

944    let now = parts.copied().unwrap_or_default();
945    PARTS
946        .iter()
947        .map(|(key, label, help)| {

950            Row {
951                key: (*key).to_owned(),
952                label: (*label).to_owned(),
953                help: (*help).to_owned(),
954                kind: RowKind::Toggle,
955                value: Value::On(on),
956                here: on != was,
957                from: None,
958            }
959        })
960        .collect()
961}

963/// A story's or a stream's rows: how it draws its lines, as `lines` says or
964/// as it always has; the prompts and what was typed only on the `story`.
965fn lines_rows(lines: Option<&Lines>, kind: LinesKind) -> Vec<Row> {
966    let story = kind == LinesKind::Story;
967    let default = Lines::default();
968    let now = lines.copied().unwrap_or_default();
969    let row = |key: &str, label: &str, help: &str, kind: RowKind, value: Value, here: bool| Row {
970        key: key.to_owned(),
971        label: label.to_owned(),
972        help: help.to_owned(),
973        kind,
974        value,
975        here,
976        from: None,
977    };
978    let switch = |key: &str, label: &str, help: &str, now: bool, was: bool| {
979        row(
980            key,
981            label,
982            help,
983            RowKind::Toggle,
984            Value::On(now),
985            now != was,
986        )
987    };
988    let stamps = STAMPS
989        .iter()
990        .find(|(each, ..)| *each == now.stamps)
991        .map_or("", |(_, value, _)| value);
992    let hours = HOURS
993        .iter()
994        .find(|(each, ..)| *each == now.hours)
995        .map_or("", |(_, value, _)| value);
996    let mut rows = vec![
997        row(
998            "stamps",
999            "Timestamps",
1000            "When each line arrived, on your clock: before it, after it, or not at all.",
1001            choice(STAMPS.iter().map(|(_, value, called)| (*value, *called))),
1002            Value::Text(stamps.to_owned()),
1003            now.stamps != default.stamps,
1004        ),
1005        switch(
1006            "seconds",
1007            "With seconds",
1008            "7:08:05 rather than 7:08.",
1009            now.seconds,
1010            default.seconds,
1011        ),
1012        row(
1013            "hours",
1014            "Clock",
1015            "7:08 PM, or 19:08.",
1016            choice(HOURS.iter().map(|(_, value, called)| (*value, *called))),
1017            Value::Text(hours.to_owned()),
1018            now.hours != default.hours,
1019        ),
1020        switch(
1021            "wrap",
1022            "Word wrap",
1023            "A long line wraps; off, the window scrolls sideways.",
1024            now.wrap,
1025            default.wrap,
1026        ),
1027    ];
1028    if story {
1029        rows.push(switch(
1030            "prompts",
1031            "Prompts",
1032            "The game's prompt, >, after what it says.",
1033            now.prompts,
1034            default.prompts,
1035        ));
1036        rows.push(switch(
1037            "echo",
1038            "What you type",
1039            "Each line you send, after the prompt it followed.",
1040            now.echo,
1041            default.echo,
1042        ));
1043    }
1044    if kind == LinesKind::Bounty {
1045        rows.push(switch(
1046            "summary",
1047            "Summary",
1048            "The task in brief, up to four lines: what kind, what and how many, where.",
1049            now.summary,
1050            default.summary,
1051        ));
1052    }
1053    rows
1054}

1065/// The overlays to choose from: none, then each image by its file's name.
1066fn overlay_choice(overlays: &[PathBuf]) -> RowKind {
1067    let mut images: Vec<(String, String)> = vec![(String::new(), "None".to_owned())];
1068    images.extend(overlays.iter().map(|path| {
1069        let mut called = path.file_name().map_or_else(
1070            || path.display().to_string(),
1071            |name| name.to_string_lossy().into_owned(),
1072        );
1073        // A pack's picture, told from the player's own of the same name.
1074        if path.parent().is_some_and(|folder| {
1075            folder.ends_with(std::path::Path::new("global").join("images").join("dolls"))
1076        }) {
1077            called.push_str(" (pack)");
1078        }
1079        (path.display().to_string(), called)
1080    }));
1081    RowKind::Choice(images)
1082}

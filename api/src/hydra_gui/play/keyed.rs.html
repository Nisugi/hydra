1//! What the keys' actions do in a play window: the command input's, and
2//! the window in use's.
3
4use super::{Play, draw};
5use crate::layout::Zone;

7impl Play {
8    /// The window in use, by its placed id: the widget last clicked in,
9    /// while it is here; the story until one is.
10    pub(crate) fn in_use(&self) -> Option<u32> {
11        let placed = self.layout.as_ref()?.placed();
12        self.in_use
13            .filter(|id| placed.iter().any(|placed| placed.id == *id))
14            .or_else(|| {
15                placed
16                    .iter()
17                    .find(|placed| placed.widget.is_story())
18                    .map(|placed| placed.id)
19            })
20    }

22    /// Whether the command input has the keyboard, or nothing does, with
23    /// nothing open that a key would close or move in first: the object's
24    /// menu once it shows, a widget's, a drop-down or a menu of the bar.
25    /// The Add-a-widget list stays open beside the play, and holds the keys
26    /// only while its search has the keyboard. What an action on the input
27    /// waits for ([`Action::on_input`](crate::keys::Action::on_input)).
28    pub(crate) fn typing(&self, context: &egui::Context) -> bool {
29        let focused = context.memory(egui::Memory::focused);
30        focused.is_none_or(|id| id == self.input_id())
31            && self.asking.as_ref().is_none_or(|asking| !asking.shown)
32            && self.menu.is_none()
33            && !egui::Popup::is_any_open(context)
34    }
35
36    /// Whether a menu is open that a controller moves in: the object's
37    /// menu showing, a widget's, a drop-down or a menu of the bar.
38    pub(crate) fn menu_open(&self, context: &egui::Context) -> bool {
39        self.object_menu_shown() || self.menu.is_some() || egui::Popup::is_any_open(context)
40    }

42    /// Scroll the window in use as `scroll` asks.
43    pub(crate) fn scroll_in_use(&self, context: &egui::Context, scroll: crate::widget::Scroll) {
44        if let Some(placed) = self.in_use() {
45            crate::widget::ask_scroll(context, draw::widget_id(self.session, placed), scroll);
46        }
47    }

49    /// Do `action`, one of those on the command input or sending from it,
50    /// or on the window in use; the line to send, if it sends one.
51    pub(crate) fn act(
52        &mut self,
53        context: &egui::Context,
54        action: crate::keys::Action,
55    ) -> Option<String> {
56        use crate::keys::Action;
57        use crate::widget::Scroll;
58        let scroll = |scroll| {
59            if let Some(placed) = self.in_use() {
60                crate::widget::ask_scroll(context, draw::widget_id(self.session, placed), scroll);
61            }
62            None
63        };
64        let typed = |back: usize| self.history.ago(back);
65        match action {
66            Action::SendOrRepeat if self.input.trim().is_empty() => typed(1),
67            Action::SendOrRepeat => self.enter(),
68            Action::RepeatLast => typed(1),
69            Action::RepeatSecondLast => typed(2),
70            Action::HistoryBack => {
71                self.history.walk(true, &mut self.input);
72                self.walked = true;
73                None
74            }
75            Action::HistoryForward => {
76                self.history.walk(false, &mut self.input);
77                self.walked = true;
78                None
79            }
80            Action::ClearInput => {
81                self.input.clear();
82                self.history.reset();
83                None
84            }
85            Action::ScrollPageUp => scroll(Scroll::PageUp),
86            Action::ScrollPageDown => scroll(Scroll::PageDown),
87            Action::ScrollLineUp => scroll(Scroll::LineUp),
88            Action::ScrollLineDown => scroll(Scroll::LineDown),
89            Action::ScrollTop => scroll(Scroll::Top),
90            Action::ScrollBottom => scroll(Scroll::Bottom),
91            Action::NextWindow => self.next_window(true),
92            Action::PreviousWindow => self.next_window(false),
93            Action::NextTab => self.turn_tab(true),
94            Action::PreviousTab => self.turn_tab(false),
95            Action::NextUnreadTab => self.unread_tab(),
96            Action::Find => {
97                self.open_find();
98                None
99            }
100            Action::FindNext => {
101                self.step_find(context, true);
102                None
103            }
104            Action::FindPrevious => {
105                self.step_find(context, false);
106                None
107            }
108            Action::TargetNext | Action::TargetPrevious => {
109                // Tab keeps the keyboard on the input: with nothing holding
110                // it, egui gives it to the next widget.
111                context.memory_mut(|memory| memory.request_focus(self.input_id()));
112                // Tab finishes what is typed first (`complete.rs`); with
113                // nothing to finish, it targets.
114                if action == Action::TargetNext
115                    && self.completing.tab(&mut self.input, &self.history)
116                {
117                    self.walked = true;
118                    return None;
119                }
120                let (ids, current) = &self.targets;
121                hydra_ui::widget::target(ids, *current, action == Action::TargetNext)
122            }
123            Action::TargetClear => Some("target clear".to_owned()),
124            Action::DrawerTop => self.drawer(Zone::Top),
125            Action::DrawerBottom => self.drawer(Zone::Bottom),
126            Action::DrawerLeft => self.drawer(Zone::Left),
127            Action::DrawerRight => self.drawer(Zone::Right),
128            Action::Lock => {
129                let layout = self.layout.as_mut()?;
130                layout.locked = !layout.locked;
131                self.arranging &= !layout.locked;
132                self.save();
133                None
134            }
135            Action::Arrange => {
136                let locked = self.layout.as_ref().is_some_and(|layout| layout.locked);
137                self.arranging = !self.arranging && !locked;
138                None
139            }
140            Action::Set(_) | Action::Character(_) => None,
141        }
142    }

144    /// The window after the one in use made the one in use, or the one
145    /// before (`forward` or not), round the windows showing.
146    fn next_window(&mut self, forward: bool) -> Option<String> {
147        let showing = self.layout.as_ref()?.showing();
148        if showing.is_empty() {
149            return None;
150        }
151        let at = self
152            .in_use()
153            .and_then(|id| showing.iter().position(|shown| *shown == id))
154            .unwrap_or(0);
155        let count = showing.len();
156        let next = if forward {
157            (at + 1) % count
158        } else {
159            (at + count - 1) % count
160        };
161        self.in_use = showing.get(next).copied();
162        None
163    }
164
165    /// The next tab of the window in use's stack shown, or the one before;
166    /// it is the window in use now.
167    fn turn_tab(&mut self, forward: bool) -> Option<String> {
168        let in_use = self.in_use()?;
169        let shown = self.layout.as_mut()?.turn_tab(in_use, forward)?;
170        self.in_use = Some(shown);
171        self.save();
172        None
173    }
174
175    /// The first tab with lines unread shown, and in use.
176    fn unread_tab(&mut self) -> Option<String> {
177        let tab = *self.unread_tabs.first()?;
178        if self.layout.as_mut()?.show_tab(tab) {
179            self.in_use = Some(tab);
180            self.save();
181        }
182        None
183    }
184
185    /// The drawer at `zone` opened, or shut.
186    fn drawer(&mut self, zone: Zone) -> Option<String> {
187        let drawer = self.layout.as_mut()?.drawers.get_mut(zone)?;
188        drawer.open = !drawer.open;
189        self.save();
190        None
191    }

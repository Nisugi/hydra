1//! Arranging a custom window's inside. A widget
2//! in a custom window is bare, with no title bar to take hold of, so while
3//! the play window's Layout menu has **Arrange** on, each cell shows its
4//! outline and its widget's name and takes the pointer: dragged from inside
5//! it moves, from near an edge that edge moves, and it lands snapped to the
6//! inside's edges, the other cells and the grid, as a window does one level
7//! up (`crate::snap`). Let go outside its window, its widget leaves: into the
8//! custom window beneath, or into a standalone window of its own
9//! (`Layout::release`).
10//!
11//! A tab stack's tabs stay clickable, and each drags alone; its
12//! body drags the whole stack. Wherever something is let go the rule is the
13//! layout's (`moves.rs`): onto a widget, it joins that widget's tab stack;
14//! onto empty space, it takes a place of its own.
15//!
16//! With Arrange off, a cell takes nothing, so no press in play rearranges
17//! anything: the everyday surface stays simple.
18

27/// How far inside a cell's edge a press takes the edge rather than the cell.
28const EDGE: f32 = 5.0;
29
30/// A cell being moved or resized, or one tab of a stack being moved.
31#[derive(Clone, Copy, Debug, PartialEq)]
32pub(crate) struct CellGesture {
33    /// The custom window it is in.
34    holder: u32,
35    /// What is being dragged: a tab, or the cell showing this widget.
36    taking: Taking,
37    /// What is dragged may be a tab of a stack: a stream.
38    stacks: bool,
39    /// Where it was when the press began, from the inside's top left.
40    start: Rect,
41    /// Which edges the press took; none, and it moves.
42    grab: Grab,
43    /// Where the press began.
44    origin: Pos2,
45    /// Where the pointer was last seen.
46    last: Pos2,
47}
48
49/// Which edge of a cell a press took along one axis.
50#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
51enum Edge {
52    /// Neither.
53    #[default]
54    Neither,
55    /// The left, or the top.
56    Low,
57    /// The right, or the bottom.
58    High,
59}
60
61impl Edge {
62    /// The edge of `low..high` within [`EDGE`] of `at`.
63    fn at(low: f32, high: f32, at: f32) -> Self {
64        if at - low < EDGE {
65            Self::Low
66        } else if high - at < EDGE {
67            Self::High
68        } else {
69            Self::Neither
70        }
71    }
72
73    /// `low..high` with this edge moved by `by`, never nearer the other
74    /// than `smallest`.
75    fn apply(self, low: f32, high: f32, by: f32, smallest: f32) -> (f32, f32) {
76        match self {
77            Self::Neither => (low, high),
78            Self::Low => ((low + by).min(high - smallest), high),
79            Self::High => (low, (high + by).max(low + smallest)),
80        }
81    }
82}

84/// Which edges of a cell a press took: one along each axis, or none, when
85/// the cell moves.
86#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
87struct Grab {
88    across: Edge,
89    down: Edge,
90}
91
92impl Grab {
93    /// The edges within [`EDGE`] of `at`, in `rect`.
94    fn at(rect: Rect, at: Pos2) -> Self {
95        Self {
96            across: Edge::at(rect.min.x, rect.max.x, at.x),
97            down: Edge::at(rect.min.y, rect.max.y, at.y),
98        }
99    }
100
101    /// Whether it took no edge, so the cell moves.
102    fn moves(self) -> bool {
103        self == Self::default()
104    }
105
106    /// The pointer over a cell there.
107    fn cursor(self) -> CursorIcon {
108        match (self.across, self.down) {
109            (Edge::Neither, Edge::Neither) => CursorIcon::Grab,
110            (_, Edge::Neither) => CursorIcon::ResizeHorizontal,
111            (Edge::Neither, _) => CursorIcon::ResizeVertical,
112            (across, down) if across == down => CursorIcon::ResizeNwSe,
113            _ => CursorIcon::ResizeNeSw,
114        }
115    }
116
117    /// `start` moved by `by`: all of it, or only the edges taken, never below
118    /// a cell's smallest.
119    fn apply(self, start: Rect, by: Vec2) -> Rect {
120        if self.moves() {
121            return start.translate(by);
122        }
123        let (left, right) = self
124            .across
125            .apply(start.min.x, start.max.x, by.x, SMALLEST_CELL.x);
126        let (top, bottom) = self
127            .down
128            .apply(start.min.y, start.max.y, by.y, SMALLEST_CELL.y);
129        Rect::from_min_max(Pos2::new(left, top), Pos2::new(right, bottom))
130    }
131}

133/// A widget, or a whole stack, dragged out of its custom window and let go.
134#[derive(Clone, Copy, Debug, PartialEq)]
135pub(crate) struct Released {
136    /// The custom window it left.
137    pub(crate) holder: u32,
138    /// What left.
139    pub(crate) taking: Taking,
140    /// Where it was let go.
141    pub(crate) at: Pos2,
142}

144/// Where a gesture let go inside its own window comes to.
145#[derive(Clone, Copy, Debug, PartialEq)]
146enum Landing {
147    /// Outside the window: it leaves.
148    Out(Released),
149    /// On another cell of the window, whose stack it joins: that cell named
150    /// by a widget it holds.
151    Onto { from: Taking, into: u32 },
152    /// A tab let go on empty space in the window: a cell of its own there,
153    /// from the inside's top left.
154    Apart { tab: u32, at: Pos2 },
155}
156
157/// What every cell of one custom window shares while it is arranged.
158struct Arranging<'a> {
159    holder: u32,
160    inside: Rect,
161    grid: f32,
162    shift: bool,
163    /// Each cell's rect, from the inside's top left, as the frame began.
164    rects: &'a [Rect],
165    /// A widget in each cell, which names the cell.
166    owners: &'a [u32],
167    /// Whether each cell holds streams alone, which something may stack
168    /// onto.
169    stackable: &'a [bool],
170}
171
172/// Custom window `holder`'s cells, drawn at `inside`, arranged as the module
173/// says, their tabs as wide as `tab_looks` chose. A widget or stack let go
174/// outside the window, if one was.
175pub(crate) fn cells(
176    ui: &mut egui::Ui,
177    custom: &mut Custom,
178    (holder, tab_looks): (u32, &BTreeMap<u32, crate::widget::tab::TabLook>),
179    inside: Rect,
180    grid: f32,
181    gesture: &mut Option<CellGesture>,
182) -> Option<Released> {
183    let rects: Vec<Rect> = custom.cells.iter().map(Cell::rect).collect();
184    let owners: Vec<u32> = custom
185        .cells
186        .iter()
187        .map(|cell| cell.shown().map_or(0, |placed| placed.id))
188        .collect();
189    let stackable: Vec<bool> = custom.cells.iter().map(Cell::stacks).collect();
190    let arranging = Arranging {
191        holder,
192        inside,
193        grid,
194        shift: ui.input(|input| input.modifiers.shift),
195        rects: &rects,
196        owners: &owners,
197        stackable: &stackable,
198    };
199    let mut landing = None;
200    let mut reorder = None;
201    for (index, cell) in custom.cells.iter_mut().enumerate() {
202        let rect = rects[index].translate(inside.min.to_vec2());
203        let strip = crate::theme::tab_height(ui.ctx(), crate::widget::LINE);
204        let bar = cell.bar;
205        let widths = crate::widget::tab::widths(&cell.tabs, tab_looks);
206        let (tabs, body) = tabs_and_body(rect, &widths, strip, bar);
207        let mut clicked = None;
208        let stack = cell.tabs.clone();
209        for (at, (tab, tab_rect)) in stack.iter().zip(tabs.iter().copied()).enumerate() {
210            let response = ui.interact(
211                tab_rect,
212                Id::new(("arrange-tab", holder, tab.id)),
213                Sense::click_and_drag(),
214            );
215            if response.clicked() {
216                clicked = Some(at);
217            }
218            // Over its own strip, a dragged tab takes the place it is let
219            // go over; the gesture's landing is nowhere then.
220            if let Some(pointer) = response.interact_pointer_pos()
221                && rect.contains(pointer)
222                && let Some(to) = tab_at(&tabs, pointer, bar.vertical())
223                && to != at
224            {
225                if response.dragged() {
226                    super::draw::slot_mark(ui, &tabs, at, to, bar.vertical());
227                }
228                if response.drag_stopped() {
229                    reorder = Some((index, at, to));
230                }
231            }
232            let taking = (Taking::Tab(tab.id), tab.widget.stacks());
233            let name = tab.widget.name();
234            landing =
235                landing.or(arranging.follow(ui, &response, gesture, taking, index, &name, cell));
236        }
237        if let Some(at) = clicked {
238            cell.showing = at;
239        }
240        let Some(placed) = cell.shown().cloned() else {
241            continue;
242        };
243        let names: Vec<_> = cell.tabs.iter().map(|tab| tab.widget.name()).collect();
244        let names = names.join(", ");
245        let response = overlay(ui, body, holder, placed.id, &names);
246        let taking = (Taking::Cell(placed.id), cell.stacks());
247        landing = landing.or(arranging.follow(ui, &response, gesture, taking, index, &names, cell));
248    }
249    if let Some((cell, at, to)) = reorder
250        && let Some(cell) = custom.cells.get_mut(cell)
251    {
252        cell.move_tab(at, to);
253    }
254    match landing? {
255        Landing::Out(released) => return Some(released),
256        Landing::Onto { from, into } => custom.stack_onto(from, into),
257        Landing::Apart { tab, at } => {
258            if let Some(tab) = custom.take(tab) {
259                custom.land(vec![tab], 0, at);
260            }
261        }
262    }
263    None
264}

266/// A cell's body while arranging: washed, outlined and named, taking the
267/// pointer; its name said to a screen reader, and a test.
268fn overlay(ui: &egui::Ui, body: Rect, holder: u32, placed: u32, names: &str) -> egui::Response {
269    let response = ui.interact(
270        body,
271        Id::new(("arrange-cell", holder, placed)),
272        Sense::drag(),
273    );
274    response.widget_info(|| egui::WidgetInfo::labeled(egui::WidgetType::Other, true, names));
275    let painter = ui.painter();
276    let accent = theme::color(ui.ctx(), T::Accent);
277    let veil = theme::color(ui.ctx(), T::Veil).gamma_multiply(170.0 / 255.0);
278    painter.rect_filled(body, 0.0, veil);
279    painter.rect_stroke(
280        body,
281        0.0,
282        Stroke::new(1.0, accent),
283        egui::StrokeKind::Inside,
284    );
285    painter.text(
286        body.center(),
287        Align2::CENTER_CENTER,
288        names,
289        FontId::proportional(12.0),
290        accent,
291    );
292    if let Some(at) = response.hover_pos() {
293        ui.ctx().set_cursor_icon(Grab::at(body, at).cursor());
294    }
295    response
296}

298impl Arranging<'_> {
299    /// Follow `response` for `taking` (and whether it may stack), cell
300    /// `index`'s, named `name`: a gesture begun on it, the cell following a
301    /// cell's gesture, and where the gesture lands when it is let go, when
302    /// that is not where the cell already is.
303    #[allow(
304        clippy::too_many_arguments,
305        reason = "one call per handle; the rest is the window's, in self"
306    )]
307    fn follow(
308        &self,
309        ui: &egui::Ui,
310        response: &egui::Response,
311        gesture: &mut Option<CellGesture>,
312        (taking, stacks): (Taking, bool),
313        index: usize,
314        name: &str,
315        cell: &mut Cell,
316    ) -> Option<Landing> {
317        if response.drag_started()
318            && let Some(origin) = ui.input(|input| input.pointer.press_origin())
319        {
320            let grab = match taking {
321                Taking::Tab(_) => Grab::default(),
322                Taking::Cell(_) => Grab::at(response.rect, origin),
323            };
324            *gesture = Some(CellGesture {
325                holder: self.holder,
326                taking,
327                stacks,
328                start: self.rects[index],
329                grab,
330                origin,
331                last: origin,
332            });
333        }
334        let ours = gesture
335            .as_mut()
336            .filter(|ours| ours.holder == self.holder && ours.taking == taking)?;
337        if let Some(at) = response.interact_pointer_pos() {
338            ours.last = at;
339        }
340        let pointer = ours.last - self.inside.min.to_vec2();
341        let outside = ours.grab.moves() && !self.inside.contains(ours.last);
342        // The cell it would stack onto: never its own, which a moved cell
343        // lies over by now, and only a cell of streams, for a stream.
344        let onto = self
345            .rects
346            .iter()
347            .enumerate()
348            .rev()
349            .find(|(at, rect)| *at != index && self.stackable[*at] && stacks_at(**rect, pointer))
350            .map(|(at, _)| at)
351            .filter(|_| ours.grab.moves() && ours.stacks);
352        let (snapped, guides) = self.snapped(ours, index);
353        let (grab, last) = (ours.grab, ours.last);
354        if response.dragged() {
355            if outside || matches!(taking, Taking::Tab(_)) {
356                ghost(ui.ctx(), last, name);
357            }
358            if let Some(onto) = onto {
359                // Let go here, it stacks: the cell it would join lights up.
360                let target = self.rects[onto].translate(self.inside.min.to_vec2());
361                let accent = theme::color(ui.ctx(), T::Accent);
362                ui.painter()
363                    .rect_filled(target, 0.0, accent.gamma_multiply(0.35));
364            }
365            if matches!(taking, Taking::Cell(_)) {
366                cell.set(snapped);
367                if !outside {
368                    self.guide(ui, &guides);
369                }
370            }
371            return None;
372        }
373        if !response.drag_stopped() {
374            return None;
375        }
376        *gesture = None;
377        if outside {
378            return Some(Landing::Out(Released {
379                holder: self.holder,
380                taking,
381                at: last,
382            }));
383        }
384        if let Some(onto) = onto {
385            return Some(Landing::Onto {
386                from: taking,
387                into: self.owners[onto],
388            });
389        }
390        match taking {
391            Taking::Tab(tab) if !self.rects[index].contains(pointer) => {
392                Some(Landing::Apart { tab, at: pointer })
393            }
394            Taking::Tab(_) => None,
395            Taking::Cell(_) => {
396                let bounds = Rect::from_min_size(Pos2::ZERO, self.inside.size());
397                cell.set(kept_in(snapped, bounds, grab));
398                None
399            }
400        }
401    }
402
403    /// Where `ours`, on cell `index`, puts the cell now: snapped to the
404    /// inside's edges, the other cells and the grid, unless Shift is held.
405    fn snapped(&self, ours: &CellGesture, index: usize) -> (Rect, Vec<snap::Guide>) {
406        let now = ours.grab.apply(ours.start, ours.last - ours.origin);
407        if self.shift {
408            return (now, Vec::new());
409        }
410        let siblings: Vec<Rect> = self
411            .rects
412            .iter()
413            .enumerate()
414            .filter(|(at, _)| *at != index)
415            .map(|(_, rect)| *rect)
416            .collect();
417        let bounds = Rect::from_min_size(Pos2::ZERO, self.inside.size());
418        snap::snap(ours.start, now, bounds, &siblings, SMALLEST_CELL, self.grid)
419    }
420
421    /// A line for each snap a cell's gesture has engaged, in its window.
422    fn guide(&self, ui: &egui::Ui, guides: &[snap::Guide]) {
423        let accent = Stroke::new(1.5, theme::color(ui.ctx(), T::Accent));
424        for guide in guides {
425            if guide.vertical {
426                ui.painter()
427                    .vline(self.inside.min.x + guide.at, self.inside.y_range(), accent);
428            } else {
429                ui.painter()
430                    .hline(self.inside.x_range(), self.inside.min.y + guide.at, accent);
431            }
432        }
433    }
434}

436/// Where a cell lands, inside its window's inside `bounds`: a move slid back
437/// in, a resize cut at the edge it was dragged past.
438fn kept_in(rect: Rect, bounds: Rect, grab: Grab) -> Rect {
439    if !grab.moves() {
440        return rect.intersect(bounds);
441    }
442    let over = |low: f32, high: f32, min: f32, max: f32| {
443        if low < min {
444            min - low
445        } else if high > max {
446            (max - high).max(min - low)
447        } else {
448            0.0
449        }
450    };
451    rect.translate(Vec2::new(
452        over(rect.min.x, rect.max.x, bounds.min.x, bounds.max.x),
453        over(rect.min.y, rect.max.y, bounds.min.y, bounds.max.y),
454    ))
455}

457/// While a widget is dragged where it would leave its cell -- out of its
458/// window, or a tab anywhere -- its name at the pointer, over everything,
459/// where it would land.
460fn ghost(context: &egui::Context, at: Pos2, name: &str) {
461    let painter = context.layer_painter(LayerId::new(Order::Tooltip, Id::new("arrange-ghost")));
462    let rect = Rect::from_center_size(at, Vec2::new(140.0, 24.0));
463    let accent = theme::color(context, T::Accent);
464    painter.rect_filled(rect, theme::corner(context), accent.gamma_multiply(0.85));
465    painter.text(
466        rect.center(),
467        Align2::CENTER_CENTER,
468        name,
469        FontId::proportional(13.0),
470        readable_on(accent),
471    );
472}

486    /// A press near an edge takes that edge, near a corner both, and inside
487    /// the cell none, so the cell moves; each shows its pointer.
488    #[test]
489    fn a_press_takes_the_edges_it_is_near() {
490        let grab = |across, down| Grab { across, down };
491        assert!(at(140.0, 120.0).moves());
492        assert_eq!(at(140.0, 120.0).cursor(), CursorIcon::Grab);
493        assert_eq!(at(102.0, 120.0), grab(Edge::Low, Edge::Neither));
494        assert_eq!(at(102.0, 120.0).cursor(), CursorIcon::ResizeHorizontal);
495        assert_eq!(at(178.0, 120.0), grab(Edge::High, Edge::Neither));
496        assert_eq!(at(140.0, 102.0), grab(Edge::Neither, Edge::Low));
497        assert_eq!(at(140.0, 138.0), grab(Edge::Neither, Edge::High));
498        assert_eq!(at(140.0, 138.0).cursor(), CursorIcon::ResizeVertical);
499        assert_eq!(at(102.0, 102.0).cursor(), CursorIcon::ResizeNwSe);
500        assert_eq!(at(178.0, 138.0).cursor(), CursorIcon::ResizeNwSe);
501        assert_eq!(at(178.0, 102.0).cursor(), CursorIcon::ResizeNeSw);
502        assert_eq!(at(102.0, 138.0).cursor(), CursorIcon::ResizeNeSw);
503    }

505    /// A move carries the whole cell; an edge moves alone, never past the
506    /// smallest a cell can be.
507    #[test]
508    fn a_grab_moves_what_it_took() {
509        let by = Vec2::new(10.0, 5.0);
510        let rect = |x0: f32, y0: f32, x1: f32, y1: f32| {
511            Rect::from_min_max(Pos2::new(x0, y0), Pos2::new(x1, y1))
512        };
513        let moved = |across, down, by| Grab { across, down }.apply(cell(), by);
514        assert_eq!(Grab::default().apply(cell(), by), cell().translate(by));
515        assert_eq!(
516            moved(Edge::Low, Edge::Neither, by),
517            rect(110.0, 100.0, 180.0, 140.0)
518        );
519        assert_eq!(
520            moved(Edge::High, Edge::Neither, by),
521            rect(100.0, 100.0, 190.0, 140.0)
522        );
523        assert_eq!(
524            moved(Edge::Neither, Edge::Low, by),
525            rect(100.0, 105.0, 180.0, 140.0)
526        );
527        assert_eq!(
528            moved(Edge::Neither, Edge::High, by),
529            rect(100.0, 100.0, 180.0, 145.0)
530        );
531        let far = Vec2::new(500.0, 500.0);
532        assert_eq!(moved(Edge::Low, Edge::Low, far).size(), SMALLEST_CELL);
533        assert_eq!(moved(Edge::High, Edge::High, -far).size(), SMALLEST_CELL);
534    }

1//! The play window's title bar: Hydra's own in place of the system's, when
2//! the player asks, with what it shows chosen on its page.
3//!
4//! The top bar is the title bar. With the system's frame kept it sits
5//! under the system's title bar; as Hydra's own, the system frame goes and
6//! the bar is the window's title: dragged to move the window, double-
7//! clicked to maximize it, the minimize, maximize and close buttons at its
8//! right end, the window's edges dragged to resize it.
9//!
10//! The bar has three sides, left, centre and right, and each holds a row
11//! of **items** in order: the character, its profession, how it is
12//! connected, the room's title and number, each of the menus and switches,
13//! and any **game value** the model can work out by name
14//! (`GameState::evaluate`), shown as its name and its number. The page
15//! shows each side as a row of pickers, one an item, and one more to add
16//! (`crate::titlebar_page`).
17
18use hydra_ui::settings::{Page, Row, RowKind, Value};
19use serde::{Deserialize, Serialize};

21/// The page's id, among the widgets' pages, which the menu sends to the
22/// window.
23pub(crate) const PAGE: &str = "widget:titlebar";

25/// What a game value's key begins with: `value:Stats.level`.
26pub(crate) const VALUE: &str = "value:";

28/// A thing the bar can show.
29#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
30#[serde(rename_all = "snake_case")]
31pub(crate) enum Item {
32    /// The character's name.
33    Character,
34    /// Its profession.
35    Profession,
36    /// How it is connected.
37    Connection,
38    /// The room's title.
39    Room,
40    /// The game's number for the room.
41    Number,
42    Lock,
43    Arrange,
44    Drawers,
45    Layout,
46    Keys,
47    Log,
48    /// The Menu the hub has too: the editors, Settings, the tools.
49    Menu,
50    Settings,
51    /// The − and + buttons that step the zoom.
52    Zoom,
53    /// A value the model works out by name, shown as the name and its
54    /// number: `Stats.level 100`.
55    Value(String),
56}

58impl Item {
59    /// Every fixed item, in the order a picker lists them.
60    pub(crate) const FIXED: [Item; 14] = [
61        Item::Character,
62        Item::Profession,
63        Item::Connection,
64        Item::Room,
65        Item::Number,
66        Item::Lock,
67        Item::Arrange,
68        Item::Drawers,
69        Item::Layout,
70        Item::Keys,
71        Item::Log,
72        Item::Menu,
73        Item::Settings,
74        Item::Zoom,
75    ];
76
77    /// Its key on the page and in a change: a word, or `value:<name>`.
78    pub(crate) fn key(&self) -> String {
79        match self {
80            Item::Character => "character".to_owned(),
81            Item::Profession => "profession".to_owned(),
82            Item::Connection => "connection".to_owned(),
83            Item::Room => "room".to_owned(),
84            Item::Number => "number".to_owned(),
85            Item::Lock => "lock".to_owned(),
86            Item::Arrange => "arrange".to_owned(),
87            Item::Drawers => "drawers".to_owned(),
88            Item::Layout => "layout".to_owned(),
89            Item::Keys => "keys".to_owned(),
90            Item::Log => "log".to_owned(),
91            Item::Menu => "menu".to_owned(),
92            Item::Settings => "settings".to_owned(),
93            Item::Zoom => "zoom".to_owned(),
94            Item::Value(name) => format!("{VALUE}{name}"),
95        }
96    }
97
98    /// The item `key` names; none for a word that is not one.
99    pub(crate) fn of_key(key: &str) -> Option<Item> {
100        let key = key.trim();
101        if let Some(name) = key.strip_prefix(VALUE) {
102            let name = name.trim();
103            return (!name.is_empty()).then(|| Item::Value(name.to_owned()));
104        }
105        Item::FIXED.into_iter().find(|item| item.key() == key)
106    }
107
108    /// What a player calls it.
109    pub(crate) fn label(&self) -> String {
110        match self {
111            Item::Character => "Character".to_owned(),
112            Item::Profession => "Profession".to_owned(),
113            Item::Connection => "Connection".to_owned(),
114            Item::Room => "Room title".to_owned(),
115            Item::Number => "Room number".to_owned(),
116            Item::Lock => "Lock".to_owned(),
117            Item::Arrange => "Arrange".to_owned(),
118            Item::Drawers => "Drawers menu".to_owned(),
119            Item::Layout => "Layout menu".to_owned(),
120            Item::Keys => "Keys menu".to_owned(),
121            Item::Log => "Log".to_owned(),
122            Item::Menu => "Menu".to_owned(),
123            Item::Settings => "Settings".to_owned(),
124            Item::Zoom => "Zoom − and +".to_owned(),
125            Item::Value(name) => name.clone(),
126        }
127    }
128}

165/// What the title bar shows, kept with the layout; read as [`Written`].
166#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
167#[serde(from = "Written")]
168pub(crate) struct TitleBar {
169    /// Hydra's own title bar in place of the system's.
170    #[serde(skip_serializing_if = "std::ops::Not::not")]
171    pub(crate) own: bool,
172    /// The left side's items, in order.
173    pub(crate) left: Vec<Item>,
174    /// The centre's items, in order.
175    #[serde(skip_serializing_if = "Vec::is_empty")]
176    pub(crate) center: Vec<Item>,
177    /// The right side's items, in order, before the window's own buttons.
178    pub(crate) right: Vec<Item>,
179    /// The Menu was put on the bar: at first, or once into a bar kept
180    /// before the Menu was one of its items. Taken off after, it stays off.
181    menu: bool,
182}
183
184/// A title bar as a layout file holds it, which may be from before the
185/// Menu was one of its items.
186#[derive(Deserialize)]
187struct Written {
188    #[serde(default)]
189    own: bool,
190    #[serde(default = "left_at_first")]
191    left: Vec<Item>,
192    #[serde(default)]
193    center: Vec<Item>,
194    #[serde(default = "right_at_first")]
195    right: Vec<Item>,
196    #[serde(default)]
197    menu: bool,
198}
199
200impl From<Written> for TitleBar {
201    /// The bar as written; one kept before the Menu was an item gains it,
202    /// before *Settings* on the right, unless a side holds it already.
203    fn from(written: Written) -> Self {
204        let mut bar = TitleBar {
205            own: written.own,
206            left: written.left,
207            center: written.center,
208            right: written.right,
209            menu: true,
210        };
211        let shown = Side::ALL
212            .into_iter()
213            .any(|side| bar.on(side).contains(&Item::Menu));
214        if !written.menu && !shown {
215            let at = bar
216                .right
217                .iter()
218                .position(|item| *item == Item::Settings)
219                .unwrap_or(bar.right.len());
220            bar.right.insert(at, Item::Menu);
221        }
222        bar
223    }
224}
225
226/// The left side as it starts: the character and its connection. The
227/// system's title bar names the room already; Hydra's own adds it.
228fn left_at_first() -> Vec<Item> {
229    vec![Item::Character, Item::Connection]
230}
231
232/// The right side as it starts: every menu and switch.
233fn right_at_first() -> Vec<Item> {
234    vec![
235        Item::Lock,
236        Item::Arrange,
237        Item::Drawers,
238        Item::Layout,
239        Item::Keys,
240        Item::Log,
241        Item::Menu,
242        Item::Settings,
243        Item::Zoom,
244    ]
245}
246
247impl Default for TitleBar {
248    fn default() -> Self {
249        Self {
250            own: false,
251            left: left_at_first(),
252            center: Vec::new(),
253            right: right_at_first(),
254            menu: true,
255        }
256    }
257}

259/// The page's key for the switch.
260pub(crate) const OWN: &str = "own";
261
262/// Items as a page's value or a change writes them: one key a line.
263pub(crate) fn written(items: &[Item]) -> String {
264    items.iter().map(Item::key).collect::<Vec<_>>().join("\n")
265}
266
267/// The items `written` names, one key a line; a line naming none is an
268/// error.
269pub(crate) fn read(written: &str) -> Result<Vec<Item>, String> {
270    written
271        .lines()
272        .map(str::trim)
273        .filter(|line| !line.is_empty())
274        .map(|line| {
275            Item::of_key(line).ok_or_else(|| format!("`{line}` is not a thing the bar shows."))
276        })
277        .collect()
278}
279
280impl TitleBar {
281    /// The items on `side`, in order.
282    pub(crate) fn on(&self, side: Side) -> &[Item] {
283        match side {
284            Side::Left => &self.left,
285            Side::Center => &self.center,
286            Side::Right => &self.right,
287        }
288    }
289
290    fn on_mut(&mut self, side: Side) -> &mut Vec<Item> {
291        match side {
292            Side::Left => &mut self.left,
293            Side::Center => &mut self.center,
294            Side::Right => &mut self.right,
295        }
296    }
297
298    /// Its page: the switch, then each side's items.
299    pub(crate) fn page(&self, file: String) -> Page {
300        let default = TitleBar::default();
301        let mut rows = vec![Row {
302            key: OWN.to_owned(),
303            label: "Hydra's own title bar".to_owned(),
304            help: "The top bar in place of the system's title bar: drag it to move the \
305                   window, double-click it to maximize, with the window's buttons at its \
306                   right end; the window's edges resize it. Off, the system's title bar \
307                   stays above the top bar."
308                .to_owned(),
309            kind: RowKind::Toggle,
310            value: Value::On(self.own),
311            here: self.own,
312            from: None,
313        }];
314        for side in Side::ALL {
315            rows.push(Row {
316                key: side.key().to_owned(),
317                label: side.label().to_owned(),
318                help: format!(
319                    "What the bar's {} shows, in order.",
320                    side.label().to_lowercase()
321                ),
322                kind: RowKind::Words,
323                value: Value::Text(written(self.on(side))),
324                here: self.on(side) != default.on(side),
325                from: None,
326            });
327        }
328        Page {
329            id: PAGE.to_owned(),
330            title: "Title bar".to_owned(),
331            file,
332            takes: "at once".to_owned(),
333            problem: None,
334            rows,
335        }
336    }
337
338    /// Set `key` to `to`, or back to its own. What was done.
339    pub(crate) fn set(&mut self, key: &str, to: Option<&str>) -> Result<String, String> {
340        if key == OWN {
341            self.own = match to.map(str::trim) {
342                None | Some("off") => false,
343                Some("on") => true,
344                Some(other) => return Err(format!("`{other}` is not on or off.")),
345            };
346            if self.own {
347                // Its own bar is the window's title now: the room and its
348                // number join the left, unless the player chose the left.
349                if self.left == left_at_first() {
350                    self.left.extend([Item::Room, Item::Number]);
351                }
352                return Ok("The window wears Hydra's own title bar.".to_owned());
353            }
354            return Ok("The window wears the system's title bar.".to_owned());
355        }
356        let side =
357            Side::of_key(key).ok_or_else(|| format!("The title bar has no setting {key}."))?;
358        *self.on_mut(side) = match to {
359            None => TitleBar::default().on(side).to_vec(),
360            Some(written) => read(written)?,
361        };
362        let items = self.on(side);
363        Ok(if items.is_empty() {
364            format!("The bar's {} shows nothing.", side.label().to_lowercase())
365        } else {
366            format!(
367                "The bar's {} shows {}.",
368                side.label().to_lowercase(),
369                items.iter().map(Item::label).collect::<Vec<_>>().join(", ")
370            )
371        })
372    }
373}

379    /// A side is set as its keys, one a line, a game value among them;
380    /// put back, it is its own again; a word that is nothing is refused.
381    #[test]
382    fn a_side_is_its_keys_in_order() {
383        let mut bar = TitleBar::default();
384        assert_eq!(bar.on(Side::Left), [Item::Character, Item::Connection]);
385        assert_eq!(bar.on(Side::Right).len(), 9);
386        bar.set("center", Some("room\nvalue:Stats.level\nnumber"))
387            .expect("set");
388        assert_eq!(
389            bar.on(Side::Center),
390            [
391                Item::Room,
392                Item::Value("Stats.level".to_owned()),
393                Item::Number
394            ]
395        );
396        assert!(bar.set("left", Some("weather")).is_err());
397        assert!(bar.set("middle", Some("room")).is_err());
398        bar.set("right", Some("")).expect("none");
399        assert!(bar.on(Side::Right).is_empty());
400        bar.set("right", None).expect("back");
401        bar.set("center", None).expect("back");
402        assert_eq!(bar, TitleBar::default());
403        let page = bar.page(String::new());
404        assert_eq!(page.id, PAGE);
405        assert_eq!(
406            page.rows[1].value,
407            Value::Text("character\nconnection".to_owned())
408        );
409    }

411    /// Switched on, Hydra's own bar names the room too, unless the player
412    /// chose the left already.
413    #[test]
414    fn its_own_bar_shows_the_room() {
415        let mut bar = TitleBar::default();
416        bar.set("own", Some("on")).expect("on");
417        assert!(bar.own);
418        assert_eq!(
419            bar.on(Side::Left),
420            [Item::Character, Item::Connection, Item::Room, Item::Number]
421        );
422        let mut chosen = TitleBar::default();
423        chosen.set("left", Some("profession")).expect("set");
424        chosen.set("own", Some("on")).expect("on");
425        assert_eq!(
426            chosen.on(Side::Left),
427            [Item::Profession],
428            "a choice is kept"
429        );
430    }

432    /// A layout kept before the title bar had a page reads with the bar
433    /// as it was; a bar written reads back whole.
434    #[test]
435    fn an_old_file_reads_as_the_default() {
436        let bar: TitleBar = serde_json::from_str("{}").expect("read");
437        assert_eq!(bar, TitleBar::default());
438        let mut bar = TitleBar::default();
439        bar.center.push(Item::Value("Society.rank".to_owned()));
440        let text = serde_json::to_string(&bar).expect("written");
441        let back: TitleBar = serde_json::from_str(&text).expect("read");
442        assert_eq!(back, bar);
443    }

445    /// A bar kept before the Menu was an item gains it once, before
446    /// Settings; taken off and kept, it stays off. One that shows it
447    /// already is left as it is.
448    #[test]
449    fn an_old_bar_gains_the_menu_once() {
450        let old = r#"{"left":["character"],"right":["lock","settings","zoom"]}"#;
451        let mut bar: TitleBar = serde_json::from_str(old).expect("read");
452        assert_eq!(
453            bar.on(Side::Right),
454            [Item::Lock, Item::Menu, Item::Settings, Item::Zoom]
455        );
456        bar.set("right", Some("lock\nsettings")).expect("set");
457        let text = serde_json::to_string(&bar).expect("written");
458        let back: TitleBar = serde_json::from_str(&text).expect("read");
459        assert_eq!(back.on(Side::Right), [Item::Lock, Item::Settings]);
460        let moved = r#"{"left":["menu"],"right":["log"]}"#;
461        let bar: TitleBar = serde_json::from_str(moved).expect("read");
462        assert_eq!(bar.on(Side::Right), [Item::Log]);
463        assert_eq!(bar.on(Side::Left), [Item::Menu]);
464    }

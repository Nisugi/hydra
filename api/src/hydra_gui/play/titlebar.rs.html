1//! The play window's title bar: Hydra's own in place of the system's, when
2//! the player asks, with what it shows chosen on its page.
3//!
4//! The top bar is the title bar. With the system's frame kept it sits
5//! under the system's title bar; as Hydra's own, the system frame goes and
6//! the bar is the window's title: dragged to move the window, double-
7//! clicked to maximize it, the minimize, maximize and close buttons at its
8//! right end, the window's edges dragged to resize it.
9//!
10//! The bar has three sides, left, centre and right, and each holds a row
11//! of **items** in order: the character, its profession, how it is
12//! connected, the room's title and number, each of the menus and switches,
13//! and any **game value** the model can work out by name
14//! (`GameState::evaluate`), shown as its name and its number. The page
15//! shows each side as a row of pickers, one an item, and one more to add
16//! (`crate::titlebar_page`).
17
18use hydra_ui::settings::{Page, Row, RowKind, Value};
19use serde::{Deserialize, Serialize};

21/// The page's id, among the widgets' pages, which the menu sends to the

23pub(crate) const PAGE: &str = "widget:titlebar";

25/// What a game value's key begins with: `value:Stats.level`.
26pub(crate) const VALUE: &str = "value:";

28/// A thing the bar can show.
29#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
30#[serde(rename_all = "snake_case")]
31pub(crate) enum Item {
32    /// The character's name.
33    Character,
34    /// Its profession.
35    Profession,
36    /// How it is connected.
37    Connection,
38    /// The room's title.
39    Room,
40    /// The game's number for the room.
41    Number,
42    Lock,
43    Arrange,
44    Drawers,
45    Layout,
46    Keys,
47    Log,
48    /// The Menu the hub has too: the editors, Settings, the tools.
49    Menu,
50    Settings,
51    /// The − and + buttons that step the zoom.
52    Zoom,
53    /// An icon opening or shutting the top drawer, lit while it is open.
54    TopDrawer,
55    /// The bottom drawer's icon.
56    BottomDrawer,
57    /// The left drawer's icon.
58    LeftDrawer,
59    /// The right drawer's icon.
60    RightDrawer,
61    /// A value the model works out by name, shown as the name and its
62    /// number: `Stats.level 100`.
63    Value(String),
64}

102            Item::Log => "log".to_owned(),

114    /// The item `key` names; none for a word that is not one.
115    pub(crate) fn of_key(key: &str) -> Option<Item> {
116        let key = key.trim();
117        if let Some(name) = key.strip_prefix(VALUE) {
118            let name = name.trim();
119            return (!name.is_empty()).then(|| Item::Value(name.to_owned()));
120        }
121        Item::FIXED.into_iter().find(|item| item.key() == key)
122    }

197/// What the title bar shows, kept with the layout; read as [`Written`].
198#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
199#[serde(from = "Written")]
200pub(crate) struct TitleBar {
201    /// Hydra's own title bar in place of the system's.
202    #[serde(skip_serializing_if = "std::ops::Not::not")]
203    pub(crate) own: bool,
204    /// The left side's items, in order.
205    pub(crate) left: Vec<Item>,
206    /// The centre's items, in order.
207    #[serde(skip_serializing_if = "Vec::is_empty")]
208    pub(crate) center: Vec<Item>,
209    /// The right side's items, in order, before the window's own buttons.
210    pub(crate) right: Vec<Item>,
211    /// The Menu was put on the bar: at first, or once into a bar kept
212    /// before the Menu was one of its items. Taken off after, it stays off.
213    menu: bool,
214}
215
216/// A title bar as a layout file holds it, which may be from before the
217/// Menu was one of its items.
218#[derive(Deserialize)]
219struct Written {
220    #[serde(default)]
221    own: bool,
222    #[serde(default = "left_at_first")]
223    left: Vec<Item>,
224    #[serde(default)]
225    center: Vec<Item>,
226    #[serde(default = "right_at_first")]
227    right: Vec<Item>,
228    #[serde(default)]
229    menu: bool,
230}
231
232impl From<Written> for TitleBar {
233    /// The bar as written; one kept before the Menu was an item gains it,
234    /// before *Settings* on the right, unless a side holds it already.
235    fn from(written: Written) -> Self {
236        let mut bar = TitleBar {
237            own: written.own,
238            left: written.left,
239            center: written.center,
240            right: written.right,
241            menu: true,
242        };
243        let shown = Side::ALL
244            .into_iter()
245            .any(|side| bar.on(side).contains(&Item::Menu));
246        if !written.menu && !shown {
247            let at = bar
248                .right
249                .iter()
250                .position(|item| *item == Item::Settings)
251                .unwrap_or(bar.right.len());
252            bar.right.insert(at, Item::Menu);
253        }
254        bar
255    }
256}
257
258/// The left side as it starts: the character and its connection. The
259/// system's title bar names the room already; Hydra's own adds it.
260fn left_at_first() -> Vec<Item> {
261    vec![Item::Character, Item::Connection]
262}
263
264/// The right side as it starts: the Menu, which holds every menu and
265/// switch the bar could, and the zoom, which the Menu does not. Any of the
266/// rest is added back on the *Window* page.
267fn right_at_first() -> Vec<Item> {
268    vec![Item::Menu, Item::Zoom]
269}
270
271impl Default for TitleBar {
272    fn default() -> Self {
273        Self {
274            own: false,
275            left: left_at_first(),
276            center: Vec::new(),
277            right: right_at_first(),
278            menu: true,
279        }
280    }
281}

283/// The page's key for the switch.
284pub(crate) const OWN: &str = "own";
285
286/// Whether `key` is one of the title bar's own rows -- the switch, or a
287/// side -- which its page draws as pickers, apart from the *Window* page's
288/// other rows.
289pub(crate) fn is_bar_key(key: &str) -> bool {
290    key == OWN || Side::of_key(key).is_some()
291}
292
293/// Items as a page's value or a change writes them: one key a line.
294pub(crate) fn written(items: &[Item]) -> String {
295    items.iter().map(Item::key).collect::<Vec<_>>().join("\n")
296}
297
298/// The items `written` names, one key a line; a line naming none is an
299/// error.
300pub(crate) fn read(written: &str) -> Result<Vec<Item>, String> {
301    written
302        .lines()
303        .map(str::trim)
304        .filter(|line| !line.is_empty())
305        .map(|line| {
306            Item::of_key(line).ok_or_else(|| format!("`{line}` is not a thing the bar shows."))
307        })
308        .collect()
309}
310
311impl TitleBar {
312    /// The items on `side`, in order.
313    pub(crate) fn on(&self, side: Side) -> &[Item] {
314        match side {
315            Side::Left => &self.left,
316            Side::Center => &self.center,
317            Side::Right => &self.right,
318        }
319    }
320
321    fn on_mut(&mut self, side: Side) -> &mut Vec<Item> {
322        match side {
323            Side::Left => &mut self.left,
324            Side::Center => &mut self.center,
325            Side::Right => &mut self.right,
326        }
327    }
328
329    /// Its page, the play window's *Window* page: the switch, then each
330    /// side's items; the window's arranging rows go under them.
331    pub(crate) fn page(&self, file: String) -> Page {
332        let default = TitleBar::default();
333        let mut rows = vec![Row {
334            key: OWN.to_owned(),
335            label: "Hydra's own title bar".to_owned(),
336            help: "The top bar in place of the system's title bar: drag it to move the \
337                   window, double-click it to maximize, with the window's buttons at its \
338                   right end; the window's edges resize it. Off, the system's title bar \
339                   stays above the top bar."
340                .to_owned(),
341            kind: RowKind::Toggle,
342            value: Value::On(self.own),
343            here: self.own,
344            from: None,
345            needs: None,
346        }];
347        for side in Side::ALL {
348            rows.push(Row {
349                key: side.key().to_owned(),
350                label: side.label().to_owned(),
351                help: format!(
352                    "What the bar's {} shows, in order.",
353                    side.label().to_lowercase()
354                ),
355                kind: RowKind::Words,
356                value: Value::Text(written(self.on(side))),
357                here: self.on(side) != default.on(side),
358                from: None,
359                needs: None,
360            });
361        }
362        Page {
363            id: PAGE.to_owned(),
364            title: "Window".to_owned(),
365            file,
366            takes: "at once".to_owned(),
367            problem: None,
368            rows,
369        }
370    }
371
372    /// Set `key` to `to`, or back to its own. What was done.
373    pub(crate) fn set(&mut self, key: &str, to: Option<&str>) -> Result<String, String> {
374        if key == OWN {
375            self.own = match to.map(str::trim) {
376                None | Some("off") => false,
377                Some("on") => true,
378                Some(other) => return Err(format!("`{other}` is not on or off.")),
379            };
380            if self.own {
381                // Its own bar is the window's title now: the room and its
382                // number join the left, unless the player chose the left.
383                if self.left == left_at_first() {
384                    self.left.extend([Item::Room, Item::Number]);
385                }
386                return Ok("The window wears Hydra's own title bar.".to_owned());
387            }
388            return Ok("The window wears the system's title bar.".to_owned());
389        }
390        let side =
391            Side::of_key(key).ok_or_else(|| format!("The title bar has no setting {key}."))?;
392        *self.on_mut(side) = match to {
393            None => TitleBar::default().on(side).to_vec(),
394            Some(written) => read(written)?,
395        };
396        let items = self.on(side);
397        Ok(if items.is_empty() {
398            format!("The bar's {} shows nothing.", side.label().to_lowercase())
399        } else {
400            format!(
401                "The bar's {} shows {}.",
402                side.label().to_lowercase(),
403                items.iter().map(Item::label).collect::<Vec<_>>().join(", ")
404            )
405        })
406    }
407}

413    /// A side is set as its keys, one a line, a game value among them;
414    /// put back, it is its own again; a word that is nothing is refused.
415    #[test]
416    fn a_side_is_its_keys_in_order() {
417        let mut bar = TitleBar::default();
418        assert_eq!(bar.on(Side::Left), [Item::Character, Item::Connection]);
419        assert_eq!(
420            bar.on(Side::Right),
421            [Item::Menu, Item::Zoom],
422            "the Menu and the zoom alone"
423        );
424        bar.set("center", Some("room\nvalue:Stats.level\nnumber"))
425            .expect("set");
426        assert_eq!(
427            bar.on(Side::Center),
428            [
429                Item::Room,
430                Item::Value("Stats.level".to_owned()),
431                Item::Number
432            ]
433        );
434        assert!(bar.set("left", Some("weather")).is_err());
435        assert!(bar.set("middle", Some("room")).is_err());
436        bar.set("right", Some("")).expect("none");
437        assert!(bar.on(Side::Right).is_empty());
438        bar.set("right", None).expect("back");
439        bar.set("center", None).expect("back");
440        assert_eq!(bar, TitleBar::default());
441        let page = bar.page(String::new());
442        assert_eq!(page.id, PAGE);
443        assert_eq!(
444            page.rows[1].value,
445            Value::Text("character\nconnection".to_owned())
446        );
447    }

449    /// Switched on, Hydra's own bar names the room too, unless the player
450    /// chose the left already.
451    #[test]
452    fn its_own_bar_shows_the_room() {
453        let mut bar = TitleBar::default();
454        bar.set("own", Some("on")).expect("on");
455        assert!(bar.own);
456        assert_eq!(
457            bar.on(Side::Left),
458            [Item::Character, Item::Connection, Item::Room, Item::Number]
459        );
460        let mut chosen = TitleBar::default();
461        chosen.set("left", Some("profession")).expect("set");
462        chosen.set("own", Some("on")).expect("on");
463        assert_eq!(
464            chosen.on(Side::Left),
465            [Item::Profession],
466            "a choice is kept"
467        );
468    }

470    /// A layout kept before the title bar had a page reads with the bar
471    /// as it was; a bar written reads back whole.
472    #[test]
473    fn an_old_file_reads_as_the_default() {
474        let bar: TitleBar = serde_json::from_str("{}").expect("read");
475        assert_eq!(bar, TitleBar::default());
476        let mut bar = TitleBar::default();
477        bar.center.push(Item::Value("Society.rank".to_owned()));
478        let text = serde_json::to_string(&bar).expect("written");
479        let back: TitleBar = serde_json::from_str(&text).expect("read");
480        assert_eq!(back, bar);
481    }

483    /// A bar kept before the Menu was an item gains it once, before
484    /// Settings; taken off and kept, it stays off. One that shows it
485    /// already is left as it is.
486    #[test]
487    fn an_old_bar_gains_the_menu_once() {
488        let old = r#"{"left":["character"],"right":["lock","settings","zoom"]}"#;
489        let mut bar: TitleBar = serde_json::from_str(old).expect("read");
490        assert_eq!(
491            bar.on(Side::Right),
492            [Item::Lock, Item::Menu, Item::Settings, Item::Zoom]
493        );
494        bar.set("right", Some("lock\nsettings")).expect("set");
495        let text = serde_json::to_string(&bar).expect("written");
496        let back: TitleBar = serde_json::from_str(&text).expect("read");
497        assert_eq!(back.on(Side::Right), [Item::Lock, Item::Settings]);
498        let moved = r#"{"left":["menu"],"right":["log"]}"#;
499        let bar: TitleBar = serde_json::from_str(moved).expect("read");
500        assert_eq!(bar.on(Side::Right), [Item::Log]);
501        assert_eq!(bar.on(Side::Left), [Item::Menu]);
502    }

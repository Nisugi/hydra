16/// `look`'s `key` set to `to`, or to `default`'s.
17pub(super) fn set(
18    look: &mut Look,
19    default: &Look,
20    key: &str,
21    to: Option<&str>,
22) -> Result<(), String> {
23    let on = on_off;
24    match (key, to) {
25        ("fills", Some(to)) => {
26            look.fills = FILLS
27                .iter()
28                .find(|(_, value, _)| *value == to)
29                .map(|(fills, ..)| *fills)
30                .ok_or_else(|| format!("A bar does not fill `{to}`."))?;
31            // A panel's words go above it and in its corner, unless the
32            // player put them somewhere already.
33            if look.fills == Fills::Panel && look.place == default.place {
34                look.place = Place::Corner;
35            }
36        }
37        ("fills", None) => look.fills = default.fills,
38        ("emblem", Some("second")) => look.second_emblem = true,
39        ("emblem", Some(_) | None) => look.second_emblem = false,
40        ("text", Some(to)) => {
41            look.place = PLACES
42                .iter()
43                .find(|(_, value, _)| *value == to)
44                .map(|(place, ..)| *place)
45                .ok_or_else(|| format!("A bar's text does not go `{to}`."))?;
46        }
47        ("text", None) => look.place = default.place,
48        ("label", to) => look.says.label = to.map_or(Ok(default.says.label), on)?,
49        ("numbers", to) => look.says.numbers = to.map_or(Ok(default.says.numbers), on)?,
50        ("percent", to) => look.says.percent = to.map_or(Ok(default.says.percent), on)?,
51        ("words", to) => look.says.words = to.map_or(Ok(default.says.words), on)?,
52        ("clock", to) => look.clock = to.map_or(Ok(default.clock), on)?,
53        ("color", Some(to)) => {
54            look.color =
55                Some(crate::menu::rgb(to).ok_or_else(|| format!("`{to}` is not a colour."))?);
56        }
57        ("color", None) => look.color = None,
58        ("text_color", Some(to)) => {
59            look.text_color =
60                Some(crate::menu::rgb(to).ok_or_else(|| format!("`{to}` is not a colour."))?);
61        }
62        ("text_color", None) => look.text_color = None,
63        ("outline_color", Some(to)) => {
64            look.outline_color =
65                Some(crate::menu::rgb(to).ok_or_else(|| format!("`{to}` is not a colour."))?);
66        }
67        ("outline_color", None) => look.outline_color = None,
68        ("outline", Some(to)) => {
69            look.outline = to
70                .trim()
71                .parse::<u8>()
72                .ok()
73                .filter(|width| *width <= Look::OUTLINE_MOST)
74                .ok_or_else(|| {
75                    format!("`{to}` is not a width from 0 to {}.", Look::OUTLINE_MOST)
76                })?;
77        }
78        ("outline", None) => look.outline = default.outline,
79        ("ring", Some(to)) => {
80            look.ring = to
81                .parse::<u8>()
82                .ok()
83                .filter(|width| (RING_LEAST..=100).contains(width))
84                .ok_or_else(|| format!("`{to}` is not a thickness from {RING_LEAST} to 100."))?;
85        }
86        ("ring", None) => look.ring = default.ring,
87        ("overlay", Some("") | None) => look.overlay = None,
88        ("overlay", Some(path)) => look.overlay = Some(path.to_owned()),
89        ("background", Some("") | None) => look.background = None,
90        ("background", Some(path)) => look.background = Some(path.to_owned()),
91        ("track", None) => look.track = default.track,
92        ("track", Some(to)) => {
93            look.track = TRACKS
94                .iter()
95                .find(|(_, value, _)| *value == to)
96                .map(|(track, ..)| *track)
97                .ok_or_else(|| format!("`{to}` is not a track."))?;
98        }
99        ("fill_image", Some("") | None) => look.fill_image = None,
100        ("fill_image", Some(path)) => look.fill_image = Some(path.to_owned()),
101        (THRESHOLDS, Some(written)) => look.thresholds = thresholds(written)?,
102        (THRESHOLDS, None) => look.thresholds.clone_from(&default.thresholds),
103        (key, _) => return Err(format!("A bar has no setting {key}.")),
104    }
105    Ok(())
106}

111/// The thresholds a table `written` says, each row a percent and a colour,
112/// lowest first; a row with both empty left out.
113pub(crate) fn thresholds(written: &str) -> Result<Vec<Threshold>, String> {
114    let rows: Vec<Vec<String>> =
115        serde_json::from_str(written).map_err(|_| "not rows of cells".to_owned())?;
116    let mut thresholds = rows
117        .into_iter()
118        .filter_map(|mut cells| {
119            cells.resize(2, String::new());
120            let (from, color) = (cells[0].trim().to_owned(), cells[1].trim().to_owned());
121            if from.is_empty() && color.is_empty() {
122                return None;
123            }
124            Some(
125                from.trim_end_matches('%')
126                    .parse::<u8>()
127                    .ok()
128                    .filter(|from| *from <= 100)
129                    .ok_or_else(|| format!("{from:?} is not a percent from 0 to 100"))
130                    .and_then(|from| {
131                        crate::menu::rgb(&color)
132                            .map(|color| Threshold { from, color })
133                            .ok_or_else(|| format!("{color:?} is not a colour #rrggbb"))
134                    }),
135            )
136        })
137        .collect::<Result<Vec<_>, _>>()?;
138    thresholds.sort_by_key(|threshold| threshold.from);
139    Ok(thresholds)
140}

142/// The thresholds row: from each percent up, the bar's colour.
143fn thresholds_row(now: &Look, default: &Look) -> Row {
144    Row {
145        key: THRESHOLDS.to_owned(),
146        label: "Colour by how full".to_owned(),
147        help: "From each percent up, the bar fills in that row's colour until a higher \
148               row's percent; below them all, its Colour. As many rows as you like."
149            .to_owned(),
150        kind: RowKind::Table(vec![Column::words("From %"), Column::colour("Colour")]),
151        value: Value::Table(
152            now.thresholds
153                .iter()
154                .map(|threshold| {
155                    vec![
156                        threshold.from.to_string(),
157                        crate::menu::hex(threshold.color),
158                    ]
159                })
160                .collect(),
161        ),
162        here: now.thresholds != default.thresholds,
163        from: None,
164        needs: None,
165    }
166}

186/// A bar widget's rows: how it draws, as `look` says, or its kind's own;
187/// `clock`, the pulse's clock row; `emblems`, its stat's two emblems'
188/// names, when it has a second.
189pub(super) fn bar_rows(
190    look: Option<&Look>,
191    default: &Look,
192    pictures: &Pictures,
193    (clock, emblems): (bool, Option<(&str, &str)>),
194) -> Vec<Row> {
195    let now = look.unwrap_or(default);
196    let row = |key: &str, label: &str, help: &str, kind: RowKind, value: Value, here: bool| Row {
197        key: key.to_owned(),
198        label: label.to_owned(),
199        help: help.to_owned(),
200        kind,
201        value,
202        here,
203        from: None,
204        needs: None,
205    };
206    let says = |key: &str, label: &str, help: &str, now: bool, default: bool| {
207        row(
208            key,
209            label,
210            help,
211            RowKind::Toggle,
212            Value::On(now),
213            now != default,
214        )
215    };
216    vec![
217        row(
218            "fills",
219            "Fills",
220            "Across or upright and from which edge, or round: an orb or a ring. A bar takes its whole space, a round one the square in its middle: size it by its window or cell.",
221            choice(FILLS.iter().map(|(_, value, called)| (*value, *called))),
222            Value::Text(fills_value(now.fills)),
223            now.fills != default.fills,
224        ),
225        row(
226            "ring",
227            "Ring thickness",
228            "A ring's thickness, in percent of its radius: 100 is a disc.",
229            RowKind::Whole {
230                min: u32::from(RING_LEAST),
231                max: 100,
232            },
233            Value::Text(now.ring.to_string()),
234            now.ring != default.ring,
235        ),
236        row(
237            "text",
238            "Text",
239            "Where its words go: inside, beside it, or none.",
240            choice(PLACES.iter().map(|(_, value, called)| (*value, *called))),
241            Value::Text(place_value(now.place)),
242            now.place != default.place,
243        ),
244        says(
245            "label",
246            "Says its label",
247            "Its name: HP, MP, Stance, Mind.",
248            now.says.label,
249            default.says.label,
250        ),
251        // A bar the game words has its word to say, and no current/max.
252        if default.says.words {
253            says(
254                "words",
255                "Says the game's word",
256                "What the game calls it: offensive, Light, clear as a bell.",
257                now.says.words,
258                default.says.words,
259            )
260        } else {
261            says(
262                "numbers",
263                "Says current/max",
264                "350/400, when the game has said both.",
265                now.says.numbers,
266                default.says.numbers,
267            )
268        },
269        says(
270            "percent",
271            "Says its percent",
272            "87%.",
273            now.says.percent,
274            default.says.percent,
275        ),
276        row(
277            "color",
278            "Colour",
279            "Its fill's colour; empty, the theme's.",
280            RowKind::Color,
281            Value::Text(now.color.map(crate::menu::hex).unwrap_or_default()),
282            now.color.is_some(),
283        ),
284    ]
285    .into_iter()
286    .chain(text_rows(now, default))
287    .chain([thresholds_row(now, default)])
288    .chain(clock.then(|| clock_row(now, default)))
289    .chain(image_rows(now, pictures))
290    .chain(emblems.map(|names| emblem_row(now, names)))
291    .map(|row| Row {
292        needs: rests_on(&row.key, now),
293        ..row
294    })
295    .collect()
296}

298/// Why the row `key` of a bar looking as `now` does nothing now, when it
299/// does nothing: what it shapes is not drawn.
300fn rests_on(key: &str, now: &Look) -> Option<String> {
301    match key {
302        "ring" if now.fills != Fills::Ring => Some("Set Fills to a ring first."),
303        "label" | "words" | "numbers" | "percent" | "text_color" | "outline_color" | "outline"
304            if now.place == Place::Hidden =>
305        {
306            Some("Set Text to show the words first.")
307        }
308        "track" if now.fills == Fills::Orb => Some("An orb has no track: it wears its glass."),
309        "track" if now.background.is_some() => {
310            Some("Take the background away first: the track shows only without one.")
311        }
312        _ => None,
313    }
314    .map(str::to_owned)
315}

379/// A bar widget's images, as `now` has them: the fill's, the one under it
380/// and the one over it, each from the pool. An orb's are the orb art, each
381/// from its own folder: the liquids, the glasses, the rims. A ring's or a
382/// bar's are the strips, fills and backgrounds, and the overlay kits.

384    let (orb, panel) = (now.fills == Fills::Orb, now.fills == Fills::Panel);
385    let row = |key: &str, label: &str, help: &str, set: &Option<String>, kind: RowKind| Row {
386        key: key.to_owned(),
387        label: label.to_owned(),
388        help: help.to_owned(),
389        kind,
390        value: Value::Text(set.clone().unwrap_or_default()),
391        here: set.is_some(),
392        from: None,
393        needs: None,
394    };
395    vec![
396        row(
397            "fill_image",
398            "Fill image",
399            "Grey art the fill uncovers as it fills, in the fill's colour: a liquid. A ring's or a bar's is a strip, wrapped round or laid along it; an upright bar turns it a quarter.",
400            &now.fill_image,
401            if orb {
402                art_choice(&pictures.orbs.fills)
403            } else {
404                art_choice(&pictures.strips.fills)
405            },
406        ),
407        row(
408            "background",
409            "Background",
410            "An image under the fill, in place of the empty part: an orb's glass, or a ring's or a bar's track from a pack. An orb with none wears the glass Hydra ships; a ring or a bar with none draws its track.",
411            &now.background,
412            if orb {
413                art_choice(&pictures.orbs.backgrounds)
414            } else {
415                art_choice(&pictures.strips.backgrounds)
416            },
417        ),
418        Row {
419            key: "track".to_owned(),
420            label: "Track".to_owned(),
421            help: "What a ring or a bar shows under its fill with no background chosen: a tube shaded across, the tube with a cast of the bar's colour, or the fill's own strip, faint, brought to full brightness as it fills.".to_owned(),
422            kind: choice(TRACKS.iter().map(|(_, value, called)| (*value, *called))),
423            value: Value::Text(
424                TRACKS
425                    .iter()
426                    .find(|(track, ..)| *track == now.track)
427                    .map_or("", |(_, value, _)| value)
428                    .to_owned(),
429            ),
430            here: now.track != Track::default(),
431            from: None,
432            needs: None,
433        },
434        row(
435            "overlay",
436            "Overlay",
437            "Art laid over the bar and its fill: an orb's rim, or a vessel -- an hourglass, a flask, a lantern -- whose glass the orb's fill shows through, with no background under it; a ring's or a bar's kit, rails along it or round it and caps at a bar's ends, the fill kept between them.",
438            &now.overlay,
439            if orb {
440                rim_or_vessel_choice(&pictures.orbs.overlays, &pictures.orbs.vessels)
441            } else if panel {
442                art_choice(&pictures.emblems)
443            } else {
444                art_choice(&pictures.strips.overlays)
445            },

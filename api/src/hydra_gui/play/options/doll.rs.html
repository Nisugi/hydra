10/// The Injuries widget's rows: its picture, any in the data folder's
11/// `dolls`, or none for the body drawn in code. `now`: the game's clock,
12/// unix seconds, where it is known (the time of day the sky shows).
13pub(super) fn doll_rows(look: Option<&DollLook>, dolls: &[PathBuf], now: Option<u32>) -> Vec<Row> {
14    #[cfg(not(feature = "doll-infinite"))]
15    let _ = now;
16    let picture = look.and_then(|look| look.picture.clone());
17    let style = look.map(|look| look.style).unwrap_or_default();
18    let mut styles = vec![
19        ("doll".to_owned(), "Doll".to_owned()),
20        ("text".to_owned(), "Text".to_owned()),
21    ];
22    if cfg!(feature = "doll-infinite") {
23        styles.push(("infinite".to_owned(), "Infinite".to_owned()));
24    }
25    let mut rows = vec![Row {
26        key: "style".to_owned(),
27        label: "Style".to_owned(),
28        help: "The Doll: dots on a picture or a body, or a picture's own art. Text: a line \
29               for each part hurt or scarred. Infinite: a puppet that moves as your \
30               character does."
31            .to_owned(),
32        kind: RowKind::Choice(styles),
33        value: Value::Text(
34            match style {
35                Style::Doll => "doll",
36                Style::Text => "text",
37                Style::Infinite => "infinite",
38            }
39            .to_owned(),
40        ),
41        here: style != Style::Doll,
42        from: None,
43    }];
44    rows.push(Row {
45        key: "picture".to_owned(),
46        label: "Picture".to_owned(),
47        help: "A picture for the doll: your own from the dolls folder in Hydra's data \
48               folder, or a pack's; None draws a body. Calibrate it from the doll's \
49               right-click menu."
50            .to_owned(),
51        kind: overlay_choice(dolls),
52        value: Value::Text(picture.clone().unwrap_or_default()),
53        here: picture.is_some(),
54        from: None,
55    });
56    #[cfg(feature = "doll-infinite")]
57    if style == Style::Infinite {
58        rows.extend(figure_rows(look));
59        let backdrop = look.map_or(Backdrop::Day, |look| look.backdrop);
60        let lit = light_rows(look, backdrop, now);
61        rows.push(Row {
62            key: "backdrop".to_owned(),
63            label: "Backdrop".to_owned(),
64            help: "What the puppet stands in front of: the game's own sky now, looking \
65                   south, its sun and moons where they stand; a dark display; black; or a \
66                   colour of your own."
67                .to_owned(),
68            kind: RowKind::Choice(
69                BACKDROPS
70                    .iter()
71                    .map(|(_, value, called)| ((*value).to_owned(), (*called).to_owned()))
72                    .collect(),
73            ),
74            value: Value::Text(backdrop_value(backdrop)),
75            here: backdrop != Backdrop::Day,
76            from: None,
77        });
78        rows.extend(lit);
79        if backdrop == Backdrop::Colour {
80            let colour = look.and_then(|look| look.colour.clone());
81            rows.push(Row {
82                key: "colour".to_owned(),
83                label: "Colour".to_owned(),
84                help: "The colour behind the puppet.".to_owned(),
85                kind: RowKind::Color,
86                value: Value::Text(colour.clone().unwrap_or_else(|| "#000000".to_owned())),
87                here: colour.is_some(),
88                from: None,
89            });
90        }
91        rows.extend(status_rows(look));
92    }
93    rows
94}

153/// The statuses to put on the puppet, a row of buttons a group, each
154/// lit while it is on: on until it is clicked again, or until Settings
155/// closes. The character's own show whatever these are.
156#[cfg(feature = "doll-infinite")]
157fn status_rows(look: Option<&DollLook>) -> Vec<Row> {
158    let shown = look.map_or(&[][..], |look| &look.shown);
159    STATUS_GROUPS
160        .iter()
161        .map(|(group, statuses)| {
162            let words: Vec<(String, String)> = statuses
163                .iter()
164                .map(|status| (status.id().to_owned(), capitalised(status.id())))
165                .collect();
166            let on: Vec<String> = words
167                .iter()
168                .filter(|(id, _)| shown.contains(id))
169                .map(|(id, _)| id.clone())
170                .collect();
171            Row {
172                key: format!("statuses_{}", group.to_ascii_lowercase()),
173                label: (*group).to_owned(),
174                help: "Put a status on the puppet to see how it looks: on until it is \
175                       clicked again, or until Settings closes. Your character's own show \
176                       whatever these are."
177                    .to_owned(),
178                kind: RowKind::Buttons(words),
179                here: !on.is_empty(),
180                value: Value::List(on),
181                from: None,
182            }
183        })
184        .collect()
185}

187/// `word` with its first letter a capital: a status's name on its button.
188#[cfg(feature = "doll-infinite")]
189fn capitalised(word: &str) -> String {
190    let mut called = word.to_owned();
191    if let Some(first) = called.get_mut(..1) {
192        first.make_ascii_uppercase();
193    }
194    called
195}

197/// The statuses a row of buttons' `key` holds (`statuses_body`: the body's),
198/// when it is one.
199#[cfg(feature = "doll-infinite")]
200fn shown_group(key: &str) -> Option<&'static [injury_doll::state::Status]> {
201    let group = key.strip_prefix("statuses_")?;
202    STATUS_GROUPS
203        .iter()
204        .find(|(name, _)| name.eq_ignore_ascii_case(group))
205        .map(|(_, statuses)| *statuses)
206}

208/// The Infinite puppet's race and figure rows: the playable races
209/// `gs_studio` installed, and the chosen race's figures.
210#[cfg(feature = "doll-infinite")]
211fn figure_rows(look: Option<&DollLook>) -> [Row; 2] {
212    let races = crate::widget::infinite::races();
213    let chosen = look.and_then(|look| look.figure.clone());
214    let shown = crate::widget::infinite::figure(&races, chosen.as_deref());
215    let race = races.iter().find(|race| {
216        race.figures
217            .iter()
218            .any(|one| Some(&one.form) == shown.as_ref())
219    });
220    [
221        Row {
222            key: "race".to_owned(),
223            label: "Race".to_owned(),
224            help: "The playable race the Infinite puppet is drawn as, of those gs_studio \
225                   has installed in Hydra's data folder."
226                .to_owned(),
227            kind: RowKind::Choice(
228                races
229                    .iter()
230                    .map(|race| (race.name.clone(), race.name.clone()))
231                    .collect(),
232            ),
233            value: Value::Text(race.map(|race| race.name.clone()).unwrap_or_default()),
234            here: chosen.is_some(),
235            from: None,
236        },
237        Row {
238            key: "figure".to_owned(),
239            label: "Figure".to_owned(),
240            help: "Which of the race's figures: painted, or on a grid that shows where \
241                   each wound lies."
242                .to_owned(),
243            kind: RowKind::Choice(race.map_or_else(Vec::new, |race| {
244                race.figures
245                    .iter()
246                    .map(|one| (one.form.clone(), one.variant.clone()))
247                    .collect()
248            })),
249            value: Value::Text(shown.unwrap_or_default()),
250            here: chosen.is_some(),
251            from: None,
252        },
253    ]
254}

256/// The Infinite puppet's light: where it comes from and how high (not
257/// under the sky, whose sun and moons stand where they are), and how
258/// strong; each a slider, at the backdrop's own until moved. Under the
259/// sky, the time of day it is lit at: the game's own now (`now`, its
260/// clock in unix seconds; noon before it is known) until moved.
261#[cfg(feature = "doll-infinite")]
262fn light_rows(look: Option<&DollLook>, backdrop: Backdrop, now: Option<u32>) -> Vec<Row> {
263    let (from, height) = crate::widget::infinite::key_light();
264    let slider = |key: &str,
265                  label: &str,
266                  help: &str,
267                  (min, max, step): (f64, f64, f64),
268                  set: Option<f64>,
269                  own: f64| Row {
270        key: key.to_owned(),
271        label: label.to_owned(),
272        help: help.to_owned(),
273        kind: RowKind::Slider { min, max, step },
274        value: Value::Text(set.unwrap_or(own).to_string()),
275        here: set.is_some(),
276        from: None,
277    };
278    let mut rows = Vec::new();
279    if backdrop != Backdrop::Day {
280        rows.push(slider(
281            "light_from",
282            "Light from",
283            "Where the light comes from, degrees round the puppet: 0 from the front, 90 from the right, -90 from the left, 180 from behind.",
284            (-180.0, 180.0, 5.0),
285            look.and_then(|look| look.light_from).map(f64::from),
286            f64::from(from).round(),
287        ));
288        rows.push(slider(
289            "light_height",
290            "Light height",
291            "How high the light stands, degrees over the horizon: 0 level with the puppet, 90 straight overhead.",
292            (0.0, 90.0, 1.0),
293            look.and_then(|look| look.light_height).map(f64::from),
294            f64::from(height).round(),
295        ));
296    }
297    rows.push(slider(
298        "light_strength",
299        "Light strength",
300        "How strong the light on the puppet is, percent of its backdrop's own.",
301        (0.0, 300.0, 5.0),
302        look.and_then(|look| look.light_strength).map(f64::from),
303        100.0,
304    ));
305    if backdrop == Backdrop::Day {
306        let set = look
307            .and_then(|look| look.hour)
308            .map(|minutes| f64::from(minutes) / 60.0);
309        // the game's hour now, to the quarter
310        let own = now.map_or(12.0, |now| {
311            (f64::from(injury_doll::base::elanthian_time::elanthian_hour_f(
312                f64::from(now),
313            )) * 4.0)
314                .round()
315                / 4.0
316        });
317        rows.push(Row {
318            key: "hour".to_owned(),
319            label: "Time of day".to_owned(),
320            help: "The hour the sky lights the puppet at, the sun where it stands then: the \
321                   game's own hour until moved."
322                .to_owned(),
323            kind: RowKind::TimeOfDay,
324            value: Value::Text(set.unwrap_or(own).to_string()),
325            here: set.is_some(),
326            from: None,
327        });
328    }
329    rows
330}

332/// Set the Infinite puppet's light `key` to `to`, or back to its own; None
333/// for another key.
334#[cfg(feature = "doll-infinite")]
335fn light_set(look: &mut DollLook, key: &str, to: Option<&str>) -> Option<Result<(), String>> {
336    let number = |what: &str| -> Result<Option<f32>, String> {
337        to.map(str::trim)
338            .filter(|to| !to.is_empty())
339            .map(|to| {
340                to.trim_end_matches(['\u{b0}', '%'])
341                    .parse::<f32>()
342                    .ok()
343                    .filter(|n| n.is_finite())
344                    .ok_or_else(|| format!("`{to}` is not {what}."))
345            })
346            .transpose()
347    };
348    #[expect(
349        clippy::cast_possible_truncation,
350        clippy::cast_sign_loss,
351        reason = "wrapped or clamped into the setting's range first"
352    )]
353    let done = match key {
354        "light_from" => number("a direction in degrees").map(|n| {
355            look.light_from = n.map(|deg| ((deg + 180.0).rem_euclid(360.0) - 180.0).round() as i16);
356        }),
357        "light_height" => number("a height in degrees").map(|n| {
358            look.light_height = n.map(|deg| deg.clamp(0.0, 90.0).round() as i16);
359        }),
360        "light_strength" => number("a strength in percent").map(|n| {
361            look.light_strength = n.map(|pct| pct.clamp(0.0, 300.0).round() as u16);
362        }),
363        // a time of day, as hours (`14.25`) or as a clock reads it (`14:15`)
364        "hour" => {
365            let clock = to
366                .and_then(|to| to.split_once(':'))
367                .and_then(|(hours, minutes)| {
368                    Some(
369                        hours.trim().parse::<f32>().ok()?
370                            + minutes.trim().parse::<f32>().ok()? / 60.0,
371                    )
372                })
373                .filter(|hours| hours.is_finite());
374            let hours = match clock {
375                Some(hours) => Ok(Some(hours)),
376                None => number("a time of day"),
377            };
378            hours.map(|n| {
379                look.hour = n.map(|hours| (hours.rem_euclid(24.0) * 60.0).round() as u16 % 1440);
380            })
381        }
382        _ => return None,
383    };
384    Some(done)
385}

387/// The Injuries widget's backdrops: each, how it is written, its name.
388const BACKDROPS: [(Backdrop, &str, &str); 4] = [
389    (Backdrop::Day, "day", "The sky (south)"),
390    (Backdrop::Display, "display", "Display"),
391    (Backdrop::Black, "black", "Black"),
392    (Backdrop::Colour, "colour", "Solid colour"),
393];

395/// How `backdrop` is written on the page.
396#[cfg_attr(
397    not(feature = "doll-infinite"),
398    expect(dead_code, reason = "the doll's page")
399)]
400fn backdrop_value(backdrop: Backdrop) -> String {
401    BACKDROPS
402        .iter()
403        .find(|(each, ..)| *each == backdrop)
404        .map_or("", |(_, value, _)| value)
405        .to_owned()
406}

408/// Set the Injuries widget's `key` to `to`, or back to its own.
409#[cfg_attr(
410    feature = "doll-infinite",
411    expect(
412        clippy::too_many_lines,
413        reason = "one arm per setting of the widget, the Infinite doll's statuses with them"
414    )
415)]
416pub(super) fn doll_set(look: &mut DollLook, key: &str, to: Option<&str>) -> Result<(), String> {
417    #[cfg(feature = "doll-infinite")]
418    if let Some(group) = shown_group(key) {
419        // the group's statuses on now, a comma between each
420        let ids: Vec<&str> = group.iter().map(|status| status.id()).collect();
421        let on: Vec<&str> = to.map_or_else(Vec::new, |to| {
422            to.split(',')
423                .map(str::trim)
424                .filter(|one| !one.is_empty())
425                .collect()
426        });
427        if let Some(other) = on.iter().find(|one| !ids.contains(*one)) {
428            return Err(format!("Injuries has no status {other} there."));
429        }
430        look.shown.retain(|one| !ids.contains(&one.as_str()));
431        look.shown.extend(on.into_iter().map(str::to_owned));
432        return Ok(());
433    }
434    match key {
435        "picture" => {
436            look.picture = to.filter(|path| !path.is_empty()).map(str::to_owned);
437            Ok(())
438        }
439        #[cfg(feature = "doll-infinite")]
440        "race" => {
441            look.figure = match to.filter(|race| !race.is_empty()) {
442                None => None,
443                Some(name) => Some(
444                    crate::widget::infinite::races()
445                        .iter()
446                        .find(|race| race.name == name)
447                        .and_then(|race| race.figures.first())
448                        .map(|one| one.form.clone())
449                        .ok_or_else(|| format!("Injuries has no race {name}."))?,
450                ),
451            };
452            Ok(())
453        }
454        #[cfg(feature = "doll-infinite")]
455        "figure" => {
456            let figure = to.filter(|figure| !figure.is_empty());
457            if let Some(figure) = figure
458                && !crate::widget::infinite::races()
459                    .iter()
460                    .flat_map(|race| &race.figures)
461                    .any(|one| one.form == figure)
462            {
463                return Err(format!("Injuries has no figure {figure}."));
464            }
465            look.figure = figure.map(str::to_owned);
466            Ok(())
467        }
468        #[cfg(feature = "doll-infinite")]
469        "light_from" | "light_height" | "light_strength" | "hour" => light_set(look, key, to)
470            .unwrap_or_else(|| Err(format!("Injuries has no setting {key}."))),
471        "facing" => {
472            look.facing = match to.map(str::trim).filter(|to| !to.is_empty()) {
473                None => None,
474                Some(to) => {
475                    let deg: f32 = to
476                        .trim_end_matches('\u{b0}')
477                        .parse()
478                        .ok()
479                        .filter(|deg: &f32| deg.is_finite())
480                        .ok_or_else(|| format!("`{to}` is not a facing in degrees."))?;
481                    #[expect(
482                        clippy::cast_possible_truncation,
483                        reason = "wrapped to -180..180 first"
484                    )]
485                    Some(((deg + 180.0).rem_euclid(360.0) - 180.0).round() as i16)
486                }
487            };
488            Ok(())
489        }
490        "backdrop" => {
491            look.backdrop = match to {
492                None => Backdrop::Day,
493                Some(to) => BACKDROPS
494                    .iter()
495                    .find(|(_, value, _)| *value == to)
496                    .map(|(each, ..)| *each)
497                    .ok_or_else(|| format!("Injuries has no backdrop {to}."))?,
498            };
499            Ok(())
500        }
501        "colour" => {
502            look.colour = match to.filter(|colour| !colour.is_empty()) {
503                None => None,
504                Some(to) => Some(
505                    crate::menu::rgb(to)
506                        .map(crate::menu::hex)
507                        .ok_or_else(|| format!("`{to}` is not a colour."))?,
508                ),
509            };
510            Ok(())
511        }
512        "style" => {
513            look.style = match to {
514                None | Some("doll") => Style::Doll,
515                Some("infinite") => Style::Infinite,
516                Some("text") => Style::Text,
517                Some(other) => return Err(format!("Injuries has no style {other}.")),
518            };
519            Ok(())
520        }
521        _ => Err(format!("Injuries has no setting {key}.")),
522    }

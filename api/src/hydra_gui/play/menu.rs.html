1//! A play window's two menus.
2//!
3//! **Add a widget**, from the Layout menu: every kind in one list, grouped
4//! by what they show, each group a fold, closed until opened; each kind
5//! says whether one is shown already. Typing finds a kind in every group
6//! at once, listed under its group's name. A click adds it in a standalone
7//! window of its own. Above them the presets, Hydra's and those saved: a
8//! click places a copy. At the bottom, closed, the **Advanced** place: the
9//! character what is added follows, when another than the window's own --
10//! a party's vitals in one window, say.
11//!
12//! **A right-click** on a window or a widget in one opens its menu: remove
13//! it, or add a tab beside it of a kind of its own group (a stream beside a
14//! stream); on the combat stream, whether fights stay in the story too;
15//! rename, save as a preset or remove a custom window; and one
16//! **Advanced** entry, closed,
17//! for the character the widget follows. On a line of the game's, it first
18//! offers to send the lines like it to another stream: a stream rule made
19//! from it, brought to the Streams page to be looked over and saved. Neither menu offers a story to
20//! another character: no window mixes two characters' story.
21

40/// A window's or widget's right-click menu, open.

46    placed: Option<u32>,
47    /// Where it opened.
48    at: Pos2,
49    /// A custom window's new title, while it is being typed.
50    renaming: Option<String>,
51    /// The name a custom window is saved as a preset under, while typed.

61/// What a menu was asked to do.
62enum Act {
63    Remove(u32),
64    RemoveWindow,
65    /// The window behind every other.
66    SendToBack,
67    Rename(String),
68    Follow(u32, Option<String>),
69    Save(String),
70    /// The widget's own settings page.
71    Settings(u32),
72    /// The custom window's own settings page.

74    /// A tab of this kind beside the widget.
75    AddTab(u32, Widget),
76    /// The widget's tab moved along its strip.
77    MoveTab(u32, crate::layout::Shift),

80    /// A rule sending the lines like the one it was opened on to this
81    /// stream; one of the player's own to name, for none.

83    /// The lookout's line it was opened on forgotten (`forget`) or shown

88    /// The line it was opened on is part of a fight (`true`), or not.

93    /// The Add-a-widget list, while it is open, over the play `area`, with
94    /// `others` the characters a widget could follow and `state` the
95    /// character's, a field of which may be picked as a widget. Whether it
96    /// changed the layout.
97    pub(super) fn add_list(
98        &mut self,
99        context: &egui::Context,
100        area: Rect,
101        (others, state): (&[Character], Option<&hydra_model::GameState>),
102        library: &Library,
103        received: &[Widget],
104    ) -> bool {
105        let Some(shown) = self.layout.as_ref().map(kinds_shown) else {
106            return false;
107        };
108        let Some(adding) = self.adding.as_mut() else {
109            return false;
110        };
111        let every = kinds(received);
112        let mut open = true;
113        let mut chosen = None;
114        let mut preset = None;
115        let mut forget = None;
116        // Above every widget window, those in the foreground among them,
117        // so a window it adds to is never drawn over it.
118        let shown = egui::Window::new("Add a widget")
119            .id(Id::new(("add-widget", self.session)))
120            .open(&mut open)
121            .collapsible(false)
122            .order(egui::Order::Foreground)
123            .default_size([260.0, 420.0])
124            .default_pos(area.right_top() + vec2(-290.0, 10.0))
125            .show(context, |ui| {
126                ui.add(
127                    egui::TextEdit::singleline(&mut adding.search).hint_text("Find a widget..."),
128                );
129                let search = adding.search.to_lowercase();
130                // As wide as the window, so its bar is at the window's edge.
131                egui::ScrollArea::vertical()
132                    .auto_shrink([false, true])
133                    .show(ui, |ui| {
134                        presets(
135                            ui,
136                            (self.session, &search),
137                            library,
138                            &mut preset,
139                            &mut forget,
140                        );
141                        let mut offer = |ui: &mut egui::Ui, kind: &Widget| {
142                            ui.horizontal(|ui| {
143                                let own_only = adding.whose.is_some() && kind.is_story();
144                                if ui
145                                    .add_enabled(!own_only, egui::Button::new(kind.name()))
146                                    .clicked()
147                                {
148                                    chosen = Some(kind.clone());
149                                }
150                                if shown.contains(kind) {
151                                    ui.weak("shown");
152                                }
153                            });
154                        };
155                        for group in Group::ALL {
156                            let kinds: Vec<&Widget> = every
157                                .iter()
158                                .filter(|kind| kind.group() == group)
159                                .filter(|kind| kind.name().to_lowercase().contains(&search))
160                                .collect();
161                            if kinds.is_empty() {
162                                continue;
163                            }
164                            let heading = group.name().to_uppercase();
165                            fold(ui, (self.session, &heading), &search, |ui| {
166                                for kind in kinds {
167                                    offer(ui, kind);
168                                }
169                            });
170                        }
171                        if let Some(field) = field_fold(ui, (self.session, &search), state) {
172                            chosen = Some(field);
173                        }
174                        advanced(ui, &mut adding.whose, others);
175                    });
176            });
177        if let Some(shown) = shown {
178            context.move_to_top(shown.response.layer_id);
179        }
180        let whose = adding_whose(self.adding.as_ref());
181        if !open {
182            self.adding = None;
183        }
184        if let Some(name) = forget {
185            self.out = Some(Asked::ForgetPreset(name));
186        }
187        let Some(layout) = self.layout.as_mut() else {
188            return false;
189        };
190        if let Some(preset) = preset {
191            layout.add_preset(&preset, whose.as_deref());
192            return true;
193        }
194        let Some(kind) = chosen else {
195            return false;
196        };
197        layout.add_widget(kind, whose);
198        true
199    }

201    /// A right-click on a window or a widget in one, at the play `area`,
202    /// opens its menu; the menu, while open, over everything. `own` are the
203    /// streams of the player's own a line could be sent to; `combat_kept`
204    /// whether fights stay in the story too, `None` with no rule taking
205    /// them. Whether it changed the layout.
206    pub(super) fn right_click(
207        &mut self,
208        context: &egui::Context,
209        area: Rect,
210        (others, received, own, combat_kept): (&[Character], &[Widget], &[String], Option<bool>),
211    ) -> bool {
212        let (secondary, pressed, pointer, escape) = context.input(|input| {
213            (
214                input.pointer.secondary_clicked(),
215                input.pointer.primary_pressed(),
216                input.pointer.interact_pos(),
217                input.key_pressed(egui::Key::Escape),
218            )
219        });
220        if escape {
221            self.menu = None;
222        }
223        if secondary
224            && let Some(at) = pointer
225            && let Some((holder, placed)) = self.under(context, area, at)
226        {
227            self.menu = Some(Menu {
228                holder,
229                placed,
230                at,
231                renaming: None,
232                saving: None,
233                tabbing: false,
234                line: crate::widget::lines::pointed(context),
235                sending: false,
236            });
237        }
238        let (Some(menu), Some(layout)) = (self.menu.as_mut(), self.layout.as_ref()) else {
239            return false;
240        };
241        let mut act = None;
242        let shown = egui::Area::new(Id::new(("play-menu", self.session)))
243            .order(Order::Foreground)
244            .fixed_pos(menu.at)
245            .show(context, |ui| {
246                egui::Frame::menu(ui.style()).show(ui, |ui| {
247                    ui.set_min_width(170.0);
248                    act = items(ui, menu, layout, (others, received, own), combat_kept);
249                });
250            });
251        let outside = pressed && pointer.is_some_and(|at| !shown.response.rect.contains(at));
252        let Some(act) = act else {
253            if outside {
254                self.menu = None;
255            }
256            return false;
257        };
258        let holder = menu.holder;
259        let line = menu.line.take();
260        self.menu = None;
261        if let Some(asked) = asked_of(&act, line.as_ref(), holder) {
262            self.out = Some(asked);
263            return false;
264        }
265        let Some(layout) = self.layout.as_mut() else {
266            return false;
267        };
268        if let Act::Save(name) = &act {
269            if let Some(Holds::Custom(custom)) = layout.holder(holder).map(|found| &found.holds) {
270                self.out = Some(Asked::SavePreset(Preset::of(name, custom)));
271            }
272            return false;
273        }
274        if let Act::Calibrate(placed, picture) = act {
275            self.calibrating = Some(super::calibrator::Calibrator::open(placed, picture));
276            return false;
277        }
278        match act {
279            Act::Remove(placed) => layout.remove_widget(holder, placed),
280            Act::RemoveWindow => layout.remove_window(holder),
281            Act::SendToBack => {
282                layout.send_to_back(holder);
283                sink(context, super::holders::id(self.session, holder));
284            }
285            Act::Rename(title) => layout.rename(holder, &title),
286            Act::Follow(placed, who) => layout.follow(placed, who),
287            Act::AddTab(beside, kind) => {
288                let _ = layout.add_tab(holder, beside, kind);
289            }
290            Act::MoveTab(placed, shift) => {
291                let _ = layout.move_tab(placed, shift);
292            }
293            Act::Save(_)
294            | Act::Settings(_)
295            | Act::WindowSettings
296            | Act::Calibrate(..)
297            | Act::SendLike(_)
298            | Act::Lookout(_)
299            | Act::KeepCombat(_)
300            | Act::FightPart(_) => {}
301        }
302        true
303    }

305    /// The window at `at` on top, and the widget there in it, if any: a
306    /// standalone window's own, or the cell's in a custom window.
307    fn under(&self, context: &egui::Context, area: Rect, at: Pos2) -> Option<(u32, Option<u32>)> {
308        let layout = self.layout.as_ref()?;
309        let on_top = context.layer_id_at(at)?.id;
310        let holder = layout
311            .holders
312            .iter()
313            .find(|holder| on_top == super::holders::id(self.session, holder.id))?;
314        let placed = match &holder.holds {
315            Holds::One(placed) => Some(placed.id),
316            Holds::Custom(custom) => {
317                let inside = self
318                    .insides
319                    .iter()
320                    .find(|(id, _)| *id == holder.id)
321                    .map(|(_, inside)| inside.translate(area.min.to_vec2()))?;
322                custom
323                    .cells
324                    .iter()
325                    .rev()
326                    .find(|cell| cell.rect().translate(inside.min.to_vec2()).contains(at))
327                    .and_then(|cell| cell.shown().map(|shown| shown.id))
328            }
329        };
330        Some((holder.id, placed))
331    }
332}

334/// One group of the list, under `heading`, a fold closed until opened, in
335/// play window `session`'s list; while `search` holds words, every group
336/// lies open under its name, so all that is found shows at once.
337fn fold(
338    ui: &mut egui::Ui,
339    (session, heading): (u32, &str),
340    search: &str,
341    add_contents: impl FnOnce(&mut egui::Ui),
342) {
343    if search.is_empty() {
344        egui::CollapsingHeader::new(RichText::new(heading).small())
345            .id_salt(("add-widget-group", session, heading))
346            .default_open(false)
347            .show(ui, add_contents);
348    } else {
349        ui.label(RichText::new(heading).small().weak());
350        add_contents(ui);
351    }
352}

354/// The fold a value of the game state is picked under.
355pub(crate) const FIELD_FOLD: &str = "A VALUE FROM THE GAME STATE";
356
357/// Any one value of the game state, as a Field widget: the room's id, the
358/// silver, a count; the one picked, if one was. Not while the list is
359/// narrowed to something else: the picker has a filter of its own.
360fn field_fold(
361    ui: &mut egui::Ui,
362    (session, search): (u32, &str),
363    state: Option<&hydra_model::GameState>,
364) -> Option<Widget> {
365    let wanted =
366        search.is_empty() || FIELD_FOLD.to_lowercase().contains(search) || "field".contains(search);
367    if !wanted {
368        return None;
369    }
370    let mut chosen = None;
371    fold(ui, (session, FIELD_FOLD), search, |ui| {
372        ui.weak("One value, on a line of its own: pick it from the game state.");
373        let key = Id::new(("play-field-pick", session));
374        if let Some(path) = crate::widget::field::pick(ui, state, key) {
375            chosen = Some(Widget::Field(path));
376        }
377    });
378    chosen
379}

381/// The presets, Hydra's and those saved, whose names hold `search`: a click
382/// on one chooses it, a player's own can be forgotten.
383fn presets(
384    ui: &mut egui::Ui,
385    (session, search): (u32, &str),
386    library: &Library,
387    chosen: &mut Option<Preset>,
388    forget: &mut Option<String>,
389) {
390    let found = |preset: &&Preset| preset.name.to_lowercase().contains(search);
391    let hydras = Preset::hydras();
392    let hydras: Vec<&Preset> = hydras.iter().filter(found).collect();
393    let saved: Vec<&Preset> = library.presets().iter().filter(found).collect();
394    if hydras.is_empty() && saved.is_empty() {
395        return;
396    }
397    fold(ui, (session, "PRESETS"), search, |ui| {
398        for preset in hydras {
399            if ui.button(&preset.name).clicked() {
400                *chosen = Some(preset.clone());
401            }
402        }
403        for preset in saved {
404            ui.horizontal(|ui| {
405                if ui.button(&preset.name).clicked() {
406                    *chosen = Some(preset.clone());
407                }
408                if ui
409                    .small_button("Forget")
410                    .on_hover_text("Forget this preset; windows placed from it stay")
411                    .clicked()
412                {
413                    *forget = Some(preset.name.clone());
414                }
415            });
416        }
417    });
418    // Outside the fold, so it is seen with the fold closed.
419    if let Some(why) = library.not_saved() {
420        ui.colored_label(
421            crate::theme::color(ui.ctx(), crate::theme::T::Wrong),
422            format!("Presets not saved: {why}"),
423        );
424    }
425}

441/// The Advanced place, closed until opened: which character `whose` is, of
442/// this one and `others`.

444    egui::CollapsingHeader::new("Advanced")

446        .show(ui, |ui| {

462/// A menu's items, for what it was opened on; what was asked, if anything.
463fn items(
464    ui: &mut egui::Ui,
465    menu: &mut Menu,
466    layout: &Layout,
467    (others, received, own): (&[Character], &[Widget], &[String]),
468    combat_kept: Option<bool>,
469) -> Option<Act> {
470    let holder = layout.holder(menu.holder)?;
471    let mut act = line_items(ui, menu, own);
472    let widget = menu
473        .placed
474        .and_then(|placed| widget_in(layout, menu.holder, placed));
475    if let (Some(placed), Some(widget)) = (menu.placed, widget) {
476        act = act.or(widget_items(ui, menu, layout, (placed, widget), received));
477        if *widget == Widget::Stream(hydra_session::stream_rules::COMBAT.to_owned())
478            && let Some(kept) = combat_kept
479        {
480            let mut keep = kept;
481            if ui
482                .checkbox(&mut keep, "Keep fights in the story")
483                .on_hover_text(
484                    "Each fight's lines and prompt shown in the story as well as here;                      off, they leave the story while this window is open",
485                )
486                .changed()
487            {
488                act = Some(Act::KeepCombat(keep));
489            }
490        }
491    }
492    ui.separator();
493    if ui
494        .button("Send to back")
495        .on_hover_text("This window behind every other")
496        .clicked()
497    {
498        act = Some(Act::SendToBack);
499    }
500    if let Holds::Custom(custom) = &holder.holds {
501        ui.weak(&custom.title);
502        match &mut menu.renaming {
503            Some(title) if menu.saving.is_none() => {
504                // Enter is the menu's, taken before the field sees it: a
505                // field that keeps the keyboard never loses it to Enter.
506                let enter = ui
507                    .input_mut(|input| input.consume_key(egui::Modifiers::NONE, egui::Key::Enter));
508                ui.text_edit_singleline(title).request_focus();
509                if enter {
510                    act = Some(Act::Rename(title.clone()));
511                }
512            }
513            _ => {
514                if ui.button("Rename...").clicked() {
515                    menu.renaming = Some(custom.title.clone());
516                    menu.saving = None;
517                }
518            }
519        }
520        if ui
521            .button("Window settings...")
522            .on_hover_text(
523                "This window's own settings: its name, its title, how solid it is, its layer",
524            )
525            .clicked()
526        {
527            act = Some(Act::WindowSettings);
528        }
529        match &mut menu.saving {
530            Some(name) => {
531                ui.label("Save as preset:");
532                let enter = ui
533                    .input_mut(|input| input.consume_key(egui::Modifiers::NONE, egui::Key::Enter));
534                ui.text_edit_singleline(name).request_focus();
535                if enter && !name.trim().is_empty() {
536                    act = Some(Act::Save(name.clone()));
537                }
538            }
539            None => {
540                if ui.button("Save as preset...").clicked() {
541                    menu.saving = Some(custom.title.clone());
542                    menu.renaming = None;
543                }
544            }
545        }
546        if ui.button("Remove window").clicked() {
547            act = Some(Act::RemoveWindow);
548        }
549    }
550    if let (Some(placed), Some(widget)) = (menu.placed, widget)
551        && !widget.is_story()
552    {
553        ui.separator();
554        egui::CollapsingHeader::new("Advanced")
555            .default_open(false)
556            .show(ui, |ui| {
557                let now = layout.follows.get(&placed).cloned();
558                let mut whose = now.clone();
559                ui.label("Show for");
560                ui.radio_value(&mut whose, None, "This character");
561                for other in others {
562                    let key = other.key(others);
563                    ui.radio_value(&mut whose, Some(key.clone()), key);
564                }
565                if whose != now {
566                    act = Some(Act::Follow(placed, whose));
567                }
568            });
569    }
570    act
571}

573/// What `act`, chosen on the menu of `holder`'s window opened on `line`
574/// when it was opened on one, asks of the window: a settings page, a stream
575/// rule made from the line, or the lookout's word on it; `None` for an act
576/// of the layout's.
577fn asked_of(act: &Act, line: Option<&crate::widget::lines::Pointed>, holder: u32) -> Option<Asked> {
578    match (act, line) {
579        (Act::Settings(placed), _) => Some(Asked::Settings(Some(super::options::page_id(*placed)))),
580        (Act::WindowSettings, _) => Some(Asked::Settings(Some(super::options::window_page_id(
581            holder,
582        )))),
583        (Act::SendLike(to), Some(line)) => Some(Asked::StreamRule(Box::new(line.brought(to)))),
584        (Act::KeepCombat(keep), _) => Some(Asked::KeepCombat(*keep)),
585        (Act::FightPart(fight), Some(line)) => Some(Asked::FightPart(*fight, line.runs.clone())),
586        (Act::Lookout(word), Some(line)) => {
587            let text: String = line.runs.iter().map(|run| run.text.as_str()).collect();
588            Some(Asked::Command(format!("lookout {word} {}", text.trim())))
589        }
590        _ => None,
591    }
592}

594/// The items for the line of the game's the menu was opened on, if it was:
595/// *Send lines like this to*, then the streams, `own` among them.
596fn line_items(ui: &mut egui::Ui, menu: &mut Menu, own: &[String]) -> Option<Act> {
597    let line = menu.line.as_ref()?;
598    let mut act = None;
599    if line.shown == hydra_ui::lookout_line::STREAM {
600        if ui
601            .button("Forget this line")
602            .on_hover_text("Learned now: the lookout does not show it again")
603            .clicked()
604        {
605            act = Some(Act::Lookout("forget"));
606        }
607        if ui
608            .button("Always show this line")
609            .on_hover_text("Shown in the lookout stream however often it comes")
610            .clicked()
611        {
612            act = Some(Act::Lookout("always"));
613        }
614        ui.separator();
615    }
616    // A line of the story's or the combat stream's: the player's word on
617    // whether the lines like it are a fight's.
618    let main = hydra_session::player_log::MAIN;
619    if line.shown == main || line.shown == hydra_session::stream_rules::COMBAT {
620        if ui
621            .button("Part of this fight")
622            .on_hover_text(
623                "Lines like this go with the fights they come in, or come alone in, from now on",
624            )
625            .clicked()
626        {
627            act = Some(Act::FightPart(true));
628        }
629        if ui
630            .button("Not part of this fight")
631            .on_hover_text("Lines like this stay where they are, whatever comes with them")
632            .clicked()
633        {
634            act = Some(Act::FightPart(false));
635        }
636        ui.separator();
637    }
638    if menu.sending {
639        ui.label("Send lines like this to:");
640        act = crate::stream_editor::send_to(ui, Some(&line.shown), own).map(Act::SendLike);
641    } else if ui
642        .button("Send lines like this to...")
643        .on_hover_text(
644            "A stream rule made from this line, its names and numbers loosened, to look over \
645             and save on the Streams page",
646        )
647        .clicked()
648    {
649        menu.sending = true;
650    }
651    ui.separator();
652    act
653}

655/// The window whose area is `window` behind every other window of the play
656/// area: every other is raised in its present order, which leaves this one
657/// lowest. The windows' layers are the middle order; what floats above
658/// them is left alone.
659fn sink(context: &egui::Context, window: Id) {
660    let others: Vec<egui::LayerId> = context.memory(|memory| {
661        memory
662            .layer_ids()
663            .filter(|layer| layer.order == Order::Middle && layer.id != window)
664            .collect()
665    });
666    for layer in others {
667        context.move_to_top(layer);
668    }
669}

671/// A widget's own items: its settings, a tab beside it, and removing it.
672fn widget_items(
673    ui: &mut egui::Ui,
674    menu: &mut Menu,
675    layout: &Layout,
676    (placed, widget): (u32, &Widget),
677    received: &[Widget],
678) -> Option<Act> {
679    let mut act = None;
680    ui.weak(widget.name());
681    if super::options::has_page(widget)
682        && ui
683            .button("Settings...")
684            .on_hover_text("This widget's own settings: how it draws")
685            .clicked()
686    {
687        act = Some(Act::Settings(placed));
688    }
689    if let Some(picture) = layout
690        .dolls
691        .get(&placed)
692        .and_then(|look| look.picture.clone())
693        && ui
694            .button("Calibrate doll...")
695            .on_hover_text("Where each part is on the doll's picture")
696            .clicked()
697    {
698        act = Some(Act::Calibrate(placed, picture));
699    }
700    let beside = tab_kinds(layout, menu.holder, (placed, widget), received);
701    if !beside.is_empty() {
702        if menu.tabbing {
703            ui.label("Add a tab:");
704            for kind in beside {
705                if ui.button(kind.name()).clicked() {
706                    act = Some(Act::AddTab(placed, kind));
707                }
708            }
709        } else if ui
710            .button("Add a tab...")
711            .on_hover_text("Another widget beside this one, a tab each")
712            .clicked()
713        {
714            menu.tabbing = true;
715        }
716    }
717    // In a stack of two or more, its tab moves along the strip.
718    let stacked = match layout.holder(menu.holder).map(|found| &found.holds) {
719        Some(Holds::Custom(custom)) => custom
720            .cells
721            .iter()
722            .find(|cell| cell.tabs.iter().any(|tab| tab.id == placed))
723            .map(|cell| {
724                (
725                    cell.tabs.len(),
726                    cell.tabs.iter().position(|tab| tab.id == placed),
727                )
728            }),
729        _ => None,
730    };
731    if let Some((tabs, Some(at))) = stacked
732        && tabs > 1
733    {
734        use crate::layout::Shift;
735        for (label, shift, can) in [
736            ("Move tab left", Shift::Left, at > 0),
737            ("Move tab right", Shift::Right, at + 1 < tabs),
738            ("Move tab to front", Shift::Front, at > 0),
739        ] {
740            if ui.add_enabled(can, egui::Button::new(label)).clicked() {
741                act = Some(Act::MoveTab(placed, shift));
742            }
743        }
744    }
745    if ui.button("Remove").clicked() {
746        act = Some(Act::Remove(placed));
747    }
748    act
749}

751/// Every kind: the catalog's, the streams named among them, and any other
752/// stream or game dialog this character has `received`: a stream that
753/// comes with a position the game gives appears for the characters that
754/// hold it, and a dialog once the game has sent it.
755fn kinds(received: &[Widget]) -> Vec<Widget> {
756    let mut every = Widget::all();
757    for kind in received {
758        if !every.contains(kind) && !kind.is_hidden() {
759            every.push(kind.clone());
760        }
761    }
762    every
763}

765/// The kinds a tab beside widget `placed` could be: of its own kin
766/// (`Widget::kin`), the
767/// story only for the window's own character, and none already beside it.
768pub(super) fn tab_kinds(
769    layout: &Layout,
770    holder: u32,
771    (placed, widget): (u32, &Widget),
772    received: &[Widget],
773) -> Vec<Widget> {
774    let beside: Vec<&Widget> = match layout.holder(holder).map(|found| &found.holds) {
775        Some(Holds::Custom(custom)) => custom
776            .cells
777            .iter()
778            .find(|cell| cell.tabs.iter().any(|tab| tab.id == placed))
779            .map(|cell| cell.tabs.iter().map(|tab| &tab.widget).collect())
780            .unwrap_or_default(),
781        Some(Holds::One(one)) => vec![&one.widget],
782        None => Vec::new(),
783    };
784    let following = layout.follows.contains_key(&placed);
785    kinds(received)
786        .into_iter()
787        .filter(|kind| widget.stacks() && kind.stacks() && !beside.contains(&kind))
788        .filter(|kind| !(following && kind.is_story()))
789        .collect()
790}

793fn widget_in(layout: &Layout, holder: u32, placed: u32) -> Option<&Widget> {
794    layout
795        .holder(holder)?
796        .holds
797        .placed()
798        .find(|one| one.id == placed)
799        .map(|one| &one.widget)

119impl Drawing<'_> {
120    /// How effect list or Timers panel `widget`, placed as `placed`, draws
121    /// its rows: its own choices over every effects window's.
122    fn rows_of(
123        &self,
124        placed: u32,
125        widget: &crate::widget::Widget,
126    ) -> Option<crate::widget::effects::EffectRows> {
127        let own = match widget {
128            crate::widget::Widget::Effects(_) => self.effects.get(&placed).cloned(),
129            crate::widget::Widget::Timers => self.timers.get(&placed).map(|look| look.rows.clone()),
130            _ => return None,
131        };
132        Some(own.unwrap_or_default().over(&self.every_effects.rows))
133    }
134
135    /// What the player chose for the widget placed as `placed` on its own
136    /// page.
137    fn chosen(&self, placed: &Placed) -> crate::widget::Chosen {
138        let (placed, widget) = (placed.id, &placed.widget);
139        crate::widget::Chosen {
140            look: self
141                .looks
142                .get(&placed)
143                .cloned()
144                .zip(widget.bar_token())
145                .map(|(look, token)| look.themed(token)),
146            room: self.rooms.get(&placed).copied(),
147            lines: self.lines.get(&placed).copied(),
148            doll: self.dolls.get(&placed).cloned(),
149            // The Uberbar's doll wears the picture of the first Injuries
150            // widget placed, when one is; its numbers are the window's own
151            // character's day, so another's shows none.
152            uberbar: (*widget == crate::widget::Widget::Uberbar).then(|| {
153                crate::widget::uberbar::UberbarShown {
154                    look: self.uberbars.get(&placed).cloned().unwrap_or_default(),
155                    facts: (!self.follows.contains_key(&placed))
156                        .then_some(self.uberbar_facts)
157                        .flatten(),
158                    doll: self.dolls.values().next().cloned(),
159                }
160            }),
161            // The window's own character's aim is remembered; another's
162            // is only what its own window has heard.
163            aim: (*widget == crate::widget::Widget::Aim).then(|| crate::widget::aim::AimShown {
164                look: self.aims.get(&placed).copied().unwrap_or_default(),
165                remembered: (!self.follows.contains_key(&placed))
166                    .then(|| self.remembered_aim.map(str::to_owned))
167                    .flatten(),
168            }),
169            list: self.lists.get(&placed).copied(),
170            label: self.labels.get(&placed).cloned(),
171            roster: self.rosters.get(&placed).cloned(),
172            quick: self.quicks.get(&placed).cloned(),
173            hotbar: (*widget == crate::widget::Widget::Hotbar).then(|| {
174                crate::widget::hotbar::Shown::of(self.hotbars, self.hotbar_looks.get(&placed))
175            }),
176            // The window's own character's: another's are its window's.
177            tracked: (widget.is_effects() && !self.follows.contains_key(&placed))
178                .then(|| self.tracked.clone()),
179            circles: widget.is_effects().then(|| {
180                self.circles
181                    .cloned()
182                    .unwrap_or_else(hydra_ui::effects::circle_palette)
183            }),
184            timers: self.timers.get(&placed).cloned(),
185            effects: self.rows_of(placed, widget),
186            chip: self.chips.get(&placed).cloned(),
187            minimap: self.minimaps.get(&placed).copied(),
188            indicator: match widget {
189                // Every game indicator follows what every indicator does,
190                // its look or none.
191                crate::widget::Widget::Indicator(_) => {
192                    Some(self.indicators.get(&placed).cloned().unwrap_or_default())
193                }
194                crate::widget::Widget::CustomIndicator => {
195                    self.customs.get(&placed).map(|custom| custom.look.clone())
196                }
197                _ => None,
198            }
199            .map(|look| crate::widget::IndicatorShown {
200                look,
201                every: self.every_indicator.clone(),
202                icons: crate::widget::hotbar::Icons {
203                    dir: self.hotbars.icons_dir(),
204                    sheets: self.hotbars.sheets.clone(),
205                    size: None,
206                },
207                styled: match (widget, self.indicator_style) {
208                    (crate::widget::Widget::Indicator(which), Some(style)) => Some(
209                        std::path::Path::new(style)
210                            .join(format!("{}.png", which.id()))
211                            .to_string_lossy()
212                            .into_owned(),
213                    ),
214                    _ => None,
215                },
216            }),
217            custom: self.customs.get(&placed).cloned(),
218            compass: self
219                .compasses
220                .get(&placed)
221                .and_then(|look| look.picture.clone()),
222            profiles: (*widget == crate::widget::Widget::Trainer
223                && !self.follows.contains_key(&placed))
224            .then(|| self.profiles.names()),
225            containers: (*widget == crate::widget::Widget::Containers
226                && !self.follows.contains_key(&placed))
227            .then(|| self.containers.clone()),
228        }
229    }

259/// The top bar: who, how connected, the keys, and the layout's controls:
260/// its menu, the drawers, Arrange and Lock. What the hands hold and the clocks are
261/// widgets, in the layout with the rest.
262/// The story's banners, a highlight's alerts, floating over every window
263/// as the *Banners* page says -- where, how wide, how solid, their colours,
264/// how long -- the newest last; nothing under them moves. A click takes
265/// them away, and they are not shown again (`gone`). The window is drawn
266/// again when the oldest is due to go.
267pub(super) fn alerts(
268    ui: &mut egui::Ui,
269    session: u32,
270    story: &hydra_ui::Story,
271    gone: &mut Option<std::time::Instant>,
272) {
273    use hydra_board::banner::{Across, Down};
274    let now = std::time::Instant::now();
275    let look = crate::play::banner(ui.ctx());
276    let lasting = std::time::Duration::from_secs(u64::from(look.seconds));
277    let alerts: Vec<_> = story
278        .alerts_within(now, lasting)
279        .into_iter()
280        .filter(|(at, _)| gone.is_none_or(|gone| *at > gone))
281        .collect();
282    let Some(oldest) = alerts.first().map(|(at, _)| *at) else {
283        return;
284    };
285    let ctx = ui.ctx().clone();
286    let ground = look
287        .ground
288        .map_or_else(|| theme::color(&ctx, T::Warning), theme::rgb);
289    let words = look
290        .text
291        .map_or_else(|| theme::readable_on(ground), theme::rgb);
292    let ground = ground.gamma_multiply(look.opacity);
293    let screen = ctx.content_rect();
294    let width = (screen.width() * look.width).max(120.0);
295    let align = match (look.across, look.down) {
296        (Across::Left, Down::Top) => egui::Align2::LEFT_TOP,
297        (Across::Centre, Down::Top) => egui::Align2::CENTER_TOP,
298        (Across::Right, Down::Top) => egui::Align2::RIGHT_TOP,
299        (Across::Left, Down::Middle) => egui::Align2::LEFT_CENTER,
300        (Across::Centre, Down::Middle) => egui::Align2::CENTER_CENTER,
301        (Across::Right, Down::Middle) => egui::Align2::RIGHT_CENTER,
302        (Across::Left, Down::Bottom) => egui::Align2::LEFT_BOTTOM,
303        (Across::Centre, Down::Bottom) => egui::Align2::CENTER_BOTTOM,
304        (Across::Right, Down::Bottom) => egui::Align2::RIGHT_BOTTOM,
305    };
306    // Clear of the top bar and the command line, and of the window's edge.
307    let offset = match look.down {
308        Down::Top => egui::vec2(0.0, 36.0),
309        Down::Middle => egui::Vec2::ZERO,
310        Down::Bottom => egui::vec2(0.0, -40.0),
311    } + match look.across {
312        Across::Left => egui::vec2(8.0, 0.0),
313        Across::Centre => egui::Vec2::ZERO,
314        Across::Right => egui::vec2(-8.0, 0.0),
315    };
316    let shown = egui::Area::new(egui::Id::new(("play-alerts", session)))
317        .order(egui::Order::Foreground)
318        .anchor(align, offset)
319        .interactable(true)
320        .show(&ctx, |ui| {
321            ui.set_width(width);
322            for (_, text) in &alerts {
323                egui::Frame::new()
324                    .fill(ground)
325                    .inner_margin(6.0)
326                    .corner_radius(theme::corner(&ctx))
327                    .show(ui, |ui| {
328                        ui.set_width(width - 12.0);
329                        ui.colored_label(words, *text);
330                    });
331            }
332        });
333    let clicked = ui
334        .interact(
335            shown.response.rect,
336            egui::Id::new(("play-alerts-click", session)),
337            egui::Sense::click(),
338        )
339        .on_hover_text("Click to take it away")
340        .clicked();
341    if clicked || shown.response.clicked() {
342        *gone = Some(now);
343        ctx.request_repaint();
344        return;
345    }
346    let left = lasting.saturating_sub(now.saturating_duration_since(oldest));
347    ctx.request_repaint_after(left);
348}

350pub(super) fn top(
351    ui: &mut egui::Ui,
352    view: &PlayView<'_>,
353    title: &TitleBar,
354    drawers: &mut crate::layout::Drawers,
355    (arranging, locked, guides): (&mut bool, &mut bool, &mut crate::layout::Guides),
356    unsaved: Option<&str>,
357) -> Option<Top> {
358    let mut asked = None;
359    // As Hydra's own title bar, the bar's own ground moves the window when
360    // dragged and maximizes it when double-clicked: taken before the
361    // buttons, which, added after, win the press where they are.
362    if title.own {
363        title_ground(ui);
364    }
365    // Three sides across the bar: the left reads from its edge, the right
366    // from its edge back, the window's own buttons outermost; the centre on
367    // the bar's middle, slid aside only as far as it must to clear them.
368    let middle = ui.max_rect().center().x;
369    ui.horizontal(|ui| {
370        for item in title.on(Side::Left) {
371            let pressed = bar_item(ui, view, item, drawers, (arranging, locked, guides));
372            asked = asked.or(pressed);
373        }
374        let left_end = ui.cursor().min.x;
375        ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
376            if title.own {
377                window_buttons(ui);
378            }
379            for item in title.on(Side::Right).iter().rev() {
380                let pressed = bar_item(ui, view, item, drawers, (arranging, locked, guides));
381                asked = asked.or(pressed);
382            }
383            if let Some(why) = unsaved {
384                ui.colored_label(theme::color(ui.ctx(), T::Wrong), why);
385            }
386            if let Some(on) = view.numlock {
387                ui.weak(if on { "NumLock on" } else { "NumLock off" });
388            }
389            if view.set != 0 {
390                ui.weak(format!("Set {}", view.set)).on_hover_text(
391                    "The macro set in use over set 0; Alt+0 goes back to set 0 alone",
392                );
393            }
394            let center = title.on(Side::Center);
395            if !center.is_empty() {
396                let right_start = ui.cursor().max.x;
397                let row = ui.max_rect().y_range();
398                let placed = centre_at(ui, middle, (left_end, right_start), row, |ui| {
399                    for item in center {
400                        let pressed =
401                            bar_item(ui, view, item, drawers, (arranging, locked, guides));
402                        asked = asked.or(pressed);
403                    }
404                });
405                ui.advance_cursor_after_rect(placed);
406            }
407        });
408    });
409    asked
410}

412/// Draw the title bar's centre with `draw`, its middle at `middle`, slid
413/// right or left just far enough to stay between `left_end` and
414/// `right_start`, the sides' items; with no room for it there, right after
415/// the left side. Its width is the last frame's, kept in egui's memory,
416/// since a group's width is known only once drawn; a changed width asks for
417/// another frame. Where it drew.
418fn centre_at(
419    ui: &mut egui::Ui,
420    middle: f32,
421    (left_end, right_start): (f32, f32),
422    row: egui::Rangef,
423    draw: impl FnOnce(&mut egui::Ui),
424) -> egui::Rect {
425    let key = ui.id().with("title-centre-width");
426    let wide = ui.data(|data| data.get_temp::<f32>(key)).unwrap_or(0.0);
427    let gap = ui.spacing().item_spacing.x;
428    let (least, most) = (left_end, right_start - gap - wide);
429    let from = if least <= most {
430        (middle - wide / 2.0).clamp(least, most)
431    } else {
432        least
433    };
434    let room = egui::Rect::from_x_y_ranges(from..=right_start.max(from), row);
435    let mut child = ui.new_child(
436        UiBuilder::new()
437            .max_rect(room)
438            .layout(egui::Layout::left_to_right(egui::Align::Center)),
439    );
440    draw(&mut child);
441    let drawn = child.min_rect();
442    if (drawn.width() - wide).abs() > 0.5 {
443        ui.data_mut(|data| data.insert_temp(key, drawn.width()));
444        ui.ctx().request_repaint();
445    }
446    drawn
447}

449/// As Hydra's own title bar, the bar's ground moves the window when
450/// dragged and maximizes it when double-clicked.
451fn title_ground(ui: &mut egui::Ui) {
452    let bar = ui.available_rect_before_wrap();
453    let ground = ui.interact(
454        bar,
455        ui.id().with("title-ground"),
456        egui::Sense::click_and_drag(),
457    );
458    if ground.drag_started_by(egui::PointerButton::Primary) {
459        ui.ctx().send_viewport_cmd(egui::ViewportCommand::StartDrag);
460    }
461    if ground.double_clicked_by(egui::PointerButton::Primary) {
462        let maximized = ui.input(|input| input.viewport().maximized.unwrap_or(false));
463        ui.ctx()
464            .send_viewport_cmd(egui::ViewportCommand::Maximized(!maximized));
465    }
466}

468/// `item` as a word of the bar, when it is one: whether it was.
469fn word(ui: &mut egui::Ui, view: &PlayView<'_>, item: &Item) -> bool {
470    let state = view.snapshot.map(|snapshot| &snapshot.state);
471    match item {
472        Item::Character => {
473            ui.strong(view.name);
474        }
475        Item::Profession => {
476            let profession = state.and_then(|state| state.character.identity.profession.as_deref());
477            ui.weak(profession.unwrap_or("?"));
478        }
479        Item::Connection => {
480            let (word, token) = crate::hub::connection(view.lifecycle);
481            ui.colored_label(theme::color(ui.ctx(), token), word);
482        }
483        Item::Room => {
484            let room = state.and_then(|state| state.room.title.as_deref());
485            ui.colored_label(
486                theme::color(ui.ctx(), T::RoomName),
487                room.unwrap_or("Room unknown"),
488            );
489        }
490        Item::Number => {
491            let number = state.and_then(|state| state.room.id.as_deref());
492            ui.weak(number.unwrap_or("?"));
493        }
494        // A game value: its name, and its number as the model works it
495        // out now, whole when it is whole.
496        Item::Value(name) => {
497            let value = state.and_then(|state| state.evaluate(name));
498            let said = match value {
499                Some(n) if n.fract() == 0.0 => format!("{name} {n:.0}"),
500                Some(n) => format!("{name} {n:.1}"),
501                None => format!("{name} ?"),
502            };
503            ui.weak(said);
504        }
505        _ => return false,
506    }
507    true
508}

510/// The − and + that step the zoom of every window, reading − then + whichever
511/// way the side lays them out: what was pressed.
512fn zoom_buttons(ui: &mut egui::Ui) -> Option<Top> {
513    let plain = |ui: &mut egui::Ui, glyph: &str, said: &str, by: i8| {
514        ui.add(egui::Button::new(glyph).frame(false))
515            .on_hover_text(said)
516            .clicked()
517            .then_some(Top::Zoom(by))
518    };
519    let smaller = ("\u{2212}", "Draw everything smaller", -1);
520    let larger = ("+", "Draw everything larger", 1);
521    let (first, second) = if ui.layout().main_dir() == egui::Direction::RightToLeft {
522        (larger, smaller)
523    } else {
524        (smaller, larger)
525    };
526    plain(ui, first.0, first.1, first.2).or_else(|| plain(ui, second.0, second.1, second.2))
527}

540/// One item of the bar: a word, or a button, and what the button asked
541/// when pressed.
542fn bar_item(
543    ui: &mut egui::Ui,
544    view: &PlayView<'_>,
545    item: &Item,
546    drawers: &mut crate::layout::Drawers,
547    (arranging, locked, guides): (&mut bool, &mut bool, &mut crate::layout::Guides),
548) -> Option<Top> {
549    if word(ui, view, item) {
550        return None;
551    }
552    let mut asked = None;
553    match item {
554        Item::Settings => {
555            if ui
556                .button("Settings")
557                .on_hover_text("This character's settings, its keys and its widgets' pages")
558                .clicked()
559            {
560                asked = Some(Top::Settings);
561            }
562        }
563        Item::Zoom => asked = zoom_buttons(ui),
564        Item::Menu => {
565            let place = crate::main_menu::Place::Play {
566                arranging: &mut *arranging,
567                locked: &mut *locked,
568                guides: &mut *guides,
569            };
570            asked = crate::main_menu::show(ui, place).map(|choice| match choice {
571                crate::main_menu::Choice::Layout(ask) => layout_top(ask),
572                other => Top::Menu(other),
573            });
574        }
575        Item::Log => {
576            if ui
577                .button("Log")
578                .on_hover_text("What this character read, kept a file a day")
579                .clicked()
580            {
581                asked = Some(Top::Log);
582            }
583        }
584        Item::Keys => {
585            ui.menu_button("Keys", |ui| {
586                for said in view.keys {
587                    ui.label(said);
588                }
589                if ui.button("Read the keybinds again").clicked() {
590                    asked = Some(Top::ReloadKeys);
591                    ui.close();
592                }
593                if ui.button("Change the keys...").clicked() {
594                    asked = Some(Top::Keys);
595                    ui.close();
596                }
597            });
598        }
599        Item::Layout => {
600            asked = crate::main_menu::menu_button(ui, "Layout", |ui| {
601                crate::main_menu::layout_entries(ui, (arranging, locked), guides)
602            })
603            .flatten()
604            .map(layout_top);
605        }
606        Item::Drawers => {
607            crate::main_menu::menu_button(ui, "Drawers", |ui| {
608                if super::drawers::menu(ui, drawers) {
609                    asked = Some(Top::Drawers);
610                }
611            });
612        }
613        // On the bar, lit while it is on, so the mode is seen
614        // and left in one click: a new custom window turns it
615        // on, and with the switch only in the Layout menu a
616        // player could be left stuck in it.
617        Item::Arrange => {
618            ui.add_enabled_ui(!*locked, |ui| {
619                ui.toggle_value(arranging, "Arrange")
620                    .on_hover_text("Move, resize and drag widgets in and out of custom windows");
621            });
622        }
623        Item::Lock => {
624            ui.toggle_value(locked, "Lock")
625                .on_hover_text("Keep every window where it is: none is dragged or resized");
626        }
627        Item::TopDrawer | Item::BottomDrawer | Item::LeftDrawer | Item::RightDrawer => {
628            if let Some(zone) = item.drawer()
629                && let Some(drawer) = drawers.get_mut(zone)
630                && drawer_icon(ui, zone, drawer.open)
631            {
632                drawer.open = !drawer.open;
633                asked = Some(Top::Drawers);
634            }
635        }
636        Item::Character
637        | Item::Profession
638        | Item::Connection
639        | Item::Room
640        | Item::Number
641        | Item::Value(_) => {}
642    }
643    asked
644}

646/// Drawer `zone`'s icon on the bar: a small box with the drawer's edge
647/// filled, in the accent while it is `open`, faint while it is shut.
648/// Whether it was clicked.
649fn drawer_icon(ui: &mut egui::Ui, zone: crate::layout::Zone, open: bool) -> bool {
650    use crate::layout::Zone;
651    let size = egui::Vec2::splat(ui.spacing().interact_size.y);
652    let (rect, response) = ui.allocate_exact_size(size, egui::Sense::click());
653    let name = zone.name();
654    response.widget_info(|| egui::WidgetInfo::selected(egui::WidgetType::Button, true, open, name));
655    let hover = if open {
656        format!("Shut the {}", name.to_lowercase())
657    } else {
658        format!("Open the {}", name.to_lowercase())
659    };
660    let response = response.on_hover_text(hover);
661    let visuals = ui.style().interact(&response);
662    let edge = visuals.fg_stroke.color;
663    let fill = if open {
664        theme::color(ui.ctx(), T::Accent)
665    } else {
666        ui.visuals().weak_text_color()
667    };
668    let painter = ui.painter();
669    let box_ = rect.shrink(3.0);
670    painter.rect_stroke(
671        box_,
672        1.0,
673        egui::Stroke::new(1.0, edge),
674        egui::StrokeKind::Inside,
675    );
676    let thick = (box_.width() / 3.0).round();
677    let strip = match zone {
678        Zone::Top => egui::Rect::from_min_size(box_.min, egui::vec2(box_.width(), thick)),
679        Zone::Bottom => {
680            egui::Rect::from_min_max(egui::pos2(box_.min.x, box_.max.y - thick), box_.max)
681        }
682        Zone::Left => egui::Rect::from_min_size(box_.min, egui::vec2(thick, box_.height())),
683        Zone::Right | Zone::Main => {
684            egui::Rect::from_min_max(egui::pos2(box_.max.x - thick, box_.min.y), box_.max)
685        }
686    };
687    painter.rect_filled(strip, 1.0, fill);
688    response.clicked()
689}

693fn window_buttons(ui: &mut egui::Ui) {
694    let maximized = ui.input(|input| input.viewport().maximized.unwrap_or(false));
695    let plain = |ui: &mut egui::Ui, glyph: &str, said: &str| {

716    if plain(ui, "\u{2014}", "Minimize") {
717        ui.ctx()
718            .send_viewport_cmd(egui::ViewportCommand::Minimized(true));
719    }

722/// How close to the window's edge a press resizes it, points.
723const EDGE: f32 = 6.0;
724
725/// With no system frame, the window's edges resize it: the pointer near
726/// one shows the resize cursor, and a press there begins the resize.
727pub(super) fn resize_edges(ctx: &egui::Context) {
728    use egui::{CursorIcon, ResizeDirection, ViewportCommand};
729    let maximized = ctx.input(|input| input.viewport().maximized.unwrap_or(false));
730    if maximized {
731        return;
732    }
733    let Some(at) = ctx.input(|input| input.pointer.hover_pos()) else {
734        return;
735    };
736    let screen = ctx.viewport_rect();
737    let (left, right) = (at.x - screen.min.x < EDGE, screen.max.x - at.x < EDGE);
738    let (top, bottom) = (at.y - screen.min.y < EDGE, screen.max.y - at.y < EDGE);
739    let (direction, cursor) = match (left, right, top, bottom) {
740        (true, _, true, _) => (ResizeDirection::NorthWest, CursorIcon::ResizeNwSe),
741        (_, true, true, _) => (ResizeDirection::NorthEast, CursorIcon::ResizeNeSw),
742        (true, _, _, true) => (ResizeDirection::SouthWest, CursorIcon::ResizeNeSw),
743        (_, true, _, true) => (ResizeDirection::SouthEast, CursorIcon::ResizeNwSe),
744        (true, _, _, _) => (ResizeDirection::West, CursorIcon::ResizeHorizontal),
745        (_, true, _, _) => (ResizeDirection::East, CursorIcon::ResizeHorizontal),
746        (_, _, true, _) => (ResizeDirection::North, CursorIcon::ResizeVertical),
747        (_, _, _, true) => (ResizeDirection::South, CursorIcon::ResizeVertical),
748        _ => return,
749    };
750    ctx.set_cursor_icon(cursor);
751    if ctx.input(|input| input.pointer.primary_pressed()) {
752        ctx.send_viewport_cmd(ViewportCommand::BeginResize(direction));
753    }
754}

761/// What a window holds, filling it: its window is the size its layout says,
762/// never its content's. One widget is given everything; a custom window's
763/// widgets are drawn bare in their cells, once the cells are kept to the
764/// inside it has now (`Custom::fit`). Nothing spills out of its cell: a
765/// one-line widget stays one line, and the rest scroll. A tab stack shows a
766/// tab for each of its widgets, a click on one showing it, each not showing
767/// with what it has said since it last showed. Its inside, where it drew.
768pub(super) fn holder(
769    ui: &mut egui::Ui,
770    holds: &mut Holds,
771    drawing: &mut Drawing<'_>,
772) -> egui::Rect {
773    let inside = ui.available_rect_before_wrap();
774    ui.set_min_size(inside.size());
775    let area = inside;
776    match holds {
777        Holds::One(placed) => shown(ui, placed, drawing),
778        Holds::Custom(custom) => {
779            // Not to the pass egui lays a window out in at its narrowest as
780            // a resize begins, then discards: kept to it, a narrow cell came
781            // back from the scaling at its smallest, not its width.
782            // Resized with Arrange on, a window leaves its widgets their
783            // size, room made or taken for arranging them; with it off they
784            // scale with the window.
785            if ui.is_sizing_pass() {
786            } else if drawing.arranging {
787                custom.keep(area.size());
788            } else {
789                custom.fit(area.size());
790            }
791            if custom.cells.is_empty() {
792                ui.new_child(UiBuilder::new().max_rect(area.shrink(4.0)))
793                    .weak("Empty. With Arrange on, in the Layout menu, drop widgets here.");
794            }
795            let gap = theme::shape(ui.ctx()).pane_gap;
796            let strip = theme::tab_height(ui.ctx(), crate::widget::LINE);
797            for cell in &mut custom.cells {
798                let whole = cell.rect().translate(area.min.to_vec2()).intersect(area);
799                // Half the gap off each side: the gap between two.
800                let at = whole.shrink(gap / 2.0);
801                if !at.is_positive() {
802                    continue;
803                }
804                let widths = crate::widget::tab::widths(&cell.tabs, drawing.tabs);
805                let height = cell.tab_height.map_or(strip, f32::from);
806                let (tabs, body) = tabs_and_body(at, &widths, height, cell.bar);
807                let reorder = tab_strip(ui, cell, &tabs, drawing);
808                if let Some((at, to)) = reorder
809                    && cell.move_tab(at, to)
810                {
811                    drawing.reordered = true;
812                }
813                let Some(placed) = cell.shown() else {
814                    continue;
815                };
816                let mut child =
817                    ui.new_child(UiBuilder::new().max_rect(body).id_salt(("cell", placed.id)));
818                shown(&mut child, placed, drawing);
819            }
820            ui.advance_cursor_after_rect(inside);
821        }
822    }
823    inside
824}

826/// `cell`'s tab strip drawn, a tab for each of its widgets at `tabs`: a
827/// click shows that tab, and a tab dragged along the strip and let go asks
828/// to move, as (from, to). With Arrange on the strip is arranging's.
829fn tab_strip(
830    ui: &mut egui::Ui,
831    cell: &mut crate::layout::Cell,
832    tabs: &[egui::Rect],
833    drawing: &mut Drawing<'_>,
834) -> Option<(usize, usize)> {
835    let bar = cell.bar;
836    let style = cell.look.unwrap_or_else(|| theme::shape(ui.ctx()).tabs);
837    let mut reorder = None;
838    for (index, (tab, rect)) in cell.tabs.iter().zip(tabs.iter().copied()).enumerate() {
839        let showing = index == cell.showing;
840        let look = drawing.tabs.get(&tab.id);
841        let quiet = look.is_some_and(|look| look.quiet);
842        // A tab showing is read; a quiet one is never counted.
843        let unread = unread(drawing.read, tab, &drawing.seen).filter(|_| !showing && !quiet);
844        if unread.is_some() {
845            drawing.unread.push(tab.id);
846        }
847        let name = crate::widget::tab::name(look, &tab.widget.name(), unread);
848        let color = match (showing, unread.is_some()) {
849            (true, _) => look.and_then(|look| look.showing),
850            (false, true) => look.and_then(|look| look.unread),
851            (false, false) => None,
852        };
853        let name = match color {
854            Some([r, g, b]) => egui::RichText::new(name).color(egui::Color32::from_rgb(r, g, b)),
855            None => egui::RichText::new(name),
856        };
857        // Text taller than its tab is drawn smaller, to fit.
858        let font = egui::TextStyle::Button.resolve(ui.style());
859        let row = ui.fonts_mut(|fonts| fonts.row_height(&font));
860        let room = rect.height() - 2.0 * ui.spacing().button_padding.y;
861        let name = if row > room && room > 0.0 {
862            name.size(font.size * room / row)
863        } else {
864            name
865        };
866        let mut child = ui.new_child(UiBuilder::new().max_rect(rect).id_salt(("tab", tab.id)));
867        // A button is never shorter than egui's interact size: a tab is its
868        // strip's height, however short.
869        let least = &mut child.spacing_mut().interact_size.y;
870        *least = least.min(rect.height());
871        let button = match style {
872            TabStyle::Plain | TabStyle::Separated => egui::Button::selectable(showing, name),
873            TabStyle::Boxed | TabStyle::Filled => {
874                tab_ground(ui, rect, style, showing);
875                egui::Button::new(name).frame(false)
876            }
877        }
878        .wrap_mode(egui::TextWrapMode::Truncate);
879        let response = child.add_sized(rect.size(), button);
880        if response.clicked() && cell.showing != index {
881            cell.showing = index;
882            // The tab shown is kept with the layout, as a key turning it is.
883            drawing.reordered = true;
884        }
885        if drawing.arranging {
886            continue;
887        }
888        let response = response.interact(egui::Sense::drag());
889        if let Some(at) = response.interact_pointer_pos()
890            && let Some(to) = crate::layout::tab_at(tabs, at, bar.vertical())
891            && to != index
892        {
893            if response.dragged() {
894                slot_mark(ui, tabs, index, to, bar.vertical());
895            }
896            if response.drag_stopped() {
897                reorder = Some((index, to));
898            }
899        }
900    }
901    if style == TabStyle::Separated {
902        tab_separators(ui, tabs, bar.vertical());
903    }
904    reorder
905}

907/// A tab at `rect` given its ground and edge under its words, as `style`
908/// draws a tab `showing` or not: boxed, an edge round it and the shown one
909/// on its ground; filled, every one on the tab ground and the shown one on
910/// its own. Under the pointer a tab not shown takes the raised ground.
911fn tab_ground(ui: &egui::Ui, rect: egui::Rect, style: TabStyle, showing: bool) {
912    let ctx = ui.ctx();
913    let rect = rect.shrink(1.0);
914    let corner = f32::from(theme::shape(ctx).corners.controls);
915    let pointed = !showing && ui.rect_contains_pointer(rect);
916    let fill = match (style, showing, pointed) {
917        (_, true, _) => theme::color(ctx, T::TabShown),
918        (_, false, true) => theme::color(ctx, T::Raised),
919        (TabStyle::Filled, false, false) => theme::color(ctx, T::Tab),
920        _ => egui::Color32::TRANSPARENT,
921    };
922    let edge = match style {
923        TabStyle::Boxed => egui::Stroke::new(1.0, theme::color(ctx, T::Line)),
924        _ => egui::Stroke::NONE,
925    };
926    ui.painter()
927        .rect(rect, corner, fill, edge, egui::StrokeKind::Inside);
928}

930/// A line between each two of the strip's tabs at `tabs`: across a strip
931/// down a side (`vertical`), else upright.
932fn tab_separators(ui: &egui::Ui, tabs: &[egui::Rect], vertical: bool) {
933    let stroke = egui::Stroke::new(1.0, theme::color(ui.ctx(), T::Line));
934    for pair in tabs.windows(2) {
935        let tab = pair[0];
936        if vertical {
937            ui.painter()
938                .hline(tab.x_range().shrink(4.0), tab.max.y, stroke);
939        } else {
940            ui.painter()
941                .vline(tab.max.x, tab.y_range().shrink(3.0), stroke);
942        }
943    }
944}

946/// The place in the strip a tab dragged from `from` would take at `to`,
947/// marked: a line at the edge of that tab it would go beside, across the
948/// strip when it runs down a side (`vertical`).
949pub(super) fn slot_mark(
950    ui: &egui::Ui,
951    tabs: &[egui::Rect],
952    from: usize,
953    to: usize,
954    vertical: bool,
955) {
956    let Some(rect) = tabs.get(to) else {
957        return;
958    };
959    let stroke = egui::Stroke::new(2.0, theme::color(ui.ctx(), T::Accent));
960    if vertical {
961        let y = if to > from { rect.max.y } else { rect.min.y };
962        ui.painter().hline(rect.x_range(), y, stroke);
963    } else {
964        let x = if to > from { rect.max.x } else { rect.min.x };
965        ui.painter().vline(x, rect.y_range(), stroke);
966    }
967}

969/// `placed` drawn into `ui`, showing: so what it has said is read, up to
970/// now. The one way a widget is drawn, standalone, alone in a cell or a tab.
971/// One that follows another character draws from that character, named on
972/// it, or says that character is not running; a story never follows.
973fn shown(ui: &mut egui::Ui, placed: &Placed, drawing: &mut Drawing<'_>) {
974    let rect = ui.max_rect();
975    drawing.rects.push((placed.id, rect));
976    if ui.input(|input| input.pointer.primary_pressed()) && ui.rect_contains_pointer(rect) {
977        drawing.pressed = Some(placed.id);
978    }
979    if drawing.in_use == Some(placed.id) {
980        // The window in use, which a scrolling key acts on.
981        let stroke = egui::Stroke::new(1.0, theme::color(ui.ctx(), T::Accent).gamma_multiply(0.5));
982        ui.painter().rect_stroke(
983            rect,
984            theme::corner(ui.ctx()),
985            stroke,
986            egui::StrokeKind::Inside,
987        );
988    }
989    // The theme's room round the widget, inside its pane.
990    let inside = padded(rect, theme::shape(ui.ctx()).pane_padding);
991    if inside == rect {
992        drawn(ui, placed, drawing);
993    } else {
994        let mut inner = ui.new_child(UiBuilder::new().max_rect(inside));
995        drawn(&mut inner, placed, drawing);
996        ui.advance_cursor_after_rect(rect);
997    }
998}

1000/// `rect` with `padding` taken off each side, but never so much that less
1001/// than a line is left either way: the theme's room round a widget gives
1002/// way in a pane too small for both, so a one-line widget keeps its line.
1003pub(super) fn padded(rect: egui::Rect, padding: f32) -> egui::Rect {
1004    let room = |size: f32| padding.min((size - crate::widget::LINE) / 2.0).max(0.0);
1005    rect.shrink2(egui::vec2(room(rect.width()), room(rect.height())))
1006}

1008/// `placed` drawn, as [`shown`] says: in its face, when its kind or it has
1009/// one of its own; an effects window's own over every effects window's.
1010fn drawn(ui: &mut egui::Ui, placed: &Placed, drawing: &mut Drawing<'_>) {
1011    let key = placed.widget.type_key();
1012    let own = drawing.faces.get(&placed.id);
1013    let every = &drawing.every_effects.face;
1014    let followed = (placed.widget.is_effects_window() && !every.is_empty())
1015        .then(|| own.map_or_else(|| every.clone(), |own| every.under(own)));
1016    if let Some(face) = theme::face(ui.ctx(), &key, followed.as_ref().or(own)) {
1017        let families = crate::fonts::loaded(ui.ctx());
1018        ui.scope(|ui| {
1019            theme::faced(ui.style_mut(), &face, &families);
1020            drawn_in(ui, placed, drawing);
1021        });
1022    } else {
1023        drawn_in(ui, placed, drawing);
1024    }
1025}

1027/// `placed` drawn in the face `ui` has.
1028fn drawn_in(ui: &mut egui::Ui, placed: &Placed, drawing: &mut Drawing<'_>) {
1029    let id = widget_id(drawing.session, placed.id);
1030    // The window's own command line, whoever else the widget would follow.
1031    if placed.widget == crate::widget::Widget::Input
1032        && let Some(typing) = &mut drawing.typing
1033    {
1034        drawing.entered |= super::command_line::command_line(ui, typing);
1035        return;
1036    }
1037    let follows = drawing
1038        .follows
1039        .get(&placed.id)
1040        .filter(|_| !placed.widget.is_story());
1041    let Some(who) = follows else {
1042        if let Some(count) = placed.widget.count(&drawing.seen) {
1043            drawing.read.insert(placed.id, count);
1044        }
1045        let chosen = drawing.chosen(placed);
1046        match placed.widget.draw_with(ui, &drawing.seen, id, &chosen) {
1047            Some(crate::widget::Clicked::Set { key, to }) => {
1048                drawing.set = Some((placed.id, key, to));
1049            }
1050            Some(line) => drawing.sent = Some(line),
1051            None => {}
1052        }
1053        return;
1054    };
1055    // A bare name two characters share -- one name on two games -- names
1056    // neither: which is meant is for the player to say.
1057    let mut named = drawing.others.iter().filter(|other| other.is(who));
1058    let (one, another) = (named.next(), named.next());
1059    if another.is_some() {
1060        ui.weak(format!(
1061            "More than one {who} is running: choose which in this window's Show for."
1062        ));
1063        return;
1064    }
1065    match one {
1066        Some(other) => {
1067            let seen = Seen {
1068                snapshot: other.snapshot.as_deref(),
1069                // Named by its game too when the follow does: one name
1070                // may be on two games.
1071                who: Some(if who.contains(':') { who } else { &other.name }),
1072                minimap: other.minimap.as_ref(),
1073                travel: None,
1074                notes: None,
1075                trainer: None,
1076                ..drawing.seen
1077            };
1078            // Another character's widget sends nothing on this one's.
1079            let chosen = drawing.chosen(placed);
1080            let _ = placed.widget.draw_with(ui, &seen, id, &chosen);
1081        }
1082        None => {
1083            ui.weak(format!("{who} is not running."));

1085    }

1088/// A window's title: its widget's name, a custom indicator's its own in
1089/// `customs`, and whose when it follows another character; or the custom
1090/// window's own.
1091pub(super) fn title(
1092    holds: &Holds,
1093    follows: &BTreeMap<u32, String>,
1094    customs: &BTreeMap<u32, crate::widget::custom_indicator::CustomIndicator>,
1095) -> String {
1096    match holds {
1097        Holds::One(placed) => {
1098            // A custom indicator's window is called by its own name.
1099            let name = customs.get(&placed.id).map_or_else(
1100                || placed.widget.name(),
1101                |custom| std::borrow::Cow::Borrowed(custom.name()),
1102            );
1103            match follows.get(&placed.id) {
1104                Some(who) if !placed.widget.is_story() => format!("{name} ({who})"),
1105                _ => name.into_owned(),
1106            }
1107        }
1108        Holds::Custom(custom) => custom.title.clone(),
1109    }
1110}

112impl Drawing<'_> {
113    /// How effect list or Timers panel `widget`, placed as `placed`, draws
114    /// its rows: its own choices over every effects window's.
115    fn rows_of(
116        &self,
117        placed: u32,
118        widget: &crate::widget::Widget,
119    ) -> Option<crate::widget::effects::EffectRows> {
120        let own = match widget {
121            crate::widget::Widget::Effects(_) => self.effects.get(&placed).cloned(),
122            crate::widget::Widget::Timers => self.timers.get(&placed).map(|look| look.rows.clone()),
123            _ => return None,
124        };
125        Some(own.unwrap_or_default().over(&self.every_effects.rows))
126    }
127
128    /// What the player chose for the widget placed as `placed` on its own
129    /// page.
130    fn chosen(&self, placed: &Placed) -> crate::widget::Chosen {
131        let (placed, widget) = (placed.id, &placed.widget);
132        crate::widget::Chosen {
133            look: self
134                .looks
135                .get(&placed)
136                .cloned()
137                .zip(widget.bar_token())
138                .map(|(look, token)| look.themed(token)),
139            room: self.rooms.get(&placed).copied(),
140            lines: self.lines.get(&placed).copied(),
141            doll: self.dolls.get(&placed).cloned(),
142            // The window's own character's aim is remembered; another's
143            // is only what its own window has heard.
144            aim: (*widget == crate::widget::Widget::Aim).then(|| crate::widget::aim::AimShown {
145                look: self.aims.get(&placed).copied().unwrap_or_default(),
146                remembered: (!self.follows.contains_key(&placed))
147                    .then(|| self.remembered_aim.map(str::to_owned))
148                    .flatten(),
149            }),
150            list: self.lists.get(&placed).copied(),
151            label: self.labels.get(&placed).cloned(),
152            roster: self.rosters.get(&placed).cloned(),
153            quick: self.quicks.get(&placed).cloned(),
154            hotbar: (*widget == crate::widget::Widget::Hotbar).then(|| {
155                crate::widget::hotbar::Shown::of(self.hotbars, self.hotbar_looks.get(&placed))
156            }),
157            // The window's own character's: another's are its window's.
158            tracked: (widget.is_effects() && !self.follows.contains_key(&placed))
159                .then(|| self.tracked.clone()),
160            circles: widget.is_effects().then(|| {
161                self.circles
162                    .cloned()
163                    .unwrap_or_else(crate::widget::circles::palette)
164            }),
165            timers: self.timers.get(&placed).cloned(),
166            effects: self.rows_of(placed, widget),
167            chip: self.chips.get(&placed).cloned(),
168            minimap: self.minimaps.get(&placed).copied(),
169            indicator: match widget {
170                // Every game indicator follows what every indicator does,
171                // its look or none.
172                crate::widget::Widget::Indicator(_) => {
173                    Some(self.indicators.get(&placed).cloned().unwrap_or_default())
174                }
175                crate::widget::Widget::CustomIndicator => {
176                    self.customs.get(&placed).map(|custom| custom.look.clone())
177                }
178                _ => None,
179            }
180            .map(|look| crate::widget::IndicatorShown {
181                look,
182                every: self.every_indicator.clone(),
183                icons: crate::widget::hotbar::Icons {
184                    dir: self.hotbars.icons_dir(),
185                    sheets: self.hotbars.sheets.clone(),
186                    size: None,
187                },
188                styled: match (widget, self.indicator_style) {
189                    (crate::widget::Widget::Indicator(which), Some(style)) => Some(
190                        std::path::Path::new(style)
191                            .join(format!("{}.png", which.id()))
192                            .to_string_lossy()
193                            .into_owned(),
194                    ),
195                    _ => None,
196                },
197            }),
198            custom: self.customs.get(&placed).cloned(),
199            compass: self
200                .compasses
201                .get(&placed)
202                .and_then(|look| look.picture.clone()),
203            profiles: (*widget == crate::widget::Widget::Trainer
204                && !self.follows.contains_key(&placed))
205            .then(|| self.profiles.names()),
206        }
207    }
208}

237/// The top bar: who, how connected, the keys, and the layout's controls:
238/// its menu, the drawers, Arrange and Lock. What the hands hold and the clocks are
239/// widgets, in the layout with the rest.
240/// The story's banners, a highlight's alerts, floating over every window
241/// as the *Banners* page says -- where, how wide, how solid, their colours,
242/// how long -- the newest last; nothing under them moves. A click takes
243/// them away, and they are not shown again (`gone`). The window is drawn
244/// again when the oldest is due to go.
245pub(super) fn alerts(
246    ui: &mut egui::Ui,
247    session: u32,
248    story: &hydra_ui::Story,
249    gone: &mut Option<std::time::Instant>,
250) {
251    use hydra_board::banner::{Across, Down};
252    let now = std::time::Instant::now();
253    let look = crate::play::banner(ui.ctx());
254    let lasting = std::time::Duration::from_secs(u64::from(look.seconds));
255    let alerts: Vec<_> = story
256        .alerts_within(now, lasting)
257        .into_iter()
258        .filter(|(at, _)| gone.is_none_or(|gone| *at > gone))
259        .collect();
260    let Some(oldest) = alerts.first().map(|(at, _)| *at) else {
261        return;
262    };
263    let ctx = ui.ctx().clone();
264    let ground = look
265        .ground
266        .map_or_else(|| theme::color(&ctx, T::Warning), theme::rgb);
267    let words = look
268        .text
269        .map_or_else(|| theme::readable_on(ground), theme::rgb);
270    let ground = ground.gamma_multiply(look.opacity);
271    let screen = ctx.content_rect();
272    let width = (screen.width() * look.width).max(120.0);
273    let align = match (look.across, look.down) {
274        (Across::Left, Down::Top) => egui::Align2::LEFT_TOP,
275        (Across::Centre, Down::Top) => egui::Align2::CENTER_TOP,
276        (Across::Right, Down::Top) => egui::Align2::RIGHT_TOP,
277        (Across::Left, Down::Middle) => egui::Align2::LEFT_CENTER,
278        (Across::Centre, Down::Middle) => egui::Align2::CENTER_CENTER,
279        (Across::Right, Down::Middle) => egui::Align2::RIGHT_CENTER,
280        (Across::Left, Down::Bottom) => egui::Align2::LEFT_BOTTOM,
281        (Across::Centre, Down::Bottom) => egui::Align2::CENTER_BOTTOM,
282        (Across::Right, Down::Bottom) => egui::Align2::RIGHT_BOTTOM,
283    };
284    // Clear of the top bar and the command line, and of the window's edge.
285    let offset = match look.down {
286        Down::Top => egui::vec2(0.0, 36.0),
287        Down::Middle => egui::Vec2::ZERO,
288        Down::Bottom => egui::vec2(0.0, -40.0),
289    } + match look.across {
290        Across::Left => egui::vec2(8.0, 0.0),
291        Across::Centre => egui::Vec2::ZERO,
292        Across::Right => egui::vec2(-8.0, 0.0),
293    };
294    let shown = egui::Area::new(egui::Id::new(("play-alerts", session)))
295        .order(egui::Order::Foreground)
296        .anchor(align, offset)
297        .interactable(true)
298        .show(&ctx, |ui| {
299            ui.set_width(width);
300            for (_, text) in &alerts {
301                egui::Frame::new()
302                    .fill(ground)
303                    .inner_margin(6.0)
304                    .corner_radius(theme::corner(&ctx))
305                    .show(ui, |ui| {
306                        ui.set_width(width - 12.0);
307                        ui.colored_label(words, *text);
308                    });
309            }
310        });
311    let clicked = ui
312        .interact(
313            shown.response.rect,
314            egui::Id::new(("play-alerts-click", session)),
315            egui::Sense::click(),
316        )
317        .on_hover_text("Click to take it away")
318        .clicked();
319    if clicked || shown.response.clicked() {
320        *gone = Some(now);
321        ctx.request_repaint();
322        return;
323    }
324    let left = lasting.saturating_sub(now.saturating_duration_since(oldest));
325    ctx.request_repaint_after(left);
326}

328pub(super) fn top(
329    ui: &mut egui::Ui,
330    view: &PlayView<'_>,
331    title: &TitleBar,
332    drawers: &mut crate::layout::Drawers,
333    (arranging, locked, guides): (&mut bool, &mut bool, &mut crate::layout::Guides),
334    unsaved: Option<&str>,
335) -> Option<Top> {
336    let mut asked = None;
337    // As Hydra's own title bar, the bar's own ground moves the window when
338    // dragged and maximizes it when double-clicked: taken before the
339    // buttons, which, added after, win the press where they are.
340    if title.own {
341        title_ground(ui);
342    }
343    // Three sides across the bar: the left reads from its edge, the right
344    // from its edge back, the window's own buttons outermost; the centre on
345    // the bar's middle, slid aside only as far as it must to clear them.
346    let middle = ui.max_rect().center().x;
347    ui.horizontal(|ui| {
348        for item in title.on(Side::Left) {
349            let pressed = bar_item(ui, view, item, drawers, (arranging, locked, guides));
350            asked = asked.or(pressed);
351        }
352        let left_end = ui.cursor().min.x;
353        ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
354            if title.own {
355                window_buttons(ui);
356            }
357            for item in title.on(Side::Right).iter().rev() {
358                let pressed = bar_item(ui, view, item, drawers, (arranging, locked, guides));
359                asked = asked.or(pressed);
360            }
361            if let Some(why) = unsaved {
362                ui.colored_label(theme::color(ui.ctx(), T::Wrong), why);
363            }
364            if let Some(on) = view.numlock {
365                ui.weak(if on { "NumLock on" } else { "NumLock off" });
366            }
367            if view.set != 0 {
368                ui.weak(format!("Set {}", view.set)).on_hover_text(
369                    "The macro set in use over set 0; Alt+0 goes back to set 0 alone",
370                );
371            }
372            let center = title.on(Side::Center);
373            if !center.is_empty() {
374                let right_start = ui.cursor().max.x;
375                let row = ui.max_rect().y_range();
376                let placed = centre_at(ui, middle, (left_end, right_start), row, |ui| {
377                    for item in center {
378                        let pressed =
379                            bar_item(ui, view, item, drawers, (arranging, locked, guides));
380                        asked = asked.or(pressed);
381                    }
382                });
383                ui.advance_cursor_after_rect(placed);
384            }
385        });
386    });
387    asked
388}

390/// Draw the title bar's centre with `draw`, its middle at `middle`, slid
391/// right or left just far enough to stay between `left_end` and
392/// `right_start`, the sides' items; with no room for it there, right after
393/// the left side. Its width is the last frame's, kept in egui's memory,
394/// since a group's width is known only once drawn; a changed width asks for
395/// another frame. Where it drew.
396fn centre_at(
397    ui: &mut egui::Ui,
398    middle: f32,
399    (left_end, right_start): (f32, f32),
400    row: egui::Rangef,
401    draw: impl FnOnce(&mut egui::Ui),
402) -> egui::Rect {
403    let key = ui.id().with("title-centre-width");
404    let wide = ui.data(|data| data.get_temp::<f32>(key)).unwrap_or(0.0);
405    let gap = ui.spacing().item_spacing.x;
406    let (least, most) = (left_end, right_start - gap - wide);
407    let from = if least <= most {
408        (middle - wide / 2.0).clamp(least, most)
409    } else {
410        least
411    };
412    let room = egui::Rect::from_x_y_ranges(from..=right_start.max(from), row);
413    let mut child = ui.new_child(
414        UiBuilder::new()
415            .max_rect(room)
416            .layout(egui::Layout::left_to_right(egui::Align::Center)),
417    );
418    draw(&mut child);
419    let drawn = child.min_rect();
420    if (drawn.width() - wide).abs() > 0.5 {
421        ui.data_mut(|data| data.insert_temp(key, drawn.width()));
422        ui.ctx().request_repaint();
423    }
424    drawn
425}

427/// As Hydra's own title bar, the bar's ground moves the window when
428/// dragged and maximizes it when double-clicked.
429fn title_ground(ui: &mut egui::Ui) {
430    let bar = ui.available_rect_before_wrap();
431    let ground = ui.interact(
432        bar,
433        ui.id().with("title-ground"),
434        egui::Sense::click_and_drag(),
435    );
436    if ground.drag_started_by(egui::PointerButton::Primary) {
437        ui.ctx().send_viewport_cmd(egui::ViewportCommand::StartDrag);
438    }
439    if ground.double_clicked_by(egui::PointerButton::Primary) {
440        let maximized = ui.input(|input| input.viewport().maximized.unwrap_or(false));
441        ui.ctx()
442            .send_viewport_cmd(egui::ViewportCommand::Maximized(!maximized));
443    }
444}

446/// `item` as a word of the bar, when it is one: whether it was.
447fn word(ui: &mut egui::Ui, view: &PlayView<'_>, item: &Item) -> bool {
448    let state = view.snapshot.map(|snapshot| &snapshot.state);
449    match item {
450        Item::Character => {
451            ui.strong(view.name);
452        }
453        Item::Profession => {
454            let profession = state.and_then(|state| state.character.identity.profession.as_deref());
455            ui.weak(profession.unwrap_or("?"));
456        }
457        Item::Connection => {
458            let (word, token) = crate::hub::connection(view.lifecycle);
459            ui.colored_label(theme::color(ui.ctx(), token), word);
460        }
461        Item::Room => {
462            let room = state.and_then(|state| state.room.title.as_deref());
463            ui.colored_label(
464                theme::color(ui.ctx(), T::RoomName),
465                room.unwrap_or("Room unknown"),
466            );
467        }
468        Item::Number => {
469            let number = state.and_then(|state| state.room.id.as_deref());
470            ui.weak(number.unwrap_or("?"));
471        }
472        // A game value: its name, and its number as the model works it
473        // out now, whole when it is whole.
474        Item::Value(name) => {
475            let value = state.and_then(|state| state.evaluate(name));
476            let said = match value {
477                Some(n) if n.fract() == 0.0 => format!("{name} {n:.0}"),
478                Some(n) => format!("{name} {n:.1}"),
479                None => format!("{name} ?"),
480            };
481            ui.weak(said);
482        }
483        _ => return false,
484    }
485    true
486}

488/// The − and + that step the zoom of every window, reading − then + whichever
489/// way the side lays them out: what was pressed.
490fn zoom_buttons(ui: &mut egui::Ui) -> Option<Top> {
491    let plain = |ui: &mut egui::Ui, glyph: &str, said: &str, by: i8| {
492        ui.add(egui::Button::new(glyph).frame(false))
493            .on_hover_text(said)
494            .clicked()
495            .then_some(Top::Zoom(by))
496    };
497    let smaller = ("\u{2212}", "Draw everything smaller", -1);
498    let larger = ("+", "Draw everything larger", 1);
499    let (first, second) = if ui.layout().main_dir() == egui::Direction::RightToLeft {
500        (larger, smaller)
501    } else {
502        (smaller, larger)
503    };
504    plain(ui, first.0, first.1, first.2).or_else(|| plain(ui, second.0, second.1, second.2))
505}

518/// One item of the bar: a word, or a button, and what the button asked
519/// when pressed.
520fn bar_item(
521    ui: &mut egui::Ui,
522    view: &PlayView<'_>,
523    item: &Item,
524    drawers: &mut crate::layout::Drawers,
525    (arranging, locked, guides): (&mut bool, &mut bool, &mut crate::layout::Guides),
526) -> Option<Top> {
527    if word(ui, view, item) {
528        return None;
529    }
530    let mut asked = None;
531    match item {
532        Item::Settings => {
533            if ui
534                .button("Settings")
535                .on_hover_text("This character's settings, its keys and its widgets' pages")
536                .clicked()
537            {
538                asked = Some(Top::Settings);
539            }
540        }
541        Item::Zoom => asked = zoom_buttons(ui),
542        Item::Menu => {
543            let place = crate::main_menu::Place::Play {

624/// Drawer `zone`'s icon on the bar: a small box with the drawer's edge
625/// filled, in the accent while it is `open`, faint while it is shut.
626/// Whether it was clicked.
627fn drawer_icon(ui: &mut egui::Ui, zone: crate::layout::Zone, open: bool) -> bool {
628    use crate::layout::Zone;
629    let size = egui::Vec2::splat(ui.spacing().interact_size.y);
630    let (rect, response) = ui.allocate_exact_size(size, egui::Sense::click());
631    let name = zone.name();
632    response.widget_info(|| egui::WidgetInfo::selected(egui::WidgetType::Button, true, open, name));
633    let hover = if open {
634        format!("Shut the {}", name.to_lowercase())
635    } else {
636        format!("Open the {}", name.to_lowercase())
637    };
638    let response = response.on_hover_text(hover);
639    let visuals = ui.style().interact(&response);
640    let edge = visuals.fg_stroke.color;
641    let fill = if open {
642        theme::color(ui.ctx(), T::Accent)
643    } else {
644        ui.visuals().weak_text_color()
645    };
646    let painter = ui.painter();
647    let box_ = rect.shrink(3.0);
648    painter.rect_stroke(
649        box_,
650        1.0,
651        egui::Stroke::new(1.0, edge),
652        egui::StrokeKind::Inside,
653    );
654    let thick = (box_.width() / 3.0).round();
655    let strip = match zone {
656        Zone::Top => egui::Rect::from_min_size(box_.min, egui::vec2(box_.width(), thick)),
657        Zone::Bottom => {
658            egui::Rect::from_min_max(egui::pos2(box_.min.x, box_.max.y - thick), box_.max)
659        }
660        Zone::Left => egui::Rect::from_min_size(box_.min, egui::vec2(thick, box_.height())),
661        Zone::Right | Zone::Main => {
662            egui::Rect::from_min_max(egui::pos2(box_.max.x - thick, box_.min.y), box_.max)
663        }
664    };
665    painter.rect_filled(strip, 1.0, fill);
666    response.clicked()
667}

700/// How close to the window's edge a press resizes it, points.
701const EDGE: f32 = 6.0;
702
703/// With no system frame, the window's edges resize it: the pointer near
704/// one shows the resize cursor, and a press there begins the resize.
705pub(super) fn resize_edges(ctx: &egui::Context) {
706    use egui::{CursorIcon, ResizeDirection, ViewportCommand};
707    let maximized = ctx.input(|input| input.viewport().maximized.unwrap_or(false));
708    if maximized {
709        return;
710    }
711    let Some(at) = ctx.input(|input| input.pointer.hover_pos()) else {
712        return;
713    };
714    let screen = ctx.viewport_rect();
715    let (left, right) = (at.x - screen.min.x < EDGE, screen.max.x - at.x < EDGE);
716    let (top, bottom) = (at.y - screen.min.y < EDGE, screen.max.y - at.y < EDGE);
717    let (direction, cursor) = match (left, right, top, bottom) {
718        (true, _, true, _) => (ResizeDirection::NorthWest, CursorIcon::ResizeNwSe),
719        (_, true, true, _) => (ResizeDirection::NorthEast, CursorIcon::ResizeNeSw),
720        (true, _, _, true) => (ResizeDirection::SouthWest, CursorIcon::ResizeNeSw),
721        (_, true, _, true) => (ResizeDirection::SouthEast, CursorIcon::ResizeNwSe),
722        (true, _, _, _) => (ResizeDirection::West, CursorIcon::ResizeHorizontal),
723        (_, true, _, _) => (ResizeDirection::East, CursorIcon::ResizeHorizontal),
724        (_, _, true, _) => (ResizeDirection::North, CursorIcon::ResizeVertical),
725        (_, _, _, true) => (ResizeDirection::South, CursorIcon::ResizeVertical),
726        _ => return,
727    };
728    ctx.set_cursor_icon(cursor);
729    if ctx.input(|input| input.pointer.primary_pressed()) {
730        ctx.send_viewport_cmd(ViewportCommand::BeginResize(direction));
731    }
732}

739/// What a window holds, filling it: its window is the size its layout says,
740/// never its content's. One widget is given everything; a custom window's
741/// widgets are drawn bare in their cells, once the cells are kept to the
742/// inside it has now (`Custom::fit`). Nothing spills out of its cell: a
743/// one-line widget stays one line, and the rest scroll. A tab stack shows a
744/// tab for each of its widgets, a click on one showing it, each not showing
745/// with what it has said since it last showed. Its inside, where it drew.
746pub(super) fn holder(
747    ui: &mut egui::Ui,
748    holds: &mut Holds,
749    drawing: &mut Drawing<'_>,
750) -> egui::Rect {
751    let inside = ui.available_rect_before_wrap();
752    ui.set_min_size(inside.size());
753    let area = inside;
754    match holds {
755        Holds::One(placed) => shown(ui, placed, drawing),
756        Holds::Custom(custom) => {
757            // Not to the pass egui lays a window out in at its narrowest as
758            // a resize begins, then discards: kept to it, a narrow cell came
759            // back from the scaling at its smallest, not its width.
760            // Resized with Arrange on, a window leaves its widgets their
761            // size, room made or taken for arranging them; with it off they
762            // scale with the window.
763            if ui.is_sizing_pass() {
764            } else if drawing.arranging {
765                custom.keep(area.size());
766            } else {
767                custom.fit(area.size());
768            }
769            if custom.cells.is_empty() {
770                ui.new_child(UiBuilder::new().max_rect(area.shrink(4.0)))
771                    .weak("Empty. With Arrange on, in the Layout menu, drop widgets here.");
772            }
773            let gap = theme::shape(ui.ctx()).pane_gap;
774            let strip = theme::tab_height(ui.ctx(), crate::widget::LINE);
775            for cell in &mut custom.cells {
776                let whole = cell.rect().translate(area.min.to_vec2()).intersect(area);
777                // Half the gap off each side: the gap between two.
778                let at = whole.shrink(gap / 2.0);
779                if !at.is_positive() {
780                    continue;
781                }
782                let widths = crate::widget::tab::widths(&cell.tabs, drawing.tabs);
783                let height = cell.tab_height.map_or(strip, f32::from);
784                let (tabs, body) = tabs_and_body(at, &widths, height, cell.bar);
785                let reorder = tab_strip(ui, cell, &tabs, drawing);
786                if let Some((at, to)) = reorder
787                    && cell.move_tab(at, to)
788                {
789                    drawing.reordered = true;
790                }
791                let Some(placed) = cell.shown() else {
792                    continue;
793                };
794                let mut child =
795                    ui.new_child(UiBuilder::new().max_rect(body).id_salt(("cell", placed.id)));
796                shown(&mut child, placed, drawing);
797            }
798            ui.advance_cursor_after_rect(inside);
799        }
800    }
801    inside
802}

804/// `cell`'s tab strip drawn, a tab for each of its widgets at `tabs`: a
805/// click shows that tab, and a tab dragged along the strip and let go asks
806/// to move, as (from, to). With Arrange on the strip is arranging's.
807fn tab_strip(
808    ui: &mut egui::Ui,
809    cell: &mut crate::layout::Cell,
810    tabs: &[egui::Rect],
811    drawing: &mut Drawing<'_>,
812) -> Option<(usize, usize)> {
813    let bar = cell.bar;
814    let style = cell.look.unwrap_or_else(|| theme::shape(ui.ctx()).tabs);
815    let mut reorder = None;
816    for (index, (tab, rect)) in cell.tabs.iter().zip(tabs.iter().copied()).enumerate() {
817        let showing = index == cell.showing;
818        let look = drawing.tabs.get(&tab.id);
819        let quiet = look.is_some_and(|look| look.quiet);
820        // A tab showing is read; a quiet one is never counted.
821        let unread = unread(drawing.read, tab, &drawing.seen).filter(|_| !showing && !quiet);
822        if unread.is_some() {
823            drawing.unread.push(tab.id);
824        }
825        let name = crate::widget::tab::name(look, &tab.widget.name(), unread);
826        let color = match (showing, unread.is_some()) {
827            (true, _) => look.and_then(|look| look.showing),
828            (false, true) => look.and_then(|look| look.unread),
829            (false, false) => None,
830        };
831        let name = match color {
832            Some([r, g, b]) => egui::RichText::new(name).color(egui::Color32::from_rgb(r, g, b)),
833            None => egui::RichText::new(name),
834        };
835        // Text taller than its tab is drawn smaller, to fit.
836        let font = egui::TextStyle::Button.resolve(ui.style());
837        let row = ui.fonts_mut(|fonts| fonts.row_height(&font));
838        let room = rect.height() - 2.0 * ui.spacing().button_padding.y;
839        let name = if row > room && room > 0.0 {
840            name.size(font.size * room / row)
841        } else {
842            name
843        };
844        let mut child = ui.new_child(UiBuilder::new().max_rect(rect).id_salt(("tab", tab.id)));
845        // A button is never shorter than egui's interact size: a tab is its
846        // strip's height, however short.
847        let least = &mut child.spacing_mut().interact_size.y;
848        *least = least.min(rect.height());
849        let button = match style {
850            TabStyle::Plain | TabStyle::Separated => egui::Button::selectable(showing, name),
851            TabStyle::Boxed | TabStyle::Filled => {
852                tab_ground(ui, rect, style, showing);
853                egui::Button::new(name).frame(false)
854            }
855        }
856        .wrap_mode(egui::TextWrapMode::Truncate);
857        let response = child.add_sized(rect.size(), button);
858        if response.clicked() && cell.showing != index {
859            cell.showing = index;
860            // The tab shown is kept with the layout, as a key turning it is.
861            drawing.reordered = true;
862        }
863        if drawing.arranging {
864            continue;
865        }
866        let response = response.interact(egui::Sense::drag());
867        if let Some(at) = response.interact_pointer_pos()
868            && let Some(to) = crate::layout::tab_at(tabs, at, bar.vertical())
869            && to != index
870        {
871            if response.dragged() {
872                slot_mark(ui, tabs, index, to, bar.vertical());
873            }
874            if response.drag_stopped() {
875                reorder = Some((index, to));
876            }
877        }
878    }
879    if style == TabStyle::Separated {
880        tab_separators(ui, tabs, bar.vertical());
881    }
882    reorder
883}

885/// A tab at `rect` given its ground and edge under its words, as `style`
886/// draws a tab `showing` or not: boxed, an edge round it and the shown one
887/// on its ground; filled, every one on the tab ground and the shown one on
888/// its own. Under the pointer a tab not shown takes the raised ground.
889fn tab_ground(ui: &egui::Ui, rect: egui::Rect, style: TabStyle, showing: bool) {
890    let ctx = ui.ctx();
891    let rect = rect.shrink(1.0);
892    let corner = f32::from(theme::shape(ctx).corners.controls);
893    let pointed = !showing && ui.rect_contains_pointer(rect);
894    let fill = match (style, showing, pointed) {
895        (_, true, _) => theme::color(ctx, T::TabShown),
896        (_, false, true) => theme::color(ctx, T::Raised),
897        (TabStyle::Filled, false, false) => theme::color(ctx, T::Tab),
898        _ => egui::Color32::TRANSPARENT,
899    };
900    let edge = match style {
901        TabStyle::Boxed => egui::Stroke::new(1.0, theme::color(ctx, T::Line)),
902        _ => egui::Stroke::NONE,
903    };
904    ui.painter()
905        .rect(rect, corner, fill, edge, egui::StrokeKind::Inside);
906}

908/// A line between each two of the strip's tabs at `tabs`: across a strip
909/// down a side (`vertical`), else upright.
910fn tab_separators(ui: &egui::Ui, tabs: &[egui::Rect], vertical: bool) {
911    let stroke = egui::Stroke::new(1.0, theme::color(ui.ctx(), T::Line));
912    for pair in tabs.windows(2) {
913        let tab = pair[0];
914        if vertical {
915            ui.painter()
916                .hline(tab.x_range().shrink(4.0), tab.max.y, stroke);
917        } else {
918            ui.painter()
919                .vline(tab.max.x, tab.y_range().shrink(3.0), stroke);
920        }
921    }
922}

924/// The place in the strip a tab dragged from `from` would take at `to`,
925/// marked: a line at the edge of that tab it would go beside, across the
926/// strip when it runs down a side (`vertical`).
927pub(super) fn slot_mark(
928    ui: &egui::Ui,
929    tabs: &[egui::Rect],
930    from: usize,
931    to: usize,
932    vertical: bool,
933) {
934    let Some(rect) = tabs.get(to) else {
935        return;
936    };
937    let stroke = egui::Stroke::new(2.0, theme::color(ui.ctx(), T::Accent));
938    if vertical {
939        let y = if to > from { rect.max.y } else { rect.min.y };
940        ui.painter().hline(rect.x_range(), y, stroke);
941    } else {
942        let x = if to > from { rect.max.x } else { rect.min.x };
943        ui.painter().vline(x, rect.y_range(), stroke);
944    }
945}

947/// `placed` drawn into `ui`, showing: so what it has said is read, up to
948/// now. The one way a widget is drawn, standalone, alone in a cell or a tab.
949/// One that follows another character draws from that character, named on
950/// it, or says that character is not running; a story never follows.
951fn shown(ui: &mut egui::Ui, placed: &Placed, drawing: &mut Drawing<'_>) {
952    let rect = ui.max_rect();
953    drawing.rects.push((placed.id, rect));
954    if ui.input(|input| input.pointer.primary_pressed()) && ui.rect_contains_pointer(rect) {
955        drawing.pressed = Some(placed.id);
956    }
957    if drawing.in_use == Some(placed.id) {
958        // The window in use, which a scrolling key acts on.
959        let stroke = egui::Stroke::new(1.0, theme::color(ui.ctx(), T::Accent).gamma_multiply(0.5));
960        ui.painter().rect_stroke(
961            rect,
962            theme::corner(ui.ctx()),
963            stroke,
964            egui::StrokeKind::Inside,
965        );
966    }
967    // The theme's room round the widget, inside its pane.
968    let inside = padded(rect, theme::shape(ui.ctx()).pane_padding);
969    if inside == rect {
970        drawn(ui, placed, drawing);
971    } else {
972        let mut inner = ui.new_child(UiBuilder::new().max_rect(inside));
973        drawn(&mut inner, placed, drawing);
974        ui.advance_cursor_after_rect(rect);
975    }
976}

978/// `rect` with `padding` taken off each side, but never so much that less
979/// than a line is left either way: the theme's room round a widget gives
980/// way in a pane too small for both, so a one-line widget keeps its line.
981pub(super) fn padded(rect: egui::Rect, padding: f32) -> egui::Rect {
982    let room = |size: f32| padding.min((size - crate::widget::LINE) / 2.0).max(0.0);
983    rect.shrink2(egui::vec2(room(rect.width()), room(rect.height())))
984}

986/// `placed` drawn, as [`shown`] says: in its face, when its kind or it has
987/// one of its own; an effects window's own over every effects window's.
988fn drawn(ui: &mut egui::Ui, placed: &Placed, drawing: &mut Drawing<'_>) {
989    let key = placed.widget.type_key();
990    let own = drawing.faces.get(&placed.id);
991    let every = &drawing.every_effects.face;
992    let followed = (placed.widget.is_effects_window() && !every.is_empty())
993        .then(|| own.map_or_else(|| every.clone(), |own| every.under(own)));
994    if let Some(face) = theme::face(ui.ctx(), &key, followed.as_ref().or(own)) {
995        let families = crate::fonts::loaded(ui.ctx());
996        ui.scope(|ui| {
997            theme::faced(ui.style_mut(), &face, &families);
998            drawn_in(ui, placed, drawing);
999        });
1000    } else {
1001        drawn_in(ui, placed, drawing);
1002    }
1003}

1005/// `placed` drawn in the face `ui` has.
1006fn drawn_in(ui: &mut egui::Ui, placed: &Placed, drawing: &mut Drawing<'_>) {
1007    let id = widget_id(drawing.session, placed.id);
1008    // The window's own command line, whoever else the widget would follow.
1009    if placed.widget == crate::widget::Widget::Input
1010        && let Some(typing) = &mut drawing.typing
1011    {
1012        drawing.entered |= super::command_line::command_line(ui, typing);
1013        return;
1014    }
1015    let follows = drawing
1016        .follows
1017        .get(&placed.id)
1018        .filter(|_| !placed.widget.is_story());
1019    let Some(who) = follows else {
1020        if let Some(count) = placed.widget.count(&drawing.seen) {
1021            drawing.read.insert(placed.id, count);
1022        }
1023        let chosen = drawing.chosen(placed);
1024        match placed.widget.draw_with(ui, &drawing.seen, id, &chosen) {
1025            Some(crate::widget::Clicked::Set { key, to }) => {
1026                drawing.set = Some((placed.id, key, to));
1027            }
1028            Some(line) => drawing.sent = Some(line),
1029            None => {}
1030        }
1031        return;
1032    };
1033    // A bare name two characters share -- one name on two games -- names
1034    // neither: which is meant is for the player to say.
1035    let mut named = drawing.others.iter().filter(|other| other.is(who));
1036    let (one, another) = (named.next(), named.next());
1037    if another.is_some() {
1038        ui.weak(format!(
1039            "More than one {who} is running: choose which in this window's Show for."
1040        ));
1041        return;
1042    }
1043    match one {
1044        Some(other) => {
1045            let seen = Seen {
1046                snapshot: other.snapshot.as_deref(),
1047                // Named by its game too when the follow does: one name
1048                // may be on two games.
1049                who: Some(if who.contains(':') { who } else { &other.name }),
1050                minimap: other.minimap.as_ref(),
1051                travel: None,
1052                notes: None,
1053                trainer: None,
1054                ..drawing.seen
1055            };
1056            // Another character's widget sends nothing on this one's.
1057            let chosen = drawing.chosen(placed);
1058            let _ = placed.widget.draw_with(ui, &seen, id, &chosen);
1059        }
1060        None => {
1061            ui.weak(format!("{who} is not running."));
1062        }
1063    }
1064}

1090/// What `placed` has said since it last showed, when it counts and that is

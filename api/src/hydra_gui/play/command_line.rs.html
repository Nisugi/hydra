1//! The command line: what is typed, sent on Enter, with the ghost of a
2//! line sent before after it. One field, drawn at the window's foot, or
3//! inside a Command line widget when one is placed: the window lends it
4//! what is typed ([`Typing`]) and sends what was entered once its widgets
5//! are drawn. Drawn inside the widget, it is that widget's window's own and
6//! goes where the window goes, above or behind the others as it is.
7
8use egui::Id;
9use hydra_ui::History;
10
11use super::complete;
12
13/// What the command line edits and reads, lent to whatever draws it.
14pub(super) struct Typing<'a> {
15    /// The line being typed.
16    pub(super) input: &'a mut String,
17    /// Up or down filled the line, or a key did: its cursor goes to the end.
18    pub(super) walked: &'a mut bool,
19    /// What was sent, which the ghost is taken from.
20    pub(super) history: &'a History,
21    /// The field's id: the keys' and the focus's way to it.
22    pub(super) id: Id,
23}
24
25/// The command line in `ui`, filling its width: whether Enter was pressed
26/// in it, the line then left for the window to send.
27pub(super) fn command_line(ui: &mut egui::Ui, typing: &mut Typing<'_>) -> bool {
28    let id = typing.id;
29    let output = egui::TextEdit::singleline(&mut *typing.input)
30        .id(id)
31        // Tab is the keys', never egui's to move the keyboard on.
32        .lock_focus(true)
33        .hint_text("Type a command")
34        .desired_width(f32::INFINITY)
35        .show(ui);
36    let response = output.response.response.clone();
37    if std::mem::take(typing.walked) {
38        let mut state = egui::TextEdit::load_state(ui.ctx(), id).unwrap_or_default();
39        let end = egui::text::CCursor::new(typing.input.chars().count());
40        state
41            .cursor
42            .set_char_range(Some(egui::text::CCursorRange::one(end)));
43        state.store(ui.ctx(), id);
44    }
45    ghost(ui, typing, &output);
46    let entered = response.lost_focus() && ui.input(|input| input.key_pressed(egui::Key::Enter));
47    if entered {
48        response.request_focus();
49    }
50    entered
51}
52
53/// The ghost after what is typed (`complete.rs`), in the weak colour,
54/// while the cursor stands at the end with nothing selected.
55fn ghost(ui: &egui::Ui, typing: &Typing<'_>, output: &egui::text_edit::TextEditOutput) {
56    let Some(range) = output.cursor_range else {
57        return;
58    };
59    let end = typing.input.chars().count();
60    if range.primary.index.0 != end || range.secondary.index.0 != end {
61        return;
62    }
63    let Some(rest) = complete::ghost(typing.input, typing.history) else {
64        return;
65    };
66    let font = egui::TextStyle::Body.resolve(ui.style());
67    let galley = ui
68        .painter()
69        .layout_no_wrap(rest, font, ui.visuals().weak_text_color());
70    let at = egui::pos2(
71        output.galley_pos.x + output.galley.rect.width(),
72        output.galley_pos.y,
73    );
74    // Inside the field, not the text's own clip, which hugs what is typed
75    // and would cut the ghost away.
76    let field = output.response.rect.shrink2(egui::vec2(0.0, 1.0));
77    ui.painter()
78        .with_clip_rect(field)
79        .galley(at, galley, ui.visuals().weak_text_color());
80}

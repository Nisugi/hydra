1//! The doll calibrator: where each part sits on a doll
2//! picture, and how its dots look, saved **into the picture**
3//! (`calibration.rs`).
4//!
5//! Opened from an Injuries widget's right-click menu once it has a picture.
6//! The parts are listed, the calibrated ones marked; a click on the picture
7//! puts the chosen part there and moves on to the next. The picture is drawn
8//! by the widget's own renderer (`widget/doll.rs`, `drawn`) with the
9//! calibration as it stands, every part showing a preview wound or scar, so
10//! what is calibrated is what the widget will show: a canvas of its own
11//! could put dots where the widget draws art.
12
13use std::collections::BTreeMap;
14use std::path::Path;
15
16use egui::{Pos2, Rect, Stroke, Vec2};
17use hydra_model::Injury;
18
19use crate::calibration::{self, Calibration};
20use crate::widget::doll::{self, PARTS};

22/// The calibrator of one Injuries widget's picture, while it is open.
23#[derive(Debug)]
24pub(crate) struct Calibrator {
25    /// The Injuries widget it was opened from.
26    pub(crate) placed: u32,
27    /// The picture's path.
28    picture: String,
29    calibration: Calibration,
30    /// The part the next click places, by its place in [`PARTS`].
31    chosen: usize,
32    /// A click moves on to the next part.
33    advance: bool,
34    /// The preview shows scars, else wounds.
35    scars: bool,
36    /// The preview's rank.
37    rank: u8,
38    /// What saving last said.
39    said: Option<String>,
40    /// Where the picture was last drawn, on the screen.
41    drawn_at: Option<Rect>,
42}
43
44impl Calibrator {
45    /// Open on the picture at `picture`, its calibration as saved.
46    pub(crate) fn open(placed: u32, picture: String) -> Self {
47        let calibration = Calibration::of_picture(Path::new(&picture));
48        Self {
49            placed,
50            picture,
51            calibration,
52            chosen: 0,
53            advance: true,
54            scars: false,
55            rank: 2,
56            said: None,
57            drawn_at: None,
58        }
59    }
60
61    /// Draw it in a window of its own, `id` its window's; `false` once it is
62    /// closed.
63    pub(crate) fn show(&mut self, context: &egui::Context, id: egui::Id) -> bool {
64        let mut open = true;
65        egui::Window::new("Calibrate doll")
66            .id(id)
67            .open(&mut open)
68            .default_size([560.0, 520.0])
69            .show(context, |ui| self.contents(ui));
70        open
71    }
72
73    fn contents(&mut self, ui: &mut egui::Ui) {
74        ui.horizontal_top(|ui| {
75            ui.vertical(|ui| {
76                // A column of its own, so its separator and sliders do not
77                // take the picture's room.
78                ui.set_max_width(210.0);
79                self.parts(ui);
80                ui.separator();
81                self.style(ui);
82            });
83            ui.add_space(8.0);
84            ui.vertical(|ui| self.canvas(ui));
85        });
86    }
87
88    /// The parts, the chosen one and the calibrated ones marked.
89    fn parts(&mut self, ui: &mut egui::Ui) {
90        ui.label("Click where each part is:");
91        for (at, part) in PARTS.iter().enumerate() {
92            let placed = if self.calibration.anchors.contains_key(part.id) {
93                "\u{2022} "
94            } else {
95                "   "
96            };
97            if ui
98                .selectable_label(self.chosen == at, format!("{placed}{}", part.name))
99                .clicked()
100            {
101                self.chosen = at;
102            }
103        }
104        ui.checkbox(&mut self.advance, "Then the next part");
105        if ui
106            .button("Use default")
107            .on_hover_text("This part back where the doll puts it on any picture")
108            .clicked()
109        {
110            self.calibration.anchors.remove(PARTS[self.chosen].id);
111        }
112        if ui.button("Reset all").clicked() {
113            self.calibration.anchors.clear();
114        }
115    }
116
117    /// The picture, drawn as the widget draws it, and where a click puts
118    /// the chosen part.
119    fn canvas(&mut self, ui: &mut egui::Ui) {
120        let Some(texture) = crate::pictures::picture(ui.ctx(), &self.picture) else {
121            ui.label(format!("{} could not be read.", self.picture));
122            return;
123        };
124        let preview = self.preview();
125        let layers = crate::doll_art::layers(ui.ctx(), &self.picture);
126        let (rect, response) = ui
127            .allocate_ui(Vec2::new(300.0, 340.0), |ui| {
128                doll::drawn(
129                    ui,
130                    Some(&preview),
131                    Some(doll::Over {
132                        texture: &texture,
133                        calibration: &self.calibration,
134                        layers: &layers,
135                    }),
136                    egui::Sense::click(),
137                )
138            })
139            .inner;
140        self.drawn_at = Some(rect);
141        let part = &PARTS[self.chosen];
142        let (x, y) = self.calibration.anchor(part);
143        let marked = rect.min + Vec2::new(x * rect.width(), y * rect.height());
144        ui.painter().circle_stroke(
145            marked,
146            12.0,
147            Stroke::new(2.0, crate::theme::color(ui.ctx(), crate::theme::T::Mark)),
148        );
149        if response.clicked()
150            && let Some(at) = response.interact_pointer_pos()
151            && let Some(put) = fraction(rect, at)
152        {
153            self.calibration.anchors.insert(part.id, put);
154            self.said = None;
155            if self.advance {
156                self.chosen = (self.chosen + 1) % PARTS.len();
157            }
158        }
159    }
160
161    /// Every part hurt at the preview's rank, wounds or scars.
162    fn preview(&self) -> BTreeMap<String, Injury> {
163        let injury = if self.scars {
164            Injury {
165                wound: 0,
166                scar: self.rank,
167            }
168        } else {
169            Injury {
170                wound: self.rank,
171                scar: 0,
172            }
173        };
174        PARTS
175            .iter()
176            .map(|part| (part.id.to_owned(), injury))
177            .collect()
178    }
179
180    /// The dots' size and opacity, the preview, and saving.
181    fn style(&mut self, ui: &mut egui::Ui) {
182        ui.add(egui::Slider::new(&mut self.calibration.diameter, 0.02..=0.2).text("dot size"));
183        ui.add(egui::Slider::new(&mut self.calibration.opacity, 0.2..=1.0).text("opacity"));
184        ui.horizontal(|ui| {
185            ui.radio_value(&mut self.scars, false, "Wounds");
186            ui.radio_value(&mut self.scars, true, "Scars");
187        });
188        ui.add(egui::Slider::new(&mut self.rank, 1..=3).text("preview rank"));
189        if ui
190            .button("Save")
191            .on_hover_text("Into the picture itself, so it travels with it")
192            .clicked()
193        {
194            calibration::write(Path::new(&self.picture), &self.calibration);
195            self.said = Some("Saved into the picture.".to_owned());
196        }
197        // The batch that writes it may yet fail, and say why.
198        if let Some(why) = crate::store::failed(Path::new(&self.picture)) {
199            ui.colored_label(ui.visuals().warn_fg_color, format!("Not saved: {why}"));
200        } else if let Some(said) = &self.said {
201            ui.label(said.as_str());
202        }
203    }
204}

206impl super::Play {
207    /// The calibrator, while it is open.
208    pub(super) fn calibrator(&mut self, context: &egui::Context) {
209        if let Some(open) = &mut self.calibrating
210            && !open.show(
211                context,
212                egui::Id::new(("calibrate", self.session, open.placed)),
213            )
214        {
215            self.calibrating = None;
216        }
217    }
218}

220/// Where `at` lies on the picture drawn in `rect`, as fractions of it, or
221/// `None` off it.
222pub(crate) fn fraction(rect: Rect, at: Pos2) -> Option<(f32, f32)> {
223    if !rect.contains(at) || rect.width() <= 0.0 || rect.height() <= 0.0 {
224        return None;
225    }
226    Some((
227        (at.x - rect.min.x) / rect.width(),
228        (at.y - rect.min.y) / rect.height(),
229    ))
230}

238    /// A picture of the test's own, in a folder of its own.
239    fn picture(name: &str) -> Option<std::path::PathBuf> {
240        let dir =
241            std::env::temp_dir().join(format!("hydra-calibrator-{}-{name}", std::process::id()));
242        let _ = std::fs::remove_dir_all(&dir);
243        std::fs::create_dir_all(&dir).ok()?;
244        let path = dir.join("doll.png");
245        image::RgbaImage::from_pixel(100, 200, image::Rgba([40, 60, 90, 255]))
246            .save(&path)
247            .ok()?;
248        Some(path)
249    }

251    /// The calibrator open on it: the chosen part placed where the picture
252    /// is clicked, the next part chosen, and Save writing it into the
253    /// picture, where the widget reads it.
254    #[test]
255    fn a_click_places_the_part_and_save_writes_it_into_the_picture() {
256        let path = picture("click").unwrap();
257        let open = Calibrator::open(7, path.to_string_lossy().into_owned());
258        let mut harness = egui_kittest::Harness::builder()
259            .with_size((640.0, 600.0))
260            .build_ui_state(
261                |ui, calibrator: &mut Calibrator| {
262                    let _ = calibrator.show(ui.ctx(), egui::Id::new("calibrate"));
263                },
264                open,
265            );
266        harness.run();
267        crate::fixture::picture(&mut harness, "calibrator");
268
269        let drawn = harness.state().drawn_at.expect("the picture drawn");
270        // A click: pressed and let go where it was pressed.
271        harness.hover_at(drawn.center());
272        harness.step();
273        harness.drag_at(drawn.center());
274        harness.step();
275        harness.drop_at(drawn.center());
276        harness.run();
277        let placed = harness.state().calibration.anchors.get("head").copied();
278        let (x, y) = placed.expect("the head placed");
279        assert!((x - 0.5).abs() < 0.02 && (y - 0.5).abs() < 0.02, "{x}, {y}");
280        assert_eq!(harness.state().chosen, 1, "on to the left eye");
281
282        harness.get_by_label("Save").click();
283        harness.run();
284        assert_eq!(
285            harness.state().said.as_deref(),
286            Some("Saved into the picture."),
287            "saving said so"
288        );
289        let saved = crate::calibration::Calibration::of_picture(&path);
290        assert_eq!(saved.anchors.get("head").copied(), placed, "in the picture");
291        assert!(
292            crate::calibration::embedded(&std::fs::read(&path).unwrap()).is_some(),
293            "and in no file beside it"
294        );
295        assert!(!path.with_extension("toml").exists());
296        let _ = std::fs::remove_dir_all(path.parent().unwrap());
297    }

299    /// A click is a fraction of the drawn picture, whatever its size: the
300    /// same place on a picture drawn small or large.
301    #[test]
302    fn a_click_is_a_fraction_of_the_picture_at_any_size() {
303        let small = Rect::from_min_max(Pos2::new(10.0, 20.0), Pos2::new(70.0, 140.0));
304        let large = Rect::from_min_max(Pos2::new(0.0, 0.0), Pos2::new(300.0, 600.0));
305        assert_eq!(fraction(small, Pos2::new(25.0, 50.0)), Some((0.25, 0.25)));
306        assert_eq!(fraction(large, Pos2::new(75.0, 150.0)), Some((0.25, 0.25)));
307        assert_eq!(
308            fraction(small, Pos2::new(5.0, 50.0)),
309            None,
310            "off the picture"
311        );
312    }

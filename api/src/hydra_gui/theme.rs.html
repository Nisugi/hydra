1//! The palette as egui draws it: a token's colour, the theme worn, and the
2//! one place in this crate a colour is made from a palette.
3//!
4//! Every widget asks here for a colour by its meaning ([`color`]) and never
5//! writes one. The app [`wear`]s a theme: its outfit is kept in egui's own
6//! data, where [`palette`] reads it, and egui's visuals are set from its
7//! surfaces and text, so every stock control follows. Until one is worn,
8//! [`palette`] answers with a bare palette ([`Palette::bare`]).
9

17/// Where the outfit worn is kept in egui's data.
18fn key() -> egui::Id {
19    egui::Id::new("theme")

22/// The outfit worn for the window `ctx` is drawing: its character's own
23/// ([`wearing`]), else Hydra's, else Hydra's before it had themes.
24pub(crate) fn outfit(ctx: &egui::Context) -> Outfit {
25    let viewport = ctx.viewport_id();
26    ctx.data(|data| {
27        data.get_temp::<Outfit>(key().with(viewport))
28            .or_else(|| data.get_temp::<Outfit>(key()))
29    })
30    .unwrap_or_default()
31}

33/// The palette in force for what `ctx` is drawing.
34pub(crate) fn palette(ctx: &egui::Context) -> Palette {
35    outfit(ctx).palette
36}

38/// The story's text style: the theme's story font at its story size.
39pub(crate) fn story_style() -> TextStyle {
40    TextStyle::Name("story".into())
41}

43/// The story's font, as `style` has it; the body's where no theme set it.
44pub(crate) fn story_font(style: &egui::Style) -> FontId {
45    style
46        .text_styles
47        .get(&story_style())
48        .cloned()
49        .unwrap_or_else(|| TextStyle::Body.resolve(style))
50}

52/// The family a theme's font `named` is drawn in: its own once egui has it
53/// (`families`, [`crate::fonts::loaded`]), else `or`.
54fn family(named: Option<&String>, or: FontFamily, families: &BTreeSet<String>) -> FontFamily {
55    match named {
56        Some(name) if families.contains(name) => FontFamily::Name(name.as_str().into()),
57        _ => or,
58    }
59}

61/// `style`'s text styles from `kind`: the UI's font at its size for the
62/// body, the buttons and, scaled, the small and the headings; the fixed
63/// width font at its size; and the story's ([`story_style`]). A font egui
64/// does not have yet (`families`) is its own.
65pub(crate) fn typed(style: &mut egui::Style, kind: &Type, families: &BTreeSet<String>) {
66    let ui = family(kind.ui_font.as_ref(), FontFamily::Proportional, families);
67    let story = family(
68        kind.story_font.as_ref().or(kind.ui_font.as_ref()),
69        FontFamily::Proportional,
70        families,
71    );
72    let mono = family(kind.mono_font.as_ref(), FontFamily::Monospace, families);
73    let scale = kind.ui_size / 13.0;
74    style.text_styles = [
75        (TextStyle::Small, FontId::new(9.0 * scale, ui.clone())),
76        (TextStyle::Body, FontId::new(kind.ui_size, ui.clone())),
77        (TextStyle::Button, FontId::new(kind.ui_size, ui.clone())),
78        (TextStyle::Heading, FontId::new(18.0 * scale, ui)),
79        (TextStyle::Monospace, FontId::new(kind.mono_size, mono)),
80        (story_style(), FontId::new(kind.story_size, story)),
81    ]
82    .into();
83}

85/// The face a widget of kind `key` draws its words in: the theme's for the
86/// kind, with `own`, the widget's own page's, over it; `None` when neither
87/// says anything.
88pub(crate) fn face(ctx: &egui::Context, key: &str, own: Option<&Face>) -> Option<Face> {
89    let kind = outfit(ctx).kind;
90    let theirs = kind.widgets.get(key).cloned().unwrap_or_default();
91    let face = own.map_or_else(|| theirs.clone(), |own| theirs.under(own));
92    (!face.is_empty()).then_some(face)
93}

95/// `style` with `face` worn for one widget: its body, buttons and story in
96/// the face's font and size, where it sets them.
97pub(crate) fn faced(style: &mut egui::Style, face: &Face, families: &BTreeSet<String>) {
98    for text in [TextStyle::Body, TextStyle::Button, story_style()] {
99        let mut font = text.resolve(style);
100        if let Some(size) = face.size {
101            font.size = size;
102        }
103        if face.font.is_some() {
104            font.family = family(face.font.as_ref(), font.family.clone(), families);
105        }
106        style.text_styles.insert(text, font);
107    }
108}

143/// Wear `outfit`: every token answers from its palette, and egui's style
144/// is set from it -- the visuals from its surfaces and text, the corners,
145/// edges, spacing and buttons from its shape -- so every stock control
146/// follows.
147pub(crate) fn wear(ctx: &egui::Context, outfit: &Outfit) {
148    ctx.data_mut(|data| data.insert_temp(key(), outfit.clone()));
149    let style = style(ctx, outfit);
150    // egui keeps a style for dark and one for light and picks by its own
151    // theme; ours decides which, so the computer's mode changing does not
152    // swap the visuals out from under the palette.
153    let theme = if style.visuals.dark_mode {
154        egui::Theme::Dark
155    } else {
156        egui::Theme::Light
157    };
158    ctx.set_theme(theme);
159    ctx.set_style_of(theme, style);
160}
161
162/// egui's style for `outfit`: its visuals from the palette, its shape and
163/// its type, in the fonts egui has now.
164fn style(ctx: &egui::Context, outfit: &Outfit) -> egui::Style {
165    let mut style = (*ctx.global_style()).clone();
166    style.visuals = visuals(&outfit.palette);
167    shaped(&mut style, &outfit.shape);
168    typed(&mut style, &outfit.kind, &crate::fonts::loaded(ctx));
169    style
170}

172/// `outfit` worn by the window `ctx` is drawing until the [`Worn`] is
173/// dropped: the outfit answers for that window, and egui's style is the
174/// outfit's meanwhile and put back after, so a play window wears its
175/// character's own theme while the rest wear Hydra's. With no outfit,
176/// nothing changes.
177pub(crate) fn wearing(ctx: &egui::Context, outfit: Option<&Outfit>) -> Option<Worn> {
178    let outfit = outfit?;
179    let viewport = ctx.viewport_id();
180    let theme = ctx.theme();
181    let before = ctx.global_style();
182    let style = style(ctx, outfit);
183    ctx.data_mut(|data| data.insert_temp(key().with(viewport), outfit.clone()));
184    ctx.set_style_of(theme, style);
185    Some(Worn {
186        ctx: ctx.clone(),
187        viewport,
188        theme,
189        before,
190    })
191}
192
193/// An outfit worn by one window, put off when dropped.
194pub(crate) struct Worn {
195    ctx: egui::Context,
196    viewport: egui::ViewportId,
197    theme: egui::Theme,
198    before: std::sync::Arc<egui::Style>,
199}
200
201impl Drop for Worn {
202    fn drop(&mut self) {
203        self.ctx.set_style_of(self.theme, (*self.before).clone());
204        self.ctx
205            .data_mut(|data| data.remove::<Outfit>(key().with(self.viewport)));
206    }
207}

209/// `style`, from egui's own spacing, given `shape`: each kind's corners,
210/// the edges' width, the density, the scrollbar, the shadows, and how a
211/// button is drawn at rest.
212pub(crate) fn shaped(style: &mut egui::Style, shape: &Shape) {
213    let factor = shape.density.factor();
214    let plain = egui::style::Spacing::default();
215    let spacing = &mut style.spacing;
216    spacing.item_spacing = plain.item_spacing * factor;
217    spacing.button_padding = plain.button_padding * factor;
218    spacing.scroll.bar_width = shape.scrollbar;
219    let visuals = &mut style.visuals;
220    let windows = CornerRadius::same(shape.corners.windows);
221    visuals.window_corner_radius = windows;
222    visuals.menu_corner_radius = windows;
223    visuals.window_stroke.width = shape.stroke;
224    if !shape.shadows {
225        visuals.window_shadow = egui::Shadow::NONE;
226        visuals.popup_shadow = egui::Shadow::NONE;
227    }
228    let controls = CornerRadius::same(shape.corners.controls);
229    for state in [
230        &mut visuals.widgets.noninteractive,
231        &mut visuals.widgets.inactive,
232        &mut visuals.widgets.hovered,
233        &mut visuals.widgets.active,
234        &mut visuals.widgets.open,
235    ] {
236        state.corner_radius = controls;
237        state.bg_stroke.width = shape.stroke;
238    }
239    // A button's ground at rest is egui's weak fill; under the pointer it
240    // fills in every style.
241    let rest = &mut visuals.widgets.inactive;
242    match shape.buttons {
243        Buttons::Filled => {}
244        Buttons::Outlined => rest.weak_bg_fill = Color32::TRANSPARENT,
245        Buttons::Flat => {
246            rest.weak_bg_fill = Color32::TRANSPARENT;
247            rest.bg_stroke = Stroke::NONE;
248        }
249    }
250}

252/// egui's visuals from `palette`: dark or light by its canvas, the
253/// surfaces on every control, the text and lines, the selection, the
254/// link, a warning and a wrong.
255pub(crate) fn visuals(palette: &Palette) -> Visuals {
256    let of = |token| rgb(palette.get(token));
257    let (canvas, surface, raised, inset) =
258        (of(T::Canvas), of(T::Surface), of(T::Raised), of(T::Inset));
259    let (line, strong, text, muted) = (of(T::Line), of(T::LineStrong), of(T::Text), of(T::Muted));
260    let dark = readable_on(canvas) == Color32::WHITE;
261    let mut visuals = if dark {
262        Visuals::dark()
263    } else {
264        Visuals::light()
265    };
266    visuals.override_text_color = Some(text);
267    visuals.weak_text_color = Some(muted);
268    visuals.window_fill = canvas;
269    visuals.panel_fill = canvas;
270    visuals.faint_bg_color = surface;
271    visuals.extreme_bg_color = inset;
272    visuals.code_bg_color = inset;
273    visuals.window_stroke = Stroke::new(1.0, strong);
274    visuals.hyperlink_color = of(T::Link);
275    visuals.warn_fg_color = of(T::Warning);
276    visuals.error_fg_color = of(T::Wrong);
277    visuals.selection.bg_fill = of(T::Selection);
278    visuals.selection.stroke = Stroke::new(1.0, text);
279    let widgets = &mut visuals.widgets;
280    for (state, fill, edge) in [
281        (&mut widgets.noninteractive, canvas, line),
282        (&mut widgets.inactive, surface, line),
283        (&mut widgets.hovered, raised, strong),
284        (&mut widgets.active, raised, strong),
285        (&mut widgets.open, surface, line),
286    ] {
287        state.bg_fill = fill;
288        state.weak_bg_fill = fill;
289        state.bg_stroke = Stroke::new(state.bg_stroke.width.max(1.0), edge);
290        state.fg_stroke = Stroke::new(state.fg_stroke.width, text);
291    }
292    widgets.noninteractive.fg_stroke.color = muted;
293    visuals
294}

296/// A palette's colour as egui's.
297pub(crate) fn rgb([red, green, blue]: Rgb) -> Color32 {
298    Color32::from_rgb(red, green, blue)
299}

301/// Black or white, whichever reads on `behind`: by its luminance, the ITU
302/// weights.
303pub(crate) fn readable_on(behind: Color32) -> Color32 {
304    let [r, g, b, _] = behind.to_array();
305    let luminance = 0.299 * f32::from(r) + 0.587 * f32::from(g) + 0.114 * f32::from(b);
306    if luminance > 140.0 {
307        Color32::BLACK
308    } else {
309        Color32::WHITE
310    }
311}

1//! Pictures the window draws from the player's files: a bar's images, a
2//! doll's picture and its overlays, a hotbar's icons.
3//!
4//! Read and decoded off the window's thread (`crate::disk`): a picture is
5//! drawn from what was last read of it, and the first time it is asked for
6//! it is not drawn until it has been. A picture the player changes is read
7//! again, and one that failed to load -- not there yet, or half copied --
8//! is tried again: its file's modification time is looked at **at most
9//! once a second**, and the picture read again only when that changed.
10

47/// A picture's opacity, kept a quarter its size each way: each kept value
48/// the most solid of the pixels it stands for, so a thin line is not lost.
49#[derive(Debug, PartialEq, Eq)]
50pub(crate) struct Solid {
51    width: usize,
52    height: usize,
53    alpha: Vec<u8>,
54}
55
56/// How many of a picture's pixels, each way, one value of a [`Solid`]
57/// stands for.
58const SOLID_STEP: u32 = 4;
59
60impl Solid {
61    /// The opacity of the picture at `path`; none when it cannot be read.
62    fn read(path: &Path) -> Option<Self> {
63        let image = image::open(path).ok()?.to_rgba8();
64        Some(Self::of(&image))
65    }
66
67    /// `image`'s opacity, kept a quarter its size.
68    pub(crate) fn of(image: &image::RgbaImage) -> Self {
69        let width = image.width().div_ceil(SOLID_STEP);
70        let height = image.height().div_ceil(SOLID_STEP);
71        let mut alpha = vec![0u8; usize::try_from(width * height).unwrap_or(0)];
72        for (x, y, pixel) in image.enumerate_pixels() {
73            let at = usize::try_from((y / SOLID_STEP) * width + x / SOLID_STEP).unwrap_or(0);
74            if let Some(kept) = alpha.get_mut(at) {
75                *kept = (*kept).max(pixel.0[3]);
76            }
77        }
78        Self {
79            width: usize::try_from(width).unwrap_or(0),
80            height: usize::try_from(height).unwrap_or(0),
81            alpha,
82        }
83    }
84
85    /// How solid it is at `(across, down)`, each a fraction of its width
86    /// and height: 0 outside it.
87    pub(crate) fn at(&self, (across, down): (f32, f32)) -> u8 {
88        if !(0.0..1.0).contains(&across) || !(0.0..1.0).contains(&down) {
89            return 0;
90        }
91        #[expect(
92            clippy::cast_possible_truncation,
93            clippy::cast_sign_loss,
94            clippy::cast_precision_loss,
95            reason = "a fraction of a few hundred, checked in range above"
96        )]
97        let (x, y) = (
98            (across * self.width as f32) as usize,
99            (down * self.height as f32) as usize,
100        );
101        self.alpha.get(y * self.width + x).copied().unwrap_or(0)
102    }
103
104    /// The part of it at least `least` solid, as fractions of its width and
105    /// height; none when no part is.
106    pub(crate) fn bounds(&self, least: u8) -> Option<egui::Rect> {
107        let mut found: Option<(usize, usize, usize, usize)> = None;
108        for (at, alpha) in self.alpha.iter().enumerate() {
109            if *alpha < least {
110                continue;
111            }
112            let (x, y) = (at % self.width.max(1), at / self.width.max(1));
113            found = Some(found.map_or((x, y, x, y), |(left, top, right, bottom)| {
114                (left.min(x), top.min(y), right.max(x), bottom.max(y))
115            }));
116        }
117        let (left, top, right, bottom) = found?;
118        #[expect(
119            clippy::cast_precision_loss,
120            reason = "a picture is a few hundred cells"
121        )]
122        let fraction = |cell: usize, of: usize| cell as f32 / of.max(1) as f32;
123        Some(egui::Rect::from_min_max(
124            egui::pos2(fraction(left, self.width), fraction(top, self.height)),
125            egui::pos2(
126                fraction(right + 1, self.width),
127                fraction(bottom + 1, self.height),
128            ),
129        ))
130    }
131}

133/// [`picture`], at `now`.
134fn picture_at(context: &egui::Context, path: &str, now: Instant) -> Option<egui::TextureHandle> {
135    let owned = context.clone();
136    kept_at(context, "picture", path, now, move |path| {
137        read(&owned, path)
138    })

141/// What `read` makes of the file at `path`, kept by egui under `kind` and
142/// read again only when the file's modification time has changed, looked
143/// at no oftener than [`LOOK_AGAIN`].
144fn kept_at<T: Clone + Send + Sync + 'static>(
145    context: &egui::Context,
146    kind: &'static str,
147    path: &str,
148    now: Instant,
149    read: impl FnOnce(&str) -> Option<T> + Send + 'static,
150) -> Option<T> {
151    kept_watching_at(context, (kind, path), path, now, read)
152}
153
154/// What `read` makes of `watched`, kept by egui under `kind` and `key` and
155/// read again when `watched`'s modification time has changed: a folder's,
156/// for what is in it (`doll_art.rs`).
157pub(crate) fn kept_watching<T: Clone + Send + Sync + 'static>(
158    context: &egui::Context,
159    kind: &'static str,
160    key: &str,
161    watched: &str,
162    read: impl FnOnce(&str) -> Option<T> + Send + 'static,
163) -> Option<T> {
164    kept_watching_at(context, (kind, key), watched, Instant::now(), read)
165}
166
167fn kept_watching_at<T: Clone + Send + Sync + 'static>(
168    context: &egui::Context,
169    (kind, key): (&'static str, &str),
170    path: &str,
171    now: Instant,
172    read: impl FnOnce(&str) -> Option<T> + Send + 'static,
173) -> Option<T> {
174    crate::disk::Disk::hydras()
175        .kept_at(
176            &format!("{kind}:{key}"),
177            Path::new(path),
178            (now, Some(&crate::repaint(context))),
179            move |path| read(&path.to_string_lossy()),
180        )
181        .map(|(_, made)| made)
182}

191/// A PNG, or any picture the `image` crate reads, as a texture.
192fn read(context: &egui::Context, path: &str) -> Option<egui::TextureHandle> {
193    read_as(context, path, false)
194}

196/// The picture at `path` as a texture, `grey` or in its colours.
197fn read_as(context: &egui::Context, path: &str, grey: bool) -> Option<egui::TextureHandle> {
198    let bytes = std::fs::read(Path::new(path)).ok()?;
199    let mut image = image::load_from_memory(&bytes).ok()?;
200    // **No larger than a texture can be**: egui asserts it and the GPU refuses it, either way on the
201    // window thread, which every character's session shares.
202    let most = u32::try_from(context.input(|input| input.max_texture_side)).unwrap_or(u32::MAX);
203    if image.width() > most || image.height() > most {
204        image = image.resize(most, most, image::imageops::FilterType::Triangle);
205    }
206    let mut image = image.to_rgba8();
207    if grey {
208        for pixel in image.pixels_mut() {
209            let [red, green, blue, alpha] = pixel.0;
210            let luma =
211                (u32::from(red) * 299 + u32::from(green) * 587 + u32::from(blue) * 114) / 1000;
212            let luma = u8::try_from(luma).unwrap_or(u8::MAX);
213            pixel.0 = [luma, luma, luma, alpha];
214        }
215    }
216    let size = [
217        usize::try_from(image.width()).ok()?,
218        usize::try_from(image.height()).ok()?,
219    ];
220    let pixels = egui::ColorImage::from_rgba_unmultiplied(size, image.as_raw());
221    let name = if grey {
222        format!("{path}#grey")
223    } else {
224        path.to_owned()
225    };
226    Some(context.load_texture(name, pixels, egui::TextureOptions::LINEAR))
227}

229#[cfg(test)]
230mod tests {
231    use super::{LOOK_AGAIN, picture_at};
232    use std::time::Duration;
233    use std::time::{Instant, SystemTime};

235    /// A `width` by 1 PNG at `path`, stamped `modified`.
236    fn png(path: &std::path::Path, width: u32, modified: SystemTime) -> Option<()> {
237        image::RgbaImage::new(width, 1).save(path).ok()?;
238        std::fs::File::options()
239            .write(true)
240            .open(path)
241            .ok()?
242            .set_modified(modified)
243            .ok()
244    }
245
246    fn folder(name: &str) -> std::path::PathBuf {
247        let dir =
248            std::env::temp_dir().join(format!("hydra-pictures-{}-{name}", std::process::id()));
249        let _ = std::fs::remove_dir_all(&dir);
250        let _ = std::fs::create_dir_all(&dir);
251        dir
252    }
253
254    #[test]
255    fn a_changed_picture_is_read_again_and_an_unchanged_one_kept() {
256        let dir = folder("changed");
257        let path = dir.join("a.png");
258        let then = SystemTime::UNIX_EPOCH + Duration::from_secs(1_000_000);
259        png(&path, 2, then).unwrap();
260        let (context, start) = (egui::Context::default(), Instant::now());
261        let at = path.to_string_lossy().into_owned();
262
263        let first = picture_at(&context, &at, start).unwrap();
264        let later = start + LOOK_AGAIN * 2;
265        let kept = picture_at(&context, &at, later).unwrap();
266        assert_eq!(kept.id(), first.id(), "unchanged: the same texture");
267
268        png(&path, 5, then + Duration::from_mins(1)).unwrap();
269        let soon = picture_at(&context, &at, later + Duration::from_millis(10)).unwrap();
270        assert_eq!(soon.size()[0], 2, "not looked at again within the second");
271        let again = picture_at(&context, &at, later + LOOK_AGAIN * 2).unwrap();
272        assert_eq!(again.size()[0], 5, "changed: read again");
273        let _ = std::fs::remove_dir_all(&dir);
274    }

276    /// A picture wider than the largest texture the window can hold would
277    /// kill the window thread, and with it every character; its path is
278    /// saved, so again at every start. It is shrunk to fit.
279    #[test]
280    fn a_picture_too_big_for_a_texture_is_shrunk_to_fit() {
281        let dir = folder("big");
282        let path = dir.join("big.png");
283        let context = egui::Context::default();
284        let most = context.input(|input| input.max_texture_side);
285        let wide = u32::try_from(most).unwrap() + 1000;
286        png(&path, wide, SystemTime::now()).unwrap();
287        let at = path.to_string_lossy().into_owned();
288        let read = picture_at(&context, &at, Instant::now()).expect("read");
289        assert!(read.size()[0] <= most, "{:?} within {most}", read.size());
290        let _ = std::fs::remove_dir_all(&dir);
291    }
292
293    #[test]
294    fn a_picture_that_was_not_there_is_read_once_it_is() {
295        let dir = folder("missing");
296        let path = dir.join("late.png");
297        let (context, start) = (egui::Context::default(), Instant::now());
298        let at = path.to_string_lossy().into_owned();
299        assert!(picture_at(&context, &at, start).is_none());
300        png(&path, 3, SystemTime::now()).unwrap();
301        let read = picture_at(&context, &at, start + LOOK_AGAIN * 2);
302        assert_eq!(read.map(|t| t.size()[0]), Some(3));
303        let _ = std::fs::remove_dir_all(&dir);
304    }

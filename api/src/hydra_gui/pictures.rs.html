1//! Pictures the window draws from the player's files: a bar's images, a
2//! doll's picture and its overlays, a hotbar's icons.
3//!
4//! Read and decoded off the window's thread (`crate::disk`): a picture is
5//! drawn from what was last read of it, and the first time it is asked for
6//! it is not drawn until it has been. A picture the player changes is read
7//! again, and one that failed to load -- not there yet, or half copied --
8//! is tried again: its file's modification time is looked at **at most
9//! once a second**, and the picture read again only when that changed.
10

17/// The picture at `path` as a texture, or `None` when it cannot be read.
18pub(crate) fn picture(context: &egui::Context, path: &str) -> Option<egui::TextureHandle> {
19    picture_at(context, path, Instant::now())
20}

22/// [`picture`], grey: a hotbar's icon drawn so, or dimmed.
23pub(crate) fn grey_picture(context: &egui::Context, path: &str) -> Option<egui::TextureHandle> {
24    let owned = context.clone();
25    kept_at(context, "grey-picture", path, Instant::now(), move |path| {
26        read_as(&owned, path, true)
27    })
28}

39/// [`picture`], at `now`.
40fn picture_at(context: &egui::Context, path: &str, now: Instant) -> Option<egui::TextureHandle> {
41    let owned = context.clone();
42    kept_at(context, "picture", path, now, move |path| {
43        read(&owned, path)
44    })
45}

47/// What `read` makes of the file at `path`, kept by egui under `kind` and
48/// read again only when the file's modification time has changed, looked
49/// at no oftener than [`LOOK_AGAIN`].
50fn kept_at<T: Clone + Send + Sync + 'static>(
51    context: &egui::Context,
52    kind: &'static str,
53    path: &str,
54    now: Instant,
55    read: impl FnOnce(&str) -> Option<T> + Send + 'static,
56) -> Option<T> {
57    kept_watching_at(context, (kind, path), path, now, read)
58}
59
60/// What `read` makes of `watched`, kept by egui under `kind` and `key` and
61/// read again when `watched`'s modification time has changed: a folder's,
62/// for what is in it (`doll_art.rs`).
63pub(crate) fn kept_watching<T: Clone + Send + Sync + 'static>(
64    context: &egui::Context,
65    kind: &'static str,
66    key: &str,
67    watched: &str,
68    read: impl FnOnce(&str) -> Option<T> + Send + 'static,
69) -> Option<T> {
70    kept_watching_at(context, (kind, key), watched, Instant::now(), read)
71}
72
73fn kept_watching_at<T: Clone + Send + Sync + 'static>(
74    context: &egui::Context,
75    (kind, key): (&'static str, &str),
76    path: &str,
77    now: Instant,
78    read: impl FnOnce(&str) -> Option<T> + Send + 'static,
79) -> Option<T> {
80    crate::disk::Disk::hydras()
81        .kept_at(
82            &format!("{kind}:{key}"),
83            Path::new(path),
84            (now, Some(&crate::repaint(context))),
85            move |path| read(&path.to_string_lossy()),
86        )
87        .map(|(_, made)| made)
88}

97/// A PNG, or any picture the `image` crate reads, as a texture.
98fn read(context: &egui::Context, path: &str) -> Option<egui::TextureHandle> {
99    read_as(context, path, false)
100}

102/// The picture at `path` as a texture, `grey` or in its colours.
103fn read_as(context: &egui::Context, path: &str, grey: bool) -> Option<egui::TextureHandle> {
104    let bytes = std::fs::read(Path::new(path)).ok()?;
105    let mut image = image::load_from_memory(&bytes).ok()?;
106    // **No larger than a texture can be**: egui asserts it and the GPU refuses it, either way on the
107    // window thread, which every character's session shares.
108    let most = u32::try_from(context.input(|input| input.max_texture_side)).unwrap_or(u32::MAX);
109    if image.width() > most || image.height() > most {
110        image = image.resize(most, most, image::imageops::FilterType::Triangle);
111    }
112    let mut image = image.to_rgba8();
113    if grey {
114        for pixel in image.pixels_mut() {
115            let [red, green, blue, alpha] = pixel.0;
116            let luma =
117                (u32::from(red) * 299 + u32::from(green) * 587 + u32::from(blue) * 114) / 1000;
118            let luma = u8::try_from(luma).unwrap_or(u8::MAX);
119            pixel.0 = [luma, luma, luma, alpha];
120        }
121    }
122    let size = [
123        usize::try_from(image.width()).ok()?,
124        usize::try_from(image.height()).ok()?,
125    ];
126    let pixels = egui::ColorImage::from_rgba_unmultiplied(size, image.as_raw());
127    let name = if grey {
128        format!("{path}#grey")
129    } else {
130        path.to_owned()
131    };
132    Some(context.load_texture(name, pixels, egui::TextureOptions::LINEAR))
133}

135#[cfg(test)]
136mod tests {
137    use super::{LOOK_AGAIN, picture_at};
138    use std::time::Duration;
139    use std::time::{Instant, SystemTime};
140
141    /// A `width` by 1 PNG at `path`, stamped `modified`.
142    fn png(path: &std::path::Path, width: u32, modified: SystemTime) -> Option<()> {
143        image::RgbaImage::new(width, 1).save(path).ok()?;
144        std::fs::File::options()
145            .write(true)
146            .open(path)
147            .ok()?
148            .set_modified(modified)
149            .ok()
150    }
151
152    fn folder(name: &str) -> std::path::PathBuf {
153        let dir =
154            std::env::temp_dir().join(format!("hydra-pictures-{}-{name}", std::process::id()));
155        let _ = std::fs::remove_dir_all(&dir);
156        let _ = std::fs::create_dir_all(&dir);
157        dir
158    }
159
160    #[test]
161    fn a_changed_picture_is_read_again_and_an_unchanged_one_kept() {
162        let dir = folder("changed");
163        let path = dir.join("a.png");
164        let then = SystemTime::UNIX_EPOCH + Duration::from_secs(1_000_000);
165        png(&path, 2, then).unwrap();
166        let (context, start) = (egui::Context::default(), Instant::now());
167        let at = path.to_string_lossy().into_owned();
168
169        let first = picture_at(&context, &at, start).unwrap();
170        let later = start + LOOK_AGAIN * 2;
171        let kept = picture_at(&context, &at, later).unwrap();
172        assert_eq!(kept.id(), first.id(), "unchanged: the same texture");
173
174        png(&path, 5, then + Duration::from_mins(1)).unwrap();
175        let soon = picture_at(&context, &at, later + Duration::from_millis(10)).unwrap();
176        assert_eq!(soon.size()[0], 2, "not looked at again within the second");
177        let again = picture_at(&context, &at, later + LOOK_AGAIN * 2).unwrap();
178        assert_eq!(again.size()[0], 5, "changed: read again");
179        let _ = std::fs::remove_dir_all(&dir);
180    }
181
182    /// A picture wider than the largest texture the window can hold would
183    /// kill the window thread, and with it every character; its path is
184    /// saved, so again at every start. It is shrunk to fit.
185    #[test]
186    fn a_picture_too_big_for_a_texture_is_shrunk_to_fit() {
187        let dir = folder("big");
188        let path = dir.join("big.png");
189        let context = egui::Context::default();
190        let most = context.input(|input| input.max_texture_side);
191        let wide = u32::try_from(most).unwrap() + 1000;
192        png(&path, wide, SystemTime::now()).unwrap();
193        let at = path.to_string_lossy().into_owned();
194        let read = picture_at(&context, &at, Instant::now()).expect("read");
195        assert!(read.size()[0] <= most, "{:?} within {most}", read.size());
196        let _ = std::fs::remove_dir_all(&dir);
197    }
198
199    #[test]
200    fn a_picture_that_was_not_there_is_read_once_it_is() {
201        let dir = folder("missing");
202        let path = dir.join("late.png");
203        let (context, start) = (egui::Context::default(), Instant::now());
204        let at = path.to_string_lossy().into_owned();
205        assert!(picture_at(&context, &at, start).is_none());
206        png(&path, 3, SystemTime::now()).unwrap();
207        let read = picture_at(&context, &at, start + LOOK_AGAIN * 2);
208        assert_eq!(read.map(|t| t.size()[0]), Some(3));
209        let _ = std::fs::remove_dir_all(&dir);
210    }

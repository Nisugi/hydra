1//! Keybinds as the window takes them: a key, and the macro it does on the
2//! character whose play window has the keyboard. The keybinds themselves --
3//! the files, Hydra's defaults, the macro sets, the writer -- are
4//! `hydra_board::keys`'s; here, each press is named and taken.
5//!
6//! **A key is named by its winit code** -- `Numpad8`, `F13`, `KeyA`,
7//! `ArrowUp` -- the physical key, not the character it types, so a numpad
8//! key is never its digit: egui folds `Numpad8` into `8`, which is why the
9//! numpad comes through Hydra's eframe fork instead
10//! (`eframe::Frame::numpad_keys`), with `NumLock` read from each press.
11//!
12//! The keys come three ways. The numpad's through the fork's numpad channel,
13//! which catches only the numpad keys that are bound, so an unbound one
14//! still types. The keys egui has no name for -- Pause, Scroll Lock, Print
15//! Screen, Caps Lock, the context-menu key -- through the fork's key capture
16//! (`eframe::Frame::set_key_capture`), which catches only those
17//! bound, so an unbound Caps Lock is left to egui. Every other key through
18//! egui, which names F1-F35, the arrows and the rest; a bound key's press is
19//! taken before anything else sees it. A key that types -- a letter, a digit,
20//! a mark, with no Ctrl, Alt or Cmd -- never fires a binding, or nobody could
21//! type it.
22//!
23//! **`NumLock` is nobody's.** On Windows and Linux it is the operating
24//! system's switch for the numpad; on macOS, which has none, winit calls the
25//! Clear key `NumLock`, and Clear is Hydra's switch there, the numpad typing
26//! or sending its keys (`app.rs`, `numpad_mode`).

45/// A chord as the window names a press: its key, and egui's modifiers.
46pub(crate) trait ChordOf {
47    /// `key` held with `modifiers`.
48    fn of(key: &str, modifiers: Modifiers) -> Self;
49}
50
51impl ChordOf for Chord {
52    fn of(key: &str, modifiers: Modifiers) -> Self {
53        let bit = |on: bool, bit: u8| if on { bit } else { 0 };
54        Chord::new(
55            key,
56            bit(modifiers.ctrl, CTRL)
57                | bit(modifiers.shift, SHIFT)
58                | bit(modifiers.alt, ALT)
59                | bit(modifiers.mac_cmd, CMD),
60        )
61    }
62}

64/// The keys in effect, as egui and the fork take them.
65pub(crate) trait EguiKeys {
66    /// The keys egui has no name for that are bound, whatever the
67    /// modifiers: what the fork is told to catch (`eframe::Frame::
68    /// set_key_capture`), so one unbound is left to egui.
69    fn key_capture(self) -> HashSet<winit::keyboard::KeyCode>;
70
71    /// Take from `input` every key press this binds, and the macros they
72    /// do, in order: taken, so no widget sees a bound key. A key whose macro
73    /// `leaves` is not taken, and does nothing here; what it typed goes too.
74    /// A held key's repeats do a send again ([`repeats`]); any other macro
75    /// once a press, its repeats taken and done nothing with.
76    fn take(self, input: &mut egui::InputState, leaves: impl Fn(&Macro) -> bool) -> Vec<Macro>;
77}
78
79impl EguiKeys for Keys<'_> {
80    fn key_capture(self) -> HashSet<winit::keyboard::KeyCode> {
81        self.bound()
82            .filter_map(|(chord, _)| CAPTURED.iter().find(|(name, _)| *name == chord.key))
83            .map(|(_, code)| *code)
84            .collect()
85    }
86
87    fn take(self, input: &mut egui::InputState, leaves: impl Fn(&Macro) -> bool) -> Vec<Macro> {
88        let (mut lines, mut typed) = (Vec::new(), false);
89        input.events.retain(|event| {
90            let egui::Event::Key {
91                key,
92                physical_key,
93                pressed: true,
94                repeat,
95                modifiers,
96            } = event
97            else {
98                return true;
99            };
100            let Some(name) = winit_name(physical_key.unwrap_or(*key)) else {
101                return true;
102            };
103            match self.does(&Chord::of(&name, *modifiers)) {
104                Some(made) if !leaves(made) => {
105                    if !*repeat || repeats(made) {
106                        lines.push(made.clone());
107                    }
108                    typed |= types(&name);
109                    false
110                }
111                _ => true,
112            }
113        });
114        if typed {
115            input
116                .events
117                .retain(|event| !matches!(event, egui::Event::Text(_)));
118        }
119        lines
120    }
121}

1//! The hub's cards: every character's card drawn from one design, which the
2//! player arranges as a custom window's inside is arranged.
3//!
4//! A design is a custom window's inside (`layout::Custom`): widgets in
5//! cells, each a free rect, the inside's size the card's. Any widget of the
6//! catalog draws on a card from the character's seat -- its snapshot and
7//! story -- and the card's own parts ([`CardPart`]) are the hub's: the name
8//! and how it is connected, together or each alone, where it is and who
9//! with, and its buttons. A
10//! Field shows any one value of the game state (`widget/field.rs`), every
11//! field of a card read from one print of its character's state. Each line
12//! of a card is drawn in a row, so they all sit at one height. A
13//! widget that needs a play window's own -- its options, its hotbars, its
14//! map -- draws as it does with none. A bar on a card says its label and
15//! percent, not its numbers: a card is small.
16//!
17//! *Cards* on the hub's Menu opens the editor: the catalog to add from,
18//! a field picked from the game state among it, the card's size, and the card itself, drawn for a character chosen and
19//! arranged as a custom window is (`play::arrange`): a cell dragged moves,
20//! an edge dragged sizes it, and one let go off the card is taken away.
21//! The design is kept in the data folder's [`FILE`] as it changes; with no
22//! file, or one that does not read, the card is Hydra's own ([`hydras`]).
23

95/// Hydra's own card: the name and how it is connected; the four vitals;
96/// where it is and who with; its buttons.
97pub(crate) fn hydras() -> Custom {
98    let rows = [
99        vec![Widget::Card(CardPart::Name)],
100        vec![
101            Widget::Health,
102            Widget::Mana,
103            Widget::Stamina,
104            Widget::Spirit,
105        ],
106        vec![Widget::Card(CardPart::Whereabouts)],
107        vec![Widget::Card(CardPart::Buttons)],
108    ];
109    let mut id = 0;
110    let rows = rows
111        .into_iter()
112        .map(|row| {
113            row.into_iter()
114                .map(|widget| {
115                    id += 1;
116                    Placed { id, widget }
117                })
118                .collect()
119        })
120        .collect();
121    Custom::rows("Card", rows, Vec2::new(312.0, 84.0))
122}

124impl Cards {
125    /// The design kept in `data`, the data folder; Hydra's own card with
126    /// none, or one that does not read.
127    #[must_use]
128    pub(crate) fn load(data: &Path) -> Self {
129        let file = data.join(FILE);
130        let design = std::fs::read_to_string(&file)
131            .ok()
132            .and_then(|text| serde_json::from_str::<Kept>(&text).ok())
133            .map_or_else(hydras, |kept| kept.card);
134        Self {
135            design,
136            file: Some(file),
137            ..Self::default()
138        }
139    }
140
141    /// Write the design, with the next batch of saves.
142    fn save(&mut self) {
143        let Some(file) = &self.file else { return };
144        let Some(data) = file.parent() else { return };
145        let kept = Kept {
146            version: VERSION,
147            card: self.design.clone(),
148        };
149        if let Ok(text) = serde_json::to_string_pretty(&kept) {
150            self.saves.write(data, file.clone(), text.into_bytes());
151        }
152    }

154    /// The editor's inside, over the characters `seats`, the first of them
155    /// unless another is chosen: the catalog, the card's size, and the card,
156    /// arranged. Kept as it changes.
157    pub(crate) fn edit(&mut self, ui: &mut egui::Ui, seats: &[Arc<Seat>]) {
158        let before = self.design.clone();
159        let shown = seats
160            .iter()
161            .find(|seat| Some(seat.id) == self.editor.shows)
162            .or_else(|| seats.first())
163            .map(AsRef::as_ref);
164        egui::Panel::left(Id::new("cards-catalog"))
165            .resizable(true)
166            .default_size(260.0)
167            .show(ui, |ui| self.catalog(ui, shown));
168        egui::CentralPanel::default().show(ui, |ui| {
169            ui.horizontal(|ui| {
170                ui.label("Drawn for");
171                egui::ComboBox::from_id_salt("cards-shows")
172                    .selected_text(shown.map_or("nobody playing", |seat| seat.name.as_str()))
173                    .show_ui(ui, |ui| {
174                        for seat in seats {
175                            ui.selectable_value(
176                                &mut self.editor.shows,
177                                Some(seat.id),
178                                seat.name.as_str(),
179                            );
180                        }
181                    });
182                ui.separator();
183                let mut size = self.design.inside();
184                ui.label("Width");
185                ui.add(egui::DragValue::new(&mut size.x).range(SMALLEST.x..=LARGEST.x));
186                ui.label("Height");
187                ui.add(egui::DragValue::new(&mut size.y).range(SMALLEST.y..=LARGEST.y));
188                if size != self.design.inside() {
189                    self.design.keep(size);
190                }
191                ui.separator();
192                if ui
193                    .button("Hydra's card")
194                    .on_hover_text("Start again from the card Hydra comes with")
195                    .clicked()
196                {
197                    self.design = hydras();
198                }
199            });
200            ui.weak(
201                "Every character's card is drawn from this one. Drag a part to move it, \
202                 an edge to size it; let it go off the card to take it away.",
203            );
204            ui.add_space(8.0);
205            egui::ScrollArea::both()
206                .id_salt("cards-card")
207                .auto_shrink(false)
208                .show(ui, |ui| self.arranged(ui, shown));
209        });
210        if self.design != before {
211            self.save();
212        }
213    }

215    /// The card in the editor, drawn for `shown`, arranged.
216    fn arranged(&mut self, ui: &mut egui::Ui, shown: Option<&Seat>) {
217        egui::Frame::group(ui.style()).show(ui, |ui| {
218            let (inside, _) = ui.allocate_exact_size(self.design.inside(), egui::Sense::hover());
219            if let Some(seat) = shown {
220                // Drawn to look at: nothing on it acts while it is arranged.
221                cells(ui, seat, &self.design, inside, false);
222            }
223            let released = crate::play::arrange::cells(
224                ui,
225                &mut self.design,
226                0,
227                inside,
228                GRID,
229                &mut self.editor.gesture,
230            );
231            // Let go off the card: taken away.
232            if let Some(released) = released {
233                match released.taking {
234                    Taking::Tab(placed) => drop(self.design.take(placed)),
235                    Taking::Cell(placed) => drop(self.design.take_cell(placed)),
236                }
237            }
238        });
239    }

241    /// The catalog to add from: the card's own parts, then every widget a
242    /// play window's list offers but the command line, by its groups, then
243    /// any one value of the game state, picked from `shown`'s.
244    fn catalog(&mut self, ui: &mut egui::Ui, shown: Option<&Seat>) {
245        egui::ScrollArea::vertical()
246            .id_salt("cards-catalog-list")
247            .show(ui, |ui| {
248                ui.strong("Add to the card");
249                let mut chosen = None;
250                egui::CollapsingHeader::new(Group::Card.name())
251                    .default_open(true)
252                    .show(ui, |ui| {
253                        for part in CardPart::ALL {
254                            let widget = Widget::Card(part);
255                            if ui.button(widget.name()).clicked() {
256                                chosen = Some(widget);
257                            }
258                        }
259                    });
260                let every = Widget::all();
261                for group in Group::ALL {
262                    let kinds: Vec<&Widget> = every
263                        .iter()
264                        .filter(|kind| {
265                            kind.group() == group && !kind.is_hidden() && **kind != Widget::Input
266                        })
267                        .collect();
268                    if kinds.is_empty() {
269                        continue;
270                    }
271                    egui::CollapsingHeader::new(group.name()).show(ui, |ui| {
272                        for kind in kinds {
273                            if ui.button(kind.name()).clicked() {
274                                chosen = Some(kind.clone());
275                            }
276                        }
277                    });
278                }
279                egui::CollapsingHeader::new("A value from the game state").show(ui, |ui| {
280                    let snapshot = shown.and_then(Seat::snapshot);
281                    let state = snapshot.as_deref().map(|snapshot| &snapshot.state);
282                    let key = shown.map_or(Id::new("card-state"), printed);
283                    if let Some(path) = crate::widget::field::pick(ui, state, key) {
284                        chosen = Some(Widget::Field(path));
285                    }
286                });
287                if let Some(widget) = chosen {
288                    self.add(widget);
289                }
290            });
291    }

293    /// `widget` put on the card below everything on it, the card grown to
294    /// hold it.
295    fn add(&mut self, widget: Widget) {
296        let design = &mut self.design;
297        let id = design
298            .cells
299            .iter()
300            .flat_map(|cell| &cell.tabs)
301            .map(|placed| placed.id)
302            .max()
303            .unwrap_or(0)
304            + 1;
305        let bottom = design
306            .cells
307            .iter()
308            .map(|cell| cell.rect().max.y)
309            .fold(0.0, f32::max);
310        let inside = design.inside();
311        let asks = widget.size();
312        let grown = Vec2::new(
313            inside.x.max(asks.x.min(LARGEST.x)),
314            inside.y.max(bottom + asks.y).min(LARGEST.y),
315        );
316        design.keep(grown);
317        design.land(vec![Placed { id, widget }], 0, pos2(0.0, bottom));
318    }

321/// `seat`'s card drawn from `design`, with how quick its window is when
322/// `perf`: what the player asked of the hub, if anything.
323pub(crate) fn card(
324    ui: &mut egui::Ui,
325    seat: &Seat,
326    design: &Custom,
327    perf: bool,
328) -> Option<HubAction> {
329    let (frame, framed) = surround(ui);
330    let shown = frame.show(ui, |ui| {
331        let (inside, _) = ui.allocate_exact_size(design.inside(), egui::Sense::hover());
332        let asked = cells(ui, seat, design, inside, true);
333        if perf {
334            let (perf, lichs_text) = {
335                let story = seat.story();
336                let lichs_text =
337                    story.lich_running() && story.view() == hydra_session::lich::View::Lich;
338                (story.perf.clone(), lichs_text)
339            };
340            crate::hub::stages(ui, seat.id, &perf, lichs_text);
341        }
342        asked
343    });
344    if let Some(framed) = framed {
345        let bound = shown.response.rect.shrink(crate::frames::inset(&framed));
346        crate::frames::paint(ui.ctx(), ui.painter(), &framed, &[bound]);
347    }
348    shown.inner
349}

351/// What a card is drawn in: the theme's frame round it, with the room it
352/// needs, when the theme names one; egui's group frame otherwise.
353pub(crate) fn surround(
354    ui: &egui::Ui,
355) -> (egui::Frame, Option<std::sync::Arc<hydra_ui::frame::Frame>>) {
356    match crate::frames::worn(ui.ctx()) {
357        Some(framed) => {
358            let margin = crate::frames::margin(&framed);
359            (egui::Frame::new().inner_margin(margin), Some(framed))
360        }
361        None => (egui::Frame::group(ui.style()), None),
362    }
363}

365/// `design`'s cells drawn for `seat` at `inside`, each kept to its own rect.
366/// What the player asked of the hub, if anything; nothing, and nothing sent,
367/// unless the card `acts`.
368fn cells(
369    ui: &mut egui::Ui,
370    seat: &Seat,
371    design: &Custom,
372    inside: Rect,
373    acts: bool,
374) -> Option<HubAction> {
375    let snapshot = seat.snapshot();
376    let mut asked = None;
377    for cell in &design.cells {
378        let Some(placed) = cell.shown() else { continue };
379        let at = cell
380            .rect()
381            .translate(inside.min.to_vec2())
382            .intersect(inside);
383        if at.width() < 1.0 || at.height() < 1.0 {
384            continue;
385        }
386        let layout = egui::Layout::top_down(egui::Align::Min);
387        let mut drawn = ui.new_child(egui::UiBuilder::new().max_rect(at).layout(layout));
388        drawn.set_clip_rect(at.intersect(ui.clip_rect()));
389        let id = Id::new(("card", seat.id, placed.id));
390        match &placed.widget {
391            Widget::Field(path) => {
392                let state = snapshot.as_deref().map(|snapshot| &snapshot.state);
393                crate::widget::field::draw(&mut drawn, state, path, printed(seat));
394            }
395            Widget::Card(part) => {
396                let done = own(&mut drawn, seat, *part, snapshot.as_deref());
397                if acts {
398                    asked = asked.or(done);
399                }
400            }
401            widget => {
402                let line = drawn_widget(&mut drawn, seat, widget, snapshot.as_deref(), id);
403                if let (true, Some(line)) = (acts, line) {
404                    seat.send(&line);
405                }
406            }
407        }
408    }
409    asked
410}

418/// One of the catalog's widgets drawn from `seat`: a line it asked sent,
419/// if one was.
420fn drawn_widget(
421    ui: &mut egui::Ui,
422    seat: &Seat,
423    widget: &Widget,
424    snapshot: Option<&Snapshot>,
425    id: Id,
426) -> Option<String> {
427    let trainer = (*widget == Widget::Trainer).then(|| seat.trainer());
428    let story = seat.story();
429    let seen = Seen {
430        snapshot,
431        story: &story,
432        who: None,
433        open: &[],
434        hydra: false,
435        tags: None,
436        minimap: None,
437        travel: None,
438        trainer: trainer.as_ref(),
439    };
440    // A bar says its label and percent: a card is small.
441    let look = widget.bar_look().map(|mut look| {
442        look.says.numbers = false;
443        look
444    });
445    let chosen = Chosen {
446        look,
447        ..Chosen::default()
448    };
449    match widget.draw_with(ui, &seen, id, &chosen) {
450        Some(Clicked::Send(line)) => Some(line),
451        _ => None,
452    }
453}

455/// One of the card's own parts, drawn from `seat`: what the player asked
456/// of the hub, if anything.
457fn own(
458    ui: &mut egui::Ui,
459    seat: &Seat,
460    part: CardPart,
461    snapshot: Option<&Snapshot>,
462) -> Option<HubAction> {
463    let lifecycle = seat.lifecycle();
464    match part {
465        CardPart::Name => {
466            ui.horizontal(|ui| {
467                ui.strong(&seat.name);
468                connection(ui, lifecycle);
469            });
470            None
471        }
472        CardPart::Character => {
473            ui.horizontal(|ui| ui.strong(&seat.name));
474            None
475        }
476        CardPart::Connection => {
477            ui.horizontal(|ui| connection(ui, lifecycle));
478            None
479        }
480        CardPart::Whereabouts => {
481            let said = if lifecycle == State::Closed {
482                // Why it closed: Hydra's last word on it, a refused login, say.
483                seat.story().said.back().cloned().unwrap_or_default()
484            } else {
485                whereabouts(snapshot, lifecycle)
486            };
487            // In a row, as the name and a field are: every line of a card
488            // sits at the same height.
489            ui.horizontal(|ui| {
490                ui.add(egui::Label::new(said.as_str()).truncate())
491                    .on_hover_text(said.as_str());
492            });
493            None
494        }
495        CardPart::Buttons => {
496            let mut asked = None;
497            ui.horizontal(|ui| {
498                if lifecycle == State::Closed {
499                    if ui.button("Reconnect").clicked() {
500                        asked = Some(HubAction::Ask(HubRequest::Reconnect(seat.id)));
501                    }
502                    if ui.button("Remove").clicked() {
503                        asked = Some(HubAction::Ask(HubRequest::Remove(seat.id)));
504                    }
505                    if ui.button("Log").clicked() {
506                        asked = Some(HubAction::Log(seat.id));
507                    }
508                    return;
509                }
510                if ui.button("Window").clicked() {
511                    asked = Some(HubAction::Show(seat.id));
512                }
513                if ui.button("Log").clicked() {
514                    asked = Some(HubAction::Log(seat.id));
515                }
516                if ui.button("Quit").clicked() {
517                    asked = Some(HubAction::Ask(HubRequest::Remove(seat.id)));
518                }
519            });
520            asked
521        }
522    }
523}

531/// Where the character is and who with: its round time, the room, whom it
532/// leads or follows.
533fn whereabouts(snapshot: Option<&Snapshot>, lifecycle: State) -> String {
534    let state = snapshot.map(|snapshot| &snapshot.state);
535    let view = state.map(|state| PlayView::project(state, lifecycle, state.game_time_now()));
536    let mut facts = Vec::new();
537    if let Some(seconds) = view
538        .as_ref()
539        .and_then(|view| view.roundtime)
540        .filter(|s| *s > 0)
541    {
542        facts.push(format!("RT {seconds}s"));
543    }
544    facts.push(
545        view.as_ref()
546            .and_then(|view| view.room.title.clone())
547            .unwrap_or_else(|| "Room unknown".to_owned()),
548    );
549    if let Some(group) = state.and_then(group_text) {
550        facts.push(group);
551    }
552    facts.join(" · ")
553}

555/// Who a character leads or follows; nothing alone, or while nobody has
556/// said.
557fn group_text(state: &hydra_model::GameState) -> Option<String> {
558    let group = &state.group;
559    match group.leader() {
560        Leader::Unknown => None,
561        Leader::You if group.is_empty() => None,
562        Leader::You => Some(format!(
563            "leading {}",
564            group
565                .members()
566                .iter()
567                .map(|member| member.noun.as_str())
568                .collect::<Vec<_>>()
569                .join(", ")
570        )),
571        Leader::Other(leader) => Some(format!("following {}", leader.noun)),
572    }
573}

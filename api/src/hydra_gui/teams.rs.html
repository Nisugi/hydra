1//! The hub's Teams tab: the teams kept, each started with a click, and the
2//! editor that makes and changes them.
3//!
4//! A team is roster characters started together, each of its windows kept
5//! where it was put for that team (`crate::placement::seat_key`). One
6//! character per account, since an account plays one at a time: once one is
7//! ticked, the others on its account cannot be. Each team starts its
8//! characters in its own mode: their play windows, headless, or for the
9//! web page (`TeamMode`).
10
11use hydra_ui::{HubRequest, RosterCard, TeamCard, TeamMode};
12
13use crate::hub::{HubAction, HubView};
14use crate::launch::{game_name, on_table, roster_name};
15
16/// What the Teams tab is editing, which outlives a frame.
17#[derive(Debug, Default)]
18pub(crate) struct Teams {
19    /// The team being made or changed; `None` while none is.
20    draft: Option<Draft>,
21}
22
23/// A team being made or changed.
24#[derive(Debug, Default)]
25struct Draft {
26    /// The name of the team it changes; `None` for a new one.
27    was: Option<String>,
28    name: String,
29    /// The characters ticked, each as the roster names it: `GAME:Name`.
30    members: Vec<String>,
31    mode: TeamMode,
32}
33
34impl Draft {
35    /// `team`, to change.
36    fn of(team: &TeamCard) -> Self {
37        Self {
38            was: Some(team.name.clone()),
39            name: team.name.clone(),
40            members: team.members.clone(),
41            mode: team.mode,
42        }
43    }
44
45    fn has(&self, member: &str) -> bool {
46        self.members.iter().any(|m| m.eq_ignore_ascii_case(member))
47    }
48
49    /// Whether it can be kept: a name a key can hold, and a character.
50    fn ready(&self) -> bool {
51        let name = self.name.trim();
52        !name.is_empty() && !name.contains(':') && !self.members.is_empty()
53    }
54}
55
56/// What the editor was asked, this frame.
57enum Edited {
58    Save,
59    Cancel,
60}
61
62/// The roster card `member` (`GAME:Name`) is, when it is on the roster.
63fn card_of<'a>(roster: &'a [RosterCard], member: &str) -> Option<&'a RosterCard> {
64    roster
65        .iter()
66        .find(|card| roster_name(&card.game, &card.character).eq_ignore_ascii_case(member))
67}
68
69/// `member` (`GAME:Name`) as the hub says it: its name alone.
70fn said(member: &str) -> &str {
71    member.split_once(':').map_or(member, |(_, name)| name)
72}

74impl Teams {
75    /// Draw the Teams tab over `view`; what the player asked for goes in
76    /// `asked`.
77    pub(crate) fn show(
78        &mut self,
79        ui: &mut egui::Ui,
80        view: &HubView<'_>,
81        asked: &mut Option<HubAction>,
82    ) {
83        egui::ScrollArea::vertical()
84            .id_salt("hub-teams")
85            .auto_shrink(false)
86            .show(ui, |ui| {
87                if let Some(draft) = &mut self.draft {
88                    match edit(ui, draft, view.roster) {
89                        Some(Edited::Save) => {
90                            *asked = Some(HubAction::Ask(HubRequest::SaveTeam {
91                                was: draft.was.clone(),
92                                team: TeamCard {
93                                    name: draft.name.trim().to_owned(),
94                                    members: draft.members.clone(),
95                                    mode: draft.mode,
96                                },
97                            }));
98                            self.draft = None;
99                        }
100                        Some(Edited::Cancel) => self.draft = None,
101                        None => {}
102                    }
103                    ui.separator();
104                } else if ui.button("New team").clicked() {
105                    self.draft = Some(Draft::default());
106                }
107                if view.teams.is_empty() {
108                    ui.weak(
109                        "No team yet. A team starts its characters together, \
110                         each window where you put it for that team.",
111                    );
112                }
113                for team in view.teams {
114                    if let Some(action) = self.card(ui, team, view) {
115                        *asked = Some(action);
116                    }
117                }
118            });
119    }
120
121    /// One team: its name and characters, and Launch, Edit and Forget.
122    fn card(
123        &mut self,
124        ui: &mut egui::Ui,
125        team: &TeamCard,
126        view: &HubView<'_>,
127    ) -> Option<HubAction> {
128        let mut asked = None;
129        egui::Frame::group(ui.style()).show(ui, |ui| {
130            ui.strong(&team.name);
131            let members: Vec<String> = team
132                .members
133                .iter()
134                .map(|member| {
135                    let playing = member
136                        .split_once(':')
137                        .is_some_and(|(game, name)| on_table(view, game, name));
138                    if playing {
139                        format!("{} (playing)", said(member))
140                    } else {
141                        said(member).to_owned()
142                    }
143                })
144                .collect();
145            ui.weak(format!("{} · {}", members.join(", "), team.mode.said()));
146            ui.horizontal(|ui| {
147                if ui.button(format!("Launch {}", team.name)).clicked() {
148                    asked = Some(HubAction::Ask(HubRequest::LaunchTeam(team.name.clone())));
149                }
150                if ui.button("Edit").clicked() {
151                    self.draft = Some(Draft::of(team));
152                }
153                if ui.button("Forget").clicked() {
154                    asked = Some(HubAction::Ask(HubRequest::ForgetTeam(team.name.clone())));
155                }
156            });
157        });
158        asked
159    }
160}

162/// The editor: the team's name, and the roster to tick its characters
163/// from, one per account; then Save and Cancel.
164fn edit(ui: &mut egui::Ui, draft: &mut Draft, roster: &[RosterCard]) -> Option<Edited> {
165    ui.strong(if draft.was.is_some() {
166        "Change the team"
167    } else {
168        "A new team"
169    });
170    ui.horizontal(|ui| {
171        let label = ui.label("Name");
172        ui.text_edit_singleline(&mut draft.name)
173            .labelled_by(label.id);
174    });
175    if draft.name.contains(':') {
176        ui.weak("A team's name cannot hold a colon.");
177    }
178    ui.horizontal(|ui| {
179        ui.label("Starts with");
180        for mode in TeamMode::ALL {
181            ui.radio_value(&mut draft.mode, mode, mode.said())
182                .on_hover_text(match mode {
183                    TeamMode::Gui => {
184                        "Their play windows open; with Hydra resting, those alone, the hub left hidden"
185                    }
186                    TeamMode::Headless => "No play window: each card opens one",
187                    TeamMode::Web => {
188                        "No play window on this computer: the web page is served for them"
189                    }
190                });
191        }
192    });
193    ui.weak("One character per account: an account plays one at a time.");
194    let mut cards: Vec<&RosterCard> = roster.iter().collect();
195    cards.sort_by_key(|card| (!card.favourite, card.character.to_lowercase()));
196    for card in cards {
197        let name = roster_name(&card.game, &card.character);
198        let ticked = draft.has(&name);
199        // Another ticked character on this account, which this one cannot
200        // join.
201        let taken = draft
202            .members
203            .iter()
204            .filter(|member| !member.eq_ignore_ascii_case(&name))
205            .filter_map(|member| card_of(roster, member))
206            .find(|other| other.account.eq_ignore_ascii_case(&card.account));
207        let mut on = ticked;
208        let text = format!(
209            "{} — {} · {}",
210            card.character,
211            game_name(&card.game),
212            card.account
213        );
214        let response = ui.add_enabled(
215            ticked || taken.is_none(),
216            egui::Checkbox::new(&mut on, text),
217        );
218        if let Some(other) = taken {
219            response
220                .on_disabled_hover_text(format!("{} is on this account already", other.character));
221        }
222        if on && !ticked {
223            draft.members.push(name);
224        } else if !on && ticked {
225            draft
226                .members
227                .retain(|member| !member.eq_ignore_ascii_case(&name));
228        }
229    }
230    // A character the team keeps that the roster has since forgotten, to
231    // untick.
232    let gone: Vec<String> = draft
233        .members
234        .iter()
235        .filter(|member| card_of(roster, member).is_none())
236        .cloned()
237        .collect();
238    for member in gone {
239        let mut on = true;
240        ui.checkbox(&mut on, format!("{} — not on the roster", said(&member)));
241        if !on {
242            draft.members.retain(|kept| *kept != member);
243        }
244    }
245    let mut done = None;
246    ui.horizontal(|ui| {
247        if ui
248            .add_enabled(draft.ready(), egui::Button::new("Save team"))
249            .clicked()
250        {
251            done = Some(Edited::Save);
252        }
253        if ui.button("Cancel").clicked() {
254            done = Some(Edited::Cancel);
255        }
256    });
257    done
258}

287    pub(crate) fn harness<'a>(kept: Vec<TeamCard>) -> Harness<'a, Scene> {
288        let scene = Scene {
289            teams: Teams::default(),
290            // Nisugi and Nerten share an account; Ashryn has her own.
291            roster: vec![
292                card("Nisugi", "one"),
293                card("Nerten", "one"),
294                card("Ashryn", "two"),
295            ],
296            kept,
297            asked: Vec::new(),
298        };
299        let design = crate::card::hydras();
300        let mut harness = Harness::builder().with_size((700.0, 600.0)).build_ui_state(
301            move |ui, scene: &mut Scene| {
302                let merged = hydra_ui::MergedHistory::default();
303                let view = HubView {
304                    merged: &merged,
305                    seats: &[],
306                    offered: &[],
307                    roster: &scene.roster,
308                    teams: &scene.kept,
309                    listing: None,
310                    said: None,
311                    update: None,
312                    card: &design,
313                };
314                let mut asked = None;
315                scene.teams.show(ui, &view, &mut asked);
316                scene.asked.extend(asked);
317            },
318            scene,
319        );
320        harness.run();
321        harness
322    }

324    /// A new team: once Nisugi is ticked, Nerten on the same account cannot
325    /// be; Ashryn can, and the team is kept with the two, headless as
326    /// chosen.
327    #[test]
328    fn a_team_takes_one_character_per_account() {
329        let mut harness = harness(Vec::new());
330        harness.get_by_label("New team").click();
331        harness.run();
332        harness
333            .get_by_role(egui::accesskit::Role::TextInput)
334            .click();
335        harness.run();
336        harness
337            .get_by_role(egui::accesskit::Role::TextInput)
338            .type_text("Hunt");
339        harness.run();
340        harness.get_by_label("Nisugi — Prime · one").click();
341        harness.run();
342        // Nisugi has the account: Nerten does not tick.
343        harness.get_by_label("Nerten — Prime · one").click();
344        harness.run();
345        harness.get_by_label("Ashryn — Prime · two").click();
346        harness.run();
347        harness.get_by_label("headless").click();
348        harness.run();
349        harness.get_by_label("Save team").click();
350        harness.run();
351        assert_eq!(
352            harness.state().asked,
353            [HubAction::Ask(HubRequest::SaveTeam {
354                was: None,
355                team: TeamCard {
356                    name: "Hunt".to_owned(),
357                    members: vec!["GS3:Nisugi".to_owned(), "GS3:Ashryn".to_owned()],
358                    mode: TeamMode::Headless,
359                },
360            })]
361        );
362    }

364    /// A team kept is launched, changed and forgotten from its card, which
365    /// says how it starts.
366    #[test]
367    fn a_kept_team_launches_from_its_card() {
368        let hunt = TeamCard {
369            name: "Hunt".to_owned(),
370            members: vec!["GS3:Nisugi".to_owned(), "GS3:Ashryn".to_owned()],
371            mode: TeamMode::Web,
372        };
373        let mut harness = harness(vec![hunt]);
374        assert!(
375            harness
376                .query_by_label("Nisugi, Ashryn · web page")
377                .is_some()
378        );
379        harness.get_by_label("Launch Hunt").click();
380        harness.run();
381        harness.get_by_label("Forget").click();
382        harness.run();
383        assert_eq!(
384            harness.state().asked,
385            [
386                HubAction::Ask(HubRequest::LaunchTeam("Hunt".to_owned())),
387                HubAction::Ask(HubRequest::ForgetTeam("Hunt".to_owned())),
388            ]
389        );
390        harness.get_by_label("Edit").click();
391        harness.run();
392        assert!(
393            harness.query_by_label("Change the team").is_some(),
394            "the editor opens on it"
395        );
396    }

1//! The commands a play window does itself, since what they change is the
2//! window's: `.vellum import`, Vellum's whole folder into Hydra; `.keys
3//! import`, a Wrayth key set or Vellum's keybinds into this character's
4//! keys or every character's, or Vellum's whole folder; and `.doll`, its Injuries widgets turned, or a
5//! folder's doll pictures brought into Hydra's `dolls` folder; and
6//! `.controller`, the settings menu opened at its page; `.hotbars`, the
7//! hotbar editor opened, or the hotbars files read again. Answered in
8//! the story, as a command is. And `.hotbar`, a hotbar's button pressed,
9//! which the command line claims, typed or put by a script, and the window
10//! presses on its character's hotbars.

26/// Whether `rest`, a line's words after its command symbol, is `.hotbar`.
27fn is_hotbar_word(rest: &str) -> bool {
28    rest.split_whitespace()
29        .next()
30        .is_some_and(|word| word.eq_ignore_ascii_case("hotbar"))
31}

57    /// `line`, typed on `seat`'s window, done here when it is `.keys` or
58    /// `.doll` with the character's own command symbol: whether it was.
59    pub(super) fn window_command(&mut self, seat: &Seat, line: &str) -> bool {
60        let line = line.trim();
61        let Some(rest) = line.strip_prefix(self.symbol_of(seat)) else {
62            return false;
63        };
64        // An alias named like one of these runs in its place.
65        if seat.handle().is_some_and(|handle| handle.alias_fits(line)) {
66            return false;
67        }
68        let (word, rest) = rest
69            .trim_start()
70            .split_once(char::is_whitespace)
71            .unwrap_or((rest.trim_start(), ""));
72        let rest = rest.trim();
73        let said = if word.eq_ignore_ascii_case("vellum") {
74            self.vellum_command(seat, rest)
75        } else if word.eq_ignore_ascii_case("keys") {
76            self.keys_command(seat, rest)
77        } else if word.eq_ignore_ascii_case("wizard") {
78            self.wizard_command(seat, rest)
79        } else if word.eq_ignore_ascii_case("alias")
80            && let Some(trigger) = rest.strip_prefix("edit")
81            && (trigger.is_empty() || trigger.starts_with(char::is_whitespace))
82        {
83            // `.alias edit [trigger]`: the Aliases page, for this character,
84            // at the trigger. The rest of `.alias` is Hydra's command line's.
85            let trigger = trigger.trim();
86            if !trigger.is_empty() {
87                self.alias_editor.want(trigger);
88            }
89            self.editors.open_at(
90                Some(format!("{}:{}", seat.game, seat.name)),
91                crate::editors_menu::Editor::Aliases,
92            );
93            vec![format!(
94                "Aliases: the editor is open{}.",
95                if trigger.is_empty() {
96                    String::new()
97                } else {
98                    format!(" at {trigger}")
99                }
100            )]
101        } else if word.eq_ignore_ascii_case("doll") {
102            self.doll_command(seat, rest)
103        } else if word.eq_ignore_ascii_case("controller") {
104            self.controller_command(seat, rest)
105        } else if word.eq_ignore_ascii_case("hotbars") {
106            self.hotbars_command(seat, rest)
107        } else if word.eq_ignore_ascii_case("savelayout") {
108            self.layout_command(seat, rest, true)
109        } else if word.eq_ignore_ascii_case("loadlayout") {
110            self.layout_command(seat, rest, false)
111        } else if word.eq_ignore_ascii_case("map") {
112            self.map_command(rest)
113        } else {
114            return false;
115        };
116        let kind = if said.len() > 1 {
117            NoticeKind::Warn
118        } else {
119            NoticeKind::Info
120        };
121        let mut story = seat.story();
122        story.hear(&Event::Claimed(line.to_owned()));
123        story.push(hydra_ui::Shown::Said(Notice::prose(kind, said).answering()));
124        true
125    }

127    /// Each `.hotbar` line the characters' command lines claimed, pressed
128    /// on its character's hotbars: what each sends, as a click would; why
129    /// not, said in its story.
130    pub(super) fn hotbar_pressed(&mut self) -> Vec<(Arc<Seat>, Asked)> {
131        let seats: Vec<Arc<Seat>> = self
132            .characters
133            .values()
134            .map(|character| Arc::clone(&character.seat))
135            .collect();
136        let mut asks = Vec::new();
137        for seat in seats {
138            let lines = seat.hotbar_presses();
139            if lines.is_empty() {
140                continue;
141            }
142            let symbol = self.symbol_of(&seat);
143            let Some(character) = self.characters.get_mut(&seat.id) else {
144                continue;
145            };
146            for line in lines {
147                let after_symbol = line
148                    .trim()
149                    .strip_prefix(symbol)
150                    .map(str::trim_start)
151                    .unwrap_or_default();
152                let (first, rest) = after_symbol
153                    .split_once(char::is_whitespace)
154                    .map_or((after_symbol, ""), |(first, rest)| (first, rest.trim()));
155                let is_macro = first.eq_ignore_ascii_case("macro");
156                // `.macro` alone, and its help, list, show and alias, are
157                // the command line's own answers: not the window's to run.
158                if is_macro && !is_macro_run(rest) {
159                    continue;
160                }
161                let pressed = {
162                    let story = seat.story();
163                    match (is_macro, crate::hotbars::named(rest), story.world()) {
164                        (_, _, None) => Err("Not yet: the character is not known yet.".to_owned()),
165                        (true, _, Some(world)) => character.play.macro_run(rest, &world),
166                        (false, None, _) => Err(HOTBAR_HELP.to_owned()),
167                        (false, Some(named), Some(world)) => {
168                            character.play.hotbar_press(named, &world)
169                        }
170                    }
171                };
172                // A button that presses buttons would press for ever.
173                let pressed = pressed.and_then(|command| {
174                    let own = command
175                        .trim()
176                        .strip_prefix(symbol)
177                        .is_some_and(is_hotbar_word);
178                    if own {
179                        Err("Not sent: a hotbar button never presses another.".to_owned())
180                    } else {
181                        Ok(command)
182                    }
183                });
184                match pressed {
185                    Ok(command) => asks.push((Arc::clone(&seat), Asked::Chain(command))),
186                    Err(why) => {
187                        let said = Notice::line(NoticeKind::Warn, why).answering();
188                        seat.story().push(hydra_ui::Shown::Said(said));
189                    }
190                }
191            }
192        }
193        asks
194    }

196    /// `.controller`: the settings menu opened at the Controller page, the
197    /// character's own, or with `global` every character's; `.controller
198    /// reload` reads the controller files again.
199    fn controller_command(&mut self, seat: &Seat, rest: &str) -> Vec<String> {
200        if rest.eq_ignore_ascii_case("reload") {
201            self.pad.read(&self.data);
202            let mut said = vec!["The controller files are read again.".to_owned()];
203            said.extend(self.pad.said.iter().cloned());
204            return said;
205        }
206        let whose =
207            (!rest.eq_ignore_ascii_case("global")).then(|| format!("{}:{}", seat.game, seat.name));
208        let said = if whose.is_some() {
209            "The Editors window is open at the Controller, on this character's own."
210        } else {
211            "The Editors window is open at the Controller, on every character's."
212        };
213        self.editors
214            .open_at(whose, crate::editors_menu::Editor::Controller);
215        vec![said.to_owned()]
216    }

233    /// `.doll`, its words after it `rest`: what was done.
234    fn doll_command(&mut self, seat: &Seat, rest: &str) -> Vec<String> {
235        let (verb, after) = rest.split_once(char::is_whitespace).unwrap_or((rest, ""));
236        let after = after.trim();
237        match verb.to_ascii_lowercase().as_str() {
238            "face" => {
239                let Some(deg) = after
240                    .trim_end_matches('\u{b0}')
241                    .parse::<f32>()
242                    .ok()
243                    .filter(|deg| deg.is_finite())
244                else {
245                    return vec![format!(
246                        "Doll: `{after}` is not a facing in degrees. {DOLL_HELP}"
247                    )];
248                };
249                #[expect(
250                    clippy::cast_possible_truncation,
251                    reason = "wrapped to -180..180 first"
252                )]
253                let deg = ((deg + 180.0).rem_euclid(360.0) - 180.0).round() as i16;
254                let turned = self
255                    .characters
256                    .get_mut(&seat.id)
257                    .map_or(0, |character| character.play.turn_dolls(deg));
258                vec![match turned {
259                    0 => "Doll: this window has no Injuries widget to turn.".to_owned(),
260                    _ => format!("Doll: turned to {deg}\u{b0}."),
261                }]
262            }
263            "import" if !after.is_empty() => {
264                let folder = Path::new(after.trim_matches('"'));
265                match crate::doll_import::import_dolls(folder, &self.data.join("dolls")) {
266                    Ok(brought) => vec![doll_said(&brought)],
267                    Err(why) => vec![format!("Doll: {why}")],
268                }
269            }
270            _ => vec![DOLL_HELP.to_owned()],
271        }
272    }

274    /// `.savelayout <name>` (`saving`) and `.loadlayout [<name>]`: this
275    /// window's whole layout kept under a name for any character, and one
276    /// so kept taken as this window's; `.loadlayout` alone lists them.
277    fn layout_command(&mut self, seat: &Seat, name: &str, saving: bool) -> Vec<String> {
278        let name = name.trim().trim_matches('"');
279        let Some(character) = self.characters.get_mut(&seat.id) else {
280            return vec!["This character has no window.".to_owned()];
281        };
282        let play = &mut character.play;
283        if saving {
284            if name.is_empty() {
285                return vec![
286                    "savelayout <name>: save this window's layout, every window \
287                             sized and placed as it is, for any character to load."
288                        .to_owned(),
289                ];
290            }
291            return vec![play.save_layout_as(name).unwrap_or_else(|why| why)];
292        }
293        if name.is_empty() {
294            let names = play.named_layouts();
295            return vec![if names.is_empty() {
296                "No layouts are saved by name; savelayout <name> saves this one.".to_owned()
297            } else {
298                format!(
299                    "loadlayout <name> takes one of these as this window's: {}.",
300                    names.join(", ")
301                )
302            }];
303        }
304        vec![play.load_layout_named(name).unwrap_or_else(|why| why)]
305    }

337    /// `file` imported: a Wrayth settings file's key sets, or a Vellum
338    /// keybinds file, into `seat`'s character's keys or every character's
339    /// (`every`); or Vellum's folder ([`Self::import_vellum_folder`]). What
340    /// was done, a line at a time.
341    fn import(&mut self, seat: &Seat, file: &Path, every: bool) -> Vec<String> {
342        if file.is_dir() {
343            return self.import_vellum_folder(file);
344        }
345        let text = match std::fs::read_to_string(file) {
346            Ok(text) => text,
347            Err(why) => return vec![format!("Keys: {} could not be read: {why}", file.display())],
348        };
349        let whom = if every {
350            self.every_characters()
351        } else {
352            Whom {
353                path: self.mine_path(seat),
354                whose: Whose::Character,
355                named: format!("{}'s keys", seat.name),
356            }
357        };
358        let name = super::keyed::name_of(file);
359        // Wrayth's settings are XML; Vellum's keybinds, TOML.
360        if text.trim_start().starts_with('<') {
361            let (imported, said) = wrayth::read(&text);
362            return self.bind_imported(&whom, &imported, said, ("Wrayth", &name));
363        }
364        self.vellum_into(&text, &whom, &name)
365    }

396    /// `.wizard import [global] <folder>`: a Wizard front end folder into
397    /// Hydra: its scripts as this character's aliases, or everyone's with
398    /// `global`; its profiles' macros as this character's keys, or every
399    /// character's; and its highlights, asked of Hydra with `.trigger
400    /// import`. What was done, a line at a time.
401    fn wizard_command(&mut self, seat: &Seat, rest: &str) -> Vec<String> {
402        let (verb, given) = rest.split_once(char::is_whitespace).unwrap_or((rest, ""));
403        let (every, given) = match given.trim().strip_prefix("global") {
404            Some(after) if after.starts_with(char::is_whitespace) => (true, after.trim()),
405            _ => (false, given.trim()),
406        };
407        let given = given.trim_matches('"');
408        if !verb.eq_ignore_ascii_case("import") || given.is_empty() {
409            return vec![WIZARD_HELP.to_owned()];
410        }
411        let folder = Path::new(given);
412        if !folder.is_dir() {
413            return vec![format!("There is no folder at {}.", folder.display())];
414        }
415        let mut lines = self.wizard_scripts(seat, folder, every);
416        let profiles = hydra_board::wizard_import::profiles(folder);
417        lines.extend(self.wizard_keys(seat, &profiles, every));
418        if profiles
419            .values()
420            .any(|p| p.strings.is_some() || p.names.is_some())
421        {
422            lines.push("Highlights: asked of Hydra; its answer follows.".to_owned());
423            let symbol = self.symbol_of(seat);
424            seat.send(&format!("{symbol}trigger import {}", folder.display()));
425        }
426        if lines.is_empty() {
427            lines.push(format!(
428                "Nothing of the Wizard's in {}: no .Cmd or .Wiz scripts, no .Mac, .His or .Nam.",
429                folder.display()
430            ));
431        }
432        lines
433    }

435    /// The Wizard scripts of `folder` as aliases: this character's, or
436    /// everyone's when `every`. What was done.
437    fn wizard_scripts(&self, seat: &Seat, folder: &Path, every: bool) -> Vec<String> {
438        let (scripts, mut said) = hydra_board::wizard_import::scripts(folder);
439        if scripts.is_empty() {
440            return said;
441        }
442        let key = seat
443            .instance
444            .as_deref()
445            .map(|instance| hydra_session::alias::character(instance, &seat.name));
446        let (whose, named) = match (&key, every) {
447            (_, true) => (
448                hydra_session::alias::Whose::Everyone,
449                "everyone's".to_owned(),
450            ),
451            (Some(key), false) => (
452                hydra_session::alias::Whose::Own(key),
453                format!("{}'s", seat.name),
454            ),
455            (None, false) => {
456                return vec![
457                    "This character's game is not known yet; try once it is in.".to_owned(),
458                ];
459            }
460        };
461        let count = scripts.len();
462        let skipped: usize = scripts.iter().map(|s| s.skipped.len()).sum();
463        // Each typed with the symbol, as the Wizard ran them: `s` stays south.
464        let symbol = self.symbol_of(seat);
465        let mut cleared = 0;
466        let put = hydra_session::alias::change(&self.data, |file| {
467            for script in &scripts {
468                let trigger = format!("{symbol}{}", script.trigger);
469                file.put(whose, &trigger, &script.target)
470                    .map_err(|why| format!("{trigger}: {why}"))?;
471                // The same script brought in before without the symbol.
472                let before = file
473                    .of(whose)
474                    .and_then(|aliases| aliases.get(&script.trigger))
475                    .is_some_and(|target| *target == script.target);
476                if before {
477                    file.remove(whose, &script.trigger);
478                    cleared += 1;
479                }
480            }
481            Ok(())
482        });
483        let mut lines = vec![match put {
484            Ok(()) => format!(
485                "Aliases: {count} script(s) brought in as {named}{}.",
486                if skipped > 0 {
487                    format!(", {skipped} line(s) skipped")
488                } else {
489                    String::new()
490                }
491            ),
492            Err(why) => format!("Aliases: {why}. None was brought in."),
493        }];
494        if cleared > 0 {
495            lines.push(format!(
496                "{cleared} of them brought in before without the {symbol}: those taken away."
497            ));
498        }
499        // A script named like one of Hydra's commands: the alias runs in its
500        // place.
501        let commands = seat
502            .handle()
503            .map(hydra_session::SessionHandle::command_words)
504            .unwrap_or_default();
505        for script in &scripts {
506            if commands
507                .iter()
508                .any(|word| word.eq_ignore_ascii_case(&script.trigger))
509            {
510                lines.push(format!(
511                    "{symbol}{} is also one of Hydra's commands: your alias wins, so \
512                     {symbol}{} runs the script.",
513                    script.trigger, script.trigger
514                ));
515            }
516            if let Some(seconds) = script.every {
517                lines.push(format!(
518                    "{symbol}{} looped every {seconds} seconds: run it as \
519                     `{symbol}every {seconds} {symbol}{}`.",
520                    script.trigger, script.trigger
521                ));
522            }
523        }
524        lines.append(&mut said);
525        lines
526    }

528    /// Each profile's `.Mac` as keys: this character's, or every
529    /// character's when `every`. What was done.
530    fn wizard_keys(
531        &mut self,
532        seat: &Seat,
533        profiles: &std::collections::BTreeMap<String, hydra_board::wizard_import::Profile>,
534        every: bool,
535    ) -> Vec<String> {
536        let mut lines = Vec::new();
537        for (name, profile) in profiles {
538            let Some(path) = &profile.macros else {
539                continue;
540            };
541            let text = match std::fs::read_to_string(path) {
542                Ok(text) => text,
543                Err(why) => {
544                    lines.push(format!("Keys: {} could not be read: {why}", path.display()));
545                    continue;
546                }
547            };
548            let (imported, said) = hydra_board::keys::wizard::read(&text);
549            let whom = if every {
550                self.every_characters()
551            } else {
552                Whom {
553                    path: self.mine_path(seat),
554                    whose: Whose::Character,
555                    named: format!("{}'s keys", seat.name),
556                }
557            };
558            lines.extend(self.bind_imported(&whom, &imported, said, ("Wizard", name)));
559        }
560        lines
561    }

563    /// `.vellum import [folder]`: Vellum's folder, the home folder's unless
564    /// named, into Hydra: the keys ([`Self::import_vellum_folder`]), the
565    /// files (`hydra_board::vellum_import`), and the highlights, asked of
566    /// Hydra with `.trigger import`. What was done, a line at a time.
567    fn vellum_command(&mut self, seat: &Seat, rest: &str) -> Vec<String> {
568        let (verb, given) = rest.split_once(char::is_whitespace).unwrap_or((rest, ""));
569        if !verb.eq_ignore_ascii_case("import") {
570            return vec![VELLUM_HELP.to_owned()];
571        }
572        let given = Some(given.trim()).filter(|given| !given.is_empty());
573        let Some(folder) = hydra_board::vellum_import::folder(given) else {
574            return vec!["There is no home folder to find Vellum's in.".to_owned()];
575        };
576        if !folder.is_dir() {
577            return vec![format!(
578                "There is no Vellum folder at {}.",
579                folder.display()
580            )];
581        }
582        let mut known: Vec<hydra_board::vellum_import::Known> = self
583            .characters
584            .values()
585            .map(|character| (character.seat.game.clone(), character.seat.name.clone()))
586            .collect();
587        for card in self.sessions.glance(std::time::Instant::now()).roster {
588            let card = (card.game, card.character);
589            if !known.contains(&card) {
590                known.push(card);
591            }
592        }
593        let playing: Vec<hydra_board::vellum_import::Known> = self
594            .sessions
595            .seats()
596            .iter()
597            .filter(|seated| seated.lifecycle() != hydra_session::State::Closed)
598            .map(|seated| (seated.game.clone(), seated.name.clone()))
599            .collect();
600        let mut lines = vec![format!("Importing Vellum's folder, {}:", folder.display())];
601        lines.extend(self.import_vellum_folder(&folder));
602        lines.extend(hydra_board::vellum_import::import(
603            &folder, &self.data, &known, &playing,
604        ));
605        lines.push("Highlights: asked of Hydra; its answer follows.".to_owned());
606        let symbol = self.symbol_of(seat);
607        seat.send(&format!("{symbol}trigger import {}", folder.display()));
608        lines
609    }
610
611    /// Vellum's folder, `.vellum-fe`: its `global/keybinds.toml` into every
612    /// character's keys, then each character's own,
613    /// `profiles/<name>/keybinds.toml`, into theirs -- a character seated or
614    /// on the roster, by name, on whichever game it plays. A profile of a
615    /// character Hydra does not know is named, not imported.
616    fn import_vellum_folder(&mut self, folder: &Path) -> Vec<String> {
617        let global = folder.join("global").join("keybinds.toml");
618        let profiles = folder.join("profiles");
619        if !global.is_file() && !profiles.is_dir() {
620            return vec![format!(
621                "Keys: {} holds no Vellum keybinds: Vellum keeps them in global/keybinds.toml \
622                 and profiles/<character>/keybinds.toml.",
623                folder.display()
624            )];
625        }
626        let mut lines = Vec::new();
627        if let Ok(text) = std::fs::read_to_string(&global) {
628            let whom = self.every_characters();
629            lines.extend(self.vellum_into(&text, &whom, "global/keybinds.toml"));
630        }
631        // Every character Hydra knows, by name, and its keys file.
632        let mut known: Vec<(String, std::path::PathBuf)> = self
633            .characters
634            .values()
635            .map(|character| (character.seat.name.clone(), self.mine_path(&character.seat)))
636            .collect();
637        for card in self.sessions.glance(std::time::Instant::now()).roster {
638            let path = self.mine_path_of(&card.game, &card.character);
639            if !known.iter().any(|(_, kept)| *kept == path) {
640                known.push((card.character, path));
641            }
642        }
643        let mut folders: Vec<std::path::PathBuf> = std::fs::read_dir(&profiles)
644            .into_iter()
645            .flatten()
646            .filter_map(|entry| entry.ok().map(|entry| entry.path()))
647            .filter(|profile| profile.join("keybinds.toml").is_file())
648            .collect();
649        folders.sort();
650        let mut strangers = Vec::new();
651        for profile in folders {
652            let name = super::keyed::name_of(&profile);
653            let Ok(text) = std::fs::read_to_string(profile.join("keybinds.toml")) else {
654                continue;
655            };
656            let theirs: Vec<(String, std::path::PathBuf)> = known
657                .iter()
658                .filter(|(character, _)| character.eq_ignore_ascii_case(&name))
659                .cloned()
660                .collect();
661            if theirs.is_empty() {
662                strangers.push(name);
663                continue;
664            }
665            for (character, path) in theirs {
666                self.mine_now(&path);
667                let whom = Whom {
668                    path,
669                    whose: Whose::Character,
670                    named: format!("{character}'s keys"),
671                };
672                let shown = format!("profiles/{name}/keybinds.toml");
673                lines.extend(self.vellum_into(&text, &whom, &shown));
674            }
675        }
676        if !strangers.is_empty() {
677            lines.push(format!(
678                "Not characters Hydra knows, so not imported: {}.",
679                strangers.join(", ")
680            ));
681        }
682        lines
683    }
684
685    /// `imported`, another front end's keys, bound in `whom`'s keys file,
686    /// with `said`, what could not be, after; `from` the front end and the
687    /// file's name, for what is said. What was done, a line at a time.
688    fn bind_imported(
689        &mut self,
690        whom: &Whom,
691        imported: &[Imported],
692        mut said: Vec<String>,
693        (from, file): (&str, &str),
694    ) -> Vec<String> {
695        let (path, whose) = (whom.path.clone(), whom.whose);
696        let every = whose == Whose::Every;
697        // A key already doing what it would -- in that file as it is now,
698        // its own line counted -- is left as it is.
699        let mine = (!every)
700            .then(|| self.mine.get(&path).map(|(mine, _)| mine))
701            .flatten();
702        let (mut changes, mut already, mut moved) = (Vec::new(), 0, Vec::new());
703        for key in imported {
704            let beneath = self.keys.beneath(whose, &key.chord);
705            if key.set == 0 && self.keys.in_file(whose, mine, &key.chord) == Some(&key.does) {
706                already += 1;
707                continue;
708            }
709            if key.set == 0 && beneath.is_some_and(|beneath| *beneath != key.does) {
710                moved.push(key.chord.written());
711            }
712            changes.push(KeyChange::Bind {
713                key: key.chord.written(),
714                does: key.does.clone(),
715                was: None,
716                place: Place {
717                    set: key.set,
718                    every,
719                },
720            });
721        }
722        let name = super::keyed::name_of(&path);
723        let old = match self.keys_text(&path) {
724            Ok(old) => old,
725            Err(why) => return vec![why],
726        };
727        let (text, made, mut refused) =
728            match write::made_all(&old, (&name, whose), &changes, (&self.keys, "Keys")) {
729                Ok(done) => done,
730                Err(why) => return vec![why],
731            };
732        // The same keys made to the file with the next batch of saves.
733        let keys = self.keys.clone();
734        self.keys_written(
735            &path,
736            text,
737            Box::new(move |now: &str| {
738                write::made_all(now, (&name, whose), &changes, (&keys, "Keys"))
739                    .map(|(text, _, _)| text)
740            }),
741        );
742        let mut lines = vec![format!(
743            "Keys: {made} imported from {file} into {}; {already} already so.",
744            whom.named
745        )];
746        if !moved.is_empty() {
747            lines.push(format!(
748                "These now do what {from} had them do, not Hydra's: {}. Restore one on the Keybinds page for Hydra's.",
749                moved.join(", ")
750            ));
751        }
752        said.append(&mut refused);
753        if !said.is_empty() {
754            lines.push("Not imported:".to_owned());
755            lines.append(&mut said);
756        }
757        lines
758    }
759}

761/// Where an import's keys go: the keys file, whose it is, and how it is
762/// named in what is said.
763struct Whom {
764    /// The keys file.
765    path: std::path::PathBuf,
766    /// Every character's, or one character's own.
767    whose: Whose,
768    /// `every character's keys`, `Ashryn's keys`.
769    named: String,
770}

772/// What a doll import says it brought.
773fn doll_said(brought: &crate::doll_import::DollsImported) -> String {
774    let mut said = format!(
775        "Doll: from {}, {} picture(s) and {} overlay(s) into Hydra's dolls folder; {} \
776         calibrated from the file beside it. Pick one on the Injuries widget's page.",
777        brought.from.display(),
778        brought.pictures,
779        brought.overlays,
780        brought.calibrated
781    );
782    if !brought.kept.is_empty() {
783        let _ = write!(
784            said,
785            " Already there, so left as they are: {}.",
786            brought.kept.join(", ")
787        );
788    }
789    said

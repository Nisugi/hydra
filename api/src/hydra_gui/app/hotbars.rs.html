1//! The hotbar editor's page in Settings: over the hotbars of the character
2//! Settings shows, or everyone's while it shows Hydra's own; and what the
3//! editor asks done, through that character's hotbars, which every other
4//! character reads again within a second.
5
6use std::sync::Arc;
7
8use hydra_ui::hotbar::Bar;
9
10use super::App;
11use crate::hotbar_editor::{Asked, Folder};
12use crate::seat::Seat;
13
14impl App {
15    /// The character the editor works through, and whether its own bars
16    /// are shown: the character Settings shows, while it plays; with
17    /// Hydra's own shown, the first that plays, its bars everyone's alone.
18    /// None while no character it could work through plays.
19    pub(super) fn hotbar_session(&self) -> Option<(u32, bool)> {
20        match self.settings.character() {
21            Some(_) => self.shown_play().map(|session| (session, true)),
22            None => self
23                .characters
24                .keys()
25                .next()
26                .map(|session| (*session, false)),
27        }
28    }
29
30    /// The bars the editor shows from `session`'s hotbars, read fresh: each
31    /// with whether it is the character's own, and those alone that are
32    /// everyone's when its own are not shown; its seat, and the icons
33    /// folder.
34    pub(super) fn hotbar_bars(
35        &mut self,
36        context: &egui::Context,
37        session: Option<(u32, bool)>,
38    ) -> (Vec<(Bar, bool)>, Option<Arc<Seat>>, Folder) {
39        let Some((session, own)) = session else {
40            return (Vec::new(), None, Folder::default());
41        };
42        let Some(character) = self.characters.get_mut(&session) else {
43            return (Vec::new(), None, Folder::default());
44        };
45        let hotbars = character.play.hotbars_mut();
46        hotbars.fresh(Some(context));
47        let bars: Vec<(Bar, bool)> = hotbars
48            .file
49            .bars
50            .iter()
51            .map(|bar| (bar.clone(), hotbars.is_own(&bar.name)))
52            .filter(|(_, mine)| own || !mine)
53            .collect();
54        let seat = Arc::clone(&character.seat);
55        // The folder as last listed, once a second: not listed every frame.
56        let folder = self.icon_folder_of(Some(session), std::time::Instant::now());
57        (bars, Some(seat), folder)
58    }
59
60    /// Do what the editor asked, on `session`'s hotbars, and tell it what
61    /// was done.
62    pub(super) fn hotbar_asked(&mut self, session: Option<u32>, asked: Asked) {
63        let Some(character) = session.and_then(|session| self.characters.get_mut(&session)) else {
64            self.hotbar_editor
65                .said("Hotbars are kept with a character: start one first.".to_owned());
66            return;
67        };
68        let hotbars = character.play.hotbars_mut();
69        match asked {
70            Asked::Save { bar, own, was } => {
71                let was = was.as_ref().map(|(name, own)| (name.as_str(), *own));
72                match hotbars.save_bar(&bar, own, was) {
73                    Ok(()) => {
74                        self.hotbar_editor.kept();
75                        self.hotbar_editor.said(format!("{} saved.", bar.name));
76                    }
77                    Err(why) => self.hotbar_editor.said(format!("Not saved: {why}")),
78                }
79            }
80            Asked::CopyIn(adding) => {
81                let sheet = adding.sheet.trim();
82                let sheet = (!sheet.is_empty()).then_some((sheet, adding.cell));
83                match hotbars.copy_in(std::path::Path::new(adding.file.trim()), sheet) {
84                    Ok(done) => {
85                        self.hotbar_editor.copied();
86                        self.hotbar_editor.said(done);
87                    }
88                    Err(why) => self.hotbar_editor.said(format!("Not copied in: {why}")),
89                }
90            }
91            Asked::Remove { name, own } => match hotbars.remove_bar(&name, own) {
92                Ok(()) => {
93                    self.hotbar_editor.removed();
94                    self.hotbar_editor.said(format!("{name} taken away."));
95                }
96                Err(why) => self.hotbar_editor.said(format!("Not taken away: {why}")),
97            },
98        }
99    }
100
101    /// `.hotbars` on `seat`'s window: Settings opened at its Hotbars page;
102    /// `.hotbars reload`, every character's hotbars read again now.
103    pub(super) fn hotbars_command(&mut self, seat: &Seat, rest: &str) -> Vec<String> {
104        if rest.eq_ignore_ascii_case("reload") {
105            for character in self.characters.values_mut() {
106                character.play.hotbars_mut().reload();
107            }
108            return vec!["The hotbars files are read again.".to_owned()];
109        }
110        self.settings.open_at(
111            Some(format!("{}:{}", seat.game, seat.name)),
112            Some(crate::settings_menu::Outside::Hotbars.id()),
113        );
114        vec!["Settings is open at this character's hotbars.".to_owned()]
115    }
116}

23    /// The Editors window, while it is open; the editor showing drawn in it.
24    pub(super) fn editors_window(&mut self, context: &egui::Context, glance: &Glance) {
25        self.editors.take_asked(context);
26        let showing = self.editors.showing();
27        self.editors_shown(showing);
28        if !self.editors.open {
29            return;
30        }
31        if self.editors.take_raise() {
32            context.send_viewport_cmd_to(
33                crate::editors_menu::viewport(),
34                egui::ViewportCommand::Focus,
35            );
36        }
37        // What the editor showing draws from, read before the window's view
38        // holds the window's own.
39        let (trigger_file, (hotbar_edited, macro_edited)) = self.editor_files(showing);
40        let (bars, hotbar_seat, folder, macro_names) =
41            self.hotbar_bars(context, hotbar_edited.as_ref());
42        let (macro_defs, macro_seat) = self.macro_defs(context, macro_edited.as_ref());
43        let stream_rules =
44            (showing == Some(Editor::Streams)).then(|| self.sessions.stream_rules().current());
45        let keyed = self.editors_keys();
46        // The hotbar editor's character, copied: no story is held while the
47        // window is drawn, since a page clicked to is drawn in the same
48        // frame and may read any story (the card editor's widgets do).
49        let hotbar_tried = Tried::of(hotbar_seat.as_deref());
50        let hotbar_world = hotbar_tried.world();
51        let hotbar_view = hotbar_edited
52            .as_ref()
53            .map(|edited| crate::hotbar_editor::View {
54                bars,
55                macros: macro_names,
56                character: edited.own_name(),
57                world: hotbar_world.as_ref(),
58                folder,
59            });
60        let macro_tried = Tried::of(macro_seat.as_deref());
61        let macro_world = macro_tried.world();
62        let macro_view = macro_edited
63            .as_ref()
64            .map(|edited| crate::macro_editor::View {
65                macros: macro_defs,
66                character: edited.own_name(),
67                world: macro_world.as_ref(),
68            });
69        if self.font_names.is_none() {
70            self.font_names = Some(self.fonts.names());
71        }
72        let none = crate::fonts::Names::default();
73        let fonts = self.font_names.as_ref().unwrap_or(&none);
74        let view = EditorsView {
75            roster: &glance.roster,
76            keys: KeysView {
77                bound: &keyed.bound,
78                numpad_always: self.keys.numpad_always(),
79                said: &keyed.said,
80                caught: self.caught_for_page.as_deref(),
81                character: keyed.whose,
82                chosen: keyed.chosen,
83            },
84            controller: crate::pad::ControllerView {
85                file: &keyed.controller.0,
86                beneath: keyed.whose.then_some(&keyed.controller.1),
87                from: &keyed.controller.2,
88                connected: self.pad.connected(),
89                captured: keyed.captured,
90                sticks: self.pad.stick_values(),
91            },
92        };
93        let menu = &mut self.editors;
94        let mut editors = Editors {
95            triggers: &mut self.highlighter,
96            streams: (
97                &mut self.stream_editor,
98                stream_rules
99                    .as_deref()
100                    .map(hydra_session::stream_rules::Compiled::rules),
101            ),
102            themes: (&mut self.themer, &self.themes),
103            lookout: (&mut self.lookout_editor, &self.data),
104            hotbars: &mut self.hotbar_editor,
105            macros: &mut self.macro_editor,
106            cards: (&mut self.cards, &glance.seats),
107            trigger_file: trigger_file.as_deref(),
108            hotbar_view: hotbar_view.as_ref(),
109            macro_view: macro_view.as_ref(),
110            fonts,
111            asked: EditorsAsked::default(),
112        };
113        // What every pass asks, kept: a click is in the first pass alone.
114        let mut asked = Vec::new();
115        let closed = self.placements.show(
116            context,
117            EDITORS,
118            crate::editors_menu::viewport(),
119            ([1100.0, 760.0], format!("Editors — {TITLE}")),
120            |ui| {
121                egui::CentralPanel::default().show(ui, |ui| {
122                    asked.extend(menu.show(ui, &view, &mut |ui, editor| editors.show(ui, editor)));
123                });
124            },
125        );
126        let editors_asked = editors.asked;
127        self.editors_done(closed, asked, editors_asked, (hotbar_edited, macro_edited));
128    }

130    /// What the editor showing draws from: the triggers, once read, while
131    /// their page shows; the character the hotbar editor and the macro
132    /// editor each work through, while its page shows, the hotbar editor's
133    /// while the Icons page does too; none for a page not showing.
134    fn editor_files(
135        &self,
136        showing: Option<Editor>,
137    ) -> (
138        Option<Arc<hydra_ui::highlight::HighlightFile>>,
139        (Shown, Shown),
140    ) {
141        let trigger_file =
142            (showing == Some(Editor::Triggers)).then(|| self.sessions.highlights().file());
143        // The Icons page lists the icons folder and copies in through the
144        // hotbars of whoever the window works for, as the hotbar editor does.
145        let hotbars =
146            matches!(showing, Some(Editor::Hotbars | Editor::Icons)).then(|| self.edited());
147        let macros = (showing == Some(Editor::Macros)).then(|| self.edited());
148        (trigger_file, (hotbars, macros))
149    }

151    /// The Editors window drawn: closed, when `closed`; what its pages and
152    /// the editors asked, done, the hotbar editor's through `hotbars` and
153    /// the macro editor's through `macros`.
154    fn editors_done(
155        &mut self,
156        closed: bool,
157        asked: Vec<MenuAsked>,
158        editors: EditorsAsked,
159        (hotbars, macros): (Shown, Shown),
160    ) {
161        if closed {
162            self.editors.open = false;
163        }
164        for asked in asked {
165            self.menu_asked(asked);
166        }
167        for ask in editors.triggers {
168            self.highlight_asked(ask);
169        }
170        for ask in editors.streams {
171            self.stream_asked(&ask);
172        }
173        for ask in editors.themes {
174            self.theme_asked(ask);
175        }
176        for ask in editors.hotbars {
177            self.hotbar_asked(hotbars.as_ref(), ask);
178        }
179        for ask in editors.macros {
180            self.macro_asked(macros.as_ref(), ask);
181        }
182    }

184    /// What the Keybinds and Controller pages draw from: the keys and the
185    /// controller file of whoever the window works for, or every
186    /// character's.
187    fn editors_keys(&mut self) -> Keyed {
188        let whose = self.editors_whose();
189        let controller = self.controller_files(whose.as_ref());
190        let file = whose
191            .as_ref()
192            .map(|(game, name)| self.mine_path_of(game, name));
193        self.load_mine(file.as_deref());
194        let read = file.as_ref().and_then(|file| self.mine.get(file));
195        let mine = read.map(|(mine, _)| mine);
196        let set = self.editors.keys_set();
197        let mut bound = self.keys.rows(set, mine);
198        // A macro's key is on the page too, as `macro: <name>`, in the set
199        // it names or every set, so a key two things claim is seen.
200        if let Some((game, name)) = &whose
201            && let Some(character) = self
202                .characters
203                .values()
204                .find(|character| character.seat.game == *game && character.seat.name == *name)
205        {
206            for made in &character.play.hotbars().macros.macros {
207                if let Some(key) = &made.key
208                    && made.set.is_none_or(|own| own == set)
209                {
210                    bound.push(crate::keys::KeyRow {
211                        key: key.clone(),
212                        does: Some(crate::keys::Macro::Run(made.name.clone())),
213                        default: None,
214                        from: Some(crate::keys::Whose::Character),
215                        beneath: None,
216                    });
217                }
218            }
219        }
220        let said = match &whose {
221            Some((_, name)) => self.keys_said(read, name),
222            None => self.keys_said.clone(),
223        };
224        let chosen = mine.map_or(0, |mine| mine.chosen);
225        Keyed {
226            whose: whose.is_some(),
227            bound,
228            said,
229            chosen,
230            controller,
231            captured: self.pad_for_page.take(),
232        }
233    }

235    /// The controller file the Controller page shows for `whose`
236    /// character, or every character's for none; every character's, beneath
237    /// a character's; and where the file is.
238    fn controller_files(
239        &mut self,
240        whose: Option<&(String, String)>,
241    ) -> (
242        crate::pad::ControllerFile,
243        crate::pad::ControllerFile,
244        String,
245    ) {
246        let (every, from_file) = self.pad.every_file();
247        let every = every.clone();
248        let file = self.data.join(crate::pad::FILE);
249        let Some((game, name)) = whose else {
250            let from = if from_file {
251                format!("Every character's: {}", file.display())
252            } else {
253                format!("Hydra's own, until a change writes {}", file.display())
254            };
255            return (every.clone(), every, from);
256        };
257        let own = crate::pad::character_path(&self.layouts, hydra_net::instance(game), name);
258        let from = format!("This character's own, over everyone's: {}", own.display());
259        (self.pad.own_file(&own).clone(), every, from)
260    }

262    /// The editors open as their pages show: the theme editor's draft worn
263    /// only while *Themes* shows, put off as the page is left or the window
264    /// closes.
265    fn editors_shown(&mut self, showing: Option<Editor>) {
266        let themes = showing == Some(Editor::Themes);
267        if themes && !self.themer.open {
268            self.open_theme_editor();
269        }
270        if !themes {
271            self.themer.open = false;
272            if self.preview.take().is_some() {
273                self.worn = None;
274            }
275        }
276        self.highlighter.open = showing == Some(Editor::Triggers);
277        self.hotbar_editor.open = showing == Some(Editor::Hotbars);
278        self.macro_editor.open = showing == Some(Editor::Macros);
279        self.cards.editor.open = showing == Some(Editor::Cards);
280    }

282    /// Do what the Streams page asked, and tell it what was done: at once
283    /// to the rules every session sorts by, and to their file with the next
284    /// batch of saves.
285    fn stream_asked(&mut self, ask: &crate::stream_editor::Asked) {
286        let made = ask.clone();
287        let done = hydra_board::stream_rules::change(
288            &self.data,
289            &self.sessions.stream_rules(),
290            move |rules| made.made(rules),
291        );
292        match done {
293            Ok(said) => {
294                self.stream_editor.done(ask);
295                self.stream_editor.said(said);
296            }
297            Err(why) => self.stream_editor.said(why),
298        }
299    }

308    /// The game and name of the character the Editors window works for;
309    /// `None` while it works for every character.
310    pub(super) fn editors_whose(&self) -> Option<(String, String)> {
311        let (game, name) = self.editors.character()?.split_once(':')?;
312        Some((game.to_owned(), name.to_owned()))

354/// The editors the window draws on their pages, and what each draws from.
355struct Editors<'a, 'w> {
356    triggers: &'a mut crate::highlight_editor::Editor,
357    /// The stream rules' editor, and the rules while its page shows.
358    streams: (
359        &'a mut crate::stream_editor::Editor,
360        Option<&'a hydra_session::stream_rules::Rules>,
361    ),
362    themes: (
363        &'a mut crate::theme_editor::Editor,
364        &'a hydra_ui::theme::Themes,
365    ),
366    hotbars: &'a mut crate::hotbar_editor::Editor,
367    macros: &'a mut crate::macro_editor::Editor,
368    cards: (&'a mut crate::card::Cards, &'a [Arc<crate::seat::Seat>]),
369    /// The lookout's ignore list's page, and the data folder it is kept in.
370    lookout: (
371        &'a mut crate::lookout_editor::LookoutEditor,
372        &'a std::path::Path,
373    ),
374    /// The triggers, once read, while their page shows.
375    trigger_file: Option<&'a hydra_ui::highlight::HighlightFile>,
376    /// What the hotbar editor draws over, while its page or the Icons page
377    /// shows.
378    hotbar_view: Option<&'a crate::hotbar_editor::View<'a, 'w>>,
379    /// What the macro editor draws over, while its page shows.
380    macro_view: Option<&'a crate::macro_editor::View<'a, 'w>>,
381    fonts: &'a crate::fonts::Names,
382    asked: EditorsAsked,
383}

385impl Editors<'_, '_> {
386    /// `editor` drawn in `ui`; the Keybinds and Controller pages are the
387    /// window's own.
388    fn show(&mut self, ui: &mut egui::Ui, editor: Editor) {
389        match editor {
390            Editor::Triggers => {
391                if let Some(file) = self.trigger_file {
392                    self.asked.triggers.extend(self.triggers.show(ui, file));
393                }
394            }
395            Editor::Streams => {
396                if let (editor, Some(rules)) = (&mut *self.streams.0, self.streams.1) {
397                    self.asked.streams.extend(editor.show(ui, rules));
398                }
399            }
400            Editor::Themes => {
401                let (editor, themes) = (&mut *self.themes.0, self.themes.1);
402                self.asked
403                    .themes
404                    .extend(editor.show(ui, themes, self.fonts));
405            }
406            Editor::Hotbars => match self.hotbar_view {
407                Some(view) => self.asked.hotbars.extend(self.hotbars.show(ui, view)),
408                None => {
409                    ui.weak("Hotbars are kept with a character: there is none to work through.");
410                }
411            },
412            Editor::Macros => match self.macro_view {
413                Some(view) => self.asked.macros.extend(self.macros.show(ui, view)),
414                None => {
415                    ui.weak("Macros are kept with a character: there is none to work through.");
416                }
417            },
418            Editor::Cards => self.cards.0.edit(ui, self.cards.1),
419            Editor::Lookout => {
420                let (editor, data) = (&mut *self.lookout.0, self.lookout.1);
421                editor.show(ui, data);
422            }
423            // The editors above scroll themselves; these are scrolled here,
424            // as a settings page is.
425            Editor::Packs => scrolled(ui, editor, crate::theme_editor::packs::show),
426            Editor::Icons => {
427                if let Some(view) = self.hotbar_view {
428                    let hotbars = &mut *self.hotbars;
429                    let asked = &mut self.asked.hotbars;
430                    scrolled(ui, editor, |ui| {
431                        asked.extend(hotbars.icons(ui, &view.folder));
432                    });
433                }
434            }
435            Editor::Keybinds | Editor::Controller => {}
436        }
437    }
438}

440/// `editor`'s page drawn by `draw` in a scroll area filling what the window
441/// has left, so a page taller than the window scrolls.
442fn scrolled(ui: &mut egui::Ui, editor: Editor, draw: impl FnOnce(&mut egui::Ui)) {
443    egui::ScrollArea::vertical()
444        .id_salt(("editors-page", editor.title()))
445        .auto_shrink([false, false])
446        .show(ui, draw);

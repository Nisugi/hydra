21/// The character a custom indicator's rule is tried on as it is built, on
22/// its page, and a hotbar button's look in the hotbar editor: its state,
23/// its flags and custom statuses, and its effects' names to pick from.
24#[derive(Default)]
25pub(super) struct Tried {
26    /// What it knows, once its feed has seen anything.
27    snapshot: Option<std::sync::Arc<hydra_session::Snapshot>>,
28    /// The flags its triggers set.
29    flags: hydra_ui::condition::Flags,
30    /// The custom statuses its triggers lit.
31    statuses: hydra_ui::condition::Flags,
32    /// Its effects' names, each once.
33    names: Vec<String>,
34}
35
36impl Tried {
37    /// `seat`'s, read now; nothing for none.
38    pub(super) fn of(seat: Option<&crate::seat::Seat>) -> Self {
39        let Some(seat) = seat else {
40            return Self::default();
41        };
42        let (flags, statuses) = {
43            let story = seat.story();
44            (story.flags.clone(), story.statuses.clone())
45        };
46        let snapshot = seat.snapshot();
47        let names: std::collections::BTreeSet<String> = snapshot
48            .iter()
49            .flat_map(|snapshot| snapshot.state.effects.iter())
50            .map(|(_, effect)| effect.text.trim().to_owned())
51            .filter(|name| !name.is_empty())
52            .collect();
53        Self {
54            snapshot,
55            flags,
56            statuses,
57            names: names.into_iter().collect(),
58        }
59    }
60
61    /// The names the title bar's game values are picked from: what the
62    /// character's state can work out now.
63    fn values(&self) -> Vec<String> {
64        self.snapshot
65            .as_ref()
66            .map(|snapshot| snapshot.state.value_names())
67            .unwrap_or_default()
68    }
69
70    /// The world a condition is tried in, once its state is known.
71    pub(super) fn world(&self) -> Option<hydra_ui::condition::World<'_>> {
72        let unix = std::time::SystemTime::now()
73            .duration_since(std::time::UNIX_EPOCH)
74            .map_or(0, |since| {
75                i64::try_from(since.as_secs()).unwrap_or(i64::MAX)
76            });
77        self.snapshot
78            .as_ref()
79            .map(|snapshot| hydra_ui::condition::World {
80                state: &snapshot.state,
81                flags: &self.flags,
82                statuses: &self.statuses,
83                unix,
84            })
85    }
86}

89    /// The settings menu, in a window of its own, while it is open.
90    pub(super) fn settings_window(&mut self, context: &egui::Context, glance: &Glance) {
91        if !self.settings.open {
92            return;
93        }
94        if self.settings.take_raise() {
95            context.send_viewport_cmd_to(viewport(), egui::ViewportCommand::Focus);
96        }
97        let gathered = self.gather();
98        let (values, world) = (gathered.tried.values(), gathered.tried.world());
99        let none = crate::fonts::Names::default();
100        let fonts = self.font_names.as_ref().unwrap_or(&none);
101        let view = MenuView {
102            roster: &glance.roster,
103            pages: glance
104                .settings
105                .as_ref()
106                .map(|(whose, pages)| (whose.as_str(), pages.as_slice())),
107            said: glance.said.as_deref(),
108            own: &gathered.own,
109            widgets: &gathered.widgets,
110            shelves: &gathered.shelves,
111            values: &values,
112            fonts,
113            icons: &gathered.icons,
114            live: crate::highlight_editor::when::Live {
115                world: world.as_ref(),
116                names: &gathered.tried.names,
117            },
118        };
119        let menu = &mut self.settings;
120        // What every pass asks, kept: a click is in the first pass alone.
121        let mut asked = Vec::new();
122        let closed = self.placements.show(
123            context,
124            SETTINGS,
125            viewport(),
126            ([1100.0, 760.0], format!("Settings — {TITLE}")),
127            |ui| {
128                egui::CentralPanel::default().show(ui, |ui| {
129                    asked.extend(menu.show(ui, &view));
130                });
131            },
132        );
133        self.settings_done(closed, asked);
134    }

136    /// The Settings window drawn: closed, when `closed`; what the menu
137    /// asked, done.
138    fn settings_done(&mut self, closed: bool, asked: Vec<MenuAsked>) {
139        if closed {
140            self.settings.open = false;
141            // the statuses put on a puppet from its page come off with it
142            for character in self.characters.values_mut() {
143                character.play.forget_shown();
144            }
145        }
146        for asked in asked {
147            self.menu_asked(asked);
148        }
149    }

151    /// What the menu draws from this frame that the window holds: Hydra's
152    /// own page, the shown character's widget pages and its state, the
153    /// icons folder.
154    fn gather(&mut self) -> Gathered {
155        let mut own = self
156            .own
157            .pages(&self.themes.names(), self.theme_problem.as_deref());
158        hydra_board::lookout::with_trigger_row(&mut own, &self.sessions.highlights());
159        let pictures = self.pictures();
160        if self.font_names.is_none() {
161            self.font_names = Some(self.fonts.names());
162        }
163        let icons = self.icon_folder(std::time::Instant::now());
164        let shown = self
165            .shown_play()
166            .and_then(|session| self.characters.get(&session));
167        let widgets = shown
168            .map(|character| {
169                let now = character
170                    .seat
171                    .snapshot()
172                    .and_then(|snapshot| snapshot.state.game_time_now());
173                character.play.widget_pages_at(&pictures, now)
174            })
175            .unwrap_or_default();
176        let shelves = shown
177            .map(|character| character.play.widget_shelves())
178            .unwrap_or_default();
179        Gathered {
180            own,
181            widgets,
182            shelves,
183            tried: Tried::of(shown.map(|character| &*character.seat)),
184            icons,
185        }
186    }

188    /// What the Menu was asked for, done: from `seat`'s play window, its
189    /// settings opened, or an editor working for its character; from the
190    /// hub, whoever's were shown, or for whomever it worked for.
191    pub(super) fn menu_chosen(
192        &mut self,
193        choice: crate::main_menu::Choice,
194        seat: Option<&crate::seat::Seat>,
195    ) {
196        use crate::main_menu::Choice;
197        match (choice, seat) {
198            (Choice::Settings, Some(seat)) => {
199                let whose = format!("{}:{}", seat.game, seat.name);
200                self.settings.open_at(Some(whose), None);
201            }
202            (Choice::Settings, None) => self.settings.open_at(None, None),
203            (Choice::Pair, _) => self.pairing.open(),
204            (Choice::Editor(editor), Some(seat)) => {
205                let whose = format!("{}:{}", seat.game, seat.name);
206                self.editors.open_at(Some(whose), editor);
207            }
208            (Choice::Editor(editor), None) => self.editors.open(editor),
209            // From a play window, its own character's.
210            (Choice::Open(Tool::Log), Some(seat)) => self.open_log(seat),
211            (Choice::Open(Tool::Reports), Some(seat)) => self.open_reports(Some(seat)),
212            (Choice::Open(tool), _) => self.open_tool(tool),
213            // A play window's own, done by the window.
214            (Choice::Layout(_), _) => {}
215            (Choice::Performance(on), _) => self.hub.perf = on,
216        }
217    }

219    /// A window of its own a *Tools* entry asked for: the Map, the Survey,
220    /// or the Log of the character the menu shows.
221    fn open_tool(&mut self, tool: Tool) {
222        match tool {
223            Tool::Map => {
224                self.explorer.open = true;
225                self.mapped.wanted(&self.data);
226            }
227            Tool::Survey => self.surveyor.open = true,
228            Tool::Log => {
229                let seat = self
230                    .shown_play()
231                    .and_then(|session| self.characters.get(&session))
232                    .map(|character| Arc::clone(&character.seat));
233                if let Some(seat) = seat {
234                    self.open_log(&seat);
235                }
236            }
237            Tool::Reports => {
238                let seat = self
239                    .shown_play()
240                    .and_then(|session| self.characters.get(&session))
241                    .map(|character| Arc::clone(&character.seat));
242                self.open_reports(seat.as_deref());
243            }
244        }
245    }

247    /// The icons folder as an icon's row offers it, its sheets the shown
248    /// character's hotbars'; read again once a second at most, `now` being
249    /// now, so a picture copied in is offered soon after.
250    fn icon_folder(&mut self, now: std::time::Instant) -> crate::hotbar_editor::icon::Folder {
251        let session = self.shown_play();
252        self.icon_folder_of(session, now)
253    }
254
255    /// [`Self::icon_folder`], its sheets the hotbars' of `session`'s
256    /// character: read again once a second, or when the character differs.
257    pub(super) fn icon_folder_of(
258        &mut self,
259        session: Option<u32>,
260        now: std::time::Instant,
261    ) -> crate::hotbar_editor::icon::Folder {
262        let fresh = self.icon_folder.as_ref().is_some_and(|(read, whose, _)| {
263            *whose == session
264                && now.saturating_duration_since(*read) < std::time::Duration::from_secs(1)
265        });
266        if !fresh {
267            let playing = session
268                .and_then(|session| self.characters.get(&session))
269                .map(|character| {
270                    let hotbars = character.play.hotbars();
271                    crate::hotbar_editor::icon::Folder::read(
272                        hotbars.icons_dir(),
273                        hotbars.sheets.clone(),
274                    )
275                });
276            // Not playing: the folder is everyone's, read through the
277            // hotbars of whoever is shown, as the Icons page copies in.
278            let folder = playing.unwrap_or_else(|| {
279                let edited = self.edited();
280                let hotbars = self.hotbars_of(&edited);
281                hotbars.fresh(None);
282                crate::hotbar_editor::icon::Folder::read(
283                    hotbars.icons_dir(),
284                    hotbars.sheets.clone(),
285                )
286            });
287            self.icon_folder = Some((now, session, folder));
288        }
289        self.icon_folder
290            .as_ref()
291            .map(|(_, _, folder)| folder.clone())
292            .unwrap_or_default()
293    }

295    /// Do what the menu asked: the binary's through it, the rest here,
296    /// saying what was done.
297    pub(super) fn menu_asked(&mut self, asked: MenuAsked) {
298        let said = match asked {
299            MenuAsked::Binary(request) => return self.sessions.ask(request, &self.runtime),
300            MenuAsked::EditTheme(name) => {
301                let name = if name.is_empty() {
302                    self.own.theme().to_owned()
303                } else {
304                    name
305                };
306                self.edit_theme(&name);
307                return;
308            }
309            MenuAsked::ReloadKeys => {
310                self.read_keys();
311                self.editors
312                    .tell("The keybinds files are read again.".to_owned());
313                return;
314            }
315            // The lookout's trigger is the highlights', which the window
316            // holds.
317            MenuAsked::Own { key, to } if key == hydra_board::lookout::TRIGGER_KEY => {
318                hydra_board::lookout::set_trigger(
319                    &self.data,
320                    &self.sessions.highlights(),
321                    to.as_deref() == Some("on"),
322                )
323            }
324            MenuAsked::Own { key, to } => {
325                let said = self.own.change(&key, to.as_deref());
326                // Worn again at the next frame, as the change says.
327                self.worn = None;
328                self.sessions.want_web(self.own.web());
329                said
330            }
331            // The keys' and the controller's pages are the Editors
332            // window's: it says what was done.
333            MenuAsked::Key { character, change } => {
334                let mine = character
335                    .as_deref()
336                    .and_then(|character| character.split_once(':'))
337                    .map(|(game, name)| (self.mine_path_of(game, name), name.to_owned()));
338                let said = self.key_change_for(mine.as_ref(), &change);
339                self.editors.tell(said.unwrap_or_else(|why| why));
340                return;
341            }
342            MenuAsked::Controller { character, asked } => match asked {
343                crate::pad::PageAsked::Test(buzz) => {
344                    self.pad.play(buzz, std::time::Instant::now());
345                    return;
346                }
347                crate::pad::PageAsked::Change(change) => {
348                    let own = character
349                        .as_deref()
350                        .and_then(|character| character.split_once(':'))
351                        .map(|(game, name)| {
352                            crate::pad::character_path(
353                                &self.layouts,
354                                hydra_net::instance(game),
355                                name,
356                            )
357                        });
358                    let said = self.pad.change(&self.data, own.as_deref(), &change);
359                    self.editors.tell(said.unwrap_or_else(|why| why));
360                    return;
361                }
362            },
363            MenuAsked::Widget { page, key, to } => self
364                .shown_play()
365                .and_then(|session| self.characters.get_mut(&session))
366                .map_or_else(
367                    || Err("That character has no window open to change.".to_owned()),
368                    |character| character.play.widget_change(&page, &key, to.as_deref()),
369                ),
370        };
371        self.settings.tell(said.unwrap_or_else(|why| why));
372    }

374    /// The game and name of the character the menu shows; `None` while it
375    /// shows Hydra's own.
376    pub(super) fn menu_whose(&self) -> Option<(String, String)> {
377        let (game, name) = self.settings.character()?.split_once(':')?;
378        Some((game.to_owned(), name.to_owned()))
379    }
380
381    /// The session of the character the menu shows, when it is seated.
382    pub(super) fn shown_play(&self) -> Option<u32> {
383        let (game, name) = self.menu_whose()?;
384        self.playing_as(&game, &name)
385    }
386
387    /// The session of `game`'s character `name`, when it is seated.
388    pub(super) fn playing_as(&self, game: &str, name: &str) -> Option<u32> {
389        self.characters
390            .values()
391            .find(|character| {
392                character.seat.game.eq_ignore_ascii_case(game)
393                    && character.seat.name.eq_ignore_ascii_case(name)
394            })
395            .map(|character| character.seat.id)
396    }

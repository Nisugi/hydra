18impl App {
19    /// The settings menu, in a window of its own, while it is open.
20    pub(super) fn settings_window(&mut self, context: &egui::Context, glance: &Glance) {
21        if !self.settings.open {
22            return;
23        }
24        let own = [self
25            .own
26            .page(&self.themes.names(), self.theme_problem.as_deref())];
27        let whose = self.menu_whose();
28        let (controller_file, every_controller, from) = self.controller_files(whose.as_ref());
29        let file = whose
30            .as_ref()
31            .map(|(game, name)| self.mine_path_of(game, name));
32        self.load_mine(file.as_deref());
33        let read = file.as_ref().and_then(|file| self.mine.get(file));
34        let mine = read.map(|(mine, _)| mine);
35        let bound = self.keys.rows(self.settings.keys_set(), mine);
36        let keys_said = match &whose {
37            Some((_, name)) => self.keys_said(read, name),
38            None => self.keys_said.clone(),
39        };
40        let chosen = mine.map_or(0, |mine| mine.chosen);
41        let pictures = self.pictures();
42        if self.font_names.is_none() {
43            self.font_names = Some(self.fonts.names());
44        }
45        let none = crate::fonts::Names::default();
46        let fonts = self.font_names.as_ref().unwrap_or(&none);
47        let shown = self
48            .shown_play()
49            .and_then(|session| self.characters.get(&session));
50        let widgets = shown
51            .map(|character| {
52                let now = character
53                    .seat
54                    .snapshot()
55                    .and_then(|snapshot| snapshot.state.game_time_now());
56                character.play.widget_pages_at(&pictures, now)
57            })
58            .unwrap_or_default();
59        // The names the title bar's game values are picked from: what the
60        // character's state can work out now.
61        let values = shown
62            .and_then(|character| character.seat.snapshot())
63            .map(|snapshot| snapshot.state.value_names())
64            .unwrap_or_default();
65        let beneath = whose.is_some().then_some(&every_controller);
66        let captured = self.pad_for_page.take();
67        let view = MenuView {
68            roster: &glance.roster,
69            pages: glance
70                .settings
71                .as_ref()
72                .map(|(whose, pages)| (whose.as_str(), pages.as_slice())),
73            said: glance.said.as_deref(),
74            own: &own,
75            keys: KeysView {
76                bound: &bound,
77                numpad_always: self.keys.numpad_always(),
78                said: &keys_said,
79                caught: self.caught_for_page.as_deref(),
80                character: whose.is_some(),
81                chosen,
82            },
83            controller: crate::pad::ControllerView {
84                file: &controller_file,
85                beneath,
86                from: &from,
87                connected: self.pad.connected(),
88                captured,
89                sticks: self.pad.stick_values(),
90            },
91            widgets: &widgets,
92            values: &values,
93            fonts,
94        };
95        let menu = &mut self.settings;
96        // What every pass asks, kept: a click is in the first pass alone.
97        let mut asked = Vec::new();
98        let closed = self.placements.show(
99            context,
100            SETTINGS,
101            viewport(),
102            ([760.0, 560.0], format!("Settings — {TITLE}")),
103            |ui| {
104                egui::CentralPanel::default().show(ui, |ui| asked.extend(menu.show(ui, &view)));
105            },
106        );
107        if closed {
108            self.settings.open = false;
109            // the statuses put on a puppet from its page come off with it
110            for character in self.characters.values_mut() {
111                character.play.forget_shown();
112            }
113        }
114        for asked in asked {
115            self.menu_asked(asked);
116        }
117    }

119    /// Do what the menu asked: the binary's through it, the rest here,
120    /// saying what was done.
121    pub(super) fn menu_asked(&mut self, asked: MenuAsked) {
122        let said = match asked {
123            MenuAsked::Binary(request) => return self.sessions.ask(request, &self.runtime),
124            MenuAsked::Own { key, to } => {
125                let said = self.own.change(&key, to.as_deref());
126                // Worn again at the next frame, as the change says.
127                self.worn = None;
128                self.sessions.want_web(self.own.web());
129                said
130            }
131            MenuAsked::Key { character, change } => {
132                let mine = character
133                    .as_deref()
134                    .and_then(|character| character.split_once(':'))
135                    .map(|(game, name)| (self.mine_path_of(game, name), name.to_owned()));
136                self.key_change_for(mine.as_ref(), &change)
137            }
138            MenuAsked::Controller { character, asked } => match asked {
139                crate::pad::PageAsked::Test(buzz) => {
140                    self.pad.play(buzz, std::time::Instant::now());
141                    return;
142                }
143                crate::pad::PageAsked::Change(change) => {
144                    let own = character
145                        .as_deref()
146                        .and_then(|character| character.split_once(':'))
147                        .map(|(game, name)| {
148                            crate::pad::character_path(
149                                &self.layouts,
150                                hydra_net::instance(game),
151                                name,
152                            )
153                        });
154                    self.pad.change(&self.data, own.as_deref(), &change)
155                }
156            },
157            MenuAsked::Widget { page, key, to } => self
158                .shown_play()
159                .and_then(|session| self.characters.get_mut(&session))
160                .map_or_else(
161                    || Err("That character has no window open to change.".to_owned()),
162                    |character| character.play.widget_change(&page, &key, to.as_deref()),
163                ),
164        };
165        self.settings.tell(said.unwrap_or_else(|why| why));
166    }

168    /// The controller file the *Controller* page shows for `whose`
169    /// character, or every character's for none; every character's, beneath
170    /// a character's; and where the file is.
171    fn controller_files(
172        &mut self,
173        whose: Option<&(String, String)>,
174    ) -> (
175        crate::pad::ControllerFile,
176        crate::pad::ControllerFile,
177        String,
178    ) {
179        let (every, from_file) = self.pad.every_file();
180        let every = every.clone();
181        let file = self.data.join(crate::pad::FILE);
182        let Some((game, name)) = whose else {
183            let from = if from_file {
184                format!("Every character's: {}", file.display())
185            } else {
186                format!("Hydra's own, until a change writes {}", file.display())
187            };
188            return (every.clone(), every, from);
189        };
190        let own = crate::pad::character_path(&self.layouts, hydra_net::instance(game), name);
191        let from = format!("This character's own, over everyone's: {}", own.display());
192        (self.pad.own_file(&own).clone(), every, from)
193    }

195    /// The game and name of the character the menu shows; `None` while it
196    /// shows Hydra's own.
197    fn menu_whose(&self) -> Option<(String, String)> {
198        let (game, name) = self.settings.character()?.split_once(':')?;
199        Some((game.to_owned(), name.to_owned()))
200    }
201
202    /// The session of the character the menu shows, when it is seated.
203    fn shown_play(&self) -> Option<u32> {
204        let (game, name) = self.menu_whose()?;
205        self.characters
206            .values()
207            .find(|character| {
208                character.seat.game.eq_ignore_ascii_case(&game)
209                    && character.seat.name.eq_ignore_ascii_case(&name)
210            })
211            .map(|character| character.seat.id)
212    }

1//! A character's own theme and accent, read from its settings file: the
2//! `theme` section the binary's *Theme* page and `.theme` write
3//! ([`Chosen`]), which its play window wears in place of Hydra's; and
4//! whether its creatures' tags are shown (`.targetid`, [`Shown`]), and how
5//! its container looks are sorted (`.sorter`, [`Sorting`]), read with it.
6//!
7//! The file is the binary's to write; this reads it off the window's
8//! thread (`crate::disk`), once a second at most and only when its modified
9//! time has moved, and keeps the outfit worked out from it until the file,
10//! Hydra's theme or the themes change.
11
12use std::collections::HashMap;
13use std::path::{Path, PathBuf};
14
15use hydra_model::targetid::{Shown, Style};
16use hydra_session::settings_store;
17use hydra_ui::history::Kept;
18use hydra_ui::sorter::{Sorted, Sorting};
19use hydra_ui::theme::{Chosen, Outfit, Themes};
20
21/// One character's look, as its file last said.
22#[derive(Debug, Default)]
23struct Looked {
24    /// The file's read the outfit was worked out from, by its generation;
25    /// `None`, none yet, or not since Hydra's theme changed.
26    seen: Option<u64>,
27    /// The outfit its window wears; `None` when it chose nothing, or what
28    /// it chose cannot be worn, so the window wears Hydra's.
29    outfit: Option<Outfit>,
30    /// How its creatures' tags look, while shown.
31    tags: Option<Style>,
32    /// How its container looks are sorted; `None`, not.
33    sorting: Option<Sorted>,
34    /// The fewest characters a typed line needs to be kept in its history.
35    history_min: usize,
36}
37
38/// The characters' looks, by settings file.
39#[derive(Debug, Default)]
40pub(super) struct Looks {
41    looked: HashMap<PathBuf, Looked>,
42}
43
44impl Looks {
45    /// Forget every outfit worked out, so each is worked out again over the
46    /// theme Hydra wears now.
47    pub(super) fn wear_again(&mut self) {
48        for (file, looked) in &mut self.looked {
49            looked.seen = None;
50            crate::disk::Disk::hydras().look_again(&key(file));
51        }
52    }
53
54    /// The outfit the character `name` on `instance` wears, its file in
55    /// `dir`, read again when it has changed; `None`, Hydra's own, `hydras`.
56    pub(super) fn outfit(
57        &mut self,
58        dir: &Path,
59        (instance, name): (&str, &str),
60        themes: &Themes,
61        hydras: &str,
62    ) -> Option<Outfit> {
63        let file = settings_store::settings_path(dir, instance, name)?;
64        let (owned, instance, name) = (dir.to_owned(), instance.to_owned(), name.to_owned());
65        let kept = crate::disk::Disk::hydras().kept(&key(&file), &file, None, move |_| {
66            Some(settings_store::load(&owned, &instance, &name).ok())
67        });
68        let looked = self.looked.entry(file).or_default();
69        if let Some((generation, read)) = kept
70            && looked.seen != Some(generation)
71        {
72            {
73                looked.seen = Some(generation);
74                let chosen: Chosen = read
75                    .as_ref()
76                    .and_then(|file| file.section(Chosen::SECTION).ok())
77                    .unwrap_or_default();
78                looked.tags = read
79                    .as_ref()
80                    .and_then(|file| file.section::<Shown>(Shown::SECTION).ok())
81                    .and_then(|shown| shown.style());
82                looked.sorting = read
83                    .as_ref()
84                    .and_then(|file| file.section::<Sorting>(Sorting::SECTION).ok())
85                    .and_then(|sorting| sorting.sorted());
86                looked.history_min = read
87                    .as_ref()
88                    .and_then(|file| file.section::<Kept>(Kept::SECTION).ok())
89                    .unwrap_or_default()
90                    .min_length();
91                looked.outfit = (chosen != Chosen::default())
92                    .then(|| themes.outfit_for(hydras, &chosen).ok())
93                    .flatten();
94            }
95        }
96        looked.outfit.clone()
97    }
98
99    /// How the character `name` on `instance` shows its creatures' tags, as
100    /// its file said when [`Looks::outfit`] last read it; `None`, not shown.
101    pub(super) fn tags(&self, dir: &Path, (instance, name): (&str, &str)) -> Option<Style> {
102        let file = settings_store::settings_path(dir, instance, name)?;
103        self.looked.get(&file).and_then(|looked| looked.tags)
104    }
105
106    /// How the character `name` on `instance` sorts its container looks, as
107    /// its file said when [`Looks::outfit`] last read it; `None`, not.
108    pub(super) fn sorting(&self, dir: &Path, (instance, name): (&str, &str)) -> Option<Sorted> {
109        let file = settings_store::settings_path(dir, instance, name)?;
110        self.looked.get(&file).and_then(|looked| looked.sorting)
111    }
112
113    /// The fewest characters a line the character `name` on `instance`
114    /// types needs to be kept in its history, as its file said when
115    /// [`Looks::outfit`] last read it; the default until then.
116    pub(super) fn history_min(&self, dir: &Path, (instance, name): (&str, &str)) -> usize {
117        settings_store::settings_path(dir, instance, name)
118            .and_then(|file| self.looked.get(&file))
119            .map_or_else(|| Kept::default().min_length(), |looked| looked.history_min)
120    }
121
122    /// Every outfit a character wears now.
123    pub(super) fn outfits(&self) -> impl Iterator<Item = &Outfit> {
124        self.looked
125            .values()
126            .filter_map(|looked| looked.outfit.as_ref())
127    }
128}
129
130/// What a character's settings file is kept under.
131fn key(file: &Path) -> String {
132    format!("settings:{}", file.display())
133}

135#[cfg(test)]
136mod tests {
137    use hydra_ui::theme::{Theme, Token};
138
139    use super::*;
140
141    #[test]
142    fn a_characters_own_theme_and_accent_are_read_and_worn() {
143        let dir = std::env::temp_dir().join(format!("hydra-looks-{}", std::process::id()));
144        let _ = std::fs::remove_dir_all(&dir);
145        std::fs::create_dir_all(&dir).expect("made");
146        let themes = Themes::built_in();
147        let who = ("Prime", "Ashryn");
148        let mut looks = Looks::default();
149        assert_eq!(
150            looks.outfit(&dir, who, &themes, Theme::DEFAULT),
151            None,
152            "no file: Hydra's"
153        );
154        let mut settings = settings_store::SettingsFile::new(who.0, who.1);
155        let chosen = Chosen {
156            theme: Some(Theme::LIGHT.to_owned()),
157            accent: Some("#c9733a".to_owned()),
158        };
159        settings.set_section(Chosen::SECTION, &chosen).expect("set");
160        settings_store::save(&dir, &settings).expect("saved");
161        looks.wear_again();
162        let palette = looks
163            .outfit(&dir, who, &themes, Theme::DEFAULT)
164            .expect("its own")
165            .palette;
166        assert_eq!(palette.get(Token::Accent), [0xc9, 0x73, 0x3a]);
167        assert_eq!(
168            palette.get(Token::Canvas),
169            themes
170                .outfit(Theme::LIGHT)
171                .expect("light")
172                .palette
173                .get(Token::Canvas)
174        );
175        assert_eq!(looks.outfits().count(), 1);
176        let _ = std::fs::remove_dir_all(&dir);
177    }
178
179    /// Creature tags off until the file says on; read with the look.
180    #[test]
181    fn creature_tags_are_read_from_the_file() {
182        let dir = std::env::temp_dir().join(format!("hydra-looks-tags-{}", std::process::id()));
183        let _ = std::fs::remove_dir_all(&dir);
184        std::fs::create_dir_all(&dir).expect("made");
185        let (themes, who) = (Themes::built_in(), ("Prime", "Ashryn"));
186        let mut looks = Looks::default();
187        let _ = looks.outfit(&dir, who, &themes, Theme::DEFAULT);
188        assert_eq!(looks.tags(&dir, who), None, "off until asked");
189        let mut settings = settings_store::SettingsFile::new(who.0, who.1);
190        settings
191            .set_section(
192                Shown::SECTION,
193                &Shown {
194                    enabled: Some(true),
195                },
196            )
197            .expect("set");
198        settings_store::save(&dir, &settings).expect("saved");
199        looks.wear_again();
200        let _ = looks.outfit(&dir, who, &themes, Theme::DEFAULT);
201        assert_eq!(looks.tags(&dir, who), Some(Style::default()));
202        let _ = std::fs::remove_dir_all(&dir);
203    }

1//! A pressed key's macro, done: commands sent on the
2//! character as if typed, the command input filled, or one of Hydra's
3//! actions asked of the window as its own button would ask it. A send
4//! macro's commands after a wait are kept here and sent when it is over,
5//! each frame sending those due and asking for a frame at the next.
6//!
7//! **The keys the fork catches**: the numpad's, through
8//! its numpad channel, and those egui has no name for -- Pause, Scroll Lock
9//! and the rest -- through its key capture, pinned at `ed8b264`. Each is told
10//! only the bound keys, so an unbound one still types or reaches egui; while
11//! the Keys page waits for a key it is told every one, and the first press
12//! goes to the page. On macOS the Clear key, which winit calls `NumLock`, is
13//! always caught: it switches the numpad between typing and sending its
14//! keys, as `NumLock` does elsewhere. Each is told the keys of the play
15//! window that last had the keyboard, its character's own among them.
16

33/// How the fork catches the numpad: everything while the Keys page waits
34/// for a key, so none is typed as its digit; on macOS, which has no
35/// `NumLock`, all of it or none as the Clear key last switched it; elsewhere
36/// `NumLock` decides, unless the keybinds file says `numpad = "always"`.
37///
38/// `clear` is how Clear last switched the numpad, on a Mac; `None`
39/// elsewhere.
40pub(super) fn numpad_mode(waiting: bool, clear: Option<bool>, always: bool) -> NumpadCaptureMode {
41    match (waiting, clear) {
42        (true, _) | (false, Some(true)) => NumpadCaptureMode::Always,
43        (false, Some(false)) => NumpadCaptureMode::Off,
44        (false, None) if always => NumpadCaptureMode::Always,
45        (false, None) => NumpadCaptureMode::NumLockAware,
46    }
47}

49/// The macros the keys pressed in a play window this frame do, and whether
50/// it has the keyboard: those egui saw, taken from its input before anything
51/// draws so no widget sees a bound key, then, while it has the keyboard,
52/// those the fork `caught`. One on the command input while something else
53/// has the keyboard is left to that.
54pub(super) fn pressed(
55    ui: &egui::Ui,
56    keys: keys::Keys<'_>,
57    play: &mut Play,
58    caught: &mut Vec<keys::Chord>,
59) -> (Vec<Macro>, bool) {
60    let typing = play.typing(ui.ctx());
61    let aside = |made: &Macro| matches!(made, Macro::Act(action) if action.on_input() && !typing);
62    let mut pressed = ui.ctx().input_mut(|input| keys.take(input, aside));
63    // A hotbar button's own key, when no keybind holds it.
64    let bound = |chord: &keys::Chord| keys.does(chord).is_some();
65    let hotbars = play.hotbars_mut();
66    hotbars.keybinds_hold(bound);
67    pressed.extend(ui.ctx().input_mut(|input| hotbars.take(input, bound)));
68    let focused = ui.input(|input| input.focused);
69    if focused {
70        pressed.extend(
71            caught
72                .drain(..)
73                .filter_map(|chord| keys.does(&chord).cloned())
74                .filter(|made| !aside(made)),
75        );
76    }
77    (pressed, focused)
78}

91/// What a play window asks for when a key performs `action`: what its own
92/// button asks. An action on its command input or its windows it does
93/// itself (`Play::act`), and a line it sends is asked to be sent.
94pub(super) fn asked(context: &egui::Context, window: &mut Play, action: Action) -> Option<Asked> {
95    Some(match action {
96        Action::Set(set) => Asked::UseSet(set),
97        Action::Character(n) => Asked::Character(n),
98        _ => return window.act(context, action).map(Asked::Send),
99    })
100}

103    /// Read the keybinds file again, and say what it bound; each
104    /// character's own is read again off the window's thread, and replaces
105    /// what is held once it has been. A file a change of the player's waits
106    /// to be written to is taken as Hydra holds it.
107    pub(super) fn read_keys(&mut self) {
108        let every = self.keys_file.clone();
109        for path in self.mine.keys() {
110            crate::disk::Disk::hydras().look_again(&mine_key(path));
111        }
112        if !crate::store::pending(&every) {
113            self.key_texts.remove(&every);
114        }
115        if !self.key_texts.contains_key(&every) {
116            self.key_texts
117                .insert(every.clone(), std::fs::read_to_string(&every).ok());
118        }
119        let text = self.key_texts.get(&every).cloned().flatten();
120        self.take_keys(text.as_deref().unwrap_or_default());
121    }

123    /// Every character's keys, as `text` says, in force; what they bound
124    /// said.
125    fn take_keys(&mut self, text: &str) {
126        let file = &self.keys_file;
127        let (keys, problems) = crate::keys::Keybinds::read(text);
128        self.keys_said = std::iter::once(match keys.changed() {
129            0 => format!(
130                "{} keys bound, all Hydra's: change them here, and the changes are kept in {}.",
131                keys.len(),
132                file.display()
133            ),
134            changed => format!(
135                "{} keys bound, {changed} of them yours, from {}.",
136                keys.len(),
137                file.display()
138            ),
139        })
140        .chain(problems)
141        .collect();
142        self.clear_sends = keys.numpad_always();
143        self.keys = keys;
144        self.catch_again = true;
145    }

158    /// The character's keys file at `file`, as last read: off the window's
159    /// thread (`crate::disk`), so until it has been the character has none
160    /// of its own. A newer read replaces what is held, unless a change of
161    /// the player's waits to be written to it, and the keys caught natively
162    /// are worked out again.
163    pub(super) fn load_mine(&mut self, file: Option<&Path>) {
164        let Some(file) = file else {
165            return;
166        };
167        let read = crate::disk::Disk::hydras().kept(&mine_key(file), file, None, |path| {
168            Some(Arc::new(std::fs::read_to_string(path).ok()))
169        });
170        let Some((generation, text)) = read else {
171            return;
172        };
173        let newer = self
174            .mine_read
175            .get(file)
176            .is_none_or(|&read| generation > read);
177        if !newer || crate::store::pending(file) {
178            return;
179        }
180        let text = Option::clone(&text);
181        let mine = KeyFile::read(text.as_deref().unwrap_or_default(), Whose::Character);
182        self.mine.insert(file.to_owned(), mine);
183        self.mine_read.insert(file.to_owned(), generation);
184        self.key_texts.insert(file.to_owned(), text);
185        self.catch_again = true;
186    }

188    /// The character's keys file at `path` read now, when it has not been
189    /// read yet: an import writes into a character whose window is not
190    /// open. A read off the window's thread that comes after replaces it
191    /// as [`Self::load_mine`] says.
192    pub(super) fn mine_now(&mut self, path: &Path) {
193        if path == self.keys_file || self.key_texts.contains_key(path) {
194            return;
195        }
196        let text = std::fs::read_to_string(path).ok();
197        let mine = KeyFile::read(text.as_deref().unwrap_or_default(), Whose::Character);
198        self.mine.insert(path.to_owned(), mine);
199        self.key_texts.insert(path.to_owned(), text);
200        self.catch_again = true;
201    }

203    /// The text Hydra holds of the keys file at `path`, nothing for none;
204    /// a character's asked for when it has not been.
205    ///
206    /// # Errors
207    ///
208    /// It has not been read yet.
209    fn key_text(&mut self, path: &Path) -> Result<String, String> {
210        if path != self.keys_file {
211            self.load_mine(Some(path));
212        }
213        match self.key_texts.get(path) {
214            Some(text) => Ok(text.clone().unwrap_or_default()),
215            None => Err("Nothing was changed: the keys file is still being read.".to_owned()),
216        }
217    }

219    /// The keys file at `path` made `text`: in force at once, and `redo`
220    /// made to the file with the next batch of saves.
221    pub(super) fn keys_written(&mut self, path: &Path, text: String, redo: crate::store::Edit) {
222        if path == self.keys_file {
223            self.take_keys(&text);
224        } else {
225            let mine = KeyFile::read(&text, Whose::Character);
226            self.mine.insert(path.to_owned(), mine);
227            self.catch_again = true;
228        }
229        self.key_texts.insert(path.to_owned(), Some(text));
230        let dir = path.parent().unwrap_or(Path::new(".")).to_owned();
231        crate::store::edit(&dir, path, redo);
232    }

234    /// What a play window says of its keys: every character's, then how
235    /// many of `name`'s own there are and the set it uses, and what is
236    /// wrong in its file.
237    pub(super) fn keys_said(
238        &self,
239        mine: Option<&(KeyFile, Vec<String>)>,
240        name: &str,
241    ) -> Vec<String> {
242        let mut said = self.keys_said.clone();
243        if let Some((mine, problems)) = mine {
244            if mine.len() > 0 {
245                said.push(format!("{} of {name}'s own.", mine.len()));
246            }
247            if mine.chosen != 0 {
248                said.push(format!("Macro set {} in use, over set 0.", mine.chosen));
249            }
250            said.extend(problems.iter().cloned());
251        }
252        said
253    }

288    /// `made`, a send macro, on `seat` from `now`: its commands before any
289    /// wait sent at once, the rest kept until their waits are over.
290    pub(super) fn send_macro(&mut self, seat: &Seat, made: &Macro, now: Instant) {
291        let Ok(steps) = made.steps() else {
292            return;
293        };
294        let mut at = now;
295        for step in steps {
296            match step {
297                Step::Wait(wait) => at += wait,
298                Step::Line(line) if at <= now => seat.send(&line),
299                Step::Line(line) => self.later.push(Later {
300                    at,
301                    session: seat.id,
302                    line,
303                }),
304            }
305        }
306    }

308    /// The kept commands whose wait is over by `now`, sent in the order
309    /// they were kept, to their characters among `seats`; one whose
310    /// character has gone is dropped. How long until the next is due.
311    pub(super) fn send_due(&mut self, seats: &[Arc<Seat>], now: Instant) -> Option<Duration> {
312        let (due, kept): (Vec<Later>, Vec<Later>) =
313            self.later.drain(..).partition(|later| later.at <= now);
314        self.later = kept;
315        for later in due {
316            if let Some(seat) = seats.iter().find(|seat| seat.id == later.session) {
317                seat.send(&later.line);
318            }
319        }
320        self.later
321            .iter()
322            .map(|later| later.at.saturating_duration_since(now))
323            .min()
324    }

331    /// Tell the fork which keys to catch, when that has changed: the Keys
332    /// page began or stopped waiting for a key, the keybinds were read
333    /// again, Clear switched the numpad, or the keyboard went to another
334    /// play window or to none.
335    pub(super) fn catch(&mut self, frame: &mut eframe::Frame) {
336        let waiting = self.page_waiting();
337        if waiting != self.page_waits {
338            self.page_waits = waiting;
339            self.catch_again = true;
340        }
341        if !std::mem::take(&mut self.catch_again) {
342            return;
343        }
344        let mac = cfg!(target_os = "macos");
345        frame.set_numpad_capture_mode(numpad_mode(
346            waiting,
347            mac.then_some(self.clear_sends),
348            self.keys.numpad_always(),
349        ));
350        let (numpad, captured) = self.keys_to_catch(waiting, mac);
351        frame.set_numpad_capture_keys(numpad);
352        frame.set_key_capture(captured);
353    }

355    /// What the fork is to catch: the numpad keys by the fork's names, all
356    /// of them for `None`, and the keys egui has no name for. While the
357    /// Keys page is `waiting` for a key, every one; while a play window has
358    /// the keyboard, those its keys bind; while none has, none of them. On
359    /// a Mac (`mac`), always Clear, the numpad's switch.
360    pub(super) fn keys_to_catch(
361        &mut self,
362        waiting: bool,
363        mac: bool,
364    ) -> (Option<HashSet<String>>, HashSet<KeyCode>) {
365        let file = self.focused.as_ref().map(|(_, file)| file.clone());
366        self.load_mine(file.as_deref());
367        let mine = file
368            .as_ref()
369            .and_then(|file| self.mine.get(file))
370            .map(|(mine, _)| mine);
371        let keys = self.keys.of(mine);
372        let (numpad, mut captured) = match (waiting, self.focused.is_some()) {
373            (true, _) => (None, keys::capturable()),
374            (false, true) => (Some(keys.numpad_caught()), keys.key_capture()),
375            (false, false) => (Some(HashSet::new()), HashSet::new()),
376        };
377        if mac {
378            captured.insert(keys::NUM_LOCK);
379        }
380        (numpad, captured)
381    }

383    /// The fork's presses this frame, of the numpad (`numpad`) and of the
384    /// keys egui has no name for (`captured`): `NumLock` as the numpad's
385    /// show it; on a Mac (`mac`), Clear switching the numpad; the bound
386    /// ones, for the play window with the keyboard to do; or, while the
387    /// Keys page waits for a key, the first press, for it.
388    pub(super) fn caught_pressed(
389        &mut self,
390        numpad: &[eframe::NumpadKeyEvent],
391        captured: &[eframe::CapturedKeyEvent],
392        mac: bool,
393    ) {
394        if let Some(on) = numpad.iter().rev().find_map(|event| event.numlock_on) {
395            self.numlock = Some(on);
396        }
397        let clear = winit::keyboard::PhysicalKey::Code(keys::NUM_LOCK);
398        let (switches, captured): (Vec<_>, Vec<_>) = captured
399            .iter()
400            .partition(|event| event.physical_key == clear);
401        if mac && switches.iter().any(|event| event.pressed && !event.repeat) {
402            self.clear_sends = !self.clear_sends;
403            self.catch_again = true;
404        }
405        if self.page_waiting() {
406            self.caught.clear();
407            let numpad = numpad
408                .iter()
409                .filter(|event| !event.repeat)
410                .find_map(keys::numpad_chord);
411            let other = captured
412                .iter()
413                .filter(|event| !event.repeat)
414                .find_map(|event| keys::captured_chord(event));
415            self.caught_for_page = numpad.or(other).map(|chord| chord.written());
416            return;
417        }
418        self.caught_for_page = None;
419        let mine = self
420            .focused
421            .as_ref()
422            .and_then(|(_, file)| self.mine.get(file))
423            .map(|(mine, _)| mine);
424        let keys = self.keys.of(mine);
425        // A held key's repeats only do a send again (`keys::repeats`).
426        self.caught = numpad
427            .iter()
428            .filter_map(|event| Some((keys::numpad_caught(event)?, event.repeat)))
429            .chain(
430                captured
431                    .iter()
432                    .filter_map(|event| Some((keys::captured_chord(event)?, event.repeat))),
433            )
434            .filter(|(chord, repeat)| {
435                keys.does(chord)
436                    .is_some_and(|made| !repeat || keys::repeats(made))
437            })
438            .map(|(chord, _)| chord)
439            .collect();
440    }

469    fn key_change_made(
470        &mut self,
471        mine: Option<&(PathBuf, String)>,
472        change: &KeyChange,
473    ) -> Result<String, String> {
474        let every = self.keys_file.clone();
475        let file = |place_every: bool| match (place_every, mine) {
476            (true, _) => Ok((
477                (every.clone(), Whose::Every),
478                "Every character's keys".to_owned(),
479            )),
480            (false, Some((mine, name))) => {
481                Ok(((mine.clone(), Whose::Character), format!("{name}'s keys")))
482            }
483            (false, None) => Err("Pick a character to change its own keys.".to_owned()),
484        };
485        let place_every = match change {
486            KeyChange::Bind { place, .. }
487            | KeyChange::Unbind { place, .. }
488            | KeyChange::Restore { place, .. } => place.every,
489            KeyChange::NumpadAlways(_) => true,
490            KeyChange::Choose(_) => false,
491            KeyChange::Share {
492                key,
493                does,
494                set,
495                every: to_every,
496            } => {
497                let ((to, to_whose), said) = file(*to_every)?;
498                let ((from, from_whose), _) = file(!*to_every)?;
499                let place = |every| Place { set: *set, every };
500                let bound = KeyChange::Bind {
501                    key: key.clone(),
502                    does: does.clone(),
503                    was: None,
504                    place: place(*to_every),
505                };
506                let taken = KeyChange::Restore {
507                    key: key.clone(),
508                    place: place(!*to_every),
509                };
510                let (to_name, from_name) = (name_of(&to), name_of(&from));
511                let (old_to, old_from) = (self.key_text(&to)?, self.key_text(&from)?);
512                let (new_to, new_from, _) = keys::write::made_pair(
513                    ((&to_name, &old_to), to_whose, &bound),
514                    ((&from_name, &old_from), from_whose, &taken),
515                    (&self.keys, &said),
516                )?;
517                let keys = self.keys.clone();
518                for (path, text, name, whose, made) in [
519                    (to, new_to, to_name, to_whose, bound),
520                    (from, new_from, from_name, from_whose, taken),
521                ] {
522                    let (keys, said) = (keys.clone(), said.clone());
523                    let redo = Box::new(move |now: &str| {
524                        keys::write::made(now, (&name, whose), &made, (&keys, &said))
525                            .map(|(text, _)| text)
526                    });
527                    self.keys_written(&path, text, redo);
528                }
529                return Ok(format!(
530                    "{said}: {key} is {} now.",
531                    if *to_every {
532                        "every character's"
533                    } else {
534                        "only this character's"
535                    }
536                ));
537            }
538        };
539        let ((path, whose), said) = file(place_every)?;
540        let name = name_of(&path);
541        let old = self.key_text(&path)?;
542        let (text, done) = keys::write::made(&old, (&name, whose), change, (&self.keys, &said))?;
543        let (keys, made) = (self.keys.clone(), change.clone());
544        let redo = Box::new(move |now: &str| {
545            keys::write::made(now, (&name, whose), &made, (&keys, &said)).map(|(text, _)| text)
546        });
547        self.keys_written(&path, text, redo);
548        Ok(done)
549    }

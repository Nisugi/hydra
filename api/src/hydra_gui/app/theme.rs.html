12/// What `worn` holds while the editor's draft is worn: no theme's name.
13const PREVIEW: &str = "\u{1}preview";
14
15impl App {
16    /// Wear the theme Hydra's own settings choose, once, and again after a
17    /// change: the one chosen, or the light one while the computer is in
18    /// light mode and the setting follows it; the editor's draft over
19    /// either while it is worn. A theme that cannot be worn is said on
20    /// Hydra's Hub page, and Despana is worn.
21    pub(super) fn wear_theme(&mut self, context: &egui::Context) {
22        self.read_themes_again();
23        // Fonts given last frame are egui's now: worn again, in them.
24        let fonts = crate::fonts::loaded(context);
25        if fonts != self.worn_fonts {
26            self.worn_fonts = fonts;
27            self.worn = None;
28        }
29        if let Some(outfit) = self.preview.clone() {
30            if self.worn.as_deref() != Some(PREVIEW) {
31                self.put_on(context, &outfit);
32                self.worn = Some(PREVIEW.to_owned());
33            }
34            return;
35        }
36        let light =
37            self.own.follow_computer() && context.system_theme() == Some(egui::Theme::Light);
38        let wanted = if light {
39            self.own.light_theme()
40        } else {
41            self.own.theme()
42        };
43        if self.worn.as_deref() == Some(wanted) {
44            return;
45        }
46        let outfit = match self.themes.outfit(wanted) {
47            Ok(outfit) => {
48                self.theme_problem = None;
49                outfit
50            }
51            Err(why) => {
52                self.theme_problem = Some(format!("{wanted} is not worn: {why}"));
53                self.themes.outfit(Theme::DEFAULT).unwrap_or_default()
54            }
55        };
56        self.worn = Some(wanted.to_owned());
57        self.put_on(context, &outfit);
58    }

60    /// Wear `outfit`, its fonts given to egui first: those new to it are
61    /// drawn from the next frame, when it is worn again. Each character's
62    /// own look is worked out again over it.
63    fn put_on(&mut self, context: &egui::Context, outfit: &hydra_ui::theme::Outfit) {
64        self.hub_kind = outfit.kind.clone();
65        self.looks.wear_again();
66        self.give_fonts(context);
67        crate::theme::wear(context, outfit);
68        self.sessions.wore(outfit.palette.clone());
69    }

71    /// Give egui the fonts Hydra's theme and each character's own name.
72    pub(super) fn give_fonts(&mut self, context: &egui::Context) {
73        let kinds: Vec<&hydra_ui::theme::Type> = std::iter::once(&self.hub_kind)
74            .chain(self.looks.outfits().map(|outfit| &outfit.kind))
75            .collect();
76        if self.fonts.give(context, &kinds) {
77            context.request_repaint();
78            if !self.fonts.missing.is_empty() && self.preview.is_none() {
79                self.theme_problem = Some(format!(
80                    "Not found, drawn in egui's own: {}",
81                    self.fonts.missing.join(", ")
82                ));
83            }
84        }
85    }

87    /// The theme Hydra wears, by name, for a character's own accent to go
88    /// over: the one chosen while the editor's draft is worn.
89    pub(super) fn hydras_theme(&self) -> String {
90        self.worn
91            .clone()
92            .filter(|worn| worn != PREVIEW)
93            .unwrap_or_else(|| self.own.theme().to_owned())
94    }

96    /// Read the themes folder again when a file in it has come, gone or
97    /// changed, once a second at most, off the window's thread
98    /// (`crate::disk`): a theme `.theme import` wrote, or one edited by
99    /// hand, can be worn at once.
100    fn read_themes_again(&mut self) {
101        let folder = self.data.join("themes");
102        let key = format!("themes:{}", folder.display());
103        let disk = crate::disk::Disk::hydras();
104        let had = self.theme_files.clone();
105        // Shared, not copied: every theme would otherwise be copied out
106        // every frame.
107        let read = disk.kept(&key, &folder, None, move |folder| {
108            let files = theme_files(folder);
109            // Read whole only when what is there has changed.
110            let themes = (files != had).then(|| Themes::load(folder));
111            Some(std::sync::Arc::new((files, themes)))
112        });
113        // Listed again at each look, once a second: a file edited in place
114        // does not change the folder's own time.
115        disk.remake(&key);
116        let Some((_, read)) = read else {
117            return;
118        };
119        let (files, themes) = &*read;
120        if *files == self.theme_files || crate::store::pending(&folder) {
121            return;
122        }
123        let first = self.theme_files.is_empty() && self.worn.is_none();
124        self.theme_files.clone_from(files);
125        if !first && let Some(themes) = themes {
126            self.themes = themes.clone();
127            self.worn = None;
128        }
129    }

131    /// Open the theme editor, its draft starting from the theme worn.
132    pub(super) fn open_theme_editor(&mut self) {
133        if self.font_names.is_none() {
134            self.font_names = Some(self.fonts.names());
135        }
136        if self.themer.open() {
137            let worn = self
138                .worn
139                .clone()
140                .filter(|worn| worn != PREVIEW)
141                .unwrap_or_else(|| self.own.theme().to_owned());
142            self.themer.start_from(&self.themes, &worn);
143        }
144    }

146    /// The Editors window opened at the theme editor, its draft starting
147    /// from the theme named `name`, open as it was or not.
148    pub(super) fn edit_theme(&mut self, name: &str) {
149        self.editors.open(crate::editors_menu::Editor::Themes);
150        if self.font_names.is_none() {
151            self.font_names = Some(self.fonts.names());
152        }
153        self.themer.open();
154        self.themer.start_from(&self.themes, name);
155    }

157    /// Do what the theme editor asked: a theme saved, its draft worn or
158    /// put off, a Wrayth theme imported and opened in it.
159    pub(super) fn theme_asked(&mut self, ask: Asked) {
160        {
161            match ask {
162                Asked::Save(theme) => {
163                    let said = self.save_theme(&theme);
164                    self.themer.said(said);
165                }
166                Asked::Preview(outfit) => {
167                    self.preview = outfit;
168                    self.worn = None;
169                }
170                Asked::Import { file, name } => {
171                    let folder = self.data.join("themes");
172                    let name = (!name.is_empty()).then_some(name);
173                    let said = hydra_ui::theme::import_wrayth(&folder, &file, name.as_deref());
174                    let said = match said {
175                        Ok((made, said)) => {
176                            self.themes = Themes::load(&folder);
177                            // Opened in the editor, to see and change.
178                            self.themer.start_from(&self.themes, &made);
179                            said
180                        }
181                        Err(why) => why,
182                    };
183                    self.themer.said(said);
184                }
185            }
186        }
187    }

189    /// Write `theme` to the themes folder with the next batch of saves,
190    /// and in use at once; what was done, for the editor.
191    pub(super) fn save_theme(&mut self, theme: &Theme) -> String {
192        let folder = self.data.join("themes");
193        let stem = hydra_ui::theme::file_stem(&theme.name);
194        let path = folder.join(format!("{stem}.toml"));
195        // Two names may make one file name ("Storm 2.0", "Storm 2-0"): the
196        // other theme is not written over.
197        if let Some(there) = self
198            .themes
199            .names()
200            .into_iter()
201            .find(|name| hydra_ui::theme::file_stem(name) == stem && *name != theme.name)
202        {
203            return format!(
204                "{} already holds the theme {there}: give this one another name.",
205                path.display()
206            );
207        }
208        crate::store::whole(&folder, &path, theme.to_toml().into_bytes());
209        self.themes.add(theme.clone());
210        self.worn = None;
211        format!(
212            "{} written. Choose it on Settings, Hydra's Hub page.",
213            path.display()
214        )
215    }

218/// The `.toml` files in the themes folder `folder`, each with when it was
219/// last changed, in order.
220fn theme_files(folder: &Path) -> Vec<(PathBuf, Option<SystemTime>)> {
221    let mut files: Vec<(PathBuf, Option<SystemTime>)> = std::fs::read_dir(folder)
222        .map(|entries| {
223            entries
224                .filter_map(Result::ok)
225                .map(|entry| {
226                    let modified = entry.metadata().and_then(|m| m.modified()).ok();
227                    (entry.path(), modified)
228                })
229                .filter(|(path, _)| path.extension().is_some_and(|ext| ext == "toml"))
230                .collect()
231        })
232        .unwrap_or_default();
233    files.sort();
234    files
235}

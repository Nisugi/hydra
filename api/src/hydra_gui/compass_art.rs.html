1//! A compass set's pictures, for the Compass widget: the rose at rest, and
2//! over it a lit layer for each way out the room has.
3//!
4//! **A set is a folder**: `compass.toml`, `rose.png`, and a picture for each
5//! way its `ways` lists, named by the way -- `n`, `ne`, `e`, `se`, `s`, `sw`,
6//! `w`, `nw`, `up`, `down`, `out` -- each the rose's own size and drawn over
7//! all of it:
8//!
9//! ```toml
10//! layout = "hub"
11//! ways = ["n", "ne", "e", "se", "s", "sw", "w", "nw", "up", "down", "out"]
12//! ```
13//!
14//! A pack installs its set in the data folder's `global/images/compass/<set>/`
15//! (`frames/packs.rs`), and a player's own set may go there too. A way the
16//! file leaves out has no layer; a file that names no ways has all eleven.
17//!
18//! **Two layouts, read alike.** In a `hub` set the eight points meet at a
19//! centre disc that is `out`, with `up` and `down` in a column beside them;
20//! in a `column` set the points meet at a bare centre that is no way's, and
21//! `out` is the disc in the column between `up` and `down`. Each way's
22//! picture is where the art puts it, so Hydra draws and answers both the
23//! same, and does not read `layout` at all.
24//!
25//! **A click goes to the way whose layer is most solid under the pointer.**
26//! A point's glow reaches into its neighbours near the middle, so a layer a
27//! click only touches does not say which way it is on; each way's own part
28//! is solid where its glow is faint. Where no layer is solid enough, a
29//! click goes nowhere.
30//!
31//! Until a set's file and its rose are read, and for a set that cannot be,
32//! the compass is drawn in words (`widget/room.rs`).
33

41/// The ways a set may light, as its files are named.
42pub(crate) const WAYS: [&str; 11] = [
43    "n", "ne", "e", "se", "s", "sw", "w", "nw", "up", "down", "out",
44];

46/// How solid a layer must be under the pointer for the pointer to be on its
47/// way, out of 255: a glow's faint edge is not.
48const SOLID: u8 = 32;

50/// The key of the Compass widget's page row for its picture.
51pub(crate) const PICTURE: &str = "compass_picture";

53/// How a Compass widget is drawn, chosen on its page.
54#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
55pub(crate) struct CompassLook {
56    /// The compass set it is drawn as, by its folder's path; none, in
57    /// words.
58    #[serde(default, skip_serializing_if = "Option::is_none")]
59    pub(crate) picture: Option<String>,
60}

62/// Where packs install compass sets under the data folder `data`.
63pub(crate) fn pool(data: &Path) -> PathBuf {
64    data.join("global").join("images").join("compass")
65}
66
67/// Every compass set in `data`'s pool: each folder holding a
68/// `compass.toml`, in name order.
69pub(crate) fn sets(data: &Path) -> Vec<PathBuf> {
70    let mut found: Vec<PathBuf> = std::fs::read_dir(pool(data))
71        .map(|entries| {
72            entries
73                .filter_map(Result::ok)
74                .map(|entry| entry.path())
75                .filter(|path| path.join("compass.toml").is_file())
76                .collect()
77        })
78        .unwrap_or_default();
79    found.sort();
80    found
81}

83/// What a set is called: its folder's name, words apart.
84pub(crate) fn name(path: &Path) -> String {
85    path.file_name()
86        .map_or_else(String::new, |name| name.to_string_lossy().replace('_', " "))
87}

89/// A set's file, as read: the ways it has layers for.
90#[derive(Clone, Debug, PartialEq, Eq)]
91pub(crate) struct Set {
92    /// Its ways, each one [`WAYS`] names, in the file's order.
93    pub(crate) ways: Vec<String>,
94}
95
96impl Set {
97    /// Read from a set's `compass.toml`; a way it does not know is left
98    /// out, and no ways at all means every one. None when it is not TOML.
99    pub(crate) fn of_toml(text: &str) -> Option<Self> {
100        let table = text.parse::<toml::Table>().ok()?;
101        let named: Vec<String> = table
102            .get("ways")
103            .and_then(toml::Value::as_array)
104            .map(|ways| {
105                ways.iter()
106                    .filter_map(toml::Value::as_str)
107                    .map(str::to_ascii_lowercase)
108                    .filter(|way| WAYS.contains(&way.as_str()))
109                    .collect()
110            })
111            .unwrap_or_default();
112        let ways = if named.is_empty() {
113            WAYS.map(str::to_owned).to_vec()
114        } else {
115            named
116        };
117        Some(Self { ways })
118    }
119}

121/// The Compass widget's page rows: its picture, words or a set.
122pub(crate) fn rows(look: Option<&CompassLook>, sets: &[PathBuf]) -> Vec<Row> {
123    let picture = look.and_then(|look| look.picture.clone());
124    let mut choices = vec![(String::new(), "Words".to_owned())];
125    choices.extend(
126        sets.iter()
127            .map(|path| (path.display().to_string(), name(path))),
128    );
129    vec![Row {
130        key: PICTURE.to_owned(),
131        label: "Picture".to_owned(),
132        help: "Words, or a compass set: its rose, with each way out the room has lit. A set \
133               is a folder in global/images/compass in Hydra's data folder, where a pack \
134               puts it."
135            .to_owned(),
136        kind: RowKind::Choice(choices),
137        value: Value::Text(picture.clone().unwrap_or_default()),
138        here: picture.is_some(),
139        from: None,
140    }]
141}

143/// `key`, the Compass widget's page row, set to `to` in `look`; back to
144/// words with none.
145///
146/// # Errors
147///
148/// The key is not the compass's.
149pub(crate) fn set(look: &mut CompassLook, key: &str, to: Option<&str>) -> Result<(), String> {
150    if key != PICTURE {
151        return Err(format!("The compass has no `{key}`."));
152    }
153    look.picture = to
154        .map(str::trim)
155        .filter(|path| !path.is_empty())
156        .map(str::to_owned);
157    Ok(())
158}

160/// One way of a set, as drawn: what it is called, whether the room has it,
161/// and how solid its layer is.
162struct Layer {
163    word: &'static str,
164    open: bool,
165    solid: Option<std::sync::Arc<crate::pictures::Solid>>,
166}

168/// What a set's compass did as it was drawn.
169#[derive(Debug, PartialEq, Eq)]
170pub(crate) struct Drawn {
171    /// The way a click went, if one did.
172    pub(crate) went: Option<String>,
173}
174
175/// The compass for the character in `state`, drawn as the set in the
176/// folder `dir`: the rose, each way the room has lit over it. A click on a
177/// lit way goes that way when `goes`; never for another character's. None
178/// when the set or its rose cannot be drawn yet, for the words compass
179/// instead.
180pub(crate) fn compass(
181    ui: &mut egui::Ui,
182    state: Option<&GameState>,
183    dir: &str,
184    goes: bool,
185) -> Option<Drawn> {
186    let folder = Path::new(dir);
187    let toml = folder.join("compass.toml");
188    let set = crate::pictures::kept_watching(
189        ui.ctx(),
190        "compass-set",
191        dir,
192        &toml.to_string_lossy(),
193        |path| {
194            std::fs::read_to_string(path)
195                .ok()
196                .and_then(|text| Set::of_toml(&text))
197        },
198    )?;
199    let rose = crate::pictures::picture(ui.ctx(), &folder.join("rose.png").to_string_lossy())?;
200    let exits = state.and_then(|state| state.room.exits.as_ref());
201
202    // The rose as large as the widget lets it be, keeping its shape.
203    let room = ui.available_size();
204    let size = rose.size_vec2();
205    let scale = (room.x / size.x).min(room.y / size.y).max(0.0);
206    let sense = if goes { Sense::click() } else { Sense::hover() };
207    let (area, rose_response) = ui.allocate_exact_size(size * scale, sense);
208    if exits.is_none() {
209        rose_response.widget_info(|| {
210            egui::WidgetInfo::labeled(egui::WidgetType::Label, true, "Exits unknown")
211        });
212    }
213    let whole = Rect::from_min_max(pos2(0.0, 0.0), pos2(1.0, 1.0));
214    ui.painter().image(rose.id(), area, whole, Color32::WHITE);
215
216    let mut layers = Vec::with_capacity(set.ways.len());
217    for way in &set.ways {
218        let path = folder.join(format!("{way}.png"));
219        let path = path.to_string_lossy();
220        let word = word(way);
221        let open =
222            exits.is_some_and(|exits| exits.iter().any(|exit| hydra_ui::exit_word(exit) == word));
223        if open && let Some(lit) = crate::pictures::picture(ui.ctx(), &path) {
224            ui.painter().image(lit.id(), area, whole, Color32::WHITE);
225        }
226        layers.push(Layer {
227            word,
228            open,
229            solid: crate::pictures::solid(ui.ctx(), &path),
230        });
231    }
232    Some(Drawn {
233        went: clicked(ui, area, (&rose_response, &layers), goes),
234    })
235}

237/// What a way is called, as the game's exits are said in full.
238fn word(way: &str) -> &'static str {
239    match way {
240        "n" => "north",
241        "ne" => "northeast",
242        "e" => "east",
243        "se" => "southeast",
244        "s" => "south",
245        "sw" => "southwest",
246        "w" => "west",
247        "nw" => "northwest",
248        "up" => "up",
249        "down" => "down",
250        _ => "out",
251    }
252}

254/// The way whose layer is most solid at `at` on the compass drawn in
255/// `area`, and whether the room has it; none where no layer is solid
256/// enough.
257fn under(at: Pos2, area: Rect, layers: &[Layer]) -> Option<(&'static str, bool)> {
258    let fraction = (
259        (at.x - area.min.x) / area.width().max(1.0),
260        (at.y - area.min.y) / area.height().max(1.0),
261    );
262    layers
263        .iter()
264        .filter_map(|layer| {
265            let solid = layer.solid.as_ref()?.at(fraction);
266            (solid >= SOLID).then_some((solid, layer.word, layer.open))
267        })
268        .max_by_key(|(solid, ..)| *solid)
269        .map(|(_, word, open)| (word, open))
270}

272/// Each way a button over its layer, for a screen reader and a pointer
273/// alike, and the rose's own response: the lit way a click went, when
274/// `goes`. A click is the way under the pointer's, whichever button took
275/// it; one with no pointer, a screen reader's, is its own button's.
276fn clicked(
277    ui: &egui::Ui,
278    area: Rect,
279    (rose, layers): (&egui::Response, &[Layer]),
280    goes: bool,
281) -> Option<String> {
282    let pointer = ui.ctx().pointer_interact_pos();
283    let beneath = pointer.and_then(|at| under(at, area, layers));
284    let mut went = rose.clicked().then_some(beneath).flatten();
285    if rose.hovered()
286        && let Some((word, _)) = beneath
287    {
288        rose.clone().on_hover_text(word);
289    }
290    let sense = if goes { Sense::click() } else { Sense::hover() };
291    for layer in layers {
292        let Some(bounds) = layer.solid.as_ref().and_then(|solid| solid.bounds(SOLID)) else {
293            continue;
294        };
295        let rect = Rect::from_min_max(
296            area.min + bounds.min.to_vec2() * area.size(),
297            area.min + bounds.max.to_vec2() * area.size(),
298        );
299        let response = ui.interact(rect, rose.id.with(layer.word), sense);
300        let (word, open) = (layer.word, layer.open);
301        response.widget_info(|| egui::WidgetInfo::labeled(egui::WidgetType::Button, open, word));
302        if response.hovered()
303            && let Some((under, _)) = beneath
304        {
305            response.clone().on_hover_text(under);
306        }
307        if response.clicked() {
308            went = if pointer.is_some() {
309                beneath
310            } else {
311                Some((word, open))
312            };
313        }
314    }
315    went.filter(|(_, open)| *open && goes)
316        .map(|(word, _)| word.to_owned())
317}

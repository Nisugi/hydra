1//! The settings menu, the one every way in opens: the main settings button,
2//! and a widget's or window's right-click menu, which opens the same menu
3//! at that setting's page.
4//!
5//! The menu around the pages: the pages down the side -- Hydra's own,
6//! *General* and *Hub*, always; under them the character the picker names,
7//! its pages from the binary and its play window's own, *Drawers* and
8//! *Window*; then, under *Widgets*, the widgets' own pages by category, a
9//! custom window's under its name with its own *Window* page first
10//! (`crate::play::Shelved`). The page drawing is [`crate::menu::Menu`]'s.
11//!
12//! Settings chooses how things behave; the editors, which make things, are
13//! a window of their own (`crate::editors_menu`), and the windows of their
14//! own -- the Map, the Log, the Reports, Pair -- are the Menu's.
15//!
16//! A character is picked from the roster, running or not. Its pages come
17//! from the binary, which applies a change through the writer that
18//! character's settings file has; the window makes the changes to its own
19//! pages and to a widget's page itself.
20

45/// What the settings menu asks for.
46#[derive(Clone, Debug, PartialEq)]
47pub(crate) enum MenuAsked {
48    /// Of the binary: a character's pages, or a change to one of them.
49    Binary(HubRequest),
50    /// A change to one of Hydra's own pages, *General* or *Hub*.
51    Own {
52        /// The row's key.
53        key: String,
54        /// Its value as the menu writes it; `None` for its default.
55        to: Option<String>,
56    },
57    /// A change to the keys: every character's (`None`), or a character's
58    /// own, as the roster names it.
59    Key {
60        /// The character.
61        character: Option<String>,
62        /// The change.
63        change: KeyChange,
64    },
65    /// Of the *Controller* page: every character's file (`None`), or a
66    /// character's own, as the roster names it.
67    Controller {
68        character: Option<String>,
69        asked: crate::pad::PageAsked,
70    },
71    /// A change on a widget's page, made by the character's play window.
72    Widget {
73        /// The page's id.
74        page: String,
75        /// The row's key.
76        key: String,
77        /// Its value; `None` for the kind's own.
78        to: Option<String>,
79    },
80    /// The keybinds files read again, from the Keybinds page: changed
81    /// outside Hydra.
82    ReloadKeys,
83    /// The theme editor opened on the theme a character's *Theme* page
84    /// names; empty for Hydra's, which the character wears when it names
85    /// none.
86    EditTheme(String),
87}

89/// What the menu draws from this frame.
90#[derive(Clone, Copy, Debug)]
91pub(crate) struct MenuView<'a> {
92    /// Every character on the roster.
93    pub(crate) roster: &'a [RosterCard],
94    /// The pages the binary last gave, and whose.
95    pub(crate) pages: Option<(&'a str, &'a [Page])>,
96    /// What the binary answered the last request.
97    pub(crate) said: Option<&'a str>,
98    /// Hydra's own pages: *General* and *Hub*.
99    pub(crate) own: &'a [Page],
100    /// The widgets' own pages, when the character shown has a play window.
101    pub(crate) widgets: &'a [Page],
102    /// Where each of them stands in the list.
103    pub(crate) shelves: &'a [crate::play::Shelved],
104    /// The fonts a font's row offers.
105    pub(crate) fonts: &'a crate::fonts::Names,
106    /// The icons an icon's row offers: the shown character's icons folder.
107    pub(crate) icons: &'a crate::hotbar_editor::icon::Folder,
108    /// The shown character, whom a condition's row tries its tests on and
109    /// a game value's row picks from.
110    pub(crate) live: crate::highlight_editor::when::Live<'a, 'a>,
111    /// The names the model works a value out by, for the title bar's game
112    /// values (`GameState::value_names`).
113    pub(crate) values: &'a [String],
114}

132/// The settings menu's state, which outlives a frame.
133#[derive(Debug, Default)]
134pub(crate) struct SettingsMenu {
135    /// Whether its window is open.
136    pub(crate) open: bool,
137    /// Whose settings: a character as the roster names it, or `None` for
138    /// Hydra's own.
139    character: Option<String>,
140    /// The page showing, by its id.
141    page: Option<String>,
142    /// The character whose pages were last asked for, since it opened.
143    asked: Option<String>,
144    /// Its window is to be brought to the front, as it was opened.
145    raise: bool,
146    /// What the menu last said: what a change did, or why it was refused.
147    note: Option<String>,
148    /// The pages' rows as they are being typed.
149    rows: crate::menu::Menu,
150    /// The *Title bar* page's own drawing (`crate::titlebar_page`).
151    titlebar: crate::titlebar_page::TitleBarPage,
152}

154impl SettingsMenu {
155    /// Open on `character`'s settings (`GAME:Name`), or Hydra's own, at
156    /// `page` when one is named: a page named by the start of its id is the
157    /// first of its kind. A character's pages are asked for afresh.
158    pub(crate) fn open_at(&mut self, character: Option<String>, page: Option<&str>) {
159        self.open = true;
160        self.raise = true;
161        self.asked = None;
162        if self.character != character {
163            self.note = None;
164        }
165        self.character = character;
166        if let Some(page) = page {
167            self.page = Some(page.to_owned());
168        }
169    }

171    /// Whether its window is to be brought to the front: once, as it was
172    /// opened.
173    pub(crate) fn take_raise(&mut self) -> bool {
174        std::mem::take(&mut self.raise)
175    }

177    /// Whose settings are shown, as the roster names them; `None` for
178    /// Hydra's own.
179    pub(crate) fn character(&self) -> Option<&str> {
180        self.character.as_deref()
181    }
182
183    /// Say `said` at the top of the menu.
184    pub(crate) fn tell(&mut self, said: String) {
185        self.note = Some(said);

188    /// Draw the menu over `view`: what it asks, when anything.
189    pub(crate) fn show(&mut self, ui: &mut egui::Ui, view: &MenuView<'_>) -> Option<MenuAsked> {
190        self.picker(ui, view);
191        if let Some(said) = self.note.as_deref().or(view.said) {
192            ui.weak(said);
193        }
194        ui.separator();
195        let character = self.character.clone();
196        let mut asked = None;
197        if let Some(character) = &character
198            && self.asked.as_deref() != Some(character.as_str())
199        {
200            self.asked = Some(character.clone());
201            asked = Some(MenuAsked::Binary(HubRequest::Settings(character.clone())));
202        }
203        let theirs: &[Page] = match (&character, view.pages) {
204            (Some(character), Some((whose, pages))) if whose == character => pages,
205            _ => &[],
206        };
207        let groups = groups(view, character.as_deref(), theirs);
208        self.choose_page(&groups, character.is_some(), theirs);
209        ui.horizontal_top(|ui| {
210            self.contents(ui, &groups);
211            ui.separator();
212            ui.vertical(|ui| {
213                if let Some(page) = view
214                    .own
215                    .iter()
216                    .find(|page| self.page.as_deref() == Some(page.id.as_str()))
217                {
218                    if let Some(change) = self.rows.page(ui, page, view.pickers()) {
219                        self.note = None;
220                        asked = Some(MenuAsked::Own {
221                            key: change.key,
222                            to: change.to,
223                        });
224                    }
225                    return;
226                }
227                let Some(character) = &character else {
228                    return;
229                };
230                let page = theirs
231                    .iter()
232                    .chain(view.widgets)
233                    .find(|page| self.page.as_deref() == Some(page.id.as_str()));
234                let Some(page) = page else {
235                    ui.weak("Reading the settings...");
236                    return;
237                };
238                let mut edit = false;
239                let change = egui::ScrollArea::vertical()
240                    .id_salt("settings-page")
241                    .auto_shrink([false, false])
242                    .show(ui, |ui| {
243                        // The Window page's title bar is drawn its own way,
244                        // its sides as rows of pickers; its other rows under
245                        // it as any page's are.
246                        if page.id == crate::play::titlebar::PAGE {
247                            let bar = self.titlebar.show(ui, page, view.values);
248                            let rest = Page {
249                                rows: page
250                                    .rows
251                                    .iter()
252                                    .filter(|row| !crate::play::titlebar::is_bar_key(&row.key))
253                                    .cloned()
254                                    .collect(),
255                                ..page.clone()
256                            };
257                            bar.or(self.rows.rows(ui, &rest, view.pickers()))
258                        } else if page.id == THEME {
259                            // The editor offered under the rows, in the
260                            // page's scroll.
261                            let change = self.rows.headed(ui, page, view.pickers());
262                            ui.add_space(8.0);
263                            edit = ui
264                                .button("Edit this theme...")
265                                .on_hover_text(
266                                    "Open the theme editor on this theme, in the Editors \
267                                     window",
268                                )
269                                .clicked();
270                            change
271                        } else {
272                            self.rows.page(ui, page, view.pickers())
273                        }
274                    })
275                    .inner;
276                if edit {
277                    asked = Some(MenuAsked::EditTheme(theme_named(page)));
278                }
279                if let Some(mut change) = change {
280                    self.note = None;
281                    change.character.clone_from(character);
282                    asked = Some(if page.id.starts_with(crate::play::WIDGET_PAGE) {
283                        widget_asked(page, change)
284                    } else {
285                        MenuAsked::Binary(HubRequest::Change(change))
286                    });
287                }
288            });
289        });
290        asked
291    }

293    /// The page to show, when the one asked for is not among `groups`: one
294    /// named by the start of its id, the first of its kind; else, with a
295    /// `character` shown, its *General* page, or nothing while `theirs`, its
296    /// pages, are still being read; else Hydra's first.
297    fn choose_page(&mut self, groups: &[Group], character: bool, theirs: &[Page]) {
298        let pages = || groups.iter().flat_map(Group::every);
299        let listed = |id: &str| pages().any(|(page, _)| page == id);
300        if self.page.as_deref().is_some_and(listed) {
301            return;
302        }
303        let started = self.page.as_deref().and_then(|start| {
304            pages()
305                .find(|(page, _)| page.starts_with(start))
306                .map(|(page, _)| page.clone())
307        });
308        let default = if character {
309            (!theirs.is_empty()).then(|| {
310                theirs
311                    .iter()
312                    .find(|page| page.id == GENERAL)
313                    .or_else(|| theirs.first())
314                    .map(|page| page.id.clone())
315            })
316        } else {
317            Some(pages().next().map(|(page, _)| page.clone()))
318        };
319        // While its pages are read, the page asked for waits for them.
320        if started.is_none() && default.is_none() {
321            return;
322        }
323        self.page = started.or(default.flatten());
324    }

326    /// Whose settings, under Hydra's own: none, or a roster character's.
327    fn picker(&mut self, ui: &mut egui::Ui, view: &MenuView<'_>) {
328        let roster = view.roster;
329        ui.horizontal(|ui| {
330            let label = ui.label("Settings for");
331            let before = self.character.clone();
332            egui::ComboBox::from_id_salt("settings-character")
333                .selected_text(called(roster, self.character.as_deref()))
334                .show_ui(ui, |ui| {
335                    ui.selectable_value(&mut self.character, None, called(roster, None));
336                    for card in roster {
337                        let name = roster_name(card);
338                        let shown = called(roster, Some(&name));
339                        ui.selectable_value(&mut self.character, Some(name), shown);
340                    }
341                })
342                .response
343                .labelled_by(label.id);
344            if self.character != before {
345                self.note = None;
346            }
347        });
348    }

350    /// The pages' titles down the side, `groups` each under its heading,
351    /// one showing.
352    fn contents(&mut self, ui: &mut egui::Ui, groups: &[Group]) {
353        let showing = self.page.clone();
354        // Scrolled, so a long layout never asks for a taller menu.
355        egui::ScrollArea::vertical()
356            .id_salt("settings-contents")
357            // As wide as its list, as tall as the menu.
358            .auto_shrink([true, false])
359            .max_width(176.0)
360            .show(ui, |ui| {
361                // Down the side, whatever the row around it lays out.
362                ui.vertical(|ui| {
363                    ui.set_width(170.0);
364                    for group in groups {
365                        self.group(ui, group, &group.name, showing.as_deref());
366                    }
367                });
368            });
369    }
370
371    /// `group`, its pages and then the groups under it, open at first when
372    /// it says or when the page `showing` is in it; `path` its headings
373    /// down to it, which name its fold.
374    fn group(&mut self, ui: &mut egui::Ui, group: &Group, path: &str, showing: Option<&str>) {
375        let open = group.open || group.every().any(|(id, _)| showing == Some(id.as_str()));
376        egui::CollapsingHeader::new(&group.name)
377            .id_salt(("settings-group", path))
378            .default_open(open)
379            .show(ui, |ui| {
380                for (id, title) in &group.pages {
381                    self.entry(ui, id, title, showing);
382                }
383                for under in &group.groups {
384                    let path = format!("{path} › {}", under.name);
385                    self.group(ui, under, &path, showing);
386                }
387            });
388    }
389
390    /// One page's line in the contents.
391    fn entry(&mut self, ui: &mut egui::Ui, id: &str, title: &str, showing: Option<&str>) {
392        if ui.selectable_label(showing == Some(id), title).clicked() {
393            self.page = Some(id.to_owned());
394        }
395    }
396}

398/// What `change` on a widget's `page` asks of the widget: a text row's
399/// words as typed, for the widget keeps them so, where a character's
400/// settings file takes them quoted.
401fn widget_asked(page: &Page, change: Change) -> MenuAsked {
402    let words = page
403        .rows
404        .iter()
405        .any(|row| row.key == change.key && row.kind == RowKind::Text);
406    MenuAsked::Widget {
407        page: change.page,
408        key: change.key,
409        to: change.to.map(|to| {
410            if words {
411                crate::menu::unquoted(&to)
412            } else {
413                to
414            }
415        }),
416    }
417}

419/// A heading of the contents, the pages under it, and whether it stands
420/// open unless one of its pages shows.
421struct Group {
422    name: String,
423    pages: Vec<(String, String)>,
424    /// The headings under it, after its pages.
425    groups: Vec<Group>,
426    open: bool,
427}
428
429impl Group {
430    /// A heading `name` over `pages`, open at first when `open`.
431    fn of(name: &str, pages: Vec<(String, String)>, open: bool) -> Self {
432        Self {
433            name: name.to_owned(),
434            pages,
435            groups: Vec::new(),
436            open,
437        }
438    }
439
440    /// Every page under it, its own and its headings' in turn.
441    fn every(&self) -> Box<dyn Iterator<Item = &(String, String)> + '_> {
442        Box::new(
443            self.pages
444                .iter()
445                .chain(self.groups.iter().flat_map(Group::every)),
446        )
447    }
448
449    /// The heading `path` names under it, made folded where it is not.
450    fn under(&mut self, path: &[String]) -> &mut Group {
451        let Some((first, rest)) = path.split_first() else {
452            return self;
453        };
454        let at = self
455            .groups
456            .iter()
457            .position(|group| group.name == *first)
458            .unwrap_or_else(|| {
459                self.groups.push(Group::of(first, Vec::new(), false));
460                self.groups.len() - 1
461            });
462        self.groups[at].under(rest)
463    }
464}

466/// The id the binary gives a character's macros page, which the macro
467/// editor stands for in the Editors window.
468const MACROS: &str = "macros";
469
470/// The id the binary gives a character's *General* page, which Settings
471/// opens on for a character.
472const GENERAL: &str = "general";
473
474/// The id the binary gives a character's *Theme* page, under which
475/// Settings offers the theme editor.
476const THEME: &str = "theme";
477
478/// The theme a character's *Theme* `page` names: its *Theme* row's, empty
479/// for Hydra's.
480fn theme_named(page: &Page) -> String {
481    page.rows
482        .iter()
483        .find(|row| row.key == "theme")
484        .and_then(|row| match &row.value {
485            Value::Text(name) => Some(name.clone()),
486            _ => None,
487        })
488        .unwrap_or_default()
489}

491/// What the picker calls the character the roster names `name`; none,
492/// none.
493fn called(roster: &[RosterCard], name: Option<&str>) -> String {
494    let Some(name) = name else {
495        return "No character".to_owned();
496    };
497    roster
498        .iter()
499        .find(|card| roster_name(card) == name)
500        .map_or_else(
501            || name.to_owned(),
502            |card| {
503                format!(
504                    "{} ({})",
505                    card.character,
506                    crate::launch::game_name(&card.game)
507                )
508            },
509        )
510}

512/// The contents' groups: Hydra's own pages, open; then, with a `character`
513/// shown, its own, open -- `theirs`, the binary's, but the macros, and its
514/// play window's own, *Drawers* and *Window*, by title -- then *Widgets*,
515/// folded: each widget's page where the play window shelves it, under its
516/// category and its custom window's name.
517fn groups(view: &MenuView<'_>, character: Option<&str>, theirs: &[Page]) -> Vec<Group> {
518    let titled = |page: &Page| (page.id.clone(), page.title.clone());
519    let mut groups = vec![Group::of(
520        "Hydra",
521        view.own.iter().map(titled).collect(),
522        true,
523    )];
524    let Some(character) = character else {
525        return groups;
526    };
527    let window = |page: &&Page| {
528        page.id == crate::play::titlebar::PAGE || page.id == crate::play::DRAWERS_PAGE
529    };
530    let mut own: Vec<(String, String)> = theirs
531        .iter()
532        .filter(|page| page.id != MACROS)
533        .chain(view.widgets.iter().filter(window))
534        .map(titled)
535        .collect();
536    own.sort_by_key(|(_, title)| title.to_lowercase());
537    groups.push(Group::of(&called(view.roster, Some(character)), own, true));
538    // The widgets' pages, each where the play window shelves it; one it
539    // does not, under *Widgets* itself.
540    let mut widgets = Group::of("Widgets", Vec::new(), false);
541    for shelved in view.shelves {
542        if view.widgets.iter().any(|page| page.id == shelved.page) {
543            widgets
544                .under(&shelved.path)
545                .pages
546                .push((shelved.page.clone(), shelved.title.clone()));
547        }
548    }
549    for page in view.widgets.iter().filter(|page| !window(page)) {
550        if !view.shelves.iter().any(|shelved| shelved.page == page.id) {
551            widgets.pages.push(titled(page));
552        }
553    }
554    if widgets.every().next().is_some() {
555        groups.push(widgets);
556    }
557    groups
558}

566    /// Words typed on a widget's page reach the widget as typed, not in the
567    /// quotes a character's settings file takes them in; a number as it is.
568    #[test]
569    fn a_widgets_words_are_kept_as_typed() {
570        let row = |key: &str, kind| Row {
571            key: key.to_owned(),
572            label: String::new(),
573            help: String::new(),
574            kind,
575            value: Value::Unset,
576            here: false,
577            from: None,
578            needs: None,
579        };
580        let page = Page {
581            id: "widget:3".to_owned(),
582            title: "Right hand".to_owned(),
583            file: String::new(),
584            takes: String::new(),
585            problem: None,
586            rows: vec![
587                row("label", RowKind::Text),
588                row("size", RowKind::Whole { min: 1, max: 9 }),
589            ],
590        };
591        let asked = |key: &str, typed: &str| {
592            let kind = &page
593                .rows
594                .iter()
595                .find(|row| row.key == key)
596                .expect("its row")
597                .kind;
598            let change = Change {
599                character: String::new(),
600                page: page.id.clone(),
601                key: key.to_owned(),
602                to: Some(crate::menu::typed(kind, typed).expect("typed")),
603            };
604            let MenuAsked::Widget { to, .. } = widget_asked(&page, change) else {
605                panic!("a widget's change");
606            };
607            to
608        };
609        assert_eq!(asked("label", "[R]").as_deref(), Some("[R]"));
610        let quoting = r#"say "hi" \o/"#;
611        assert_eq!(asked("label", quoting).as_deref(), Some(quoting));
612        assert_eq!(asked("size", "4").as_deref(), Some("4"));
613    }

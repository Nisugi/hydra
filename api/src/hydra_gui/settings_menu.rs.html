1//! The settings menu, the one every way in opens: the main settings button,
2//! and a widget's or window's right-click menu, which opens the same menu
3//! at that setting's page.
4//!
5//! The menu around the pages: the picker of whose settings, the pages down
6//! the side, Hydra's own *Window* and *Keys* pages, a character's pages from
7//! the binary, and the widgets' own pages of the character's play window.
8//! The page drawing is [`crate::menu::Menu`]'s.
9//!
10//! **Every editor is a page here**, whoever's settings are shown: Triggers,
11//! Themes, Cards and Packs under *Editors*, and Hotbars among Hydra's or the
12//! character's own, its bars and everyone's. The window draws those pages
13//! itself ([`Outside`]), as it holds the editors. Under *Tools*, the web
14//! page's address is a page too, and the Map, the Survey and the
15//! character's Log are windows of their own, opened from here; the
16//! *Performance* switch is here as well.
17//!
18//! A character is picked from the roster, running or not. Its pages come
19//! from the binary, which applies a change through the writer that
20//! character's settings file has; the window makes the changes to its own
21//! pages, to the keys and to a widget's page itself.
22

40/// A page the window draws itself, not the menu: an editor it holds, or the
41/// web page's address.
42#[derive(Clone, Copy, Debug, PartialEq, Eq)]
43pub(crate) enum Outside {
44    /// The triggers' editor.
45    Triggers,
46    /// The theme editor: making themes.
47    Themes,
48    /// The hotbar editor, over the character's bars and everyone's.
49    Hotbars,
50    /// The cards' editor.
51    Cards,
52    /// The packs the studio's catalogue offers.
53    Packs,
54    /// The web page's address and its code.
55    Web,
56}

73            Outside::Cards => "cards",
74            Outside::Packs => "packs",
75            Outside::Web => "web",

82            Outside::Triggers => "Triggers",
83            Outside::Themes => "Themes",
84            Outside::Hotbars => "Hotbars",
85            Outside::Cards => "Cards",
86            Outside::Packs => "Packs",
87            Outside::Web => "Web page",
88        }

91    /// The page whose id is `id`, when it is one of these.
92    pub(crate) fn of(id: &str) -> Option<Self> {
93        [
94            Outside::Triggers,
95            Outside::Themes,
96            Outside::Hotbars,
97            Outside::Cards,
98            Outside::Packs,
99            Outside::Web,

101        .into_iter()
102        .find(|page| page.id() == id)

119pub(crate) enum MenuAsked {

121    Binary(HubRequest),
122    /// A change to Hydra's own *Window* page.
123    Own {
124        /// The row's key.
125        key: String,
126        /// Its value as the menu writes it; `None` for its default.
127        to: Option<String>,

129    /// A change to the keys: every character's (`None`), or a character's
130    /// own, as the roster names it.

132        /// The character.
133        character: Option<String>,
134        /// The change.
135        change: KeyChange,
136    },
137    /// Of the *Controller* page: every character's file (`None`), or a
138    /// character's own, as the roster names it.

140        character: Option<String>,
141        asked: crate::pad::PageAsked,
142    },
143    /// A change on a widget's page, made by the character's play window.
144    Widget {
145        /// The page's id.
146        page: String,
147        /// The row's key.

149        /// Its value; `None` for the kind's own.
150        to: Option<String>,
151    },
152    /// A window of its own opened: the Log, the shown character's.
153    Open(Tool),
154    /// The *Performance* switch flipped: each live card shows how quick its

306    /// Draw the menu over `view`, `outside` drawing a page the window draws
307    /// itself ([`Outside`]): what it asks, when anything.
308    pub(crate) fn show(
309        &mut self,
310        ui: &mut egui::Ui,
311        view: &MenuView<'_>,
312        outside: &mut dyn FnMut(&mut egui::Ui, Outside),
313    ) -> Option<MenuAsked> {
314        self.picker(ui, view);
315        if let Some(said) = self.note.as_deref().or(view.said) {
316            ui.weak(said);
317        }
318        ui.separator();
319        let Some(character) = self.character.clone() else {
320            return self.hydra(ui, view, outside);
321        };
322        let mut asked = None;
323        if self.asked.as_deref() != Some(character.as_str()) {
324            self.asked = Some(character.clone());
325            asked = Some(MenuAsked::Binary(HubRequest::Settings(character.clone())));
326        }
327        let theirs = match view.pages {
328            Some((whose, pages)) if whose == character => pages,
329            _ => &[],
330        };
331        let mut titles: Vec<(&str, &str)> = theirs
332            .iter()
333            .map(|page| (page.id.as_str(), page.title.as_str()))
334            .collect();
335        titles.push((KEYS, "Keys"));
336        titles.push((CONTROLLER, "Controller"));
337        titles.push((Outside::Hotbars.id(), Outside::Hotbars.title()));
338        titles.extend(
339            view.widgets
340                .iter()
341                .map(|page| (page.id.as_str(), page.title.as_str())),
342        );
343        titles.extend(everyones());
344        // Its own pages come first, so the menu waits for them, unless it
345        // was opened at its keys, a widget's page or the window's own.
346        let elsewhere = self.page.as_deref().is_some_and(|page| {
347            page == KEYS
348                || page == CONTROLLER
349                || page.starts_with(crate::play::WIDGET_PAGE)
350                || Outside::of(page).is_some()
351        });
352        if theirs.is_empty() && !elsewhere {
353            ui.weak("Reading the settings...");
354            return asked;
355        }
356        ui.horizontal_top(|ui| {
357            asked = asked.take().or(self.contents(ui, &titles, view));
358            ui.separator();
359            ui.vertical(|ui| {
360                if let Some(page) = self.page.as_deref().and_then(Outside::of) {
361                    outside(ui, page);
362                    return;
363                }
364                if self.page.as_deref() == Some(KEYS) {
365                    if let Some(change) = self.keys.show(ui, &view.keys, &mut self.note) {
366                        asked = Some(MenuAsked::Key {
367                            character: Some(character.clone()),
368                            change,
369                        });
370                    }
371                    return;
372                }
373                if self.page.as_deref() == Some(CONTROLLER) {
374                    if let Some(now) = self.controller.show(ui, &view.controller) {
375                        asked = Some(MenuAsked::Controller {
376                            character: Some(character.clone()),
377                            asked: now,
378                        });
379                    }
380                    return;
381                }
382                let page = theirs
383                    .iter()
384                    .chain(view.widgets)
385                    .find(|page| self.page.as_deref() == Some(page.id.as_str()));
386                let change = page.and_then(|page| {
387                    egui::ScrollArea::vertical()
388                        .id_salt("settings-page")
389                        .auto_shrink([false, false])
390                        .show(ui, |ui| {
391                            // The title bar's page is drawn its own way:
392                            // its sides as rows of pickers.
393                            if page.id == crate::play::titlebar::PAGE {
394                                self.titlebar.show(ui, page, view.values)
395                            } else {
396                                self.rows.page(ui, page, view.pickers())
397                            }
398                        })
399                        .inner
400                });
401                if let (Some(page), Some(mut change)) = (page, change) {
402                    self.note = None;
403                    change.character.clone_from(&character);
404                    asked = Some(if page.id.starts_with(crate::play::WIDGET_PAGE) {
405                        MenuAsked::Widget {
406                            page: change.page,
407                            key: change.key,
408                            to: change.to,
409                        }
410                    } else {
411                        MenuAsked::Binary(HubRequest::Change(change))
412                    });
413                }
414            });
415        });
416        asked
417    }

419    /// Whose settings: Hydra's own, or a roster character's.
420    fn picker(&mut self, ui: &mut egui::Ui, view: &MenuView<'_>) {
421        let called = |name: Option<&str>| {
422            let Some(name) = name else {
423                return "Hydra (every character)".to_owned();
424            };
425            view.roster
426                .iter()
427                .find(|card| roster_name(card) == name)
428                .map_or_else(
429                    || name.to_owned(),
430                    |card| {
431                        format!(
432                            "{} ({})",
433                            card.character,
434                            crate::launch::game_name(&card.game)
435                        )
436                    },
437                )
438        };
439        ui.horizontal(|ui| {
440            let label = ui.label("Settings for");
441            let before = self.character.clone();
442            egui::ComboBox::from_id_salt("settings-character")
443                .selected_text(called(self.character.as_deref()))
444                .show_ui(ui, |ui| {
445                    ui.selectable_value(&mut self.character, None, called(None));
446                    for card in view.roster {
447                        let name = roster_name(card);
448                        let shown = called(Some(&name));
449                        ui.selectable_value(&mut self.character, Some(name), shown);
450                    }
451                })
452                .response
453                .labelled_by(label.id);
454            if self.character != before {
455                self.note = None;
456            }
457        });
458    }

460    /// The pages' titles down the side, one showing: the one asked for, by
461    /// its id or the start of it, or the first; under them the *Tools*.
462    /// What a tool asked, when one did.
463    fn contents(
464        &mut self,
465        ui: &mut egui::Ui,
466        titles: &[(&str, &str)],
467        view: &MenuView<'_>,
468    ) -> Option<MenuAsked> {
469        // The web page's address is a page, though it stands among the
470        // tools.
471        if self
472            .page
473            .as_deref()
474            .is_none_or(|id| id != Outside::Web.id() && titles.iter().all(|(page, _)| *page != id))
475        {
476            let started = self.page.as_deref().and_then(|start| {
477                titles
478                    .iter()
479                    .find(|(page, _)| page.starts_with(start))
480                    .map(|(page, _)| *page)
481            });
482            self.page = started
483                .or_else(|| titles.first().map(|(id, _)| *id))
484                .map(str::to_owned);
485        }
486        // Grouped: the character's own pages, then its window's, then
487        // each custom window's widgets under the window's name, folded;
488        // and scrolled, so a long layout never asks for a taller menu.
489        let groups = grouped(titles);
490        let showing = self.page.clone();
491        let mut asked = None;
492        egui::ScrollArea::vertical()
493            .id_salt("settings-contents")
494            // As wide as its list, as tall as the menu.
495            .auto_shrink([true, false])
496            .max_width(176.0)
497            .show(ui, |ui| {
498                // Down the side, whatever the row around it lays out.
499                ui.vertical(|ui| {
500                    ui.set_width(170.0);
501                    for group in &groups {
502                        let Some(name) = &group.name else {
503                            for (id, title) in &group.pages {
504                                self.entry(ui, id, title, showing.as_deref());
505                            }
506                            continue;
507                        };
508                        let open = group
509                            .pages
510                            .iter()
511                            .any(|(id, _)| showing.as_deref() == Some(id.as_str()));
512                        egui::CollapsingHeader::new(name)
513                            .id_salt(("settings-group", name))
514                            .default_open(open || name == EDITORS)
515                            .show(ui, |ui| {
516                                for (id, title) in &group.pages {
517                                    self.entry(ui, id, title, showing.as_deref());
518                                }
519                            });
520                    }
521                    asked = self.tools(ui, view, showing.as_deref());
522                });
523            });
524        asked
525    }

527    /// The *Tools*: the web page's address, a page; the Map, the Survey and
528    /// the shown character's Log, windows of their own; the *Performance*
529    /// switch. What one asked, when one did.
530    fn tools(
531        &mut self,
532        ui: &mut egui::Ui,
533        view: &MenuView<'_>,
534        showing: Option<&str>,
535    ) -> Option<MenuAsked> {
536        let mut asked = None;
537        egui::CollapsingHeader::new("Tools")
538            .id_salt("settings-tools")
539            .default_open(true)
540            .show(ui, |ui| {
541                let web = Outside::Web;
542                self.entry(ui, web.id(), web.title(), showing);
543                let windows = [
544                    (
545                        Tool::Map,
546                        "Map",
547                        "The whole map, and where every character stands",
548                        true,
549                    ),
550                    (
551                        Tool::Survey,
552                        "Survey",
553                        "The map a survey is making, drawn as its rooms are walked",
554                        true,
555                    ),
556                    (
557                        Tool::Log,
558                        "Log",
559                        "What this character read, kept a file a day",
560                        view.playing,
561                    ),
562                ];
563                for (tool, name, hover, can) in windows {
564                    let opened = ui
565                        .add_enabled(can, egui::Button::selectable(false, format!("{name} ↗")))
566                        .on_hover_text(hover)
567                        .on_disabled_hover_text("Pick a character that is playing to see its Log.");
568                    if opened.clicked() {
569                        asked = Some(MenuAsked::Open(tool));
570                    }
571                }
572                let mut perf = view.perf;
573                if ui
574                    .checkbox(&mut perf, "Performance")
575                    .on_hover_text(
576                        "How quick each character's window is, stage by stage, on its hub \
577                         card; logged every 10 seconds while on, in the data folder's perf \
578                         folder",
579                    )
580                    .changed()
581                {
582                    asked = Some(MenuAsked::Performance(perf));
583                }
584            });
585        asked
586    }

595    /// Hydra's own pages: *Window*, drawn as any other page, *Keys* and
596    /// *Controller*, every character's, and everyone's hotbars; then the
597    /// editors, which `outside` draws.
598    fn hydra(
599        &mut self,
600        ui: &mut egui::Ui,
601        view: &MenuView<'_>,
602        outside: &mut dyn FnMut(&mut egui::Ui, Outside),
603    ) -> Option<MenuAsked> {
604        let mut titles: Vec<(&str, &str)> = view
605            .own
606            .iter()
607            .map(|page| (page.id.as_str(), page.title.as_str()))
608            .collect();
609        titles.push((KEYS, "Keys"));
610        titles.push((CONTROLLER, "Controller"));
611        titles.push((Outside::Hotbars.id(), Outside::Hotbars.title()));
612        titles.extend(everyones());
613        let mut asked = None;
614        ui.horizontal_top(|ui| {
615            asked = self.contents(ui, &titles, view);
616            ui.separator();
617            ui.vertical(|ui| {
618                if let Some(page) = self.page.as_deref().and_then(Outside::of) {
619                    outside(ui, page);
620                    return;
621                }
622                if self.page.as_deref() == Some(KEYS) {
623                    asked = self
624                        .keys
625                        .show(ui, &view.keys, &mut self.note)
626                        .map(|change| MenuAsked::Key {
627                            character: None,
628                            change,
629                        });
630                    return;
631                }
632                if self.page.as_deref() == Some(CONTROLLER) {
633                    asked = self.controller.show(ui, &view.controller).map(|now| {
634                        MenuAsked::Controller {
635                            character: None,
636                            asked: now,
637                        }
638                    });
639                    return;
640                }
641                let page = view
642                    .own
643                    .iter()
644                    .find(|page| self.page.as_deref() == Some(page.id.as_str()));
645                if let Some(page) = page
646                    && let Some(change) = self.rows.page(ui, page, view.pickers())
647                {
648                    self.note = None;
649                    asked = Some(MenuAsked::Own {
650                        key: change.key,
651                        to: change.to,
652                    });
653                }
654            });
655        });
656        asked
657    }

670/// The editors' pages, which every scope lists.
671fn everyones<'a>() -> impl Iterator<Item = (&'a str, &'a str)> {
672    Outside::EDITORS
673        .into_iter()
674        .map(|page| -> (&'a str, &'a str) { (page.id(), page.title()) })
675}

677/// `titles` grouped for the contents: the character's own pages first,
678/// unheaded; then each window's widgets under the window's name, read off
679/// a widget page's title, `Health (Vitals)`; widgets standing alone under
680/// *Widgets*; the window's own pages, the title bar, under *Window*; and
681/// the editors under *Editors*, last.
682fn grouped(titles: &[(&str, &str)]) -> Vec<Group> {
683    let mut own = Vec::new();
684    let mut window = Vec::new();
685    let mut alone = Vec::new();
686    let mut editors = Vec::new();
687    let mut windows: Vec<Group> = Vec::new();
688    for (id, title) in titles {
689        let (id, title) = ((*id).to_owned(), (*title).to_owned());
690        if Outside::of(&id).is_some_and(|page| Outside::EDITORS.contains(&page)) {
691            editors.push((id, title));
692        } else if !id.starts_with(crate::play::WIDGET_PAGE) {
693            own.push((id, title));
694        } else if id == crate::play::titlebar::PAGE {
695            window.push((id, title));
696        } else if let Some((name, holder)) = title
697            .rsplit_once(" (")
698            .and_then(|(name, rest)| rest.strip_suffix(')').map(|holder| (name, holder)))
699        {
700            let holder = holder.to_owned();
701            match windows
702                .iter_mut()
703                .find(|group| group.name.as_deref() == Some(&holder))
704            {
705                Some(group) => group.pages.push((id, name.to_owned())),
706                None => windows.push(Group {
707                    name: Some(holder),
708                    pages: vec![(id, name.to_owned())],
709                }),
710            }
711        } else {
712            alone.push((id, title));
713        }
714    }
715    let mut groups = vec![Group {
716        name: None,
717        pages: own,
718    }];
719    if !window.is_empty() {
720        groups.push(Group {
721            name: Some("Window".to_owned()),
722            pages: window,
723        });
724    }
725    if !alone.is_empty() {
726        groups.push(Group {
727            name: Some("Widgets".to_owned()),
728            pages: alone,
729        });
730    }
731    groups.extend(windows);
732    if !editors.is_empty() {
733        groups.push(Group {
734            name: Some(EDITORS.to_owned()),
735            pages: editors,
736        });
737    }
738    groups
739}

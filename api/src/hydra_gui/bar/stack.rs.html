1//! The vitals stack: a spirit orb in the centre, with health, mana and
2//! stamina rings round it, stacked outward, Hydra sizing and centring them
3//! in the square it is given. Fills and tracks only: no rims, no kits, no
4//! words inside, so the rings read at a glance; the words are said to a
5//! screen reader.
6//!
7//! Everything is measured in the orb's own units, its liquid `LIQUID`
8//! across: each ring's band `BAND` wide, with `GAP` between the orb's
9//! square and the first ring and between one ring and the next.
10
11use egui::{Color32, Rect, Vec2};
12
13use super::shape::{self, LIQUID as LIQUID_SHARE, Paint};
14use super::{Amount, Background, Fills, Track};
15
16/// The orb's liquid across, in the stack's units.
17const LIQUID: f32 = 904.0;
18/// A ring's band, across.
19const BAND: f32 = 160.0;
20/// Between the orb's square and the first ring, and between rings.
21const GAP: f32 = 40.0;
22/// How many rings round the orb.
23const RINGS: usize = 3;
24
25/// One of the stack's four: how full, and its colour.
26#[derive(Clone, Copy, Debug)]
27pub struct Part {
28    /// The vital, as the game gave it; none, and it draws empty.
29    pub amount: Option<Amount>,
30    /// Its fill's colour.
31    pub color: Color32,
32}
33
34/// The stack's art, shared by its rings: a fill strip, a strip background,
35/// and the track drawn with neither.
36#[derive(Clone, Copy, Debug)]
37pub struct Art {
38    /// The rings' fill strip, read so it repeats.
39    pub fill: Option<(egui::TextureId, Vec2)>,
40    /// The rings' track from a pack.
41    pub background: Option<Background>,
42    /// What the rings show under their fill with no background.
43    pub track: Track,
44    /// The orb's glass.
45    pub glass: egui::TextureId,
46}
47
48/// The stack as drawn: the spirit orb in the centre and, outward, the
49/// health, mana and stamina rings.
50pub struct Stack {
51    /// Spirit, the orb.
52    pub spirit: Part,
53    /// Health, mana and stamina, the rings from the orb out.
54    pub rings: [Part; RINGS],
55    /// What they draw with.
56    pub art: Art,
57}
58
59/// The stack's radius in its units: the orb's square to its edge, then
60/// each ring and the gap before it.
61#[allow(clippy::cast_precision_loss)]
62const fn reach() -> f32 {
63    LIQUID / LIQUID_SHARE / 2.0 + (GAP + BAND) * RINGS as f32
64}
65
66impl Stack {
67    /// The square the stack fills in `area`: the largest centred in it.
68    #[must_use]
69    pub fn square(area: Rect) -> Rect {
70        Rect::from_center_size(area.center(), Vec2::splat(area.width().min(area.height())))
71    }
72
73    /// Each ring's square and band, from the orb out, in `square`: the
74    /// band as a share of the ring's radius, which the ring painter takes.
75    fn rings_in(square: Rect) -> [(Rect, f32); RINGS] {
76        let unit = square.width() / 2.0 / reach();
77        let orb_edge = LIQUID / LIQUID_SHARE / 2.0;
78        let mut rings = [(Rect::NOTHING, 0.0); RINGS];
79        for (at, ring) in rings.iter_mut().enumerate() {
80            #[allow(clippy::cast_precision_loss)]
81            let outer = orb_edge + (GAP + BAND) * (at as f32 + 1.0);
82            *ring = (
83                Rect::from_center_size(square.center(), Vec2::splat(outer * 2.0 * unit)),
84                BAND / outer,
85            );
86        }
87        rings
88    }
89
90    /// The orb's square in `square`.
91    fn orb_in(square: Rect) -> Rect {
92        let unit = square.width() / 2.0 / reach();
93        Rect::from_center_size(square.center(), Vec2::splat(LIQUID / LIQUID_SHARE * unit))
94    }
95
96    /// Paint it in `area`, centred.
97    pub fn paint(&self, painter: &egui::Painter, area: Rect) {
98        let square = Self::square(area);
99        if !square.is_positive() {
100            return;
101        }
102        let trough = painter.ctx().global_style().visuals.extreme_bg_color;
103        let share = |part: &Part| {
104            part.amount
105                .map_or(0.0, |amount| shape::share(amount.percent))
106        };
107        let orb = Paint {
108            trough,
109            fill: self.spirit.color,
110            background: Some(Background::Whole(self.art.glass)),
111            track: Track::default(),
112            kit: None,
113            fill_image: None,
114            ring: 0.0,
115        };
116        shape::paint(
117            painter,
118            Fills::Orb,
119            Self::orb_in(square),
120            share(&self.spirit),
121            &orb,
122        );
123        for ((frame, band), part) in Self::rings_in(square).into_iter().zip(&self.rings) {
124            let ring = Paint {
125                trough,
126                fill: part.color,
127                background: self.art.background,
128                track: self.art.track,
129                kit: None,
130                fill_image: self.art.fill,
131                ring: band,
132            };
133            shape::paint(painter, Fills::Ring, frame, share(part), &ring);
134        }
135    }
136
137    /// What it shows, for a screen reader: each vital's percent.
138    #[must_use]
139    pub fn words(&self) -> String {
140        let say = |label: &str, part: &Part| match part.amount {
141            Some(amount) => format!("{label} {}%", amount.percent),
142            None => format!("{label} ?"),
143        };
144        [
145            say("Sp", &self.spirit),
146            say("HP", &self.rings[0]),
147            say("MP", &self.rings[1]),
148            say("SP", &self.rings[2]),
149        ]
150        .join(", ")
151    }
152}

154#[cfg(test)]
155mod tests {
156    use super::*;
157
158    /// The rings sit outside the orb with even gaps, each band the same
159    /// width, the outermost reaching the square's edge.
160    #[test]
161    fn the_rings_stack_outward_with_even_gaps_to_the_edge() {
162        let square = Rect::from_min_size(egui::pos2(0.0, 0.0), Vec2::splat(400.0));
163        let unit = 200.0 / reach();
164        let orb = Stack::orb_in(square);
165        assert!((orb.width() - LIQUID / LIQUID_SHARE * unit).abs() < 0.01);
166        let rings = Stack::rings_in(square);
167        let mut edge = orb.width() / 2.0;
168        for (frame, band) in rings {
169            let outer = frame.width() / 2.0;
170            let inner = outer * (1.0 - band);
171            assert!(
172                (inner - edge - GAP * unit).abs() < 0.01,
173                "a gap of {GAP} units"
174            );
175            assert!(
176                (outer - inner - BAND * unit).abs() < 0.01,
177                "a band of {BAND} units"
178            );
179            edge = outer;
180        }
181        assert!((edge - 200.0).abs() < 0.01, "to the square's edge");
182    }

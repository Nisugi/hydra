1//! Art laid over a bar, after its fill and before its text: a whole image
2//! stretched, or a frame, a nine-slice; or a kit ([`Kit`]), rails along a
3//! bar or round a ring with caps at a bar's ends, the fill kept to the
4//! channel between the rails.
5
6use egui::{Color32, Rect, Vec2};
7
8/// A picture of `size` pixels, as a texture.
9pub type Picture = (egui::TextureId, Vec2);
10
11/// A bar or ring overlay kit: looping rails, and a cap for each end of a
12/// bar, drawn at the scale the bar's thickness makes of the kit's height.
13/// The fill and its track keep to the channel: `rail` pixels in from each
14/// long side and, on a bar, `cap` pixels in from each end, at that scale.
15/// A cap picture may be wider than `cap`: its ornament reaches over the
16/// channel. A ring has no caps.
17#[derive(Clone, Copy, Debug, PartialEq)]
18pub struct Kit {
19    /// The rails, read so they repeat: along a bar, or round a ring with
20    /// their top rail outside.
21    pub rails: Picture,
22    /// The cap at a bar's left end (its foot, upright).
23    pub cap_left: Picture,
24    /// The cap at a bar's right end (its head, upright).
25    pub cap_right: Picture,
26    /// How far in from each long side the channel starts, in the kit's
27    /// pixels.
28    pub rail: f32,
29    /// How far in from each end of a bar the channel starts, in the kit's
30    /// pixels.
31    pub cap: f32,
32}
33
34impl Kit {
35    /// What one of the kit's pixels is in points on a bar `thickness`
36    /// points across: the kit is drawn at its height.
37    #[must_use]
38    pub fn scale(&self, thickness: f32) -> f32 {
39        thickness / self.rails.1.y.max(1.0)
40    }
41}
42
43/// Art laid over a bar: a texture of `size` pixels, stretched whole, or
44/// framed -- a nine-slice, whose corners keep their size and whose edges
45/// stretch only along the bar, with the centre left clear so the fill shows
46/// through, as a skin's frame lies over a bar.
47#[derive(Clone, Copy, Debug, PartialEq)]
48pub struct Overlay {
49    texture: egui::TextureId,
50    size: Vec2,
51    /// Border widths in the texture's pixels, top, right, bottom, left; `None`
52    /// for the whole image stretched.
53    insets: Option<[f32; 4]>,
54}
55
56impl Overlay {
57    /// The whole image, stretched over the bar: a gloss, a texture.
58    #[must_use]
59    pub fn stretched(texture: egui::TextureId, size: Vec2) -> Self {
60        Self {
61            texture,
62            size,
63            insets: None,
64        }
65    }
66
67    /// A frame: `insets` pixels of each edge -- top, right, bottom, left --
68    /// are border, drawn at their own size; the middle is left clear.
69    #[must_use]
70    pub fn framed(texture: egui::TextureId, size: Vec2, insets: [f32; 4]) -> Self {
71        Self {
72            texture,
73            size,
74            insets: Some(insets),
75        }
76    }
77
78    pub(super) fn paint(&self, painter: &egui::Painter, bar: Rect) {
79        let whole = Rect::from_min_max(egui::pos2(0.0, 0.0), egui::pos2(1.0, 1.0));
80        let patches = match self.insets {
81            None => vec![(bar, whole)],
82            Some(insets) => nine_slice(self.size, insets, bar),
83        };
84        for (dest, uv) in patches {
85            painter.image(self.texture, dest, uv, Color32::WHITE);
86        }
87    }
88}
89
90/// The eight border patches of a nine-slice over `rect`, each as a
91/// destination and the part of the texture it shows. Insets too
92/// wide for `rect` shrink in proportion, so opposite borders never overlap.
93pub(super) fn nine_slice(texture: Vec2, insets: [f32; 4], rect: Rect) -> Vec<(Rect, Rect)> {
94    if texture.x <= 0.0 || texture.y <= 0.0 || !rect.is_positive() {
95        return Vec::new();
96    }
97    let [top, right, bottom, left] = insets.map(|inset| inset.max(0.0));
98    let shrink = |a: f32, b: f32, room: f32| {
99        if a + b > room {
100            let by = room / (a + b);
101            (a * by, b * by)
102        } else {
103            (a, b)
104        }
105    };
106    let (dt, db) = shrink(top, bottom, rect.height());
107    let (dl, dr) = shrink(left, right, rect.width());
108    let dx = [rect.min.x, rect.min.x + dl, rect.max.x - dr, rect.max.x];
109    let dy = [rect.min.y, rect.min.y + dt, rect.max.y - db, rect.max.y];
110    let ux = [
111        0.0,
112        (left / texture.x).min(1.0),
113        1.0 - (right / texture.x).min(1.0),
114        1.0,
115    ];
116    let uy = [
117        0.0,
118        (top / texture.y).min(1.0),
119        1.0 - (bottom / texture.y).min(1.0),
120        1.0,
121    ];
122    let mut patches = Vec::with_capacity(8);
123    for row in 0..3 {
124        for col in 0..3 {
125            if row == 1 && col == 1 {
126                continue;
127            }
128            let dest = Rect::from_min_max(
129                egui::pos2(dx[col], dy[row]),
130                egui::pos2(dx[col + 1], dy[row + 1]),
131            );
132            let uv = Rect::from_min_max(
133                egui::pos2(ux[col], uy[row]),
134                egui::pos2(ux[col + 1], uy[row + 1]),
135            );
136            if dest.width() > 0.0 && dest.height() > 0.0 && uv.width() > 0.0 && uv.height() > 0.0 {
137                patches.push((dest, uv));
138            }
139        }
140    }
141    patches
142}

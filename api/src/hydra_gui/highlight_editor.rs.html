1//! The highlights editor, Settings' *Triggers* page: every highlight, by
2//! category, on the left, each switched on or off where it is listed; the
3//! one chosen, or a new one *New* made ready for its kind, in a form on the
4//! right (`form.rs`), with a line to try it on.
5//!
6//! Nothing is written here: the editor asks the app ([`Asked`]), which
7//! makes it to the highlights at once and to the file with the next batch
8//! of saves.
9

18/// What the editor asks of the app.
19#[derive(Clone, Debug, PartialEq)]
20pub(crate) enum Asked {
21    /// Keep `rule` as `name`, in place of `was` when it was one.
22    Save {
23        /// The name it had; `None`, a new one.
24        was: Option<String>,
25        /// Its name.
26        name: String,
27        /// The highlight.
28        rule: Box<Rule>,
29    },
30    /// Take the highlight called this away.
31    Remove(String),
32    /// Switch the highlight or category called this on or off.
33    Switch(String, bool),
34}
35
36impl Asked {
37    /// What it asks made to `file`: what to say of it.
38    ///
39    /// # Errors
40    ///
41    /// Why it cannot be made.
42    pub(crate) fn made(&self, file: &mut HighlightFile) -> Result<String, String> {
43        match self {
44            Self::Save { was, name, rule } => file
45                .put(was.as_deref(), name, (**rule).clone())
46                .map(|()| format!("{name} saved.")),
47            Self::Remove(name) => file.remove(name).map(|name| format!("{name} taken away.")),
48            Self::Switch(name, on) => file.switch(name, *on),
49        }
50    }
51}

53/// The highlights editor.
54#[derive(Debug, Default)]
55pub(crate) struct Editor {
56    /// Whether it is showing.
57    pub(crate) open: bool,
58    /// What the list is narrowed to.
59    filter: String,
60    /// The highlight in the form: its name as kept, or `None` for a new one.
61    chosen: Option<String>,
62    /// The form.
63    draft: Draft,
64    /// What the editor has to say: why a save did nothing, or that it did.
65    notice: Option<String>,
66    /// The highlights picked in the list, by name, to take away.
67    picks: crate::picks::Picks<String>,
68}
69
70impl Editor {
71    /// What is said under the form.
72    pub(crate) fn said(&mut self, notice: String) {
73        self.notice = Some(notice);
74    }
75
76    /// The form's highlight was kept as `name`: it is the one chosen now.
77    pub(crate) fn kept(&mut self, name: &str) {
78        self.chosen = Some(name.to_owned());
79        name.clone_into(&mut self.draft.name);
80    }
81
82    /// The highlight called `name` was taken away: the form starts a new one.
83    pub(crate) fn removed(&mut self, name: &str) {
84        if self.chosen.as_deref() == Some(name) {
85            self.chosen = None;
86            self.draft = Draft::default();
87        }
88    }
89
90    /// Draw the editor into `ui` over `file`. What it asks of the app.
91    pub(crate) fn show(&mut self, ui: &mut egui::Ui, file: &HighlightFile) -> Vec<Asked> {
92        let mut asked = Vec::new();
93        egui::Panel::left(egui::Id::new("highlights-list"))
94            .resizable(true)
95            .default_size(300.0)
96            .show(ui, |ui| asked.extend(self.list(ui, file)));
97        egui::CentralPanel::default().show(ui, |ui| {
98            egui::ScrollArea::vertical()
99                .id_salt("highlight-form")
100                .show(ui, |ui| {
101                    let categories = file.category_names();
102                    asked.extend(self.draft.show(ui, self.chosen.as_deref(), &categories));
103                    if let Some(notice) = &self.notice {
104                        ui.separator();
105                        ui.label(notice.as_str());
106                    }
107                });
108        });
109        if asked
110            .iter()
111            .any(|ask| matches!(ask, Asked::Save { .. } | Asked::Remove(_)))
112        {
113            self.notice = None;
114        }
115        asked
116    }
117
118    /// The list: a filter, *New*, and each category with its highlights.
119    fn list(&mut self, ui: &mut egui::Ui, file: &HighlightFile) -> Vec<Asked> {
120        let mut asked = Vec::new();
121        ui.horizontal(|ui| {
122            ui.menu_button("New", |ui| {
123                crate::form::hint(ui, "What should it do?");
124                for kind in Kind::ALL {
125                    let chose = ui
126                        .horizontal(|ui| {
127                            let chose = ui.button(kind.name()).clicked();
128                            crate::form::hint(ui, kind.about());
129                            chose
130                        })
131                        .inner;
132                    if chose {
133                        self.chosen = None;
134                        self.picks.clear();
135                        self.draft = Draft::new(kind);
136                        self.notice = None;
137                        ui.close();
138                    }
139                }
140            });
141            ui.add(
142                egui::TextEdit::singleline(&mut self.filter)
143                    .hint_text("Find")
144                    .desired_width(f32::INFINITY),
145            );
146        });
147        ui.label(format!("{} trigger(s)", file.rules.len()));
148        let filter = self.filter.trim().to_lowercase();
149        // Every row the filter shows, category by category, as listed.
150        let order: Vec<String> = file
151            .category_names()
152            .iter()
153            .flat_map(|category| {
154                file.rules
155                    .iter()
156                    .filter(|(name, rule)| rule.category == *category && shows(name, rule, &filter))
157                    .map(|(name, _)| name.clone())
158            })
159            .collect();
160        self.picks.keep(&order);
161        asked.extend(
162            self.picks
163                .remove_bar(ui, "triggers")
164                .into_iter()
165                .map(Asked::Remove),
166        );
167        ui.separator();
168        egui::ScrollArea::vertical()
169            .id_salt("highlights-scroll")
170            .auto_shrink(false)
171            .show(ui, |ui| {
172                for category in file.category_names() {
173                    let rules: Vec<(&String, &Rule)> = file
174                        .rules
175                        .iter()
176                        .filter(|(_, rule)| rule.category == category)
177                        .filter(|(name, rule)| shows(name, rule, &filter))
178                        .collect();
179                    if rules.is_empty() && !filter.is_empty() {
180                        continue;
181                    }
182                    let mut on = file.category_on(&category);
183                    let header =
184                        egui::CollapsingHeader::new(format!("{category} ({})", rules.len()))
185                            .id_salt(("highlight-category", &category))
186                            .default_open(!filter.is_empty());
187                    ui.horizontal(|ui| {
188                        if ui
189                            .checkbox(&mut on, "")
190                            .on_hover_text("The whole category on or off")
191                            .changed()
192                        {
193                            asked.push(Asked::Switch(category.clone(), on));
194                        }
195                        ui.vertical(|ui| {
196                            header.show(ui, |ui| {
197                                for (name, rule) in rules {
198                                    asked.extend(self.row(ui, name, rule, &order));
199                                }
200                            });
201                        });
202                    });
203                }
204            });
205        asked
206    }
207
208    /// The highlight `name` switched `on` or off in the list: asked of the
209    /// app, and in the form too when it is the one there, so saving the form
210    /// does not switch it back.
211    fn switched(&mut self, name: &str, on: bool) -> Asked {
212        if self.chosen.as_deref() == Some(name) {
213            self.draft.set_off(!on);
214        }
215        Asked::Switch(name.to_owned(), on)
216    }
217
218    /// One highlight in the list: its switch, a swatch of its colours, and
219    /// its name, which a click puts in the form.
220    fn row(
221        &mut self,
222        ui: &mut egui::Ui,
223        name: &str,
224        rule: &Rule,
225        order: &[String],
226    ) -> Option<Asked> {
227        let mut asked = None;
228        ui.horizontal(|ui| {
229            let mut on = !rule.off;
230            if ui.checkbox(&mut on, "").changed() {
231                asked = Some(self.switched(name, on));
232            }
233            let mut sample = egui::RichText::new(" Aa ");
234            if let Some(color) = rule
235                .color
236                .as_deref()
237                .and_then(|hex| egui::Color32::from_hex(hex).ok())
238            {
239                sample = sample.color(color);
240            }
241            if let Some(ground) = rule
242                .background
243                .as_deref()
244                .and_then(|hex| egui::Color32::from_hex(hex).ok())
245            {
246                sample = sample.background_color(ground);
247            }
248            ui.label(sample);
249            let chosen = self.chosen.as_deref() == Some(name);
250            let key = name.to_owned();
251            if ui
252                .selectable_label(chosen || self.picks.has(&key), name)
253                .clicked()
254                && self.picks.click(ui, &key, order)
255                && !chosen
256            {
257                self.chosen = Some(key);
258                self.draft = Draft::of(name, rule);
259                self.notice = None;
260            }
261        });
262        asked
263    }
264}

266/// Whether the highlight `name` shows under `filter` (lower case): its
267/// name, category or words hold it.
268fn shows(name: &str, rule: &Rule, filter: &str) -> bool {
269    filter.is_empty()
270        || [
271            Some(name),
272            Some(rule.category.as_str()),
273            rule.text.as_deref(),
274            rule.regex.as_deref(),
275        ]
276        .into_iter()
277        .flatten()
278        .any(|said| said.to_lowercase().contains(filter))
279}

1//! The highlights editor: every highlight, by category, on the left, each
2//! switched on or off where it is listed; the one chosen, or a new one, in
3//! a form on the right; and a line to try it on.
4//!
5//! Nothing is written here: the editor asks the app ([`Asked`]), which
6//! makes it to the highlights at once and to the file with the next batch
7//! of saves.
8
9mod form;

12use hydra_ui::highlight::{HighlightFile, Rule};
13
14pub(crate) use form::Draft;

16/// What the editor asks of the app.
17#[derive(Clone, Debug, PartialEq)]
18pub(crate) enum Asked {
19    /// Keep `rule` as `name`, in place of `was` when it was one.
20    Save {
21        /// The name it had; `None`, a new one.
22        was: Option<String>,
23        /// Its name.
24        name: String,
25        /// The highlight.
26        rule: Box<Rule>,
27    },
28    /// Take the highlight called this away.
29    Remove(String),
30    /// Switch the highlight or category called this on or off.
31    Switch(String, bool),
32}
33
34impl Asked {
35    /// What it asks made to `file`: what to say of it.
36    ///
37    /// # Errors
38    ///
39    /// Why it cannot be made.
40    pub(crate) fn made(&self, file: &mut HighlightFile) -> Result<String, String> {
41        match self {
42            Self::Save { was, name, rule } => file
43                .put(was.as_deref(), name, (**rule).clone())
44                .map(|()| format!("{name} saved.")),
45            Self::Remove(name) => file.remove(name).map(|name| format!("{name} taken away.")),
46            Self::Switch(name, on) => file.switch(name, *on),
47        }
48    }
49}

51/// The highlights editor.
52#[derive(Debug, Default)]
53pub(crate) struct Editor {
54    /// Whether it is showing.
55    pub(crate) open: bool,
56    /// What the list is narrowed to.
57    filter: String,
58    /// The highlight in the form: its name as kept, or `None` for a new one.
59    chosen: Option<String>,
60    /// The form.
61    draft: Draft,
62    /// What the editor has to say: why a save did nothing, or that it did.
63    notice: Option<String>,
64}
65
66impl Editor {
67    /// What is said under the form.
68    pub(crate) fn said(&mut self, notice: String) {
69        self.notice = Some(notice);
70    }
71
72    /// The form's highlight was kept as `name`: it is the one chosen now.
73    pub(crate) fn kept(&mut self, name: &str) {
74        self.chosen = Some(name.to_owned());
75        name.clone_into(&mut self.draft.name);
76    }
77
78    /// The highlight called `name` was taken away: the form starts a new one.
79    pub(crate) fn removed(&mut self, name: &str) {
80        if self.chosen.as_deref() == Some(name) {
81            self.chosen = None;
82            self.draft = Draft::default();
83        }
84    }
85
86    /// Draw the editor into `ui` over `file`. What it asks of the app.
87    pub(crate) fn show(&mut self, ui: &mut egui::Ui, file: &HighlightFile) -> Vec<Asked> {
88        let mut asked = Vec::new();
89        egui::Panel::left(egui::Id::new("highlights-list"))
90            .resizable(true)
91            .default_size(300.0)
92            .show(ui, |ui| asked.extend(self.list(ui, file)));
93        egui::CentralPanel::default().show(ui, |ui| {
94            egui::ScrollArea::vertical()
95                .id_salt("highlight-form")
96                .show(ui, |ui| {
97                    let categories = file.category_names();
98                    asked.extend(self.draft.show(ui, self.chosen.as_deref(), &categories));
99                    if let Some(notice) = &self.notice {
100                        ui.separator();
101                        ui.label(notice.as_str());
102                    }
103                });
104        });
105        if asked
106            .iter()
107            .any(|ask| matches!(ask, Asked::Save { .. } | Asked::Remove(_)))
108        {
109            self.notice = None;
110        }
111        asked
112    }
113
114    /// The list: a filter, *New*, and each category with its highlights.
115    fn list(&mut self, ui: &mut egui::Ui, file: &HighlightFile) -> Vec<Asked> {
116        let mut asked = Vec::new();
117        ui.horizontal(|ui| {
118            if ui
119                .button("New")
120                .on_hover_text("A highlight of your own")
121                .clicked()
122            {
123                self.chosen = None;
124                self.draft = Draft::default();
125                self.notice = None;
126            }
127            ui.add(
128                egui::TextEdit::singleline(&mut self.filter)
129                    .hint_text("Find")
130                    .desired_width(f32::INFINITY),
131            );
132        });
133        ui.label(format!("{} highlight(s)", file.rules.len()));
134        ui.separator();
135        let filter = self.filter.trim().to_lowercase();
136        egui::ScrollArea::vertical()
137            .id_salt("highlights-scroll")
138            .auto_shrink(false)
139            .show(ui, |ui| {
140                for category in file.category_names() {
141                    let rules: Vec<(&String, &Rule)> = file
142                        .rules
143                        .iter()
144                        .filter(|(_, rule)| rule.category == category)
145                        .filter(|(name, rule)| shows(name, rule, &filter))
146                        .collect();
147                    if rules.is_empty() && !filter.is_empty() {
148                        continue;
149                    }
150                    let mut on = file.category_on(&category);
151                    let header =
152                        egui::CollapsingHeader::new(format!("{category} ({})", rules.len()))
153                            .id_salt(("highlight-category", &category))
154                            .default_open(!filter.is_empty());
155                    ui.horizontal(|ui| {
156                        if ui
157                            .checkbox(&mut on, "")
158                            .on_hover_text("The whole category on or off")
159                            .changed()
160                        {
161                            asked.push(Asked::Switch(category.clone(), on));
162                        }
163                        ui.vertical(|ui| {
164                            header.show(ui, |ui| {
165                                for (name, rule) in rules {
166                                    asked.extend(self.row(ui, name, rule));
167                                }
168                            });
169                        });
170                    });
171                }
172            });
173        asked
174    }
175
176    /// The highlight `name` switched `on` or off in the list: asked of the
177    /// app, and in the form too when it is the one there, so saving the form
178    /// does not switch it back.
179    fn switched(&mut self, name: &str, on: bool) -> Asked {
180        if self.chosen.as_deref() == Some(name) {
181            self.draft.set_off(!on);
182        }
183        Asked::Switch(name.to_owned(), on)
184    }
185
186    /// One highlight in the list: its switch, a swatch of its colours, and
187    /// its name, which a click puts in the form.
188    fn row(&mut self, ui: &mut egui::Ui, name: &str, rule: &Rule) -> Option<Asked> {
189        let mut asked = None;
190        ui.horizontal(|ui| {
191            let mut on = !rule.off;
192            if ui.checkbox(&mut on, "").changed() {
193                asked = Some(self.switched(name, on));
194            }
195            let mut sample = egui::RichText::new(" Aa ");
196            if let Some(color) = rule
197                .color
198                .as_deref()
199                .and_then(|hex| egui::Color32::from_hex(hex).ok())
200            {
201                sample = sample.color(color);
202            }
203            if let Some(ground) = rule
204                .background
205                .as_deref()
206                .and_then(|hex| egui::Color32::from_hex(hex).ok())
207            {
208                sample = sample.background_color(ground);
209            }
210            ui.label(sample);
211            let chosen = self.chosen.as_deref() == Some(name);
212            if ui.selectable_label(chosen, name).clicked() && !chosen {
213                self.chosen = Some(name.to_owned());
214                self.draft = Draft::of(name, rule);
215                self.notice = None;
216            }
217        });
218        asked
219    }
220}
221
222/// Whether the highlight `name` shows under `filter` (lower case): its
223/// name, category or words hold it.
224fn shows(name: &str, rule: &Rule, filter: &str) -> bool {
225    filter.is_empty()
226        || [
227            Some(name),
228            Some(rule.category.as_str()),
229            rule.text.as_deref(),
230            rule.regex.as_deref(),
231        ]
232        .into_iter()
233        .flatten()
234        .any(|said| said.to_lowercase().contains(filter))
235}

10/// Lay out one area: its own rooms, plus the neighbours a stranded room
11/// needs to stay attached (`cena_map_layout::areas::layout_rooms`), as the
12/// mapper's window and gate lay it out; the engine's scene, which a place
13/// is opened on ([`open_place`]). `None` when nothing in it is a place.
14pub(crate) fn lay_out(
15    map: &Map,
16    area: &str,
17    rooms: &[RoomId],
18    placeable: &HashSet<RoomId>,
19) -> Option<cena_map_layout::MapScene> {
20    sheet(
21        area,
22        &Map::from_rooms(cena_map_layout::areas::layout_rooms(rooms, map, placeable)).ok()?,
23    )
24}

26/// The places `area`'s sheet leaves off (`cena_map_layout::hidden`), each
27/// place's rooms: Rawknuckle's in the Hinterwilds is one.
28pub(crate) fn places(map: &Map, rooms: &[RoomId], placeable: &HashSet<RoomId>) -> Vec<Vec<RoomId>> {
29    let Ok(subset) = Map::from_rooms(cena_map_layout::areas::layout_rooms(rooms, map, placeable))
30    else {
31        return Vec::new();
32    };
33    let hidden = cena_map_layout::hidden::hidden_rooms(&subset);
34    cena_map_layout::hidden::place_rooms(&subset, &hidden)
35}

37/// A place entered, named `key`: opened on its area's sheet `area` where
38/// its dot is, at half the streets' scale (`cena_map_layout::open`), or, with no dot of `area`'s leading in, on a sheet of
39/// its own.
40pub(crate) fn open_place(
41    map: &Map,
42    key: &str,
43    area: &cena_map_layout::MapScene,
44    rooms: &[RoomId],
45) -> Option<MapScene> {
46    let rooms = rooms
47        .iter()
48        .filter_map(|&id| map.room(id).cloned())
49        .collect();
50    let alone = sheet(key, &Map::from_rooms(rooms).ok()?)?;
51    Some(
52        match cena_map_layout::open::open_place(
53            area,
54            &alone,
55            cena_map_layout::interior_shelf::TOWN_SCALE,
56        ) {
57            Some(opened) => convert(map, key, &opened.scene),
58            None => convert(map, key, &alone),
59        },
60    )
61}

63/// Lay out `subset` as the sheet named `name`; `None` when it has no room.
64fn sheet(name: &str, subset: &Map) -> Option<cena_map_layout::MapScene> {
65    if subset.rooms().is_empty() {
66        return None;
67    }
68    let layout = cena_map_layout::generate_layout(subset);
69    Some(cena_map_layout::build_scene(name, &layout, subset))
70}

72/// The engine's scene as the minimap's: rooms, lines with their bends, the
73/// way-in dots, and which building each room and line is in.
74pub(crate) fn convert(map: &Map, area: &str, scene: &cena_map_layout::MapScene) -> MapScene {
75    let sheet = &scene.sheet;
76    // Units are the streets and each building; the window only needs to
77    // know the buildings, numbered from 0.
78    let mut buildings: Vec<String> = Vec::new();
79    let mut building_of: HashMap<usize, usize> = HashMap::new();
80    for (unit, u) in scene.units.iter().enumerate() {
81        if u.kind == UnitKind::Building {
82            building_of.insert(unit, buildings.len());
83            buildings.push(u.name.clone());
84        }
85    }
86    #[allow(clippy::cast_precision_loss)]
87    let point = |x: i32, y: i32| (x as f32, y as f32);
88    let rooms = sheet
89        .rooms
90        .iter()
91        .map(|r| SceneRoom {
92            id: r.id.0,
93            cell: (r.cell.x, r.cell.y),
94            title: r.title.clone(),
95            building: building_of.get(&r.unit).copied(),
96            paths: map
97                .room(r.id)
98                .and_then(|room| room.paths.last().cloned())
99                .unwrap_or_default(),
100            marks: marks(map, r.id),
101        })
102        .collect();
103    let edges = sheet
104        .edges
105        .iter()
106        .map(|e| {
107            let mut path = vec![point(e.a.x, e.a.y)];
108            path.extend(e.via.iter().map(|p| (p.x, p.y)));
109            path.push(point(e.b.x, e.b.y));
110            SceneEdge {
111                a: e.a_room.0,
112                b: e.b_room.0,
113                kind: match e.kind {
114                    SceneEdgeKind::Directional => EdgeKind::Directional,
115                    SceneEdgeKind::Connector => EdgeKind::Connector,
116                    SceneEdgeKind::Stub => EdgeKind::Stub,
117                },
118                path,
119                label: e.label.clone(),
120                building: e.unit.and_then(|u| building_of.get(&u).copied()),
121            }
122        })
123        .collect();
124    let doors = sheet
125        .doors
126        .iter()
127        .map(|d| SceneDoor {
128            street: d.street.0,
129            inside: d.inside.0,
130            at: (d.at.x, d.at.y),
131            place: d.place.clone(),
132            named: d.named(),
133            marks: marks(map, d.inside),
134        })
135        .collect();
136    let labels = sheet
137        .labels
138        .iter()
139        .map(|l| SceneLabel {
140            text: l.text.clone(),
141            at: point(l.cell.x, l.cell.y),
142            building: l.unit.and_then(|u| building_of.get(&u).copied()),
143        })
144        .collect();
145    MapScene {
146        area: area.to_owned(),
147        rooms,
148        edges,
149        doors,
150        buildings,
151        labels,
152        min: (sheet.min.x, sheet.min.y),
153        max: (sheet.max.x, sheet.max.y),
154    }
155}

157/// What room `id` is: its map tags' kinds of place, each once.
158fn marks(map: &Map, id: RoomId) -> Vec<String> {
159    let mut marks: Vec<String> = map
160        .room(id)
161        .into_iter()
162        .flat_map(|room| &room.tags)
163        .filter_map(|tag| {
164            Some(match tag.as_str() {
165                "bank" => "bank",
166                "furrier" => "furrier",
167                "gemshop" => "gemshop",
168                "pawnshop" => "pawnshop",
169                "advguild" | "advguard" | "advguard2" | "advpickup" => "advguild",
170                "locksmith" => "locksmith",
171                "npchealer" | "npccleric" => "healer",
172                "herbalist" => "herbalist",
173                "alchemist" => "alchemist",
174                _ => return None,
175            })
176        })
177        .map(str::to_owned)
178        .collect();
179    marks.sort();
180    marks.dedup();
181    marks
182}

184#[cfg(test)]
185mod tests {
186    use super::*;
187
188    /// The map Hydra ships, decoded.
189    fn shipped() -> Map {
190        cena_map::binary::decode(cena_gs_map::GS_MAP).expect("the shipped map decodes")
191    }
192
193    /// A small area of the shipped map lays out whole: every room once,
194    /// every line between two drawn rooms along its own path.
195    #[test]
196    fn an_area_of_the_shipped_map_lays_out() {
197        let map = shipped();
198        let areas = cena_map_layout::areas::baked(&map);
199        let placeable = cena_map_layout::regions::placeable_rooms(&map);
200        let rooms = &areas["icemule-trace-ranger-guild"];
201        let scene = lay_out(&map, "icemule-trace-ranger-guild", rooms, &placeable)
202            .map(|engine| convert(&map, "icemule-trace-ranger-guild", &engine))
203            .expect("the guild is a place");
204
205        assert_eq!(scene.area, "icemule-trace-ranger-guild");
206        assert!(!scene.rooms.is_empty());
207        let ids: HashSet<u32> = scene.rooms.iter().map(|r| r.id).collect();
208        assert_eq!(ids.len(), scene.rooms.len(), "a room drawn twice");
209        for edge in &scene.edges {
210            assert!(ids.contains(&edge.a) && ids.contains(&edge.b));
211            assert!(edge.path.len() >= 2);
212            let start = scene.room(edge.a).expect("drawn").cell;
213            #[allow(clippy::cast_precision_loss)]
214            let start = (start.0 as f32, start.1 as f32);
215            assert_eq!(edge.path[0], start, "a line starts at its room");
216        }
217        for room in &scene.rooms {
218            assert!(room.cell.0 >= scene.min.0 && room.cell.0 <= scene.max.0);
219            assert!(room.cell.1 >= scene.min.1 && room.cell.1 <= scene.max.1);
220        }
221    }

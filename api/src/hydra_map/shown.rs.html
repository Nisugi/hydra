1//! The route each character's minimap draws: put here by travel
2//! (`.route2`, `hydra-travel`), read by the minimap, and kept until it is
3//! cleared or another takes its place. And the course each character's
4//! ocean chart draws, put by `.sail`.
5//!
6//! One board for the process, as the map is one ([`crate::shared`]): the
7//! commands that put a route and the windows that draw it are on different
8//! threads and never meet otherwise. A character is named as the roster
9//! names it, `GAME:Name`, without regard to case.
10
11use std::collections::HashMap;
12use std::sync::{Arc, Mutex, OnceLock};
13
14/// A route to draw: the rooms in order, the one it starts from first.
15#[derive(Debug, Clone, PartialEq, Eq)]
16pub struct ShownRoute {
17    /// The rooms, by the map's numbers: where it starts, each it passes,
18    /// and the goal last.
19    pub rooms: Vec<u32>,
20}
21
22impl ShownRoute {
23    /// The goal.
24    #[must_use]
25    pub fn goal(&self) -> Option<u32> {
26        self.rooms.last().copied()
27    }
28
29    /// What is left of it standing in `room`: the rooms from there on, or
30    /// the whole of it when `room` is not on it. The goal alone at the goal.
31    #[must_use]
32    pub fn from(&self, room: u32) -> &[u32] {
33        match self.rooms.iter().rposition(|on| *on == room) {
34            Some(at) => &self.rooms[at..],
35            None => &self.rooms,
36        }
37    }
38}

40/// A course to draw on the ocean's chart: the rooms in order, the one it
41/// starts from first, and the port it makes for.
42#[derive(Debug, Clone, PartialEq, Eq)]
43pub struct ShownCourse {
44    /// The ocean's rooms, by the game's numbers: where it starts, each it
45    /// passes, and the port's last.
46    pub rooms: Vec<i64>,
47    /// The port's name.
48    pub port: String,
49}
50
51impl ShownCourse {
52    /// What is left of it with the ship in `room`: the rooms from there
53    /// on, or the whole of it when `room` is not on it.
54    #[must_use]
55    pub fn from(&self, room: i64) -> &[i64] {
56        match self.rooms.iter().rposition(|on| *on == room) {
57            Some(at) => &self.rooms[at..],
58            None => &self.rooms,
59        }
60    }
61}

63type Courses = Mutex<HashMap<String, Arc<ShownCourse>>>;
64
65fn courses() -> &'static Courses {
66    static COURSES: OnceLock<Courses> = OnceLock::new();
67    COURSES.get_or_init(Courses::default)
68}
69
70/// Show `course` on `character`'s chart, or none.
71pub fn show_course(character: &str, course: Option<ShownCourse>) {
72    let Ok(mut courses) = courses().lock() else {
73        return;
74    };
75    let key = character.to_lowercase();
76    match course {
77        Some(course) => courses.insert(key, Arc::new(course)),
78        None => courses.remove(&key),
79    };
80}
81
82/// The course `character`'s chart draws, if any.
83#[must_use]
84pub fn shown_course(character: &str) -> Option<Arc<ShownCourse>> {
85    courses()
86        .lock()
87        .ok()?
88        .get(&character.to_lowercase())
89        .cloned()
90}

92type Board = Mutex<HashMap<String, Arc<ShownRoute>>>;
93
94fn board() -> &'static Board {
95    static BOARD: OnceLock<Board> = OnceLock::new();
96    BOARD.get_or_init(Board::default)
97}

99/// Show `route` on `character`'s minimap, or none.
100pub fn show_route(character: &str, route: Option<ShownRoute>) {
101    let Ok(mut board) = board().lock() else {
102        return;
103    };
104    let key = character.to_lowercase();
105    match route {
106        Some(route) => board.insert(key, Arc::new(route)),
107        None => board.remove(&key),
108    };
109}
110
111/// The route `character`'s minimap draws, if any.
112#[must_use]
113pub fn shown_route(character: &str) -> Option<Arc<ShownRoute>> {
114    board().lock().ok()?.get(&character.to_lowercase()).cloned()
115}

117#[cfg(test)]
118mod tests {
119    use super::*;
120
121    #[test]
122    fn a_route_is_shown_until_cleared_by_any_spelling() {
123        let route = ShownRoute {
124            rooms: vec![1, 2, 3],
125        };
126        show_route("GS3:Shown", Some(route.clone()));
127        assert_eq!(shown_route("gs3:shown").as_deref(), Some(&route));
128        show_route("gs3:SHOWN", None);
129        assert_eq!(shown_route("GS3:Shown"), None);
130    }
131
132    #[test]
133    fn what_is_left_is_from_where_you_stand() {
134        let route = ShownRoute {
135            rooms: vec![1, 2, 3],
136        };
137        assert_eq!(route.from(2), [2, 3]);
138        assert_eq!(route.from(3), [3]);
139        assert_eq!(route.from(9), [1, 2, 3]);
140        assert_eq!(route.goal(), Some(3));
141    }

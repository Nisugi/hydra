1//! One area of the map as the minimap draws it: rooms at cells, the lines
2//! between them by kind, and a dot for each way into a place not drawn.
3//! Plain data, no engine type and no egui, so the painter is a pure
4//! function of it.
5//!
6//! Cells are the layout's sheet cells: one step between two street rooms is
7//! several cells, so a building's rooms fit between them. A point is a cell
8//! position that need not be a cell's centre: a bend, or a dot beside a room.
9

14/// Where a character is on the map, as its minimap draws it.
15#[derive(Clone, Debug, PartialEq)]
16pub enum MinimapView {
17    /// Its area, and the room it is in.
18    Here {
19        /// The area, laid out.
20        scene: Arc<MapScene>,
21        /// The room's number.
22        room: u32,
23        /// The maps next door, each placed beside this one where a walk
24        /// joins them.
25        next_door: Vec<NextDoor>,
26        /// The route travel shows on it, if any ([`crate::show_route`]).
27        route: Option<Arc<crate::ShownRoute>>,
28    },
29    /// Aboard a ship: where it is on the ocean, and the deck.
30    Ocean {
31        /// Where the ship is.
32        sailing: crate::Sailing,
33        /// The way it last moved.
34        heading: Option<crate::Heading>,
35        /// Where the wheel is turned, as the game put it: `to the south`,
36        /// `toward a port`, `nowhere in particular`; `None` when not known.
37        wheel: Option<String>,
38        /// The deck plan, laid out, and the room of it the character is
39        /// in; `None` until it is laid out.
40        deck: Option<(Arc<MapScene>, u32)>,
41        /// The course travel shows on it, if any ([`crate::show_course`]).
42        course: Option<Arc<crate::ShownCourse>>,
43    },
44    /// Nothing to draw yet, and why, to say where the map would be.
45    Waiting(String),
46}

48/// A map next door: its sheet, and where its cells sit on this one's.
49#[derive(Clone, Debug, PartialEq)]
50pub struct NextDoor {
51    /// The map, laid out.
52    pub scene: Arc<MapScene>,
53    /// Added to each of its cells and points to draw it beside this one.
54    pub offset: (f32, f32),
55    /// How many maps out it is: 1 beside this one, 2 beside one of those.
56    pub ring: u8,
57}

59/// One laid-out area.
60#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
61pub struct MapScene {
62    /// The area's name, as baked into the map; `area@room` for a place
63    /// opened on its area.
64    pub area: String,
65    /// Every room drawn.
66    pub rooms: Vec<SceneRoom>,
67    /// The lines between them.
68    pub edges: Vec<SceneEdge>,
69    /// A dot for each way into a place left off the sheet.
70    pub doors: Vec<SceneDoor>,
71    /// The buildings drawn with the streets: a room's `building` indexes
72    /// here.
73    pub buildings: Vec<String>,
74    /// The buildings' names, where they are drawn.
75    #[serde(default)]
76    pub labels: Vec<SceneLabel>,
77    /// The least cell across and down any room reaches.
78    pub min: (i32, i32),
79    /// The greatest.
80    pub max: (i32, i32),
81    /// Where each room is among `rooms`, by its number, built the first time
82    /// one is asked for: the minimap asks for the rooms of a route every
83    /// frame.
84    #[serde(skip)]
85    pub by_id: ById,
86}

88/// Where each room is among a scene's rooms, by its number, once asked.
89/// Built from the rooms, so two scenes with the same rooms compare equal
90/// whether or not either has been asked yet.
91#[derive(Clone, Debug, Default)]
92pub struct ById(std::sync::OnceLock<std::collections::HashMap<u32, usize>>);
93
94impl PartialEq for ById {
95    fn eq(&self, _: &Self) -> bool {
96        true
97    }
98}

100/// One room on the sheet.
101#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
102pub struct SceneRoom {

104    pub id: u32,
105    /// Its cell.
106    pub cell: (i32, i32),
107    /// The room's title, `[Town Square, Center]`.
108    pub title: String,

117    #[serde(default)]
118    pub marks: Vec<String>,

121/// How a line is drawn.
122#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
123pub enum EdgeKind {
124    /// An exit with a compass direction: a solid line.
125    Directional,
126    /// An exit with no direction (`go door`, `climb rope`): a dashed line,
127    /// which may bend round what is in its way.
128    Connector,
129    /// A link too long to draw across the sheet: a short mark at each end,
130    /// each labelled with the room at the other.
131    Stub,
132}

134/// One line between two rooms.
135#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
136pub struct SceneEdge {
137    /// One end's room number.
138    pub a: u32,
139    /// The other's.
140    pub b: u32,
141    /// How it is drawn.
142    pub kind: EdgeKind,
143    /// The line from `a` to `b`, both ends and every bend, as points.

181impl MapScene {
182    /// The room with number `id`, if it is drawn here.
183    #[must_use]
184    pub fn room(&self, id: u32) -> Option<&SceneRoom> {
185        let by_id = self.by_id.0.get_or_init(|| {
186            self.rooms
187                .iter()
188                .enumerate()
189                .map(|(at, r)| (r.id, at))
190                .collect()
191        });
192        by_id.get(&id).and_then(|&at| self.rooms.get(at))
193    }
194}

196#[cfg(test)]
197mod tests {
198    use super::*;
199
200    /// A room is found by its number as a scan would find it, and none for
201    /// a number no room has.
202    #[test]
203    fn a_room_is_found_by_its_number() {
204        let room = |id, cell| SceneRoom {
205            id,
206            cell,
207            title: String::new(),
208            building: None,
209            paths: String::new(),
210            marks: Vec::new(),
211        };
212        let scene = MapScene {
213            rooms: vec![room(7, (0, 0)), room(9, (1, 0)), room(3, (2, 0))],
214            ..MapScene::default()
215        };
216        for id in [7, 9, 3, 4] {
217            assert_eq!(
218                scene.room(id).map(|r| r.cell),
219                scene.rooms.iter().find(|r| r.id == id).map(|r| r.cell),
220                "{id}"
221            );
222        }
223        assert_eq!(scene.room(9).map(|r| r.cell), Some((1, 0)));
224    }

1//! One area of the map as the minimap draws it: rooms at cells, the lines
2//! between them by kind, and a dot for each way into a place not drawn.
3//! Plain data, no engine type and no egui, so the painter is a pure
4//! function of it.
5//!
6//! Cells are the layout's sheet cells: one step between two street rooms is
7//! several cells, so a building's rooms fit between them. A point is a cell
8//! position that need not be a cell's centre: a bend, or a dot beside a room.
9
10use std::sync::Arc;
11
12use serde::{Deserialize, Serialize};

14/// Where a character is on the map, as its minimap draws it.
15#[derive(Clone, Debug, PartialEq)]
16pub enum MinimapView {
17    /// Its area, and the room it is in.
18    Here {
19        /// The area, laid out.
20        scene: Arc<MapScene>,
21        /// The room's number.
22        room: u32,
23        /// The maps next door, each placed beside this one where a walk
24        /// joins them.
25        next_door: Vec<NextDoor>,
26        /// The route travel shows on it, if any ([`crate::show_route`]).
27        route: Option<Arc<crate::ShownRoute>>,
28    },
29    /// Aboard a ship: where it is on the ocean, and the deck.
30    Ocean {
31        /// Where the ship is.
32        sailing: crate::Sailing,
33        /// The way it last moved.
34        heading: Option<crate::Heading>,
35        /// Where the wheel is turned, as the game put it: `to the south`,
36        /// `toward a port`, `nowhere in particular`; `None` when not known.
37        wheel: Option<String>,
38        /// The deck plan, laid out, and the room of it the character is
39        /// in; `None` until it is laid out.
40        deck: Option<(Arc<MapScene>, u32)>,
41        /// The course travel shows on it, if any ([`crate::show_course`]).
42        course: Option<Arc<crate::ShownCourse>>,
43    },
44    /// Nothing to draw yet, and why, to say where the map would be.
45    Waiting(String),
46}

48/// A map next door: its sheet, and where its cells sit on this one's.
49#[derive(Clone, Debug, PartialEq)]
50pub struct NextDoor {
51    /// The map, laid out.
52    pub scene: Arc<MapScene>,
53    /// Added to each of its cells and points to draw it beside this one.
54    pub offset: (f32, f32),
55    /// How many maps out it is: 1 beside this one, 2 beside one of those.
56    pub ring: u8,
57}

59/// One laid-out area.
60#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
61pub struct MapScene {
62    /// The area's name, as baked into the map; `area@room` for a place
63    /// opened on its area.
64    pub area: String,
65    /// Every room drawn.
66    pub rooms: Vec<SceneRoom>,
67    /// The lines between them.
68    pub edges: Vec<SceneEdge>,
69    /// A dot for each way into a place left off the sheet.
70    pub doors: Vec<SceneDoor>,
71    /// The buildings drawn with the streets: a room's `building` indexes
72    /// here.
73    pub buildings: Vec<String>,
74    /// The buildings' names, where they are drawn.
75    #[serde(default)]
76    pub labels: Vec<SceneLabel>,
77    /// The least cell across and down any room reaches.
78    pub min: (i32, i32),
79    /// The greatest.
80    pub max: (i32, i32),
81}
82
83/// One room on the sheet.
84#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
85pub struct SceneRoom {
86    /// The map's room number.
87    pub id: u32,
88    /// Its cell.
89    pub cell: (i32, i32),
90    /// The room's title, `[Town Square, Center]`.
91    pub title: String,
92    /// The building it is in, by index into [`MapScene::buildings`];
93    /// `None` for a street or the country between.
94    pub building: Option<usize>,
95    /// Its ways out as the game lists them (`Obvious paths: north, east`),
96    /// for the card a hover shows.
97    #[serde(default)]
98    pub paths: String,
99    /// What kind of place it is: `bank`, `healer`, `gemshop`...
100    #[serde(default)]
101    pub marks: Vec<String>,
102}

104/// How a line is drawn.
105#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
106pub enum EdgeKind {
107    /// An exit with a compass direction: a solid line.
108    Directional,
109    /// An exit with no direction (`go door`, `climb rope`): a dashed line,
110    /// which may bend round what is in its way.
111    Connector,
112    /// A link too long to draw across the sheet: a short mark at each end,
113    /// each labelled with the room at the other.
114    Stub,
115}
116
117/// One line between two rooms.
118#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
119pub struct SceneEdge {
120    /// One end's room number.
121    pub a: u32,
122    /// The other's.
123    pub b: u32,
124    /// How it is drawn.
125    pub kind: EdgeKind,
126    /// The line from `a` to `b`, both ends and every bend, as points.
127    pub path: Vec<(f32, f32)>,
128    /// What a connector says, when it is worth saying (`dock`, `gate`).
129    pub label: Option<String>,
130    /// The building the line is inside, when both ends are in one.
131    pub building: Option<usize>,
132}
133
134/// A way into a place not drawn: a dot beside the street room it is
135/// entered from.
136#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
137pub struct SceneDoor {
138    /// The street room.
139    pub street: u32,
140    /// The first room inside.
141    pub inside: u32,
142    /// Where the dot is, as a point.
143    pub at: (f32, f32),
144    /// The place behind it (`Angargreft`, `Trader's Bank`).
145    pub place: String,
146    /// Whether the place is big enough to name on the map.
147    pub named: bool,
148    /// What the place is, as its first room marks it.
149    #[serde(default)]
150    pub marks: Vec<String>,
151}
152
153/// A building's name, drawn beside it.
154#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
155pub struct SceneLabel {
156    /// The name.
157    pub text: String,
158    /// Where it is drawn, as a point: its building's top-left.
159    pub at: (f32, f32),
160    /// The building it names, by index into [`MapScene::buildings`].
161    pub building: Option<usize>,
162}

164impl MapScene {
165    /// The room with number `id`, if it is drawn here.
166    #[must_use]
167    pub fn room(&self, id: u32) -> Option<&SceneRoom> {
168        self.rooms.iter().find(|r| r.id == id)
169    }

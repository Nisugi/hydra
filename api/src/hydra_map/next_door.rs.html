1//! The maps next door: a neighbouring area's sheet, drawn beside the one a
2//! character is on.
3//!
4//! Each map keeps its own sheet; one next door is drawn beside it where a
5//! walk joins them: the two rooms of the first compass walk found one
6//! street step apart in its direction, pushed on along it off anything
7//! already drawn. A character crossing is then drawn on the other sheet,
8//! the first placed by the same walk the other way, so little moves.
9
10use std::collections::{BTreeMap, HashSet};
11use std::sync::{Arc, PoisonError};
12
13use cena_map::{Map, RoomId};
14
15use crate::atlas::Atlas;
16use crate::scene::{MapScene, NextDoor};

18/// One street step, in cells.
19const STEP: i32 = cena_map_layout::interior_shelf::TOWN_SCALE;
20
21/// How many steps a map is pushed off another before it is left there.
22const MOST_PUSHES: i32 = 12;
23
24/// A walk from one map into another: the room either side, and the step
25/// its direction takes.
26#[derive(Clone, Copy, Debug, PartialEq, Eq)]
27pub(crate) struct Join {
28    here: RoomId,
29    there: RoomId,
30    step: (i32, i32),
31}
32
33/// A map drawn: its name, its sheet, where its cells sit, and how many
34/// maps out it is.
35#[derive(Clone)]
36struct Placed {
37    name: String,
38    scene: Arc<MapScene>,
39    at: (i32, i32),
40    ring: u8,
41}
42
43impl Atlas {
44    /// The maps next door to the sheet `sheet` (an area, or a place opened
45    /// on one), `rings` deep: each placed as its first walk from a map
46    /// already placed says. A map not laid out yet is asked for and left
47    /// out until it is.
48    #[must_use]
49    pub fn next_door(&self, map: &Map, sheet: &str, rings: usize) -> Vec<NextDoor> {
50        let base = sheet.split('@').next().unwrap_or(sheet);
51        let Some(home) = self.laid(base) else {
52            return Vec::new();
53        };
54        let mut taken = cells(&home, (0, 0));
55        let mut placed = vec![Placed {
56            name: base.to_owned(),
57            scene: home,
58            at: (0, 0),
59            ring: 0,
60        }];
61        let mut ring = vec![0];
62        for out in 1..=rings {
63            let mut next = Vec::new();
64            for from in ring {
65                let Placed {
66                    name, scene, at, ..
67                } = placed[from].clone();
68                for (other, join) in self.joins(map, &name) {
69                    if placed.iter().any(|p| p.name == other) {
70                        continue;
71                    }
72                    let Some(there) = self.laid(&other) else {
73                        self.ask(&other);
74                        continue;
75                    };
76                    let Some(beside) = beside(&scene, at, &there, join, &taken) else {
77                        continue;
78                    };
79                    taken.extend(cells(&there, beside));
80                    next.push(placed.len());
81                    placed.push(Placed {
82                        name: other,
83                        scene: there,
84                        at: beside,
85                        ring: u8::try_from(out).unwrap_or(u8::MAX),
86                    });
87                }
88            }
89            ring = next;
90        }
91        placed
92            .into_iter()
93            .skip(1)
94            .map(|p| {
95                #[allow(clippy::cast_precision_loss)]
96                let offset = (p.at.0 as f32, p.at.1 as f32);
97                NextDoor {
98                    scene: p.scene,
99                    offset,
100                    ring: p.ring,
101                }
102            })
103            .collect()
104    }

106    /// Each map a walk leads into from `area`, by its first compass walk,
107    /// least rooms first; kept once worked out.
108    fn joins(&self, map: &Map, area: &str) -> Vec<(String, Join)> {
109        if let Some(known) = self
110            .joins
111            .read()
112            .unwrap_or_else(PoisonError::into_inner)
113            .get(area)
114        {
115            return known.clone();
116        }
117        let dirs = self
118            .dirs
119            .get_or_init(|| cena_map_layout::DirectionMap::build(map));
120        let mut first: BTreeMap<String, Join> = BTreeMap::new();
121        for &here in self.rooms.get(area).into_iter().flatten() {
122            for exit in map.room(here).into_iter().flat_map(|room| &room.exits) {
123                let Some(other) = self.area_of.get(&exit.to.0).filter(|o| *o != area) else {
124                    continue;
125                };
126                let Some(dir) = dirs.get(here, exit.to).filter(|d| d.is_compass()) else {
127                    continue;
128                };
129                let join = Join {
130                    here,
131                    there: exit.to,
132                    step: dir.offset(),
133                };
134                first
135                    .entry(other.clone())
136                    .and_modify(|kept| {
137                        if (join.here, join.there) < (kept.here, kept.there) {
138                            *kept = join;
139                        }
140                    })
141                    .or_insert(join);
142            }
143        }
144        let joins: Vec<(String, Join)> = first.into_iter().collect();
145        self.joins
146            .write()
147            .unwrap_or_else(PoisonError::into_inner)
148            .insert(area.to_owned(), joins.clone());
149        joins
150    }
151}

153/// Where `there` goes beside `here` (drawn at `offset`) by `join`: its room
154/// one step from `here`'s in the walk's direction, pushed on along it off
155/// the cells `taken`. `None` when either room is not drawn.
156fn beside(
157    here: &MapScene,
158    offset: (i32, i32),
159    there: &MapScene,
160    join: Join,
161    taken: &HashSet<(i32, i32)>,
162) -> Option<(i32, i32)> {
163    let from = here.room(join.here.0)?.cell;
164    let to = there.room(join.there.0)?.cell;
165    let (dx, dy) = (join.step.0 * STEP, join.step.1 * STEP);
166    let mut at = (from.0 + offset.0 + dx - to.0, from.1 + offset.1 + dy - to.1);
167    for _ in 0..MOST_PUSHES {
168        if !cells(there, at).iter().any(|c| near(taken, *c)) {
169            break;
170        }
171        at = (at.0 + dx, at.1 + dy);
172    }
173    Some(at)
174}
175
176/// The cells `scene`'s rooms are drawn in, moved by `at`.
177fn cells(scene: &MapScene, at: (i32, i32)) -> HashSet<(i32, i32)> {
178    scene
179        .rooms
180        .iter()
181        .map(|r| (r.cell.0 + at.0, r.cell.1 + at.1))
182        .collect()
183}
184
185/// Whether a cell of `taken` is `cell` or beside it.
186fn near(taken: &HashSet<(i32, i32)>, cell: (i32, i32)) -> bool {
187    (-1..=1).any(|dx| (-1..=1).any(|dy| taken.contains(&(cell.0 + dx, cell.1 + dy))))
188}

212    /// The two smallest maps a compass walk joins in the map Hydra ships
213    /// (found, not named: a map changes as areas are put on one, as Kraken
214    /// Manor and its Wine Cellar went onto Solhaven's): from the one, the
215    /// other is drawn beside it, no room of it on or beside one of the
216    /// first's.
217    #[test]
218    fn the_smallest_pair_is_drawn_side_by_side() {
219        let whole = cena_map::binary::decode(cena_gs_map::GS_MAP).expect("decodes");
220        let sheets = cena_map_layout::areas::baked(&whole);
221        let sheet_of: std::collections::HashMap<RoomId, &String> = sheets
222            .iter()
223            .flat_map(|(name, rooms)| rooms.iter().map(move |&id| (id, name)))
224            .collect();
225        let dirs = cena_map_layout::DirectionMap::build(&whole);
226        let mut smallest: Option<(usize, String, String)> = None;
227        for (name, rooms) in &sheets {
228            for &id in rooms {
229                for exit in whole.room(id).into_iter().flat_map(|room| &room.exits) {
230                    let Some(other) = sheet_of.get(&exit.to).filter(|o| **o != name) else {
231                        continue;
232                    };
233                    if !dirs
234                        .get(id, exit.to)
235                        .is_some_and(cena_map_layout::Dir::is_compass)
236                    {
237                        continue;
238                    }
239                    let pair = (
240                        rooms.len() + sheets[*other].len(),
241                        name.clone(),
242                        (*other).clone(),
243                    );
244                    if smallest.as_ref().is_none_or(|least| pair < *least) {
245                        smallest = Some(pair);
246                    }
247                }
248            }
249        }
250        let (_, here, there) = smallest.expect("two maps a walk joins");
251        let rooms = [&here, &there]
252            .iter()
253            .flat_map(|sheet| &sheets[*sheet])
254            .filter_map(|&id| whole.room(id).cloned())
255            .collect();
256        let map = Arc::new(Map::from_rooms(rooms).expect("a subset"));
257        let dir = std::env::temp_dir().join(format!("hydra-map-next-door-{}", std::process::id()));
258        let _ = std::fs::remove_dir_all(&dir);
259        let atlas = Atlas::start(&map, "5555555555555555", &dir);
260        let deadline = std::time::Instant::now() + std::time::Duration::from_mins(1);
261        let next = loop {
262            let next = atlas.next_door(&map, &here, 1);
263            if !next.is_empty() || std::time::Instant::now() > deadline {
264                break next;
265            }
266            std::thread::sleep(std::time::Duration::from_millis(50));
267        };
268        assert_eq!(next.len(), 1, "{there} next door to {here}");
269        assert_eq!(next[0].scene.area, there);
270        let first = cells(&atlas.laid(&here).expect("laid"), (0, 0));
271        #[allow(clippy::cast_possible_truncation)]
272        let at = (next[0].offset.0 as i32, next[0].offset.1 as i32);
273        assert!(
274            cells(&next[0].scene, at).iter().all(|c| !near(&first, *c)),
275            "no room of {there} on {here}'s"
276        );
277        let _ = std::fs::remove_dir_all(&dir);
278    }

280    /// East from room 1 to room 2: the map next door has room 2 one step
281    /// east of room 1; where that is taken, it is pushed on east.
282    #[test]
283    fn a_map_next_door_is_a_step_along_its_walk() {
284        let here = sheet("a", &[(1, (0, 0)), (3, (0, 4))]);
285        let there = sheet("b", &[(2, (10, 10)), (4, (14, 10))]);
286        let east = Join {
287            here: RoomId(1),
288            there: RoomId(2),
289            step: (1, 0),
290        };
291        let taken = cells(&here, (0, 0));
292        assert_eq!(beside(&here, (0, 0), &there, east, &taken), Some((-6, -10)));
293        let crowded: HashSet<(i32, i32)> = taken.into_iter().chain([(4, 0)]).collect();
294        assert_eq!(
295            beside(&here, (0, 0), &there, east, &crowded),
296            Some((-2, -10))
297        );
298    }

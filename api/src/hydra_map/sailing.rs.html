1//! Where on the ocean a character's ship is.
2//!
3//! The model keeps what the game said of the ship
4//! ([`hydra_model::state::ship`]); this turns it into a room of the
5//! [`Ocean`].
6//!
7//! # A ship starts from a port
8//!
9//! The only way onto the ocean is from a pier, and every pier belongs to
10//! one port, which is one room of the ocean. So the room a ship departs
11//! into is known, and each move the game tells of leads from it along the
12//! ocean's own links. The game's number for a room, when the player shows
13//! them, is a check on that reckoning and not something it needs.
14//!
15//! # The ladder
16//!
17//! The last room the game showed for the ship is the start: by its number;
18//! else, in port, by the pier's title, or by the pier the character stepped
19//! aboard from; else, off a port, by how the port looks from the water or
20//! by the pier the ship was boarded from. Moves
21//! told since are followed from there. A room shown with nothing to say
22//! which it is leaves the position as it was reckoned. The model keeps only
23//! the latest moves; when one since the start is no longer kept, the ship
24//! is lost rather than reckoned from the wrong room.
25

27use hydra_model::state::ship::{Ship, Waters, aboard};

29use crate::ocean::{Heading, Ocean};

31/// Where a character's ship is.
32#[derive(Clone, Copy, Debug, PartialEq, Eq)]
33pub enum Sailing {
34    /// The character is not aboard a ship.
35    Ashore,
36    /// The ship lies at a pier.
37    Docked {
38        /// Its port's room of the ocean, when the port is known.
39        port: Option<i64>,
40    },
41    /// The ship is on the ocean.
42    At {
43        /// The ocean room.
44        room: i64,
45        /// The game stated the room; `false` when it was reckoned from the
46        /// moves since.
47        stated: bool,
48    },
49    /// The ship is at sea and nothing says where.
50    Lost,
51}
52
53/// A start to reckon from: an ocean room or a pier, and the move it was
54/// true after.
55#[derive(Clone, Copy, Debug)]
56struct Start {
57    place: Place,
58    as_of: u32,
59    stated: bool,
60}
61
62#[derive(Clone, Copy, Debug)]
63enum Place {
64    Sea(i64),
65    Pier(Option<i64>),
66}
67
68/// One character's ship, followed across the ocean.
69#[derive(Clone, Copy, Debug, Default)]
70pub struct Navigator {
71    /// The port last known to hold the ship: its room of the ocean.
72    port: Option<i64>,
73    /// Where the ship was last reckoned to be, and after which move.
74    reckoned: Option<(i64, u32)>,
75    /// The last fix given.
76    last: Option<Sailing>,
77}
78
79impl Navigator {
80    /// Where the ship is, by what `state` has been told.
81    pub fn fix(&mut self, ocean: &Ocean, state: &GameState) -> Sailing {
82        // Between a room's number and its title the room has no name: the
83        // character is where it was a moment ago.
84        if state.room.title.is_none()
85            && let Some(last) = self.last
86        {
87            return last;
88        }
89        let fix = self.fix_now(ocean, state);
90        self.last = Some(fix);
91        fix
92    }
93
94    fn fix_now(&mut self, ocean: &Ocean, state: &GameState) -> Sailing {
95        let land = state
96            .room
97            .id
98            .as_deref()
99            .and_then(|id| id.parse::<i64>().ok());
100        if aboard(&state.room).is_none() {
101            // A pier stood on is the port a ship boarded here lies in.
102            if let Some(port) = land.and_then(|land| ocean.port_of_pier(land)) {
103                self.port = Some(port.uid);
104            }
105            self.reckoned = None;
106            return Sailing::Ashore;
107        }
108        let ship = &state.ship;
109        let Some(start) = self.start(ocean, ship) else {
110            return Sailing::Lost;
111        };
112        let from = match start.place {
113            Place::Pier(port) => {
114                if port.is_some() {
115                    self.port = port;
116                }
117                // Moves told after a pier was shown begin at its port.
118                if !ship.legs.iter().any(|leg| leg.ordinal > start.as_of) {
119                    self.reckoned = None;
120                    return Sailing::Docked { port: self.port };
121                }
122                let Some(port) = self.port else {
123                    return Sailing::Lost;
124                };
125                port
126            }
127            Place::Sea(room) => room,
128        };
129        let mut room = from;
130        let mut stated = start.stated;
131        let mut after = start.as_of;
132        for leg in ship.legs.iter().filter(|leg| leg.ordinal > start.as_of) {
133            // The model keeps only the latest moves: one between the start
134            // and the first kept is gone, and the room is not known.
135            if leg.ordinal != after.wrapping_add(1) {
136                self.reckoned = None;
137                return Sailing::Lost;
138            }
139            after = leg.ordinal;
140            let next =
141                Heading::named(&leg.heading).and_then(|heading| ocean.room(room)?.toward(heading));
142            let Some(next) = next else {
143                self.reckoned = None;
144                return Sailing::Lost;
145            };
146            room = next;
147            stated = false;
148        }
149        self.reckoned = Some((room, ship.moves));
150        Sailing::At { room, stated }
151    }
152
153    /// The latest thing that says where the ship is.
154    fn start(&self, ocean: &Ocean, ship: &Ship) -> Option<Start> {
155        let seen = ship.waters.as_ref().and_then(|waters| {
156            Some(Start {
157                place: self.place_of(ocean, waters)?,
158                as_of: waters.as_of,
159                stated: waters.uid.is_some(),
160            })
161        });
162        let berthed = ship.berth.as_ref().map(|berth| Start {
163            place: Place::Pier(
164                ocean
165                    .port_of_pier_titled(&berth.pier)
166                    .or_else(|| ocean.port_looking(berth.port.as_deref()?))
167                    .map(|port| port.uid),
168            ),
169            as_of: berth.as_of,
170            stated: true,
171        });
172        let reckoned = self.reckoned.map(|(room, as_of)| Start {
173            place: Place::Sea(room),
174            as_of,
175            stated: false,
176        });
177        // The pier stepped aboard from, when it is one: what a navigator
178        // that never saw the pier knows of the port.
179        let boarded = ship.boarded.and_then(|boarded| {
180            Some(Start {
181                place: Place::Pier(Some(ocean.port_of_pier(boarded.from)?.uid)),
182                as_of: boarded.as_of,
183                stated: true,
184            })
185        });
186        // The latest wins; at a tie, what the game showed over what was
187        // reckoned, and a pier come in to over the water before it; the
188        // pier stepped aboard from yields to all.
189        let latest = [boarded, reckoned, seen, berthed]
190            .into_iter()
191            .flatten()
192            .max_by_key(|start| start.as_of);
193        // Aboard with nothing said yet: the ship lies where it was boarded.
194        latest.or(Some(Start {
195            place: Place::Pier(None),
196            as_of: ship.moves,
197            stated: false,
198        }))
199    }
200
201    /// The place a room seen from the deck is, when anything says.
202    fn place_of(&self, ocean: &Ocean, waters: &Waters) -> Option<Place> {
203        if let Some(uid) = waters.uid {
204            if ocean.room(uid).is_some() {
205                return Some(Place::Sea(uid));
206            }
207            if let Some(port) = ocean.port_of_pier(uid) {
208                return Some(Place::Pier(Some(port.uid)));
209            }
210        }
211        if !waters.at_sea {
212            let port = ocean.port_of_pier_titled(&waters.title);
213            return Some(Place::Pier(port.map(|port| port.uid)));
214        }
215        // Open water beside a port is that port's room.
216        let looks = waters.port.as_deref()?;
217        ocean
218            .port_looking(looks)
219            .map(|port| port.uid)
220            .or(self.port)
221            .map(Place::Sea)
222    }
223}

242    /// Every fix, in order, as the session is read frame by frame: each
243    /// change once.
244    fn fixes(session: &str) -> Vec<Sailing> {
245        let ocean = ocean().as_ref().expect("the tables read");
246        let mut parser = Parser::new();
247        let mut state = GameState::default();
248        let mut navigator = Navigator::default();
249        let mut seen = Vec::new();
250        for frame in parser.push_bytes(session.as_bytes()) {
251            state.apply(&frame);
252            let fix = navigator.fix(ocean, &state);
253            if seen.last() != Some(&fix) {
254                seen.push(fix);
255            }
256        }
257        seen
258    }

268    /// The room numbers taken out, as the game prints rooms to a player who
269    /// does not show them.
270    fn unnumbered(session: &str) -> String {
271        let mut out = String::with_capacity(session.len());
272        let mut rest = session;
273        while let Some(at) = rest.find("] (") {
274            let (before, after) = rest.split_at(at + 1);
275            out.push_str(before);
276            let close = after.find(')').unwrap_or(0);
277            let number = after.get(2..close).unwrap_or_default();
278            if number
279                .trim_start_matches('-')
280                .bytes()
281                .all(|b| b.is_ascii_digit())
282            {
283                rest = after.get(close + 1..).unwrap_or_default();
284            } else {
285                rest = after;
286            }
287        }
288        out.push_str(rest);
289        out
290    }

364    /// More moves than the model keeps, since the last place the game said:
365    /// a navigator that followed them all knows where the ship is; a fresh
366    /// one, with the first moves gone, does not guess.
367    #[test]
368    fn moves_no_longer_kept_leave_the_ship_lost() {
369        let ocean = ocean().as_ref().expect("the tables read");
370        let mut parser = Parser::new();
371        let mut state = GameState::default();
372        let mut following = Navigator::default();
373        let mut session = unnumbered(&[BOARD, DECK_WALK, DEPART].concat());
374        // North twice, to 7116355, then back and forth: the 65th move ends
375        // a room south of it.
376        for at in 0..65 {
377            let heading = if at < 2 || at % 2 == 1 {
378                "north"
379            } else {
380                "south"
381            };
382            session.push_str("The mistwood sloop cuts through the ocean, heading ");
383            session.push_str(heading);
384            session.push_str(" at full speed.\r\n<prompt time=\"1\">&gt;</prompt>\r\n");
385        }
386        for frame in parser.push_bytes(session.as_bytes()) {
387            state.apply(&frame);
388            let _ = following.fix(ocean, &state);
389        }
390        assert_eq!(state.ship.moves, 65);
391        assert_eq!(following.fix(ocean, &state), at(7_116_361, false));
392        assert_eq!(Navigator::default().fix(ocean, &state), Sailing::Lost);
393    }

395    /// Without the ocean watched no room is shown after a move, and the
396    /// ship is brought in with no pier shown either.
397    #[test]
398    fn an_unwatched_voyage_is_reckoned_and_its_docking_known() {
399        let voyage = VOYAGE
400            .split_inclusive('\n')
401            .scan(false, |watching, line| {
402                // Drop what the watched stream carried, a line at a time.
403                let opens = line.contains("<pushStream id=\"familiar\"");
404                let shown = *watching || opens;
405                if opens {
406                    *watching = true;
407                }
408                if line.contains("<popStream/>") {
409                    *watching = false;
410                }
411                Some(if shown {
412                    line.split("<pushStream id=\"familiar\"")
413                        .next()
414                        .unwrap_or("")
415                } else {
416                    line
417                })
418            })
419            .collect::<String>();
420        assert!(!voyage.contains("familiar"));
421        let fixes = fixes(&[BOARD, DECK_WALK, DEPART, &voyage].concat());
422        assert_eq!(fixes.last(), Some(&DOCKED), "{fixes:#?}");
423        assert!(fixes.contains(&at(7_116_355, false)), "{fixes:#?}");
424        assert!(!fixes.contains(&Sailing::Lost), "{fixes:#?}");
425    }

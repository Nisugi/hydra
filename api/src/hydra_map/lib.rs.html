38/// The variable naming a map file to use in place of the one built in: a
39/// map being curated in the mapper.
40pub const MAP_ENV: &str = "HYDRA_MAP";
41
42/// The map, and the fingerprint of its bytes the layout cache is kept by.
43#[derive(Debug, Clone)]
44pub struct Loaded {
45    /// The map.
46    pub map: Arc<Map>,
47    /// A fingerprint of its bytes, in hex.
48    pub hash: String,
49}
50
51/// The map: the file [`MAP_ENV`] names, else the one built in.
52///
53/// # Errors
54///
55/// The file cannot be read, or is not a map.
56pub fn load() -> Result<Loaded, String> {
57    match std::env::var_os(MAP_ENV) {
58        Some(path) => {
59            let bytes = std::fs::read(&path)
60                .map_err(|why| format!("{} cannot be read: {why}", Path::new(&path).display()))?;
61            decode(&bytes)
62        }
63        None => decode(cena_gs_map::GS_MAP),
64    }
65}
66
67/// The map, loaded the first time anyone asks and kept for the process:
68/// [`load`]'s answer, shared by the minimaps and travel. Blocks while it
69/// loads, so ask it off any thread that draws.
70pub fn shared() -> &'static Result<Loaded, String> {
71    static SHARED: OnceLock<Result<Loaded, String>> = OnceLock::new();
72    SHARED.get_or_init(|| hydra_mem::part(hydra_mem::Part::Map, load))
73}
74
75/// The map in `bytes`.
76///
77/// # Errors
78///
79/// It is not a map.
80pub fn decode(bytes: &[u8]) -> Result<Loaded, String> {
81    let map =
82        cena_map::binary::decode(bytes).map_err(|why| format!("the map cannot be used: {why}"))?;
83    Ok(Loaded {
84        map: Arc::new(map),
85        hash: fingerprint(bytes),
86    })
87}
88
89/// FNV-1a over `bytes`, as 16 hex digits: enough to tell one map file from
90/// another for the cache, with no crate for it.
91fn fingerprint(bytes: &[u8]) -> String {
92    let mut hash: u64 = 0xcbf2_9ce4_8422_2325;
93    for byte in bytes {
94        hash ^= u64::from(*byte);
95        hash = hash.wrapping_mul(0x0100_0000_01b3);
96    }
97    format!("{hash:016x}")
98}

100#[cfg(test)]
101mod tests {

103
104    #[test]
105    fn the_built_in_map_decodes_and_a_bad_one_is_refused() {
106        let loaded = decode(cena_gs_map::GS_MAP).expect("the built-in map");
107        assert!(
108            loaded.map.rooms().len() > 30_000,
109            "{}",
110            loaded.map.rooms().len()
111        );
112        assert_eq!(loaded.hash.len(), 16);
113        assert!(decode(b"not a map").is_err());
114        assert_ne!(fingerprint(b"a"), fingerprint(b"b"));
115    }

1//! One page on the socket: paired first, then shown the hub or a character,
2//! and what it asks done.
3//!
4//! The page shows a character from its seat, as the window does: the same
5//! story, so what was said before the page came is there, and the
6//! highlights and triggers have already run, once, in the seat's feed. A
7//! line typed on the page is sent as one typed in the window; a link's
8//! command, or an object menu's entry, is checked as a line from the game
9//! is ([`hydra_ui::game_line`]).
10//!
11//! What changed is looked for every `TICK` while the page shows a
12//! character, and every `HUB_TICK` on the hub: each message is sent only
13//! when it differs from the last of its kind. The streams the page shows
14//! (`thoughts`, `familiar`) are followed as the story is, once it names them.
15//! Its workspace is kept with the character's ([`crate::workspace`]).
16//! Its highlights editor changes the highlights as the window's does
17//! ([`crate::highlights`]).
18

36/// How often a page showing a character is told what changed.
37const TICK: Duration = Duration::from_millis(100);
38/// How often a page showing the hub is.
39const HUB_TICK: Duration = Duration::from_millis(500);
40/// How long a page has to say its pairing.
41const HELLO_FOR: Duration = Duration::from_secs(10);
42
43/// The most a hello may be: a pairing and its wrapping, nothing like a
44/// kilobyte. Longer, it is refused before it is read.
45const HELLO_MOST: usize = 1024;
46
47/// How often a quiet connection is pinged, and how long one may say nothing
48/// -- no message, no pong -- before it is taken for gone: a phone that
49/// slept or left the network, which would otherwise be followed for ever.
50const PING_EVERY: Duration = Duration::from_secs(30);
51const SILENT_FOR: Duration = Duration::from_secs(90);
52/// The longest a write to the page, or its close, is waited on: a page
53/// that stops reading has its connection ended past it, so the loop that
54/// hears its silence and the web page's switch is never held.
55const WRITE_FOR: Duration = Duration::from_secs(10);
56/// How long an object's menu is waited for.
57const MENU_FOR: Duration = Duration::from_secs(10);
58/// How often a panel open on the page, the inventory or the tasks, is
59/// looked at for a change.
60const PANEL_EVERY: Duration = Duration::from_millis(300);
61/// How often a page's hotbars files are looked at for a change.
62const HOTBARS_EVERY: Duration = Duration::from_secs(1);
63
64/// How often the Window page's picture settings are read again, while a
65/// character is shown.
66const PICTURES_EVERY: Duration = Duration::from_secs(2);
67/// The first number a page's object menus are asked by: well clear of a
68/// window's, which count from 1, so neither takes the other's answer.
69const FIRST_MENU: u32 = 900_000;

74    number: u32,

108/// A character's skill trainer, open on the page.
109#[derive(Default)]
110struct TrainerPanel {
111    /// Its goal profiles, read when it opened and kept as the page changes
112    /// them.
113    profiles: hydra_board::trainer::GoalProfiles,
114    watched: Watched,
115}
116
117impl TrainerPanel {
118    /// The panel for `seat`, opened now: its profiles read from `data`.
119    fn open(data: &std::path::Path, seat: &Seat) -> Self {
120        let profiles = hydra_board::trainer::GoalProfiles::load(
121            &data.join("layouts"),
122            seat.instance.as_deref(),
123            &seat.name,
124        );
125        Self {
126            profiles,
127            watched: Watched::default(),
128        }
129    }
130
131    /// The trainer to send the page, as `seat` has it, when it is time to
132    /// look and it changed.
133    fn changed(&mut self, seat: &Seat) -> Option<Out> {
134        if !self.watched.due() {
135            return None;
136        }
137        let out = crate::trainer::trainer(&seat.trainer(), self.profiles.names());
138        self.watched
139            .changed(&out)
140            .then(|| Out::Trainer(Box::new(out)))
141    }
142}

144/// A panel open on the page: looked at every [`PANEL_EVERY`] at most, and
145/// sent only when it is other than as last sent.
146#[derive(Default)]
147struct Watched {
148    /// When it was last looked at; none, it is looked at next.
149    looked: Option<Instant>,
150    /// What was last sent, as sent.
151    sent: Option<String>,
152}
153
154impl Watched {
155    /// Whether it is time to look again; if so, looked at now.
156    fn due(&mut self) -> bool {
157        if self.looked.is_some_and(|at| at.elapsed() < PANEL_EVERY) {
158            return false;
159        }
160        self.looked = Some(Instant::now());
161        true
162    }
163
164    /// Whether `out` is other than as last sent; if so, it is the last.
165    fn changed(&mut self, out: &impl serde::Serialize) -> bool {
166        let said = serde_json::to_string(out).unwrap_or_default();
167        if self.sent.as_ref() == Some(&said) {
168            return false;
169        }
170        self.sent = Some(said);
171        true
172    }
173}

196/// The tasks to send the page, as `story` and `state` have them, when
197/// they are open on it and changed.
198fn tasks_changed(
199    tasks: Option<&mut Watched>,
200    story: &hydra_ui::Story,
201    state: Option<&hydra_model::GameState>,
202) -> Option<Out> {
203    let watched = tasks?;
204    if !watched.due() {
205        return None;
206    }
207    let out = crate::tasks::tasks(story, state);
208    watched.changed(&out).then(|| Out::Tasks(Box::new(out)))
209}

211/// A character's hotbars as the page has them: read again when a file of
212/// them changes, looked at every [`HOTBARS_EVERY`].
213#[derive(Default)]
214struct Hotbars {
215    file: hydra_ui::hotbar::HotbarFile,
216    problems: Vec<String>,
217    /// When each file last changed, as last looked.
218    changed: Vec<Option<std::time::SystemTime>>,
219    /// When the files were last looked at.
220    looked: Option<Instant>,
221    /// The bars last sent, as sent.
222    sent: Option<String>,
223}
224
225impl Hotbars {
226    /// The bars to send the page, as `story` has the character now, when
227    /// they look other than as last sent.
228    fn changed_bars(&mut self, story: &hydra_ui::Story) -> Option<Out> {
229        let world = story.world()?;
230        let bars = wire::hotbars(&self.file, &world);
231        let said = serde_json::to_string(&bars).unwrap_or_default();
232        if self.sent.as_ref() == Some(&said) {
233            return None;
234        }
235        self.sent = Some(said);
236        Some(Out::Hotbars {
237            bars,
238            problems: self.problems.clone(),
239        })
240    }
241
242    /// Read the files again if one changed since last looked, at most
243    /// every [`HOTBARS_EVERY`].
244    fn fresh(&mut self, data: &std::path::Path, seat: &Seat) {
245        if self.looked.is_some_and(|at| at.elapsed() < HOTBARS_EVERY) {
246            return;
247        }
248        self.looked = Some(Instant::now());
249        let changed: Vec<Option<std::time::SystemTime>> =
250            hydra_board::hotbars_paths(data, seat.instance.as_deref(), &seat.name)
251                .iter()
252                .map(|path| {
253                    std::fs::metadata(path)
254                        .and_then(|meta| meta.modified())
255                        .ok()
256                })
257                .collect();
258        if self.looked.is_some() && !self.changed.is_empty() && changed == self.changed {
259            return;
260        }
261        self.changed = changed;
262        let (file, problems) = hydra_board::hotbars_for(data, seat.instance.as_deref(), &seat.name);
263        self.file = file;
264        self.problems = problems;
265        self.sent = None;
266    }
267}

297/// Answer the page on `socket` until it goes, or the web page is no longer
298/// wanted on `port` (`wanted`): only once it has said the pairing. A page
299/// on the device Hydra runs on (`here`) may log a character in.
300pub async fn run<S>(
301    socket: WebSocketStream<S>,
302    server: Server,
303    (mut wanted, port): (watch::Receiver<Option<u16>>, u16),
304    here: bool,
305) where
306    S: AsyncRead + AsyncWrite + Unpin,
307{
308    let (mut sink, mut stream) = socket.split();
309    let hello = tokio::time::timeout(HELLO_FOR, stream.next()).await;
310    let paired = match hello {
311        Ok(Some(Ok(Message::Text(text)))) if text.len() <= HELLO_MOST => {
312            match serde_json::from_str::<In>(&text) {
313                Ok(In::Hello { pairing: said }) => pairing::matches(&said, &server.pairing),
314                _ => false,
315            }
316        }
317        _ => false,
318    };
319    if !paired {
320        let refused = Out::Refused {
321            why: "This device is not paired: open the address the hub shows.".to_owned(),
322        };
323        if sent(&mut sink, message(&refused)).await {
324            closed(&mut sink).await;
325        }
326        return;
327    }
328    let themes = hydra_ui::theme::Themes::load(&server.data.join("themes")).names();
329    let mut page = Page {
330        server,
331        shown: None,
332        hub: None,
333        theme: None,
334        worn: None,
335        menus_asked: FIRST_MENU,
336        highlights: None,
337        out: vec![Out::Welcome { login: here }, Out::Themes { names: themes }],
338        here,
339        pictures: (None, wire::Pictures::default()),
340    };
341    let mut tick = tokio::time::interval(TICK);
342    tick.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
343    let mut ping = tokio::time::interval(PING_EVERY);
344    ping.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
345    let mut hub_due = Instant::now();
346    let mut heard = Instant::now();
347    loop {
348        page.look(&mut hub_due);
349        for out in page.out.drain(..) {
350            if !sent(&mut sink, message(&out)).await {
351                return;
352            }
353        }
354        tokio::select! {
355            said = stream.next() => match said {
356                Some(Ok(Message::Text(text))) => {
357                    heard = Instant::now();
358                    if let Ok(asked) = serde_json::from_str::<In>(&text) {
359                        page.asked(asked).await;
360                        hub_due = Instant::now();
361                    }
362                }
363                Some(Ok(Message::Close(_)) | Err(_)) | None => return,
364                Some(Ok(_)) => heard = Instant::now(),
365            },
366            _ = tick.tick() => {}
367            _ = ping.tick() => {
368                if heard.elapsed() > SILENT_FOR {
369                    closed(&mut sink).await;
370                    return;
371                }
372                if !sent(&mut sink, Message::Ping(Vec::new().into())).await {
373                    return;
374                }
375            }
376            changed = wanted.changed() => {
377                if changed.is_err() || *wanted.borrow() != Some(port) {
378                    closed(&mut sink).await;
379                    return;
380                }
381            }
382        }
383    }
384}

386/// What the panels open on the page have to say of `shown` -- its
387/// inventory, its skill trainer -- each only when it changed, and the link
388/// its `goals web` brought.
389fn panels(shown: &mut Shown, snapshot: Option<&hydra_session::Snapshot>) -> Vec<Out> {
390    let mut out = Vec::new();
391    if let (Some(inventory), Some(snapshot)) = (&mut shown.inventory, snapshot) {
392        out.extend(inventory.changed(&snapshot.state));
393    }
394    if let Some(panel) = &mut shown.trainer {
395        out.extend(panel.changed(&shown.seat));
396    }
397    out.extend(offered(&shown.seat));
398    out
399}

420/// `message` sent on `sink` within [`WRITE_FOR`]: whether it went.
421async fn sent<K>(sink: &mut K, message: Message) -> bool
422where
423    K: futures_util::Sink<Message> + Unpin,
424{
425    matches!(
426        tokio::time::timeout(WRITE_FOR, sink.send(message)).await,
427        Ok(Ok(()))
428    )
429}
430
431/// `sink` closed, waited on [`WRITE_FOR`] at most.
432async fn closed<K>(sink: &mut K)
433where
434    K: futures_util::Sink<Message> + Unpin,
435{
436    let _ = tokio::time::timeout(WRITE_FOR, sink.close()).await;
437}

444impl Page {
445    /// Queue the highlights, while the page's editor is open, when they
446    /// changed since it was last given them.
447    fn look_highlights(&mut self) {
448        let Some(editing) = &mut self.highlights else {
449            return;
450        };
451        let highlights = self.server.sessions.highlights();
452        let generation = highlights.generation();
453        if editing.sent == Some(generation) {
454            return;
455        }
456        editing.sent = Some(generation);
457        let (categories, rules) = crate::highlights::listed(&highlights.file());
458        self.out.push(Out::Highlights {
459            categories,
460            rules,
461            sounds: crate::highlights::sounds(&self.server.data),
462        });
463    }
464
465    /// Queue the colours the page wears, when they changed since it was
466    /// last given them: its own theme's, or the window's.
467    fn look_theme(&mut self) {
468        let palette = match &self.worn {
469            Some((_, palette)) => Some(palette.clone()),
470            None => self.server.sessions.palette(),
471        };
472        if let Some(palette) = palette {
473            let colors = palette.by_name();
474            let names = wire::names(&palette);
475            let said = serde_json::to_string(&(&colors, &names)).unwrap_or_default();
476            if self.theme.as_ref() != Some(&said) {
477                self.theme = Some(said);
478                self.out.push(Out::Theme { colors, names });
479            }
480        }
481    }
482
483    /// The story's picture settings, read again from the Window page's file
484    /// at most every [`PICTURES_EVERY`].
485    fn pictures_now(&mut self) -> wire::Pictures {
486        let (read, pictures) = &mut self.pictures;
487        if read.is_none_or(|at| at.elapsed() >= PICTURES_EVERY) {
488            *read = Some(Instant::now());
489            *pictures = wire::Pictures::load(&self.server.data);
490        }
491        pictures.clone()
492    }
493
494    /// Queue what changed since the page was last told.
495    fn look(&mut self, hub_due: &mut Instant) {
496        self.look_theme();
497        self.look_highlights();
498        let now = Instant::now();
499        if self.shown.is_none() && now >= *hub_due {
500            *hub_due = now + HUB_TICK;
501            let hub = self.hub_now();
502            let said = serde_json::to_string(&hub).unwrap_or_default();
503            if self.hub.as_ref() != Some(&said) {
504                self.hub = Some(said);
505                self.out.push(Out::Hub(hub));
506            }
507        }
508        if self.shown.is_none() {
509            return;
510        }
511        let pictures = self.pictures_now();
512        let Some(shown) = &mut self.shown else {
513            return;
514        };
515        let seated = self
516            .server
517            .sessions
518            .glance(now)
519            .seats
520            .iter()
521            .any(|seat| Arc::ptr_eq(seat, &shown.seat));
522        if !seated {
523            self.shown = None;
524            self.hub = None;
525            self.out.push(Out::Detached {
526                why: "That character is no longer on the table.".to_owned(),
527            });
528            return;
529        }
530        let snapshot = shown.seat.snapshot();
531        shown.hotbars.fresh(&self.server.data, &shown.seat);
532        let (lines, answer, streams) = {
533            let story = shown.seat.story();
534            if let Some(out) = shown.hotbars.changed_bars(&story) {
535                self.out.push(out);
536            }
537            let state = snapshot.as_ref().map(|snapshot| &snapshot.state);
538            self.out
539                .extend(tasks_changed(shown.tasks.as_mut(), &story, state));
540            let streams: Vec<Out> = shown
541                .streams
542                .iter_mut()
543                .filter_map(|(id, follow)| {
544                    let (from, lines) = follow.take(story.streams.get(id)?)?;
545                    Some(Out::Stream {
546                        id: id.clone(),
547                        from,
548                        lines,
549                    })
550                })
551                .collect();
552            let answer = shown.asking.as_ref().and_then(|asking| {
553                let number = asking.number.to_string();
554                story
555                    .menu
556                    .as_ref()
557                    .filter(|(count, menu)| Some(*count) > asking.after && menu.id == number)
558                    .map(|(_, menu)| menu.clone())
559            });
560            (shown.follow.take(&story, &pictures), answer, streams)
561        };
562        if let Some((from, lines)) = lines {
563            self.out.push(Out::Lines { from, lines });
564        }
565        self.out.extend(streams);
566        if let Some(snapshot) = &snapshot {
567            let view = wire::view(snapshot, shown.seat.lifecycle());
568            let said = serde_json::to_string(&view).unwrap_or_default();
569            if shown.view.as_ref() != Some(&said) {
570                shown.view = Some(said);
571                self.out.push(Out::View(Box::new(view)));
572            }
573        }
574        self.out.extend(panels(shown, snapshot.as_deref()));
575        if let Some(asking) = &shown.asking {
576            let generation = snapshot.as_ref().map(|snapshot| snapshot.generation);
577            if let Some(menu) = answer {
578                let groups = object_menu(
579                    &menu,
580                    (&asking.exist, &asking.noun),
581                    snapshot.as_ref().map(|snapshot| &snapshot.state),
582                );
583                self.out.push(Out::Menu {
584                    noun: asking.noun.clone(),
585                    groups: wire::menu(groups),
586                });
587                shown.asking = None;
588            } else if generation != asking.generation || asking.since.elapsed() > MENU_FOR {
589                shown.asking = None;
590            }
591        }
592    }

594    /// What the page's skill trainer asked of the shown character's, done
595    /// as the window's widget does it: a goal stepped, every goal put back,
596    /// the goals applied, or a profile loaded, saved or deleted.
597    fn trainer_asked(&mut self, asked: In) {
598        use hydra_board::trainer::{Change, GoalProfiles};
599        let Some(shown) = &mut self.shown else {
600            return;
601        };
602        let seat = Arc::clone(&shown.seat);
603        match asked {
604            In::TrainerStep { id, up, times } => seat.change_goals(Change::Step {
605                id,
606                up,
607                times: times.clamp(1, STEP_MOST),
608            }),
609            In::TrainerReset => seat.change_goals(Change::Reset),
610            In::TrainerApply => seat.apply_goals(
611                &tokio::runtime::Handle::current(),
612                self.server.sessions.wake(),
613            ),
614            In::TrainerProfile { act, name } => {
615                let Some(panel) = &mut shown.trainer else {
616                    return;
617                };
618                let name = name.trim().to_owned();
619                if name.is_empty() {
620                    return;
621                }
622                let goals = match act.as_str() {
623                    "load" => {
624                        if let Some(goals) = panel.profiles.profiles.get(&name).cloned() {
625                            seat.change_goals(Change::Aim { name, goals });
626                        }
627                        return;
628                    }
629                    "save" => match seat.trainer().goals {
630                        Some(read) => Some(read.targets()),
631                        None => return,
632                    },
633                    "delete" => None,
634                    _ => return,
635                };
636                match &goals {
637                    Some(goals) => {
638                        panel.profiles.profiles.insert(name.clone(), goals.clone());
639                    }
640                    None => {
641                        panel.profiles.profiles.remove(&name);
642                    }
643                }
644                GoalProfiles::change(
645                    &self.server.data.join("layouts"),
646                    seat.instance.as_deref(),
647                    &seat.name,
648                    (&name, goals),
649                );
650            }
651            _ => {}
652        }
653    }

723    /// Do what the page asked.
724    async fn asked(&mut self, asked: In) {
725        match asked {
726            In::Hello { .. } => {}
727            In::Hub => {
728                self.shown = None;
729                self.hub = None;
730            }
731            In::Attach { id } => self.attach(id),
732            In::Send { line } => {
733                if let Some(shown) = &self.shown {
734                    shown.seat.typed(&line, true);
735                }
736            }
737            In::Link { link } => self.link(link),
738            In::Choose { command } => self.game_said(&command),
739            asked @ (In::Characters { .. } | In::Login { .. }) => self.logging_in(asked).await,
740            In::Launch { name } => self.request(HubRequest::Add(name)).await,
741            In::LaunchTeam { name } => self.request(HubRequest::LaunchTeam(name)).await,
742            In::Reconnect { id } => self.request(HubRequest::Reconnect(id)).await,
743            In::Remove { id } => self.request(HubRequest::Remove(id)).await,
744            In::Streams { ids } => {
745                if let Some(shown) = &mut self.shown {
746                    let mut was = std::mem::take(&mut shown.streams);
747                    shown.streams = ids
748                        .into_iter()
749                        .map(|id| {
750                            let follow = was.remove(&id).unwrap_or_default();
751                            (id, follow)
752                        })
753                        .collect();
754                }
755            }
756            In::Wheel { wheel, path } => self.wheel(&wheel, &path),
757            In::Theme { name } => self.wear(name),
758            In::Hotbar { bar, button } => self.press(&bar, &button),
759            In::Button { command } => self.button(&command),
760            In::Inventory { open } => {
761                if let Some(shown) = &mut self.shown {
762                    shown.inventory = open.then(Inventory::default);
763                }
764            }
765            In::HighlightsText => self.highlights_text(),
766            In::HighlightsMerge { text } => self.highlights_merge(text),
767            In::PhoneGet { kind } => self.phone(&kind),
768            In::PhonePut { kind, text } => self.phone_put(&kind, &text),
769            In::Tasks { open } => {
770                if let Some(shown) = &mut self.shown {
771                    shown.tasks = open.then(Watched::default);
772                }
773            }
774            asked @ (In::TrainerStep { .. }
775            | In::TrainerReset
776            | In::TrainerApply
777            | In::TrainerProfile { .. }) => self.trainer_asked(asked),
778            In::Trainer { open } => {
779                if let Some(shown) = &mut self.shown {
780                    shown.trainer =
781                        open.then(|| TrainerPanel::open(&self.server.data, &shown.seat));
782                }
783            }
784            In::InventoryAct { id, act } => self.inventory_act(&id, &act),
785            In::InventoryRefresh => {
786                if let Some(shown) = &self.shown {
787                    let symbol = shown.seat.symbol().unwrap_or('.');
788                    shown.seat.send(&format!("{symbol}inventory"));
789                }
790            }
791            In::Highlights { open } => self.highlights = open.then(Editing::default),
792            In::HighlightSave { was, name, form } => {
793                let kept = name.trim().to_owned();
794                self.highlight(Some(kept), move |file| {
795                    crate::highlights::save(file, (was.as_deref(), &name), &form)
796                });
797            }
798            In::HighlightSwitch { name, on } => {
799                self.highlight(None, move |file| file.switch(&name, on));
800            }
801            In::HighlightRemove { name } => self.highlight(None, move |file| {
802                file.remove(&name).map(|name| format!("{name} taken away."))
803            }),
804            In::Workspace {
805                ask,
806                method,
807                body,
808                character,
809            } => {
810                let (status, body) = match self.workspace_of(character.as_deref()) {
811                    Some(path) => crate::workspace::answer(&path, &method, body.as_deref()).await,
812                    None => (409, None),
813                };
814                self.out.push(Out::Workspace { ask, status, body });
815            }
816        }
817    }

819    /// Show the character seated as `id`, from the end of its story.
820    fn attach(&mut self, id: u32) {
821        let glance = self.server.sessions.glance(Instant::now());
822        let Some(seat) = glance.seats.into_iter().find(|seat| seat.id == id) else {
823            self.out.push(Out::Detached {
824                why: "That character is no longer on the table.".to_owned(),
825            });
826            return;
827        };
828        let follow = Follow::from_end(&seat.story());
829        // The character's scrollback, kept by its story and the page alike.
830        let scrollback = scrollback_of(&self.server.data, &seat);
831        seat.story().scrollback(scrollback);
832        self.out.push(Out::Attached {
833            id,
834            name: seat.name.clone(),
835            game: seat.game.clone(),
836            streams: wire::streams(),
837            story_lines: scrollback.story(),
838            stream_lines: scrollback.streams(),
839        });
840        self.out.push(Out::Doll(Box::new(crate::injuries::doll())));
841        self.out.push(Out::Wheel {
842            wheel: hydra_board::touch_wheel(
843                &self.server.data,
844                seat.instance.as_deref(),
845                &seat.name,
846            ),
847        });
848        self.shown = Some(Shown {
849            seat,
850            follow,
851            view: None,
852            asking: None,
853            streams: BTreeMap::new(),
854            hotbars: Hotbars::default(),
855            inventory: None,
856            tasks: None,
857            trainer: None,
858        });
859    }

861    /// A link tapped: a command sent; an object's own command sent, or its
862    /// menu asked for. A web address the page opens itself.
863    fn link(&mut self, link: RunLink) {
864        let Some(shown) = &mut self.shown else {
865            return;
866        };
867        match link {
868            RunLink::Command { command } => self.game_said(&command),
869            RunLink::Url { .. } => {}
870            RunLink::Object { exist, noun, coord } => {
871                let snapshot = shown.seat.snapshot();
872                let state = snapshot.as_ref().map(|snapshot| &snapshot.state);
873                if let Some(command) = coord
874                    .as_deref()
875                    .and_then(|coord| link_command(coord, (&exist, &noun), state))
876                {
877                    self.game_said(&command);
878                    return;
879                }
880                self.menus_asked = self.menus_asked.wrapping_add(1).max(FIRST_MENU);
881                let number = self.menus_asked;
882                let after = shown.seat.story().menu.as_ref().map(|(count, _)| *count);
883                shown.seat.send(&format!("_menu #{exist} {number}"));
884                shown.asking = Some(Asking {
885                    number,
886                    exist,
887                    noun,
888                    generation: snapshot.as_ref().map(|snapshot| snapshot.generation),
889                    after,
890                    since: Instant::now(),
891                });
892            }
893        }
894    }

977    /// The act `act` on the inventory's item `id`, when its menu offers
978    /// it: sent as the window's Inventory panel sends it.
979    fn inventory_act(&mut self, id: &str, act: &str) {
980        let Some(shown) = &mut self.shown else {
981            return;
982        };
983        let Some(snapshot) = shown.seat.snapshot() else {
984            return;
985        };
986        let Some(act) = crate::inventory::offered(&snapshot.state.inventory_snapshot, id, act)
987        else {
988            self.out.push(Out::Said {
989                text: "Not sent: that item does not offer that; Refresh the inventory.".to_owned(),
990            });
991            return;
992        };
993        if act == hydra_ui::inventory::Act::Inspect
994            && let Some(inventory) = &mut shown.inventory
995        {
996            inventory.item = Some(id.to_owned());
997            inventory.watched.looked = None;
998        }
999        let line = match act.sent(id) {
1000            hydra_ui::inventory::Sent::Game(line) | hydra_ui::inventory::Sent::Quietly(line) => {
1001                line
1002            }
1003            hydra_ui::inventory::Sent::Hydra(line) => {
1004                format!("{}{line}", shown.seat.symbol().unwrap_or('.'))
1005            }
1006        };
1007        shown.seat.send(&line);
1008    }

1091    /// Ask the binary for `request`, as the hub does, and say its answer.
1092    /// The login form's requests: an account's characters listed, or one
1093    /// logged in. Refused for a page on another device.
1094    async fn logging_in(&mut self, asked: In) {
1095        if self.refused_here() {
1096            return;
1097        }
1098        match asked {
1099            In::Characters {
1100                account,
1101                password,
1102                game,
1103                remember,
1104            } => {
1105                let account = hydra_ui::Account {
1106                    account,
1107                    password: hydra_ui::Password::new(password.0),
1108                    game,
1109                    remember,
1110                };
1111                self.request(HubRequest::Characters(account)).await;
1112            }
1113            In::Login {
1114                account,
1115                password,
1116                game,
1117                character,
1118                remember,
1119            } => {
1120                let login = hydra_ui::Login {
1121                    account,

1123                    game,
1124                    character,
1125                    remember,
1126                };
1127                self.request(HubRequest::Login(login)).await;
1128            }
1129            _ => {}
1130        }

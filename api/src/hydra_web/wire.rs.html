1//! What crosses the page's socket, as JSON: each message an object whose
2//! `t` names it.
3//!
4//! The page draws and sends; it decides nothing about the game. A run comes
5//! with the theme colour it is drawn in already chosen ([`run_token`]), by
6//! the token's name, which the page has as a CSS variable; the theme's
7//! colours come in their own message, so a theme worn again recolours what
8//! is drawn without sending it again. Game text is text: the page sets it as
9//! text, never as markup.

663/// `picture` as the page draws it, by `pictures`: `None` when switched off
664/// or when its source is not taken, and then why.
665fn picture_out(
666    picture: &hydra_protocol::frame::InlineImage,
667    pictures: &Pictures,
668) -> Result<PictureOut, Option<String>> {
669    use base64::Engine as _;
670    use hydra_ui::picture::{Source, kind_of, source, within};
671    if !pictures.shown {
672        return Err(None);
673    }
674    let src = match source(&picture.src, pictures.folder.as_deref()).map_err(Some)? {
675        Source::Https(address) => address,
676        Source::Data { kind, base64 } => format!("data:image/{kind};base64,{base64}"),
677        Source::File(path) => {
678            if !pictures
679                .folder
680                .as_deref()
681                .is_some_and(|folder| within(&path, folder))
682            {
683                return Err(Some(format!(
684                    "{} is not a file in the pictures folder",
685                    path.display()
686                )));
687            }
688            let small = std::fs::metadata(&path).is_ok_and(|meta| meta.len() <= PAGE_FILE_MOST);
689            if !small {
690                return Err(Some(format!(
691                    "{} is too large to send to the page",
692                    path.display()
693                )));
694            }
695            let bytes = std::fs::read(&path)
696                .map_err(|e| Some(format!("{} cannot be read: {e}", path.display())))?;
697            let Some(kind) = kind_of(&bytes) else {
698                return Err(Some(format!(
699                    "{} is not a picture the page draws",
700                    path.display()
701                )));
702            };
703            format!(
704                "data:image/{kind};base64,{}",
705                base64::engine::general_purpose::STANDARD.encode(bytes)
706            )
707        }
708    };
709    Ok(PictureOut {
710        src,
711        alt: picture.alt.clone(),
712        width: picture.width,
713        height: picture.height,
714    })
715}

853    preset

892/// A line of the story as the page draws it, its picture drawn as
893/// `pictures` says.
894#[must_use]
895pub fn story_line(shown: &Shown, pictures: &Pictures) -> Line {
896    if let Shown::Picture(picture) = shown {
897        let words = || {
898            vec![plain(
899                &format!("[{}]", picture.alt.as_deref().unwrap_or("a picture")),
900                Some(Token::Muted),
901            )]
902        };
903        let (runs, picture) = match picture_out(picture, pictures) {
904            Ok(out) => (words(), Some(out)),
905            Err(Some(_)) => (words(), None),
906            Err(None) => (Vec::new(), None),
907        };
908        return Line {
909            kind: "picture",
910            stream: None,
911            runs,
912            picture,
913        };
914    }
915    let (kind, stream, runs) = match shown {
916        Shown::Game(runs) => ("game", None, runs.iter().map(run).collect()),
917        Shown::From(stream, runs) => ("game", Some(stream.clone()), runs.iter().map(run).collect()),
918        Shown::Typed { prompt, line } => (
919            "typed",
920            None,
921            vec![plain(prompt, Some(Token::Muted)), plain(line, None)],
922        ),
923        Shown::Prompt(prompt) => ("prompt", None, vec![plain(prompt, Some(Token::Muted))]),
924        Shown::Gap => (
925            "gap",
926            None,
927            vec![plain("— lines were lost here —", Some(Token::Warning))],
928        ),
929        Shown::Hydra(text) => ("hydra", None, vec![plain(text, Some(Token::Hydra))]),
930        Shown::Said(notice) => {
931            let token = match notice.kind {
932                NoticeKind::Error => Token::Wrong,
933                NoticeKind::Warn => Token::Warning,
934                NoticeKind::Info | NoticeKind::Debug => Token::Hydra,
935            };
936            let mut said = plain(&notice.lines().join("\n"), Some(token));
937            said.mono = notice.is_mono();
938            ("said", None, vec![said])
939        }
940        // Taken above.
941        Shown::Picture(_) => ("picture", None, Vec::new()),
942    };
943    Line {
944        kind,
945        stream,
946        runs,
947        picture: None,
948    }
949}

1019/// The Ebon Gate merchants at work in `state`, newest first: by the second
1020/// the game gave, else when it was heard.
1021fn merchants(state: &GameState) -> Vec<MerchantOut> {
1022    use hydra_protocol::MerchantStatus;
1023    let now = state.game_time_now();
1024    let runs = |said: &hydra_protocol::runs::Runs| -> Vec<Run> {
1025        hydra_ui::styled(&said.runs).iter().map(run).collect()
1026    };
1027    let mut at_work: Vec<_> = state.world.merchants.at_work(now).collect();
1028    at_work.sort_by_key(|merchant| std::cmp::Reverse(merchant.time.or(merchant.since)));
1029    at_work
1030        .into_iter()
1031        .map(|merchant| {
1032            let unknown = match &merchant.status {
1033                MerchantStatus::Other(word) => Some(word.clone()),
1034                _ => None,
1035            };
1036            let said = if unknown.is_some() && merchant.said != merchant.announcement {
1037                runs(&merchant.said)
1038            } else {
1039                Vec::new()
1040            };
1041            MerchantOut {
1042                name: merchant.name.clone(),
1043                room: merchant.room.clone(),
1044                status: unknown.filter(|word| !word.is_empty()),
1045                left: now
1046                    .zip(merchant.expires_at)
1047                    .map(|(now, at)| at.saturating_sub(now)),
1048                announcement: runs(&merchant.announcement),
1049                said,
1050                rules: runs(&merchant.rules),
1051            }
1052        })
1053        .collect()
1054}

1119/// `state`'s room, everything in it a link.
1120fn room(state: &GameState) -> RoomOut {
1121    let room = &state.room;

1124            .iter()

1126                text: match &item.status {
1127                    Some(status) => format!("{} ({status})", item.text),

1130                token: Some(token.name()),
1131                bold: token == Token::Creature,
1132                link: Some(RunLink::Object {

1146        creatures: linked(&room.creatures, Token::Creature),

1148        players: linked(&room.players, Token::Player),

1150            exits
1151                .iter()
1152                .map(|exit| hydra_ui::exit_word(exit).to_owned())

1337    /// A picture line carries its picture as the settings say: an https
1338    /// address and a data URI as written, a file of the folder as a data
1339    /// URI; one outside the folder, too large, or not taken, by its words;
1340    /// and nothing at all while pictures are switched off.
1341    #[test]
1342    fn a_picture_line_carries_its_picture_as_the_settings_say() {
1343        let root = std::env::temp_dir().join(format!("hydra-web-pictures-{}", std::process::id()));
1344        let folder = root.join("pictures");
1345        std::fs::create_dir_all(&folder).expect("made");
1346        let png = b"\x89PNG\r\n\x1a\nthe rest of a picture";
1347        std::fs::write(folder.join("tile.png"), png).expect("written");
1348        std::fs::write(root.join("outside.png"), png).expect("written");
1349        let mut big = png.to_vec();
1350        big.resize(usize::try_from(PAGE_FILE_MOST).expect("small") + 1, 0);
1351        std::fs::write(folder.join("big.png"), big).expect("written");
1352        std::fs::write(folder.join("words.png"), b"not a picture").expect("written");
1353
1354        let line = |src: &str, pictures: &Pictures| {
1355            story_line(
1356                &Shown::Picture(hydra_protocol::frame::InlineImage {
1357                    stream: String::new(),
1358                    src: src.to_owned(),
1359                    alt: Some("A tile".to_owned()),
1360                    width: Some(32),
1361                    height: None,
1362                    attrs: Vec::new(),
1363                }),
1364                pictures,
1365            )
1366        };
1367        let on = Pictures {
1368            shown: true,
1369            folder: Some(folder.clone()),
1370        };
1371        let src = |line: &Line| line.picture.as_ref().map(|picture| picture.src.clone());
1372        let words = |line: &Line| {
1373            line.runs
1374                .iter()
1375                .map(|run| run.text.as_str())
1376                .collect::<String>()
1377        };
1378
1379        let https = line("https://example.com/a.png", &on);
1380        assert_eq!(https.kind, "picture");
1381        assert_eq!(src(&https).as_deref(), Some("https://example.com/a.png"));
1382        assert_eq!(
1383            https.picture.as_ref().and_then(|picture| picture.width),
1384            Some(32)
1385        );
1386        assert_eq!(
1387            src(&line("data:image/gif;base64,R0lG ODlh", &on)).as_deref(),
1388            Some("data:image/gif;base64,R0lGODlh")
1389        );
1390        let file = src(&line("tile.png", &on)).expect("sent");
1391        assert!(file.starts_with("data:image/png;base64,"), "{file}");
1392
1393        for (refused, why) in [
1394            (
1395                root.join("outside.png").to_string_lossy().into_owned(),
1396                "outside",
1397            ),
1398            ("big.png".to_owned(), "too large"),
1399            ("words.png".to_owned(), "not a picture"),
1400            ("http://example.com/a.png".to_owned(), "not https"),
1401        ] {
1402            let sent = line(&refused, &on);
1403            assert_eq!(sent.picture, None, "{why}");
1404            assert_eq!(words(&sent), "[A tile]", "{why}");
1405        }
1406        let off = line("https://example.com/a.png", &Pictures::default());
1407        assert_eq!(off.picture, None);
1408        assert!(off.runs.is_empty(), "switched off: nothing");
1409        assert_eq!(
1410            line(
1411                "tile.png",
1412                &Pictures {
1413                    shown: true,
1414                    folder: None
1415                }
1416            )
1417            .picture,
1418            None,
1419            "no folder"
1420        );
1421        let _ = std::fs::remove_dir_all(&root);
1422    }

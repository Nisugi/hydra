1//! What crosses the page's socket, as JSON: each message an object whose
2//! `t` names it.
3//!
4//! The page draws and sends; it decides nothing about the game. A run comes
5//! with the theme colour it is drawn in already chosen ([`run_token`]), by
6//! the token's name, which the page has as a CSS variable; the theme's
7//! colours come in their own message, so a theme worn again recolours what
8//! is drawn without sending it again. Game text is text: the page sets it as
9//! text, never as markup.

891/// `picture` as the page draws it, by `pictures`: `None` when switched off
892/// or when its source is not taken, and then why.
893fn picture_out(
894    picture: &hydra_protocol::frame::InlineImage,
895    pictures: &Pictures,
896) -> Result<PictureOut, Option<String>> {
897    use base64::Engine as _;
898    use hydra_ui::picture::{Source, kind_of, source, within};
899    if !pictures.shown {
900        return Err(None);
901    }
902    let src = match source(&picture.src, pictures.folder.as_deref()).map_err(Some)? {
903        Source::Https(address) => address,
904        Source::Data { kind, base64 } => format!("data:image/{kind};base64,{base64}"),
905        Source::File(path) => {
906            if !pictures
907                .folder
908                .as_deref()
909                .is_some_and(|folder| within(&path, folder))
910            {
911                return Err(Some(format!(
912                    "{} is not a file in the pictures folder",
913                    path.display()
914                )));
915            }
916            let small = std::fs::metadata(&path).is_ok_and(|meta| meta.len() <= PAGE_FILE_MOST);
917            if !small {
918                return Err(Some(format!(
919                    "{} is too large to send to the page",
920                    path.display()
921                )));
922            }
923            let bytes = std::fs::read(&path)
924                .map_err(|e| Some(format!("{} cannot be read: {e}", path.display())))?;
925            let Some(kind) = kind_of(&bytes) else {
926                return Err(Some(format!(
927                    "{} is not a picture the page draws",
928                    path.display()
929                )));
930            };
931            format!(
932                "data:image/{kind};base64,{}",
933                base64::engine::general_purpose::STANDARD.encode(bytes)
934            )
935        }
936    };
937    Ok(PictureOut {
938        src,
939        alt: picture.alt.clone(),
940        width: picture.width,
941        height: picture.height,
942    })
943}

1147/// A line of the story as the page draws it, its picture drawn as
1148/// `pictures` says.
1149#[must_use]
1150pub fn story_line(shown: &Shown, pictures: &Pictures) -> Line {
1151    if let Shown::Picture(picture) = shown {
1152        let words = || {
1153            vec![plain(
1154                &format!("[{}]", picture.alt.as_deref().unwrap_or("a picture")),
1155                Some(Token::Muted),
1156            )]
1157        };
1158        let (runs, picture) = match picture_out(picture, pictures) {
1159            Ok(out) => (words(), Some(out)),
1160            Err(Some(_)) => (words(), None),
1161            Err(None) => (Vec::new(), None),
1162        };
1163        return Line {
1164            kind: "picture",
1165            stream: None,
1166            runs,
1167            picture,
1168            help: None,
1169        };
1170    }
1171    let (kind, stream, runs) = match shown {
1172        Shown::Game(runs) => ("game", None, runs.iter().map(run).collect()),
1173        Shown::From(stream, runs) => ("game", Some(stream.clone()), runs.iter().map(run).collect()),
1174        Shown::Typed { prompt, line } => (
1175            "typed",
1176            None,
1177            vec![run(&StyledRun::prompt(prompt)), plain(line, None)],
1178        ),
1179        Shown::Prompt(prompt) => ("prompt", None, vec![run(&StyledRun::prompt(prompt))]),
1180        Shown::Gap => (
1181            "gap",
1182            None,
1183            vec![plain("— lines were lost here —", Some(Token::Warning))],
1184        ),
1185        Shown::Hydra(text) => ("hydra", None, vec![plain(text, Some(Token::Hydra))]),
1186        Shown::Said(notice) if matches!(notice.body, hydra_session::notice::Body::Help { .. }) => {
1187            return help_line(notice);
1188        }
1189        Shown::Said(notice) => {
1190            let token = match notice.kind {
1191                NoticeKind::Error => Token::Wrong,
1192                NoticeKind::Warn => Token::Warning,
1193                NoticeKind::Info | NoticeKind::Debug => Token::Hydra,
1194            };
1195            let mut said = plain(&notice.lines().join("\n"), Some(token));
1196            said.mono = notice.is_mono();
1197            ("said", None, vec![said])
1198        }
1199        // Taken above.
1200        Shown::Picture(_) => ("picture", None, Vec::new()),
1201    };
1202    Line {
1203        kind,
1204        stream,
1205        runs,
1206        picture: None,
1207        help: None,
1208    }
1209}

1215    let Body::Help { parts, .. } = &notice.body else {
1216        return line(&Shown::Hydra(notice.lines().join("\n")));

1221            HelpPart::Heading(heading) => HelpOut::Heading(heading.clone()),

1227        })
1228        .collect();
1229    let mut said = plain(&notice.lines().join("\n"), Some(Token::Hydra));
1230    said.mono = true;
1231    Line {
1232        kind: "help",
1233        stream: None,
1234        runs: vec![said],
1235        picture: None,
1236        help: Some(help),
1237    }

1409/// The Ebon Gate merchants at work in `state`, newest first: by the second
1410/// the game gave, else when it was heard.
1411fn merchants(state: &GameState) -> Vec<MerchantOut> {

1414    let runs = |said: &hydra_protocol::runs::Runs| -> Vec<Run> {
1415        hydra_ui::styled(&said.runs).iter().map(run).collect()

1418    at_work.sort_by_key(|merchant| std::cmp::Reverse(merchant.time.or(merchant.since)));
1419    at_work
1420        .into_iter()
1421        .map(|merchant| {
1422            let unknown = match &merchant.status {
1423                MerchantStatus::Other(word) => Some(word.clone()),

1427                runs(&merchant.said)

1432                name: merchant.name.clone(),
1433                room: merchant.room.clone(),
1434                status: unknown.filter(|word| !word.is_empty()),
1435                left: now
1436                    .zip(merchant.expires_at)

1969/// Every stream read alongside the story, for the page, and `own`, those
1970/// the stream rules make, after them.
1971#[must_use]
1972pub fn streams(own: &[String]) -> Vec<StreamOut> {
1973    let game = hydra_ui::story::STREAMS
1974        .iter()
1975        .map(|(id, name)| ((*id).to_owned(), (*name).to_owned()));
1976    let own = hydra_ui::story::own_streams(own.iter().map(String::as_str))
1977        .into_iter()
1978        .map(|id| (id.clone(), id));
1979    game.chain(own)
1980        .map(|(id, name)| StreamOut { id, name })
1981        .collect()

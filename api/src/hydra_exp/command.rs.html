9/// What `.exp help` says.

13    "exp hour               the last hour's gains and rate",
14    "exp daily              ten days of totals, a line each",
15    "exp last               the last ten pulses; exp instant, the last ten bounties, brooch rubs and lumps",
16    "exp timing             how far apart the pulses came",
17    "exp level              the level reached, what is left, and when at today's pace",
18    "exp favor              Voln favor as last seen; exp resource, the profession's points",

25/// How many pulses `last` and `instant` show.
26const SHOWN: usize = 10;
27/// How many normal pulses `timing` looks at.
28const TIMED: usize = 1000;

30/// What was asked.
31#[derive(Debug, Clone, PartialEq, Eq)]
32pub enum Asked {
33    /// The words.
34    Help,
35    /// This session.
36    Session,
37    /// The last pulses; `true` for lumps and bounties only.
38    Last(bool),
39    /// Today.
40    Today,
41    /// Ten days.
42    Daily,
43    /// The last hour.
44    Hour,
45    /// Pulse timing.
46    Timing,
47    /// The level and its ETA.
48    Level,
49    /// The newest snapshots of a kind.
50    Snapshots(SnapshotKind),
51    /// The whole report from one time to another, Unix seconds.
52    Report(i64, i64),
53    /// A gain typed in.
54    AddGain(i64),
55    /// Drop rows with no gain.
56    Clean,
57    /// Ask before dropping everything.
58    Reset,
59    /// Drop everything.
60    ConfirmReset,
61}

63/// The words after `exp`, at `now`. `Err` says what was wrong.
64///
65/// # Errors
66///
67/// A word that is not one of the commands, or a time not understood.
68pub fn parse(rest: &str, now: i64) -> Result<Asked, String> {
69    let words = quoted_words(rest);
70    let word = words.first().map(|w| w.to_ascii_lowercase());
71    Ok(match word.as_deref() {
72        None | Some("help") => Asked::Help,
73        Some("session") => Asked::Session,
74        Some("last") => Asked::Last(false),
75        Some("instant") => Asked::Last(true),
76        Some("today") => Asked::Today,
77        Some("daily") => Asked::Daily,
78        Some("hour") => Asked::Hour,
79        Some("timing") => Asked::Timing,
80        Some("level") => Asked::Level,
81        Some("favor") => Asked::Snapshots(SnapshotKind::Favor),
82        Some("resource" | "resources") => Asked::Snapshots(SnapshotKind::Resource),
83        Some("suffused") => Asked::Snapshots(SnapshotKind::Suffused),
84        Some("report") => {
85            let (Some(from), Some(to)) = (words.get(1), words.get(2)) else {
86                return Err("say the span: exp report <from> <to>, as 13:05 15:02, \"07-20 13:05\" \"07-22 15:02\" or \"2026-07-20 13:05:52\" \"2026-07-22 15:02:22\".".to_owned());
87            };
88            let from = parse_time(from, now).ok_or_else(|| format!("`{from}` is not a time."))?;
89            let to = parse_time(to, now).ok_or_else(|| format!("`{to}` is not a time."))?;
90            Asked::Report(from, to)
91        }
92        Some("addgain") => {
93            let gain = words
94                .get(1)
95                .and_then(|w| w.parse::<i64>().ok())
96                .ok_or_else(|| "say the gain: exp addgain <n>.".to_owned())?;
97            Asked::AddGain(gain)
98        }
99        Some("clean") => Asked::Clean,
100        Some("reset") => Asked::Reset,
101        Some("confirm_reset") => Asked::ConfirmReset,
102        Some(other) => {
103            return Err(format!(
104                "`{other}` is not an exp word; exp help lists them."
105            ));
106        }
107    })
108}

110/// Words, a quoted run of them one word.
111fn quoted_words(text: &str) -> Vec<String> {
112    let mut words = Vec::new();
113    let mut rest = text.trim();
114    while !rest.is_empty() {
115        if let Some(after) = rest.strip_prefix('"') {
116            let (word, tail) = after.split_once('"').unwrap_or((after, ""));
117            words.push(word.to_owned());
118            rest = tail.trim_start();
119        } else {
120            let (word, tail) = rest.split_once(char::is_whitespace).unwrap_or((rest, ""));
121            words.push(word.to_owned());
122            rest = tail.trim_start();
123        }
124    }
125    words
126}

128/// `HH:MM[:SS]` today, `MM-DD HH:MM[:SS]` this year, or `YYYY-MM-DD
129/// HH:MM[:SS]`, on the game's clock, at `now`.
130#[must_use]
131pub fn parse_time(text: &str, now: i64) -> Option<i64> {
132    let text = text.trim();
133    let (day, clock) = match text.split_once(char::is_whitespace) {
134        Some((day, clock)) => (Some(day), clock.trim()),
135        None => (None, text),
136    };
137    let mut parts = clock.split(':');
138    let h: i64 = parts.next()?.parse().ok()?;
139    let m: i64 = parts.next()?.parse().ok()?;
140    let s: i64 = parts.next().map_or(Some(0), |s| s.parse().ok())?;
141    if parts.next().is_some() || h > 23 || m > 59 || s > 59 {
142        return None;
143    }
144    let (year, month, day_of) = eastern::date(now);
145    let date = match day {
146        None => (year, month, day_of),
147        Some(day) => {
148            let mut parts = day.split('-');
149            let first: i64 = parts.next()?.parse().ok()?;
150            let second: i64 = parts.next()?.parse().ok()?;
151            match parts.next() {
152                None => (year, first, second),
153                Some(third) => (first, second, third.parse().ok()?),
154            }
155        }
156    };
157    if !(1..=12).contains(&date.1) || !(1..=31).contains(&date.2) {
158        return None;
159    }
160    Some(eastern::midnight(date) + h * 3600 + m * 60 + s)
161}

163/// Answer `asked` for `who` out of `ledger`, at `now`. Lines, in fixed
164/// width. The reports read; `addgain`, `clean` and `reset` write.
165///
166/// # Errors
167///
168/// The ledger cannot be read or written.
169pub fn answer(
170    ledger: &Ledger,
171    writer: &Writer,
172    who: &Who,
173    asked: &Asked,
174    now: i64,
175) -> Result<Vec<String>, String> {
176    let reader = ledger.reader().map_err(|e| e.to_string())?;
177    Ok(match *asked {
178        Asked::AddGain(gain) => {
179            let latest = reader.last_pulses(who, 1).map_err(|e| e.to_string())?;
180            let (exp, axp) = latest.first().map_or((0, 0), |p| (p.exp, p.axp));
181            let pulse = Pulse {
182                who: who.clone(),
183                stamp: Stamp {
184                    at: now,
185                    game_time: None,
186                },
187                exp,
188                axp,
189                exp_gained: gain,
190                axp_gained: 0,
191                instant: 0,
192                fxp: None,
193                fxp_max: None,
194                until_next: None,
195                lumnis: None,
196                rpa: None,
197                fashlonae: None,
198                mind_percent: None,
199                mind_text: None,
200                source: Source::Manual,
201            };
202            writer
203                .try_record(Row::Pulse(pulse))
204                .map_err(|e| e.to_string())?;
205            vec![format!("Added an experience gain of {gain} by hand.")]
206        }
207        Asked::Clean => {
208            let dropped = ledger.prune_pulses(who, false).map_err(|e| e.to_string())?;
209            vec![format!("Dropped {dropped} rows with no experience gained.")]
210        }
211        Asked::Reset => vec![
212            "This drops every experience row of this character's, for good.".to_owned(),
213            "To go on, say: exp confirm_reset".to_owned(),
214        ],
215        Asked::ConfirmReset => {
216            let dropped = ledger.prune_pulses(who, true).map_err(|e| e.to_string())?;
217            vec![format!("Dropped {dropped} rows; the next bar starts over.")]
218        }
219        _ => return read(ledger, who, asked, now),
220    })
221}

223/// A whole line answered, `exp` and its words, as a window asks: the
224/// lines, or what was wrong. Only the reports: a word that writes is
225/// refused.
226///
227/// # Errors
228///
229/// The line is not `exp`, a word is not one of the reports, or the ledger
230/// cannot be read.
231pub fn run(ledger: &Ledger, who: &Who, line: &str, now: i64) -> Result<Vec<String>, String> {
232    let line = line.trim().trim_start_matches(['.', ';']);
233    let (word, rest) = line.split_once(char::is_whitespace).unwrap_or((line, ""));
234    if !word.eq_ignore_ascii_case("exp") {
235        return Err(format!("{word} is not an experience report."));
236    }
237    read(ledger, who, &parse(rest, now)?, now)
238}

240/// A report read for `who` out of `ledger`, at `now`. Lines, in fixed
241/// width.
242///
243/// # Errors
244///
245/// The ledger cannot be read, or `asked` writes.
246pub fn read(ledger: &Ledger, who: &Who, asked: &Asked, now: i64) -> Result<Vec<String>, String> {
247    let name = who.character.as_str();
248    let reader = ledger.reader().map_err(|e| e.to_string())?;
249    let day = report::day_start(now);
250    let read = |from: i64, to: i64| reader.pulses(who, from, to).map_err(|e| e.to_string());
251    Ok(match *asked {
252        Asked::Help => HELP.iter().map(|&l| l.to_owned()).collect(),
253        Asked::Session => {
254            let rows = read(now - eastern::DAY, i64::MAX)?;
255            report::session(name, &rows)
256        }
257        Asked::Last(only_instant) => {
258            let mut rows = reader
259                .last_pulses(who, if only_instant { 1000 } else { SHOWN })
260                .map_err(|e| e.to_string())?;
261            if only_instant {
262                rows.retain(|p| {
263                    matches!(p.source, Source::Instant | Source::Bounty | Source::Brooch)
264                });
265                rows.truncate(SHOWN);
266            }
267            report::last(name, &rows, only_instant)
268        }
269        Asked::Today => report::today(name, &read(day, i64::MAX)?),
270        Asked::Hour => report::hour(name, &read(now - eastern::HOUR, i64::MAX)?),
271        Asked::Daily => report::daily(&read(now - 10 * eastern::DAY, i64::MAX)?, now),
272        Asked::Timing => {
273            let rows = reader.last_pulses(who, TIMED).map_err(|e| e.to_string())?;
274            report::timing(&rows)
275        }
276        Asked::Level => {
277            let latest = reader.last_pulses(who, 1).map_err(|e| e.to_string())?;
278            report::level(name, latest.first(), &read(day, i64::MAX)?, now)
279        }
280        Asked::Snapshots(kind) => {
281            let latest = reader.latest_snapshots(who).map_err(|e| e.to_string())?;
282            let today = reader
283                .snapshots(who, kind, day, i64::MAX)
284                .map_err(|e| e.to_string())?;
285            report::snapshots(kind, &latest, &today)
286        }
287        Asked::Report(from, to) => report::range(name, &read(from, to + 1)?, from, to),
288        Asked::AddGain(_) | Asked::Clean | Asked::Reset | Asked::ConfirmReset => {
289            return Err("that changes the ledger: it is typed, never clicked.".to_owned());
290        }
291    })
292}

298    #[test]
299    fn words_are_read_with_their_quotes() {
300        let now = eastern::midnight((2026, 7, 22)) + 10 * 3600;
301        assert_eq!(parse("", now), Ok(Asked::Help));
302        assert_eq!(parse("Instant", now), Ok(Asked::Last(true)));
303        assert_eq!(parse("addgain 250", now), Ok(Asked::AddGain(250)));
304        assert!(parse("addgain", now).is_err());
305        assert!(parse("bogus", now).is_err());
306        assert_eq!(
307            parse("report 13:05 15:02", now),
308            Ok(Asked::Report(
309                eastern::midnight((2026, 7, 22)) + 13 * 3600 + 5 * 60,
310                eastern::midnight((2026, 7, 22)) + 15 * 3600 + 2 * 60
311            ))
312        );
313        assert_eq!(
314            parse("report \"07-20 13:05\" \"2025-07-22 15:02:22\"", now),
315            Ok(Asked::Report(
316                eastern::midnight((2026, 7, 20)) + 13 * 3600 + 5 * 60,
317                eastern::midnight((2025, 7, 22)) + 15 * 3600 + 2 * 60 + 22
318            ))
319        );
320        assert!(parse("report 13:05", now).is_err());
321        assert!(parse("report 25:00 15:02", now).is_err());
322    }

324    #[test]
325    fn the_answers_come_out_of_a_ledger() {
326        let dir = std::env::temp_dir().join(format!("hydra-exp-cmd-{}", std::process::id()));
327        let _ = std::fs::remove_dir_all(&dir);
328        let ledger = Ledger::open(Ledger::path_in(&dir)).unwrap();
329        let writer = ledger.writer().unwrap();
330        let who = Who::new("prime", "Nisugi");
331        let now = eastern::midnight((2026, 7, 22)) + 10 * 3600;
332
333        let said = answer(&ledger, &writer, &who, &Asked::Level, now).unwrap();
334        assert_eq!(said, ["No pulses recorded yet for Nisugi."]);
335        let said = answer(&ledger, &writer, &who, &Asked::AddGain(300), now).unwrap();
336        assert_eq!(said, ["Added an experience gain of 300 by hand."]);
337        // The writer's thread has the row; the next read waits for it.
338        tokio::runtime::Builder::new_current_thread()
339            .build()
340            .unwrap()
341            .block_on(writer.flush())
342            .unwrap();
343        let said = answer(&ledger, &writer, &who, &Asked::Today, now).unwrap();
344        assert_eq!(said[1], "  Total gained: 300");
345        assert_eq!(said[2], "    Normal:     300");
346        let said = answer(&ledger, &writer, &who, &Asked::Reset, now).unwrap();
347        assert_eq!(said.len(), 2);
348        let said = answer(&ledger, &writer, &who, &Asked::ConfirmReset, now).unwrap();
349        assert_eq!(said, ["Dropped 1 rows; the next bar starts over."]);
350        let _ = std::fs::remove_dir_all(&dir);
351    }

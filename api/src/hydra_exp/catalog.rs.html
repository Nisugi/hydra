1//! The reports named, for a window that lists them: each with the words
2//! that ask for it, a line saying what it shows, and what it takes. Only
3//! the reports: what writes (`addgain`, `clean`, `reset`) is typed.
4
5/// What a report takes beyond its words.
6#[derive(Debug, Clone, Copy, PartialEq, Eq)]
7pub enum Takes {
8    /// Nothing.
9    Nothing,
10    /// A span: two times, as `exp report` takes them.
11    Span,
12}
13
14/// One report.
15#[derive(Debug, Clone, Copy, PartialEq, Eq)]
16pub struct Entry {
17    /// What the list calls it.
18    pub name: &'static str,
19    /// The words after `exp` that ask for it.
20    pub words: &'static str,
21    /// A line on what it shows.
22    pub blurb: &'static str,
23    /// What goes after the words.
24    pub takes: Takes,
25}

133impl Entry {
134    /// The command line that asks for it with `argument` after its words:
135    /// `exp today`, `exp report "07-20 13:05" "07-22 15:02"`.
136    #[must_use]
137    pub fn command(&self, argument: &str) -> String {
138        let mut line = String::from("exp");
139        for part in [self.words, argument.trim()] {
140            if !part.is_empty() {
141                line.push(' ');
142                line.push_str(part);
143            }
144        }
145        line
146    }
147
148    /// Whether `query`'s words all appear in its name or its blurb.
149    #[must_use]
150    pub fn matches(&self, query: &str) -> bool {
151        let text = format!("{} {} {}", self.name, self.words, self.blurb).to_lowercase();
152        query
153            .split_whitespace()
154            .all(|word| text.contains(&word.to_lowercase()))
155    }

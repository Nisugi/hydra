19/// The level a total has attained; 100 at the cap.
20#[must_use]
21pub fn level_for(exp: u64) -> usize {
22    LEVEL_TOTALS
23        .iter()
24        .rposition(|&total| exp >= total)
25        .unwrap_or(0)
26}
27
28/// `level N (X.X% through)`: how far from N to N+1 the total is.
29#[must_use]
30pub fn label(exp: u64) -> String {
31    let level = level_for(exp);
32    if level >= 100 {
33        return "level 100 (cap)".to_owned();
34    }
35    let span = LEVEL_TOTALS[level + 1] - LEVEL_TOTALS[level];
36    #[allow(clippy::cast_precision_loss)]
37    let through = (exp - LEVEL_TOTALS[level]) as f64 * 100.0 / span as f64;
38    format!("level {level} ({through:.1}% through)")
39}

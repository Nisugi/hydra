1//! The bars heard and the looks taken, and the rows they yield.
2//!
3//! The tracker keeps an `Experience` of its own, fed every mind-state bar
4//! as it comes ([`Tracker::heard`]), so a burst of bars is still one pulse
5//! each, in order. A look ([`Tracker::look`]) compares those totals with the
6//! totals last recorded and yields a pulse when they moved; it also reads
7//! the character's state for the guild's count and for favor and resource
8//! points, yielding a snapshot when one changed. The first bar sets the
9//! baseline and yields no pulse, since the gain before it is unknown.
10//!
11//! # How a pulse's source is told
12//!
13//! The game's lines say it. The guild taking a task as done counts up in
14//! the model (`bounty.finished`): a gain within [`BOUNTY_WINDOW`] of that
15//! count rising is the task's reward. A brooch rubbed counts up too
16//! (`brooch_rubs`), and `[You have instantly absorbed N experience
17//! points!]` sums what was absorbed at once (`instant_absorbed`), whatever
18//! brought it. A pulse records that sum since the last pulse as its
19//! `instant`, and its source is bounty, brooch, instant or normal in that
20//! order of what the lines said. No threshold guesses.
21
22use std::collections::HashMap;
23
24use hydra_ledger::{Pulse, Row, Snapshot, SnapshotKind, Source, Stamp, Who};
25use hydra_model::GameState;
26use hydra_model::state::character::experience::Experience;
27use hydra_protocol::frame::ProgressBar;
28
29/// Seconds after the guild takes a task as done within which a gain is
30/// the task's reward.
31pub const BOUNTY_WINDOW: i64 = 15;
32
33/// What a snapshot said, to tell the next from it.
34type Said = (Option<i64>, Option<i64>, Option<i64>);
35
36/// The bars heard, the totals last recorded, and the snapshots last taken
37/// down.
38#[derive(Debug, Clone)]
39pub struct Tracker {
40    who: Who,
41    /// The experience as the bars heard say it.
42    xp: Experience,
43    /// Experience and ascension experience as last recorded; `None` before
44    /// the first bar.
45    last: Option<(u64, u64)>,
46    /// `bounty.finished` as of the last look.
47    finished: u32,
48    /// When the count last rose, and the gain since has not been claimed.
49    bounty_at: Option<i64>,
50    /// `brooch_rubs` as of the last look.
51    rubs: u32,
52    /// When a brooch was last rubbed, and the gain since has not been claimed.
53    brooch_at: Option<i64>,
54    /// `instant_absorbed` as of the last pulse.
55    absorbed: u64,
56    /// What each snapshot last said, by kind and name.
57    said: HashMap<(SnapshotKind, Option<String>), Said>,
58}
59
60impl Tracker {
61    /// A tracker for `who`, with nothing heard yet.
62    #[must_use]
63    pub fn new(who: Who) -> Self {
64        Self {
65            who,
66            xp: Experience::default(),
67            last: None,
68            finished: 0,
69            bounty_at: None,
70            rubs: 0,
71            brooch_at: None,
72            absorbed: 0,
73            said: HashMap::new(),
74        }
75    }
76
77    /// The experience as the session already holds it, for a tracker that
78    /// starts after bars have come: the baseline, with no gain.
79    pub fn seed(&mut self, xp: &Experience) {
80        self.xp = xp.clone();
81    }
82
83    /// Snapshots already on file, so the first look does not write them
84    /// again unchanged.
85    pub fn knows(&mut self, snapshots: &[Snapshot]) {
86        for s in snapshots {
87            self.said
88                .insert((s.kind, s.name.clone()), (s.value, s.weekly, s.total));
89        }
90    }
91
92    /// A bar of the `expr` dialog heard; `true` when it was the mind-state
93    /// bar, and a look is due.
94    pub fn heard(&mut self, bar: &ProgressBar) -> bool {
95        self.xp.apply_bar(bar) && bar.id == "mindState"
96    }
97
98    /// Look, as of `stamp`, with `state` for what the bars do not say: the
99    /// rows it yields.
100    pub fn look(&mut self, state: &GameState, stamp: Stamp) -> Vec<Row> {
101        let mut rows = Vec::new();
102        self.note_finished(state.bounty.finished, stamp.at);
103        self.note_rubbed(state.character.experience.brooch_rubs, stamp.at);
104        let absorbed = state.character.experience.instant_absorbed;
105        if let Some(pulse) = self.pulse(stamp, absorbed) {
106            rows.push(Row::Pulse(pulse));
107        }
108        rows.extend(self.snapshots(state, stamp).into_iter().map(Row::Snapshot));
109        rows
110    }
111
112    /// The guild's count rising opens a bounty window; a reconnect's
113    /// cleared count is taken as the new base.
114    fn note_finished(&mut self, finished: u32, at: i64) {
115        if finished > self.finished {
116            self.bounty_at = Some(at);
117        }
118        self.finished = finished;
119    }
120
121    /// A brooch rubbed opens its window, as the guild's word does.
122    fn note_rubbed(&mut self, rubs: u32, at: i64) {
123        if rubs > self.rubs {
124            self.brooch_at = Some(at);
125        }
126        self.rubs = rubs;
127    }
128
129    fn pulse(&mut self, stamp: Stamp, absorbed: u64) -> Option<Pulse> {
130        let xp = &self.xp;
131        let exp = xp.experience?;
132        let axp = xp.ascension_experience.unwrap_or(0);
133        let (before, before_asc) = self.last.replace((exp, axp))?;
134        let exp_gained =
135            i64::try_from(exp).unwrap_or(i64::MAX) - i64::try_from(before).unwrap_or(0);
136        let axp_gained =
137            i64::try_from(axp).unwrap_or(i64::MAX) - i64::try_from(before_asc).unwrap_or(0);
138        if exp_gained == 0 && axp_gained == 0 {
139            return None;
140        }
141        // What the game said was absorbed at once since the last pulse; a
142        // reconnect's cleared sum is taken as the new base.
143        let instant = if absorbed >= self.absorbed {
144            i64::try_from(absorbed - self.absorbed).unwrap_or(i64::MAX)
145        } else {
146            i64::try_from(absorbed).unwrap_or(i64::MAX)
147        };
148        self.absorbed = absorbed;
149        let within = |opened: Option<i64>| opened.is_some_and(|at| stamp.at - at <= BOUNTY_WINDOW);
150        let source = if within(self.bounty_at.take()) {
151            Source::Bounty
152        } else if within(self.brooch_at.take()) {
153            Source::Brooch
154        } else if instant > 0 {
155            Source::Instant
156        } else {
157            Source::Normal
158        };
159        Some(Pulse {
160            who: self.who.clone(),
161            stamp,
162            exp,
163            axp,
164            exp_gained,
165            axp_gained,
166            instant,
167            fxp: xp.field_experience,
168            fxp_max: xp.field_experience_max,
169            until_next: xp.until_next,
170            lumnis: xp.lumnis,
171            rpa: xp.rpa.clone(),
172            fashlonae: xp.fashlonae,
173            mind_percent: xp.mind_percent,
174            mind_text: xp.mind_state.clone(),
175            source,
176        })
177    }
178
179    fn snapshots(&mut self, state: &GameState, stamp: Stamp) -> Vec<Snapshot> {
180        let mut out = Vec::new();
181        if let Some(favor) = state.character.currency.voln_favor {
182            self.take(
183                &mut out,
184                stamp,
185                SnapshotKind::Favor,
186                None,
187                (Some(favor), None, None),
188            );
189        }
190        let standing = &state.character.standing;
191        let name = standing.resource_type.map(|r| r.as_str().to_owned());
192        if let Some(amounts) = standing.resources {
193            self.take(
194                &mut out,
195                stamp,
196                SnapshotKind::Resource,
197                name.clone(),
198                (
199                    None,
200                    Some(i64::from(amounts.weekly)),
201                    Some(i64::from(amounts.total)),
202                ),
203            );
204        }
205        if let Some(suffused) = standing.suffused {
206            self.take(
207                &mut out,
208                stamp,
209                SnapshotKind::Suffused,
210                name,
211                (Some(i64::from(suffused)), None, None),
212            );
213        }
214        out
215    }
216
217    /// A snapshot, when it says something new.
218    fn take(
219        &mut self,
220        out: &mut Vec<Snapshot>,
221        stamp: Stamp,
222        kind: SnapshotKind,
223        name: Option<String>,
224        said: Said,
225    ) {
226        let key = (kind, name.clone());
227        if self.said.get(&key) == Some(&said) {
228            return;
229        }
230        self.said.insert(key, said);
231        out.push(Snapshot {
232            who: self.who.clone(),
233            stamp,
234            kind,
235            name,
236            value: said.0,
237            weekly: said.1,
238            total: said.2,
239        });
240    }
241}

259    /// `wire` into the state and the tracker, as the recorder feeds them:
260    /// a look after each mind-state bar and at the end.
261    fn fed(tracker: &mut Tracker, state: &mut GameState, wire: &str, t: i64) -> Vec<Row> {
262        let mut rows = Vec::new();
263        for frame in Parser::new().push_bytes(wire.as_bytes()) {
264            state.apply(&frame);
265            if let Frame::ProgressBar(bar) = &frame
266                && tracker.heard(bar)
267            {
268                rows.extend(tracker.look(state, at(t)));
269            }
270        }
271        rows.extend(tracker.look(state, at(t)));
272        rows
273    }

277            "<dialogData id='expr'><progressBar id='mindState' value='100' text='must rest' \

279             exp='{exp}' until_next='1543'/></dialogData>\n<prompt time=\"1\">&gt;</prompt>\n"

285    #[test]
286    fn the_first_bar_is_the_baseline_and_a_moved_total_is_a_pulse() {
287        let mut state = GameState::default();
288        let mut tracker = Tracker::new(who());
289        assert!(tracker.look(&state, at(0)).is_empty(), "nothing known");
290        assert!(
291            fed(&mut tracker, &mut state, &bar(1000, 50), 10).is_empty(),
292            "baseline"
293        );
294        assert!(
295            fed(&mut tracker, &mut state, &bar(1000, 50), 20).is_empty(),
296            "no change"
297        );
298        let rows = fed(&mut tracker, &mut state, &bar(1063, 50), 30);
299        let [Row::Pulse(pulse)] = rows.as_slice() else {
300            panic!("one pulse, got {rows:?}");
301        };
302        assert_eq!((pulse.exp_gained, pulse.axp_gained), (63, 0));
303        assert_eq!(pulse.source, Source::Normal);
304        assert_eq!((pulse.lumnis, pulse.rpa.as_deref()), (Some(3), Some("2")));
305        assert_eq!(pulse.mind_text.as_deref(), Some("must rest"));
306        assert_eq!(pulse.stamp, at(30));
307    }

309    /// Bars that arrive together are still one pulse each, in order, since
310    /// the tracker hears the bars rather than the state after them.
311    #[test]
312    fn a_burst_of_bars_is_a_pulse_each() {
313        let mut state = GameState::default();
314        let mut tracker = Tracker::new(who());
315        let burst = format!("{}{}{}", bar(1000, 0), bar(1060, 0), bar(1120, 0));
316        let rows = fed(&mut tracker, &mut state, &burst, 5);
317        let gains: Vec<_> = rows
318            .iter()
319            .map(|r| match r {
320                Row::Pulse(p) => p.exp_gained,
321                _ => panic!("only pulses here"),
322            })
323            .collect();
324        assert_eq!(gains, [60, 60]);
325    }

345    #[test]
346    fn a_gain_after_the_guild_s_word_is_a_bounty_and_a_lump_is_instant() {
347        let mut state = GameState::default();
348        let mut tracker = Tracker::new(who());
349        fed(&mut tracker, &mut state, &bar(1000, 0), 0);
350        fed(&mut tracker, &mut state, DONE, 5);
351        let rows = fed(&mut tracker, &mut state, &bar(2500, 0), 12);
352        assert!(matches!(rows.as_slice(), [Row::Pulse(p)] if p.source == Source::Bounty));
353        // The game says what was absorbed at once; the rest of the gain
354        // was the pulse.
355        fed(&mut tracker, &mut state, ABSORBED, 35);
356        let rows = fed(&mut tracker, &mut state, &bar(4060, 0), 40);
357        assert!(
358            matches!(rows.as_slice(), [Row::Pulse(p)] if p.source == Source::Instant && p.instant == 1500 && p.exp_gained == 1560),
359            "{rows:?}"
360        );
361        // A brooch rubbed names the lump.
362        fed(&mut tracker, &mut state, RUBBED, 45);
363        fed(&mut tracker, &mut state, ABSORBED, 46);
364        let rows = fed(&mut tracker, &mut state, &bar(5560, 0), 48);
365        assert!(
366            matches!(rows.as_slice(), [Row::Pulse(p)] if p.source == Source::Brooch && p.instant == 1500),
367            "{rows:?}"
368        );
369        fed(&mut tracker, &mut state, DONE, 50);
370        let rows = fed(
371            &mut tracker,
372            &mut state,
373            &bar(5620, 0),
374            50 + BOUNTY_WINDOW + 1,
375        );
376        assert!(
377            matches!(rows.as_slice(), [Row::Pulse(p)] if p.source == Source::Normal),
378            "the window closed"
379        );
380    }

384        let mut state = GameState::default();
385        let mut tracker = Tracker::new(who());
386        tracker.knows(&[Snapshot {

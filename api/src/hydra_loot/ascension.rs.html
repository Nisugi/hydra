1//! The creatures whose searches count toward ascension gemstones and dust,
2//! from the list kept in `data/ascension.txt`: the grounds' own, and the
3//! Onslaught's invaders apart, since whether those count is not settled.
4
5/// The list as the crate carries it.
6const LIST: &str = include_str!("../data/ascension.txt");
7
8/// Where a listed creature belongs.
9#[derive(Debug, Clone, Copy, PartialEq, Eq)]
10pub enum Listing {
11    /// One of the ascension hunting grounds' own.
12    Grounds,
13    /// An Empyrean Onslaught invader.
14    Onslaught,
15}
16
17/// Whether a creature named `name` in a search line is on the list, and
18/// where it belongs. The listed words must appear whole in the name.
19#[must_use]
20pub fn listed(name: &str) -> Option<Listing> {
21    let words: Vec<String> = name
22        .split(|c: char| !c.is_alphanumeric() && c != '-')
23        .filter(|word| !word.is_empty())
24        .map(str::to_ascii_lowercase)
25        .collect();
26    let mut section = Listing::Grounds;
27    for line in LIST.lines() {
28        let line = line.trim();
29        if line.is_empty() || line.starts_with('#') {
30            continue;
31        }
32        match line {
33            "[grounds]" => section = Listing::Grounds,
34            "[onslaught]" => section = Listing::Onslaught,
35            entry => {
36                let entry: Vec<&str> = entry.split(' ').collect();
37                if words
38                    .windows(entry.len())
39                    .any(|window| window.iter().zip(&entry).all(|(a, b)| a == b))
40                {
41                    return Some(section);
42                }
43            }
44        }
45    }
46    None
47}

49#[cfg(test)]
50mod tests {
51    use super::*;
52
53    #[test]
54    fn the_list_is_matched_whole_word_without_the_adjective() {
55        assert_eq!(listed("gigas berserker"), Some(Listing::Grounds));
56        assert_eq!(
57            listed("a brawny gigas shield-maiden"),
58            Some(Listing::Grounds)
59        );
60        assert_eq!(listed("hapless charmed corsair"), Some(Listing::Grounds));
61        assert_eq!(listed("sanguine ooze"), Some(Listing::Grounds));
62        assert_eq!(listed("sanguine oozeling"), None, "whole words only");
63        assert_eq!(listed("goliath plunderer"), Some(Listing::Onslaught));
64        assert_eq!(listed("triton warden"), None);
65    }
66
67    #[test]
68    fn every_line_of_the_list_is_a_heading_a_comment_or_a_name() {
69        for line in LIST.lines().map(str::trim) {
70            assert!(
71                line.is_empty()
72                    || line.starts_with('#')
73                    || line == "[grounds]"
74                    || line == "[onslaught]"
75                    || line
76                        .chars()
77                        .all(|c| c.is_ascii_lowercase() || c == ' ' || c == '-'),
78                "{line:?}"
79            );
80        }
81    }

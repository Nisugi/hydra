1//! `.loot`: what was found and what it came to, from the ledger.
2//!
3//! The day and the month are the game's, Eastern. An item's path is its
4//! events in order: found on what, appraised for how much, sold for how
5//! much where. Items are shared across the household by their ids
6//! ([`crate::identity`]), so a sale counts for whoever sold, and "of which
7//! found by" names who found.
8

15/// What `.loot help` says.
16pub const HELP: &[&str] = &[
17    "loot             this character today: searches, mugs and boxes, items found and sold, and the last ten items' paths",
18    "loot all         every character today: items found and sold, and whose finds each sold",
19    "loot creatures   this month's creatures by what they carried: searches, silver, items, and what the items sold for",
20    "loot cap         this month's items as appraised against what they sold for",
21    "loot gems        ascension gemstones: searches this week and month, since the last, the average, everyone's standing, the last ten",
22    "loot dust        gemstone dust, the same way",
23    "loot keys        keys and locks found, against every search",
24    "loot legendary   legendary treasures, against every search",
25    "The game's day turns at midnight Eastern. Nothing is ever sent to the game: it is all read off searches, boxes, appraisals and sales as they happen.",
26];

28/// How many items the character's own report lists.
29const SHOWN: usize = 10;

31/// The words after `loot`, answered for `who` at `now`.
32///
33/// # Errors
34///
35/// A word that is not one of the commands, or the ledger cannot be read.
36pub fn answer(ledger: &Ledger, who: &Who, rest: &str, now: i64) -> Result<Vec<String>, String> {
37    let word = rest.split_whitespace().next().map(str::to_ascii_lowercase);
38    let reader = ledger.reader().map_err(|e| e.to_string())?;
39    let conn = reader.connection();
40    let (today, tomorrow) = day_bounds(now);
41    let (month, next_month) = month_bounds(now);
42    let done = match word.as_deref() {
43        None => own(conn, who, today, tomorrow),
44        Some("all") => all(conn, &who.instance, today, tomorrow),
45        Some("creatures") => creatures(conn, &who.instance, month, next_month),
46        Some("cap") => cap(conn, who, month, next_month),
47        Some(word) if let Some(find) = crate::finds::Find::parse(word) => {
48            crate::finds::report(conn, who, find, now)
49        }
50        Some("help") => return Ok(HELP.iter().map(|&line| line.to_owned()).collect()),
51        Some(other) => {
52            return Err(format!(
53                "`{other}` is not a loot word; loot help lists them."
54            ));
55        }
56    };
57    done.map_err(|e| e.to_string())
58}

60/// A report asked for on a command line, `loot ...`, answered for `who`
61/// at `now`: what a window that lists the reports runs.
62///
63/// # Errors
64///
65/// The line is not a loot line, or the ledger cannot be read.
66pub fn run(ledger: &Ledger, who: &Who, line: &str, now: i64) -> Result<Vec<String>, String> {
67    let line = line.trim().trim_start_matches(['.', ';']);
68    let (word, rest) = line.split_once(char::is_whitespace).unwrap_or((line, ""));
69    if !word.eq_ignore_ascii_case("loot") {
70        return Err(format!("{word} is not a loot report."));
71    }
72    answer(ledger, who, rest, now)
73}

75/// The Eastern day `now` falls in, as seconds: its start, and the next's.
76fn day_bounds(now: i64) -> (i64, i64) {
77    let date = eastern::date(now);
78    let days = eastern::days_from_civil(date);
79    (
80        eastern::midnight(date),
81        eastern::midnight(eastern::civil_from_days(days + 1)),
82    )
83}
84
85/// The Eastern month `now` falls in, as seconds: its start, and the next's.
86fn month_bounds(now: i64) -> (i64, i64) {
87    let (y, m, _) = eastern::date(now);
88    let next = if m == 12 {
89        (y + 1, 1, 1)
90    } else {
91        (y, m + 1, 1)
92    };
93    (eastern::midnight((y, m, 1)), eastern::midnight(next))
94}

96/// One character's day: the counts, then the last items' paths.
97fn own(conn: &Connection, who: &Who, from: i64, to: i64) -> Result<Vec<String>, Error> {
98    let count = |kind: &str| -> Result<(i64, i64), Error> {
99        conn.query_row(
100            "SELECT COUNT(*), COALESCE(SUM(amount), 0) FROM loot_events
101             WHERE instance = ?1 AND character = ?2 AND kind = ?3 AND at >= ?4 AND at < ?5",
102            params![who.instance, who.character, kind, from, to],
103            |row| Ok((row.get(0)?, row.get(1)?)),
104        )
105    };
106    let (searched, search_silver) = count("searched")?;
107    let (mugged, mug_silver) = count("mugged")?;
108    let (boxes, _) = count("box_opened")?;
109    let (_, box_silver) = count("box_coins")?;
110    let (found, _) = count("found")?;
111    let (sold, sold_silver) = count("sold")?;
112    let mut lines = vec![
113        format!(
114            "{}, today ({}):",
115            who.character,
116            eastern::format(eastern::date(from))
117        ),
118        format!(
119            "searched {searched}, mugged {mugged}, opened {boxes} boxes: {} silver found, {found} items",
120            commas(search_silver + mug_silver + box_silver)
121        ),
122        format!("sold {sold} items for {} silver", commas(sold_silver)),
123    ];
124    lines.extend(paths(conn, who, SHOWN)?);
125    Ok(lines)
126}

128/// The last `limit` items this character saw, each with its path.
129fn paths(conn: &Connection, who: &Who, limit: usize) -> Result<Vec<String>, Error> {
130    let mut items = conn.prepare_cached(
131        "SELECT id, name FROM items
132         WHERE instance = ?1 AND character = ?2
133         ORDER BY last_seen DESC, id DESC LIMIT ?3",
134    )?;
135    let limit = i64::try_from(limit).unwrap_or(i64::MAX);
136    let rows: Vec<(i64, String)> = items
137        .query_map(params![who.instance, who.character, limit], |row| {
138            Ok((row.get(0)?, row.get(1)?))
139        })?
140        .collect::<Result<_, _>>()?;
141    let mut events = conn.prepare_cached(
142        "SELECT kind, subject, amount, character FROM loot_events
143         WHERE item_id = ?1 ORDER BY at, id",
144    )?;
145    let mut lines = Vec::new();
146    for (id, name) in rows {
147        let steps: Vec<String> = events
148            .query_map(params![id], |row| {
149                let kind: String = row.get(0)?;
150                let subject: Option<String> = row.get(1)?;
151                let amount: Option<i64> = row.get(2)?;
152                let by: String = row.get(3)?;
153                Ok(step(&kind, subject.as_deref(), amount, &by, &who.character))
154            })?
155            .collect::<Result<_, _>>()?;
156        lines.push(format!("{name}: {}", steps.join(" > ")));
157    }
158    Ok(lines)
159}

161/// One step of an item's path, in words.
162fn step(kind: &str, subject: Option<&str>, amount: Option<i64>, by: &str, own: &str) -> String {
163    let who = if by == own {
164        String::new()
165    } else {
166        format!(" by {by}")
167    };
168    let subject = subject.unwrap_or("");
169    match kind {
170        "found" => format!("found on {subject}{who}"),
171        "box_opened" => format!("opened{who}"),
172        "box_coins" => format!("{} silver inside{who}", commas(amount.unwrap_or(0))),
173        "appraised" => format!(
174            "appraised {} at {subject}{who}",
175            commas(amount.unwrap_or(0))
176        ),
177        "sold" => format!("sold {} at {subject}{who}", commas(amount.unwrap_or(0))),
178        "skinned" => format!("skinned from {subject}{who}"),
179        other => format!("{other}{who}"),
180    }
181}

183/// Every character's day: found, sold, and whose finds were sold.
184fn all(conn: &Connection, instance: &str, from: i64, to: i64) -> Result<Vec<String>, Error> {
185    let mut characters = conn.prepare_cached(
186        "SELECT DISTINCT character FROM loot_events
187         WHERE instance = ?1 AND at >= ?2 AND at < ?3 ORDER BY character",
188    )?;
189    let names: Vec<String> = characters
190        .query_map(params![instance, from, to], |row| row.get(0))?
191        .collect::<Result<_, _>>()?;
192    if names.is_empty() {
193        return Ok(vec![format!(
194            "No loot is on file today ({}).",
195            eastern::format(eastern::date(from))
196        )]);
197    }
198    let mut lines = vec![format!(
199        "Loot today ({}):",
200        eastern::format(eastern::date(from))
201    )];
202    for name in names {
203        let who = Who::new(instance, name.as_str());
204        let (found, silver): (i64, i64) = conn.query_row(
205            "SELECT
206                (SELECT COUNT(*) FROM loot_events WHERE instance = ?1 AND character = ?2
207                    AND kind = 'found' AND at >= ?3 AND at < ?4),
208                (SELECT COALESCE(SUM(amount), 0) FROM loot_events WHERE instance = ?1
209                    AND character = ?2 AND kind IN ('searched', 'mugged', 'box_coins')
210                    AND at >= ?3 AND at < ?4)",
211            params![who.instance, who.character, from, to],
212            |row| Ok((row.get(0)?, row.get(1)?)),
213        )?;
214        let (sold, sold_silver): (i64, i64) = conn.query_row(
215            "SELECT COUNT(*), COALESCE(SUM(amount), 0) FROM loot_events
216             WHERE instance = ?1 AND character = ?2 AND kind = 'sold' AND at >= ?3 AND at < ?4",
217            params![who.instance, who.character, from, to],
218            |row| Ok((row.get(0)?, row.get(1)?)),
219        )?;
220        let mut finders = conn.prepare_cached(
221            "SELECT f.character, COUNT(*), COALESCE(SUM(s.amount), 0) FROM loot_events s
222             JOIN loot_events f ON f.item_id = s.item_id AND f.kind = 'found'
223                AND f.character <> s.character
224             WHERE s.instance = ?1 AND s.character = ?2 AND s.kind = 'sold'
225                AND s.at >= ?3 AND s.at < ?4
226             GROUP BY f.character ORDER BY f.character",
227        )?;
228        let others: Vec<(String, i64, i64)> = finders
229            .query_map(params![who.instance, who.character, from, to], |row| {
230                Ok((row.get(0)?, row.get(1)?, row.get(2)?))
231            })?
232            .collect::<Result<_, _>>()?;
233        let mut line = format!(
234            "{name}: found {found} items and {} silver; sold {sold} items for {}",
235            commas(silver),
236            commas(sold_silver)
237        );
238        for (finder, count, amount) in others {
239            let _ = write!(
240                line,
241                ", of which {count} found by {finder} ({})",
242                commas(amount)
243            );
244        }
245        lines.push(line);
246    }
247    Ok(lines)
248}

250/// The month's creatures by what they carried.
251fn creatures(conn: &Connection, instance: &str, from: i64, to: i64) -> Result<Vec<String>, Error> {
252    let mut stmt = conn.prepare_cached(
253        "SELECT e.subject,
254            SUM(CASE WHEN e.kind IN ('searched', 'mugged') THEN 1 ELSE 0 END),
255            COALESCE(SUM(CASE WHEN e.kind IN ('searched', 'mugged') THEN e.amount END), 0),
256            SUM(CASE WHEN e.kind = 'found' THEN 1 ELSE 0 END),
257            COALESCE((SELECT SUM(s.amount) FROM loot_events s
258                WHERE s.kind = 'sold' AND s.item_id IN
259                    (SELECT f.item_id FROM loot_events f WHERE f.kind = 'found'
260                        AND f.subject = e.subject AND f.instance = e.instance
261                        AND f.at >= ?2 AND f.at < ?3)), 0)
262         FROM loot_events e
263         WHERE e.instance = ?1 AND e.kind IN ('searched', 'mugged', 'found')
264            AND e.subject IS NOT NULL AND e.at >= ?2 AND e.at < ?3
265         GROUP BY e.subject
266         ORDER BY 5 DESC, 3 DESC, 2 DESC",
267    )?;
268    let rows: Vec<(String, i64, i64, i64, i64)> = stmt
269        .query_map(params![instance, from, to], |row| {
270            Ok((
271                row.get(0)?,
272                row.get(1)?,
273                row.get(2)?,
274                row.get(3)?,
275                row.get(4)?,
276            ))
277        })?
278        .collect::<Result<_, _>>()?;
279    if rows.is_empty() {
280        return Ok(vec![
281            "No creature searched this month is on file.".to_owned(),
282        ]);
283    }
284    let mut lines = vec![format!(
285        "{:<32} {:>8} {:>12} {:>6} {:>12}",
286        "creature", "searches", "silver", "items", "sold for"
287    )];
288    for (name, searches, silver, items, sold) in rows {
289        lines.push(format!(
290            "{name:<32} {searches:>8} {:>12} {items:>6} {:>12}",
291            commas(silver),
292            commas(sold)
293        ));
294    }
295    Ok(lines)
296}

298/// The month's items as appraised against what they sold for.
299fn cap(conn: &Connection, who: &Who, from: i64, to: i64) -> Result<Vec<String>, Error> {
300    let (items, appraised, sold_items, sold): (i64, i64, i64, i64) = conn.query_row(
301        "SELECT
302            (SELECT COUNT(*) FROM items WHERE instance = ?1 AND character = ?2
303                AND first_seen >= ?3 AND first_seen < ?4),
304            (SELECT COALESCE(SUM(best), 0) FROM
305                (SELECT MAX(COALESCE(worth, amount)) AS best FROM loot_events
306                 WHERE instance = ?1 AND character = ?2 AND kind = 'appraised'
307                    AND at >= ?3 AND at < ?4 GROUP BY item_id)),
308            (SELECT COUNT(*) FROM loot_events WHERE instance = ?1 AND character = ?2
309                AND kind = 'sold' AND at >= ?3 AND at < ?4),
310            (SELECT COALESCE(SUM(amount), 0) FROM loot_events WHERE instance = ?1
311                AND character = ?2 AND kind = 'sold' AND at >= ?3 AND at < ?4)",
312        params![who.instance, who.character, from, to],
313        |row| Ok((row.get(0)?, row.get(1)?, row.get(2)?, row.get(3)?)),
314    )?;
315    let (y, m, _) = eastern::date(from);
316    Ok(vec![
317        format!("{}, {y:04}-{m:02}:", who.character),
318        format!(
319            "{items} items first seen; appraised at {} in all",
320            commas(appraised)
321        ),
322        format!("{sold_items} sold for {}", commas(sold)),
323    ])
324}

326/// A number with thousands separated.
327fn commas(n: i64) -> String {
328    let digits = n.abs().to_string();
329    let mut out = String::new();
330    for (i, c) in digits.chars().enumerate() {
331        if i > 0 && (digits.len() - i).is_multiple_of(3) {
332            out.push(',');
333        }
334        out.push(c);
335    }
336    if n < 0 { format!("-{out}") } else { out }
337}

378    #[test]
379    fn a_find_sold_by_another_is_credited_to_the_finder() {
380        let scratch = Scratch::new("report");
381        let ledger = Ledger::open(scratch.file()).unwrap_or_else(|e| panic!("{e}"));
382        let zoleta = Who::new("prime", "Zoleta");
383        let nisugi = Who::new("prime", "Nisugi");
384        let mut hers = Recorder::open(&ledger, zoleta.clone()).unwrap_or_else(|e| panic!("{e}"));
385        let mut his = Recorder::open(&ledger, nisugi.clone()).unwrap_or_else(|e| panic!("{e}"));
386        let ruby = item("892682691", "flickering snowfire ruby", "ruby");
387        hers.record(
388            &LootEvent::Searched {
389                creature: item("892673796", "gigas berserker", "berserker"),
390                silver: Some(563),
391                items: vec![ruby.clone()],
392            },
393            NOON,
394        )
395        .unwrap_or_else(|e| panic!("{e}"));
396        his.record(
397            &LootEvent::Sold {
398                item: ruby,
399                silver: 4000,
400                shop: "The jeweler Etaenia".to_owned(),
401            },
402            NOON + 600,
403        )
404        .unwrap_or_else(|e| panic!("{e}"));
405        let lines = answer(&ledger, &nisugi, "all", NOON + 700).unwrap_or_else(|e| panic!("{e}"));
406        assert_eq!(lines[0], "Loot today (2025-10-09):");
407        assert_eq!(
408            lines[1],
409            "Nisugi: found 0 items and 0 silver; sold 1 items for 4,000, of which 1 found by Zoleta (4,000)"
410        );
411        assert_eq!(
412            lines[2],
413            "Zoleta: found 1 items and 563 silver; sold 0 items for 0"
414        );
415        let own = answer(&ledger, &zoleta, "", NOON + 700).unwrap_or_else(|e| panic!("{e}"));
416        assert_eq!(own[0], "Zoleta, today (2025-10-09):");
417        assert_eq!(
418            own[1],
419            "searched 1, mugged 0, opened 0 boxes: 563 silver found, 1 items"
420        );
421        assert_eq!(own[2], "sold 0 items for 0 silver");
422        assert_eq!(
423            own[3],
424            "flickering snowfire ruby: found on gigas berserker > sold 4,000 at The jeweler Etaenia by Nisugi"
425        );
426        let creatures =
427            answer(&ledger, &zoleta, "creatures", NOON + 700).unwrap_or_else(|e| panic!("{e}"));
428        assert!(creatures[1].starts_with("gigas berserker"), "{creatures:?}");
429        assert!(creatures[1].ends_with("4,000"), "{creatures:?}");
430        let cap = answer(&ledger, &nisugi, "cap", NOON + 700).unwrap_or_else(|e| panic!("{e}"));
431        assert_eq!(cap[2], "1 sold for 4,000");
432        assert!(answer(&ledger, &nisugi, "weekly", NOON).is_err());
433    }

435    #[test]
436    fn the_day_and_the_month_are_eastern() {
437        let (from, to) = day_bounds(NOON);
438        assert_eq!(eastern::format(eastern::date(from)), "2025-10-09");
439        assert_eq!(eastern::format(eastern::date(to)), "2025-10-10");
440        let (from, to) = month_bounds(NOON);
441        assert_eq!(eastern::format(eastern::date(from)), "2025-10-01");
442        assert_eq!(eastern::format(eastern::date(to)), "2025-11-01");
443        let (_, to) = month_bounds(eastern::midnight((2025, 12, 15)));
444        assert_eq!(eastern::format(eastern::date(to)), "2026-01-01");
445    }

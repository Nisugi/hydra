1//! The keeper: one a character, holding which item row each id of the
2//! session names, and writing both as they are first seen.
3//!
4//! An id seen for the first time with no item to belong to makes a new
5//! item. An id learned to be an old item's new one ([`Identity::renamed`])
6//! joins that item's ids. Every sight moves the item's and the id's
7//! `last_seen`.
8
9use std::collections::HashMap;
10
11use hydra_ledger::{Error, Ledger, Who};
12use rusqlite::{Connection, params};
13
14/// How long an id seen by another character still names the same thing:
15/// a day, well past any hunt and short of the game reusing it.
16const SAME_ID_SECONDS: i64 = 24 * 60 * 60;
17
18/// An item's row id in the ledger.
19#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
20pub struct ItemId(pub i64);
21
22/// The keeper of one character's items.
23#[derive(Debug)]
24pub struct Identity {
25    conn: Connection,
26    who: Who,
27    /// The item each id of this session names.
28    by_exist: HashMap<String, ItemId>,
29}
30
31impl Identity {
32    /// A keeper for `who`, writing to `ledger`, knowing no id yet.
33    ///
34    /// # Errors
35    ///
36    /// The ledger cannot be opened for writing.
37    pub fn open(ledger: &Ledger, who: Who) -> Result<Self, Error> {
38        Ok(Self {
39            conn: ledger.writing()?,
40            who,
41            by_exist: HashMap::new(),
42        })
43    }
44
45    /// The item the id `exist` names, seen at `at` as `name` (its noun
46    /// `noun`): the one already known for it, or a new one made now.
47    ///
48    /// # Errors
49    ///
50    /// The ledger cannot be written.
51    pub fn item(&mut self, exist: &str, name: &str, noun: &str, at: i64) -> Result<ItemId, Error> {
52        if let Some(id) = self.by_exist.get(exist).copied() {
53            self.seen(id, exist, at)?;
54            return Ok(id);
55        }
56        // Another character of the household saw this id lately: the game
57        // names one thing by one id for everyone in the room, so what they
58        // found is what this character holds.
59        if let Some(id) = self.seen_by_another(exist, at)? {
60            self.wears(id, exist, at)?;
61            return Ok(id);
62        }
63        self.conn.execute(
64            "INSERT INTO items (instance, character, name, noun, first_seen, last_seen)
65             VALUES (?1, ?2, ?3, ?4, ?5, ?5)",
66            params![self.who.instance, self.who.character, name, noun, at],
67        )?;
68        let id = ItemId(self.conn.last_insert_rowid());
69        self.wears(id, exist, at)?;
70        Ok(id)
71    }
72
73    /// The connection the keeper writes on, for rows written beside its own.
74    #[must_use]
75    pub fn connection(&self) -> &Connection {
76        &self.conn
77    }
78
79    /// The item an id names, if the keeper has seen the id.
80    #[must_use]
81    pub fn known(&self, exist: &str) -> Option<ItemId> {
82        self.by_exist.get(exist).copied()
83    }
84
85    /// The item of this character's that last wore the id `exist`, whenever
86    /// that was: an id of a session now over.
87    ///
88    /// # Errors
89    ///
90    /// The ledger cannot be read.
91    pub fn wore(&self, exist: &str) -> Result<Option<ItemId>, Error> {
92        let mut stmt = self.conn.prepare_cached(
93            "SELECT item_ids.item_id FROM item_ids
94             JOIN items ON items.id = item_ids.item_id
95             WHERE item_ids.exist_id = ?1 AND items.instance = ?2 AND items.character = ?3
96             ORDER BY item_ids.last_seen DESC LIMIT 1",
97        )?;
98        let found = stmt
99            .query_row(
100                params![exist, self.who.instance, self.who.character],
101                |row| row.get::<_, i64>(0),
102            )
103            .map(ItemId);
104        match found {
105            Ok(id) => Ok(Some(id)),
106            Err(rusqlite::Error::QueryReturnedNoRows) => Ok(None),
107            Err(e) => Err(e),
108        }
109    }
110
111    /// The id `exist`, seen at `at`, is the item `id`'s new one: after a
112    /// relog or a return from the pool.
113    ///
114    /// # Errors
115    ///
116    /// The ledger cannot be written.
117    pub fn renamed(&mut self, id: ItemId, exist: &str, at: i64) -> Result<(), Error> {
118        self.wears(id, exist, at)
119    }
120
121    /// Every id of this session forgotten: the session ended, and the next
122    /// one's ids are new.
123    pub fn session_ended(&mut self) {
124        self.by_exist.clear();
125    }
126
127    /// The item of this instance that wore the id `exist` within the last
128    /// day, if one did: another character's, seen in the same room.
129    fn seen_by_another(&self, exist: &str, at: i64) -> Result<Option<ItemId>, Error> {
130        let mut stmt = self.conn.prepare_cached(
131            "SELECT item_ids.item_id FROM item_ids
132             JOIN items ON items.id = item_ids.item_id
133             WHERE item_ids.exist_id = ?1 AND items.instance = ?2
134               AND item_ids.last_seen > ?3
135             ORDER BY item_ids.last_seen DESC LIMIT 1",
136        )?;
137        let found = stmt
138            .query_row(
139                params![exist, self.who.instance, at - SAME_ID_SECONDS],
140                |row| row.get::<_, i64>(0),
141            )
142            .map(ItemId);
143        match found {
144            Ok(id) => Ok(Some(id)),
145            Err(rusqlite::Error::QueryReturnedNoRows) => Ok(None),
146            Err(e) => Err(e),
147        }
148    }
149
150    /// The item `id` wears the id `exist` from `at`.
151    fn wears(&mut self, id: ItemId, exist: &str, at: i64) -> Result<(), Error> {
152        self.conn.execute(
153            "INSERT INTO item_ids (item_id, exist_id, first_seen, last_seen)
154             VALUES (?1, ?2, ?3, ?3)
155             ON CONFLICT (item_id, exist_id) DO UPDATE SET last_seen = excluded.last_seen",
156            params![id.0, exist, at],
157        )?;
158        self.by_exist.insert(exist.to_owned(), id);
159        self.seen(id, exist, at)
160    }
161
162    /// The item `id`, as `exist`, seen again at `at`.
163    fn seen(&self, id: ItemId, exist: &str, at: i64) -> Result<(), Error> {
164        self.conn.execute(
165            "UPDATE items SET last_seen = MAX(last_seen, ?2) WHERE id = ?1",
166            params![id.0, at],
167        )?;
168        self.conn.execute(
169            "UPDATE item_ids SET last_seen = MAX(last_seen, ?3) WHERE item_id = ?1 AND exist_id = ?2",
170            params![id.0, exist, at],
171        )?;
172        Ok(())
173    }
174}

201    fn ids_of(conn: &Connection, item: ItemId) -> Vec<(String, i64, i64)> {
202        let mut stmt = conn
203            .prepare("SELECT exist_id, first_seen, last_seen FROM item_ids WHERE item_id = ?1 ORDER BY first_seen")
204            .unwrap_or_else(|e| panic!("{e}"));
205        let rows = stmt
206            .query_map(params![item.0], |row| {
207                Ok((row.get(0)?, row.get(1)?, row.get(2)?))
208            })
209            .unwrap_or_else(|e| panic!("{e}"));
210        rows.map(|row| row.unwrap_or_else(|e| panic!("{e}")))
211            .collect()
212    }

214    #[test]
215    fn an_id_another_character_saw_lately_is_the_same_item() {
216        let scratch = Scratch::new("household");
217        let ledger = Ledger::open(scratch.file()).unwrap_or_else(|e| panic!("{e}"));
218        let mut hers =
219            Identity::open(&ledger, Who::new("prime", "Zoleta")).unwrap_or_else(|e| panic!("{e}"));
220        let mut his =
221            Identity::open(&ledger, Who::new("prime", "Nisugi")).unwrap_or_else(|e| panic!("{e}"));
222        let mut other = Identity::open(&ledger, Who::new("platinum", "Nisugi"))
223            .unwrap_or_else(|e| panic!("{e}"));
224        let ruby = hers
225            .item("892682691", "flickering snowfire ruby", "ruby", 1_000)
226            .unwrap_or_else(|e| panic!("{e}"));
227        let his_ruby = his
228            .item("892682691", "flickering snowfire ruby", "ruby", 1_300)
229            .unwrap_or_else(|e| panic!("{e}"));
230        assert_eq!(his_ruby, ruby, "the same id in the same room");
231        let elsewhere = other
232            .item("892682691", "flickering snowfire ruby", "ruby", 1_300)
233            .unwrap_or_else(|e| panic!("{e}"));
234        assert_ne!(elsewhere, ruby, "another instance's id is another thing");
235        let later = his
236            .item(
237                "892682692",
238                "some dashing samite boots",
239                "boots",
240                1_300 + SAME_ID_SECONDS,
241            )
242            .unwrap_or_else(|e| panic!("{e}"));
243        assert_ne!(later, ruby);
244    }

246    #[test]
247    fn an_id_makes_an_item_once_and_a_new_id_joins_it() {
248        let scratch = Scratch::new("identity");
249        let ledger = Ledger::open(scratch.file()).unwrap_or_else(|e| panic!("{e}"));
250        let who = Who::new("prime", "Nisugi");
251        let mut keeper = Identity::open(&ledger, who).unwrap_or_else(|e| panic!("{e}"));
252        let wand = keeper
253            .item("892676300", "white starstone wand", "wand", 100)
254            .unwrap_or_else(|e| panic!("{e}"));
255        let again = keeper
256            .item("892676300", "white starstone wand", "wand", 150)
257            .unwrap_or_else(|e| panic!("{e}"));
258        assert_eq!(wand, again);
259        assert_eq!(keeper.known("892676300"), Some(wand));
260        assert_eq!(keeper.known("1"), None);
261        // A relog: the session's ids are gone, and the wand wears a new one.
262        keeper.session_ended();
263        assert_eq!(keeper.known("892676300"), None);
264        keeper
265            .renamed(wand, "900000001", 200)
266            .unwrap_or_else(|e| panic!("{e}"));
267        assert_eq!(keeper.known("900000001"), Some(wand));
268        let conn = ledger.writing().unwrap_or_else(|e| panic!("{e}"));
269        assert_eq!(
270            ids_of(&conn, wand),
271            vec![
272                ("892676300".to_owned(), 100, 150),
273                ("900000001".to_owned(), 200, 200)
274            ]
275        );
276        let (name, last): (String, i64) = conn
277            .query_row(
278                "SELECT name, last_seen FROM items WHERE id = ?1",
279                params![wand.0],
280                |row| Ok((row.get(0)?, row.get(1)?)),
281            )
282            .unwrap_or_else(|e| panic!("{e}"));
283        assert_eq!(name, "white starstone wand");
284        assert_eq!(last, 200);
285    }

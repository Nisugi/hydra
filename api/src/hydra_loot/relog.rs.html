1//! The inventory before a logout and after the login, walked together.
2//!
3//! A relog gives everything carried a new id, but nothing moves: each
4//! container still holds the same things in the same order. So the two
5//! inventories are walked side by side, container by container, and an item
6//! with the same name in the same place in both is the same item, its old
7//! id now its new one. Two identical gems in one bag map to each other in
8//! order, which is right because they are the same to every report. An item
9//! in one inventory and not the other, dropped or picked up between the two,
10//! is left unmatched.
11//!
12//! The inventory is the game's whole-inventory answer (`_inventory
13//! manager`), asked quietly at login and again before a quit, each item
14//! with its container; the model folds the answer in, and this crate keeps
15//! what was carried in the ledger until the next login.
16
17use hydra_protocol::InventoryItem;
18
19/// One item as an inventory lists it: where it is and what it is.
20#[derive(Debug, Clone, PartialEq, Eq)]
21pub struct Carried {
22    /// The game's id for it, this session.
23    pub exist: String,
24    /// What holds it: a container's name, or the character's own hands or
25    /// body. Compared as given, so both inventories must name places the
26    /// same way.
27    pub place: String,
28    /// Its name, as the game shows it.
29    pub name: String,
30}
31
32/// The game's inventory answer as carried things, in the game's order: an
33/// item's place is its container's name, or how it sits on the body
34/// (`worn`) when the body holds it.
35#[must_use]
36pub fn carried(items: &[InventoryItem]) -> Vec<Carried> {
37    items
38        .iter()
39        .map(|item| Carried {
40            exist: item.id.clone(),
41            place: if item.parent == "player" {
42                item.relation.clone()
43            } else {
44                items
45                    .iter()
46                    .find(|holder| holder.id == item.parent)
47                    .map_or_else(|| item.parent.clone(), |holder| holder.name.clone())
48            },
49            name: item.name.clone(),
50        })
51        .collect()
52}
53
54/// Each item of `before` matched to one of `after`, as `(old id, new id)`,
55/// in `before`'s order. Within a place, items of one name are matched in
56/// the order given; one left over on either side is unmatched.
57#[must_use]
58pub fn walk(before: &[Carried], after: &[Carried]) -> Vec<(String, String)> {
59    let mut taken = vec![false; after.len()];
60    let mut pairs = Vec::new();
61    for old in before {
62        let found = after
63            .iter()
64            .enumerate()
65            .position(|(at, new)| !taken[at] && new.place == old.place && new.name == old.name);
66        if let Some(at) = found {
67            taken[at] = true;
68            pairs.push((old.exist.clone(), after[at].exist.clone()));
69        }
70    }
71    pairs
72}

74#[cfg(test)]
75mod tests {
76    use super::*;
77
78    #[test]
79    fn the_answer_s_items_take_their_container_s_name_as_their_place() {
80        let answer = [
81            InventoryItem {
82                id: "706".to_owned(),
83                relation: "worn".to_owned(),
84                parent: "player".to_owned(),
85                name: "a nacreous plumille cloak".to_owned(),
86                ..InventoryItem::default()
87            },
88            InventoryItem {
89                id: "758".to_owned(),
90                relation: "in".to_owned(),
91                parent: "706".to_owned(),
92                name: "a sapphire-set platinum crown".to_owned(),
93                ..InventoryItem::default()
94            },
95        ];
96        assert_eq!(
97            super::carried(&answer),
98            vec![
99                carried("706", "worn", "a nacreous plumille cloak"),
100                carried(
101                    "758",
102                    "a nacreous plumille cloak",
103                    "a sapphire-set platinum crown"
104                ),
105            ]
106        );
107    }
108
109    fn carried(exist: &str, place: &str, name: &str) -> Carried {
110        Carried {
111            exist: exist.to_owned(),
112            place: place.to_owned(),
113            name: name.to_owned(),
114        }
115    }
116
117    #[test]
118    fn the_same_things_in_the_same_places_pair_up_in_order() {
119        let before = [
120            carried("1", "cloak", "white starstone wand"),
121            carried("2", "cloak", "uncut diamond"),
122            carried("3", "cloak", "uncut diamond"),
123            carried("4", "right hand", "glowbark long bow"),
124            carried("5", "cloak", "dropped later"),
125        ];
126        let after = [
127            carried("11", "cloak", "white starstone wand"),
128            carried("12", "cloak", "uncut diamond"),
129            carried("13", "cloak", "uncut diamond"),
130            carried("14", "right hand", "glowbark long bow"),
131            carried("15", "cloak", "picked up later"),
132        ];
133        assert_eq!(
134            walk(&before, &after),
135            vec![
136                ("1".to_owned(), "11".to_owned()),
137                ("2".to_owned(), "12".to_owned()),
138                ("3".to_owned(), "13".to_owned()),
139                ("4".to_owned(), "14".to_owned()),
140            ]
141        );
142    }
143
144    #[test]
145    fn a_name_in_another_place_is_not_the_same_item() {
146        let before = [carried("1", "cloak", "uncut diamond")];
147        let after = [carried("11", "sack", "uncut diamond")];
148        assert!(walk(&before, &after).is_empty());
149    }

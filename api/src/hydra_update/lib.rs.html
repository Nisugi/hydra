1//! Hydra updating itself: a newer release looked for on the public releases
2//! page, fetched, its signature checked, and put in place of the running
3//! program, which is that release from its next start.
4//!
5//! The release job signs each release's checksums with the release key;
6//! Hydra holds the key's public half. A file is put in place only when the
7//! signature holds under it, the signature's signed comment names the
8//! release, and the file's hash is the one the checksums give it.
9//!
10//! - `release`: the releases listed, and the newest above this Hydra.
11//! - `signed`: the checksums, their signature checked.
12//! - `swap`: the new programs unpacked and put where the running ones are:
13//!   Hydra's, and those that come with it, such as `hydra-tui`.
14//!
15//! Hydra updates itself on Windows alone for now: elsewhere
16//! [`Releases::look`] finds nothing.
17
18mod release;
19mod signed;
20mod swap;

29/// The release key's public half: every release's checksums are signed
30/// with its secret half.
31const KEY: &str = "RWTnP4xymgNChF6vPB59oTv8tc6coKg6Svl9A5P0MtDOB/Enpl70UtLc";

33/// The public repository's releases, newest first, as GitHub's API lists
34/// them.
35const RELEASES: &str = "https://api.github.com/repos/Nisugi/hydra/releases?per_page=30";
36
37/// How long a connection may take to open, and how long the network may go
38/// quiet in the middle of an answer, before Hydra gives up on it.
39const QUIET: Duration = Duration::from_secs(30);
40
41/// The most a listing, the checksums, their signature and a release file
42/// may be: an answer larger than these is no release of Hydra's.
43const LISTING_MOST: u64 = 4 * 1024 * 1024;
44const SUMS_MOST: u64 = 64 * 1024;
45const SIGNATURE_MOST: u64 = 4 * 1024;
46const FILE_MOST: u64 = 256 * 1024 * 1024;
47
48/// The release file for this system, the program inside it, and the
49/// programs that come with it, each beside Hydra's.
50#[derive(Clone, Copy)]
51struct Build {
52    file: &'static str,
53    program: &'static str,
54    with: &'static [&'static str],
55}
56
57#[cfg(all(windows, target_arch = "x86_64"))]
58const BUILD: Option<Build> = Some(Build {
59    file: "hydra-windows-x86_64.zip",
60    program: "hydra.exe",
61    with: &["hydra-tui.exe"],
62});
63#[cfg(not(all(windows, target_arch = "x86_64")))]
64const BUILD: Option<Build> = None;

66/// The public releases, reached over a client of their own.
67#[derive(Clone, Debug)]
68pub struct Releases {
69    client: reqwest::Client,
70}
71
72impl Releases {
73    /// The releases, reached with a user agent: GitHub's API refuses a
74    /// request with none.
75    ///
76    /// # Errors
77    ///
78    /// The client could not be made: why.
79    pub fn new() -> Result<Self, String> {
80        let client = reqwest::Client::builder()
81            .user_agent(concat!("hydra/", env!("CARGO_PKG_VERSION")))
82            .connect_timeout(QUIET)
83            .read_timeout(QUIET)
84            .build()
85            .map_err(|why| format!("Hydra cannot reach the releases: {why}"))?;
86        Ok(Self { client })
87    }
88
89    /// The newest release above `own`, this Hydra's version, with a build
90    /// for this system and signed checksums: none when there is no such
91    /// release, or Hydra does not update itself here.
92    ///
93    /// # Errors
94    ///
95    /// The releases could not be listed or read: why.
96    pub async fn look(&self, own: &Version) -> Result<Option<Offer>, String> {
97        let Some(build) = BUILD else {
98            return Ok(None);
99        };
100        let listing = text(&self.client, RELEASES, LISTING_MOST).await?;
101        release::newest(&listing, own, build.file)
102    }
103
104    /// Fetch `offer`, check it against its signed checksums, and put its
105    /// program in place of `exe`, the program running: from its next start,
106    /// Hydra is that release. `progress` is told the bytes of the release
107    /// file fetched so far, and how many it has.
108    ///
109    /// # Errors
110    ///
111    /// A file could not be fetched, the signature or a hash does not hold,
112    /// or the program could not be put in place: why. Nothing is put in
113    /// place unless every check held.
114    pub async fn fetch(
115        &self,
116        offer: &Offer,
117        exe: &Path,
118        progress: impl FnMut(u64, u64),
119    ) -> Result<(), String> {
120        let Some(build) = BUILD else {
121            return Err("Hydra does not update itself on this system".to_owned());
122        };
123        let file = self.checked(offer, build, progress).await?;
124        let exe = exe.to_path_buf();
125        tokio::task::spawn_blocking(move || {
126            let mut programs = vec![(exe.clone(), swap::unpack(&file, build.program)?)];
127            // Each program that comes with Hydra, where the release has it:
128            // one before it came has none.
129            programs.extend(unpacked(&file, &exe, build.with)?);
130            swap::put_in_place(&programs)
131        })
132        .await
133        .map_err(|why| format!("The update stopped: {why}"))?
134    }
135
136    /// Put back each program that comes with Hydra and is missing beside
137    /// `exe`, from `own`'s release, this Hydra's: the names put back. An
138    /// update by a Hydra from before a program came with it brings none,
139    /// and a build of no release has none to put back.
140    ///
141    /// # Errors
142    ///
143    /// The release could not be fetched or checked, or a program put in
144    /// place: why.
145    pub async fn repair(&self, own: &Version, exe: &Path) -> Result<Vec<String>, String> {
146        let Some(build) = BUILD else {
147            return Ok(Vec::new());
148        };
149        let missing: Vec<&'static str> = build
150            .with
151            .iter()
152            .copied()
153            .filter(|name| !exe.with_file_name(name).exists())
154            .collect();
155        if missing.is_empty() {
156            return Ok(Vec::new());
157        }
158        let listing = text(&self.client, RELEASES, LISTING_MOST).await?;
159        let Some(offer) = release::exact(&listing, own, build.file)? else {
160            return Ok(Vec::new());
161        };
162        let file = self.checked(&offer, build, |_, _| {}).await?;
163        let exe = exe.to_path_buf();
164        tokio::task::spawn_blocking(move || {
165            let programs = unpacked(&file, &exe, &missing)?;
166            swap::put_in_place(&programs)?;
167            Ok(programs
168                .iter()
169                .filter_map(|(path, _)| path.file_name())
170                .map(|name| name.to_string_lossy().into_owned())
171                .collect())
172        })
173        .await
174        .map_err(|why| format!("The repair stopped: {why}"))?
175    }
176
177    /// `offer`'s release file for `build`, fetched and checked against its
178    /// signed checksums; `progress` told the bytes fetched so far, and how
179    /// many it has.
180    async fn checked(
181        &self,
182        offer: &Offer,
183        build: Build,
184        progress: impl FnMut(u64, u64),
185    ) -> Result<Vec<u8>, String> {
186        let sums = text(&self.client, &offer.sums.url, SUMS_MOST).await?;
187        let signature = text(&self.client, &offer.signature.url, SIGNATURE_MOST).await?;
188        let sums = signed::Sums::checked(&sums, &signature, &offer.tag, KEY)?;
189        let file = body(
190            &self.client,
191            &offer.file.url,
192            FILE_MOST,
193            offer.file.size,
194            progress,
195        )
196        .await?;
197        sums.check(build.file, &file)?;
198        Ok(file)
199    }
200}

202/// Each of `names` the release file `file` has, read out of it, with the
203/// path it goes to beside `exe`.
204///
205/// # Errors
206///
207/// `file` is no zip, or a program cannot be read out of it: why.
208fn unpacked(
209    file: &[u8],
210    exe: &Path,
211    names: &[&str],
212) -> Result<Vec<(std::path::PathBuf, Vec<u8>)>, String> {
213    let mut programs = Vec::new();
214    for name in names {
215        if let Some(bytes) = swap::unpack_if(file, name)? {
216            programs.push((exe.with_file_name(name), bytes));
217        }
218    }
219    Ok(programs)
220}

222/// Take away the programs an update set aside beside `exe`, Hydra's, and
223/// those that come with it, now no longer running: as Hydra starts. One
224/// still running is left for the next start.
225pub fn clear_old(exe: &Path) {
226    swap::clear_old(exe);
227    if let Some(build) = BUILD {
228        for name in build.with {
229            swap::clear_old(&exe.with_file_name(name));
230        }
231    }
232}

234/// The text at `url`, at most `most` bytes.
235async fn text(client: &reqwest::Client, url: &str, most: u64) -> Result<String, String> {
236    let bytes = body(client, url, most, 0, |_, _| {}).await?;
237    String::from_utf8(bytes).map_err(|_| format!("{url} is not text"))
238}
239
240/// The bytes at `url`, at most `most`; `progress` is told the bytes so far
241/// and how many the answer has, or `listed` when it does not say.
242async fn body(
243    client: &reqwest::Client,
244    url: &str,
245    most: u64,
246    listed: u64,
247    mut progress: impl FnMut(u64, u64),
248) -> Result<Vec<u8>, String> {
249    let too_large = || format!("{url} is larger than any release of Hydra's");
250    let mut response = client
251        .get(url)
252        .send()
253        .await
254        .and_then(reqwest::Response::error_for_status)
255        .map_err(|why| format!("{url}: {why}"))?;
256    let total = response.content_length().unwrap_or(listed);
257    if total > most {
258        return Err(too_large());
259    }
260    let mut got = Vec::with_capacity(usize::try_from(total).unwrap_or(0));
261    while let Some(chunk) = response
262        .chunk()
263        .await
264        .map_err(|why| format!("{url}: {why}"))?
265    {
266        got.extend_from_slice(&chunk);
267        let so_far = got.len() as u64;
268        if so_far > most {
269            return Err(too_large());
270        }
271        progress(so_far, total);
272    }
273    Ok(got)
274}

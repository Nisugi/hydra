1//! Hydra updating itself: a newer release looked for on the public releases
2//! page, fetched, its signature checked, and put in place of the running
3//! program, which is that release from its next start.
4//!
5//! The release job signs each release's checksums with the release key;
6//! Hydra holds the key's public half. A file is put in place only when the
7//! signature holds under it, the signature's signed comment names the
8//! release, and the file's hash is the one the checksums give it.
9//!
10//! - `release`: the releases listed, and the newest above this Hydra.
11//! - `signed`: the checksums, their signature checked.
12//! - `swap`: the new program unpacked and put where the running one is.
13//!
14//! Hydra updates itself on Windows alone for now: elsewhere
15//! [`Releases::look`] finds nothing.
16
17mod release;
18mod signed;
19mod swap;

29/// The release key's public half: every release's checksums are signed
30/// with its secret half.
31const KEY: &str = "RWTnP4xymgNChF6vPB59oTv8tc6coKg6Svl9A5P0MtDOB/Enpl70UtLc";

33/// The public repository's releases, newest first, as GitHub's API lists
34/// them.
35const RELEASES: &str = "https://api.github.com/repos/Nisugi/hydra/releases?per_page=30";
36
37/// How long a connection may take to open, and how long the network may go
38/// quiet in the middle of an answer, before Hydra gives up on it.
39const QUIET: Duration = Duration::from_secs(30);
40
41/// The most a listing, the checksums, their signature and a release file
42/// may be: an answer larger than these is no release of Hydra's.
43const LISTING_MOST: u64 = 4 * 1024 * 1024;
44const SUMS_MOST: u64 = 64 * 1024;
45const SIGNATURE_MOST: u64 = 4 * 1024;
46const FILE_MOST: u64 = 256 * 1024 * 1024;
47
48/// The release file for this system, and the program inside it.
49#[derive(Clone, Copy)]
50struct Build {
51    file: &'static str,
52    program: &'static str,
53}
54
55#[cfg(all(windows, target_arch = "x86_64"))]
56const BUILD: Option<Build> = Some(Build {
57    file: "hydra-windows-x86_64.zip",
58    program: "hydra.exe",
59});
60#[cfg(not(all(windows, target_arch = "x86_64")))]
61const BUILD: Option<Build> = None;

63/// The public releases, reached over a client of their own.
64#[derive(Clone, Debug)]
65pub struct Releases {
66    client: reqwest::Client,
67}
68
69impl Releases {
70    /// The releases, reached with a user agent: GitHub's API refuses a
71    /// request with none.
72    ///
73    /// # Errors
74    ///
75    /// The client could not be made: why.
76    pub fn new() -> Result<Self, String> {
77        let client = reqwest::Client::builder()
78            .user_agent(concat!("hydra/", env!("CARGO_PKG_VERSION")))
79            .connect_timeout(QUIET)
80            .read_timeout(QUIET)
81            .build()
82            .map_err(|why| format!("Hydra cannot reach the releases: {why}"))?;
83        Ok(Self { client })
84    }
85
86    /// The newest release above `own`, this Hydra's version, with a build
87    /// for this system and signed checksums: none when there is no such
88    /// release, or Hydra does not update itself here.
89    ///
90    /// # Errors
91    ///
92    /// The releases could not be listed or read: why.
93    pub async fn look(&self, own: &Version) -> Result<Option<Offer>, String> {
94        let Some(build) = BUILD else {
95            return Ok(None);
96        };
97        let listing = text(&self.client, RELEASES, LISTING_MOST).await?;
98        release::newest(&listing, own, build.file)
99    }
100
101    /// Fetch `offer`, check it against its signed checksums, and put its
102    /// program in place of `exe`, the program running: from its next start,
103    /// Hydra is that release. `progress` is told the bytes of the release
104    /// file fetched so far, and how many it has.
105    ///
106    /// # Errors
107    ///
108    /// A file could not be fetched, the signature or a hash does not hold,
109    /// or the program could not be put in place: why. Nothing is put in
110    /// place unless every check held.
111    pub async fn fetch(
112        &self,
113        offer: &Offer,
114        exe: &Path,
115        progress: impl FnMut(u64, u64),
116    ) -> Result<(), String> {
117        let Some(build) = BUILD else {
118            return Err("Hydra does not update itself on this system".to_owned());
119        };
120        let sums = text(&self.client, &offer.sums.url, SUMS_MOST).await?;
121        let signature = text(&self.client, &offer.signature.url, SIGNATURE_MOST).await?;
122        let sums = signed::Sums::checked(&sums, &signature, &offer.tag, KEY)?;
123        let file = body(
124            &self.client,
125            &offer.file.url,
126            FILE_MOST,
127            offer.file.size,
128            progress,
129        )
130        .await?;
131        let exe = exe.to_path_buf();
132        tokio::task::spawn_blocking(move || {
133            sums.check(build.file, &file)?;
134            let program = swap::unpack(&file, build.program)?;
135            swap::put_in_place(&exe, &program)
136        })
137        .await
138        .map_err(|why| format!("The update stopped: {why}"))?
139    }
140}

142/// The text at `url`, at most `most` bytes.
143async fn text(client: &reqwest::Client, url: &str, most: u64) -> Result<String, String> {
144    let bytes = body(client, url, most, 0, |_, _| {}).await?;
145    String::from_utf8(bytes).map_err(|_| format!("{url} is not text"))
146}
147
148/// The bytes at `url`, at most `most`; `progress` is told the bytes so far
149/// and how many the answer has, or `listed` when it does not say.
150async fn body(
151    client: &reqwest::Client,
152    url: &str,
153    most: u64,
154    listed: u64,
155    mut progress: impl FnMut(u64, u64),
156) -> Result<Vec<u8>, String> {
157    let too_large = || format!("{url} is larger than any release of Hydra's");
158    let mut response = client
159        .get(url)
160        .send()
161        .await
162        .and_then(reqwest::Response::error_for_status)
163        .map_err(|why| format!("{url}: {why}"))?;
164    let total = response.content_length().unwrap_or(listed);
165    if total > most {
166        return Err(too_large());
167    }
168    let mut got = Vec::with_capacity(usize::try_from(total).unwrap_or(0));
169    while let Some(chunk) = response
170        .chunk()
171        .await
172        .map_err(|why| format!("{url}: {why}"))?
173    {
174        got.extend_from_slice(&chunk);
175        let so_far = got.len() as u64;
176        if so_far > most {
177            return Err(too_large());
178        }
179        progress(so_far, total);
180    }
181    Ok(got)
182}

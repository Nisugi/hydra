1//! A release's checksums, their signature checked against the release key,
2//! and a release file checked against them.
3
4use minisign_verify::{PublicKey, Signature};
5use sha2::{Digest, Sha256};
6
7/// The checksums a release's signature holds for: each file's name, and its
8/// SHA-256 in lowercase hex.
9#[derive(Debug)]
10pub(crate) struct Sums(Vec<(String, String)>);
11
12impl Sums {
13    /// `sums`, the checksums file as fetched, once `signature` holds for it
14    /// under `key` and its signed comment is `Hydra <tag>`: an older
15    /// release's signed checksums cannot stand in for this one's.
16    ///
17    /// # Errors
18    ///
19    /// The key or the signature cannot be read, the signature does not
20    /// hold, or it names another release: why.
21    pub(crate) fn checked(
22        sums: &str,
23        signature: &str,
24        tag: &str,
25        key: &str,
26    ) -> Result<Self, String> {
27        let key = PublicKey::from_base64(key)
28            .map_err(|why| format!("The release key cannot be read: {why}"))?;
29        let signature = Signature::decode(signature)
30            .map_err(|why| format!("The checksums' signature cannot be read: {why}"))?;
31        key.verify(sums.as_bytes(), &signature, false)
32            .map_err(|why| format!("The checksums' signature does not hold: {why}"))?;
33        let named = format!("Hydra {tag}");
34        if signature.trusted_comment() != named {
35            return Err(format!(
36                "The checksums are signed for \"{}\", not {named}",
37                signature.trusted_comment()
38            ));
39        }
40        // `sha256sum` writes the hash, a space, and the name after a space
41        // or, read as binary, a `*`.
42        let listed = sums
43            .lines()
44            .filter_map(|line| {
45                let (hash, name) = line.split_once(' ')?;
46                let name = name.strip_prefix([' ', '*'])?;
47                Some((name.to_owned(), hash.to_ascii_lowercase()))
48            })
49            .collect();
50        Ok(Self(listed))
51    }
52
53    /// Whether `bytes`, the release file `name`, has the hash the checksums
54    /// give it.
55    ///
56    /// # Errors
57    ///
58    /// The checksums do not list `name`, or its hash differs: why.
59    pub(crate) fn check(&self, name: &str, bytes: &[u8]) -> Result<(), String> {
60        let Some((_, wanted)) = self.0.iter().find(|(listed, _)| listed == name) else {
61            return Err(format!("The release's checksums do not list {name}"));
62        };
63        if hex(&Sha256::digest(bytes)) == *wanted {
64            Ok(())
65        } else {
66            Err(format!(
67                "{name} is not the file the release signed: its hash differs"
68            ))
69        }
70    }
71}
72
73/// `bytes` in lowercase hex.
74fn hex(bytes: &[u8]) -> String {
75    const DIGITS: &[u8; 16] = b"0123456789abcdef";
76    bytes
77        .iter()
78        .flat_map(|&byte| {
79            [
80                DIGITS[usize::from(byte >> 4)],
81                DIGITS[usize::from(byte & 0xf)],
82            ]
83        })
84        .map(char::from)
85        .collect()
86}

88#[cfg(test)]
89mod tests {
90    use super::*;
91    use std::io::Cursor;
92
93    /// The checksums of v0.1.0-beta.1 as the release published them.
94    const RECORDED: &str = include_str!("../tests/fixtures/SHA256SUMS.txt");
95
96    /// SHA-256 of `abc`, the standard's own example.
97    const ABC: &str = "ba7816bf8f01cfea414140de5dae2223b00361a396177a9cb410ff61f20015ad";
98
99    /// A key of the test's own: its public half, as Hydra holds one, and a
100    /// signer of text under a signed comment, as the release job signs.
101    fn key() -> (String, impl Fn(&str, &str) -> String) {
102        let pair = minisign::KeyPair::generate_unencrypted_keypair().unwrap();
103        let public = pair.pk.to_base64();
104        let sign = move |text: &str, comment: &str| {
105            minisign::sign(
106                Some(&pair.pk),
107                &pair.sk,
108                Cursor::new(text.as_bytes()),
109                Some(comment),
110                None,
111            )
112            .unwrap()
113            .into_string()
114        };
115        (public, sign)
116    }
117
118    /// Signed under the key and naming the release, the checksums hold, and
119    /// a file is checked against them: its own bytes pass, any other fail.
120    #[test]
121    fn a_file_is_checked_against_checksums_that_hold() {
122        let (public, sign) = key();
123        let sums = format!("{ABC}  hydra-windows-x86_64.zip\n");
124        let signature = sign(&sums, "Hydra v0.1.0-beta.2");
125        let checked = Sums::checked(&sums, &signature, "v0.1.0-beta.2", &public).unwrap();
126        assert_eq!(checked.check("hydra-windows-x86_64.zip", b"abc"), Ok(()));
127        assert!(checked.check("hydra-windows-x86_64.zip", b"abd").is_err());
128        assert!(checked.check("hydra-linux-x86_64.tar.gz", b"abc").is_err());
129    }
130
131    /// The recorded checksums are read line by line: a file they list is
132    /// checked against its hash, not taken as unlisted.
133    #[test]
134    fn the_recorded_checksums_are_read() {
135        let (public, sign) = key();
136        let signature = sign(RECORDED, "Hydra v0.1.0-beta.1");
137        let checked = Sums::checked(RECORDED, &signature, "v0.1.0-beta.1", &public).unwrap();
138        assert_eq!(checked.0.len(), 4);
139        let said = checked
140            .check("hydra-windows-x86_64.zip", b"not the zip")
141            .unwrap_err();
142        assert!(said.contains("hash differs"), "{said}");
143    }
144
145    /// A name read as binary, `*` before it, is read as the name.
146    #[test]
147    fn a_name_read_as_binary_is_read() {
148        let (public, sign) = key();
149        let sums = format!("{ABC} *hydra-windows-x86_64.zip\n");
150        let signature = sign(&sums, "Hydra v0.1.0-beta.2");
151        let checked = Sums::checked(&sums, &signature, "v0.1.0-beta.2", &public).unwrap();
152        assert_eq!(checked.check("hydra-windows-x86_64.zip", b"abc"), Ok(()));
153    }
154
155    /// Signed by another key, the checksums are refused.
156    #[test]
157    fn another_keys_signature_is_refused() {
158        let (public, _) = key();
159        let (_, other) = key();
160        let sums = format!("{ABC}  hydra-windows-x86_64.zip\n");
161        let signature = other(&sums, "Hydra v0.1.0-beta.2");
162        assert!(Sums::checked(&sums, &signature, "v0.1.0-beta.2", &public).is_err());
163    }
164
165    /// Checksums changed after they were signed are refused.
166    #[test]
167    fn checksums_changed_after_signing_are_refused() {
168        let (public, sign) = key();
169        let sums = format!("{ABC}  hydra-windows-x86_64.zip\n");
170        let signature = sign(&sums, "Hydra v0.1.0-beta.2");
171        let changed = sums.replace("ba78", "0000");
172        let said = Sums::checked(&changed, &signature, "v0.1.0-beta.2", &public).unwrap_err();
173        assert!(said.contains("does not hold"), "{said}");
174    }
175
176    /// An older release's checksums, signed and sound, are refused for a
177    /// newer release: the signed comment names the one they were made for.
178    #[test]
179    fn checksums_signed_for_another_release_are_refused() {
180        let (public, sign) = key();
181        let sums = format!("{ABC}  hydra-windows-x86_64.zip\n");
182        let signature = sign(&sums, "Hydra v0.1.0-beta.1");
183        let said = Sums::checked(&sums, &signature, "v0.1.0-beta.2", &public).unwrap_err();
184        assert!(said.contains("v0.1.0-beta.1"), "{said}");
185    }
186
187    /// A signature that is no signature is refused, not taken as one.
188    #[test]
189    fn a_signature_unread_is_refused() {
190        let (public, _) = key();
191        let sums = format!("{ABC}  hydra-windows-x86_64.zip\n");
192        assert!(Sums::checked(&sums, "Not Found", "v0.1.0-beta.2", &public).is_err());
193    }

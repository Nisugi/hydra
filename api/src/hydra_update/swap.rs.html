1//! The new programs taken out of their release file and put where the
2//! running ones are: Hydra's, and those that come with it.
3//!
4//! Windows lets no one write over or take away a program while it runs, but
5//! lets it be renamed. So each new program is written beside its old one
6//! first; then each old one is set aside under a name of its own --
7//! `hydra.old.exe`, or `hydra.old-1.exe` while a terminal still runs the
8//! one an update set aside before -- and each new one takes its name. A step
9//! that fails puts every program back as it was, so the programs are never
10//! of two releases. What was set aside goes as Hydra next starts, once
11//! nothing runs it ([`clear_old`]).
12
13use std::io::{Cursor, Read as _, Write as _};
14use std::path::{Path, PathBuf};

16/// The program `program`, read out of `zip`, the release file.
17///
18/// # Errors
19///
20/// `zip` is no zip, or has no `program`: why.
21pub(crate) fn unpack(zip: &[u8], program: &str) -> Result<Vec<u8>, String> {
22    let mut archive = zip::ZipArchive::new(Cursor::new(zip))
23        .map_err(|why| format!("The release file cannot be opened: {why}"))?;
24    let mut entry = archive
25        .by_name(program)
26        .map_err(|why| format!("The release file has no {program}: {why}"))?;
27    let mut bytes = Vec::with_capacity(usize::try_from(entry.size()).unwrap_or(0));
28    entry
29        .read_to_end(&mut bytes)
30        .map_err(|why| format!("{program} cannot be read out of the release file: {why}"))?;
31    Ok(bytes)
32}

34/// The program `program`, read out of `zip`, when it is there: none when
35/// the release has no such program.
36///
37/// # Errors
38///
39/// `zip` is no zip, or `program` cannot be read out of it: why.
40pub(crate) fn unpack_if(zip: &[u8], program: &str) -> Result<Option<Vec<u8>>, String> {
41    let archive = zip::ZipArchive::new(Cursor::new(zip))
42        .map_err(|why| format!("The release file cannot be opened: {why}"))?;
43    if archive.index_for_name(program).is_none() {
44        return Ok(None);
45    }
46    unpack(zip, program).map(Some)
47}

49/// Put each program in place of its path, all or none: from their next
50/// start, the paths are the new programs. A path with no program yet is
51/// given one.
52///
53/// # Errors
54///
55/// A program could not be written, or a path could not be moved aside or
56/// given its new program: why. Every path is then as it was, unless even
57/// putting one back failed, which the error says, naming what to rename.
58pub(crate) fn put_in_place(programs: &[(PathBuf, Vec<u8>)]) -> Result<(), String> {
59    // Each written beside its path first.
60    let mut staged: Vec<PathBuf> = Vec::new();
61    for (exe, bytes) in programs {
62        let new = beside(exe, "new");
63        let written = write(&new, bytes).and_then(|()| runnable(exe, &new));
64        staged.push(new);
65        if let Err(why) = written {
66            remove_all(&staged);
67            return Err(why);
68        }
69    }
70    // Each old one set aside, then each new one given its name.
71    let mut done = Done::default();
72    let swapped = programs
73        .iter()
74        .zip(&staged)
75        .try_for_each(|((exe, _), new)| done.swap(exe, new));
76    remove_all(&staged);
77    match swapped {
78        // What no program runs goes at once; one still running is left for
79        // the next start.
80        Ok(()) => {
81            for (_, old) in &done.aside {
82                let _ = std::fs::remove_file(old);
83            }
84            Ok(())
85        }
86        Err(why) => match done.undo() {
87            Ok(()) => Err(why),
88            Err(stuck) => Err(format!("{why}; and {stuck}")),
89        },
90    }
91}

93/// What a swap has done, to undo.
94#[derive(Default)]
95struct Done {
96    /// Each path set aside, and where.
97    aside: Vec<(PathBuf, PathBuf)>,
98    /// Each path given its new program.
99    placed: Vec<PathBuf>,
100}
101
102impl Done {
103    /// `exe` set aside, when it is there, and `new`, written beside it,
104    /// given its name.
105    fn swap(&mut self, exe: &Path, new: &Path) -> Result<(), String> {
106        if exe.exists() {
107            let old = free_old(exe);
108            aside(exe, &old)
109                .map_err(|why| format!("{} cannot be moved aside: {why}", exe.display()))?;
110            self.aside.push((exe.to_path_buf(), old));
111        }
112        std::fs::rename(new, exe)
113            .map_err(|why| format!("{} cannot be put in place: {why}", exe.display()))?;
114        self.placed.push(exe.to_path_buf());
115        Ok(())
116    }
117
118    /// Every path as it was: each new program taken away, each old one
119    /// given its name back.
120    fn undo(self) -> Result<(), String> {
121        remove_all(&self.placed);
122        let mut stuck = Vec::new();
123        for (exe, old) in self.aside.iter().rev() {
124            if std::fs::rename(old, exe).is_err() {
125                stuck.push(format!(
126                    "{} could not be put back: rename it to {} by hand before Hydra's next start",
127                    old.display(),
128                    exe.display()
129                ));
130            }
131        }
132        if stuck.is_empty() {
133            Ok(())
134        } else {
135            Err(stuck.join("; "))
136        }
137    }
138}

140/// `bytes` written to `path`, all of it on the disk.
141fn write(path: &Path, bytes: &[u8]) -> Result<(), String> {
142    std::fs::File::create(path)
143        .and_then(|mut file| {
144            file.write_all(bytes)?;
145            file.sync_all()
146        })
147        .map_err(|why| format!("{} cannot be written: {why}", path.display()))
148}
149
150/// `new` made runnable where it goes: `exe`'s permissions when there is an
151/// `exe`, a program's otherwise.
152fn runnable(exe: &Path, new: &Path) -> Result<(), String> {
153    let made = match std::fs::metadata(exe) {
154        Ok(running) => std::fs::set_permissions(new, running.permissions()),
155        #[cfg(unix)]
156        Err(_) => {
157            use std::os::unix::fs::PermissionsExt as _;
158            std::fs::set_permissions(new, std::fs::Permissions::from_mode(0o755))
159        }
160        #[cfg(not(unix))]
161        Err(_) => Ok(()),
162    };
163    made.map_err(|why| format!("{} cannot be made runnable: {why}", new.display()))
164}

166/// Each of `paths` taken away, where it is.
167fn remove_all(paths: &[PathBuf]) {
168    for path in paths {
169        let _ = std::fs::remove_file(path);
170    }
171}

173/// The name `exe` is set aside under: `old`, or the first of `old-1`,
174/// `old-2` and on that is free, one a program still runs being kept.
175fn free_old(exe: &Path) -> PathBuf {
176    (0_u32..1000)
177        .map(|at| {
178            if at == 0 {
179                beside(exe, "old")
180            } else {
181                beside(exe, &format!("old-{at}"))
182            }
183        })
184        .find(|old| match std::fs::remove_file(old) {
185            Ok(()) => true,
186            Err(why) => why.kind() == std::io::ErrorKind::NotFound,
187        })
188        .unwrap_or_else(|| beside(exe, "old"))
189}

191/// `exe` renamed `old`. Windows holds a program's file a moment as it starts,
192/// and a virus scanner as it reads a file new to it: one held so is tried
193/// again for up to a second.
194fn aside(exe: &Path, old: &Path) -> std::io::Result<()> {
195    /// Windows' "being used by another process".
196    const SHARING_VIOLATION: i32 = 32;
197    let mut tries = 0;
198    loop {
199        match std::fs::rename(exe, old) {
200            Err(why) if why.raw_os_error() == Some(SHARING_VIOLATION) && tries < 20 => {
201                tries += 1;
202                std::thread::sleep(std::time::Duration::from_millis(50));
203            }
204            done => return done,
205        }
206    }
207}

209/// Take away every program an update set aside beside `exe` --
210/// `hydra.old.exe`, `hydra.old-1.exe` and on -- now no longer running: as
211/// Hydra starts. One still running is left for the next start.
212pub(crate) fn clear_old(exe: &Path) {
213    let (Some(folder), Some(stem)) = (exe.parent(), exe.file_stem()) else {
214        return;
215    };
216    let Ok(found) = std::fs::read_dir(folder) else {
217        return;
218    };
219    let lead = format!("{}.old", stem.to_string_lossy());
220    let tail = exe.extension().map_or_else(String::new, |extension| {
221        format!(".{}", extension.to_string_lossy())
222    });
223    for entry in found.flatten() {
224        let name = entry.file_name();
225        let name = name.to_string_lossy();
226        let Some(between) = name
227            .strip_prefix(lead.as_str())
228            .and_then(|rest| rest.strip_suffix(tail.as_str()))
229        else {
230            continue;
231        };
232        let numbered = between
233            .strip_prefix('-')
234            .is_some_and(|at| !at.is_empty() && at.bytes().all(|b| b.is_ascii_digit()));
235        if between.is_empty() || numbered {
236            let _ = std::fs::remove_file(entry.path());
237        }
238    }
239}

241/// `exe` with `mark` before its extension: `hydra.exe` and `new` make
242/// `hydra.new.exe`; `hydra` makes `hydra.new`.
243fn beside(exe: &Path, mark: &str) -> PathBuf {
244    let stem = exe.file_stem().unwrap_or_default().to_string_lossy();
245    let name = match exe.extension() {
246        Some(extension) => format!("{stem}.{mark}.{}", extension.to_string_lossy()),
247        None => format!("{stem}.{mark}"),
248    };
249    exe.with_file_name(name)
250}

263    /// A zip as the Windows release is made: `hydra.exe` in it, deflated.
264    fn release_zip(program: &[u8]) -> Vec<u8> {
265        let mut zip = zip::ZipWriter::new(Cursor::new(Vec::new()));
266        let options = zip::write::SimpleFileOptions::default()
267            .compression_method(zip::CompressionMethod::Deflated);
268        zip.start_file("hydra.exe", options).unwrap();
269        zip.write_all(program).unwrap();
270        zip.finish().unwrap().into_inner()
271    }

413    /// The second program failing to be set aside, after the first was put
414    /// in place, puts the first back: the two are never of two releases.
415    #[cfg(windows)]
416    #[test]
417    fn a_failure_after_the_first_puts_it_back() {
418        let dir = scratch("undo");
419        let tui = dir.join("hydra-tui.exe");
420        std::fs::write(&tui, b"old terminal").unwrap();
421        // A folder a program runs in cannot be renamed: Hydra's place, here.
422        let hydra = dir.join("held");
423        std::fs::create_dir_all(&hydra).unwrap();
424        let mut running = run_from(&dir.join("pinger.exe"));
425        running.kill().unwrap();
426        running.wait().unwrap();
427        let mut holding = std::process::Command::new(dir.join("pinger.exe"))
428            .args(["-n", "30", "127.0.0.1"])
429            .current_dir(&hydra)
430            .stdout(std::process::Stdio::null())
431            .spawn()
432            .unwrap();
433        let failed = put_in_place(&[
434            (tui.clone(), b"new terminal".to_vec()),
435            (hydra.clone(), b"new hydra".to_vec()),
436        ]);
437        holding.kill().unwrap();
438        holding.wait().unwrap();
439        assert!(failed.is_err());
440        assert_eq!(std::fs::read(&tui).unwrap(), b"old terminal", "put back");
441        assert!(hydra.is_dir(), "left as it was");
442        assert!(!dir.join("hydra-tui.old.exe").exists());
443        assert!(!dir.join("hydra-tui.new.exe").exists());
444        assert!(!dir.join("held.new").exists());
445        let _ = std::fs::remove_dir_all(&dir);
446    }

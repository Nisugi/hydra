35/// A word after `attacks` or `creatures` that makes it a listing rather
36/// than one by name or id.
37const LISTING_WORDS: &[&str] = &["all", "last", "between", "recent", "page", "per"];
38
39/// What was asked.
40#[derive(Debug, Clone, PartialEq, Eq)]
41pub enum Asked {
42    /// The words.
43    Help,
44    /// A hunt's report: the latest, or by id; `full` lists every creature.
45    Session {
46        /// Which hunt; the latest when `None`.
47        id: Option<i64>,
48        /// Every creature instance listed.
49        full: bool,
50    },
51    /// The hunts over a scope, a line each.
52    Sessions(Scope),
53    /// The numbers over a scope.
54    Aggregate(Scope),
55    /// Damage by ability over a scope.
56    Abilities(Scope),
57    /// Every attack name by damage, or the newest attacks, a page.
58    Attacks(Listing),
59    /// One attack by row id.
60    AttackById(i64),
61    /// The nth instance of an attack by name, or the names.
62    AttackByName(Option<String>, Option<i64>),
63    /// Every creature by damage, or the newest, a page.
64    Creatures(Listing),
65    /// A creature's report.
66    Creature {
67        /// Its id or noun; the top twenty when `None`.
68        arg: Option<String>,
69        /// Which instance of the noun.
70        idx: Option<i64>,
71        /// The whole tree.
72        full: bool,
73        /// Every sequence, not just the ends.
74        all: bool,
75    },
76    /// A spawn tree by root.
77    Tree(i64),
78    /// A flare's analytics, or the names.
79    Flare(Option<String>),
80    /// The flares that fire with one.
81    FlareAlongside(String),
82    /// Defensive readings, or the nouns.
83    Defense(Option<String>),
84    /// HP estimates.
85    Hp(Option<String>),
86    /// An analytics report over a scope, with an assumed aim location.
87    Analytics(Kind, Scope, String),
88    /// Two hunts compared.
89    Compare(Vec<i64>),
90    /// Each bark cast and its absorbs.
91    Barkskin,
92    /// Every barrier together over a scope.
93    Intercepts(Scope),
94}

96/// The trailing scope words taken off `words`: `all`, `last <n>`,
97/// `between <id> <id>`, or an id; the latest hunt when none.
98fn scope_from(words: &mut Vec<String>) -> Scope {
99    let last = words.last().map(|w| w.to_ascii_lowercase());
100    if last.as_deref() == Some("all") {
101        words.pop();
102        return Scope::All;
103    }
104    if words.len() >= 3
105        && words[words.len() - 3].eq_ignore_ascii_case("between")
106        && let (Ok(a), Ok(b)) = (
107            words[words.len() - 2].parse::<i64>(),
108            words[words.len() - 1].parse::<i64>(),
109        )
110    {
111        words.truncate(words.len() - 3);
112        return Scope::Between(a, b);
113    }
114    if words.len() >= 2
115        && words[words.len() - 2].eq_ignore_ascii_case("last")
116        && let Ok(n) = words[words.len() - 1].parse::<usize>()
117    {
118        words.truncate(words.len() - 2);
119        return Scope::Last(n);
120    }
121    if let Some(id) = last.as_deref().and_then(|w| w.parse::<i64>().ok()) {
122        words.pop();
123        return Scope::One(id);
124    }
125    Scope::Latest
126}

128/// The words after `cstats`. `Err` says what was wrong.
129///
130/// # Errors
131///
132/// A word that is not one of the commands.
133#[allow(clippy::too_many_lines)]
134pub fn parse(rest: &str) -> Result<Asked, String> {
135    let mut words: Vec<String> = rest.split_whitespace().map(str::to_owned).collect();
136    let word = words.first().map(|w| w.to_ascii_lowercase());
137    let number = |i: usize| words.get(i).and_then(|w| w.parse::<i64>().ok());
138    Ok(match word.as_deref() {
139        None => Asked::Session {
140            id: None,
141            full: false,
142        },
143        Some("help" | "menu") => Asked::Help,
144        Some("full") => Asked::Session {
145            id: number(1),
146            full: true,
147        },
148        Some("session" | "report") => Asked::Session {
149            id: number(1),
150            full: false,
151        },
152        Some("sessions" | "hunts") => {
153            words.remove(0);
154            Asked::Sessions(if words.is_empty() {
155                Scope::Last(10)
156            } else {
157                scope_from(&mut words)
158            })
159        }
160        Some("all") => Asked::Aggregate(Scope::All),
161        Some("last") => match number(1) {
162            Some(n) if n > 0 => Asked::Aggregate(Scope::Last(usize::try_from(n).unwrap_or(1))),
163            _ => return Err("say how many: cstats last <n>.".to_owned()),
164        },
165        Some("abilities" | "ability" | "recount") => {
166            words.remove(0);
167            Asked::Abilities(scope_from(&mut words))
168        }
169        Some("attack" | "attacks") => match words.get(1).map(|w| w.to_ascii_lowercase()) {
170            None => Asked::Attacks(Listing::from_words(&[], scope_from)),
171            Some(w) if LISTING_WORDS.contains(&w.as_str()) => {
172                Asked::Attacks(Listing::from_words(&words[1..], scope_from))
173            }
174            Some(w) if w.starts_with('#') || w.parse::<i64>().is_ok() => {
175                match w.trim_start_matches('#').parse::<i64>() {
176                    Ok(id) => Asked::AttackById(id),
177                    Err(_) => return Err(format!("`{w}` is not an attack id.")),
178                }
179            }
180            Some(_) => Asked::AttackByName(words.get(1).cloned(), number(2)),
181        },
182        Some("creature" | "creatures") => match words.get(1).map(|w| w.to_ascii_lowercase()) {
183            None => Asked::Creatures(Listing::from_words(&[], scope_from)),
184            Some(w) if LISTING_WORDS.contains(&w.as_str()) => {
185                Asked::Creatures(Listing::from_words(&words[1..], scope_from))
186            }
187            Some(w) if w == "full" => Asked::Creature {
188                arg: words.get(2).cloned(),
189                idx: None,
190                full: true,
191                all: false,
192            },
193            Some(_) => {
194                let all = words.get(2).is_some_and(|w| w.eq_ignore_ascii_case("all"));
195                Asked::Creature {
196                    arg: words.get(1).cloned(),
197                    idx: if all { None } else { number(2) },
198                    full: false,
199                    all,
200                }
201            }
202        },
203        Some("tree") => match number(1) {
204            Some(id) => Asked::Tree(id),
205            None => return Err("say the root: cstats tree <id>.".to_owned()),
206        },
207        Some("flare" | "flares") => match (words.get(1), words.get(2)) {
208            (Some(name), Some(word)) if word.eq_ignore_ascii_case("alongside") => {
209                Asked::FlareAlongside(name.clone())
210            }
211            (name, _) => Asked::Flare(name.cloned()),
212        },
213        Some("def" | "defense") => Asked::Defense(words.get(1).cloned()),
214        Some("hp") => Asked::Hp(words.get(1).cloned()),
215        Some("blind" | "control" | "family" | "aim" | "accuracy" | "ttk") => {
216            let kind = match word.as_deref() {
217                Some("blind") => Kind::Blind,
218                Some("control") => Kind::Control,
219                Some("family") => Kind::Family,
220                Some("aim") => Kind::Aim,
221                Some("accuracy") => Kind::Accuracy,
222                _ => Kind::Ttk,
223            };
224            words.remove(0);
225            let scope = scope_from(&mut words);
226            Asked::Analytics(kind, scope, words.join(" "))
227        }
228        Some("barkskin" | "bark") => Asked::Barkskin,
229        Some("intercepts" | "intercept") => {
230            words.remove(0);
231            Asked::Intercepts(scope_from(&mut words))
232        }
233        Some("compare") => {
234            Asked::Compare(words[1..].iter().filter_map(|w| w.parse().ok()).collect())
235        }
236        Some(n) if n.parse::<i64>().is_ok() => Asked::Session {
237            id: n.parse().ok(),
238            full: words.get(1).is_some_and(|w| w.eq_ignore_ascii_case("full")),
239        },
240        Some(other) => {
241            return Err(format!(
242                "`{other}` is not a cstats word; cstats help lists them."
243            ));
244        }
245    })
246}

248/// A whole line answered, `cstats` and its words, as a link's click or a
249/// window asks: the lines, or what was wrong.
250///
251/// # Errors
252///
253/// The line is not `cstats`, a word is not one of the commands, or the
254/// ledger cannot be read.
255pub fn run(ledger: &Ledger, who: &Who, line: &str) -> Result<Vec<String>, String> {
256    let line = line.trim().trim_start_matches(['.', ';']);
257    let (word, rest) = line.split_once(char::is_whitespace).unwrap_or((line, ""));
258    if !word.eq_ignore_ascii_case("cstats") {
259        return Err(format!("{word} is not a combat report."));
260    }
261    answer(ledger, who, &parse(rest)?)
262}

264/// Answer `asked` for `who` out of `ledger`. Lines, in fixed width.
265///
266/// # Errors
267///
268/// The ledger cannot be read.
269pub fn answer(ledger: &Ledger, who: &Who, asked: &Asked) -> Result<Vec<String>, String> {
270    let reader = ledger.reader().map_err(|e| e.to_string())?;
271    let conn = reader.connection();
272    let said = match asked {
273        Asked::Help => {
274            let mut lines: Vec<String> = HELP.iter().map(|&l| l.to_owned()).collect();
275            lines.extend(crate::report::quick_actions(None));
276            Ok(lines)
277        }
278        Asked::Session { id, full } => crate::report::session(conn, who, *id, *full),
279        Asked::Sessions(scope) => crate::report::sessions(conn, who, scope),
280        Asked::Aggregate(scope) => crate::report::aggregate(conn, who, scope),
281        Asked::Abilities(scope) => crate::report::abilities(conn, who, scope),
282        Asked::Attacks(listing) => crate::listing::attacks(conn, who, listing),
283        Asked::AttackById(id) => crate::detail::attack_detail(conn, *id, None),
284        Asked::AttackByName(name, idx) => {
285            crate::detail::attack_by_name(conn, who, name.as_deref(), *idx)
286        }
287        Asked::Creatures(listing) => crate::listing::creatures(conn, who, listing),
288        Asked::Creature {
289            arg,
290            idx,
291            full,
292            all,
293        } => crate::detail::creature(conn, who, arg.as_deref(), *idx, *full, *all),
294        Asked::Tree(root) => crate::detail::tree(conn, *root),
295        Asked::Flare(name) => crate::analytics::flare(conn, who, name.as_deref()),
296        Asked::FlareAlongside(name) => crate::analytics::alongside(conn, who, name),
297        Asked::Defense(noun) => crate::analytics::defense(conn, who, noun.as_deref()),
298        Asked::Hp(arg) => crate::analytics::hp(conn, who, arg.as_deref()),
299        Asked::Analytics(kind, scope, location) => {
300            crate::analytics::analytics(conn, who, *kind, scope, location)
301        }
302        Asked::Compare(ids) => crate::analytics::compare(conn, who, ids),
303        Asked::Barkskin => crate::analytics::barkskin(conn, who),
304        Asked::Intercepts(scope) => crate::analytics::intercepts(conn, who, scope),
305    };
306    said.map_err(|e| e.to_string())
307}

309#[cfg(test)]
310mod tests {
311    use super::*;
312
313    #[test]
314    fn the_words_are_read() {
315        assert_eq!(
316            parse(""),
317            Ok(Asked::Session {
318                id: None,
319                full: false
320            })
321        );
322        assert_eq!(
323            parse("7 full"),
324            Ok(Asked::Session {
325                id: Some(7),
326                full: true
327            })
328        );
329        assert_eq!(parse("last 3"), Ok(Asked::Aggregate(Scope::Last(3))));
330        assert_eq!(parse("abilities all"), Ok(Asked::Abilities(Scope::All)));
331        assert_eq!(
332            parse("abilities last 2"),
333            Ok(Asked::Abilities(Scope::Last(2)))
334        );
335        assert_eq!(parse("attack #12"), Ok(Asked::AttackById(12)));
336        assert_eq!(
337            parse("attack fire 3"),
338            Ok(Asked::AttackByName(Some("fire".into()), Some(3)))
339        );
340        assert_eq!(
341            parse("creature kobold all"),
342            Ok(Asked::Creature {
343                arg: Some("kobold".into()),
344                idx: None,
345                full: false,
346                all: true
347            })
348        );
349        assert_eq!(
350            parse("aim left eye 4"),
351            Ok(Asked::Analytics(
352                Kind::Aim,
353                Scope::One(4),
354                "left eye".into()
355            ))
356        );
357        assert_eq!(parse("compare 3 5"), Ok(Asked::Compare(vec![3, 5])));
358        assert!(parse("bogus").is_err());
359    }

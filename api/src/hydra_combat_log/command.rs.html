42/// A word after `attacks` or `creatures` that makes it a listing rather
43/// than one by name or id.
44const LISTING_WORDS: &[&str] = &["all", "last", "between", "recent", "page", "per"];
45
46/// What was asked.
47#[derive(Debug, Clone, PartialEq, Eq)]
48pub enum Asked {
49    /// The words.
50    Help,
51    /// A hunt's report: the latest, or by id; `full` lists every creature.
52    Session {
53        /// Which hunt; the latest when `None`.
54        id: Option<i64>,
55        /// Every creature instance listed.
56        full: bool,
57    },
58    /// The hunts over a scope, a line each.
59    Sessions(Scope),
60    /// The numbers over a scope.
61    Aggregate(Scope),
62    /// Damage by ability over a scope.
63    Abilities(Scope),
64    /// Every attack name by damage, or the newest attacks, a page.
65    Attacks(Listing),
66    /// One attack by row id.
67    AttackById(i64),
68    /// The nth instance of an attack by name, or the names.
69    AttackByName(Option<String>, Option<i64>),
70    /// Every creature by damage, or the newest, a page.
71    Creatures(Listing),
72    /// A creature's report.
73    Creature {
74        /// Its id or noun; the top twenty when `None`.
75        arg: Option<String>,
76        /// Which instance of the noun.
77        idx: Option<i64>,
78        /// The whole tree.
79        full: bool,
80        /// Every sequence, not just the ends.
81        all: bool,
82    },
83    /// A spawn tree by root.
84    Tree(i64),
85    /// A flare's analytics, or the names.
86    Flare(Option<String>),
87    /// The flares that fire with one.
88    FlareAlongside(String),
89    /// A flare's rate with each effect up and without it.
90    FlareEffects(String),
91    /// A flare's rate by each item held.
92    FlareItems(String),
93    /// Defensive readings, or the nouns.
94    Defense(Option<String>),
95    /// HP estimates.
96    Hp(Option<String>),
97    /// An analytics report over a scope, with an assumed aim location.
98    Analytics(Kind, Scope, String),
99    /// Two hunts compared.
100    Compare(Vec<i64>),
101    /// Each bark cast and its absorbs.
102    Barkskin,
103    /// Every barrier together over a scope.
104    Intercepts(Scope),
105    /// What the swings were made with, over a scope.
106    Gear(Scope),
107    /// Unarmed positioning over a scope, and the attack shown: one of
108    /// jab, punch, kick and grapple, `combined` for them together, or
109    /// `None` for each.
110    Ucs(Scope, Option<String>),
111    /// The commands that dealt the most over a scope, the top so many.
112    Biggest(Scope, usize),
113    /// One command's attacks: its hunt, and its chunk in the hunt.
114    Chunk(i64, i64),
115    /// What was up over the hunts.
116    Effects(Scope),
117    /// One effect's recorded values over the hunts.
118    Effect(String, Scope),
119}

121/// The trailing scope words taken off `words`: `all`, `last <n>`,
122/// `between <id> <id>`, or an id; the latest hunt when none.
123fn scope_from(words: &mut Vec<String>) -> Scope {
124    let last = words.last().map(|w| w.to_ascii_lowercase());
125    if last.as_deref() == Some("all") {
126        words.pop();
127        return Scope::All;
128    }
129    if words.len() >= 3
130        && words[words.len() - 3].eq_ignore_ascii_case("between")
131        && let (Ok(a), Ok(b)) = (
132            words[words.len() - 2].parse::<i64>(),
133            words[words.len() - 1].parse::<i64>(),
134        )
135    {
136        words.truncate(words.len() - 3);
137        return Scope::Between(a, b);
138    }
139    if words.len() >= 2
140        && words[words.len() - 2].eq_ignore_ascii_case("last")
141        && let Ok(n) = words[words.len() - 1].parse::<usize>()
142    {
143        words.truncate(words.len() - 2);
144        return Scope::Last(n);
145    }
146    if let Some(id) = last.as_deref().and_then(|w| w.parse::<i64>().ok()) {
147        words.pop();
148        return Scope::One(id);
149    }
150    Scope::Latest
151}

153/// The words after `cstats`. `Err` says what was wrong.
154///
155/// # Errors
156///
157/// A word that is not one of the commands.
158#[allow(clippy::too_many_lines)]
159pub fn parse(rest: &str) -> Result<Asked, String> {
160    let mut words: Vec<String> = rest.split_whitespace().map(str::to_owned).collect();
161    let word = words.first().map(|w| w.to_ascii_lowercase());
162    let number = |i: usize| words.get(i).and_then(|w| w.parse::<i64>().ok());
163    Ok(match word.as_deref() {
164        None => Asked::Session {
165            id: None,
166            full: false,
167        },
168        Some("help" | "menu") => Asked::Help,
169        Some("full") => Asked::Session {
170            id: number(1),
171            full: true,
172        },
173        Some("session" | "report") => Asked::Session {
174            id: number(1),
175            full: false,
176        },
177        Some("sessions" | "hunts") => {
178            words.remove(0);
179            Asked::Sessions(if words.is_empty() {
180                Scope::Last(10)
181            } else {
182                scope_from(&mut words)
183            })
184        }
185        Some("all") => Asked::Aggregate(Scope::All),
186        Some("last") => match number(1) {
187            Some(n) if n > 0 => Asked::Aggregate(Scope::Last(usize::try_from(n).unwrap_or(1))),
188            _ => return Err("say how many: cstats last <n>.".to_owned()),
189        },
190        Some("abilities" | "ability" | "recount") => {
191            words.remove(0);
192            Asked::Abilities(scope_from(&mut words))
193        }
194        Some("attack" | "attacks") => match words.get(1).map(|w| w.to_ascii_lowercase()) {
195            None => Asked::Attacks(Listing::from_words(&[], scope_from)),
196            Some(w) if LISTING_WORDS.contains(&w.as_str()) => {
197                Asked::Attacks(Listing::from_words(&words[1..], scope_from))
198            }
199            Some(w) if w.starts_with('#') || w.parse::<i64>().is_ok() => {
200                match w.trim_start_matches('#').parse::<i64>() {
201                    Ok(id) => Asked::AttackById(id),
202                    Err(_) => return Err(format!("`{w}` is not an attack id.")),
203                }
204            }
205            Some(_) => Asked::AttackByName(words.get(1).cloned(), number(2)),
206        },
207        Some("creature" | "creatures") => match words.get(1).map(|w| w.to_ascii_lowercase()) {
208            None => Asked::Creatures(Listing::from_words(&[], scope_from)),
209            Some(w) if LISTING_WORDS.contains(&w.as_str()) => {
210                Asked::Creatures(Listing::from_words(&words[1..], scope_from))
211            }
212            Some(w) if w == "full" => Asked::Creature {
213                arg: words.get(2).cloned(),
214                idx: None,
215                full: true,
216                all: false,
217            },
218            Some(_) => {
219                let all = words.get(2).is_some_and(|w| w.eq_ignore_ascii_case("all"));
220                Asked::Creature {
221                    arg: words.get(1).cloned(),
222                    idx: if all { None } else { number(2) },
223                    full: false,
224                    all,
225                }
226            }
227        },
228        Some("tree") => match number(1) {
229            Some(id) => Asked::Tree(id),
230            None => return Err("say the root: cstats tree <id>.".to_owned()),
231        },
232        Some("flare" | "flares") => match (words.get(1), words.get(2)) {
233            (Some(name), Some(word)) if word.eq_ignore_ascii_case("alongside") => {
234                Asked::FlareAlongside(name.clone())
235            }
236            (Some(name), Some(word)) if word.eq_ignore_ascii_case("effects") => {
237                Asked::FlareEffects(name.clone())
238            }
239            (Some(name), Some(word)) if word.eq_ignore_ascii_case("items") => {
240                Asked::FlareItems(name.clone())
241            }
242            (name, _) => Asked::Flare(name.cloned()),
243        },
244        Some("def" | "defense") => Asked::Defense(words.get(1).cloned()),
245        Some("hp") => Asked::Hp(words.get(1).cloned()),
246        Some("blind" | "control" | "family" | "aim" | "accuracy" | "ttk") => {
247            let kind = match word.as_deref() {
248                Some("blind") => Kind::Blind,
249                Some("control") => Kind::Control,
250                Some("family") => Kind::Family,
251                Some("aim") => Kind::Aim,
252                Some("accuracy") => Kind::Accuracy,
253                _ => Kind::Ttk,
254            };
255            words.remove(0);
256            let scope = scope_from(&mut words);
257            Asked::Analytics(kind, scope, words.join(" "))
258        }
259        Some("barkskin" | "bark") => Asked::Barkskin,
260        Some("intercepts" | "intercept") => {
261            words.remove(0);
262            Asked::Intercepts(scope_from(&mut words))
263        }
264        Some("biggest" | "big") => {
265            words.remove(0);
266            // `top <n>`, anywhere: a bare number would be a hunt.
267            let top = words
268                .iter()
269                .position(|w| w.eq_ignore_ascii_case("top"))
270                .and_then(|i| {
271                    let n = words.get(i + 1)?.parse::<usize>().ok()?;
272                    words.drain(i..=i + 1);
273                    Some(n.max(1))
274                });
275            Asked::Biggest(scope_from(&mut words), top.unwrap_or(crate::biggest::SHOWN))
276        }
277        Some("chunk" | "command") => match (number(1), number(2)) {
278            (Some(hunt), Some(chunk)) => Asked::Chunk(hunt, chunk),
279            _ => return Err("say which: cstats chunk <hunt> <chunk>.".to_owned()),
280        },
281        Some("ucs" | "positioning") => {
282            words.remove(0);
283            let show = words
284                .iter()
285                .position(|w| crate::ucs::SHOWN.contains(&w.to_ascii_lowercase().as_str()))
286                .map(|i| words.remove(i).to_ascii_lowercase());
287            Asked::Ucs(scope_from(&mut words), show)
288        }
289        Some("gear" | "equipment") => {
290            words.remove(0);
291            Asked::Gear(scope_from(&mut words))
292        }
293        Some("effects" | "effect") => {
294            words.remove(0);
295            let scope = scope_from(&mut words);
296            if words.is_empty() {
297                Asked::Effects(scope)
298            } else {
299                Asked::Effect(words.join(" "), scope)
300            }
301        }
302        Some("compare") => {
303            Asked::Compare(words[1..].iter().filter_map(|w| w.parse().ok()).collect())
304        }
305        Some(n) if n.parse::<i64>().is_ok() => Asked::Session {
306            id: n.parse().ok(),
307            full: words.get(1).is_some_and(|w| w.eq_ignore_ascii_case("full")),
308        },
309        Some(other) => {
310            return Err(format!(
311                "`{other}` is not a cstats word; cstats help lists them."
312            ));
313        }
314    })
315}

317/// A whole line answered, `cstats` and its words, as a link's click or a
318/// window asks: the lines, or what was wrong.
319///
320/// # Errors
321///
322/// The line is not `cstats`, a word is not one of the commands, or the
323/// ledger cannot be read.
324pub fn run(ledger: &Ledger, who: &Who, line: &str) -> Result<Vec<String>, String> {
325    run_report(ledger, who, line).map(|report| report.text())
326}
327
328/// A whole line answered as blocks, for a window that draws them.
329///
330/// # Errors
331///
332/// The line is not `cstats`, a word is not one of the reports, or the
333/// ledger cannot be read.
334pub fn run_report(ledger: &Ledger, who: &Who, line: &str) -> Result<Report, String> {
335    let line = line.trim().trim_start_matches(['.', ';']);
336    let (word, rest) = line.split_once(char::is_whitespace).unwrap_or((line, ""));
337    if !word.eq_ignore_ascii_case("cstats") {
338        return Err(format!("{word} is not a combat report."));
339    }
340    let (rest, who) = on_both(rest, who);
341    answer_report(ledger, &who, &parse(&rest)?)
342}

344/// The words after `cstats` with `both` taken out, and whose hunts they
345/// are over: `who`'s, or `who`'s on every instance when `both` was said.
346#[must_use]
347pub fn on_both(rest: &str, who: &Who) -> (String, Who) {
348    let (both, kept): (Vec<&str>, Vec<&str>) = rest
349        .split_whitespace()
350        .partition(|w| w.eq_ignore_ascii_case("both"));
351    let who = if both.is_empty() {
352        who.clone()
353    } else {
354        crate::report::on_every_instance(who)
355    };
356    (kept.join(" "), who)
357}

359/// Answer `asked` for `who` out of `ledger`. Lines, in fixed width.
360///
361/// # Errors
362///
363/// The ledger cannot be read.
364pub fn answer(ledger: &Ledger, who: &Who, asked: &Asked) -> Result<Vec<String>, String> {
365    answer_report(ledger, who, asked).map(|report| report.text())
366}

368/// Answer `asked` for `who` out of `ledger`, as blocks: the flare and
369/// creature reports say tables and facts; the rest are lines, their grids
370/// read back as tables.
371///
372/// # Errors
373///
374/// The ledger cannot be read.
375pub fn answer_report(ledger: &Ledger, who: &Who, asked: &Asked) -> Result<Report, String> {
376    let reader = ledger.reader().map_err(|e| e.to_string())?;
377    let conn = reader.connection();
378    let lines = |said: Result<Vec<String>, crate::report::Error>| {
379        said.map(|l| crate::out::Out::from_text(&l))
380    };
381    let said = match asked {
382        Asked::Help => {
383            let mut help: Vec<String> = HELP.iter().map(|&l| l.to_owned()).collect();
384            help.extend(crate::report::quick_actions(None));
385            Ok(Report::from_lines(help))
386        }
387        Asked::Session { id, full } => lines(crate::report::session(conn, who, *id, *full)),
388        Asked::Sessions(scope) => lines(crate::report::sessions(conn, who, scope)),
389        Asked::Aggregate(scope) => lines(crate::report::aggregate(conn, who, scope)),
390        Asked::Abilities(scope) => lines(crate::report::abilities(conn, who, scope)),
391        Asked::Attacks(listing) => lines(crate::listing::attacks(conn, who, listing)),
392        Asked::AttackById(id) => lines(crate::detail::attack_detail(conn, *id, None)),
393        Asked::AttackByName(name, idx) => lines(crate::detail::attack_by_name(
394            conn,
395            who,
396            name.as_deref(),
397            *idx,
398        )),
399        Asked::Creatures(listing) => lines(crate::listing::creatures(conn, who, listing)),
400        Asked::Creature {
401            arg,
402            idx,
403            full,
404            all,
405        } => crate::detail::creature(conn, who, arg.as_deref(), *idx, *full, *all),
406        Asked::Tree(root) => lines(crate::detail::tree(conn, *root)),
407        Asked::Flare(name) => crate::analytics::flare(conn, who, name.as_deref()),
408        Asked::FlareAlongside(name) => lines(crate::analytics::alongside(conn, who, name)),
409        Asked::FlareEffects(name) => crate::analytics::flare_effects(conn, who, name),
410        Asked::FlareItems(name) => crate::analytics::flare_items(conn, who, name),
411        Asked::Defense(noun) => lines(crate::analytics::defense(conn, who, noun.as_deref())),
412        Asked::Hp(arg) => lines(crate::analytics::hp(conn, who, arg.as_deref())),
413        Asked::Analytics(kind, scope, location) => lines(crate::analytics::analytics(
414            conn, who, *kind, scope, location,
415        )),
416        Asked::Compare(ids) => lines(crate::analytics::compare(conn, who, ids)),
417        Asked::Barkskin => lines(crate::analytics::barkskin(conn, who)),
418        Asked::Intercepts(scope) => lines(crate::analytics::intercepts(conn, who, scope)),
419        Asked::Gear(scope) => lines(crate::analytics::gear(conn, who, scope)),
420        Asked::Ucs(scope, show) => crate::ucs::ucs(conn, who, scope, show.as_deref()),
421        Asked::Biggest(scope, top) => crate::biggest::biggest(conn, who, scope, *top),
422        Asked::Chunk(hunt, chunk) => crate::biggest::chunk(conn, *hunt, *chunk),
423        Asked::Effects(scope) => lines(crate::analytics::effects(conn, who, scope)),
424        Asked::Effect(name, scope) => lines(crate::analytics::effect(conn, who, name, scope)),
425    };
426    said.map_err(|e| e.to_string())
427}

429#[cfg(test)]
430mod tests {
431    use super::*;
432
433    #[test]
434    fn the_words_are_read() {
435        assert_eq!(
436            parse(""),
437            Ok(Asked::Session {
438                id: None,
439                full: false
440            })
441        );
442        assert_eq!(
443            parse("7 full"),
444            Ok(Asked::Session {
445                id: Some(7),
446                full: true
447            })
448        );
449        assert_eq!(parse("last 3"), Ok(Asked::Aggregate(Scope::Last(3))));
450        assert_eq!(parse("abilities all"), Ok(Asked::Abilities(Scope::All)));
451        assert_eq!(
452            parse("abilities last 2"),
453            Ok(Asked::Abilities(Scope::Last(2)))
454        );
455        assert_eq!(parse("attack #12"), Ok(Asked::AttackById(12)));
456        assert_eq!(
457            parse("attack fire 3"),
458            Ok(Asked::AttackByName(Some("fire".into()), Some(3)))
459        );
460        assert_eq!(
461            parse("creature kobold all"),
462            Ok(Asked::Creature {
463                arg: Some("kobold".into()),
464                idx: None,
465                full: false,
466                all: true
467            })
468        );
469        assert_eq!(
470            parse("aim left eye 4"),
471            Ok(Asked::Analytics(
472                Kind::Aim,
473                Scope::One(4),
474                "left eye".into()
475            ))
476        );
477        assert_eq!(parse("compare 3 5"), Ok(Asked::Compare(vec![3, 5])));
478        assert!(parse("bogus").is_err());
479        assert_eq!(
480            parse("ucs all Jab"),
481            Ok(Asked::Ucs(Scope::All, Some("jab".into())))
482        );
483        assert_eq!(
484            parse("ucs combined last 2"),
485            Ok(Asked::Ucs(Scope::Last(2), Some("combined".into())))
486        );
487        assert_eq!(parse("ucs"), Ok(Asked::Ucs(Scope::Latest, None)));
488        assert_eq!(parse("biggest all"), Ok(Asked::Biggest(Scope::All, 20)));
489        assert_eq!(
490            parse("biggest top 5 last 3"),
491            Ok(Asked::Biggest(Scope::Last(3), 5))
492        );
493        assert_eq!(parse("biggest 12"), Ok(Asked::Biggest(Scope::One(12), 20)));
494        assert_eq!(parse("chunk 3052 5185"), Ok(Asked::Chunk(3052, 5185)));
495        assert!(parse("chunk 3052").is_err());
496        let nisugi = Who::new("Prime", "Nisugi");
497        let (rest, who) = on_both("ucs all BOTH", &nisugi);
498        assert_eq!(rest, "ucs all");
499        assert_eq!(who, Who::new("*", "Nisugi"));
500        assert_eq!(
501            on_both("flare briar", &nisugi),
502            ("flare briar".to_owned(), nisugi)
503        );
504    }

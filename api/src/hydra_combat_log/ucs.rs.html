1//! `cstats ucs`: unarmed positioning over a scope. Every own jab, punch,
2//! kick and grapple is paired with the positioning the game said for it
3//! (said before its roll: the tier the strike was made at) and any
4//! follow-up it opened. A strike's tier after is the tier of the next
5//! strike at the same creature in the same hunt.
6//!
7//! Over the landed strikes below excellent, a tier that rose is told
8//! apart by its strike: the offered follow-up taken, or not. A rise with
9//! no follow-up taken is counted as rising without it.
10//!
11//! The arms are Bold Brawler's rank as each hunt's gemstone set said it,
12//! so a test with the gem off and on splits on its own. The game's line
13//! is kept in the status row: its tier, or the follow-up attack it named,
14//! in the row's `action`.
15

35/// A tier's word as the game says it, and its rank.
36fn tier_of(word: &str) -> Option<u8> {
37    match word {
38        "decent" => Some(1),
39        "good" => Some(2),
40        "excellent" => Some(3),
41        _ => None,
42    }
43}

45/// A tier's rank as its word.
46fn word_of(tier: u8) -> &'static str {
47    match tier {
48        1 => "decent",
49        2 => "good",
50        _ => "excellent",
51    }
52}

64/// A share with its 95% interval (Wilson's), in words: `13.6% (9.8-18.6)`.
65fn share(count: i64, of: i64) -> String {
66    if of == 0 {
67        return "-".to_owned();
68    }
69    #[allow(clippy::cast_precision_loss)]
70    let (count, of) = (count as f64, of as f64);
71    let rate = count / of;
72    let z = 1.96_f64;
73    let scale = 1.0 + z * z / of;
74    let centre = (rate + z * z / (2.0 * of)) / scale;
75    let half = z * (rate * (1.0 - rate) / of + z * z / (4.0 * of * of)).sqrt() / scale;
76    format!(
77        "{:.1}% ({:.1}-{:.1})",
78        rate * 100.0,
79        (centre - half).max(0.0) * 100.0,
80        (centre + half).min(1.0) * 100.0
81    )
82}

84impl Arm {
85    /// The landed strikes made at `tier`, and those that offered.
86    fn at(&self, tier: u8) -> (i64, i64) {
87        self.by_tier.get(&tier).copied().unwrap_or((0, 0))
88    }
89
90    /// The landed strikes below excellent, and those that offered.
91    fn below(&self) -> (i64, i64) {
92        let (d, g) = (self.at(1), self.at(2));
93        (d.0 + g.0, d.1 + g.1)
94    }
95}

97/// What one arm counted.
98#[derive(Default)]
99struct Arm {
100    hunts: std::collections::BTreeSet<i64>,
101    landed: i64,
102    by_tier: BTreeMap<u8, (i64, i64)>,
103    named: BTreeMap<String, i64>,
104    taken: i64,
105    taken_rose: i64,
106    /// Landed below excellent with the follow-up not taken, and those
107    /// whose next strike was a tier higher.
108    eligible: i64,
109    rose: i64,
110}

112/// What one strike's lines said: its tier, and the follow-up it named.
113type Slot = (Option<u8>, Option<String>);
114
115/// The UCS lines of one hunt, creature and second, as written: each
116/// line's status and its word.
117type Lines = HashMap<(i64, i64, i64), Vec<(String, Option<String>)>>;
118
119/// What the strikes of one hunt, creature and second said, in the order
120/// they were said: a second with a jab and its echo holds two.
121type Said = HashMap<(i64, i64, i64), Vec<Slot>>;
122
123/// A second's UCS lines, in order, as one slot a strike: a positioning
124/// line opens the next strike's slot, and an offer belongs to the strike
125/// whose positioning came before it.
126fn slots<'a>(lines: impl IntoIterator<Item = (&'a str, Option<&'a str>)>) -> Vec<Slot> {
127    let mut out: Vec<Slot> = Vec::new();
128    for (status, action) in lines {
129        if status == "position" {
130            out.push((action.and_then(tier_of), None));
131        } else if let Some(last) = out.last_mut().filter(|s| s.1.is_none()) {
132            last.1 = action.map(str::to_owned);
133        } else {
134            out.push((None, action.map(str::to_owned)));
135        }
136    }
137    out
138}

140/// An attack and an arm: 0 for one attack or 1 for all of them (so the
141/// whole comes last), the attack, and the arm's label. Ordered so each
142/// attack's arms sit side by side, the comparison a row apart.
143type Key = (u8, String, String);
144
145/// The rows `show` keeps: one attack's, the four together (`combined`),
146/// or each attack's; and the four together too, unless one attack is all
147/// there is, when they would only say its rows again.
148fn shown(counted: &mut BTreeMap<Key, Arm>, show: Option<&str>) {
149    match show {
150        Some("combined") => counted.retain(|(order, _, _), _| *order == 1),
151        Some(attack) => counted.retain(|(order, name, _), _| *order == 0 && name == attack),
152        None => {
153            let attacks: std::collections::BTreeSet<&String> = counted
154                .keys()
155                .filter(|(order, _, _)| *order == 0)
156                .map(|(_, name, _)| name)
157                .collect();
158            if attacks.len() == 1 {
159                counted.retain(|(order, _, _), _| *order == 0);
160            }
161        }
162    }
163}

165/// The report; `show` picks one attack's rows, `combined`, or each.
166///
167/// # Errors
168///
169/// The file cannot be read.
170pub fn ucs(
171    conn: &Connection,
172    who: &Who,
173    scope: &Scope,
174    show: Option<&str>,
175) -> Result<Report, Error> {
176    let Some(hunts) = resolve(conn, who, scope)? else {
177        return Ok(Report::from_lines(vec![
178            "No hunts recorded yet.".to_owned(),
179        ]));
180    };
181    let set = hunts.sql_in();
182    let strikes = strikes_of(conn, &set)?;
183    let mut out = Out::new();
184    out.title(format!("UCS positioning - {}", hunts.label));
185    if strikes.is_empty() {
186        out.note("No jab, punch, kick or grapple of yours is recorded over these hunts.");
187        return Ok(out.into_report());
188    }
189    let arms = arms_of(conn, &set, who.instance == crate::report::EVERY_INSTANCE)?;
190    let mut counted = count(&strikes, &arms);
191    shown(&mut counted, show);
192    let rows: Vec<Vec<String>> = counted
193        .iter()
194        .map(|((_, attack, label), arm)| {
195            let (eligible, rose) = (arm.eligible, arm.rose);
196            vec![
197                attack.clone(),
198                label.clone(),
199                arm.hunts.len().to_string(),
200                arm.landed.to_string(),
201                share(arm.at(3).0, arm.landed),
202                share(arm.below().1, arm.below().0),
203                format!("{}/{}", arm.taken_rose, arm.taken),
204                eligible.to_string(),
205                share(rose, eligible),
206            ]
207        })
208        .collect();
209    out.section(
210        "By attack and Bold Brawler rank",
211        &[
212            "Attack",
213            "Arm",
214            "Hunts",
215            "Landed",
216            "At excellent",
217            "Follow-up offered below excellent (95%)",
218            "Follow-ups taken that rose",
219            "Eligible",
220            "Rose without the follow-up (95%)",
221        ],
222        &rows,
223    );
224    out.note(
225        "Arm: Bold Brawler's rank in the hunt's gemstone set; off, not equipped; unread, the set not yet read. No strike at excellent opens a follow-up, so the offer rate counts the landed strikes below it. Decent and good offer at different rates: between arms, compare them tier by tier (the table below). Eligible: a landed strike below excellent, followed by another at the same creature, the offered follow-up not taken. Intervals are 95% (Wilson).",
226    );
227    let rows: Vec<Vec<String>> = counted
228        .iter()
229        .map(|((_, attack, label), arm)| {
230            vec![
231                attack.clone(),
232                label.clone(),
233                arm.eligible.to_string(),
234                share(arm.rose, arm.eligible),
235            ]
236        })
237        .collect();
238    out.section(
239        "Rose without the follow-up",
240        &["Attack", "Arm", "Eligible", "Rose (95%)"],
241        &rows,
242    );
243    tier_tables(&mut out, &counted);
244    Ok(out.into_report())
245}

247/// The offer rate by the tier a strike was made at, and the follow-up
248/// each strike named.
249fn tier_tables(out: &mut Out, counted: &BTreeMap<Key, Arm>) {
250    // Rows by attack, then tier or follow-up, then arm: the arms of one
251    // comparison side by side.
252    let mut rows: BTreeMap<(u8, String, u8, String), Vec<String>> = BTreeMap::new();
253    for ((order, attack, label), arm) in counted {
254        for (tier, (landed, offered)) in &arm.by_tier {
255            rows.insert(
256                (*order, attack.clone(), *tier, label.clone()),
257                vec![
258                    attack.clone(),
259                    word_of(*tier).to_owned(),
260                    label.clone(),
261                    landed.to_string(),
262                    share(*offered, *landed),
263                ],
264            );
265        }
266    }
267    out.section(
268        "Follow-up offered, by the tier the strike was made at: the fair comparison between arms",
269        &["Attack", "Tier", "Arm", "Landed", "Offered (95%)"],
270        &rows.into_values().collect::<Vec<_>>(),
271    );
272    let mut rows: BTreeMap<(u8, String, String, String), Vec<String>> = BTreeMap::new();
273    for ((order, attack, label), arm) in counted {
274        for (named, n) in &arm.named {
275            rows.insert(
276                (*order, attack.clone(), named.clone(), label.clone()),
277                vec![attack.clone(), named.clone(), label.clone(), n.to_string()],
278            );
279        }
280    }
281    out.section(
282        "The follow-up named",
283        &["Strike", "Follow-up named", "Arm", "Offers"],
284        &rows.into_values().collect::<Vec<_>>(),
285    );
286}

288/// Every landed strike counted into its arm by attack, and into the
289/// arm's whole.
290fn count(strikes: &[Strike], arms: &BTreeMap<i64, String>) -> BTreeMap<Key, Arm> {
291    // Each arm by attack, and every attack of the arm together, last.
292    let mut counted: BTreeMap<Key, Arm> = BTreeMap::new();
293    for (i, a) in strikes.iter().enumerate() {
294        let Some(label) = arms.get(&a.hunt) else {
295            continue;
296        };
297        let Some(tier) = a.tier.filter(|_| a.landed) else {
298            continue;
299        };
300        // The next strike at the same creature in the same hunt, and what
301        // it made of this one.
302        let next = strikes
303            .get(i + 1)
304            .filter(|b| b.creature == a.creature && b.hunt == a.hunt)
305            .and_then(|b| b.tier.map(|t| (b, t)));
306        for key in [
307            (0, a.name.clone(), label.clone()),
308            (1, "all".to_owned(), label.clone()),
309        ] {
310            let arm = counted.entry(key).or_default();
311            arm.landed += 1;
312            arm.hunts.insert(a.hunt);
313            let t = arm.by_tier.entry(tier).or_insert((0, 0));
314            t.0 += 1;
315            if let Some(offer) = &a.offer {
316                t.1 += 1;
317                *arm.named.entry(offer.clone()).or_insert(0) += 1;
318            }
319            let Some((b, after)) = next else {
320                continue;
321            };
322            if a.offer.as_deref() == Some(b.name.as_str()) {
323                arm.taken += 1;
324                arm.taken_rose += i64::from(after > tier);
325            } else if tier < 3 {
326                arm.eligible += 1;
327                arm.rose += i64::from(after > tier);
328            }
329        }
330    }
331    counted
332}

334/// Every own UCS strike over `set`, by creature and time, with the tier
335/// and the offer its chunk said.
336fn strikes_of(conn: &Connection, set: &str) -> Result<Vec<Strike>, Error> {
337    // The status rows of a strike's second: same hunt and creature, in the
338    // order they were written, a slot a strike.
339    let mut stmt = conn.prepare(&format!(
340        "SELECT hunt_id, creature_id, occurred_at, status, action FROM statuses
341         WHERE hunt_id {set} AND kind = 'ucs' AND status IN ('position', 'tierup')
342           AND creature_id IS NOT NULL
343         ORDER BY id"
344    ))?;
345    let mut lines: Lines = HashMap::new();
346    for row in stmt.query_map([], |r| {
347        Ok((
348            r.get::<_, i64>(0)?,
349            r.get::<_, i64>(1)?,
350            r.get::<_, i64>(2)?,
351            r.get::<_, String>(3)?,
352            r.get::<_, Option<String>>(4)?,
353        ))
354    })? {
355        let (hunt, creature, at, status, action) = row?;
356        lines
357            .entry((hunt, creature, at))
358            .or_default()
359            .push((status, action));
360    }
361    let said: Said = lines
362        .into_iter()
363        .map(|(key, rows)| {
364            (
365                key,
366                slots(rows.iter().map(|(s, a)| (s.as_str(), a.as_deref()))),
367            )
368        })
369        .collect();
370    // The strikes of one second take its slots in turn.
371    let mut taken: HashMap<(i64, i64, i64), usize> = HashMap::new();
372    let mut stmt = conn.prepare(&format!(
373        "SELECT a.hunt_id, a.creature_id, a.occurred_at, a.name,
374            EXISTS (SELECT 1 FROM hits h WHERE h.attack_id = a.id AND h.flare_id IS NULL
375                    AND h.creature_id IS NOT NULL)
376         FROM attacks a WHERE a.hunt_id {set} AND a.ours = 1 AND a.name IN ({ATTACKS})
377           AND a.creature_id IS NOT NULL
378         ORDER BY a.hunt_id, a.creature_id, a.occurred_at, a.id"
379    ))?;
380    let rows = stmt.query_map([], |r| {
381        Ok((
382            r.get::<_, i64>(0)?,
383            r.get::<_, i64>(1)?,
384            r.get::<_, i64>(2)?,
385            r.get::<_, String>(3)?,
386            r.get::<_, bool>(4)?,
387        ))
388    })?;
389    let mut strikes = Vec::new();
390    for row in rows {
391        let (hunt, creature, at, name, landed) = row?;
392        let key = (hunt, creature, at);
393        let index = taken.entry(key).or_insert(0);
394        let (tier, offer) = said
395            .get(&key)
396            .and_then(|s| s.get(*index))
397            .cloned()
398            .unwrap_or_default();
399        *index += 1;
400        strikes.push(Strike {
401            hunt,
402            creature,
403            name,
404            landed,
405            tier,
406            offer,
407        });
408    }
409    Ok(strikes)
410}

412/// Each hunt's arm: Bold Brawler's rank, `off` when not equipped, or
413/// `unread` when the set was not yet read; with the hunt's instance before
414/// it when `by_instance`, so a test run on two instances keeps its arms
415/// apart.
416fn arms_of(
417    conn: &Connection,
418    set: &str,
419    by_instance: bool,
420) -> Result<BTreeMap<i64, String>, Error> {
421    let mut stmt = conn.prepare(&format!(
422        "SELECT h.id, h.instance,
423            (SELECT g.rank FROM hunt_gemstones g WHERE g.hunt_id = h.id AND g.mnemonic = '{PROPERTY}'),
424            EXISTS (SELECT 1 FROM hunt_gemstones g WHERE g.hunt_id = h.id)
425         FROM hunts h WHERE h.id {set}"
426    ))?;
427    let rows = stmt.query_map([], |r| {
428        Ok((
429            r.get::<_, i64>(0)?,
430            r.get::<_, String>(1)?,
431            r.get::<_, Option<i64>>(2)?,
432            r.get::<_, bool>(3)?,
433        ))
434    })?;
435    let mut arms = BTreeMap::new();
436    for row in rows {
437        let (hunt, instance, rank, known) = row?;
438        let arm = match (rank, known) {
439            (Some(rank), _) => format!("rank {rank}"),
440            (None, true) => "off".to_owned(),
441            (None, false) => "unread".to_owned(),
442        };
443        arms.insert(
444            hunt,
445            if by_instance {
446                format!("{instance}, {arm}")
447            } else {
448                arm
449            },
450        );
451    }
452    Ok(arms)
453}

456mod tests {
457    use super::{Arm, Key, share, shown, slots};

460    fn rows(keys: &[(u8, &str)]) -> BTreeMap<Key, Arm> {
461        keys.iter()
462            .map(|(order, attack)| {
463                (
464                    (*order, (*attack).to_owned(), "rank 5".to_owned()),
465                    Arm::default(),
466                )
467            })
468            .collect()
469    }
470
471    fn kept(counted: &BTreeMap<Key, Arm>) -> Vec<String> {
472        counted
473            .keys()
474            .map(|(_, attack, _)| attack.clone())
475            .collect()
476    }

478    #[test]
479    fn the_report_shows_what_was_picked() {
480        let both = [(0, "jab"), (0, "kick"), (1, "all")];
481        let mut each = rows(&both);
482        shown(&mut each, None);
483        assert_eq!(kept(&each), ["jab", "kick", "all"]);
484        let mut jab = rows(&both);
485        shown(&mut jab, Some("jab"));
486        assert_eq!(kept(&jab), ["jab"]);
487        let mut combined = rows(&both);
488        shown(&mut combined, Some("combined"));
489        assert_eq!(kept(&combined), ["all"]);
490        // One attack alone: its combined rows would only repeat it.
491        let mut alone = rows(&[(0, "jab"), (1, "all")]);
492        shown(&mut alone, None);
493        assert_eq!(kept(&alone), ["jab"]);
494    }

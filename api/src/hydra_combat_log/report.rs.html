14/// `HH:MM:SS` on the game's clock.
15#[must_use]
16pub fn clock(at: i64) -> String {
17    let seconds = (at + hydra_net::eastern::offset(at)).rem_euclid(hydra_net::eastern::DAY);
18    format!(
19        "{:02}:{:02}:{:02}",
20        seconds / 3600,
21        seconds % 3600 / 60,
22        seconds % 60
23    )
24}

36/// `Xm YYs`, or `Xs` under a minute; `open` for a hunt not yet ended.

46/// Words a click sends as one of Hydra's commands, inside a report line:
47/// `command` without the symbol, which the window adds. The marker characters
48/// never occur in the game's text.
49#[must_use]
50pub fn link(text: impl Into<String>, command: impl Into<String>) -> String {
51    format!("\u{1}{}\u{2}{}\u{3}", text.into(), command.into())
52}

54/// How many characters a line shows: its links' words, not their commands.
55#[must_use]
56pub fn visible(line: &str) -> usize {
57    let mut shown = 0;
58    let mut hidden = false;
59    for c in line.chars() {
60        match c {
61            '\u{1}' | '\u{3}' => hidden = false,
62            '\u{2}' => hidden = true,
63            _ if !hidden => shown += 1,
64            _ => {}
65        }
66    }
67    shown
68}

70/// The lines as spans, each link a click that sends its command as one of
71/// Hydra's (the window adds the symbol), for a notice whose cells may be
72/// links.
73#[must_use]
74pub fn linked(lines: &[String]) -> Vec<Vec<hydra_session::notice::Span>> {
75    use hydra_session::notice::Span;
76    lines
77        .iter()
78        .map(|line| {
79            let mut spans = Vec::new();
80            let mut rest = line.as_str();
81            while let Some(start) = rest.find('\u{1}') {
82                if start > 0 {
83                    spans.push(Span::plain(&rest[..start]));
84                }
85                let after = &rest[start + 1..];
86                let Some((text, tail)) = after.split_once('\u{2}') else {
87                    rest = after;
88                    break;
89                };
90                let Some((command, tail)) = tail.split_once('\u{3}') else {
91                    spans.push(Span::plain(text));
92                    rest = "";
93                    break;
94                };
95                spans.push(Span::hydra(text, command));
96                rest = tail;
97            }
98            if !rest.is_empty() {
99                spans.push(Span::plain(rest));
100            }
101            spans
102        })
103        .collect()
104}

132/// `n/d (p%)`.
133#[must_use]
134pub fn fraction(n: i64, d: i64) -> String {
135    if d == 0 {
136        format!("{n}/{d} (-)")
137    } else {
138        #[allow(clippy::cast_precision_loss)]
139        let pct = n as f64 * 100.0 / d as f64;
140        format!("{n}/{d} ({pct:.1}%)")
141    }
142}

144/// A grid: a header, a rule, the rows, each column as wide as its widest.
145#[must_use]
146pub fn grid(cols: &[&str], rows: &[Vec<String>]) -> Vec<String> {
147    let widths: Vec<usize> = cols
148        .iter()
149        .enumerate()
150        .map(|(i, col)| {
151            rows.iter()
152                .map(|r| r.get(i).map_or(0, |c| visible(c)))
153                .max()
154                .unwrap_or(0)
155                .max(col.chars().count())
156        })
157        .collect();
158    let rule = format!(
159        "+{}+",
160        widths
161            .iter()
162            .map(|w| "-".repeat(w + 2))
163            .collect::<Vec<_>>()
164            .join("+")
165    );
166    let line = |cells: Vec<String>| {
167        format!(
168            "|{}|",
169            widths
170                .iter()
171                .enumerate()
172                .map(|(i, w)| {
173                    let cell = cells.get(i).map_or("", String::as_str);
174                    let pad = " ".repeat(w.saturating_sub(visible(cell)));
175                    format!(" {cell}{pad} ")
176                })
177                .collect::<Vec<_>>()
178                .join("|")
179        )
180    };
181    let mut out = vec![
182        rule.clone(),
183        line(cols.iter().map(|c| (*c).to_owned()).collect()),
184        rule.clone(),
185    ];
186    out.extend(rows.iter().map(|r| line(r.clone())));
187    out.push(rule);
188    out
189}

191/// A titled grid.
192#[must_use]
193pub fn section(title: &str, cols: &[&str], rows: &[Vec<String>]) -> Vec<String> {
194    let mut out = vec![title.to_owned()];
195    out.extend(grid(cols, rows));
196    out

199/// Which hunts a report is over.
200#[derive(Debug, Clone, PartialEq, Eq)]
201pub enum Scope {
202    /// The latest.
203    Latest,
204    /// One, by id.
205    One(i64),
206    /// Every hunt of this character's.
207    All,
208    /// The last N.
209    Last(usize),
210    /// Every hunt from one id to another, both in.
211    Between(i64, i64),
212}

214/// The hunts resolved: their ids, and a label for the title.
215#[derive(Debug, Clone, PartialEq, Eq)]
216pub struct Hunts {
217    /// The ids, oldest first.
218    pub ids: Vec<i64>,
219    /// `SESSION #3`, `LAST 5 SESSIONS (#2-#6)`, `ALL 12 SESSIONS`.
220    pub label: String,
221    /// The scope as typed: `3`, `last 5`, `all`.
222    pub words: String,
223}
224
225impl Hunts {
226    /// `IN (...)` over the ids, for a query's `WHERE`.
227    #[must_use]
228    pub fn sql_in(&self) -> String {
229        if self.ids.is_empty() {
230            return "IN (NULL)".to_owned();
231        }
232        format!(
233            "IN ({})",
234            self.ids
235                .iter()
236                .map(ToString::to_string)
237                .collect::<Vec<_>>()
238                .join(",")
239        )
240    }
241
242    /// The one hunt, when the scope was one.
243    #[must_use]
244    pub fn single(&self) -> Option<i64> {
245        match self.ids.as_slice() {
246            [id] => Some(*id),
247            _ => None,
248        }
249    }
250}

252/// The hunts `scope` names for `who`. `None`: nothing recorded, or no such
253/// hunt.
254///
255/// # Errors
256///
257/// The file cannot be read.
258pub fn resolve(conn: &Connection, who: &Who, scope: &Scope) -> Result<Option<Hunts>, Error> {
259    let mut stmt = conn.prepare_cached(
260        "SELECT id FROM hunts WHERE instance = ?1 AND character = ?2 ORDER BY id",
261    )?;
262    let all: Vec<i64> = stmt
263        .query_map(params![who.instance, who.character], |r| r.get(0))?
264        .collect::<Result<_, _>>()?;
265    if all.is_empty() {
266        return Ok(None);
267    }
268    Ok(match scope {
269        Scope::Latest => all.last().map(|&id| Hunts {
270            ids: vec![id],
271            label: format!("SESSION #{id}"),
272            words: id.to_string(),
273        }),
274        Scope::One(id) => all.contains(id).then(|| Hunts {
275            ids: vec![*id],
276            label: format!("SESSION #{id}"),
277            words: id.to_string(),
278        }),
279        Scope::All => Some(Hunts {
280            label: format!("ALL {} SESSIONS", all.len()),
281            ids: all,
282            words: "all".to_owned(),
283        }),
284        Scope::Last(n) => {
285            let n = (*n).clamp(1, all.len());
286            let ids = all[all.len() - n..].to_vec();
287            let (label, words) = if n == 1 {
288                (format!("SESSION #{}", ids[0]), ids[0].to_string())
289            } else {
290                (
291                    format!("LAST {n} SESSIONS (#{}-#{})", ids[0], ids[n - 1]),
292                    format!("last {n}"),
293                )
294            };
295            Some(Hunts { ids, label, words })
296        }
297        Scope::Between(a, b) => {
298            let (a, b) = (*a.min(b), *a.max(b));
299            let ids: Vec<i64> = all.into_iter().filter(|id| (a..=b).contains(id)).collect();
300            (!ids.is_empty()).then(|| Hunts {
301                label: format!("SESSIONS #{a}-#{b} ({})", ids.len()),
302                ids,
303                words: format!("between {a} {b}"),
304            })
305        }
306    })
307}

309/// A hit we dealt: its attack is ours, or it rode a flare of ours.
310pub(crate) const OWN_HIT: &str =
311    "(a.ours = 1 OR EXISTS (SELECT 1 FROM flares f WHERE f.id = h.flare_id AND f.ours = 1))";
312/// Damage we dealt to a creature.
313const OUR_DMG: &str = "h.creature_id IS NOT NULL AND (a.ours = 1 OR EXISTS (SELECT 1 FROM flares f WHERE f.id = h.flare_id AND f.ours = 1))";
314/// The kill is ours: the crediting attack `a` is our own swing, or a hit of
315/// ours on creature `c` rode it.
316const OUR_KILL: &str = "(a.ours = 1 OR EXISTS (SELECT 1 FROM hits h JOIN flares f ON f.id = h.flare_id WHERE h.attack_id = a.id AND h.creature_id = c.id AND f.ours = 1))";
317
318/// A creature's attack at us: not a nearby player's, not an unowned tick,
319/// and not what we do to ourselves (the thorns' recoil) or the room does
320/// (the cold), which the reader files as inbound from `self` and
321/// `environment`.
322pub const INBOUND: &str = "a.inbound = 1 AND a.foreign_caster = 0 AND a.unowned = 0 \
323    AND COALESCE(a.attacker, '') NOT IN ('self', 'environment') \
324    AND COALESCE(a.outcome, '') <> 'already_dead' \
325    AND NOT (COALESCE(a.attacker_exist_id, 0) < 0 AND a.outcome IS NULL \
326             AND NOT EXISTS (SELECT 1 FROM hits h WHERE h.attack_id = a.id))";
327/// The flares that stop a creature's attack on us, by name: bark, the
328/// image, the thorns' block, the evanescent shield, the stone barrier.
329pub const REACTIVE_FLARES: &str = "('barkskin', 'mirror_image_def', 'wthorns640_block', 'sward319', 'edefenseiii414_lorebenefit')";
330/// The barrier flares that stop an attack before any roll: the thorns,
331/// the image, and the two whose way is not yet known. A block says nothing
332/// of whether the attack would have hit.
333pub const BLOCK_FLARES: &str =
334    "('mirror_image_def', 'wthorns640_block', 'sward319', 'edefenseiii414_lorebenefit')";
335/// An inbound attack a barrier blocked before any roll: a block flare of
336/// ours on it, or the game's own intercept outcome with no bark absorb.
337pub const BLOCKED: &str = concat!(
338    "(EXISTS (SELECT 1 FROM flares f WHERE f.attack_id = a.id AND f.ours = 1 AND f.name IN ",
339    "('mirror_image_def', 'wthorns640_block', 'sward319', 'edefenseiii414_lorebenefit')) ",
340    "OR (a.outcome = 'intercept' AND NOT EXISTS (SELECT 1 FROM flares f ",
341    "WHERE f.attack_id = a.id AND f.ours = 1 AND f.name = 'barkskin')))"
342);
343/// An inbound attack the bark absorbed: it would have hit.
344pub const ABSORBED: &str = "EXISTS (SELECT 1 FROM flares f WHERE f.attack_id = a.id \
345    AND f.ours = 1 AND f.name = 'barkskin')";
346/// An inbound attack something of ours took before it landed: blocked
347/// before any roll, or absorbed when it would have hit.
348pub const INTERCEPTED: &str = concat!(
349    "(a.outcome = 'intercept' OR EXISTS (SELECT 1 FROM flares f ",
350    "WHERE f.attack_id = a.id AND f.ours = 1 AND f.name IN ",
351    "('barkskin', 'mirror_image_def', 'wthorns640_block', 'sward319', 'edefenseiii414_lorebenefit')))"
352);
353/// An inbound attack that landed: a hit on us, or an outcome of hit, and
354/// not intercepted.
355pub const LANDED: &str = concat!(
356    "NOT (a.outcome = 'intercept' OR EXISTS (SELECT 1 FROM flares f ",
357    "WHERE f.attack_id = a.id AND f.ours = 1 AND f.name IN ",
358    "('barkskin', 'mirror_image_def', 'wthorns640_block', 'sward319', 'edefenseiii414_lorebenefit'))) ",
359    "AND (a.outcome IN ('hit', 'ward_failed') OR EXISTS (SELECT 1 FROM hits h WHERE h.attack_id = a.id AND h.creature_id IS NULL))"
360);

362/// SQL: the effect named `name` (as its dialog shows it) was up as attack
363/// `a` came, by the hunt's intervals.
364#[must_use]
365pub fn effect_up(name: &str) -> String {
366    format!(
367        "EXISTS (SELECT 1 FROM hunt_effects e JOIN names n ON n.id = e.name_id \
368         WHERE e.hunt_id = a.hunt_id AND n.name = '{name}' \
369         AND e.began_at <= a.occurred_at AND (e.ended_at IS NULL OR a.occurred_at < e.ended_at))"
370    )
371}

373/// Inbound attacks over `hunts`: all, landed, missed and intercepted.
374///
375/// # Errors
376///
377/// The file cannot be read.
378pub fn inbound_of(conn: &Connection, hunts: &Hunts) -> Result<Inbound, Error> {
379    let set = hunts.sql_in();
380    let count = |more: &str| -> Result<i64, Error> {
381        one(
382            conn,
383            &format!("SELECT COUNT(*) FROM attacks a WHERE a.hunt_id {set} AND {INBOUND} {more}"),
384            [],
385        )
386    };
387    let all = count("")?;
388    let intercepted = count(&format!("AND {INTERCEPTED}"))?;
389    let landed = count(&format!("AND {LANDED}"))?;
390    Ok(Inbound {
391        all,
392        landed,
393        missed: all - landed - intercepted,
394        intercepted,
395    })
396}

398/// Inbound attacks, sorted.
399#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
400pub struct Inbound {
401    /// Every one ours to receive.
402    pub all: i64,
403    /// Hit us.
404    pub landed: i64,
405    /// Evaded, parried, blocked, or short of the roll.
406    pub missed: i64,
407    /// Taken by a reactive flare of ours: would have hit.
408    pub intercepted: i64,
409}

414    pub fn said(&self) -> String {

419    }

430fn hunt_row(conn: &Connection, id: i64) -> Result<Option<HuntRow>, Error> {
431    let mut stmt = conn.prepare_cached(
432        "SELECT id, started_at, ended_at,
433            (SELECT MAX(occurred_at) FROM attacks WHERE hunt_id = hunts.id)
434         FROM hunts WHERE id = ?1",
435    )?;
436    let mut rows = stmt.query_map(params![id], |r| {
437        Ok(HuntRow {
438            id: r.get(0)?,
439            started: r.get(1)?,
440            ended: r.get(2)?,
441            last: r.get(3)?,
442        })
443    })?;
444    rows.next().transpose()
445}

447fn one<T: rusqlite::types::FromSql + Default>(
448    conn: &Connection,
449    sql: &str,
450    params: impl rusqlite::Params,
451) -> Result<T, Error> {
452    match conn.query_row(sql, params, |r| r.get::<_, Option<T>>(0)) {
453        Ok(value) => Ok(value.unwrap_or_default()),
454        Err(Error::QueryReturnedNoRows) => Ok(T::default()),
455        Err(e) => Err(e),
456    }
457}

459/// Kills and assists over `hunts`: a kill, the creature died and our attack
460/// landed the blow; an assist, we damaged it and someone else did.
461fn kills_and_assists(conn: &Connection, hunts: &Hunts) -> Result<(i64, i64), Error> {
462    let set = hunts.sql_in();
463    let kills: i64 = one(
464        conn,
465        &format!(
466            "SELECT COUNT(*) FROM creatures c JOIN attacks a ON a.id = c.killed_by_attack_id
467             WHERE c.hunt_id {set} AND c.killed_at IS NOT NULL AND {OUR_KILL}"
468        ),
469        [],
470    )?;
471    let assists: i64 = one(
472        conn,
473        &format!(
474            "SELECT COUNT(*) FROM creatures c
475             WHERE c.hunt_id {set} AND c.killed_at IS NOT NULL
476               AND EXISTS (SELECT 1 FROM hits h JOIN attacks a ON a.id = h.attack_id
477                           WHERE h.creature_id = c.id AND {OWN_HIT})
478               AND (c.killed_by_attack_id IS NULL
479                    OR NOT EXISTS (SELECT 1 FROM attacks a WHERE a.id = c.killed_by_attack_id AND {OUR_KILL}))"
480        ),
481        [],
482    )?;
483    Ok((kills, assists))
484}

486/// The headline numbers over `hunts`.
487struct Totals {
488    attacks: i64,
489    sequences: i64,
490    inbound: Inbound,
491    kills: i64,
492    assists: i64,
493    dealt: i64,
494    taken: i64,
495}
496
497fn totals(conn: &Connection, hunts: &Hunts) -> Result<Totals, Error> {
498    let set = hunts.sql_in();
499    let (attacks, sequences): (i64, i64) = conn.query_row(
500        &format!(
501            "SELECT COUNT(*), COUNT(DISTINCT COALESCE(a.root_attack_id, a.id)) FROM attacks a
502             WHERE a.hunt_id {set} AND a.ours = 1"
503        ),
504        [],
505        |r| Ok((r.get(0)?, r.get(1)?)),
506    )?;
507    let inbound = inbound_of(conn, hunts)?;
508    let (kills, assists) = kills_and_assists(conn, hunts)?;
509    let dealt = one(
510        conn,
511        &format!(
512            "SELECT COALESCE(SUM(h.damage), 0) FROM hits h JOIN attacks a ON a.id = h.attack_id
513             WHERE h.hunt_id {set} AND {OUR_DMG}"
514        ),
515        [],
516    )?;
517    let taken = one(
518        conn,
519        &format!(
520            "SELECT COALESCE(SUM(h.damage), 0) FROM hits h JOIN attacks a ON a.id = h.attack_id
521             WHERE h.hunt_id {set} AND h.creature_id IS NULL AND a.inbound = 1"
522        ),
523        [],
524    )?;
525    Ok(Totals {
526        attacks,
527        sequences,
528        inbound,
529        kills,
530        assists,
531        dealt,
532        taken,
533    })
534}

539    noun: String,
540    name: String,
541    first: Option<i64>,
542    last: Option<i64>,
543    killed: Option<i64>,

549fn creatures_of(conn: &Connection, hunts: &Hunts) -> Result<Vec<CreatureRow>, Error> {
550    let set = hunts.sql_in();
551    let mut stmt = conn.prepare(&format!(
552        "SELECT c.exist_id, c.noun, c.name, c.first_seen, c.last_seen, c.killed_at,
553            (SELECT COUNT(*) FROM attacks a WHERE a.creature_id = c.id AND a.ours = 1),
554            (SELECT COALESCE(SUM(h.damage), 0) FROM hits h JOIN attacks a ON a.id = h.attack_id
555              WHERE h.creature_id = c.id AND {OWN_HIT}),
556            COALESCE((SELECT {OUR_KILL} FROM attacks a WHERE a.id = c.killed_by_attack_id), 0)
557         FROM creatures c WHERE c.hunt_id {set} ORDER BY c.noun, c.first_seen"
558    ))?;
559    let rows = stmt.query_map([], |r| {
560        Ok(CreatureRow {
561            exist: r.get(0)?,
562            noun: r
563                .get::<_, Option<String>>(1)?
564                .unwrap_or_else(|| "(unknown)".to_owned()),
565            name: r.get::<_, Option<String>>(2)?.unwrap_or_default(),
566            first: r.get(3)?,
567            last: r.get(4)?,
568            killed: r.get(5)?,
569            attacks: r.get(6)?,
570            damage: r.get(7)?,
571            our_kill: r.get::<_, i64>(8)? == 1,
572        })
573    })?;
574    rows.collect()
575}

577/// Creatures grouped by kind: `Kind Seen Kills Other-dead Alive Attacks Damage`.
578fn kind_rows(creatures: &[CreatureRow], per_hour: Option<f64>) -> Vec<Vec<String>> {
579    let mut kinds: Vec<(&str, Vec<&CreatureRow>)> = Vec::new();
580    for c in creatures {
581        match kinds.iter_mut().find(|(k, _)| *k == c.noun) {
582            Some((_, group)) => group.push(c),
583            None => kinds.push((c.noun.as_str(), vec![c])),
584        }
585    }
586    if per_hour.is_some() {
587        kinds.sort_by_key(|(_, g)| std::cmp::Reverse(g.iter().filter(|c| c.our_kill).count()));
588    }
589    kinds
590        .into_iter()
591        .map(|(noun, group)| {
592            let kills = group.iter().filter(|c| c.our_kill).count();
593            let dead = group.iter().filter(|c| c.killed.is_some()).count();
594            let mut row = vec![
595                link(noun.to_uppercase(), format!("cstats creature {noun}")),
596                group.len().to_string(),
597                kills.to_string(),
598                (dead - kills).to_string(),
599                (group.len() - dead).to_string(),
600                group.iter().map(|c| c.attacks).sum::<i64>().to_string(),
601                group.iter().map(|c| c.damage).sum::<i64>().to_string(),
602            ];
603            if let Some(hours) = per_hour {
604                #[allow(clippy::cast_precision_loss)]
605                let rate = if hours > 0.0 {
606                    format!("{:.1}", kills as f64 / hours)
607                } else {
608                    "-".to_owned()
609                };
610                row.push(rate);
611            }
612            row
613        })
614        .collect()
615}

673/// The gemstones a hunt ran with, as one line: each equipped gemstone and
674/// its properties with their ranks. Nothing when the hunt's set was never
675/// known.
676///
677/// # Errors
678///
679/// The file cannot be read.
680pub fn gemstones_line(conn: &Connection, hunt: i64) -> Result<Vec<String>, Error> {
681    let mut stmt = conn.prepare_cached(
682        "SELECT description, name, rank, ranks FROM hunt_gemstones WHERE hunt_id = ?1 ORDER BY number, id",
683    )?;
684    let rows: Vec<(String, String, Option<i64>, Option<i64>)> = stmt
685        .query_map(params![hunt], |r| {
686            Ok((r.get(0)?, r.get(1)?, r.get(2)?, r.get(3)?))
687        })?
688        .collect::<Result<_, _>>()?;
689    if rows.is_empty() {
690        return Ok(Vec::new());
691    }
692    let mut gems: Vec<(String, Vec<String>)> = Vec::new();
693    for (description, name, rank, ranks) in rows {
694        let property = match (rank, ranks) {
695            (Some(rank), Some(ranks)) => format!("{name} {rank}/{ranks}"),
696            _ => name,
697        };
698        match gems.last_mut() {
699            Some((last, properties)) if *last == description => properties.push(property),
700            _ => gems.push((description, vec![property])),
701        }
702    }
703    let said: Vec<String> = gems
704        .into_iter()
705        .map(|(description, properties)| format!("{description} ({})", properties.join(", ")))
706        .collect();
707    Ok(vec![format!("  Gemstones: {}", said.join("; "))])
708}

710/// The items worn as the hunt opened, as one line; a change during the
711/// hunt is noted with its time. Nothing when no list was known.
712///
713/// # Errors
714///
715/// The file cannot be read.
716pub fn worn_line(conn: &Connection, hunt: i64) -> Result<Vec<String>, Error> {
717    let mut stmt = conn.prepare_cached(
718        "SELECT w.began_at, w.ended_at, n.name FROM hunt_gear w JOIN names n ON n.id = w.gear_id
719         WHERE w.hunt_id = ?1 AND w.slot = 'worn' ORDER BY w.began_at, w.id",
720    )?;
721    let rows: Vec<(i64, Option<i64>, String)> = stmt
722        .query_map(params![hunt], |r| Ok((r.get(0)?, r.get(1)?, r.get(2)?)))?
723        .collect::<Result<_, _>>()?;
724    let Some((first, _, _)) = rows.first() else {
725        return Ok(Vec::new());
726    };
727    let first = *first;
728    let opened: Vec<&str> = rows
729        .iter()
730        .filter(|(at, _, _)| *at == first)
731        .map(|(_, _, name)| name.as_str())
732        .collect();
733    // A change is an item put on after the opening or taken off before
734    // the hunt's end; the close stamps every open row alike, so an end
735    // shared by all is the hunt's, not a change.
736    let ends: Vec<i64> = rows.iter().filter_map(|(_, end, _)| *end).collect();
737    let common_end = (ends.len() == rows.len())
738        .then(|| ends.iter().max().copied())
739        .flatten();
740    let changes = rows
741        .iter()
742        .flat_map(|(at, end, _)| {
743            let mut when = Vec::new();
744            if *at != first {
745                when.push(*at);
746            }
747            if let Some(end) = end
748                && Some(*end) != common_end
749            {
750                when.push(*end);
751            }
752            when
753        })
754        .collect::<std::collections::BTreeSet<_>>();
755    let mut lines = vec![format!("  Worn: {}", opened.join(", "))];
756    if !changes.is_empty() {
757        lines.push(format!(
758            "  Worn list changed at {}",
759            changes
760                .iter()
761                .map(|at| clock(*at))
762                .collect::<Vec<_>>()
763                .join(", ")
764        ));
765    }
766    Ok(lines)
767}

769/// The effects up as the hunt opened, as one line, and how many came or
770/// went during it. Nothing when none was known.
771///
772/// # Errors
773///
774/// The file cannot be read.
775pub fn effects_line(conn: &Connection, hunt: i64) -> Result<Vec<String>, Error> {
776    let started: i64 = conn.query_row(
777        "SELECT started_at FROM hunts WHERE id = ?1",
778        params![hunt],
779        |r| r.get(0),
780    )?;
781    let mut stmt = conn.prepare_cached(
782        "SELECT n.name, e.began_at FROM hunt_effects e JOIN names n ON n.id = e.name_id
783         WHERE e.hunt_id = ?1 ORDER BY e.began_at, n.kind, n.name",
784    )?;
785    let rows: Vec<(String, i64)> = stmt
786        .query_map(params![hunt], |r| Ok((r.get(0)?, r.get(1)?)))?
787        .collect::<Result<_, _>>()?;
788    if rows.is_empty() {
789        return Ok(Vec::new());
790    }
791    let opened: Vec<&str> = rows
792        .iter()
793        .filter(|(_, at)| *at == started)
794        .map(|(name, _)| name.as_str())
795        .collect();
796    let later = rows.len() - opened.len();
797    let mut lines = vec![format!("  Up: {}", opened.join(", "))];
798    if later > 0 {
799        lines.push(format!(
800            "  {later} more came up during the hunt {}",
801            link("[effects]", format!("cstats effects {hunt}"))
802        ));
803    }
804    Ok(lines)
805}

819/// A hunt's report: header, the numbers, the creatures by kind; `full`
820/// lists every creature instance instead.
821///
822/// # Errors
823///
824/// The file cannot be read.
825pub fn session(
826    conn: &Connection,
827    who: &Who,
828    id: Option<i64>,
829    full: bool,
830) -> Result<Vec<String>, Error> {
831    let scope = id.map_or(Scope::Latest, Scope::One);
832    let Some(hunts) = resolve(conn, who, &scope)? else {
833        return Ok(vec![match id {
834            Some(id) => format!("No hunt #{id}."),
835            None => "No hunts recorded yet.".to_owned(),
836        }]);
837    };
838    let Some(id) = hunts.single() else {
839        return Ok(vec!["No hunts recorded yet.".to_owned()]);
840    };
841    let Some(hunt) = hunt_row(conn, id)? else {
842        return Ok(vec![format!("No hunt #{id}.")]);
843    };
844    let t = totals(conn, &hunts)?;
845    let creatures = creatures_of(conn, &hunts)?;
846    let mut lines = vec![
847        format!("=== SESSION REPORT #{id} ==="),
848        format!("Character: {}", who.character),
849        format!("Started: {}", stamp(hunt.started)),
850        format!("Last Event: {}", stamp(hunt.last.unwrap_or(hunt.started))),
851        match hunt.ended {
852            Some(ended) => format!(
853                "Ended: {} ({})",
854                stamp(ended),
855                duration(Some(ended - hunt.started))
856            ),
857            None => "Ended: ACTIVE".to_owned(),
858        },
859        String::new(),
860        "SESSION STATS:".to_owned(),
861        format!(
862            "  Total Attacks: {}    Sequences: {}    Inbound: {}",
863            t.attacks,
864            t.sequences,
865            t.inbound.said()
866        ),
867        format!(
868            "  Kills: {}    Assists: {}    Damage Dealt: {}    Damage Taken: {}",
869            t.kills, t.assists, t.dealt, t.taken
870        ),
871    ];
872    lines.extend(kit_lines(conn, id)?);
873    lines.push(String::new());
874    if full {
875        lines.push(format!("CREATURES ENCOUNTERED ({}):", creatures.len()));
876        lines.push("-".repeat(60));
877        let mut kinds: Vec<&str> = Vec::new();
878        for c in &creatures {
879            if !kinds.contains(&c.noun.as_str()) {
880                kinds.push(c.noun.as_str());
881            }
882        }
883        for kind in kinds {
884            let group: Vec<&CreatureRow> = creatures.iter().filter(|c| c.noun == kind).collect();
885            lines.push(String::new());
886            lines.push(format!("{} ({}):", kind.to_uppercase(), group.len()));
887            for (i, c) in group.iter().enumerate() {
888                let label = c.exist.map_or_else(
889                    || "#?".to_owned(),
890                    |e| link(format!("#{e}"), format!("cstats creature {e}")),
891                );
892                let status = match (c.killed, c.our_kill, c.damage > 0) {
893                    (None, _, _) => "ALIVE",
894                    (Some(_), true, _) => "DEAD (kill)",
895                    (Some(_), false, true) => "DEAD (assist)",
896                    (Some(_), false, false) => "DEAD",
897                };
898                let lasted = match (c.first, c.last) {
899                    (Some(first), Some(last)) => duration(Some(last - first)),
900                    _ => "?".to_owned(),
901                };
902                lines.push(format!("  [{}] {label} - {}", i + 1, c.name));
903                lines.push(format!(
904                    "      Status: {status} | Attacks: {} | Damage: {} | Duration: {lasted}",
905                    c.attacks, c.damage
906                ));
907            }
908        }
909    } else {
910        lines.push(format!(
911            "CREATURES ENCOUNTERED ({}) - {} lists every instance:",
912            creatures.len(),
913            link("[full report]", format!("cstats full {id}"))
914        ));
915        lines.extend(grid(
916            &[
917                "Kind",
918                "Seen",
919                "Kills",
920                "Other-dead",
921                "Alive",
922                "Attacks",
923                "Damage",
924            ],
925            &kind_rows(&creatures, None),
926        ));
927    }
928    lines.extend(quick_actions(Some(&hunts)));
929    Ok(lines)
930}

932/// The numbers summed over a scope, with per-hour rates and a per-hunt
933/// grid; creature kinds pooled across the hunts.
934///
935/// # Errors
936///
937/// The file cannot be read.
938// The report is one piece: its header, totals and grids read in order.
939#[allow(clippy::too_many_lines)]
940pub fn aggregate(conn: &Connection, who: &Who, scope: &Scope) -> Result<Vec<String>, Error> {
941    let Some(hunts) = resolve(conn, who, scope)? else {
942        return Ok(vec!["No hunts recorded yet.".to_owned()]);
943    };
944    let mut rows = Vec::new();
945    let mut hunt_time = 0_i64;
946    let mut first = None;
947    let mut last = None;
948    for &id in &hunts.ids {
949        let Some(hunt) = hunt_row(conn, id)? else {
950            continue;
951        };
952        let end = hunt.ended.or(hunt.last).unwrap_or(hunt.started);
953        hunt_time += end - hunt.started;
954        first = first.or(Some(hunt.started));
955        last = Some(end);
956        rows.push(hunt);
957    }
958    let t = totals(conn, &hunts)?;
959    let creatures = creatures_of(conn, &hunts)?;
960    #[allow(clippy::cast_precision_loss)]
961    let hours = hunt_time as f64 / 3600.0;
962    let per_hour = |n: i64| {
963        if hunt_time > 0 {
964            #[allow(clippy::cast_precision_loss)]
965            let rate = n as f64 / hours;
966            format!("{rate:.1}")
967        } else {
968            "-".to_owned()
969        }
970    };
971    let mut lines = vec![
972        format!("=== AGGREGATE REPORT - {} ===", hunts.label),
973        format!("Character: {}", who.character),
974        format!(
975            "Hunts: {}    First: {}    Last: {}",
976            rows.len(),
977            first.map_or_else(String::new, stamp),
978            last.map_or_else(String::new, stamp)
979        ),
980        format!(
981            "Hunt time: {} (avg {} per hunt)",
982            duration(Some(hunt_time)),
983            duration(Some(
984                hunt_time / i64::try_from(rows.len().max(1)).unwrap_or(1)
985            ))
986        ),
987        String::new(),
988        "TOTALS:".to_owned(),
989        format!(
990            "  Total Attacks: {}    Sequences: {}    Inbound: {}",
991            t.attacks,
992            t.sequences,
993            t.inbound.said()
994        ),
995        format!(
996            "  Kills: {}    Assists: {}    Damage Dealt: {}    Damage Taken: {}",
997            t.kills, t.assists, t.dealt, t.taken
998        ),
999        format!(
1000            "  Per hour: {} kills    {} damage    {} attacks    {} taken",
1001            per_hour(t.kills),
1002            per_hour(t.dealt),
1003            per_hour(t.attacks),
1004            per_hour(t.taken)
1005        ),
1006        String::new(),
1007        format!("CREATURES ENCOUNTERED ({}):", creatures.len()),
1008    ];
1009    lines.extend(grid(
1010        &[
1011            "Kind",
1012            "Seen",
1013            "Kills",
1014            "Other-dead",
1015            "Alive",
1016            "Attacks",
1017            "Damage",
1018            "Kills/hr",
1019        ],
1020        &kind_rows(&creatures, Some(hours)),
1021    ));
1022    if rows.len() <= 20 {
1023        lines.push(String::new());
1024        lines.push("PER HUNT:".to_owned());
1025        let mut per = Vec::new();
1026        for hunt in &rows {
1027            let one_hunt = Hunts {
1028                ids: vec![hunt.id],
1029                label: String::new(),
1030                words: hunt.id.to_string(),
1031            };
1032            let t = totals(conn, &one_hunt)?;
1033            per.push(vec![
1034                link(hunt.id.to_string(), format!("cstats {}", hunt.id)),
1035                stamp(hunt.started),
1036                hunt.ended
1037                    .map_or_else(|| "ACTIVE".to_owned(), |e| duration(Some(e - hunt.started))),
1038                t.attacks.to_string(),
1039                t.kills.to_string(),
1040                t.dealt.to_string(),
1041            ]);
1042        }
1043        lines.extend(grid(
1044            &["ID", "Started", "Duration", "Attacks", "Kills", "Damage"],
1045            &per,
1046        ));
1047    } else {
1048        lines.push(String::new());
1049        lines.push(format!(
1050            "{} hunts; {} lists them.",
1051            rows.len(),
1052            link("[sessions]", format!("cstats sessions {}", rows.len()))
1053        ));
1054    }
1055    lines.extend(quick_actions(Some(&hunts)));
1056    Ok(lines)
1057}

1059/// Damage by ability and the flares' rollup over a scope, and the attacks
1060/// against us.
1061///
1062/// # Errors
1063///
1064/// The file cannot be read.
1065// Three grids, one query each, said in order.
1066#[allow(clippy::too_many_lines)]
1067pub fn abilities(conn: &Connection, who: &Who, scope: &Scope) -> Result<Vec<String>, Error> {
1068    let Some(hunts) = resolve(conn, who, scope)? else {
1069        return Ok(vec!["No hunts recorded yet.".to_owned()]);
1070    };
1071    let set = hunts.sql_in();
1072    let mut lines = vec![format!("=== ABILITIES - {} ===", hunts.label)];
1073    let mut stmt = conn.prepare(&format!(
1074        "SELECT a.name, a.parent, COUNT(*),
1075            COALESCE(SUM(EXISTS (SELECT 1 FROM hits h WHERE h.attack_id = a.id AND h.flare_id IS NULL AND h.creature_id IS NOT NULL)), 0),
1076            COALESCE(SUM((SELECT COUNT(*) FROM hits h WHERE h.attack_id = a.id AND h.flare_id IS NULL AND h.creature_id IS NOT NULL)), 0),
1077            COALESCE(SUM((SELECT COALESCE(SUM(h.damage), 0) FROM hits h WHERE h.attack_id = a.id AND h.flare_id IS NULL AND h.creature_id IS NOT NULL)), 0),
1078            COALESCE(SUM((SELECT COUNT(*) FROM hits h WHERE h.attack_id = a.id AND h.flare_id IS NULL AND h.creature_id IS NOT NULL AND h.crit_rank IS NOT NULL)), 0),
1079            COALESCE(SUM((SELECT COUNT(*) FROM hits h WHERE h.attack_id = a.id AND h.flare_id IS NULL AND h.creature_id IS NOT NULL AND h.fatal = 1)), 0)
1080         FROM attacks a WHERE a.hunt_id {set} AND a.ours = 1
1081         GROUP BY a.name, a.parent ORDER BY 6 DESC"
1082    ))?;
1083    let rows: Vec<Vec<String>> = stmt
1084        .query_map([], |r| {
1085            let name: String = r.get(0)?;
1086            let parent: Option<String> = r.get(1)?;
1087            let attempts: i64 = r.get(2)?;
1088            let landed: i64 = r.get(3)?;
1089            let hits: i64 = r.get(4)?;
1090            let total: i64 = r.get(5)?;
1091            let crits: i64 = r.get(6)?;
1092            let fatal: i64 = r.get(7)?;
1093            let label = match parent {
1094                Some(p) => format!("{} (via {p})", link(&name, format!("cstats attack {name}"))),
1095                None => link(&name, format!("cstats attack {name}")),
1096            };
1097            #[allow(clippy::cast_precision_loss)]
1098            let avg = if hits > 0 {
1099                format!("{:.1}", total as f64 / hits as f64)
1100            } else {
1101                "-".to_owned()
1102            };
1103            Ok(vec![
1104                label,
1105                attempts.to_string(),
1106                landed.to_string(),
1107                hits.to_string(),
1108                total.to_string(),
1109                avg,
1110                crits.to_string(),
1111                fatal.to_string(),
1112            ])
1113        })?
1114        .collect::<Result<_, _>>()?;
1115    if !rows.is_empty() {
1116        lines.push(
1117            "Damage by ability (own swings; Landed = swings with a hit, Hits = damage components):"
1118                .to_owned(),
1119        );
1120        lines.extend(grid(
1121            &[
1122                "Ability", "Attempts", "Landed", "Hits", "Dmg", "Avg", "Crits", "Fatal",
1123            ],
1124            &rows,
1125        ));
1126    }
1127    let mut stmt = conn.prepare(&format!(
1128        "SELECT f.name, COUNT(DISTINCT f.id), COUNT(h.id), COALESCE(SUM(h.damage), 0),
1129            COALESCE(SUM(h.fatal), 0)
1130         FROM flares f JOIN attacks a ON a.id = f.attack_id
1131         LEFT JOIN hits h ON h.flare_id = f.id
1132         WHERE a.hunt_id {set} AND f.ours = 1
1133         GROUP BY f.name ORDER BY 4 DESC, 2 DESC"
1134    ))?;
1135    let rows: Vec<Vec<String>> = stmt
1136        .query_map([], |r| {
1137            Ok(vec![
1138                {
1139                    let name: String = r.get(0)?;
1140                    link(&name, format!("cstats flare {name}"))
1141                },
1142                r.get::<_, i64>(1)?.to_string(),
1143                r.get::<_, i64>(2)?.to_string(),
1144                r.get::<_, i64>(3)?.to_string(),
1145                r.get::<_, i64>(4)?.to_string(),
1146            ])
1147        })?
1148        .collect::<Result<_, _>>()?;
1149    if !rows.is_empty() {
1150        lines.push(String::new());
1151        lines.push("Flares (own; cstats flare <name> for one):".to_owned());
1152        lines.extend(grid(&["Flare", "Procs", "Hits", "Dmg", "Fatal"], &rows));
1153    }
1154    let mut stmt = conn.prepare(&format!(
1155        "SELECT a.name, a.attacker, COUNT(*),
1156            COALESCE(SUM({LANDED}), 0), COALESCE(SUM({INTERCEPTED}), 0),
1157            GROUP_CONCAT(DISTINCT a.outcome),
1158            COALESCE(SUM((SELECT COALESCE(SUM(h.damage), 0) FROM hits h WHERE h.attack_id = a.id AND h.creature_id IS NULL)), 0)
1159         FROM attacks a WHERE a.hunt_id {set} AND {INBOUND}
1160         GROUP BY a.name, a.attacker ORDER BY 3 DESC"
1161    ))?;
1162    let rows: Vec<Vec<String>> = stmt
1163        .query_map([], |r| {
1164            let all: i64 = r.get(2)?;
1165            let landed: i64 = r.get(3)?;
1166            let intercepted: i64 = r.get(4)?;
1167            Ok(vec![
1168                r.get::<_, String>(0)?,
1169                r.get::<_, Option<String>>(1)?.unwrap_or_default(),
1170                all.to_string(),
1171                landed.to_string(),
1172                (all - landed - intercepted).to_string(),
1173                intercepted.to_string(),
1174                r.get::<_, Option<String>>(5)?.unwrap_or_default(),
1175                r.get::<_, i64>(6)?.to_string(),
1176            ])
1177        })?
1178        .collect::<Result<_, _>>()?;
1179    if !rows.is_empty() {
1180        lines.push(String::new());
1181        lines.push("Attacks against you:".to_owned());
1182        lines.extend(grid(
1183            &[
1184                "Attack",
1185                "Attacker",
1186                "N",

1//! Where Hydra's memory goes.
2//!
3//! Built with the `measure` feature, every allocation is counted against
4//! the part of Hydra whose code made it, for as long as it is held: [`part`]
5//! marks a stretch of code as one part's, and [`held`] says what each part
6//! holds now. Memory made outside every part is *other*. Built without it,
7//! [`part`] only runs its code and [`held`] says nothing, so no other build
8//! pays for the counting.
9//!
10//! What is counted is what Rust allocates. Memory the graphics driver or the
11//! system holds for Hydra is not, so the parts add up to less than the
12//! process's committed memory: the difference is theirs.
13//!
14//! A part is marked on stretches of code that run on one thread without
15//! pausing, since what is being counted follows the thread, or on a task
16//! ([`task`]), counted each time it runs, or on a whole thread ([`thread`]),
17//! for what it makes outside every other part. A part inside another counts
18//! as its own: the combat matching inside the game text counts as combat.
19
20/// A part of Hydra whose memory is counted on its own.
21#[derive(Clone, Copy, Debug, PartialEq, Eq)]
22pub enum Part {
23    /// The game's text read into each character's state: the parser and
24    /// the model, but for the combat matching.
25    GameText,
26    /// Combat lines matched against the attack, flare, outcome and crit
27    /// patterns, and what the matching keeps between lines, but for the
28    /// gates.
29    Combat,
30    /// The gates over the combat families and the crit patterns no first
31    /// word places, each line asked of them: what their lazy DFAs learn.
32    CombatGates,
33    /// Each character's whole state copied for its windows and the page.
34    Snapshots,
35    /// Each character's story: the lines it shows, heard from its session.
36    Story,
37    /// The highlights matched against each line.
38    Highlights,
39    /// Each character's session but for its game text and snapshots: its
40    /// connection, its commands, what it records, and what it keeps.
41    Session,
42    /// The windows' feed from each session: its events heard, its
43    /// highlights and triggers built and handed on, its lines sent.
44    Feeds,
45    /// The relay between each character and its Lich.
46    Lich,
47    /// The web page: served, and each browser on it answered.
48    WebPage,
49    /// Hydra's own commands run as tasks: travel, casting, timers, the
50    /// inventory, the survey, the bestiary, the history.
51    Commands,
52    /// The ledger's writing thread: the rows waiting, and the file written.
53    Ledger,
54    /// The windows, while they are open.
55    Windows,
56    /// The Infinite doll, drawn, and what it keeps.
57    Doll,
58    /// The map: read, and its areas laid out.
59    Map,
60    /// The windows' own threads: logs and settings written, sounds and
61    /// notifications, the logs window read.
62    WindowThreads,
63    /// Whatever else the runtime's threads run, outside every part.
64    Background,
65    /// The main thread outside the windows: starting up, and resting.
66    MainThread,
67}

69impl Part {
70    /// Every part, in the order [`held`] says them.
71    pub const ALL: [Self; 18] = [
72        Self::GameText,
73        Self::Combat,
74        Self::CombatGates,
75        Self::Snapshots,
76        Self::Story,
77        Self::Highlights,
78        Self::Session,
79        Self::Feeds,
80        Self::Lich,
81        Self::WebPage,
82        Self::Commands,
83        Self::Ledger,
84        Self::Windows,
85        Self::Doll,
86        Self::Map,
87        Self::WindowThreads,
88        Self::Background,
89        Self::MainThread,
90    ];
91
92    /// As the memory line says it.
93    #[must_use]
94    pub fn said(self) -> &'static str {
95        match self {
96            Self::GameText => "game text",
97            Self::Combat => "combat",
98            Self::CombatGates => "combat gates",
99            Self::Snapshots => "snapshots",
100            Self::Story => "story",
101            Self::Highlights => "highlights",
102            Self::Session => "session",
103            Self::Feeds => "feeds",
104            Self::Lich => "lich",
105            Self::WebPage => "web page",
106            Self::Commands => "commands",
107            Self::Ledger => "ledger",
108            Self::Windows => "windows",
109            Self::Doll => "doll",
110            Self::Map => "map",
111            Self::WindowThreads => "window threads",
112            Self::Background => "background tasks",
113            Self::MainThread => "main thread",
114        }
115    }
116}

118#[cfg(not(feature = "measure"))]
119mod off {
120    use super::Part;
121
122    /// Run `run` as `part`'s: counted, in a measuring build.
123    #[inline]
124    pub fn part<R>(_part: Part, run: impl FnOnce() -> R) -> R {
125        run()
126    }
127
128    /// The task `run` as `part`'s: counted, in a measuring build.
129    #[inline]
130    pub fn task<F: Future>(_part: Part, run: F) -> F {
131        run
132    }
133
134    /// This thread's own part, in a measuring build.
135    pub fn thread(_part: Part) {}
136
137    /// Start counting, in a measuring build.
138    pub fn start() {}
139
140    /// What each part holds now, in a measuring build; `None` in this one.
141    #[must_use]
142    pub fn held() -> Option<String> {
143        None
144    }
145}

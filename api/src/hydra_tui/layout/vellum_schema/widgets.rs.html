16/// A window's persisted show/hide state. Replaces the old `visible: bool`.
17/// `Hidden` means BOTH "don't render" AND "suppress the game from
18/// auto-spawning it" — the unified-windows rule. `Ephemeral` marks a
19/// session-only window (containers) that is never persisted and is wiped
20/// on relog; it renders like `Shown` while alive.
21#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
22pub enum WindowVisibility {
23    /// Drawn, and kept in the layout.
24    #[default]
25    Shown,
26    /// Not drawn, and not opened when the game asks for it.
27    Hidden,
28    /// Drawn for this session only, never written to the layout.
29    Ephemeral,
30}
31
32impl WindowVisibility {
33    /// Whether the window should render.
34    #[must_use]
35    pub fn is_shown(&self) -> bool {
36        matches!(self, WindowVisibility::Shown | WindowVisibility::Ephemeral)
37    }
38    /// Whether the game may auto-(re)spawn this window. Hidden suppresses it.
39    #[must_use]
40    pub fn allows_autospawn(&self) -> bool {
41        !matches!(self, WindowVisibility::Hidden)
42    }
43    /// Whether this window persists to layout.toml. Ephemeral does not.
44    #[must_use]
45    pub fn is_persistent(&self) -> bool {
46        !matches!(self, WindowVisibility::Ephemeral)
47    }
48}

50// Written as `"shown"` or `"hidden"`; read from that or from a
51// `visible = true|false`, which some layout files have. A session-only
52// window is never written.
53impl serde::Serialize for WindowVisibility {
54    fn serialize<S: serde::Serializer>(&self, s: S) -> Result<S::Ok, S::Error> {
55        s.serialize_str(match self {
56            WindowVisibility::Shown => "shown",
57            WindowVisibility::Hidden => "hidden",
58            WindowVisibility::Ephemeral => "ephemeral",
59        })
60    }
61}

63impl<'de> serde::Deserialize<'de> for WindowVisibility {
64    fn deserialize<D: serde::Deserializer<'de>>(d: D) -> Result<Self, D::Error> {
65        use serde::de::Error;
66        #[derive(serde::Deserialize)]
67        #[serde(untagged)]
68        enum Compat {
69            Bool(bool),
70            Str(String),
71        }
72        match Compat::deserialize(d)? {
73            // Legacy layout.toml: visible = true|false.
74            Compat::Bool(true) => Ok(WindowVisibility::Shown),
75            Compat::Bool(false) => Ok(WindowVisibility::Hidden),
76            Compat::Str(s) => match s.to_ascii_lowercase().as_str() {
77                "shown" | "visible" | "true" => Ok(WindowVisibility::Shown),
78                "hidden" | "false" => Ok(WindowVisibility::Hidden),
79                "ephemeral" => Ok(WindowVisibility::Ephemeral),
80                other => Err(D::Error::custom(format!(
81                    "invalid window visibility '{other}'"
82                ))),
83            },
84        }
85    }
86}

88/// What game source a window is bound to, so the client can find the ONE

96    Dialog(String),
97    /// A game stream id (thoughts, loot, bounty, ...).

101}

145// Convert from TOML array format ["left", "right"] to BorderSides struct
146impl From<Vec<String>> for BorderSides {
147    fn from(sides: Vec<String>) -> Self {
148        let mut border = Self {
149            top: false,
150            bottom: false,
151            left: false,
152            right: false,
153        };
154
155        for side in sides {
156            match side.to_lowercase().as_str() {
157                "top" => border.top = true,
158                "bottom" => border.bottom = true,
159                "left" => border.left = true,
160                "right" => border.right = true,
161                _ => {} // Ignore unknown sides
162            }
163        }
164
165        border
166    }
167}

197impl WindowBase {
198    fn horizontal_border_units_for(show: bool, sides: &BorderSides) -> u16 {
199        if !show {
200            return 0;
201        }
202        u16::from(sides.top) + u16::from(sides.bottom)
203    }
204
205    fn vertical_border_units_for(show: bool, sides: &BorderSides) -> u16 {
206        if !show {
207            return 0;
208        }
209        u16::from(sides.left) + u16::from(sides.right)
210    }
211
212    /// Number of rows consumed by borders (top + bottom)
213    #[must_use]
214    pub fn horizontal_border_units(&self) -> u16 {
215        Self::horizontal_border_units_for(self.show_border, &self.border_sides)
216    }
217
218    /// Number of columns consumed by borders (left + right)
219    #[must_use]
220    pub fn vertical_border_units(&self) -> u16 {
221        Self::vertical_border_units_for(self.show_border, &self.border_sides)
222    }
223
224    /// Rows available for the widget's interior content
225    #[must_use]
226    pub fn content_rows(&self) -> u16 {
227        self.rows
228            .get()
229            .saturating_sub(self.horizontal_border_units())
230    }
231
232    /// Columns available for the widget's interior content
233    #[must_use]
234    pub fn content_cols(&self) -> u16 {
235        self.cols.get().saturating_sub(self.vertical_border_units())
236    }
237
238    /// Apply new border visibility/sides while keeping interior size the same.
239    /// Also adjusts `min_rows/max_rows/min_cols/max_cols` proportionally (if set).
240    pub fn apply_border_configuration(&mut self, show_border: bool, border_sides: BorderSides) {
241        let prev_horizontal = self.horizontal_border_units();
242        let prev_vertical = self.vertical_border_units();
243
244        // Calculate content dimensions (interior without borders)
245        let content_rows = self.rows.get().saturating_sub(prev_horizontal).max(1);
246        let content_cols = self.cols.get().saturating_sub(prev_vertical).max(1);
247
248        // Calculate content-based min/max (if set) - None stays None
249        let content_min_rows = self
250            .min_rows
251            .map(|m| m.saturating_sub(prev_horizontal).max(1));
252        let content_max_rows = self
253            .max_rows
254            .map(|m| m.saturating_sub(prev_horizontal).max(1));
255        let content_min_cols = self
256            .min_cols
257            .map(|m| m.saturating_sub(prev_vertical).max(1));
258        let content_max_cols = self
259            .max_cols
260            .map(|m| m.saturating_sub(prev_vertical).max(1));
261
262        // Apply new border configuration
263        self.show_border = show_border && border_sides.any();
264        self.border_sides = border_sides;
265
266        let new_horizontal =
267            Self::horizontal_border_units_for(self.show_border, &self.border_sides);
268        let new_vertical = Self::vertical_border_units_for(self.show_border, &self.border_sides);
269
270        // Adjust rows/cols (minimum 1)
271        self.rows = Height::new((content_rows + new_horizontal).max(1));
272        self.cols = Width::new((content_cols + new_vertical).max(1));
273
274        // Adjust min/max if set (minimum 1, None stays None)
275        self.min_rows = content_min_rows.map(|m| (m + new_horizontal).max(1));
276        self.max_rows = content_max_rows.map(|m| (m + new_horizontal).max(1));
277        self.min_cols = content_min_cols.map(|m| (m + new_vertical).max(1));
278        self.max_cols = content_max_cols.map(|m| (m + new_vertical).max(1));
279
280        // Enforce constraints on rows/cols
281        if let Some(min_rows) = self.min_rows
282            && self.rows.get() < min_rows
283        {
284            self.rows = Height::new(min_rows);
285        }
286        if let Some(max_rows) = self.max_rows
287            && self.rows.get() > max_rows
288        {
289            self.rows = Height::new(max_rows);
290        }
291        if let Some(min_cols) = self.min_cols
292            && self.cols.get() < min_cols
293        {
294            self.cols = Width::new(min_cols);
295        }
296        if let Some(max_cols) = self.max_cols
297            && self.cols.get() > max_cols
298        {
299            self.cols = Width::new(max_cols);
300        }
301    }
302
303    /// Apply a change to an optional content row (like `show_label` for encumbrance).
304    /// When enabling (false -> true), adds 1 row; when disabling (true -> false), removes 1 row.
305    /// Also adjusts `min_rows/max_rows` proportionally (if set).
306    pub fn apply_optional_content_row(&mut self, new_show: bool, prev_show: bool) {
307        if new_show == prev_show {
308            return; // No change
309        }
310
311        // One row more or less, never under one.
312        let moved = |rows: u16| {
313            if new_show {
314                rows.saturating_add(1)
315            } else {
316                rows.saturating_sub(1).max(1)
317            }
318        };
319
320        // Adjust rows (minimum 1)
321        self.rows = Height::new(moved(self.rows.get()));
322
323        // Adjust min/max if set (minimum 1, None stays None)
324        self.min_rows = self.min_rows.map(moved);
325        self.max_rows = self.max_rows.map(moved);
326
327        // Enforce constraints
328        if let Some(min_rows) = self.min_rows
329            && self.rows.get() < min_rows
330        {
331            self.rows = Height::new(min_rows);
332        }
333        if let Some(max_rows) = self.max_rows
334            && self.rows.get() > max_rows
335        {
336            self.rows = Height::new(max_rows);
337        }
338    }
339}

773/// How a dashboard is laid out, read from its `dashboard_layout` setting.
774#[derive(Debug, Clone, Copy, PartialEq)]
775pub enum DashboardLayout {
776    /// In one row.
777    Horizontal,
778    /// In one column.
779    Vertical,
780    /// In a grid of so many rows and columns.
781    Grid {
782        /// Its rows.
783        rows: usize,
784        /// Its columns.
785        cols: usize,
786    },
787    /// Left to right, wrapping at the edge.
788    Flow,
789}
790
791impl DashboardLayout {
792    /// Parse a `dashboard_layout` string: `horizontal`/`vertical`/`flow`, or
793    /// `grid:RxC` (e.g. `grid:2x3`). Anything unrecognized falls back to
794    /// horizontal.
795    #[must_use]
796    pub fn parse(value: &str) -> Self {
797        let lower = value.to_lowercase();
798        if lower.starts_with("grid")
799            && let Some(spec) = lower.split(':').nth(1)
800        {
801            let parts: Vec<_> = spec.split('x').collect();
802            if parts.len() == 2
803                && let (Ok(r), Ok(c)) = (parts[0].parse::<usize>(), parts[1].parse::<usize>())
804                && r > 0
805                && c > 0
806            {
807                return DashboardLayout::Grid { rows: r, cols: c };
808            }
809        }
810        match lower.as_str() {
811            "vertical" => DashboardLayout::Vertical,
812            "flow" => DashboardLayout::Flow,
813            _ => DashboardLayout::Horizontal,
814        }
815    }
816}

858    /// that shares a `stack`, so a dashboard is as tall as what it holds.
859    #[must_use]
860    pub fn cell_count(&self) -> usize {
861        let mut seen: Vec<String> = Vec::new();
862        let mut count = 0;
863        for def in &self.indicators {
864            if def.stack.is_empty() {
865                count += 1;
866            } else {
867                let key = def.stack.to_lowercase();
868                if !seen.contains(&key) {
869                    seen.push(key);
870                    count += 1;
871                }
872            }
873        }
874        count
875    }

936/// How an effect's time left is written.
937#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, Serialize, Deserialize)]
938#[serde(rename_all = "lowercase")]
939pub enum TimeFormat {
940    /// `[HH:MM]` from an hour, `[MM:SS]` under.
941    #[default]
942    Vellum,
943    /// `[2h39]` from an hour, `[2m39]` under.
944    Units,
945    /// `2:39:12` from an hour, `39:12` under.
946    Clock,
947}
948
949impl TimeFormat {
950    /// `seconds` left, written so.
951    #[must_use]
952    pub fn left(self, seconds: u32) -> String {
953        let (hours, minutes, seconds) = (seconds / 3600, seconds / 60 % 60, seconds % 60);
954        match (self, hours > 0) {
955            (Self::Vellum, true) => format!("[{hours:02}:{minutes:02}]"),
956            (Self::Vellum, false) => format!("[{minutes:02}:{seconds:02}]"),
957            (Self::Units, true) => format!("[{hours}h{minutes:02}]"),
958            (Self::Units, false) => format!("[{minutes}m{seconds:02}]"),
959            (Self::Clock, true) => format!("{hours}:{minutes:02}:{seconds:02}"),
960            (Self::Clock, false) => format!("{minutes}:{seconds:02}"),
961        }
962    }
963}

1474    /// Rows in display order, paired with whether each is shown. Rows missing

1476    /// hides anything -- the checkbox is what hides.

1481            .iter()

1528
1529    /// Raw membership in the merge set -- what the config STORES. The editor
1530    /// reads and writes this; only the renderer applies the positional "first
1531    /// row cannot merge" rule. Conflating the two was a data-loss bug: the
1532    /// view snapshot read the positional answer (false for whatever row was
1533    /// first) and wrote it back on any unrelated edit, deleting the stored
1534    /// flag.
1535    #[must_use]
1536    pub fn row_merge_flag(&self, row: CardRow) -> bool {
1537        self.merged_rows.iter().any(|r| r == row.id())
1538    }
1539
1540    /// Whether this row RENDERS on the line above it: stored flag, unless the
1541    /// row is first (nothing above it to join).
1542    #[must_use]
1543    pub fn row_merged(&self, row: CardRow) -> bool {
1544        if self
1545            .ordered_rows()
1546            .first()
1547            .is_some_and(|(first, _)| *first == row)
1548        {
1549            return false;
1550        }
1551        self.row_merge_flag(row)
1552    }
1553
1554    /// Set or clear a row's merge-with-above flag.
1555    pub fn set_row_merged(&mut self, row: CardRow, merged: bool) {
1556        let present = self.merged_rows.iter().position(|r| r == row.id());
1557        match (merged, present) {
1558            (true, None) => self.merged_rows.push(row.id().to_string()),
1559            (false, Some(idx)) => {
1560                self.merged_rows.remove(idx);
1561            }
1562            _ => {}
1563        }
1564    }
1565
1566    /// Column weights sanitized for layout: at least one column, every
1567    /// weight positive. Garbage in the config degrades to equal columns
1568    /// rather than a zero-width or vanished one.
1569    #[must_use]
1570    pub fn column_weights(&self) -> Vec<f32> {
1571        let mut weights: Vec<f32> = self
1572            .card_column_weights
1573            .iter()
1574            .map(|w| if w.is_finite() && *w > 0.0 { *w } else { 1.0 })
1575            .collect();
1576        if weights.is_empty() {
1577            weights.push(1.0);
1578        }
1579        weights
1580    }
1581
1582    /// Which column a row renders in, clamped to the columns that exist --
1583    /// an assignment to a removed column lands in the last one instead of
1584    /// hiding the row.
1585    #[must_use]
1586    pub fn row_column(&self, row: CardRow) -> usize {
1587        let last = self.column_weights().len() - 1;
1588        self.card_row_columns
1589            .get(row.id())
1590            .copied()
1591            .unwrap_or(0)
1592            .min(last)
1593    }
1594
1595    /// Assign a row to a column. Column 0 is the default, so assigning it
1596    /// removes the entry rather than storing a redundant zero.
1597    pub fn set_row_column(&mut self, row: CardRow, column: usize) {
1598        if column == 0 {
1599            self.card_row_columns.remove(row.id());
1600        } else {
1601            self.card_row_columns.insert(row.id().to_string(), column);
1602        }
1603    }
1604
1605    /// One column's rows grouped into lines: each inner vec is one
1606    /// horizontal run. Order within the column follows `row_order`.
1607    ///
1608    /// Hidden rows are dropped BEFORE grouping, so hiding the row a merged
1609    /// row was attached to promotes it to its own line rather than leaving a
1610    /// dangling continuation. The same applies to rows moved to another
1611    /// column: "share line with the row above" chains only within a column.
1612    #[must_use]
1613    pub fn row_lines(&self, column: usize) -> Vec<Vec<CardRow>> {
1614        let mut lines: Vec<Vec<CardRow>> = Vec::new();
1615        for (row, shown) in self.ordered_rows() {
1616            if !shown || self.row_column(row) != column {
1617                continue;
1618            }
1619            // A merged row joins the line above only when `may_join` allows
1620            // it. This is the data-level guard: a config asking vitals to
1621            // join the RT strip self-heals to its own line instead of
1622            // rendering crushed.
1623            let joinable =
1624                self.row_merged(row) && lines.last().is_some_and(|line| row.may_join(line));
1625            match lines.last_mut() {
1626                Some(line) if joinable => line.push(row),
1627                _ => lines.push(vec![row]),
1628            }
1629        }
1630        lines
1631    }
1632
1633    /// A row moved one place up or down, the whole order written out first
1634    /// so an empty `row_order` becomes the order it stood for.
1635    pub fn move_row(&mut self, row: CardRow, up: bool) {
1636        let mut order: Vec<CardRow> = self.ordered_rows().into_iter().map(|(r, _)| r).collect();
1637        let Some(idx) = order.iter().position(|r| *r == row) else {
1638            return;
1639        };
1640        let target = if up {
1641            idx.checked_sub(1)
1642        } else if idx + 1 < order.len() {
1643            Some(idx + 1)
1644        } else {
1645            None
1646        };
1647        if let Some(target) = target {
1648            order.swap(idx, target);
1649            self.row_order = order.into_iter().map(|r| r.id().to_string()).collect();
1650        }
1651    }
1652}

1654/// Text replacement rule for perception widget
1655#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
1656pub struct TextReplacement {
1657    /// What is found: a regular expression when it holds any of a regular
1658    /// expression's marks, else the plain words.
1659    pub pattern: String,
1660    /// What it is replaced with; nothing removes it.
1661    pub replace: String,
1662}

1664/// Pre-compiled text replacement for runtime use.
1665/// Regex is compiled once at creation, not on every application.
1666#[derive(Debug, Clone)]
1667pub struct CompiledTextReplacement {
1668    /// Original pattern string (for literal matching or error fallback)
1669    pattern: String,
1670    /// Replacement text
1671    replace: String,
1672    /// Pre-compiled regex (None if pattern is literal or invalid regex)
1673    compiled_regex: Option<regex::Regex>,
1674}
1675
1676impl CompiledTextReplacement {
1677    /// Compile a `TextReplacement` into a `CompiledTextReplacement`
1678    #[must_use]
1679    pub fn compile(replacement: &TextReplacement) -> Self {
1680        let pattern = replacement.pattern.as_str();
1681        let is_regex = pattern.contains('\\')
1682            || pattern.contains('^')
1683            || pattern.contains('$')
1684            || pattern.contains('.')
1685            || pattern.contains('*')
1686            || pattern.contains('+')
1687            || pattern.contains('?')
1688            || pattern.contains('(')
1689            || pattern.contains(')')
1690            || pattern.contains('[')
1691            || pattern.contains(']')
1692            || pattern.contains('{')
1693            || pattern.contains('}')
1694            || pattern.contains('|');
1695
1696        let compiled_regex = if is_regex {
1697            // Not a pattern: matched as the words it is.
1698            regex::Regex::new(pattern).ok()
1699        } else {
1700            None
1701        };
1702
1703        Self {
1704            pattern: replacement.pattern.clone(),
1705            replace: replacement.replace.clone(),
1706            compiled_regex,
1707        }
1708    }
1709
1710    /// Apply this replacement to the given text
1711    #[must_use]
1712    pub fn apply(&self, text: &str) -> String {
1713        if let Some(ref re) = self.compiled_regex {
1714            re.replace_all(text, self.replace.as_str()).into_owned()
1715        } else {
1716            text.replace(&self.pattern, &self.replace)
1717        }
1718    }
1719}
1720
1721/// Compile a slice of `TextReplacements` into `CompiledTextReplacements`.
1722/// Call this once at config load or when replacements change.
1723pub fn compile_text_replacements(replacements: &[TextReplacement]) -> Vec<CompiledTextReplacement> {
1724    replacements
1725        .iter()
1726        .map(CompiledTextReplacement::compile)
1727        .collect()
1728}
1729
1730/// Apply pre-compiled text replacements (efficient - no regex compilation).
1731#[must_use]
1732pub fn apply_compiled_text_replacements(
1733    text: &str,
1734    replacements: &[CompiledTextReplacement],
1735) -> String {
1736    let mut result = text.to_string();
1737    for replacement in replacements {
1738        result = replacement.apply(&result);
1739    }
1740    result
1741}

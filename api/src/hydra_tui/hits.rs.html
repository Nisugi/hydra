1//! What a click finds: each frame drawn says where its links, buttons and
2//! tabs are, and a click is looked up among them, the last drawn on top.
3
4use hydra_ui::RunLink;
5use ratatui::layout::Rect;
6
7/// What a click on a place does.
8#[derive(Clone, Debug, PartialEq, Eq)]
9pub enum Target {
10    /// One of the game's links, in a character's text.
11    Link {
12        /// Whose text.
13        seat: u32,
14        /// The link.
15        link: RunLink,
16    },
17    /// One of Hydra's own commands, in a character's text: sent with its
18    /// command symbol.
19    Hydra {
20        /// Whose text.
21        seat: u32,
22        /// The command, without its symbol.
23        command: String,
24    },
25    /// A character: its screen shown.
26    Seat(u32),
27    /// The hub shown.
28    Hub,
29    /// A button of the screen's own, by what it does.
30    Button(String),
31    /// A window of the screen's layout, by its place: the focus given it.
32    Window(usize),
33    /// A tab of a window: its window's place, and its own.
34    Tab(usize, usize),
35}
36
37/// The places a click can find in the frame last drawn.
38#[derive(Debug, Default)]
39pub struct Hits(Vec<(Rect, Target)>);
40
41impl Hits {
42    /// Nothing to find: a frame is about to be drawn.
43    pub fn clear(&mut self) {
44        self.0.clear();
45    }
46
47    /// A click in `rect` finds `target`.
48    pub fn add(&mut self, rect: Rect, target: Target) {
49        self.0.push((rect, target));
50    }
51
52    /// The layout's window at `col`, `row`, under whatever is drawn in it.
53    #[must_use]
54    pub fn window_at(&self, col: u16, row: u16) -> Option<usize> {
55        self.0
56            .iter()
57            .rev()
58            .filter(|(rect, _)| {
59                col >= rect.x
60                    && col < rect.x.saturating_add(rect.width)
61                    && row >= rect.y
62                    && row < rect.y.saturating_add(rect.height)
63            })
64            .find_map(|(_, target)| match target {
65                Target::Window(at) => Some(*at),
66                _ => None,
67            })
68    }
69
70    /// What a click at `col`, `row` finds: the place drawn last there.
71    #[must_use]
72    pub fn at(&self, col: u16, row: u16) -> Option<&Target> {
73        self.0
74            .iter()
75            .rev()
76            .find(|(rect, _)| {
77                col >= rect.x
78                    && col < rect.x.saturating_add(rect.width)
79                    && row >= rect.y
80                    && row < rect.y.saturating_add(rect.height)
81            })
82            .map(|(_, target)| target)
83    }
84}

86#[cfg(test)]
87mod tests {
88    use super::*;
89
90    /// The place drawn last is the one found.
91    #[test]
92    fn the_last_drawn_is_found() {
93        let mut hits = Hits::default();
94        hits.add(Rect::new(0, 0, 10, 10), Target::Hub);
95        hits.add(Rect::new(2, 2, 2, 1), Target::Seat(3));
96        assert_eq!(hits.at(2, 2), Some(&Target::Seat(3)));
97        assert_eq!(hits.at(5, 5), Some(&Target::Hub));
98        assert_eq!(hits.at(10, 0), None);
99        hits.clear();
100        assert_eq!(hits.at(5, 5), None);
101    }
102}

1//! A terminal's key events named as the keys file names keys: by the
2//! physical key, `KeyA`, `Digit1`, `Numpad8`, `ArrowUp`, `F3`, with the
3//! modifiers held, and what the key types besides.
4//!
5//! A terminal reports characters, not keys, unless it speaks the kitty
6//! keyboard protocol: a typed `!` is said to be `Digit1` with Shift, as on
7//! a US keyboard. The numpad is told apart in the Windows console and in
8//! the terminals that speak the kitty protocol; elsewhere it types digits.
9
10use crossterm::event::{KeyCode, KeyEvent, KeyEventKind, KeyEventState, KeyModifiers};
11
12use crate::wire::{Key, Mods};

14/// The physical key that types `c` on a US keyboard, and whether Shift is
15/// held for it; a character no key there types is named by itself.
16#[must_use]
17pub fn us_key(c: char) -> (String, bool) {
18    if c.is_ascii_alphabetic() {
19        return (
20            format!("Key{}", c.to_ascii_uppercase()),
21            c.is_ascii_uppercase(),
22        );
23    }
24    if c.is_ascii_digit() {
25        return (format!("Digit{c}"), false);
26    }
27    let (name, shift) = match c {
28        ' ' => ("Space", false),
29        '-' => ("Minus", false),
30        '_' => ("Minus", true),
31        '=' => ("Equal", false),
32        '+' => ("Equal", true),
33        '[' => ("BracketLeft", false),
34        '{' => ("BracketLeft", true),
35        ']' => ("BracketRight", false),
36        '}' => ("BracketRight", true),
37        '\\' => ("Backslash", false),
38        '|' => ("Backslash", true),
39        ';' => ("Semicolon", false),
40        ':' => ("Semicolon", true),
41        '\'' => ("Quote", false),
42        '"' => ("Quote", true),
43        ',' => ("Comma", false),
44        '<' => ("Comma", true),
45        '.' => ("Period", false),
46        '>' => ("Period", true),
47        '/' => ("Slash", false),
48        '?' => ("Slash", true),
49        '`' => ("Backquote", false),
50        '~' => ("Backquote", true),
51        '!' => ("Digit1", true),
52        '@' => ("Digit2", true),
53        '#' => ("Digit3", true),
54        '$' => ("Digit4", true),
55        '%' => ("Digit5", true),
56        '^' => ("Digit6", true),
57        '&' => ("Digit7", true),
58        '*' => ("Digit8", true),
59        '(' => ("Digit9", true),
60        ')' => ("Digit0", true),
61        other => return (other.to_string(), false),
62    };
63    (name.to_owned(), shift)
64}

66/// The modifiers held, as the wire says them.
67fn mods(modifiers: KeyModifiers) -> u8 {
68    let bit = |held: bool, bit: u8| if held { bit } else { 0 };
69    bit(modifiers.contains(KeyModifiers::CONTROL), Mods::CTRL)
70        | bit(modifiers.contains(KeyModifiers::SHIFT), Mods::SHIFT)
71        | bit(modifiers.contains(KeyModifiers::ALT), Mods::ALT)
72        | bit(
73            modifiers.contains(KeyModifiers::SUPER) || modifiers.contains(KeyModifiers::META),
74            Mods::CMD,
75        )
76}

78/// The numpad's key a code is, by the keys file's name, and what it types.
79fn numpad(code: KeyCode) -> Option<(&'static str, Option<char>)> {
80    Some(match code {
81        KeyCode::Keypad0 => ("Numpad0", Some('0')),
82        KeyCode::Keypad1 => ("Numpad1", Some('1')),
83        KeyCode::Keypad2 => ("Numpad2", Some('2')),
84        KeyCode::Keypad3 => ("Numpad3", Some('3')),
85        KeyCode::Keypad4 => ("Numpad4", Some('4')),
86        KeyCode::Keypad5 | KeyCode::KeypadBegin => ("Numpad5", Some('5')),
87        KeyCode::Keypad6 => ("Numpad6", Some('6')),
88        KeyCode::Keypad7 => ("Numpad7", Some('7')),
89        KeyCode::Keypad8 => ("Numpad8", Some('8')),
90        KeyCode::Keypad9 => ("Numpad9", Some('9')),
91        KeyCode::KeypadMultiply => ("NumpadMultiply", Some('*')),
92        KeyCode::KeypadPlus => ("NumpadAdd", Some('+')),
93        KeyCode::KeypadComma => ("NumpadComma", Some(',')),
94        KeyCode::KeypadMinus => ("NumpadSubtract", Some('-')),
95        KeyCode::KeypadPeriod => ("NumpadDecimal", Some('.')),
96        KeyCode::KeypadDivide => ("NumpadDivide", Some('/')),
97        KeyCode::KeypadEqual => ("NumpadEqual", Some('=')),
98        KeyCode::KeypadEnter => ("NumpadEnter", None),
99        _ => return None,
100    })
101}

103/// The numpad key a character from the numpad is, when the terminal said
104/// it came from there (the kitty protocol's keypad flag).
105fn keypad_char(c: char) -> Option<KeyCode> {
106    Some(match c {
107        '0' => KeyCode::Keypad0,
108        '1' => KeyCode::Keypad1,
109        '2' => KeyCode::Keypad2,
110        '3' => KeyCode::Keypad3,
111        '4' => KeyCode::Keypad4,
112        '5' => KeyCode::Keypad5,
113        '6' => KeyCode::Keypad6,
114        '7' => KeyCode::Keypad7,
115        '8' => KeyCode::Keypad8,
116        '9' => KeyCode::Keypad9,
117        '*' => KeyCode::KeypadMultiply,
118        '+' => KeyCode::KeypadPlus,
119        '-' => KeyCode::KeypadMinus,
120        '.' => KeyCode::KeypadPeriod,
121        '/' => KeyCode::KeypadDivide,
122        '=' => KeyCode::KeypadEqual,
123        ',' => KeyCode::KeypadComma,
124        _ => return None,
125    })
126}

128/// `event` as the wire says a key; `None` for a release, or a key that is
129/// only a modifier or a media key.
130#[must_use]
131pub fn key(event: &KeyEvent) -> Option<Key> {
132    if event.kind == KeyEventKind::Release {
133        return None;
134    }
135    let mut held = mods(event.modifiers);
136    let keypad = event.state.contains(KeyEventState::KEYPAD);
137    let code = match event.code {
138        KeyCode::Char(c) if keypad => keypad_char(c).unwrap_or(event.code),
139        KeyCode::Enter if keypad => KeyCode::KeypadEnter,
140        other => other,
141    };
142    if let Some((name, text)) = numpad(code) {
143        return Some(Key {
144            name: name.to_owned(),
145            text: text.filter(|_| held & (Mods::CTRL | Mods::ALT | Mods::CMD) == 0),
146            mods: Mods(held),
147        });
148    }
149    let named = |name: &str| Some((name.to_owned(), None));
150    let (name, text) = match code {
151        KeyCode::Char(c) => {
152            let (name, shift) = us_key(c);
153            if shift || c.is_ascii_uppercase() {
154                held |= Mods::SHIFT;
155            }
156            let commands = held & (Mods::CTRL | Mods::ALT | Mods::CMD) != 0;
157            (name, (!commands).then_some(c))
158        }
159        KeyCode::Backspace => named("Backspace")?,
160        KeyCode::Enter => named("Enter")?,
161        KeyCode::Left => named("ArrowLeft")?,
162        KeyCode::Right => named("ArrowRight")?,
163        KeyCode::Up => named("ArrowUp")?,
164        KeyCode::Down => named("ArrowDown")?,
165        KeyCode::Home => named("Home")?,
166        KeyCode::End => named("End")?,
167        KeyCode::PageUp => named("PageUp")?,
168        KeyCode::PageDown => named("PageDown")?,
169        KeyCode::Tab => named("Tab")?,
170        KeyCode::BackTab => {
171            held |= Mods::SHIFT;
172            named("Tab")?
173        }
174        KeyCode::Delete => named("Delete")?,
175        KeyCode::Insert => named("Insert")?,
176        KeyCode::F(n) => (format!("F{n}"), None),
177        KeyCode::Esc => named("Escape")?,
178        KeyCode::CapsLock => named("CapsLock")?,
179        KeyCode::ScrollLock => named("ScrollLock")?,
180        KeyCode::NumLock => named("NumLock")?,
181        KeyCode::PrintScreen => named("PrintScreen")?,
182        KeyCode::Pause => named("Pause")?,
183        KeyCode::Menu => named("ContextMenu")?,
184        _ => return None,
185    };
186    Some(Key {
187        name,
188        text,
189        mods: Mods(held),
190    })
191}

193#[cfg(test)]
194mod tests {
195    use super::*;
196
197    fn press(code: KeyCode, modifiers: KeyModifiers) -> KeyEvent {
198        KeyEvent::new(code, modifiers)
199    }
200
201    /// A letter, a mark and a shifted mark are named by the key that types
202    /// them, and type themselves; with Ctrl they type nothing.
203    #[test]
204    fn a_typed_character_is_named_by_its_key() {
205        let a = key(&press(KeyCode::Char('a'), KeyModifiers::NONE)).expect("a key");
206        assert_eq!(
207            (a.name.as_str(), a.text, a.mods),
208            ("KeyA", Some('a'), Mods(0))
209        );
210        let bang = key(&press(KeyCode::Char('!'), KeyModifiers::SHIFT)).expect("a key");
211        assert_eq!(
212            (bang.name.as_str(), bang.text, bang.mods),
213            ("Digit1", Some('!'), Mods(Mods::SHIFT))
214        );
215        let big = key(&press(KeyCode::Char('A'), KeyModifiers::NONE)).expect("a key");
216        assert_eq!(
217            big.mods,
218            Mods(Mods::SHIFT),
219            "an uppercase letter holds Shift"
220        );
221        let ctrl_f = key(&press(KeyCode::Char('f'), KeyModifiers::CONTROL)).expect("a key");
222        assert_eq!(
223            (ctrl_f.name.as_str(), ctrl_f.text, ctrl_f.mods),
224            ("KeyF", None, Mods(Mods::CTRL))
225        );
226    }
227
228    /// The numpad is its own keys, whether the console said so or the kitty
229    /// protocol's keypad flag did; Shift+Tab is Tab with Shift.
230    #[test]
231    fn the_numpad_is_its_own_keys() {
232        let eight = key(&press(KeyCode::Keypad8, KeyModifiers::NONE)).expect("a key");
233        assert_eq!((eight.name.as_str(), eight.text), ("Numpad8", Some('8')));
234        let mut kitty = press(KeyCode::Char('8'), KeyModifiers::NONE);
235        kitty.state = KeyEventState::KEYPAD;
236        assert_eq!(key(&kitty).expect("a key").name, "Numpad8");
237        let mut enter = press(KeyCode::Enter, KeyModifiers::NONE);
238        enter.state = KeyEventState::KEYPAD;
239        assert_eq!(key(&enter).expect("a key").name, "NumpadEnter");
240        let back = key(&press(KeyCode::BackTab, KeyModifiers::SHIFT)).expect("a key");
241        assert_eq!((back.name.as_str(), back.mods), ("Tab", Mods(Mods::SHIFT)));
242        assert_eq!(
243            key(&press(KeyCode::Pause, KeyModifiers::NONE))
244                .expect("a key")
245                .name,
246            "Pause"
247        );
248    }
249
250    /// A release is not a press.
251    #[test]
252    fn a_release_is_no_key() {
253        let mut up = press(KeyCode::Char('a'), KeyModifiers::NONE);
254        up.kind = KeyEventKind::Release;
255        assert!(key(&up).is_none());
256    }

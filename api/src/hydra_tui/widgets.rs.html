1//! Each widget of the catalog, drawn in a terminal's cells: the story and
2//! the streams as text that scrolls; the vitals and the game's other gauges
3//! as bars by eighths of a cell; the clocks counting down; the hands, the

84    pub hydra: bool,
85    /// The character drawn as it is seated -- the window's own, or the one

114        self.seated

158impl State {
159    /// The wheel turned over it: three rows back (`up`) or on.
160    pub fn wheel(&mut self, up: bool) {
161        self.scroll = if up {
162            self.scroll.saturating_sub(3)
163        } else {
164            self.scroll.saturating_add(3)
165        };
166    }

168    /// Words typed into its question since it was last drawn, taken.
169    fn take_typed(&self) -> Option<String> {
170        self.typed
171            .lock()
172            .unwrap_or_else(PoisonError::into_inner)
173            .take()
174    }

285/// What marks a button's piece while its row is wrapped, so it is never
286/// joined to a neighbour drawn alike.
287const MARK: &str = "\u{0}press:";
288
289/// `row` wrapped to `width` columns, each button kept whole to itself.
290pub(crate) fn wrapped(row: &[Bit], width: u16) -> Vec<Row> {
291    let marked: Vec<Piece> = row
292        .iter()
293        .enumerate()
294        .map(|(at, bit)| match &bit.press {
295            Some(_) => Piece {
296                link: Some(RunLink::Url {
297                    href: format!("{MARK}{at}"),
298                }),
299                own: false,
300                ..bit.piece.clone()
301            },
302            None => bit.piece.clone(),
303        })
304        .collect();
305    crate::text::wrap(&marked, width)
306        .into_iter()
307        .map(|line| {
308            line.into_iter()
309                .map(|piece| {
310                    let at = match &piece.link {
311                        Some(RunLink::Url { href }) => href
312                            .strip_prefix(MARK)
313                            .and_then(|at| at.parse::<usize>().ok()),
314                        _ => None,
315                    };
316                    match at.and_then(|at| row.get(at)) {
317                        Some(pressed) => Bit {
318                            piece: Piece {
319                                link: pressed.piece.link.clone(),
320                                own: pressed.piece.own,
321                                ..piece
322                            },
323                            press: pressed.press.clone(),
324                        },
325                        None => bit(piece),
326                    }
327                })
328                .collect()
329        })
330        .collect()
331}

333/// Draw `row` at `x`, `y`, no wider than `width`: its links kept in `hits`
334/// as `seat`'s, its buttons as the widget's own. The columns it took.
335pub(crate) fn draw_bits(
336    buf: &mut Buffer,
337    (x, y, width): (u16, u16, u16),
338    row: &[Bit],
339    hits: &mut Hits,
340    seat: Option<u32>,
341) -> u16 {
342    let mut col = 0u16;
343    for each in row {
344        if col >= width {
345            break;
346        }
347        let took = draw_row(
348            buf,
349            (x + col, y, width - col),
350            std::slice::from_ref(&each.piece),
351            hits,
352            seat,
353        );
354        if let Some(press) = &each.press
355            && took > 0
356        {
357            hits.add(
358                Rect::new(x + col, y, took, 1),
359                Target::Button(format!("{PRESS}{press}")),
360            );
361        }
362        col += took;
363    }
364    col
365}

367/// `rows` drawn from the top of `area`, each wrapped to its width, scrolled
368/// as `state` holds it: how many rows are below what is shown.
369pub(crate) fn rows(
370    buf: &mut Buffer,
371    area: Rect,
372    rows: &[Row],
373    (seen, hits): (&Seen<'_>, &mut Hits),
374    state: &mut State,
375) -> Drawn {
376    let all: Vec<Row> = rows
377        .iter()
378        .flat_map(|row| wrapped(row, area.width))
379        .collect();
380    let height = usize::from(area.height);
381    state.scroll = state.scroll.min(all.len().saturating_sub(height));
382    for (at, row) in all.iter().skip(state.scroll).take(height).enumerate() {
383        let y = area.y + u16::try_from(at).unwrap_or(0);
384        draw_bits(buf, (area.x, y, area.width), row, hits, seen.linked());
385    }
386    Drawn {
387        below: all.len().saturating_sub(state.scroll + height),
388    }
389}

402/// A bar `width` cells wide, `percent` full, in `fill`.
403pub(crate) fn bar_in(width: u16, percent: u32, fill: Color, colors: &Colors) -> Vec<Piece> {
404    let width = u32::from(width);
405    let eighths = (percent.min(100) * width * 8 + 50) / 100;
406    let (whole, part) = (eighths / 8, eighths % 8);
407    let filled = Style::new().fg(fill).bg(colors.color(Token::Inset));
408    let empty = colors.text(Token::Muted).bg(colors.color(Token::Inset));
409    let mut pieces = vec![Piece::new("█".repeat(whole as usize), filled)];
410    let mut used = whole;
411    if part > 0 && used < width {
412        pieces.push(Piece::new(EIGHTHS[part as usize - 1], filled));
413        used += 1;
414    }
415    pieces.push(Piece::new(
416        " ".repeat((width - used.min(width)) as usize),
417        empty,
418    ));
419    pieces
420}

422/// A gauge as a row `width` wide: its label, a bar filling what is left,
423/// and what it says after it -- the numbers, the game's words, the clock.
424pub(crate) fn gauge_row(
425    width: u16,
426    (label, said): (&str, &str),
427    percent: Option<u32>,
428    (fill, colors): (Color, &Colors),
429) -> Row {
430    let label = format!("{label} ");
431    let said = format!(" {said}");
432    let fixed = unicode_width::UnicodeWidthStr::width(label.as_str())
433        + unicode_width::UnicodeWidthStr::width(said.as_str());
434    let room = width.saturating_sub(u16::try_from(fixed).unwrap_or(u16::MAX));
435    let mut row = vec![word(label, tinted(fill, colors))];
436    match percent {
437        Some(percent) => row.extend(
438            bar_in(room.max(3), percent, fill, colors)
439                .into_iter()
440                .map(bit),
441        ),
442        None => row.push(word("·".repeat(usize::from(room.max(3))), colors.muted())),
443    }
444    row.push(word(said, colors.canvas()));
445    row
446}

448/// An effect's row `width` wide, as a bar behind its words: `text` from the
449/// left, cut short to leave room, and `suffix` pinned to the right edge; the
450/// row's background filled from the left by `percent` in `fill`, the rest the
451/// canvas. The words are drawn in the text colour, or black or white where
452/// that would not read against what is behind them.
453pub(crate) fn effect_bar(
454    width: u16,
455    (text, suffix): (&str, &str),
456    percent: u32,
457    (fill, colors): (Color, &Colors),
458) -> Row {
459    let width = usize::from(width);
460    let suffix_len = suffix.chars().count();
461    let shown: String = match width.cmp(&(suffix_len + 1)) {
462        std::cmp::Ordering::Less => suffix.chars().take(width).collect(),
463        std::cmp::Ordering::Equal => suffix.to_owned(),
464        std::cmp::Ordering::Greater => {
465            let room = width - suffix_len - 1;
466            let cut: String = text.chars().take(room).collect();
467            let padding = width - cut.chars().count() - suffix_len;
468            format!("{cut}{}{suffix}", " ".repeat(padding))
469        }
470    };
471    let split = usize::try_from(u64::from(percent.min(100)) * width as u64 / 100).unwrap_or(0);
472    let canvas = colors.color(Token::Canvas);
473    let base = colors.color(Token::Text);
474    let behind = if split > 0 { fill } else { canvas };
475    let filled = Style::new().fg(readable(base, behind)).bg(fill);
476    let empty = Style::new().fg(readable(base, canvas)).bg(canvas);
477    let cells: Vec<char> = shown.chars().collect();
478    let split = split.min(cells.len());
479    let mut row = Vec::new();
480    if split > 0 {
481        row.push(word(cells[..split].iter().collect::<String>(), filled));
482    }
483    if split < cells.len() {
484        row.push(word(cells[split..].iter().collect::<String>(), empty));
485    }
486    row
487}

489/// `text` where it reads against `behind`, by its contrast with it; else
490/// black on a light one, white on a dark.
491fn readable(text: Color, behind: Color) -> Color {
492    let (lit, dark) = {
493        let (one, two) = (luminance(text), luminance(behind));
494        if one > two { (one, two) } else { (two, one) }
495    };
496    if (lit + 0.05) / (dark + 0.05) >= 3.0 {
497        text
498    } else if luminance(behind) > 0.5 {
499        Color::Black
500    } else {
501        Color::White
502    }
503}
504
505/// How light `color` is to the eye, 0 to 1; a colour not given as red,
506/// green and blue counts as mid-grey.
507fn luminance(color: Color) -> f64 {
508    let Color::Rgb(red, green, blue) = color else {
509        return 0.5;
510    };
511    let channel = |value: u8| {
512        let value = f64::from(value) / 255.0;
513        if value <= 0.039_28 {
514            value / 12.92
515        } else {
516            ((value + 0.055) / 1.055).powf(2.4)
517        }
518    };
519    0.2126 * channel(red) + 0.7152 * channel(green) + 0.0722 * channel(blue)
520}

556/// The lines of a text widget -- the story, a stream, Hydra's messages --
557/// each numbered, oldest first; `None` for a widget that is not text.
558#[must_use]
559pub fn text_lines(widget: &Widget, seen: &Seen<'_>) -> Option<Vec<(u64, Line)>> {
560    let colors = seen.colors;
561    match widget {
562        Widget::Story => Some(crate::story::lines(
563            seen.story,
564            colors,
565            (seen.open, seen.hydra),
566            hydra_ui::lines::Lines::default(),
567        )),
568        Widget::Stream(id) => {
569            let kept = seen.story.streams.get(id);
570            Some(kept.map_or_else(Vec::new, |kept| {
571                let first = kept.heard.saturating_sub(kept.lines.len() as u64);
572                kept.lines
573                    .iter()
574                    .enumerate()
575                    .map(|(at, (_, runs))| (first + at as u64, runs_line(runs, colors)))
576                    .collect()
577            }))
578        }
579        Widget::Hydra => {
580            let first = seen.story.told.saturating_sub(seen.story.said.len() as u64);
581            Some(
582                seen.story
583                    .said
584                    .iter()
585                    .enumerate()
586                    .map(|(at, said)| {
587                        (
588                            first + at as u64,
589                            vec![Piece::new(said.clone(), colors.text(Token::Hydra))],
590                        )
591                    })
592                    .collect(),
593            )
594        }
595        _ => None,
596    }
597}

599/// Draw `widget` in `area`, scrolled as `scroll` says when it is text; the
600/// links and buttons it draws kept in `hits`; what it keeps between frames
601/// in `kept`.
602#[allow(
603    clippy::too_many_lines,
604    reason = "the catalog's table: one arm per kind; split, it would hide which kind draws how"
605)]
606pub fn draw(
607    widget: &Widget,
608    buf: &mut Buffer,
609    area: Rect,
610    (seen, hits): (&Seen<'_>, &mut Hits),
611    (scroll, kept): (Scroll, Kept<'_>),
612) -> Drawn {
613    if area.width == 0 || area.height == 0 {
614        return Drawn::default();
615    }
616    let colors = seen.colors;
617    if let Some(lines) = text_lines(widget, seen) {
618        // A followed character's story is never shown: no window mixes two
619        // characters' text.
620        let below = draw_lines(buf, area, &lines, scroll, hits, seen.linked());
621        return Drawn { below };
622    }
623    let Kept { state, shared } = kept;
624    let listed = |rows_of: Vec<Row>, buf: &mut Buffer, hits: &mut Hits, state: &mut State| {
625        rows(buf, area, &rows_of, (seen, hits), state)
626    };
627    match widget {
628        Widget::Health
629        | Widget::Mana
630        | Widget::Stamina
631        | Widget::Spirit
632        | Widget::FieldExperience
633        | Widget::BloodPoints
634        | Widget::Stance
635        | Widget::Encumbrance
636        | Widget::Mind
637        | Widget::NextLevel
638        | Widget::Pulse => character::gauges(widget, buf, area, seen),
639        Widget::Roundtime | Widget::CastTime | Widget::Stun => {
640            character::clocks(widget, buf, area, seen);
641        }
642        Widget::RightHand | Widget::LeftHand | Widget::Prepared => {
643            character::hands(widget, buf, area, (seen, hits));
644        }
645        Widget::Aim => character::aim(buf, area, seen),
646        Widget::Indicator(which) => character::indicator(buf, area, *which, seen),
647        Widget::Statuses => character::statuses(buf, area, seen),
648        Widget::Level
649        | Widget::TrainingPoints
650        | Widget::ConvertedTrainingPoints

763/// What a button pressed asks of the screen.
764#[derive(Debug)]
765pub enum Pressed {
766    /// Nothing more: the widget changed what it shows.
767    Done,
768    /// A line sent as if typed in the command line: Hydra's and Lich's
769    /// commands as well as the game's.
770    Typed(String),
771    /// One of Hydra's own commands, its command symbol put before it.
772    Hydra(String),
773    /// A hotbar's button, by its bar's name and its id: the chain it runs.
774    Hotbar(String, String),
775    /// A line sent to the game without its echo: a quick bar switched, an
776    /// item dragged.
777    Quietly(String),
778    /// Text for the terminal to copy.
779    Copy(String),
780    /// Something to show over the screen.
781    Open(Box<dyn Overlay>),
782}
783
784/// The button `press` (after [`PRESS`]) of the widget keeping `state`
785/// pressed, on `seat`'s screen: what it changed, and what it asks.
786pub fn press(press: &str, state: &mut State, (seat, context): (&Seat, &Context)) -> Pressed {
787    let (verb, rest) = press.split_once(':').unwrap_or((press, ""));
788    match verb {
789        "typed" => return Pressed::Typed(rest.to_owned()),
790        "hotbar" => {
791            if let Some((bar, button)) = rest.split_once(bars::HOTBAR_APART) {
792                return Pressed::Hotbar(bar.to_owned(), button.to_owned());
793            }
794        }
795        "hydra" => return Pressed::Hydra(rest.to_owned()),
796        "quiet" => return Pressed::Quietly(rest.to_owned()),
797        "toggle" => {
798            if !state.open.remove(rest) {
799                state.open.insert(rest.to_owned());
800            }
801        }
802        "page" => {
803            state.page = rest.parse().unwrap_or(0);
804            state.scroll = 0;
805        }
806        "zoom" => state.zoom = minimap::zoomed(state.zoom, rest == "in"),
807        "ask" => {
808            return Pressed::Open(Box::new(ask::Question::new(rest, Arc::clone(&state.typed))));
809        }
810        "say" => {
811            *state.typed.lock().unwrap_or_else(PoisonError::into_inner) = Some(rest.to_owned());
812        }
813        "copy" => {
814            let found = rest
815                .parse::<u32>()
816                .ok()
817                .and_then(|id| crate::seat::find(&context.sessions.seats(), id).cloned());
818            if let Some(snapshot) = found.and_then(|seat| seat.snapshot()) {
819                return Pressed::Copy(format!("{:#?}", snapshot.state));
820            }
821        }
822        containers::VERB => return containers::press(rest, state, (seat, &context.data)),
823        verb if inventory::VERBS.contains(&verb) => return inventory::press(verb, rest, state),
824        verb if trainer::VERBS.contains(&verb) => {
825            trainer::press(verb, rest, state, (seat, context));
826        }
827        _ => {}
828    }
829    Pressed::Done
830}

972    /// Row `y` of `buf`'s text, trailing spaces off.
973    pub(crate) fn row_text(buf: &Buffer, y: u16) -> String {
974        let area = buf.area;
975        (area.x..area.x + area.width)
976            .map(|x| buf[(x, y)].symbol().to_owned())
977            .collect::<String>()
978            .trim_end()
979            .to_owned()
980    }
981
982    /// Every row of `buf`'s text, trailing spaces off.
983    pub(crate) fn text_of(buf: &Buffer) -> Vec<String> {
984        let area = buf.area;
985        (area.y..area.y + area.height)
986            .map(|y| row_text(buf, y))
987            .collect()
988    }

990    /// `widget` drawn for `seen` in an area `width` by `height`: its text,
991    /// and its clicks.
992    pub(crate) fn drawn(
993        widget: &Widget,
994        seen: &Seen<'_>,
995        (width, height): (u16, u16),
996    ) -> (Vec<String>, Hits) {
997        let area = Rect::new(0, 0, width, height);
998        let mut buf = Buffer::empty(area);
999        let mut hits = Hits::default();
1000        let mut states = States::default();

1002            widget,
1003            &mut buf,
1004            area,
1005            (seen, &mut hits),
1006            (Scroll::default(), states.of((0, 0))),
1007        );
1008        (text_of(&buf), hits)
1009    }

1//! A hub card, in a terminal: the design every character's card is drawn
2//! from (`hydra_board::card`, the windows' cards editor's), its cells
3//! scaled to the terminal's.
4//!
5//! The design is laid out in points; a terminal has cells. A column is
6//! [`COL`] points wide and a row [`ROW`] points tall, a line of the
7//! windows' text, so a card keeps its shape; no card is smaller than
8//! [`NARROWEST`] by [`SHORTEST`]. A terminal's letters are wider than the
9//! windows', so each part is given at least the columns its words take,
10//! and parts beside one another keep their order and a cell between them,
11//! those after pushed along and the card widened ([`layout`]). Each cell's widget draws as it does on a
12//! character's screen, from the character's state and story; the card's
13//! own parts are the hub's: the name and how it is connected, where it is
14//! and who with, and its buttons. A Field says one value of the game state,
15//! read from a print of it kept a short while for every field of the card.
16//! The four vitals say their label and percent over their bar: a card is
17//! small.

40/// The points of the design a column takes.
41pub const COL: f32 = 8.0;
42
43/// The points of the design a row takes: a line of the windows' text.
44pub const ROW: f32 = hydra_ui::widget::LINE;

46/// The fewest columns a card's inside has.
47pub const NARROWEST: u16 = 24;
48
49/// The fewest rows a card's inside has.
50pub const SHORTEST: u16 = 4;

52/// `points` of the design as cells, `per` points to a cell.
53fn cells(points: f32, per: f32) -> u16 {
54    #[expect(
55        clippy::cast_possible_truncation,
56        clippy::cast_sign_loss,
57        reason = "a design's points, kept to what a terminal holds before the cast"
58    )]
59    let cells = (points.clamp(0.0, 60_000.0) / per).round() as u16;
60    cells
61}

69/// The columns a part showing `widget` takes at the least: what its words
70/// take, its label and the most its numbers say; none for one that takes
71/// what the design gives it.
72fn needs(widget: &Widget) -> u16 {
73    let width = |text: &str| u16::try_from(text.chars().count()).unwrap_or(u16::MAX);
74    match widget {
75        // A name, its nickname and how it is connected, said at most so.
76        Widget::Card(CardPart::Name) => 30,
77        Widget::Card(CardPart::Character) => 16,
78        Widget::Card(CardPart::Connection) => width("Reconnecting"),
79        Widget::Card(CardPart::Buttons) => [false, true]
80            .into_iter()
81            .map(|closed| {
82                Button::shown(closed)
83                    .iter()
84                    .map(|button| width(button.label()) + 3)
85                    .sum::<u16>()
86                    .saturating_sub(1)
87            })
88            .max()
89            .unwrap_or(0),
90        Widget::Health => width("HP 100%"),
91        Widget::Mana => width("MP 100%"),
92        Widget::Stamina => width("SP 100%"),
93        Widget::Spirit => width("Sp 100%"),
94        Widget::FieldExperience => width("Field 100%"),
95        Widget::BloodPoints => width("Blood 100%"),
96        Widget::Indicator(which) => width(which.name()) + 2,
97        _ => 0,
98    }
99}

101/// Each of `design`'s cells' place, from the card's inside's top left --
102/// none for a cell that shows nothing -- and the card's inside, in columns
103/// and rows: each cell as scaled, at least as wide as its words
104/// (`needs`), and pushed along past those before it on its rows, a cell
105/// between them at the least; the inside the design's, as wide as its
106/// cells reach, never smaller than [`NARROWEST`] by [`SHORTEST`].
107#[must_use]
108pub fn layout(design: &Custom) -> (Vec<Option<Rect>>, (u16, u16)) {
109    let mut places: Vec<Option<Rect>> = design
110        .cells
111        .iter()
112        .map(|cell| {
113            let placed = cell.shown()?;
114            let mut at = scaled(cell.rect);
115            at.width = at.width.max(needs(&placed.widget));
116            Some(at)
117        })
118        .collect();
119    let arranged: Vec<Option<Rect>> = design
120        .cells
121        .iter()
122        .map(|cell| cell.shown().map(|_| scaled(cell.rect)))
123        .collect();
124    let mut order: Vec<usize> = (0..places.len())
125        .filter(|at| places[*at].is_some())
126        .collect();
127    order.sort_by_key(|at| arranged[*at].map_or(0, |rect| rect.x));
128    let beside = |a: Rect, b: Rect| a.y < b.bottom() && b.y < a.bottom();
129    for (done, at) in order.iter().enumerate() {
130        let (Some(was), Some(mut now)) = (arranged[*at], places[*at]) else {
131            continue;
132        };
133        let after = order[..done]
134            .iter()
135            .filter_map(|before| Some((arranged[*before]?, places[*before]?)))
136            .filter(|(their_was, _)| their_was.right() <= was.x && beside(*their_was, was))
137            .map(|(their_was, their_now)| their_now.right() + (was.x - their_was.right()).max(1))
138            .max();
139        if let Some(after) = after {
140            now.x = now.x.max(after);
141        }
142        places[*at] = Some(now);
143    }
144    let [width, height] = design.inside;
145    let reach = places
146        .iter()
147        .flatten()
148        .map(|place| place.right())
149        .max()
150        .unwrap_or(0);
151    let size = (
152        cells(width, COL).max(reach).max(NARROWEST),
153        cells(height, ROW).max(SHORTEST),
154    );
155    (places, size)
156}

158/// A cell's rect of the design -- x, y, width, height in points -- as
159/// cells from the card's inside's top left: its edges each put on the
160/// nearest cell, a column and a row at the least.
161#[must_use]
162pub fn scaled([x, y, width, height]: [f32; 4]) -> Rect {
163    let (left, top) = (cells(x, COL), cells(y, ROW));
164    let (right, bottom) = (cells(x + width, COL), cells(y + height, ROW));
165    Rect::new(
166        left,
167        top,
168        right.saturating_sub(left).max(1),
169        bottom.saturating_sub(top).max(1),
170    )
171}

173/// One of a card's buttons.
174#[derive(Clone, Copy, Debug, PartialEq, Eq)]
175pub enum Button {
176    /// Show the character's screen.
177    Window,
178    /// Open the character in a terminal of its own.
179    Terminal,
180    /// The character's log.
181    Log,
182    /// Quit the character, once the player has said yes.
183    Quit,
184    /// Log a closed character back in.
185    Reconnect,
186    /// Take a closed character off the hub.
187    Remove,
188}
189
190impl Button {
191    /// Every button, in no order a card keeps.
192    const ALL: [Self; 6] = [
193        Self::Window,
194        Self::Terminal,
195        Self::Log,
196        Self::Quit,
197        Self::Reconnect,
198        Self::Remove,
199    ];
200
201    /// The buttons a card shows: its character's own while it plays, those
202    /// for one that has closed once it has.
203    #[must_use]
204    pub const fn shown(closed: bool) -> &'static [Self] {
205        if closed {
206            &[Self::Reconnect, Self::Remove, Self::Log]
207        } else {
208            &[Self::Window, Self::Log, Self::Quit]
209        }
210    }
211
212    /// What a click on it is called, among the hub's.
213    #[must_use]
214    pub const fn verb(self) -> &'static str {
215        match self {
216            Self::Window => "window",
217            Self::Terminal => "terminal",
218            Self::Log => "log",
219            Self::Quit => "quit",
220            Self::Reconnect => "reconnect",
221            Self::Remove => "remove",
222        }
223    }
224
225    /// The button `verb` calls.
226    #[must_use]
227    pub fn of(verb: &str) -> Option<Self> {
228        Self::ALL.into_iter().find(|button| button.verb() == verb)
229    }
230
231    /// What it says.
232    #[must_use]
233    pub const fn label(self) -> &'static str {
234        match self {
235            Self::Window => "Window",
236            Self::Terminal => "Terminal",
237            Self::Log => "Log",
238            Self::Quit => "Quit",
239            Self::Reconnect => "Reconnect",
240            Self::Remove => "Remove",
241        }
242    }
243
244    /// What a click on it finds, on seat `seat`'s card.
245    #[must_use]
246    pub fn target(self, seat: u32) -> Target {
247        Target::Button(format!("hub:{}:{seat}", self.verb()))
248    }
249}

269/// Draw `seat`'s card from `design` in `inside`, the card's inside -- its
270/// nickname on the roster, when it has one, after its name; its files in
271/// the data folder `data`: each cell's widget in its scaled place, cut to
272/// the card; the buttons' and the links' places kept in `hits`.
273pub fn draw(
274    buf: &mut Buffer,
275    inside: Rect,
276    (seat, design, nickname, data): (&Seat, &Custom, Option<&str>, &Path),
277    (colors, hits, kept, now): (&Colors, &mut Hits, &mut Kept, Instant),
278) {
279    let snapshot = seat.snapshot();
280    let state = snapshot.as_deref().map(|snapshot| &snapshot.state);
281    let (spots, _) = layout(design);
282    for (cell, at) in design.cells.iter().zip(spots) {
283        let (Some(placed), Some(at)) = (cell.shown(), at) else {
284            continue;
285        };
286        let area = Rect::new(
287            inside.x.saturating_add(at.x),
288            inside.y.saturating_add(at.y),
289            at.width,
290            at.height,
291        )
292        .intersection(inside);
293        if area.is_empty() {
294            continue;
295        }
296        match &placed.widget {
297            Widget::Card(part) => own(buf, area, *part, (seat, nickname), (colors, hits)),
298            Widget::Field(path) => {
299                let said = state.map_or_else(
300                    || "—".to_owned(),
301                    |state| {
302                        let print = kept.prints.print(seat.id, state, now);
303                        printed::value(printed::top(&print.tree), path)
304                            .unwrap_or_else(|| "not found".to_owned())
305                    },
306                );
307                line(buf, area, &[Piece::new(said, colors.canvas())]);
308            }
309            widget @ (Widget::Health
310            | Widget::Mana
311            | Widget::Stamina
312            | Widget::Spirit
313            | Widget::FieldExperience
314            | Widget::BloodPoints) => {
315                let (label, amount) = match widget {
316                    Widget::Health => ("HP", state.and_then(GameState::health)),
317                    Widget::Mana => ("MP", state.and_then(GameState::mana)),
318                    Widget::Stamina => ("SP", state.and_then(GameState::stamina)),
319                    Widget::FieldExperience => {
320                        ("Field", state.and_then(hydra_ui::sheet::field_experience))
321                    }
322                    Widget::BloodPoints => ("Blood", state.and_then(hydra_ui::sheet::blood_points)),
323                    _ => ("Sp", state.and_then(GameState::spirit)),
324                };
325                let token = widget.bar_token().unwrap_or(Token::Accent);
326                vital(buf, area, (label, amount.as_ref(), token), colors);
327            }
328            widget => {
329                let story = seat.story();
330                let seen = Seen {
331                    seat: seat.id,
332                    own: true,
333                    who: None,
334                    story: &story,
335                    snapshot: snapshot.as_deref(),
336                    colors,
337                    open: &[],
338                    hydra: false,
339                    seated: Some((seat, data)),
340                    times: crate::layout::vellum_schema::TimeFormat::default(),
341                };
342                // Each card's widget keeps its own, by its seat and its id.
343                let keyed = (
344                    usize::try_from(seat.id).unwrap_or(usize::MAX),
345                    usize::try_from(placed.id).unwrap_or(usize::MAX),
346                );
347                crate::widgets::draw(
348                    widget,
349                    buf,
350                    area,
351                    (&seen, hits),
352                    (Scroll::default(), kept.widgets.of(keyed)),
353                );
354            }
355        }
356    }
357}

370/// A vital on a card: its label and percent over a bar filled as far.
371fn vital(
372    buf: &mut Buffer,
373    area: Rect,
374    (label, amount, token): (&str, Option<&Vital>, Token),
375    colors: &Colors,
376) {
377    let percent = amount.map(|vital| vital.percent.min(100));
378    let said = percent.map_or_else(
379        || format!("{label} ?"),
380        |percent| format!("{label} {percent}%"),
381    );
382    let width = usize::from(area.width);
383    let text: Vec<char> = format!("{said:^width$}").chars().collect();
384    let filled = percent.map_or(0, |percent| (u32::from(area.width) * percent + 50) / 100);
385    let full = Style::new()
386        .fg(colors.color(Token::Canvas))
387        .bg(colors.color(token));
388    let empty = colors.text(token).bg(colors.color(Token::Inset));
389    for col in 0..area.width {
390        let style = if u32::from(col) < filled { full } else { empty };
391        let c = text.get(usize::from(col)).copied().unwrap_or(' ');
392        if let Some(cell) = buf.cell_mut((area.x + col, area.y)) {
393            cell.set_char(c).set_style(style);
394        }
395    }
396}

398/// One of the card's own parts, drawn from `seat`.
399fn own(
400    buf: &mut Buffer,
401    area: Rect,
402    part: CardPart,
403    (seat, nickname): (&Seat, Option<&str>),
404    (colors, hits): (&Colors, &mut Hits),
405) {
406    let lifecycle = seat.lifecycle();
407    match part {
408        CardPart::Name => {
409            let (said, token) = hydra_board::connection(lifecycle);
410            let mut pieces = vec![Piece::new(
411                seat.name.clone(),
412                colors.text(Token::Accent).add_modifier(Modifier::BOLD),
413            )];
414            if let Some(nickname) = nickname.filter(|nickname| !nickname.is_empty()) {
415                pieces.push(Piece::new(format!(" ({nickname})"), colors.muted()));
416            }
417            pieces.push(Piece::new(format!(" {said}"), colors.text(token)));
418            line(buf, area, &pieces);
419        }
420        CardPart::Character => {
421            let mut pieces = vec![Piece::new(
422                seat.name.clone(),
423                colors.text(Token::Accent).add_modifier(Modifier::BOLD),
424            )];
425            if let Some(nickname) = nickname.filter(|nickname| !nickname.is_empty()) {
426                pieces.push(Piece::new(format!(" ({nickname})"), colors.muted()));
427            }
428            line(buf, area, &pieces);
429        }
430        CardPart::Connection => {
431            let (said, token) = hydra_board::connection(lifecycle);
432            line(
433                buf,
434                area,
435                &[Piece::new(said.to_owned(), colors.text(token))],
436            );
437        }
438        CardPart::Whereabouts => {
439            let piece = if lifecycle == State::Closed {
440                // Why it closed: Hydra's last word on it, a refused login, say.
441                let why = seat.story().said.back().cloned().unwrap_or_default();
442                Piece::new(why, colors.text(Token::Wrong))
443            } else {
444                let snapshot = seat.snapshot();
445                let state = snapshot.as_deref().map(|snapshot| &snapshot.state);
446                Piece::new(whereabouts(state), colors.canvas())
447            };
448            line(buf, area, &[piece]);
449        }
450        CardPart::Buttons => {
451            let closed = lifecycle == State::Closed;
452            let mut x = area.x;
453            for button in Button::shown(closed) {
454                let width = area.width.saturating_sub(x.saturating_sub(area.x));
455                if width == 0 {
456                    break;
457                }
458                let took = draw_row(
459                    buf,
460                    (x, area.y, width),
461                    &[Piece::new(
462                        format!("[{}]", button.label()),
463                        colors.text(Token::Link),
464                    )],
465                    &mut Hits::default(),
466                    None,
467                );
468                hits.add(Rect::new(x, area.y, took, 1), button.target(seat.id));
469                x = x.saturating_add(took + 1);
470            }
471        }
472    }
473}

475/// Where a character is and who with: its round time, the room, whom it
476/// leads or follows.
477fn whereabouts(state: Option<&GameState>) -> String {
478    let mut facts = Vec::new();
479    if let Some(seconds) = state
480        .and_then(GameState::roundtime_remaining)
481        .filter(|seconds| *seconds > 0)
482    {
483        facts.push(format!("RT {seconds}s"));
484    }
485    facts.push(
486        state
487            .and_then(|state| state.room.title.clone())
488            .unwrap_or_else(|| "Room unknown".to_owned()),
489    );
490    if let Some(group) = state.and_then(group) {
491        facts.push(group);
492    }
493    facts.join(" · ")
494}

496/// Whom a character leads or follows; nothing alone, or while nobody has
497/// said.
498fn group(state: &GameState) -> Option<String> {
499    let group = &state.group;
500    match group.leader() {
501        Leader::Unknown => None,
502        Leader::You if group.is_empty() => None,
503        Leader::You => Some(format!(
504            "leading {}",
505            group
506                .members()
507                .iter()
508                .map(|member| member.noun.as_str())
509                .collect::<Vec<_>>()
510                .join(", ")
511        )),
512        Leader::Other(leader) => Some(format!("following {}", leader.noun)),
513    }
514}

517pub const STAGES_SIZE: (u16, u16) = (42, 12);

519/// How quick `seat`'s terminal and session are, drawn in `area` under its
520/// card while *Performance* is on: each stage's median and slow end over its
521/// last samples, coloured by how they read, how many there are, and how
522/// often the screen fell behind. The stages a screen's drawing takes are a
523/// terminal's, apart from the windows'.
524pub fn stages(buf: &mut Buffer, area: Rect, seat: &Seat, colors: &Colors) {
525    use hydra_ui::perf::{Grade, took};
526    let (perf, terminal, lichs_text) = {
527        let story = seat.story();
528        let lichs_text = story.lich_running() && story.view() == hydra_session::lich::View::Lich;
529        (story.perf.clone(), story.terminal.clone(), lichs_text)
530    };
531    let graded = |grade: Grade| {
532        colors.text(match grade {
533            Grade::Quick => Token::Stamina,
534            Grade::Noticeable => Token::Warning,
535            Grade::Slow => Token::Wrong,
536        })
537    };
538    let mut lines: Vec<Vec<Piece>> = vec![vec![Piece::new(
539        format!("{:<17}{:>9}{:>9}{:>6}", "Stage", "median", "95%", "of"),
540        colors.muted(),
541    )]];
542    for (stage, samples) in perf.terminal_stages(lichs_text, &terminal) {
543        let mut line = vec![Piece::new(format!("{:<17}", stage.name), colors.canvas())];
544        match samples.spread() {
545            Some((median, slow)) => {
546                for time in [median, slow] {
547                    line.push(Piece::new(
548                        format!("{:>9}", took(time)),
549                        graded(stage.grade(time)),
550                    ));
551                }
552            }
553            None => line.push(Piece::new(format!("{:>9}{:>9}", "—", "—"), colors.muted())),
554        }
555        line.push(Piece::new(format!("{:>6}", samples.len()), colors.muted()));
556        lines.push(line);
557    }
558    lines.push(vec![
559        Piece::new("quick", graded(Grade::Quick)),
560        Piece::new(" · ", colors.muted()),
561        Piece::new("noticeable", graded(Grade::Noticeable)),
562        Piece::new(" · ", colors.muted()),
563        Piece::new("slow", graded(Grade::Slow)),
564    ]);
565    if perf.gaps > 0 {
566        lines.push(vec![Piece::new(
567            format!("Fell behind {} times", perf.gaps),
568            colors.text(Token::Wrong),
569        )]);
570    }
571    for (row, line) in lines.iter().enumerate().take(usize::from(area.height)) {
572        let y = area.y + u16::try_from(row).unwrap_or(0);
573        draw_row(
574            buf,
575            (area.x, y, area.width),
576            line,
577            &mut Hits::default(),
578            None,
579        );
580    }
581}

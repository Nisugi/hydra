1//! The hub, in a terminal: every character Hydra runs, at a glance, in four
2//! tabs -- *Live*, the characters playing, each a card drawn from the card
3//! design (`card.rs`), the shared streams' chat below them; *Offline*,
4//! those that ended this run, then the roster's characters not playing, to
5//! start; *Teams*, roster characters started together; and *New login*, a
6//! login by account.
7//!
8//! Tab and Shift+Tab, or 1 to 4, go between the tabs, as does a click on a
9//! tab's name; the line under the tabs says the keys of the one shown. A
10//! password typed here goes to the binary in a [`Password`], which never
11//! prints, and leaves the hub once asked for; the hub is printed by its
12//! `Debug`, and never shows one.

43/// How long the card design is used before its file is read again: the
44/// windows' cards editor may have changed it.
45const DESIGN_KEPT: Duration = Duration::from_secs(3);

47/// Which of the hub's tabs is shown.
48#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
49pub enum Tab {
50    /// The characters playing.
51    #[default]
52    Live,
53    /// Those that ended this run, then the roster's not playing.
54    Offline,
55    /// The teams kept.
56    Teams,
57    /// A login by account.
58    NewLogin,
59}
60
61impl Tab {
62    /// Every tab, in the order shown.
63    pub const ALL: [Self; 4] = [Self::Live, Self::Offline, Self::Teams, Self::NewLogin];
64
65    /// Its place among the tabs, from 0.
66    #[must_use]
67    pub const fn at(self) -> usize {
68        match self {
69            Self::Live => 0,
70            Self::Offline => 1,
71            Self::Teams => 2,
72            Self::NewLogin => 3,
73        }
74    }
75
76    /// The tab after it (`forward`) or before it, round.
77    #[must_use]
78    pub const fn step(self, forward: bool) -> Self {
79        let count = Self::ALL.len();
80        let at = if forward {
81            (self.at() + 1) % count
82        } else {
83            (self.at() + count - 1) % count
84        };
85        Self::ALL[at]
86    }
87}

132/// The nickname the roster gives `seat`'s character, when it gives one.
133fn nickname<'g>(glance: &'g Glance, seat: &hydra_board::Seat) -> Option<&'g str> {
134    glance

137        .find(|card| {

162#[derive(Debug, Default)]

165    pub tab: Tab,
166    /// The card or row chosen on each tab.
167    chosen: [usize; 4],
168    /// How many cards the last drawn grid held to a row.
169    per_row: usize,
170    /// The Live tab's chat is hidden.
171    chat_hidden: bool,
172    /// The card design, and when it was read.
173    design: Option<(Instant, Custom)>,
174    /// What the cards keep between frames.
175    kept: Kept,
176    /// What the Offline tab is typing.
177    offline: offline::Offline,
178    /// The New login tab's form.
179    form: login::Form,

182impl HubScreen {
183    /// Show `tab`.
184    fn switch(&mut self, tab: Tab) {
185        self.tab = tab;
186        self.offline.stop();
187    }

189    /// `key` pressed on the hub, the board as `glance` has it.
190    pub fn key(&mut self, key: &Key, glance: &Glance) -> Option<Asked> {
191        if self.tab == Tab::Offline && self.offline.editing() {
192            return self.offline.edit_key(key, glance);
193        }
194        if key.name == "Tab" && !key.mods.commands() {
195            self.switch(self.tab.step(!key.mods.has(Mods::SHIFT)));
196            return None;
197        }
198        let at = self.tab.at();
199        if self.tab == Tab::NewLogin && self.form.typing(key, self.chosen[at], glance) {
200            return None;
201        }
202        if key.name.starts_with("Digit")
203            && let Some(digit) = letter(key).and_then(|c| c.to_digit(10))
204            && let Some(tab) = usize::try_from(digit)
205                .ok()
206                .and_then(|digit| Tab::ALL.get(digit.wrapping_sub(1)))
207        {
208            self.switch(*tab);
209            return None;
210        }
211        let chosen = &mut self.chosen[at];
212        match self.tab {
213            Tab::Live => live::key(key, (chosen, self.per_row), glance, &mut self.chat_hidden),
214            Tab::Offline => self.offline.key(key, (chosen, self.per_row), glance),
215            Tab::Teams => teams::key(key, chosen, glance),
216            Tab::NewLogin => self.form.key(key, chosen, glance),
217        }
218    }

220    /// Text pasted on the hub: into the field being typed in.
221    pub fn paste(&mut self, text: &str, glance: &Glance) {
222        match self.tab {
223            Tab::Offline => self.offline.paste(text),
224            Tab::NewLogin => self
225                .form
226                .paste(text, self.chosen[Tab::NewLogin.at()], glance),
227            Tab::Live | Tab::Teams => {}
228        }
229    }

231    /// A click on one of the hub's own places, `button` as drawn
232    /// (`hub:<verb>:<what>`).
233    pub fn click(&mut self, button: &str, glance: &Glance) -> Option<Asked> {
234        let (verb, what) = button.strip_prefix("hub:")?.split_once(':')?;
235        if verb == "tab" {
236            let tab = what.parse::<usize>().ok().and_then(|at| Tab::ALL.get(at))?;
237            self.switch(*tab);
238            return None;
239        }
240        if let Some(button) = Button::of(verb) {
241            let id = what.parse::<u32>().ok()?;
242            let seat = glance.seats.iter().find(|seat| seat.id == id)?;
243            return Some(match button {
244                Button::Window => Asked::Show(id),
245                Button::Terminal => Asked::Terminal(id),
246                Button::Log => Asked::Log(id),
247                Button::Quit => quit(&seat.name, id),
248                Button::Reconnect => Asked::Ask(HubRequest::Reconnect(id)),
249                Button::Remove => Asked::Ask(HubRequest::Remove(id)),
250            });
251        }
252        if verb == "card" {
253            let id = what.parse::<u32>().ok()?;
254            return Some(Asked::Show(id));
255        }
256        let at = self.tab.at();
257        match self.tab {
258            Tab::Live | Tab::Offline => {
259                self.offline
260                    .click((verb, what), &mut self.chosen[Tab::Offline.at()], glance)
261            }
262            Tab::Teams => teams::click((verb, what), &mut self.chosen[at], glance),
263            Tab::NewLogin => self.form.click((verb, what), &mut self.chosen[at], glance),
264        }
265    }

267    /// The card design, read again once it is a few seconds old.
268    fn design(&mut self, context: &Context, now: Instant) -> Custom {
269        let stale = self
270            .design
271            .as_ref()
272            .is_none_or(|(at, _)| now.saturating_duration_since(*at) >= DESIGN_KEPT);
273        if stale {
274            self.design = Some((now, hydra_board::card::load(&context.data)));
275        }
276        self.design
277            .as_ref()
278            .map_or_else(hydra_board::card::hydras, |(_, design)| design.clone())
279    }

281    /// Draw the hub in `area`; where the cursor goes, while a field is
282    /// typed in.
283    pub fn draw(
284        &mut self,
285        buf: &mut Buffer,
286        area: Rect,
287        (glance, context): (&Glance, &Context),
288        (colors, hits): (&Colors, &mut Hits),
289    ) -> Option<(u16, u16)> {
290        if area.height < 3 || area.width < 4 {
291            return None;
292        }
293        let now = Instant::now();
294        self.kept.keep_only(&glance.seats);
295        self.header(buf, area, glance, (colors, hits));
296        let body = Rect::new(area.x, area.y + 2, area.width, area.height - 2);
297        let at = self.tab.at();
298        match self.tab {
299            Tab::Live => {
300                let design = self.design(context, now);
301                let chat = (!self.chat_hidden).then(|| context.sessions.merged());
302                self.per_row = live::draw(
303                    buf,
304                    body,
305                    (
306                        glance,
307                        &design,
308                        chat.as_ref(),
309                        &context.data,
310                        context.sessions.perf(),
311                    ),
312                    &mut self.chosen[at],
313                    (colors, hits, &mut self.kept, now),
314                );
315                None
316            }
317            Tab::Offline => {
318                let design = self.design(context, now);
319                let (per_row, cursor) = self.offline.draw(
320                    buf,
321                    body,
322                    (glance, &design, &context.data),
323                    &mut self.chosen[at],
324                    (colors, hits, &mut self.kept, now),
325                );
326                self.per_row = per_row;
327                cursor
328            }
329            Tab::Teams => {
330                teams::draw(buf, body, glance, &mut self.chosen[at], (colors, hits));
331                None
332            }
333            Tab::NewLogin => {
334                self.form
335                    .draw(buf, body, glance, &mut self.chosen[at], (colors, hits))
336            }
337        }
338    }

340    /// The tabs, what the binary said and the web page's address; under
341    /// them, the keys of the tab shown.
342    fn header(
343        &self,
344        buf: &mut Buffer,
345        area: Rect,
346        glance: &Glance,
347        (colors, hits): (&Colors, &mut Hits),
348    ) {
349        let surface = |token: Token| colors.text(token).bg(colors.color(Token::Surface));
350        let top = Rect::new(area.x, area.y, area.width, 1);
351        buf.set_style(top, surface(Token::Text));
352        let live = glance
353            .seats
354            .iter()
355            .filter(|seat| seat.lifecycle() != State::Closed)
356            .count();
357        let names = [
358            format!("Live ({live})"),
359            format!("Offline ({})", offline::count(glance)),
360            format!("Teams ({})", glance.teams.len()),
361            "New login".to_owned(),
362        ];
363        let mut x = area.x;
364        let end = area.x + area.width;
365        let mut put = |buf: &mut Buffer, hits: &mut Hits, piece: Piece, target: Option<Target>| {
366            let took = draw_row(
367                buf,
368                (x, area.y, end.saturating_sub(x)),
369                &[piece],
370                &mut Hits::default(),
371                None,
372            );
373            if let Some(target) = target {
374                hits.add(Rect::new(x, area.y, took, 1), target);
375            }
376            x = x.saturating_add(took);
377        };
378        put(
379            buf,
380            hits,
381            Piece::new(
382                " Hydra ",
383                surface(Token::Accent).add_modifier(Modifier::BOLD),
384            ),
385            None,
386        );
387        for (tab, name) in Tab::ALL.into_iter().zip(names) {
388            let style = if tab == self.tab {
389                colors.chosen().add_modifier(Modifier::BOLD)
390            } else {
391                surface(Token::Muted)
392            };
393            put(
394                buf,
395                hits,
396                Piece::new(format!(" {name} "), style),
397                Some(Target::Button(format!("hub:tab:{}", tab.at()))),
398            );
399        }
400        let web = match &glance.web_at {
401            Some(WebAt::Listening(address)) => format!(" Web page: {address} "),
402            Some(WebAt::Failed(_)) | None => String::new(),
403        };
404        let web_width = u16::try_from(web.chars().count()).unwrap_or(u16::MAX);
405        let web_x = end.saturating_sub(web_width);
406        if let Some(said) = &glance.said {
407            let room = web_x.saturating_sub(x + 1);
408            if room > 2 {
409                draw_row(
410                    buf,
411                    (x + 1, area.y, room),
412                    &[Piece::new(format!("│ {said}"), surface(Token::Hydra))],
413                    &mut Hits::default(),
414                    None,
415                );
416            }
417        }
418        if !web.is_empty() && web_x > x {
419            buf.set_string(web_x, area.y, web, surface(Token::Link));
420        }
421        say(
422            buf,
423            Rect::new(area.x + 1, area.y + 1, area.width.saturating_sub(1), 1),
424            0,
425            self.hint(),
426            colors,
427            Token::Muted,
428        );
429    }

468/// A step a key takes through a grid or a list.
469#[derive(Clone, Copy, Debug, PartialEq, Eq)]
470enum Step {
471    Left,
472    Right,
473    Up,
474    Down,
475}
476
477/// The step `key` takes: the arrows, the numpad's, or h, j, k and l.
478fn step(key: &Key) -> Option<Step> {
479    match key.name.as_str() {
480        "ArrowLeft" | "Numpad4" => return Some(Step::Left),
481        "ArrowRight" | "Numpad6" => return Some(Step::Right),
482        "ArrowUp" | "Numpad8" => return Some(Step::Up),
483        "ArrowDown" | "Numpad2" => return Some(Step::Down),
484        _ => {}
485    }
486    match letter(key) {
487        Some('h') => Some(Step::Left),
488        Some('l') => Some(Step::Right),
489        Some('k') => Some(Step::Up),
490        Some('j') => Some(Step::Down),
491        _ => None,
492    }
493}

495/// `key` moving the card chosen among `count`, `per_row` to a row:
496/// whether it was a step.
497fn moved(key: &Key, chosen: &mut usize, (count, per_row): (usize, usize)) -> bool {
498    let Some(step) = step(key) else {
499        return false;
500    };
501    let per_row = per_row.max(1);
502    let last = count.saturating_sub(1);
503    *chosen = match step {
504        Step::Left => chosen.saturating_sub(1),
505        Step::Right => (*chosen + 1).min(last),
506        Step::Up => chosen.checked_sub(per_row).unwrap_or(*chosen),
507        Step::Down => {
508            if *chosen + per_row <= last {
509                *chosen + per_row
510            } else if *chosen / per_row < last / per_row {
511                // Down from the row above the last, which is short: its
512                // last card.
513                last
514            } else {
515                *chosen
516            }
517        }
518    };
519    true
520}
521
522/// `key` moving the row chosen among `count` up or down: whether it was a
523/// step.
524fn moved_in_list(key: &Key, chosen: &mut usize, count: usize) -> bool {
525    match step(key) {
526        Some(Step::Up) => {
527            *chosen = chosen.saturating_sub(1);
528            true
529        }
530        Some(Step::Down) => {
531            *chosen = (*chosen + 1).min(count.saturating_sub(1));
532            true
533        }
534        _ => false,
535    }
536}

538/// Where cards go: as many to a row as fit, a column between them,
539/// scrolled down so the one chosen shows.
540#[derive(Debug, PartialEq, Eq)]
541struct Tiles {
542    /// How many to a row.
543    per_row: usize,
544    /// Each card shown, by its place in the list, and where.
545    places: Vec<(usize, Rect)>,
546}
547
548/// `count` cards of `size` (with their frames) laid in `area`, the one
549/// `chosen` in sight.
550fn tiles(area: Rect, (width, height): (u16, u16), count: usize, chosen: usize) -> Tiles {
551    let (width, height) = (width.max(1), height.max(1));
552    let per_row = usize::from(((area.width + 1) / (width + 1)).max(1));
553    let rows_fit = usize::from((area.height / height).max(1));
554    let first_row = (chosen / per_row).saturating_sub(rows_fit - 1);
555    let mut places = Vec::new();
556    for at in first_row * per_row..count {
557        let row = at / per_row - first_row;
558        if row >= rows_fit {
559            break;
560        }
561        let col = u16::try_from(at % per_row).unwrap_or(0);
562        let row = u16::try_from(row).unwrap_or(0);
563        let x = area.x + col * (width + 1);
564        let y = area.y + row * height;
565        let rect = Rect::new(x, y, width, height).intersection(area);
566        if !rect.is_empty() {
567            places.push((at, rect));
568        }
569    }
570    Tiles { per_row, places }
571}

594/// Draw a field to type in at `x`, `y`, `width` wide: its text, or a dot
595/// for each character of a password; where the cursor goes while it is
596/// typed in.
597fn field(
598    buf: &mut Buffer,
599    (x, y, width): (u16, u16, u16),
600    (edit, masked): (&LineEdit, bool),
601    (typing, colors): (bool, &Colors),
602) -> Option<(u16, u16)> {
603    if width == 0 {
604        return None;
605    }
606    let ground = if typing {
607        Token::Selection
608    } else {
609        Token::Inset
610    };
611    let style = colors.text(Token::Text).bg(colors.color(ground));
612    buf.set_style(Rect::new(x, y, width, 1), style);
613    let room = usize::from(width);
614    let (shown, at) = if masked {
615        let text = edit.text();
616        let before = text[..edit.cursor()].graphemes(true).count();
617        let total = text.graphemes(true).count();
618        let start = before.saturating_sub(room - 1);
619        ("•".repeat((total - start).min(room)), before - start)
620    } else {
621        edit.shown(room)
622    };
623    buf.set_stringn(x, y, &shown, room, style);
624    typing.then(|| {
625        let at = u16::try_from(at).unwrap_or(0).min(width - 1);
626        (x + at, y)
627    })
628}

630/// Draw `pieces` as a row at `x`, `y`, no wider than `width`, each with
631/// what a click on it finds: the columns taken.
632fn buttons(
633    buf: &mut Buffer,
634    (x, y, width): (u16, u16, u16),
635    pieces: Vec<(Piece, Option<Target>)>,
636    hits: &mut Hits,
637) -> u16 {
638    let mut col = 0;
639    for (piece, target) in pieces {
640        if col >= width {
641            break;
642        }
643        let took = draw_row(
644            buf,
645            (x + col, y, width - col),
646            &[piece],
647            &mut Hits::default(),
648            None,
649        );
650        if let Some(target) = target {
651            hits.add(Rect::new(x + col, y, took, 1), target);
652        }
653        col += took;
654    }
655    col
656}

664/// it is about.
665#[derive(Debug)]
666struct Confirm {
667    title: String,
668    question: String,
669    /// What yes says.
670    yes: String,
671    /// What yes asks.
672    request: HubRequest,
673    /// Where yes and no were drawn, for a click.
674    answers: [Rect; 2],
675}
676
677impl Confirm {
678    /// The question `question`, titled `title`, its yes saying `yes` and
679    /// asking `request`.
680    fn new(title: &str, question: String, yes: String, request: HubRequest) -> Self {
681        Self {
682            title: title.to_owned(),
683            question,
684            yes,
685            request,
686            answers: [Rect::default(); 2],
687        }
688    }
689
690    /// Yes: the request asked.
691    fn yes(&self, context: &Context) -> Next {
692        context.sessions.ask(self.request.clone(), &context.runtime);
693        Next::Close
694    }
695}
696
697impl Overlay for Confirm {
698    fn title(&self) -> String {
699        self.title.clone()
700    }
701
702    fn key(&mut self, key: &Key, context: &Context) -> Next {
703        if enter(key) || matches!(letter(key), Some('y' | 'Y')) {
704            self.yes(context)
705        } else if key.name == "Escape" || matches!(letter(key), Some('n' | 'N')) {
706            Next::Close
707        } else {
708            Next::Stay
709        }
710    }
711
712    fn click(&mut self, col: u16, row: u16, context: &Context) -> Next {
713        let inside = |rect: Rect| rect.contains(ratatui::layout::Position::new(col, row));
714        if inside(self.answers[0]) {
715            self.yes(context)
716        } else if inside(self.answers[1]) {
717            Next::Close
718        } else {
719            Next::Stay
720        }
721    }
722
723    fn size(&self) -> (u16, u16) {
724        (50, 30)
725    }
726
727    fn draw(
728        &mut self,
729        buf: &mut Buffer,
730        area: Rect,
731        (_, colors, hits): (&Context, &Colors, &mut Hits),
732    ) -> Option<(u16, u16)> {
733        let rows = wrap(
734            &[Piece::new(
735                self.question.clone(),
736                colors.text(Token::Warning),
737            )],
738            area.width,
739        );
740        let mut y = area.y;
741        for row in rows.iter().take(usize::from(area.height.saturating_sub(2))) {
742            draw_row(
743                buf,
744                (area.x, y, area.width),
745                row,
746                &mut Hits::default(),
747                None,
748            );
749            y += 1;
750        }
751        let y = (y + 1).min(area.y + area.height.saturating_sub(1));
752        let yes = button(&format!("y: {}", self.yes), colors);
753        let yes_width = u16::try_from(yes.text.chars().count()).unwrap_or(0);
754        let took = buttons(
755            buf,
756            (area.x, y, area.width),
757            vec![
758                (yes, None),
759                (Piece::new("   ", colors.canvas()), None),
760                (button("n: Not now", colors), None),
761            ],
762            hits,
763        );
764        self.answers = [
765            Rect::new(area.x, y, yes_width.min(took), 1),
766            Rect::new(
767                area.x + yes_width + 3,
768                y,
769                took.saturating_sub(yes_width + 3),
770                1,
771            ),
772        ];
773        None
774    }
775}

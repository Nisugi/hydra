1//! The theme's colours as a terminal draws them: each meaning's colour
2//! (`Token`) as 24-bit colour, and a run of the game's text in the colour
3//! the windows give it.
4
5use hydra_ui::StyledRun;
6use hydra_ui::run_token;
7use hydra_ui::theme::{Palette, Rgb, Token, parse_hex};
8use ratatui::style::{Color, Modifier, Style};
9
10/// The colours a screen is drawn in.
11#[derive(Clone, Debug, PartialEq)]
12pub struct Colors {
13    palette: Palette,
14}
15
16impl Default for Colors {
17    fn default() -> Self {
18        Self::of(
19            hydra_ui::theme::Themes::built_in()
20                .outfit(hydra_ui::theme::Theme::DEFAULT)
21                .unwrap_or_default()
22                .palette,
23        )
24    }
25}
26
27/// `rgb` as the terminal's colour.
28#[must_use]
29pub const fn rgb([red, green, blue]: Rgb) -> Color {
30    Color::Rgb(red, green, blue)
31}
32
33impl Colors {
34    /// The colours of `palette`.
35    #[must_use]
36    pub const fn of(palette: Palette) -> Self {
37        Self { palette }
38    }
39
40    /// The palette drawn from.
41    #[must_use]
42    pub const fn palette(&self) -> &Palette {
43        &self.palette
44    }
45
46    /// `token`'s colour.
47    #[must_use]
48    pub fn color(&self, token: Token) -> Color {
49        rgb(self.palette.get(token))
50    }
51
52    /// Text in `token`'s colour on the canvas.
53    #[must_use]
54    pub fn text(&self, token: Token) -> Style {
55        Style::new()
56            .fg(self.color(token))
57            .bg(self.color(Token::Canvas))
58    }
59
60    /// The screen's ground and plain text.
61    #[must_use]
62    pub fn canvas(&self) -> Style {
63        self.text(Token::Text)
64    }
65
66    /// Text that is there to be read past: hints, what is off.
67    #[must_use]
68    pub fn muted(&self) -> Style {
69        self.text(Token::Muted)
70    }
71
72    /// A border, and a focused one.
73    #[must_use]
74    pub fn border(&self, focused: bool) -> Style {
75        if focused {
76            self.text(Token::Accent)
77        } else {
78            self.text(Token::Line)
79        }
80    }
81
82    /// What is chosen in a list: drawn on the selection colour.
83    #[must_use]
84    pub fn chosen(&self) -> Style {
85        Style::new()
86            .fg(self.color(Token::Text))
87            .bg(self.color(Token::Selection))
88    }
89
90    /// `run` as the windows colour it: its own colour, else its meaning's
91    /// (a creature, a link, a preset), else the text's; its own ground; bold
92    /// as the game marked it. A link is known by its colour alone: nothing
93    /// on the screen is underlined.
94    #[must_use]
95    pub fn run(&self, run: &StyledRun) -> Style {
96        let fg = run
97            .color
98            .as_deref()
99            .and_then(parse_hex)
100            .unwrap_or_else(|| self.palette.get(run_token(run).unwrap_or(Token::Text)));
101        let bg = run
102            .background
103            .as_deref()
104            .and_then(parse_hex)
105            .unwrap_or_else(|| self.palette.get(Token::Canvas));
106        let mut style = Style::new().fg(rgb(fg)).bg(rgb(bg));
107        if run.bold {
108            style = style.add_modifier(Modifier::BOLD);
109        }
110        style
111    }
112}

114#[cfg(test)]
115mod tests {
116    use super::*;
117
118    /// A run's own colour wins; a creature is the creature's colour, bold;
119    /// a link is not underlined.
120    #[test]
121    fn a_run_is_coloured_as_the_windows_colour_it() {
122        let colors = Colors::default();
123        let own = StyledRun {
124            color: Some("#102030".to_owned()),
125            ..StyledRun::plain("gold")
126        };
127        assert_eq!(colors.run(&own).fg, Some(Color::Rgb(0x10, 0x20, 0x30)));
128        let creature = StyledRun {
129            bold: true,
130            ..StyledRun::plain("a kobold")
131        };
132        let style = colors.run(&creature);
133        assert_eq!(style.fg, Some(colors.color(Token::Creature)));
134        assert!(style.add_modifier.contains(Modifier::BOLD));
135        let link = StyledRun {
136            link: Some(hydra_ui::RunLink::Command {
137                command: "look".to_owned(),
138            }),
139            ..StyledRun::plain("look")
140        };
141        assert!(
142            !colors
143                .run(&link)
144                .add_modifier
145                .contains(Modifier::UNDERLINED)
146        );
147    }
148}

1//! Arranging a character's screen: its windows moved, sized, added, taken
2//! away, their tabs and frames changed, by keys or by the mouse.
3//!
4//! While arranging, the layout is fitted to the terminal, so a cell moved is
5//! a cell on the screen. The window with the focus is the one arranged:
6//!
7//! - the arrows move it, Ctrl with them by five; Shift with them sizes it
8//!   from its right and bottom edges;
9//! - Tab and Shift+Tab give the focus to the next window and the one
10//!   before;
11//! - `a` adds a window, `t` a tab to this one, chosen from the catalog;
12//!   `x` takes away the tab shown, `d` or Delete the window;
13//! - `e` edits its settings (`.editwindow`); `b` changes its frame; `f` fits the whole layout to this terminal, `r`
14//!   lays it out afresh;
15//! - Enter, Escape or F2 is done, and the layout is kept.
16//!
17//! With the mouse, a window's top edge drags it, and its bottom-right
18//! corner sizes it. Out of arranging, any edge with its border drawn sizes
19//! the window and its top edge moves it, unless the window is locked; with
20//! the layout's shared edges on, a window whose edge met the one sized is
21//! sized with it.
22

30/// What arranging asked of the screen.
31#[derive(Clone, Debug, PartialEq, Eq)]
32pub enum Asked {
33    /// Nothing more.
34    Nothing,
35    /// Arranging is done: keep the layout.
36    Done,
37    /// Show the catalog, to add a window.
38    AddWindow,
39    /// Show the catalog, to add a tab to the window with the focus.
40    AddTab,
41    /// Edit the window with the focus.
42    Edit,
43}

45/// The edges of a window a drag sizes; none when it moves the window.
46#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
47#[expect(
48    clippy::struct_excessive_bools,
49    reason = "each edge is sized or not on its own"
50)]
51pub struct Edges {
52    /// Its left edge.
53    pub left: bool,
54    /// Its right edge.
55    pub right: bool,
56    /// Its top edge.
57    pub top: bool,
58    /// Its bottom edge.
59    pub bottom: bool,
60}
61
62impl Edges {
63    /// Whether any is sized: else the window is moved.
64    #[must_use]
65    pub const fn any(self) -> bool {
66        self.left || self.right || self.top || self.bottom
67    }
68}

70/// A drag under way: the window, how, where it began and its place then.
71#[derive(Clone, Debug, PartialEq, Eq)]
72pub struct Drag {
73    /// The window dragged.
74    pub window: usize,
75    /// The edges sized; none, a move.
76    pub edges: Edges,
77    /// Where the mouse went down.
78    pub from: (u16, u16),
79    /// The window's place then.
80    pub was: Rect,
81    /// The windows sized with it, their edge meeting one sized: each its
82    /// number, its place then, and its edges that move.
83    pub sharing: Vec<(usize, Rect, Edges)>,
84}

86/// `rect` moved by `dx`, `dy` and kept within `cols` by `rows`.
87fn moved(rect: Rect, (dx, dy): (i32, i32), (cols, rows): (u16, u16)) -> Rect {
88    let clamp = |at: u16, by: i32, size: u16, most: u16| {
89        let at = i32::from(at) + by;
90        let most = i32::from(most.saturating_sub(size));
91        u16::try_from(at.clamp(0, most.max(0))).unwrap_or(0)
92    };
93    Rect::new(
94        clamp(rect.x, dx, rect.width, cols),
95        clamp(rect.y, dy, rect.height, rows),
96        rect.width,
97        rect.height,
98    )
99}

101/// `rect` with `edges` moved by `dx`, `dy`: at least a cell, and kept
102/// within `cols` by `rows`.
103fn sized_by(rect: Rect, edges: Edges, (dx, dy): (i32, i32), (cols, rows): (u16, u16)) -> Rect {
104    let (mut left, mut top) = (i32::from(rect.x), i32::from(rect.y));
105    let mut right = left + i32::from(rect.width);
106    let mut bottom = top + i32::from(rect.height);
107    if edges.left {
108        left = (left + dx).clamp(0, right - 1);
109    }
110    if edges.right {
111        right = (right + dx).clamp(left + 1, i32::from(cols).max(left + 1));
112    }
113    if edges.top {
114        top = (top + dy).clamp(0, bottom - 1);
115    }
116    if edges.bottom {
117        bottom = (bottom + dy).clamp(top + 1, i32::from(rows).max(top + 1));
118    }
119    let cell = |at: i32| u16::try_from(at.max(0)).unwrap_or(u16::MAX);
120    Rect::new(
121        cell(left),
122        cell(top),
123        cell(right - left),
124        cell(bottom - top),
125    )
126}

128/// `rect` sized by `dw`, `dh` from its right and bottom edges, at least a
129/// cell, and kept within `cols` by `rows`.
130fn sized(rect: Rect, (dw, dh): (i32, i32), (cols, rows): (u16, u16)) -> Rect {
131    let size = |size: u16, by: i32, at: u16, most: u16| {
132        let size = i32::from(size) + by;
133        let most = i32::from(most.saturating_sub(at));
134        u16::try_from(size.clamp(1, most.max(1))).unwrap_or(1)
135    };
136    Rect::new(
137        rect.x,
138        rect.y,
139        size(rect.width, dw, rect.x, cols),
140        size(rect.height, dh, rect.y, rows),
141    )
142}

144/// `key` done to `layout` while arranging, the window `focus` the one
145/// arranged; the layout fitted to `area`, `cols` by `rows`.
146pub fn key(layout: &mut Layout, focus: &mut usize, key: &Key, area: (u16, u16)) -> Asked {
147    let count = layout.windows.len();
148    let step = if key.mods.has(Mods::CTRL) { 5 } else { 1 };
149    let sizing = key.mods.has(Mods::SHIFT);
150    let by = match key.name.as_str() {
151        "ArrowLeft" => Some((-step, 0)),
152        "ArrowRight" => Some((step, 0)),
153        "ArrowUp" => Some((0, -step)),
154        "ArrowDown" => Some((0, step)),
155        _ => None,
156    };
157    if let (Some(by), Some(window)) = (by, layout.windows.get_mut(*focus)) {
158        let rect = window.rect();
159        window.place(if sizing {
160            sized(rect, by, area)
161        } else {
162            moved(rect, by, area)
163        });
164        return Asked::Nothing;
165    }
166    match (key.name.as_str(), key.text) {
167        ("Enter" | "Escape" | "F2", _) => return Asked::Done,
168        ("Tab", _) if count > 0 => {
169            *focus = if sizing {
170                (*focus + count - 1) % count
171            } else {
172                (*focus + 1) % count
173            };
174        }
175        (_, Some('a')) => return Asked::AddWindow,
176        (_, Some('e')) if *focus < count => return Asked::Edit,
177        (_, Some('t')) if *focus < count => return Asked::AddTab,
178        (_, Some('x')) => {
179            if let Some(window) = layout.windows.get_mut(*focus) {
180                let showing = window.showing.min(window.tabs.len().saturating_sub(1));
181                window.remove_tab(showing);
182            }
183        }
184        ("Delete", _) | (_, Some('d')) if *focus < count => {
185            layout.windows.remove(*focus);
186            *focus = (*focus).min(layout.windows.len().saturating_sub(1));
187        }
188        (_, Some('b')) => {
189            if let Some(window) = layout.windows.get_mut(*focus) {
190                let at = Border::ALL
191                    .iter()
192                    .position(|border| *border == window.border())
193                    .unwrap_or(0);
194                window.set_border(Border::ALL[(at + 1) % Border::ALL.len()]);
195            }
196        }
197        (_, Some('f')) => layout.fit(area),
198        (_, Some('r')) => {
199            *layout = Layout::default_for(area);
200            *focus = 0;
201        }
202        _ => {}
203    }
204    Asked::Nothing
205}

207/// A drag to `to`, the mouse at it now: the window it moves or sizes,
208/// placed.
209pub fn drag(layout: &mut Layout, drag: &Drag, to: (u16, u16), area: (u16, u16)) {
210    let by = (
211        i32::from(to.0) - i32::from(drag.from.0),
212        i32::from(to.1) - i32::from(drag.from.1),
213    );
214    if let Some(window) = layout.windows.get_mut(drag.window) {
215        window.place(if drag.edges.any() {
216            sized_by(drag.was, drag.edges, by, area)
217        } else {
218            moved(drag.was, by, area)
219        });
220    }
221    for (at, was, edges) in &drag.sharing {
222        if let Some(window) = layout.windows.get_mut(*at) {
223            window.place(sized_by(*was, *edges, by, area));
224        }
225    }
226}

228/// Where a press at `at` begins a drag of window `window`, placed at
229/// `rect`, out of arranging: on an edge whose border is drawn (`sides`),
230/// that edge sized, two at a corner; on the top edge but its corners, a
231/// move.
232#[must_use]
233pub fn grab(window: usize, rect: Rect, at: (u16, u16), sides: Edges) -> Option<Drag> {
234    let (col, row) = at;
235    let right = rect.x + rect.width.saturating_sub(1);
236    let bottom = rect.y + rect.height.saturating_sub(1);
237    let edges = Edges {
238        left: sides.left && col == rect.x,
239        right: sides.right && col == right,
240        top: sides.top && row == rect.y,
241        bottom: sides.bottom && row == bottom,
242    };
243    if !edges.any() {
244        return None;
245    }
246    // The top edge moves the window; at a corner, it sizes.
247    let edges = if edges.top && !edges.left && !edges.right {
248        Edges::default()
249    } else {
250        edges
251    };
252    Some(Drag {
253        window,
254        edges,
255        from: at,
256        was: rect,
257        sharing: Vec::new(),
258    })
259}

261/// The windows of `layout` sized with window `window` when its `edges`
262/// are: each whose opposite edge met one of them, beside it along it.
263#[must_use]
264pub fn sharing(layout: &Layout, window: usize, edges: Edges) -> Vec<(usize, Rect, Edges)> {
265    let Some(this) = layout.windows.get(window).map(Window::rect) else {
266        return Vec::new();
267    };
268    let beside = |a: Rect, b: Rect| a.y < b.bottom() && b.y < a.bottom();
269    let over = |a: Rect, b: Rect| a.x < b.right() && b.x < a.right();
270    layout
271        .windows
272        .iter()
273        .enumerate()
274        .filter(|(at, _)| *at != window)
275        .filter_map(|(at, other)| {
276            let rect = other.rect();
277            let meets = Edges {
278                left: edges.right && rect.x == this.right() && beside(rect, this),
279                right: edges.left && rect.right() == this.x && beside(rect, this),
280                top: edges.bottom && rect.y == this.bottom() && over(rect, this),
281                bottom: edges.top && rect.bottom() == this.y && over(rect, this),
282            };
283            meets.any().then_some((at, rect, meets))
284        })
285        .collect()
286}

288/// Where a press at `at` begins a drag of window `window`, placed at
289/// `rect`: on its top edge, a move; on its bottom-right corner, a size.
290#[must_use]
291pub fn drag_from(window: usize, rect: Rect, at: (u16, u16)) -> Option<Drag> {
292    let (col, row) = at;
293    let right = rect.x + rect.width.saturating_sub(1);
294    let bottom = rect.y + rect.height.saturating_sub(1);
295    let sizing = col >= right.saturating_sub(1) && col <= right && row == bottom;
296    let moving = row == rect.y && col >= rect.x && col <= right;
297    (sizing || moving).then(|| Drag {
298        window,
299        edges: Edges {
300            right: sizing,
301            bottom: sizing,
302            ..Edges::default()
303        },
304        from: at,
305        was: rect,
306        sharing: Vec::new(),
307    })
308}

310/// A new window of `widget` placed at `at` in an area `cols` by `rows`, as
311/// big as it would like within it.
312#[must_use]
313pub fn new_window(widget: Widget, at: (u16, u16), (cols, rows): (u16, u16)) -> Window {
314    let [width, height] = widget.size();
315    let one_line = height <= hydra_ui::widget::LINE;
316    let (border, chrome) = if one_line {
317        (Border::None, 0)
318    } else {
319        (Border::Rounded, 2)
320    };
321    #[expect(
322        clippy::cast_possible_truncation,
323        clippy::cast_sign_loss,
324        reason = "a widget's wished size in points, a few hundred at most, as cells"
325    )]
326    let (want_cols, want_rows) = (
327        (width / 8.0).round() as u16 + chrome,
328        (height / hydra_ui::widget::LINE).round() as u16 + chrome,
329    );
330    let col = at.0.min(cols.saturating_sub(1));
331    let row = at.1.min(rows.saturating_sub(1));
332    let width = want_cols.clamp(4, cols.saturating_sub(col).max(1));
333    let height = want_rows.clamp(1, rows.saturating_sub(row).max(1));
334    Window::new((row, col), (height, width), &[widget], border)
335}

337/// What the catalog adds: a widget, and the character it follows, by its
338/// roster name, when another than the screen's own.
339pub type Pick = (Widget, Option<String>);
340
341/// The catalog to choose a widget from: each group a menu of its widgets,
342/// opened beside it, the streams the character has heard besides those
343/// Hydra names among the streams, and each kind `shown` already marked.
344/// Last, *Advanced*: each of the `others`, by roster name, a menu of the
345/// same, added following that character; a story is the screen's own.
346#[must_use]
347pub fn catalog(streams: &[String], shown: &[Widget], others: &[String]) -> Vec<Entry<Pick>> {
348    let mut entries = groups(streams, shown, None);
349    if !others.is_empty() {
350        let followed = others
351            .iter()
352            .map(|name| {
353                Entry::sub(
354                    format!("Following {name}"),
355                    groups(streams, &[], Some(name)),
356                )
357            })
358            .collect();
359        entries.push(Entry::sub("Advanced", followed));
360    }
361    entries
362}
363
364/// The catalog's groups, each a menu of its widgets, those `shown` marked,
365/// added following `whose` when another's: no story then.
366fn groups(streams: &[String], shown: &[Widget], whose: Option<&String>) -> Vec<Entry<Pick>> {
367    let mut every: Vec<Widget> = Widget::all()
368        .into_iter()
369        .filter(|widget| !widget.is_hidden())
370        .filter(|widget| whose.is_none() || !widget.is_story())
371        .collect();
372    for stream in streams.iter().filter(|_| whose.is_none()) {
373        let widget = Widget::Stream(stream.clone());
374        if !every.contains(&widget) {
375            every.push(widget);
376        }
377    }
378    let mut entries = Vec::new();
379    for group in Group::ALL {
380        let mut in_group = every
381            .iter()
382            .filter(|widget| widget.group() == group)
383            .peekable();
384        if in_group.peek().is_none() {
385            continue;
386        }
387        let widgets = in_group
388            .map(|widget| {
389                let mut label = widget.name().into_owned();
390                if shown.contains(widget) {
391                    label.push_str(" (shown)");
392                }
393                Entry::new(label, (widget.clone(), whose.cloned()))
394            })
395            .collect();
396        entries.push(Entry::sub(group.name(), widgets));
397    }
398    entries
399}

1//! A terminal asking the Hydra running on a data folder for its screen.
2//!
3//! The running Hydra holds [`LOCK`] in the data folder for as long as it
4//! runs, and writes the loopback port and key it listens on to [`PORT`]. A
5//! terminal says the key and `tui` there, with the character to open on,
6//! and the connection stays: Hydra answers in a line, then draws a screen
7//! over it ([`crate::wire`]).
8
9use std::fs::{OpenOptions, TryLockError};
10use std::io::Write as _;
11use std::net::{Ipv4Addr, SocketAddr, TcpStream};
12use std::path::{Path, PathBuf};
13use std::time::Duration;
14
15/// The file whose lock says Hydra runs on the data folder.
16pub const LOCK: &str = "hydra.lock";
17/// Where the running process says which port and key it listens on.
18pub const PORT: &str = "hydra.port";
19/// How long a launch waits on the running process, which may be starting
20/// up and not have said its port yet: twenty tries a quarter second apart.
21pub const TRIES: u32 = 20;
22/// The wait between two tries.
23pub const BETWEEN_TRIES: Duration = Duration::from_millis(250);
24/// How long the running process waits for a launch to say its line.
25pub const SAY_WAIT: Duration = Duration::from_secs(2);
26/// The longest line a launch is heard to the end of.
27pub const LONGEST_LINE: u64 = 128;
28
29/// Where Hydra keeps its own files: `HYDRA_DATA_DIR`, or `data` beside the
30/// working directory.
31#[must_use]
32pub fn data_dir() -> PathBuf {
33    std::env::var_os("HYDRA_DATA_DIR").map_or_else(|| PathBuf::from("data"), PathBuf::from)
34}
35
36/// Whether Hydra runs on `data` now: its lock is held.
37#[must_use]
38pub fn running(data: &Path) -> bool {
39    let Ok(lock) = OpenOptions::new().write(true).open(data.join(LOCK)) else {
40        return false;
41    };
42    matches!(lock.try_lock(), Err(TryLockError::WouldBlock))
43}
44
45/// The port and key the running Hydra wrote, when they read.
46#[must_use]
47pub fn said(data: &Path) -> Option<(u16, String)> {
48    let text = std::fs::read_to_string(data.join(PORT)).ok()?;
49    let mut words = text.split_whitespace();
50    let port = words.next()?.parse().ok()?;
51    let key = words.next()?.to_owned();
52    Some((port, key))
53}
54
55/// Ask the Hydra running on `data` for a screen in this terminal, opened on
56/// `seat` (`GAME:Name`) when one is named: the connection it is drawn over.
57///
58/// # Errors
59///
60/// It did not answer, or refused: why.
61pub fn ask(data: &Path, seat: Option<&str>) -> Result<TcpStream, String> {
62    let mut refused = None;
63    for _ in 0..TRIES {
64        if let Some((port, key)) = said(data) {
65            let at = SocketAddr::from((Ipv4Addr::LOCALHOST, port));
66            if let Ok(mut stream) = TcpStream::connect_timeout(&at, Duration::from_secs(1))
67                && stream
68                    .write_all(crate::wire::ask_line(&key, seat).as_bytes())
69                    .is_ok()
70            {
71                match answer_of(&mut stream) {
72                    Ok(()) => return Ok(stream),
73                    Err(why) => refused = Some(why),
74                }
75                break;
76            }
77        }
78        std::thread::sleep(BETWEEN_TRIES);
79    }
80    Err(refused.unwrap_or_else(|| {
81        format!(
82            "Hydra on {} did not answer: is it starting?",
83            data.display()
84        )
85    }))
86}
87
88/// Hydra's answer line, read a byte at a time so nothing after it is taken.
89fn answer_of(stream: &mut TcpStream) -> Result<(), String> {
90    use std::io::Read as _;
91    let _ = stream.set_read_timeout(Some(SAY_WAIT * 5));
92    let mut line = Vec::new();
93    let mut byte = [0u8; 1];
94    while u64::try_from(line.len()).unwrap_or(u64::MAX) < LONGEST_LINE {
95        match stream.read(&mut byte) {
96            Ok(1) if byte[0] == b'\n' => break,
97            Ok(1) => line.push(byte[0]),
98            Ok(_) => return Err("Hydra closed the connection.".to_owned()),
99            Err(why) => return Err(format!("Hydra did not answer: {why}")),
100        }
101    }
102    let _ = stream.set_read_timeout(None);
103    crate::wire::answered(&String::from_utf8_lossy(&line))
104}

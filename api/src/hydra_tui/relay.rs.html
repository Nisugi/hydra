1//! Hydra's side of an attached terminal: two threads of its own.
2//!
3//! One reads what the terminal sends. The other keeps the terminal's screen:
4//! woken by what the terminal did and by the board (each character's feed
5//! wakes it as it wakes the windows), it draws the screen into bytes and
6//! sends them. What the terminal did is drawn at once; what the game did, no
7//! oftener than `PACE`; the clocks at the next whole second, or oftener in
8//! roundtime. A screen that cannot be written to for `WRITE_WAIT` is
9//! given up, as is one whose terminal never says hello.
10//!
11//! Its characters play on when it leaves: a terminal is a window, not a
12//! session.
13

28/// The least time between two frames drawn for the game alone.
29const PACE: Duration = Duration::from_millis(33);
30/// How long a terminal has to say hello.
31const HELLO_WAIT: Duration = Duration::from_secs(10);
32/// How long a frame may take to be written before the terminal is given up.
33const WRITE_WAIT: Duration = Duration::from_secs(10);
34
35/// What wakes a screen.
36enum Wakeup {
37    /// The terminal said something.
38    Up(Up),
39    /// The terminal's connection ended.
40    Gone,
41    /// The board changed.
42    Board,
43    /// Open this link, the windows being closed.
44    Url(String),
45}
46
47/// A screen as the board wakes it.
48struct Screen(Sender<Wakeup>);
49
50impl Windows for Screen {
51    fn repaint(&self) {
52        let _ = self.0.send(Wakeup::Board);
53    }
54
55    fn repaint_seat(&self, _id: u32) {
56        let _ = self.0.send(Wakeup::Board);
57    }
58
59    fn open_url(&self, url: &str) {
60        let _ = self.0.send(Wakeup::Url(url.to_owned()));
61    }
62}
63
64/// A screen attached to the board until this is dropped, however its
65/// thread ends.
66struct Attached {
67    wake: Wake,
68    number: u64,
69}
70
71impl Drop for Attached {
72    fn drop(&mut self) {
73        self.wake.detached(self.number);
74        hydra_board::store::flush();
75    }
76}

78/// Keep a screen for the terminal on `stream`, opened on the character
79/// named `seat` (`GAME:Name`) when one is, on threads of its own.
80pub fn serve(stream: TcpStream, seat: Option<String>, context: Context) {
81    let started = std::thread::Builder::new()
82        .name("hydra-tui".to_owned())
83        .spawn(move || {
84            hydra_mem::part(hydra_mem::Part::Windows, || screen(stream, seat, context));
85        });
86    if let Err(why) = started {
87        eprintln!("[hydra] a terminal's screen cannot start: {why}");
88    }
89}

91/// Read what the terminal sends into `wakes` until its connection ends.
92fn read(mut stream: TcpStream, wakes: &Sender<Wakeup>) {
93    while let Ok(up) = wire::read_up(&mut stream) {
94        let bye = up == Up::Bye;
95        if wakes.send(Wakeup::Up(up)).is_err() || bye {
96            return;
97        }
98    }
99    let _ = wakes.send(Wakeup::Gone);
100}

102/// The terminal's first word: its size, where it is, its keys.
103fn hello(wakes: &Receiver<Wakeup>) -> Option<(Size, bool)> {
104    let until = Instant::now() + HELLO_WAIT;
105    loop {
106        let left = until.saturating_duration_since(Instant::now());
107        match wakes.recv_timeout(left).ok()? {
108            Wakeup::Up(Up::Hello {
109                cols, rows, remote, ..
110            }) => return Some((Size::new(cols, rows), remote)),
111            Wakeup::Gone | Wakeup::Up(Up::Bye) => return None,
112            _ => {}
113        }
114    }
115}

117/// The screen's thread: the connection shut down however it ends, so the
118/// reading thread ends with it.
119fn screen(mut stream: TcpStream, seat: Option<String>, context: Context) {
120    keep(&mut stream, seat, context);
121    let _ = stream.shutdown(std::net::Shutdown::Both);
122}

124/// Keep the screen for the terminal on `stream` until it ends.
125fn keep(stream: &mut TcpStream, seat: Option<String>, context: Context) {
126    let _ = stream.set_write_timeout(Some(WRITE_WAIT));
127    let (wakes, woken) = mpsc::channel();
128    let Ok(reading) = stream.try_clone() else {
129        return;
130    };
131    {
132        let wakes = wakes.clone();
133        let started = std::thread::Builder::new()
134            .name("hydra-tui-read".to_owned())
135            .spawn(move || read(reading, &wakes));
136        if started.is_err() {
137            return;
138        }
139    }
140    let Some((size, remote)) = hello(&woken) else {
141        return;
142    };
143    let wake = context.sessions.wake();
144    let number = wake.attached(Arc::new(Screen(wakes)));
145    let _attached = Attached { wake, number };
146    let Ok(mut terminal) = Terminal::new(Wire::new(size)) else {
147        return;
148    };
149    let _ = terminal.clear();
150    let mut closing = context.sessions.closes();
151    let mut app = App::new(context, seat, remote);
152    let mut last_drawn: Option<Instant> = None;
153    let mut dirty = true;
154    let mut input = true;
155    loop {
156        if *closing.borrow_and_update() {
157            let _ = wire::send_down(stream, &Down::Bye("Hydra has shut down.".to_owned()));
158            return;
159        }
160        if app.leaving() {
161            let _ = wire::send_down(
162                stream,
163                &Down::Bye("Left; the characters play on.".to_owned()),
164            );
165            return;
166        }
167        let now = Instant::now();
168        let paced = last_drawn.is_none_or(|at| now.saturating_duration_since(at) >= PACE);
169        if dirty && (input || paced) {
170            let began = Instant::now();
171            if terminal.draw(|frame| app.draw(frame)).is_err() {
172                return;
173            }
174            let bytes = terminal.backend_mut().take();
175            if !bytes.is_empty() && wire::send_down(stream, &Down::Draw(bytes)).is_err() {
176                return;
177            }
178            for down in app.take_out() {
179                if wire::send_down(stream, &down).is_err() {
180                    return;
181                }
182            }
183            app.sent(began);
184            last_drawn = Some(Instant::now());
185            dirty = false;
186            input = false;
187        }
188        let wait = if dirty {
189            PACE.saturating_sub(last_drawn.map_or(PACE, |at| at.elapsed()))
190        } else {
191            app.due()
192        };
193        let first = match woken.recv_timeout(wait) {
194            Ok(wakeup) => Some(wakeup),
195            Err(RecvTimeoutError::Timeout) => None,
196            Err(RecvTimeoutError::Disconnected) => return,
197        };
198        dirty = true;
199        for wakeup in first.into_iter().chain(woken.try_iter()) {
200            match wakeup {
201                Wakeup::Up(Up::Resize { cols, rows }) => {
202                    terminal.backend_mut().resize(Size::new(cols, rows));
203                    let _ = terminal.autoresize();
204                    let _ = terminal.clear();
205                    input = true;
206                }
207                Wakeup::Up(up) => {
208                    app.up(up);
209                    input = true;
210                }
211                Wakeup::Gone => return,
212                Wakeup::Board => {}
213                Wakeup::Url(url) => {
214                    if wire::send_down(stream, &Down::OpenUrl(url)).is_err() {
215                        return;
216                    }
217                }
218            }
219        }
220    }

1//! A character's screen in a terminal, laid out in Vellum's layout format: a
2//! Vellum `layout.toml` reads as it is, and Hydra's is one.
3//!
4//! Each window ([`vellum_schema::WindowDef`], tagged by `widget_type`) has
5//! its place by row and column, its chrome (border sides and style, title
6//! and where it sits, colours) and the widget it shows with that widget's
7//! own settings. Hydra adds to the format only what Vellum has none of: a
8//! `hydra` window of one of its own widgets, a window that `follows`
9//! another character, and which `hand` a hand window shows. A window's
10//! defaults are its widget kind's: there are no window templates.
11//!
12//! The layout keeps the size of terminal it was arranged for
13//! (`terminal_width`, `terminal_height`). A terminal of another size draws
14//! it scaled, every edge where it falls in proportion, so windows that
15//! touched still touch; a window a row or three tall keeps its height.
16//! A window this build does not know is kept and written back as it was.
17//!
18//! Kept beside the character's other files as `<instance>_<name>.tui.toml`
19//! in the data folder's `layouts`, and by name for any character in
20//! `layouts/tui/`. A file that does not read is kept aside, not written
21//! over, and the screen starts from the default, Vellum's sidebar layout.
22//! A layout Hydra wrote in its earlier form is read and written in this one.

41/// A window's frame, as arranging steps through them.
42#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
43#[serde(rename_all = "snake_case")]
44pub enum Border {
45    /// None: a bar or a line, the whole of its place its widget's.
46    None,
47    /// Square corners: Vellum's `single`.
48    #[default]
49    Plain,
50    /// Round corners.
51    Rounded,
52    /// Doubled lines.
53    Double,
54    /// Thick lines.
55    Thick,
56}
57
58impl Border {
59    /// Every frame, in the order offered.
60    pub const ALL: [Self; 5] = [
61        Self::None,
62        Self::Plain,
63        Self::Rounded,
64        Self::Double,
65        Self::Thick,
66    ];
67
68    /// As a player reads it.
69    #[must_use]
70    pub const fn said(self) -> &'static str {
71        match self {
72            Self::None => "none",
73            Self::Plain => "plain",
74            Self::Rounded => "rounded",
75            Self::Double => "double",
76            Self::Thick => "thick",
77        }
78    }
79
80    /// As the layout file writes it.
81    const fn style(self) -> &'static str {
82        match self {
83            Self::None | Self::Plain => "single",
84            Self::Rounded => "rounded",
85            Self::Double => "double",
86            Self::Thick => "thick",
87        }
88    }
89}

91/// One window of the layout: its definition as the file has it, the
92/// widgets it shows worked out from it, and the tab shown.
93#[derive(Clone, Debug, PartialEq)]
94pub struct Window {
95    /// As the file has it.
96    pub def: WindowDef,
97    /// Its widgets: one, or several as tabs.
98    pub tabs: Vec<Widget>,
99    /// The tab shown.
100    pub showing: usize,
101    /// A custom window's cells, each a window placed inside it.
102    pub cells: Vec<Window>,
103}
104
105impl Window {
106    /// The window `def` defines.
107    #[must_use]
108    pub fn of(def: WindowDef) -> Self {
109        let cells = match &def {
110            WindowDef::Custom { data, .. } => data.cells.iter().cloned().map(Self::of).collect(),
111            _ => Vec::new(),
112        };
113        Self {
114            tabs: widgets_of(&def),
115            def,
116            showing: 0,
117            cells,
118        }
119    }
120
121    /// A window of `tabs` at `row`, `col`, `rows` by `cols`, framed so.
122    #[must_use]
123    pub fn new(
124        (row, col): (u16, u16),
125        (rows, cols): (u16, u16),
126        tabs: &[Widget],
127        border: Border,
128    ) -> Self {
129        let name = tabs
130            .first()
131            .map_or_else(|| "window".to_owned(), Widget::type_key);
132        let mut base = base(&name, Rect::new(col, row, cols, rows));
133        set_border(&mut base, border);
134        Self::of(def_of(tabs, base))
135    }
136
137    /// A new window of the kind the layout file names `kind` (`text`,
138    /// `compass`, `progress`...), named `name`, at `rect`, with that kind's
139    /// defaults; none for a kind there is none of.
140    #[must_use]
141    pub fn of_kind(kind: &str, name: &str, rect: Rect) -> Option<Self> {
142        WindowDef::blank(kind, base(name, rect)).map(Self::of)
143    }
144
145    /// Its place and chrome.
146    #[must_use]
147    pub fn base(&self) -> &WindowBase {
148        self.def.base()
149    }
150
151    /// The widget shown.
152    #[must_use]
153    pub fn shown(&self) -> Option<&Widget> {
154        self.tabs.get(self.showing).or_else(|| self.tabs.first())
155    }
156
157    /// Whether it is drawn: neither hidden nor a window with nothing a
158    /// terminal draws (a spacer, the command line, which is at the foot).
159    /// A custom window is drawn for its frame, its cells after it.
160    #[must_use]
161    pub fn drawn(&self) -> bool {
162        self.base().visibility != WindowVisibility::Hidden
163            && (!self.tabs.is_empty()
164                || matches!(
165                    self.def,
166                    WindowDef::Custom { .. } | WindowDef::CommandInput { .. }
167                ))
168    }
169
170    /// The edges its border is drawn on.
171    #[must_use]
172    pub fn sides(&self) -> crate::arrange::Edges {
173        let base = self.base();
174        let sides = &base.border_sides;
175        let shown = base.show_border;
176        crate::arrange::Edges {
177            left: shown && sides.left,
178            right: shown && sides.right,
179            top: shown && sides.top,
180            bottom: shown && sides.bottom,
181        }
182    }
183
184    /// How it writes an effect's time left: an effects window's choice,
185    /// else the default.
186    #[must_use]
187    pub fn times(&self) -> vellum_schema::TimeFormat {
188        match &self.def {
189            WindowDef::ActiveEffects { data, .. } => data.time_format,
190            _ => vellum_schema::TimeFormat::default(),
191        }
192    }
193
194    /// Where its widgets go when it is at `rect`: inside its frame.
195    #[must_use]
196    pub fn inside(&self, rect: Rect) -> Rect {
197        let base = self.base();
198        if base.show_border {
199            ratatui::widgets::Block::default()
200                .borders(crate::chrome::borders(&base.border_sides))
201                .inner(rect)
202        } else {
203            rect
204        }
205    }
206
207    /// Whether its widgets are drawn side by side, each its share: a
208    /// dashboard of indicators, or the four vitals in one.
209    #[must_use]
210    pub const fn across(&self) -> bool {
211        matches!(
212            self.def,
213            WindowDef::Dashboard { .. } | WindowDef::MiniVitals { .. }
214        )
215    }
216
217    /// Its frame.
218    #[must_use]
219    pub fn border(&self) -> Border {
220        let base = self.base();
221        if !base.show_border {
222            return Border::None;
223        }
224        match base.border_style.as_str() {
225            "rounded" => Border::Rounded,
226            "double" => Border::Double,
227            "thick" => Border::Thick,
228            _ => Border::Plain,
229        }
230    }
231
232    /// Framed as `border` now.
233    pub fn set_border(&mut self, border: Border) {
234        set_border(self.def.base_mut(), border);
235    }
236
237    /// Its title, when it is shown and set: else its tabs name it.
238    #[must_use]
239    pub fn title(&self) -> Option<&str> {
240        let base = self.base();
241        base.show_title.then_some(base.title.as_deref()).flatten()
242    }
243
244    /// Another character it shows, by its roster name.
245    #[must_use]
246    pub fn follows(&self) -> Option<&str> {
247        self.base().follows.as_deref()
248    }
249
250    /// Its place as arranged.
251    #[must_use]
252    pub fn rect(&self) -> Rect {
253        let base = self.base();
254        Rect::new(
255            base.col.get(),
256            base.row.get(),
257            base.cols.get(),
258            base.rows.get(),
259        )
260    }
261
262    /// Its place, moved and sized to `rect`.
263    pub fn place(&mut self, rect: Rect) {
264        let base = self.def.base_mut();
265        base.col = Col::new(rect.x);
266        base.row = Row::new(rect.y);
267        base.cols = Width::new(rect.width);
268        base.rows = Height::new(rect.height);
269    }
270
271    /// `widget` as a tab of it, shown: its widgets made one window of
272    /// tabs, its place and chrome kept.
273    pub fn push_tab(&mut self, widget: Widget) {
274        let mut tabs = self.tabs.clone();
275        tabs.push(widget);
276        self.showing = tabs.len() - 1;
277        self.retab(&tabs);
278    }
279
280    /// Its tab `at` taken away; whether it was.
281    pub fn remove_tab(&mut self, at: usize) -> bool {
282        if at >= self.tabs.len() || self.tabs.len() < 2 {
283            return false;
284        }
285        let mut tabs = self.tabs.clone();
286        tabs.remove(at);
287        self.showing = self.showing.min(tabs.len() - 1);
288        self.retab(&tabs);
289        true
290    }
291
292    /// Made a window of `tabs`, its place and chrome kept.
293    fn retab(&mut self, tabs: &[Widget]) {
294        let base = self.base().clone();
295        let showing = self.showing;
296        *self = Self::of(def_of(tabs, base));
297        self.showing = showing.min(self.tabs.len().saturating_sub(1));
298    }
299}

306        col: Col::new(rect.x),
307        rows: Height::new(rect.height),
308        cols: Width::new(rect.width),
309        show_border: true,
310        border_style: "single".to_owned(),
311        border_sides: BorderSides::default(),
312        border_color: None,
313        show_title: true,
314        title: None,
315        title_position: "top-left".to_owned(),
316        background_color: None,
317        text_color: None,
318        transparent_background: false,
319        locked: false,
320        min_rows: None,
321        max_rows: None,
322        min_cols: None,
323        max_cols: None,
324        visibility: WindowVisibility::Shown,
325        binding: None,
326        content_align: None,
327        tts_speak: false,
328        text_size: None,
329        font_family: None,
330        follows: None,
331    }
332}

340/// Whether `widget` is text read as a stream: the story, or a stream.
341const fn is_text(widget: &Widget) -> bool {
342    matches!(widget, Widget::Story | Widget::Stream(_))
343}

346fn stream_of(widget: &Widget) -> String {

348        Widget::Stream(id) => id.clone(),
349        _ => "main".to_owned(),
350    }

353/// The window showing `tabs`, with `base`'s place and chrome: text tabs as
354/// Vellum's tabbed text, one widget as Vellum's widget of the same kind,
355/// and one Vellum has none of as Hydra's own.
356fn def_of(tabs: &[Widget], base: WindowBase) -> WindowDef {
357    if tabs.len() > 1 && tabs.iter().all(is_text) {
358        let tabs = tabs
359            .iter()
360            .map(|widget| TabbedTextTab {
361                name: widget.name().into_owned(),
362                stream: Some(stream_of(widget)),
363                streams: Vec::new(),
364                show_timestamps: None,
365                ignore_activity: None,
366                timestamp_position: None,
367            })
368            .collect();
369        let mut def = blank("tabbedtext", base);
370        if let WindowDef::TabbedText { data, .. } = &mut def {
371            data.tabs = tabs;
372        }
373        return def;
374    }
375    match tabs.first() {
376        Some(widget) => def_of_widget(widget, base),
377        None => blank("spacer", base),
378    }
379}

381/// Vellum's blank window of `kind`; Hydra knows every kind it names.
382fn blank(kind: &str, base: WindowBase) -> WindowDef {
383    WindowDef::blank(kind, base.clone()).unwrap_or(WindowDef::Hydra {
384        base,
385        data: HydraWidgetData {
386            widget: Widget::Story,
387        },
388    })
389}

391/// The window showing `widget`, with `base`'s place and chrome.
392fn def_of_widget(widget: &Widget, base: WindowBase) -> WindowDef {
393    let id = |kind: &str, id: &str, base: WindowBase| {
394        let mut def = blank(kind, base);
395        match &mut def {
396            WindowDef::Progress { data, .. } => data.id = Some(id.to_owned()),
397            WindowDef::Countdown { data, .. } => data.id = Some(id.to_owned()),
398            _ => {}
399        }
400        def
401    };
402    match widget {
403        Widget::Story => text(vec!["main".to_owned()], base),
404        Widget::Stream(stream) => text(vec![stream.clone()], base),
405        Widget::Health => id("progress", "health", base),
406        Widget::Mana => id("progress", "mana", base),
407        Widget::Stamina => id("progress", "stamina", base),
408        Widget::Spirit => id("progress", "spirit", base),
409        Widget::Encumbrance => id("progress", "encumlevel", base),
410        Widget::Stance => id("progress", "pbarStance", base),
411        Widget::Mind => id("progress", "mindState", base),
412        Widget::NextLevel => id("progress", "nextLvlPB", base),
413        Widget::Roundtime => id("countdown", "roundtime", base),
414        Widget::CastTime => id("countdown", "casttime", base),
415        Widget::Stun => id("countdown", "stun", base),
416        Widget::LeftHand | Widget::RightHand | Widget::Prepared => hand(widget, base),
417        Widget::Indicator(indicator) => {
418            let mut def = blank("indicator", base);
419            if let WindowDef::Indicator { data, .. } = &mut def {
420                data.indicator_id = Some(indicator.id().to_owned());
421            }
422            def
423        }
424        Widget::Effects(category) => {
425            let mut def = blank("active_effects", base);
426            if let WindowDef::ActiveEffects { data, .. } = &mut def {
427                data.category = category.game_id().replace(' ', "");
428            }
429            def
430        }
431        Widget::Dialog(id) => {
432            let mut def = blank("dialogpanel", base);
433            if let WindowDef::DialogPanel { data, .. } = &mut def {
434                data.dialog_id.clone_from(id);
435            }
436            def
437        }
438        other => match vellums_kind(other) {
439            Some(kind) => blank(kind, base),
440            None => WindowDef::Hydra {
441                base,
442                data: HydraWidgetData {
443                    widget: other.clone(),
444                },
445            },
446        },
447    }
448}

472/// A hand window showing `widget`, which hand written out.
473fn hand(widget: &Widget, base: WindowBase) -> WindowDef {
474    let mut def = blank("hand", base);
475    if let WindowDef::Hand { data, .. } = &mut def {
476        data.hand = Some(
477            match widget {
478                Widget::LeftHand => "left",
479                Widget::Prepared => "spell",
480                _ => "right",
481            }
482            .to_owned(),
483        );
484    }
485    def
486}

505/// The indicator Vellum names `id`.
506fn indicator(id: &str) -> Option<Indicator> {
507    let id = id.to_ascii_lowercase();
508    let id = id.strip_prefix("icon").unwrap_or(&id);
509    Indicator::ALL
510        .into_iter()
511        .find(|indicator| indicator.id() == id || indicator.name().eq_ignore_ascii_case(id))
512}

514/// What a window of `def` shows, as Hydra's widgets: none for what a
515/// terminal draws nothing of here (a spacer, the command line at the foot,
516/// another program's panel).
517fn widgets_of(def: &WindowDef) -> Vec<Widget> {
518    let one = |widget: Widget| vec![widget];
519    match def {
520        WindowDef::Text { data, .. } => one(text_widget(&data.streams)),
521        WindowDef::TabbedText { data, .. } => {
522            let tabs: Vec<Widget> = data
523                .tabs
524                .iter()
525                .map(|tab| match &tab.stream {
526                    Some(stream) => text_widget(std::slice::from_ref(stream)),
527                    None => text_widget(&tab.streams),
528                })
529                .collect();
530            if tabs.is_empty() {
531                one(Widget::Story)
532            } else {
533                tabs
534            }
535        }
536        WindowDef::Hand { base, data } => {
537            let hand = data.hand.as_deref().unwrap_or(&base.name);
538            one(match hand.to_ascii_lowercase().as_str() {
539                "left" => Widget::LeftHand,
540                "spell" => Widget::Prepared,
541                _ => Widget::RightHand,
542            })
543        }
544        WindowDef::Countdown { data, .. } => one(match data.id.as_deref() {
545            Some("casttime") => Widget::CastTime,
546            Some("stun" | "stuntime") => Widget::Stun,
547            _ => Widget::Roundtime,
548        }),
549        WindowDef::Progress { base, data } => progress(data.id.as_deref().unwrap_or(&base.name))
550            .map(one)
551            .unwrap_or_default(),
552        WindowDef::Indicator { base, data } => data
553            .indicator_id
554            .as_deref()
555            .or(Some(base.name.as_str()))
556            .and_then(indicator)
557            .map(|which| one(Widget::Indicator(which)))
558            .unwrap_or_default(),
559        WindowDef::Dashboard { data, .. } => data
560            .indicators
561            .iter()
562            .filter_map(|each| indicator(&each.id))
563            .map(Widget::Indicator)
564            .collect(),
565        WindowDef::MiniVitals { .. } => vec![
566            Widget::Health,
567            Widget::Mana,
568            Widget::Stamina,
569            Widget::Spirit,
570        ],
571        WindowDef::ActiveEffects { data, .. } => {
572            one(Widget::Effects(match data.category.as_str() {
573                "Buffs" => Category::Buffs,
574                "Debuffs" => Category::Debuffs,
575                "Cooldowns" => Category::Cooldowns,
576                _ => Category::ActiveSpells,
577            }))
578        }
579        WindowDef::DialogPanel { data, .. } => one(Widget::Dialog(data.dialog_id.clone())),
580        WindowDef::Hydra { data, .. } => one(data.widget.clone()),
581        other => fixed_widget(other).map(one).unwrap_or_default(),
582    }
583}

647/// A character's screen, laid out.
648#[derive(Clone, Debug, PartialEq)]
649pub struct Layout {
650    /// The columns of the terminal it was arranged for.
651    pub cols: u16,
652    /// Its rows.
653    pub rows: u16,
654    /// Its windows, the later drawn over the earlier.
655    pub windows: Vec<Window>,
656    /// What the file holds besides.
657    pub kept: Kept,
658}

660/// The file, as Vellum writes it.
661#[derive(Serialize)]
662struct File<'a> {
663    windows: Vec<&'a WindowDef>,
664    terminal_width: u16,
665    terminal_height: u16,
666    #[serde(skip_serializing_if = "Option::is_none")]
667    base_layout: Option<&'a str>,
668    #[serde(skip_serializing_if = "Option::is_none")]
669    theme: Option<&'a str>,
670    #[serde(skip_serializing_if = "Vec::is_empty")]
671    deleted_windows: Vec<&'a WindowDef>,
672    #[serde(skip_serializing_if = "is_off")]
673    shared_edges: bool,
674}

676/// Whether `on` is off: a choice written only when on.
677#[expect(clippy::trivially_copy_pass_by_ref, reason = "serde asks a reference")]
678const fn is_off(on: &bool) -> bool {
679    !*on
680}

682impl Layout {
683    /// Every window drawn, in order: the layout's own, then each custom
684    /// window's cells. A window's number in [`Self::placed`] is its place
685    /// here; a cell's is past the layout's own.
686    #[must_use]
687    pub fn drawn(&self) -> Vec<&Window> {
688        self.windows
689            .iter()
690            .chain(self.windows.iter().flat_map(|window| &window.cells))
691            .collect()
692    }
693
694    /// Whether a window of its own holds the command line; else the line
695    /// is at the screen's foot.
696    #[must_use]
697    pub fn has_command_line(&self) -> bool {
698        self.windows.iter().any(|window| {
699            window.base().visibility != WindowVisibility::Hidden
700                && matches!(window.def, WindowDef::CommandInput { .. })
701        })
702    }
703
704    /// Each window's place in an area `cols` by `rows`, as [`Self::drawn`]
705    /// orders them: the layout's own fitted ([`Self::fitted`]), then each
706    /// custom window's cells where they sit inside it, cut to it.
707    #[must_use]
708    pub fn placed(&self, size: (u16, u16)) -> Vec<Rect> {
709        let mut placed = self.fitted(size);
710        let cells: Vec<Rect> = self
711            .windows
712            .iter()
713            .zip(&placed)
714            .flat_map(|(window, rect)| {
715                let inside = window.inside(*rect);
716                let arranged = window.inside(window.rect());
717                window
718                    .cells
719                    .iter()
720                    .map(move |cell| cell_placed(cell, arranged, inside))
721            })
722            .collect();
723        placed.extend(cells);
724        placed
725    }
726
727    /// Each of the layout's own windows' place in an area `cols` by `rows`,
728    /// in order: fitted from what it was arranged for as Vellum fits it,
729    /// the difference shared out (`resize.rs`), and none outside the area.
730    #[must_use]
731    pub fn fitted(&self, (cols, rows): (u16, u16)) -> Vec<Rect> {
732        let mut defs: Vec<WindowDef> = self
733            .windows
734            .iter()
735            .map(|window| window.def.clone())
736            .collect();
737        resize::resize(&mut defs, (self.cols, self.rows), (cols, rows));
738        defs.iter()
739            .map(|def| {
740                let base = def.base();
741                let left = base.col.get().min(cols.saturating_sub(1));
742                let top = base.row.get().min(rows.saturating_sub(1));
743                Rect::new(
744                    left,
745                    top,
746                    base.cols.get().clamp(1, (cols - left).max(1)),
747                    base.rows.get().clamp(1, (rows - top).max(1)),
748                )
749            })
750            .collect()
751    }
752
753    /// The layout fitted to an area `cols` by `rows`: every window placed
754    /// as it is drawn there, and that size taken as what it is arranged for.
755    pub fn fit(&mut self, (cols, rows): (u16, u16)) {
756        let placed = self.fitted((cols, rows));
757        for (window, rect) in self.windows.iter_mut().zip(placed) {
758            window.place(rect);
759        }
760        self.cols = cols;
761        self.rows = rows;
762    }
763
764    /// The layout a character starts from, Vellum's sidebar, whatever the
765    /// size: it is drawn scaled to the terminal.
766    #[must_use]
767    pub fn default_for(_size: (u16, u16)) -> Self {
768        Self::parse(SIDEBAR).unwrap_or_else(|_| Self {
769            cols: 120,
770            rows: 40,
771            windows: vec![Window::new(
772                (0, 0),
773                (40, 120),
774                &[Widget::Story],
775                Border::Plain,
776            )],
777            kept: Kept::default(),
778        })
779    }
780
781    /// Read from `text`, in Vellum's layout format or the earlier form
782    /// Hydra wrote.
783    ///
784    /// # Errors
785    ///
786    /// It is not a layout.
787    pub fn parse(text: &str) -> Result<Self, String> {
788        let table: toml::Table = toml::from_str(text).map_err(|why| why.to_string())?;
789        if table.contains_key("window") {
790            return earlier::read(text);
791        }
792        let mut kept = Kept {
793            base_layout: str_of(&table, "base_layout"),
794            theme: str_of(&table, "theme"),
795            shared_edges: table
796                .get("shared_edges")
797                .and_then(toml::Value::as_bool)
798                .unwrap_or(false),
799            ..Kept::default()
800        };
801        if let Some(deleted) = table.get("deleted_windows") {
802            kept.deleted_windows = deleted
803                .clone()
804                .try_into()
805                .map_err(|why: toml::de::Error| why.to_string())?;
806        }
807        let mut windows = Vec::new();
808        for each in table
809            .get("windows")
810            .and_then(toml::Value::as_array)
811            .into_iter()
812            .flatten()
813        {
814            match each.clone().try_into::<WindowDef>() {
815                Ok(def) => windows.push(Window::of(def)),
816                Err(_) => kept.unknown_windows.push(each.clone()),
817            }
818        }
819        let number = |key: &str| {
820            table
821                .get(key)
822                .and_then(toml::Value::as_integer)
823                .and_then(|n| u16::try_from(n).ok())
824        };
825        // With no size said, the size its windows reach.
826        let reach = |edge: fn(&Window) -> u16| windows.iter().map(edge).max().unwrap_or(0);
827        let cols = number("terminal_width")
828            .unwrap_or_else(|| reach(|window| window.rect().right()))
829            .max(1);
830        let rows = number("terminal_height")
831            .unwrap_or_else(|| reach(|window| window.rect().bottom()))
832            .max(1);
833        Ok(Self {
834            cols,
835            rows,
836            windows,
837            kept,
838        })
839    }
840
841    /// As its file holds it, in Vellum's layout format.
842    #[must_use]
843    pub fn to_toml(&self) -> String {
844        let file = File {
845            windows: self.windows.iter().map(|window| &window.def).collect(),
846            terminal_width: self.cols,
847            terminal_height: self.rows,
848            base_layout: self.kept.base_layout.as_deref(),
849            theme: self.kept.theme.as_deref(),
850            deleted_windows: self.kept.deleted_windows.iter().collect(),
851            shared_edges: self.kept.shared_edges,
852        };
853        let Ok(mut value) = toml::Value::try_from(&file) else {
854            return String::new();
855        };
856        if !self.kept.unknown_windows.is_empty()
857            && let Some(windows) = value
858                .as_table_mut()
859                .and_then(|table| table.get_mut("windows"))
860                .and_then(toml::Value::as_array_mut)
861        {
862            windows.extend(self.kept.unknown_windows.iter().cloned());
863        }
864        toml::to_string(&value).unwrap_or_default()
865    }
866}

868/// Where `cell` of a custom window goes, the window's inside `arranged` as
869/// the layout has it and `inside` as drawn: from the inside's top left, a
870/// cell that reaches the inside's right or bottom edge and grows
871/// (`resize::stretches`) kept to it, and none outside it.
872fn cell_placed(cell: &Window, arranged: Rect, inside: Rect) -> Rect {
873    let at = cell.rect();
874    let grows = resize::stretches(&cell.def);
875    let kept = |start: u16, size: u16, was: u16, now: u16| {
876        if grows && start.saturating_add(size) >= was {
877            now.saturating_sub(start).max(1)
878        } else {
879            size
880        }
881    };
882    Rect::new(
883        inside.x.saturating_add(at.x),
884        inside.y.saturating_add(at.y),
885        kept(at.x, at.width, arranged.width, inside.width),
886        kept(at.y, at.height, arranged.height, inside.height),
887    )
888    .intersection(inside)
889}

891/// A text field of `table`, when it has one.
892fn str_of(table: &toml::Table, key: &str) -> Option<String> {
893    table
894        .get(key)
895        .and_then(toml::Value::as_str)
896        .map(str::to_owned)
897}

899/// The form Hydra wrote its terminal layouts in before it took Vellum's,
900/// read once and written in Vellum's from then on.
901mod earlier {
902    use hydra_ui::widget::Widget;
903    use ratatui::layout::Rect;
904    use serde::Deserialize;
905
906    use super::{Border, Kept, Layout, Window, base, def_of, set_border};
907
908    /// A window as Hydra wrote it.
909    #[derive(Deserialize)]
910    struct Was {
911        row: u16,
912        col: u16,
913        rows: u16,
914        cols: u16,
915        tabs: Vec<Widget>,
916        #[serde(default)]
917        showing: usize,
918        #[serde(default)]
919        title: Option<String>,
920        #[serde(default)]
921        border: Border,
922        #[serde(default)]
923        follows: Option<String>,
924    }
925
926    /// A layout as Hydra wrote it.
927    #[derive(Deserialize)]
928    struct Earlier {
929        cols: u16,
930        rows: u16,
931        #[serde(default)]
932        window: Vec<Was>,
933    }
934
935    /// The layout in `text`, in the earlier form: each window kept, its
936    /// text tabs one window of tabs and any other its first tab.
937    pub(super) fn read(text: &str) -> Result<Layout, String> {
938        let earlier: Earlier = toml::from_str(text).map_err(|why| why.to_string())?;
939        let windows = earlier
940            .window
941            .into_iter()
942            .map(|was| {
943                let name = was
944                    .tabs
945                    .first()
946                    .map_or_else(|| "window".to_owned(), Widget::type_key);
947                let mut base = base(&name, Rect::new(was.col, was.row, was.cols, was.rows));
948                set_border(&mut base, was.border);
949                base.title = was.title;
950                base.follows = was.follows;
951                let mut window = Window::of(def_of(&was.tabs, base));
952                window.showing = was.showing.min(window.tabs.len().saturating_sub(1));
953                window
954            })
955            .collect();
956        Ok(Layout {
957            cols: earlier.cols,
958            rows: earlier.rows,
959            windows,
960            kept: Kept::default(),
961        })
962    }
963}

978/// The file of the layout named `name`, under the data folder `data`: its
979/// name in lower case, letters, digits and dashes only.
980#[must_use]
981pub fn named_path(data: &Path, name: &str) -> PathBuf {
982    let clean: String = name
983        .chars()
984        .map(|c| {
985            if c.is_ascii_alphanumeric() {
986                c.to_ascii_lowercase()
987            } else {
988                '-'
989            }
990        })
991        .collect();
992    named_dir(data).join(format!("{}.toml", clean.trim_matches('-')))
993}

995/// The names of the layouts kept by name, in order.
996#[must_use]
997pub fn named(data: &Path) -> Vec<String> {
998    let mut names: Vec<String> = std::fs::read_dir(named_dir(data))
999        .map(|entries| {
1000            entries
1001                .filter_map(Result::ok)
1002                .map(|entry| entry.path())
1003                .filter(|path| path.extension().is_some_and(|ext| ext == "toml"))
1004                .filter_map(|path| Some(path.file_stem()?.to_string_lossy().into_owned()))
1005                .collect()
1006        })
1007        .unwrap_or_default();
1008    names.sort();
1009    names
1010}

1012/// The layout in the file at `path`: `None` for no file.
1013///
1014/// # Errors
1015///
1016/// Why it does not read, the file kept aside so it is not written over.
1017pub fn load(path: &Path) -> Result<Option<Layout>, String> {
1018    let text = match std::fs::read_to_string(path) {
1019        Ok(text) => text,
1020        Err(why) if why.kind() == std::io::ErrorKind::NotFound => return Ok(None),
1021        Err(why) => return Err(format!("{} cannot be read: {why}", path.display())),
1022    };
1023    match Layout::parse(&text) {
1024        Ok(layout) => Ok(Some(layout)),
1025        Err(why) => {
1026            let kept = hydra_board::kept::keep_unread(path, |text| Layout::parse(text).is_ok());
1027            Err(match kept {
1028                Ok(()) => format!(
1029                    "{} does not read, and is kept as {}: {why}",
1030                    path.display(),
1031                    hydra_board::kept::unread(path).display()
1032                ),
1033                Err(moved) => format!("{} does not read ({why}); {moved}", path.display()),
1034            })
1035        }
1036    }
1037}

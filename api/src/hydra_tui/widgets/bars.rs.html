1//! The bars of buttons: the game's quick bars, and the player's hotbars.
2//!
3//! **Quick.** The game sends its bars as dialogs of links -- `quick`, the
4//! main one, `quick-combat`, `quick-simu` -- and a Lich script opens its
5//! own at `location='quickBar'`; the game says which shows. The pages are
6//! named along the top, the one shown marked: a press on another asks the
7//! game to switch, `_qlink change <id>`, as Wrayth does, and a script
8//! listening for it answers for its own. The page shown draws its labels,
9//! separators and links: a game link sends its command once checked, a
10//! script's as typed, a `url:` one opens that page of play.net.
11//!
12//! **Hotbar.** The character's hotbars, as its files have them: every
13//! character's `hotbars.toml` in the data folder, its own beside its layout
14//! over it. Each button in its slot of the bar's grid, its face as its
15//! state now has it -- label, colours, dimmed, the seconds it counts down
16//! -- and a press sends what that state chooses, as if typed. With more
17//! than one bar, their names along the top choose which shows.

36/// The quick bar: its pages along the top, then the page shown.
37pub(super) fn quick(seen: &Seen<'_>) -> Vec<Row> {
38    let colors = seen.colors;
39    let Some(state) = seen.state() else {
40        return vec![note("Not sent yet", colors)];
41    };
42    let look = QuickLook::default();
43    let shown = quick::showing(Some(state), &look);
44    let pages = quick::pages(Some(state), &look);
45    let mut rows = Vec::new();
46    if pages.is_empty() {
47        rows.push(note("No pages yet", colors));
48    } else {
49        let mut top: Row = Vec::new();
50        for page in &pages {
51            let text = format!(" {} ", page.name);
52            top.push(if page.key == shown {
53                word(text, colors.chosen())
54            } else if seen.own {
55                button(
56                    text,
57                    colors.muted(),
58                    format!("quiet:_qlink change {}", page.key),
59                )
60            } else {
61                word(text, colors.muted())
62            });
63        }
64        rows.push(top);
65    }
66    rows.push(parts(state, &shown, seen));
67    rows
68}

70/// The bar `id`'s parts, the game's or a script's: its labels, its
71/// separators and their words, and its links.
72fn parts(state: &GameState, id: &str, seen: &Seen<'_>) -> Row {
73    let colors = seen.colors;
74    let Some(dialog) = state.dialogs.get(id) else {
75        return note("Not sent yet", colors);
76    };
77    let mut row: Row = Vec::new();
78    let put = |bit: Bit, row: &mut Row| {
79        if !row.is_empty() {
80            row.push(word(" ", colors.canvas()));
81        }
82        row.push(bit);
83    };
84    let style = colors.text(Token::Link);
85    for part in quick::parts(dialog) {
86        let bit = match part {
87            BarPart::Label(words) => word(words, colors.canvas()),
88            BarPart::Separator(words) => {
89                word(words.unwrap_or_else(|| "│".to_owned()), colors.muted())
90            }
91            BarPart::Link {
92                words,
93                does: BarLink::Typed(command),
94            } => {
95                if seen.own {
96                    button(words, style, format!("typed:{command}"))
97                } else {
98                    word(words, style)
99                }
100            }
101            BarPart::Link {
102                words,
103                does: BarLink::Url(href),
104            } => link(words, style, Some(RunLink::Url { href })),
105            BarPart::Link {
106                words,
107                does: BarLink::Game(command),
108            } => link(words, style, Some(RunLink::Command { command })),
109        };
110        put(bit, &mut row);
111    }
112    if row.is_empty() {
113        return note("Nothing on it", colors);
114    }
115    row
116}

118/// A hotbars file read off the drawing thread and kept until it changes:
119/// none before the first read; why it does not read, when it does not.
120fn read(path: &Path) -> Result<HotbarFile, String> {
121    let key = format!("tui-hotbars:{}", path.display());
122    hydra_board::disk::Disk::hydras()
123        .kept(&key, path, None, |path| Some(HotbarFile::load(path)))
124        .map_or_else(|| Ok(HotbarFile::default()), |(_, read)| read)
125}

127/// The character's macros, everyone's with its own over them, kept read
128/// as the hotbars are: what a button that names one looks like.
129fn macros(data: &Path, instance: Option<&str>, name: &str) -> MacroFile {
130    let read = |path: &Path| {
131        let key = format!("tui-macros:{}", path.display());
132        hydra_board::disk::Disk::hydras()
133            .kept(&key, path, None, |path| Some(MacroFile::load(path)))
134            .and_then(|(_, read)| read.ok())
135            .unwrap_or_default()
136    };
137    let [every, own] = hydra_board::macros::macros_paths(data, instance, name);
138    read(&every).merged(read(&own))
139}

141/// The character's hotbars: every character's in `data`, its own over
142/// them; and why either file does not read.
143fn hotbars(data: &Path, instance: Option<&str>, name: &str) -> (HotbarFile, Vec<String>) {
144    let [every, own]: [PathBuf; 2] = hydra_board::hotbars_paths(data, instance, name);
145    let mut problems = Vec::new();
146    let mut take = |read: Result<HotbarFile, String>| {
147        read.unwrap_or_else(|why| {
148            problems.push(why);
149            HotbarFile::default()
150        })
151    };
152    let every = take(read(&every));
153    let own = take(read(&own));
154    (every.merged(own), problems)
155}

157/// The Hotbar widget: the bars to choose from, when more than one; then
158/// the bar shown, its grid a row of buttons a row of slots.
159pub(super) fn hotbar(seen: &Seen<'_>, state: &State, width: u16) -> Vec<Row> {
160    let colors = seen.colors;
161    let Some((seat, data)) = seen.seated else {
162        return vec![note(
163            "No hotbars yet: hotbars.toml in the data folder holds them.",
164            colors,
165        )];
166    };
167    let (file, problems) = hotbars(data, seat.instance.as_deref(), &seat.name);
168    let mut rows: Vec<Row> = problems
169        .into_iter()
170        .map(|why| vec![word(why, colors.text(Token::Warning))])
171        .collect();
172    if file.bars.is_empty() {
173        rows.push(note(
174            format!("No hotbars yet: {FILE} in the data folder holds them."),
175            colors,
176        ));
177        return rows;
178    }
179    let page = state.page.min(file.bars.len() - 1);
180    if file.bars.len() > 1 {
181        let names: Row = file
182            .bars
183            .iter()
184            .enumerate()
185            .map(|(at, bar)| {
186                let name = format!(" {} ", bar.title.as_deref().unwrap_or(&bar.name));
187                if at == page {
188                    word(name, colors.chosen())
189                } else {
190                    button(name, colors.muted(), format!("page:{at}"))
191                }
192            })
193            .collect();
194        rows.push(names);
195    }
196    let bar = &file.bars[page];
197    let made = macros(data, seat.instance.as_deref(), &seat.name);
198    let unknown = GameState::default();
199    let none = Flags::default();
200    let world = seen
201        .own
202        .then(|| seen.story.world())
203        .flatten()
204        .unwrap_or_else(|| World {
205            state: seen.state().unwrap_or(&unknown),
206            flags: &none,
207            statuses: &none,
208            unix: std::time::SystemTime::now()
209                .duration_since(std::time::UNIX_EPOCH)
210                .map_or(0, |since| {
211                    i64::try_from(since.as_secs()).unwrap_or(i64::MAX)
212                }),
213        });
214    let grid = bar.grid();
215    let columns = usize::from(grid.columns.max(1));
216    let cell = (usize::from(width) / columns).clamp(4, 24);
217    for slots in grid.slots.chunks(columns) {
218        let mut row: Row = Vec::new();
219        for slot in slots {
220            let Some(place) = slot else {
221                row.push(word(" ".repeat(cell), colors.canvas()));
222                continue;
223            };
224            let button_of = &bar.buttons[*place];
225            let look = hydra_ui::macros::button_look(button_of, &world, &made);
226            // The seconds after the label, in the timer's colour.
227            let counted = look.countdown.map(|seconds| format!(" {seconds}"));
228            let room = cell
229                .saturating_sub(1)
230                .saturating_sub(counted.as_ref().map_or(0, String::len));
231            let face = fitted(&look.label, room);
232            let style = face_style(&look, colors);
233            row.push(if seen.own {
234                button(
235                    face,
236                    style,
237                    format!("hotbar:{}{HOTBAR_APART}{}", bar.name, button_of.id),
238                )
239            } else {
240                word(face, style)
241            });
242            if let Some(counted) = counted {
243                let fg = look
244                    .countdown_colour
245                    .as_deref()
246                    .and_then(parse_hex)
247                    .map_or_else(|| colors.color(Token::Warning), rgb);
248                row.push(word(counted, style.fg(fg)));
249            }
250            row.push(word(" ", colors.canvas()));
251        }
252        rows.push(row);
253    }
254    rows
255}

257/// A button's face as its state has it: its colours, or the raised
258/// surface's; dimmed, muted.
259fn face_style(look: &hydra_ui::hotbar::Look, colors: &Colors) -> Style {
260    let colour = |hex: Option<&String>| hex.and_then(|hex| parse_hex(hex)).map(rgb);
261    let fg = colour(look.fg.as_ref()).unwrap_or_else(|| colors.color(Token::Text));
262    let bg = colour(look.bg.as_ref()).unwrap_or_else(|| colors.color(Token::Raised));
263    if look.dim {
264        Style::new().fg(colors.color(Token::Muted)).bg(bg)
265    } else {
266        Style::new().fg(fg).bg(bg)
267    }
268}

270/// `label` centred in `width` columns, cut when it is wider.
271fn fitted(label: &str, width: usize) -> String {
272    let label: String = label.chars().take(width).collect();
273    let used = unicode_width::UnicodeWidthStr::width(label.as_str()).min(width);
274    let before = (width - used) / 2;
275    format!(
276        "{}{label}{}",
277        " ".repeat(before),
278        " ".repeat(width - used - before)
279    )
280}

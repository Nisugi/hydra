1//! The minimap: the area the character is in, laid out (`hydra_map`), drawn
2//! in braille dots -- each cell of the terminal two dots across and four
3//! down -- and centred on the character's room, following it.
4//!
5//! The rooms are dots, the lines between them the ways out (a way with no
6//! direction dashed), the character's room a block in its own colour, and
7//! the way travel shows (`.route2`) drawn over them in the route's colour.
8//! The maps next door are drawn faint beside it. `+` and `-` along the top
9//! zoom in and out; a building's rooms are drawn at twice the streets'
10//! scale, as they are laid out at half it. Aboard a ship it says where the
11//! ship is, its heading and its wheel.
12//!
13//! The map is laid out once for the process (`hydra_map::laid`), the
14//! windows and every terminal drawing from the same, the first time a
15//! minimap asks; until then the widget says it is loading.
16

31/// Dots a cell of the map's sheet outdoors, at the zoom of its own.
32const SCALE: f32 = 1.5;
33
34/// How much each step of zoom multiplies the scale by.
35const STEP: f32 = 1.5;
36
37/// The least and the most steps of zoom from its own.
38const ZOOMS: (i8, i8) = (-3, 4);
39
40/// What a dot is, by what is drawn over what: the faintest first.
41#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
42enum Rank {
43    /// A map next door's.
44    NextDoor,
45    /// A way out.
46    Line,
47    /// A way out with no direction.
48    Connector,
49    /// A room.
50    Room,
51    /// The way travel shows.
52    Route,
53    /// The character's room.
54    You,
55}
56
57impl Rank {
58    /// The palette's token it is drawn in.
59    const fn token(self) -> Token {
60        match self {
61            Self::NextDoor => Token::MapMuted,
62            Self::Line => Token::MapLine,
63            Self::Connector => Token::MapConnector,
64            Self::Room => Token::MapRoom,
65            Self::Route => Token::MapRoute,
66            Self::You => Token::MapYou,
67        }
68    }
69}

71/// The step of zoom after `zoom`, in or out, within its bounds.
72pub(super) fn zoomed(zoom: i8, inward: bool) -> i8 {
73    let next = if inward { zoom + 1 } else { zoom - 1 };
74    next.clamp(ZOOMS.0, ZOOMS.1)
75}

77/// Draw the minimap of the character `seen` draws, followed by its
78/// follower in `followers`, zoomed as `state` keeps it; the map laid out
79/// held in `map` while the screen draws from it, asked for again each time
80/// so a map laid out again is drawn from at once.
81pub(super) fn draw(
82    buf: &mut Buffer,
83    area: Rect,
84    (seen, hits): (&Seen<'_>, &mut Hits),
85    (state, followers, map): (
86        &mut State,
87        &mut HashMap<String, Follower>,
88        &mut Option<Arc<Laid>>,
89    ),
90) {
91    let colors = seen.colors;
92    let Some((_, data)) = seen.seated else {
93        return line(buf, area, &[Piece::new("No map here.", colors.muted())]);
94    };
95    let key = seen.key().unwrap_or_default();
96    let laid = map.insert(hydra_map::laid(data));
97    let view = laid.view(
98        followers.entry(key.clone()).or_default(),
99        seen.snapshot.map(|snapshot| &snapshot.state),
100    );
101    drawn(
102        buf,
103        area,
104        (&with_route(view, &key), state.zoom),
105        (seen, hits),
106    );
107}

109/// `view` with the way travel shows on it, or the course.
110fn with_route(view: MinimapView, key: &str) -> MinimapView {
111    match view {
112        MinimapView::Here {
113            scene,
114            room,
115            next_door,
116            ..
117        } => MinimapView::Here {
118            scene,
119            room,
120            next_door,
121            route: hydra_map::shown_route(key),
122        },
123        MinimapView::Ocean {
124            sailing,
125            heading,
126            wheel,
127            deck,
128            ..
129        } => MinimapView::Ocean {
130            sailing,
131            heading,
132            wheel,
133            deck,
134            course: hydra_map::shown_course(key),
135        },
136        waiting @ MinimapView::Waiting(_) => waiting,
137    }
138}

140/// Draw `view` in `area`, zoomed `zoom` steps from its own; the zoom's
141/// buttons along the top.
142fn drawn(
143    buf: &mut Buffer,
144    area: Rect,
145    (view, zoom): (&MinimapView, i8),
146    (seen, hits): (&Seen<'_>, &mut Hits),
147) {
148    let colors = seen.colors;
149    match view {
150        MinimapView::Waiting(why) => {
151            let mut state = State::default();
152            super::rows(
153                buf,
154                area,
155                &[super::note(why.clone(), colors)],
156                (seen, hits),
157                &mut state,
158            );
159        }
160        MinimapView::Ocean {
161            sailing,
162            heading,
163            wheel,
164            course,
165            ..
166        } => {
167            let said = at_sea(*sailing, *heading, wheel.as_deref(), course.as_deref());
168            let rows: Vec<super::Row> = said
169                .into_iter()
170                .map(|text| vec![super::word(text, colors.canvas())])
171                .collect();
172            let mut state = State::default();
173            super::rows(buf, area, &rows, (seen, hits), &mut state);
174        }
175        MinimapView::Here {
176            scene,
177            room,
178            next_door,
179            route,
180        } => {
181            let inside = scene
182                .room(*room)
183                .is_some_and(|here| here.building.is_some());
184            let mut scale = SCALE * STEP.powi(i32::from(zoom));
185            if inside {
186                scale *= 2.0;
187            }
188            let mut dots = Dots::new(area.width, area.height);
189            let centre = scene.room(*room).map_or((0.0, 0.0), |here| cell(here.cell));
190            let middle = (f32::from(area.width), f32::from(area.height) * 2.0);
191            let place = |(x, y): (f32, f32)| {
192                (
193                    (x - centre.0).mul_add(scale, middle.0),
194                    (y - centre.1).mul_add(scale, middle.1),
195                )
196            };
197            for door in next_door.iter().filter(|door| door.ring == 1) {
198                let moved = |(x, y): (f32, f32)| place((x + door.offset.0, y + door.offset.1));
199                sheet(&mut dots, &door.scene, &moved, (scale, true));
200            }
201            sheet(&mut dots, scene, &place, (scale, false));
202            if let Some(route) = route {
203                let rooms: Vec<(f32, f32)> = route
204                    .from(*room)
205                    .iter()
206                    .filter_map(|id| scene.room(*id))
207                    .map(|on| place(cell(on.cell)))
208                    .collect();
209                for pair in rooms.windows(2) {
210                    dots.line(pair[0], pair[1], (Rank::Route, false));
211                }
212            }
213            let you = place(centre);
214            dots.block(you, 1, Rank::You);
215            dots.draw(buf, area, colors);
216            title_and_zoom(
217                buf,
218                area,
219                (scene.room(*room).map(|here| here.title.as_str()), colors),
220                hits,
221            );
222        }
223    }
224}

226/// The room's title along the top, and the zoom's buttons at its right.
227fn title_and_zoom(
228    buf: &mut Buffer,
229    area: Rect,
230    (title, colors): (Option<&str>, &Colors),
231    hits: &mut Hits,
232) {
233    if area.height < 3 || area.width < 8 {
234        return;
235    }
236    let style = colors.muted();
237    let zoom = " - + ";
238    let right = area.x + area.width - 5;
239    if let Some(title) = title {
240        buf.set_stringn(area.x, area.y, title, usize::from(area.width - 6), style);
241    }
242    buf.set_string(right, area.y, zoom, colors.text(Token::Accent));
243    hits.add(
244        Rect::new(right, area.y, 2, 1),
245        Target::Button(format!("{PRESS}zoom:out")),
246    );
247    hits.add(
248        Rect::new(right + 2, area.y, 3, 1),
249        Target::Button(format!("{PRESS}zoom:in")),
250    );
251}

253/// `scene`'s lines and rooms as dots, each point placed by `place`, `faint`
254/// for a map next door.
255fn sheet(
256    dots: &mut Dots,
257    scene: &MapScene,
258    place: &dyn Fn((f32, f32)) -> (f32, f32),
259    (scale, faint): (f32, bool),
260) {
261    for edge in &scene.edges {
262        let (rank, dashed) = match (faint, edge.kind) {
263            (false, EdgeKind::Directional) => (Rank::Line, false),
264            (false, EdgeKind::Connector) => (Rank::Connector, true),
265            (true, _) | (false, EdgeKind::Stub) => (Rank::NextDoor, false),
266        };
267        for pair in edge.path.windows(2) {
268            dots.line(place(pair[0]), place(pair[1]), (rank, dashed));
269        }
270    }
271    let rank = if faint { Rank::NextDoor } else { Rank::Room };
272    let size = u8::from(scale >= 2.5);
273    for room in &scene.rooms {
274        dots.block(place(cell(room.cell)), size, rank);
275    }
276}

278/// What the minimap says aboard a ship: where it is, its heading, its
279/// wheel, and the course shown.
280fn at_sea(
281    sailing: Sailing,
282    heading: Option<hydra_map::Heading>,
283    wheel: Option<&str>,
284    course: Option<&hydra_map::ShownCourse>,
285) -> Vec<String> {
286    let port_named = |uid: i64| {
287        hydra_map::ocean()
288            .as_ref()
289            .ok()
290            .and_then(|ocean| ocean.port_at(uid))
291            .map(|port| port.name.clone())
292    };
293    let mut said = vec![match sailing {
294        Sailing::Ashore => "Ashore.".to_owned(),
295        Sailing::Docked { port } => match port.and_then(port_named) {
296            Some(name) => format!("Docked at {name}."),
297            None => "Docked.".to_owned(),
298        },
299        Sailing::At { room, stated } => {
300            let near = port_named(room).map_or_else(String::new, |name| format!(", off {name}"));
301            let how = if stated { "" } else { ", reckoned" };
302            format!("At sea: room {room}{near}{how}.")
303        }
304        Sailing::Lost => "At sea, and nothing says where.".to_owned(),
305    }];
306    if let Some(heading) = heading {
307        said.push(format!(
308            "Heading {}.",
309            format!("{heading:?}").to_lowercase()
310        ));
311    }
312    if let Some(wheel) = wheel {
313        said.push(format!("The wheel is turned {wheel}."));
314    }
315    if let Some(course) = course {
316        let left = match sailing {
317            Sailing::At { room, .. } => course.from(room).len(),
318            _ => course.rooms.len(),
319        };
320        said.push(format!(
321            "Course for {}: {} room{} to go.",
322            course.port,
323            left.saturating_sub(1),
324            if left == 2 { "" } else { "s" }
325        ));
326    }
327    said
328}

330/// A picture in braille dots: each cell two dots across and four down,
331/// each dot what is drawn there, the cell in the colour of its highest.
332#[derive(Debug)]
333pub(crate) struct Dots {
334    /// Dots across and down.
335    width: usize,
336    height: usize,
337    /// Each dot, by row: what is drawn there, if anything.
338    drawn: Vec<Option<Rank>>,
339}
340
341/// Each dot's bit in its cell's braille character, by its column and row.
342const BITS: [[u8; 4]; 2] = [[0x01, 0x02, 0x04, 0x40], [0x08, 0x10, 0x20, 0x80]];
343
344impl Dots {
345    /// A picture filling `columns` by `rows` cells.
346    fn new(columns: u16, rows: u16) -> Self {
347        let (width, height) = (usize::from(columns) * 2, usize::from(rows) * 4);
348        Self {
349            width,
350            height,
351            drawn: vec![None; width * height],
352        }
353    }
354
355    /// The dot at `x`, `y` drawn as `rank`, unless something above it is
356    /// drawn there already; off the picture, nothing.
357    fn put(&mut self, x: i64, y: i64, rank: Rank) {
358        let (Ok(x), Ok(y)) = (usize::try_from(x), usize::try_from(y)) else {
359            return;
360        };
361        if x >= self.width || y >= self.height {
362            return;
363        }
364        let dot = &mut self.drawn[y * self.width + x];
365        if dot.is_none_or(|was| was < rank) {
366            *dot = Some(rank);
367        }
368    }
369
370    /// A line from `from` to `to`, a dot at each step along it; every other
371    /// dot left out when `dashed`.
372    fn line(&mut self, from: (f32, f32), to: (f32, f32), (rank, dashed): (Rank, bool)) {
373        let (dx, dy) = (to.0 - from.0, to.1 - from.1);
374        let steps = dx.abs().max(dy.abs()).ceil().clamp(1.0, 4096.0);
375        #[allow(
376            clippy::cast_possible_truncation,
377            clippy::cast_sign_loss,
378            reason = "clamped to 1-4096 above"
379        )]
380        let count = steps as u32;
381        for at in 0..=count {
382            if dashed && at % 2 == 1 {
383                continue;
384            }
385            #[allow(clippy::cast_precision_loss, reason = "a step of at most 4096")]
386            let along = at as f32 / steps;
387            self.put(
388                round(dx.mul_add(along, from.0)),
389                round(dy.mul_add(along, from.1)),
390                rank,
391            );
392        }
393    }
394
395    /// A square of dots about `at`, `size` dots out each way: a dot alone
396    /// at size 0.
397    fn block(&mut self, at: (f32, f32), size: u8, rank: Rank) {
398        let (x, y) = (round(at.0), round(at.1));
399        let size = i64::from(size);
400        for row in y - size..=y + size {
401            for col in x - size..=x + size {
402                self.put(col, row, rank);
403            }
404        }
405    }
406
407    /// The picture drawn into `area`: each cell's braille character in the
408    /// colour of the highest drawn in it; a cell with nothing drawn left as
409    /// it is.
410    fn draw(&self, buf: &mut Buffer, area: Rect, colors: &Colors) {
411        let ground = colors.color(Token::Canvas);
412        for row in 0..usize::from(area.height) {
413            for col in 0..usize::from(area.width) {
414                let mut bits = 0u8;
415                let mut top: Option<Rank> = None;
416                for (dx, column) in BITS.iter().enumerate() {
417                    for (dy, bit) in column.iter().enumerate() {
418                        let (x, y) = (col * 2 + dx, row * 4 + dy);
419                        if let Some(rank) = self.drawn.get(y * self.width + x).copied().flatten() {
420                            bits |= bit;
421                            top = top.max(Some(rank));
422                        }
423                    }
424                }
425                let Some(top) = top else {
426                    continue;
427                };
428                let glyph = char::from_u32(0x2800 + u32::from(bits)).unwrap_or(' ');
429                let (x, y) = (
430                    area.x + u16::try_from(col).unwrap_or(0),
431                    area.y + u16::try_from(row).unwrap_or(0),
432                );
433                let colour: Color = colors.color(top.token());
434                buf[(x, y)]
435                    .set_char(glyph)
436                    .set_style(Style::new().fg(colour).bg(ground));
437            }
438        }
439    }
440}

442/// A cell of the map's sheet as a point.
443#[allow(
444    clippy::cast_precision_loss,
445    reason = "a sheet's cells are a few thousand at most"
446)]
447fn cell((x, y): (i32, i32)) -> (f32, f32) {
448    (x as f32, y as f32)
449}
450
451/// `value` to the nearest whole dot.
452fn round(value: f32) -> i64 {
453    #[allow(
454        clippy::cast_possible_truncation,
455        reason = "a dot's place, far inside an i64 once clamped"
456    )]
457    let whole = value.round().clamp(-1.0e9, 1.0e9) as i64;
458    whole
459}

461/// The braille characters a line drawn from `from` to `to`, in dots, makes
462/// in a picture `columns` by `rows` cells: each row of cells as text.
463#[cfg(test)]
464fn braille_line(from: (f32, f32), to: (f32, f32), (columns, rows): (u16, u16)) -> Vec<String> {
465    let mut dots = Dots::new(columns, rows);
466    dots.line(from, to, (Rank::Line, false));
467    let area = Rect::new(0, 0, columns, rows);
468    let mut buf = Buffer::empty(area);
469    dots.draw(&mut buf, area, &Colors::default());
470    (0..rows)
471        .map(|y| {
472            (0..columns)
473                .map(|x| buf[(x, y)].symbol().to_owned())
474                .collect::<String>()
475        })
476        .collect()
477}

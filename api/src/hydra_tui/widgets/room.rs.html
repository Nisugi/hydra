27/// How the character's creatures' tags look while `.targetid` shows them,
28/// as its settings say; `None` while they do not.
29pub(super) fn tags(seen: &Seen<'_>, looks: &mut hydra_board::looks::Looks) -> Option<Tags> {
30    let (seat, data) = seen.seated?;
31    let instance = seat.instance.as_deref()?;
32    let who = (instance, seat.name.as_str());
33    let _ = looks.sorting_now(data, who);
34    looks.tags(data, who)
35}

38/// creature's colour, a link the game's.
39fn runs_row(runs: &[StyledRun], colors: &Colors) -> Row {
40    super::of_line(runs_line(runs, colors))

43/// The room as the game describes it: `[name] (number)`, its description
44/// with what is here run on after it, who else is here, the ways out.
45pub(super) fn room(seen: &Seen<'_>) -> Vec<Row> {
46    let colors = seen.colors;
47    let Some(state) = seen.state() else {
48        return vec![note("Room unknown", colors)];
49    };
50    let room = &state.room;
51    let mut rows: Vec<Row> = seen.who.map(|who| note(who, colors)).into_iter().collect();
52    rows.push(vec![word(
53        said::heading(room),
54        colors.text(Token::RoomName).add_modifier(Modifier::BOLD),
55    )]);
56    let prose = said::prose(room, (true, true));
57    if !prose.is_empty() {
58        rows.push(runs_row(&prose, colors));
59    }
60    if !room.players.is_empty() {
61        let player = colors.palette().get(Token::Player);
62        rows.push(runs_row(&said::players(room, player), colors));
63    }
64    rows.push(ways_out(Some(room), colors));
65    rows
66}

68/// The ways out as the game gave them, or as the model holds them, each a
69/// link that goes there; that they are not known, when they are not.
70fn ways_out(room: Option<&Room>, colors: &Colors) -> Row {
71    match said::ways_out(room) {
72        WaysOut::Runs(runs) => runs_row(&runs, colors),
73        WaysOut::None => vec![word(said::NO_EXITS, colors.canvas())],
74        WaysOut::Unknown => note("Exits unknown", colors),
75    }
76}

79pub(super) fn title(buf: &mut Buffer, area: Rect, seen: &Seen<'_>) {
80    let title = seen
81        .state()
82        .and_then(|state| state.room.title.as_deref())
83        .unwrap_or("Room unknown");
84    line(

105/// The ways out on one line, after the name of the character followed.
106pub(super) fn exits(buf: &mut Buffer, area: Rect, (seen, hits): (&Seen<'_>, &mut Hits)) {
107    let colors = seen.colors;
108    let mut row: Row = seen
109        .who
110        .map(|who| word(format!("{who} "), colors.canvas()))
111        .into_iter()
112        .collect();
113    row.extend(ways_out(seen.state().map(|state| &state.room), colors));
114    super::draw_bits(buf, (area.x, area.y, area.width), &row, hits, seen.linked());
115}

117/// The compass: every way out as an arrow, lit and clickable where there is
118/// one; another character's only shows.
119pub(super) fn compass(buf: &mut Buffer, area: Rect, (seen, hits): (&Seen<'_>, &mut Hits)) {
120    let colors = seen.colors;
121    let exits: Vec<String> = seen
122        .state()
123        .and_then(|state| state.room.exits.clone())
124        .unwrap_or_default();
125    // The game names them short (`n`, `ne`): each said in full first.
126    let has = |command: &str| {
127        exits
128            .iter()
129            .any(|exit| hydra_ui::exit_word(exit).eq_ignore_ascii_case(command))
130    };
131    let rows: [&[(&str, &str, &str)]; 3] = [
132        &[
133            ("up", "↑", "up"),
134            ("nw", "↖", "northwest"),
135            ("n", "▲", "north"),
136            ("ne", "↗", "northeast"),
137        ],
138        &[
139            ("", " ", ""),
140            ("w", "◀", "west"),
141            ("out", "o", "out"),
142            ("e", "▶", "east"),
143        ],
144        &[
145            ("down", "↓", "down"),
146            ("sw", "↙", "southwest"),
147            ("s", "▼", "south"),
148            ("se", "↘", "southeast"),
149        ],
150    ];
151    let left = area.x + area.width.saturating_sub(8) / 2;
152    let top = area.y + area.height.saturating_sub(3) / 2;
153    for (y, row) in rows.iter().enumerate() {
154        let y = top + u16::try_from(y).unwrap_or(0);
155        if y >= area.y + area.height {
156            break;
157        }
158        let pieces: Vec<Piece> = row
159            .iter()
160            .flat_map(|(dir, glyph, command)| {
161                let lit = !dir.is_empty() && has(command);
162                let style = if lit {
163                    colors.text(Token::Link)
164                } else {
165                    colors.text(Token::Line)
166                };
167                let link = lit.then(|| RunLink::Command {
168                    command: (*command).to_owned(),
169                });
170                [
171                    Piece::linked(*glyph, style, link),
172                    Piece::new(" ", colors.canvas()),
173                ]
174            })
175            .collect();
176        draw_row(
177            buf,
178            (left, y, area.width.min(8)),
179            &pieces,
180            hits,
181            seen.linked(),
182        );
183    }
184}

186/// A thin bar of a creature's health, eight cells, in the colour its
187/// percent reads in.
188fn health_bar(percent: u32, colors: &Colors) -> Vec<super::Bit> {
189    let mut row = vec![word(" ", colors.canvas())];
190    row.extend(
191        super::bar_in(8, percent, colors.color(said::health_band(percent)), colors)
192            .into_iter()
193            .map(bit),
194    );
195    row
196}

198/// A link to the creature `id`, its `noun`.
199fn creature_link(id: i64, noun: Option<&str>) -> RunLink {
200    RunLink::Object {
201        exist: id.to_string(),
202        noun: noun.unwrap_or_default().to_owned(),
203        coord: None,
204    }

207/// The Targets list: the hostile creatures, each a link to it, the one
208/// targeted marked `►` in the accent; its tag, its statuses, its health as
209/// a percent and a thin bar when the game states it. A dead creature, a
210/// decoy and a bare appendage are left out.
211pub(super) fn targets(seen: &Seen<'_>, tags: Option<Tags>) -> Vec<Row> {
212    let colors = seen.colors;
213    let mut rows: Vec<Row> = seen.who.map(|who| note(who, colors)).into_iter().collect();
214    let listed = seen
215        .state()
216        .and_then(|state| said::targets(state, tags, &said::status_word));
217    let Some((listed, _)) = listed else {
218        rows.push(note("Targets unknown", colors));
219        return rows;
220    };
221    if listed.is_empty() {
222        rows.push(note("No targets", colors));
223    }
224    for target in listed {
225        let style = if target.current {
226            colors.text(Token::Accent).add_modifier(Modifier::BOLD)
227        } else {
228            colors.text(Token::Creature)
229        };
230        let mut row = Vec::new();
231        if target.current {
232            row.push(word("► ", style));
233        }
234        row.push(link(
235            target.name,
236            style,
237            Some(creature_link(target.id, target.noun.as_deref())),
238        ));
239        if let Some(tag) = target.tag {
240            row.push(word(format!(" ({tag})"), colors.muted()));
241        }
242        if !target.statuses.is_empty() {
243            row.push(word(
244                format!(" {}", target.statuses.join(", ")),
245                colors.muted(),
246            ));
247        }
248        if let Some(percent) = target.health {
249            row.push(word(format!(" {percent}%"), colors.muted()));
250            row.extend(health_bar(percent, colors));
251        }
252        rows.push(row);
253    }
254    rows
255}

257/// The room's creatures that were not hostile when first seen, or its
258/// objects: across, a comma between each.
259pub(super) fn listed(widget: &Widget, seen: &Seen<'_>, tags: Option<Tags>) -> Vec<Row> {
260    let colors = seen.colors;
261    let mut rows: Vec<Row> = seen.who.map(|who| note(who, colors)).into_iter().collect();
262    let Some(state) = seen
263        .state()
264        .filter(|state| state.room.component("room objs").is_some())
265    else {
266        rows.push(note("unknown", colors));
267        return rows;
268    };
269    let (items, token) = if *widget == Widget::Objects {
270        (state.room.objects.iter().collect(), Token::Object)
271    } else {
272        (said::npcs(state), Token::Creature)
273    };
274    if items.is_empty() {
275        rows.push(note("none", colors));
276        return rows;
277    }
278    let tags = (*widget == Widget::Npcs).then_some(tags).flatten();
279    let palette = colors.palette();
280    let mut across = Vec::new();
281    for (at, item) in items.into_iter().enumerate() {
282        if at > 0 {
283            across.push(StyledRun::plain(", "));
284        }
285        across.extend(said::item_runs(
286            item,
287            palette.get(token),
288            tags.map(|style| (state, style)),
289            Some(palette.get(Token::Muted)),
290        ));
291    }
292    rows.push(runs_row(&across, colors));
293    rows
294}

296/// The players here, one a line, each a link, what they are doing after
297/// it; one the room says is dead drawn faint.
298pub(super) fn players(seen: &Seen<'_>) -> Vec<Row> {
299    let colors = seen.colors;
300    let mut rows: Vec<Row> = seen.who.map(|who| note(who, colors)).into_iter().collect();
301    let Some(state) = seen.state().filter(|state| state.room.saw_players()) else {
302        rows.push(note("Players unknown", colors));
303        return rows;
304    };
305    if state.room.players.is_empty() {
306        rows.push(note("No players", colors));
307    }
308    for player in &state.room.players {
309        let style = if said::appears_dead(player) {
310            colors.muted()
311        } else {
312            colors.text(Token::Player)
313        };
314        let mut row = vec![link(player.text.clone(), style, Some(said::object(player)))];
315        if let Some(status) = &player.status {
316            row.push(word(format!(" {}", status.as_str()), colors.muted()));
317        }
318        rows.push(row);
319    }
320    rows
321}

323/// Who is fighting here: *FRIENDLY*, the character with its stance and its
324/// friends; *FOES*, the rest. Each a link to it, whether it is dead, its
325/// statuses, and its health as a thin bar when stated.
326pub(super) fn combat(seen: &Seen<'_>) -> Vec<Row> {
327    let colors = seen.colors;
328    let mut rows: Vec<Row> = seen.who.map(|who| note(who, colors)).into_iter().collect();
329    let Some(state) = seen.state() else {
330        rows.push(note("Combat unknown", colors));
331        return rows;
332    };
333    let (friends, foes): (Vec<said::Fighter>, Vec<said::Fighter>) = said::fighters(state)
334        .into_iter()
335        .partition(|fighter| fighter.friend);
336    rows.push(note("FRIENDLY", colors));
337    let mut you = vec![word("You", colors.text(Token::Player))];
338    if let Some(stance) = &state.character.stance {
339        you.push(word(format!(" {stance}"), colors.muted()));
340    }
341    rows.push(you);
342    for friend in &friends {
343        rows.push(fighter(friend, Token::Player, colors));
344    }
345    rows.push(note("FOES", colors));
346    if foes.is_empty() {
347        rows.push(note("No foes.", colors));
348    }
349    for foe in &foes {
350        rows.push(fighter(foe, Token::Creature, colors));
351    }
352    rows
353}

355/// One fighter, in `token`'s colour.
356fn fighter(fighter: &said::Fighter, token: Token, colors: &Colors) -> Row {
357    let mut row = vec![link(
358        fighter.name.clone(),
359        colors.text(token),
360        Some(creature_link(fighter.id, fighter.noun.as_deref())),
361    )];
362    if fighter.dead {
363        row.push(word(" dead", colors.muted()));
364    }
365    if !fighter.statuses.is_empty() {
366        row.push(word(
367            format!(" {}", fighter.statuses.join(", ")),
368            colors.muted(),
369        ));
370    }
371    if let Some(percent) = fighter.health {
372        row.extend(health_bar(percent, colors));
373    }
374    row
375}

383    /// The fixture's room as the game joins it; its creature a link.
384    #[test]
385    fn the_room_is_drawn_as_the_game_joins_it() {
386        let (story, snapshot) = (hydra_ui::fixture::story(), hydra_ui::fixture::snapshot());
387        let colors = Colors::default();
388        let seen = seen_of(&story, Some(&snapshot), &colors);
389        let (text, hits) = drawn(&Widget::Room, &seen, (40, 6));
390        assert_eq!(
391            &text[..4],
392            [
393                "[Rawknuckle's, Watering Hole]",
394                "You also see a kobold",
395                "Also here: Maravel",
396                "Obvious exits: north, out",
397            ]
398        );
399        let kobold = text[1].find("kobold").expect("drawn");
400        assert!(matches!(
401            hits.at(u16::try_from(kobold).expect("narrow"), 1),
402            Some(Target::Link { link: RunLink::Object { exist, .. }, .. }) if exist == "-6"
403        ));
404    }

406    /// The kobold is a target, hostile when first seen; it is no NPC; the
407    /// player is listed with the players; the fighting lists it a foe.
408    #[test]
409    fn who_is_in_the_room_is_listed_by_kind() {
410        let (story, snapshot) = (hydra_ui::fixture::story(), hydra_ui::fixture::snapshot());
411        let colors = Colors::default();
412        let seen = seen_of(&story, Some(&snapshot), &colors);
413        let (text, _) = drawn(&Widget::Creatures, &seen, (40, 3));
414        assert_eq!(text[0], "a kobold");
415        let (text, _) = drawn(&Widget::Npcs, &seen, (40, 3));
416        assert_eq!(text[0], "none");
417        let (text, _) = drawn(&Widget::Combat, &seen, (40, 6));
418        assert_eq!(text[0], "FRIENDLY");
419        assert_eq!(text[2], "FOES");
420        assert!(text[3].starts_with("kobold"), "{text:?}");
421        let (text, _) = drawn(&Widget::Exits, &seen, (40, 1));
422        assert_eq!(text[0], "Obvious exits: north, out");
423    }

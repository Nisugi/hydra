1//! The Bestiary: the creature codex browsed in a window, the same answers
2//! `.bestiary` gives (`hydra_session::bestiary`).
3//!
4//! *Ask* opens a question over the screen that takes what `.bestiary`
5//! takes -- a name, `here`, `level 29`, `area graveyard`, `family Bear`,
6//! `undead`, `search manti` -- and the answer is drawn under it as a linked
7//! table, as the story draws one. A link that asks the bestiary again -- a
8//! level, a family, a name -- is answered here, in the widget; a `[go2]` or
9//! `[map]` goes to the character's command line as a click in the story
10//! would. *Here* and *Undead* are a press each.
11
12use hydra_session::bestiary;
13use hydra_ui::RunLink;
14use hydra_ui::theme::Token;
15
16use super::{Bit, Row, Seen, State, bit, button, note, word};
17
18/// Draw the widget: its presses, and the answer to what was last asked;
19/// what was asked since it was drawn answered first.
20pub(super) fn bestiary(seen: &Seen<'_>, state: &mut State) -> Vec<Row> {
21    let colors = seen.colors;
22    if let Some(words) = state.take_typed() {
23        let game = seen.state();
24        let answer = match bestiary::asked(&words) {
25            Ok(asked) => {
26                let uid = game.and_then(bestiary::room_uid);
27                let profession =
28                    game.and_then(|game| game.character.identity.profession.as_deref());
29                bestiary::answered(asked, uid, profession)
30            }
31            Err(why) => hydra_session::Notice::line(hydra_session::NoticeKind::Warn, why),
32        };
33        state.answer = Some((words, answer));
34        state.scroll = 0;
35    }
36    let accent = colors.text(Token::Accent);
37    let mut rows = vec![vec![
38        button("[Ask…]", accent, "ask:Ask the bestiary"),
39        word(" ", colors.canvas()),
40        button("[Here]", accent, "say:here"),
41        word(" ", colors.canvas()),
42        button("[Undead]", accent, "say:undead"),
43    ]];
44    let Some((asked, notice)) = &state.answer else {
45        rows.push(note(
46            "Ask the bestiary: a creature's name, here, level 29, undead.",
47            colors,
48        ));
49        return rows;
50    };
51    rows.push(note(format!("bestiary {asked}"), colors));
52    for line in crate::story::notice_lines(notice, colors) {
53        rows.push(line.into_iter().map(answered).collect());
54    }
55    rows
56}
57
58/// A piece of the answer: a link that asks the bestiary again, a press of
59/// the widget's own; any other as it is.
60fn answered(piece: crate::text::Piece) -> Bit {
61    let again = match (&piece.link, piece.own) {
62        (Some(RunLink::Command { command }), true) => command
63            .strip_prefix("bestiary")
64            .map(|rest| rest.trim().to_owned()),
65        _ => None,
66    };
67    match again {
68        Some(rest) => Bit {
69            piece: crate::text::Piece {
70                link: None,
71                own: false,
72                ..piece
73            },
74            press: Some(format!("say:{rest}")),
75        },
76        None => bit(piece),
77    }
78}

80#[cfg(test)]
81mod tests {
82    use hydra_ui::Story;
83    use hydra_ui::widget::Widget;
84
85    use crate::hits::Target;
86    use crate::theme::Colors;
87    use crate::widgets::tests::{seen_of, text_of};
88    use crate::widgets::{PRESS, Scroll, States, draw};
89
90    /// Asked for undead, the widget answers with the codex's list, its
91    /// presses along the top.
92    #[test]
93    fn the_bestiary_answers_what_is_asked() {
94        let story = Story::default();
95        let colors = Colors::default();
96        let snapshot = hydra_ui::fixture::snapshot();
97        let seen = seen_of(&story, Some(&snapshot), &colors);
98        let mut states = States::default();
99        let area = ratatui::layout::Rect::new(0, 0, 60, 8);
100        let mut buf = ratatui::buffer::Buffer::empty(area);
101        let mut hits = crate::hits::Hits::default();
102        draw(
103            &Widget::Bestiary,
104            &mut buf,
105            area,
106            (&seen, &mut hits),
107            (Scroll::default(), states.of((0, 0))),
108        );
109        let text = text_of(&buf);
110        assert_eq!(text[0], "[Ask…] [Here] [Undead]");
111        assert!(text[1].starts_with("Ask the bestiary"), "{text:?}");
112        assert_eq!(
113            hits.at(9, 0),
114            Some(&Target::Button(format!("{PRESS}say:here")))
115        );
116        *states.state((0, 0)).typed.lock().expect("held") = Some("undead".to_owned());
117        let mut buf = ratatui::buffer::Buffer::empty(area);
118        draw(
119            &Widget::Bestiary,
120            &mut buf,
121            area,
122            (&seen, &mut crate::hits::Hits::default()),
123            (Scroll::default(), states.of((0, 0))),
124        );
125        let text = text_of(&buf);
126        assert_eq!(text[1], "bestiary undead");
127        assert!(text.len() > 2 && !text[2].is_empty(), "{text:?}");
128        assert!(states.state((0, 0)).answer.is_some());
129    }
130}

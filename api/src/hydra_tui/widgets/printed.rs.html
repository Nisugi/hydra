1//! Everything the model holds for a character, as it prints itself: the
2//! Game state widget, for troubleshooting, and the Field widget, one value
3//! of it on a line of its own.
4//!
5//! The print is `GameState`'s `Debug`, read as a tree
6//! (`hydra_ui::printed`), so no field is left off and a field added to the
7//! model shows with no change here. The Game state widget shows it closed
8//! below its top, each branch opened and shut by its `▸`; *Find* narrows
9//! it to the lines holding some words, each with where it is; *Copy* takes
10//! the whole print for a report.
11//!
12//! A print is taken at most twice a second for each character, and every
13//! widget drawing that character's shares it.
14

26/// The most lines *Find* lists.
27const FOUND: usize = 500;
28
29/// The prints kept, by character.
30pub(super) type Prints = hydra_ui::printed::Prints<String>;
31
32/// The print of the character `seen` draws, kept in `prints` and taken
33/// again once it is old; `None` before the game has said anything.
34pub(super) fn print(seen: &Seen<'_>, prints: &mut Prints) -> Option<Arc<Printed>> {
35    let state = seen.state()?;
36    let key = seen.key().unwrap_or_default();
37    Some(prints.print(key, state, Instant::now()))
38}

40/// The Game state widget's rows: Find, Copy, and the tree or what was
41/// found.
42pub(super) fn game_state(
43    seen: &Seen<'_>,
44    state: &mut State,
45    print: Option<Arc<Printed>>,
46) -> Vec<Row> {
47    let colors = seen.colors;
48    if let Some(words) = state.take_typed() {
49        state.filter = words;
50        state.scroll = 0;
51    }
52    let mut rows: Vec<Row> = seen.who.map(|who| note(who, colors)).into_iter().collect();
53    let Some(print) = print else {
54        rows.push(note("Nothing seen yet.", colors));
55        return rows;
56    };
57    let accent = colors.text(Token::Accent);
58    let mut head = vec![button("[Find…]", accent, "ask:Find in the game state")];
59    if !state.filter.is_empty() {
60        head.push(word(" ", colors.canvas()));
61        head.push(button("[All]", accent, "say:"));
62    }
63    if let Some((seat, _)) = seen.seated {
64        head.push(word(" ", colors.canvas()));
65        head.push(button("[Copy]", accent, format!("copy:{}", seat.id)));
66    }
67    if !state.filter.is_empty() {
68        head.push(word(format!("  {}", state.filter), colors.muted()));
69    }
70    rows.push(head);
71    let lines = top(&print.tree);
72    let needle = state.filter.trim().to_lowercase();
73    if needle.is_empty() {
74        branches(lines, (&mut Vec::new(), &state.open), (seen, &mut rows));
75        return rows;
76    }
77    let found = found(lines, &needle, FOUND);
78    if found.is_empty() {
79        rows.push(note("Nothing matches.", colors));
80    }
81    for (path, said) in found {
82        rows.push(vec![
83            word(path.join(" › "), colors.muted()),
84            word(format!(" = {said}"), colors.canvas()),
85        ]);
86    }
87    rows
88}

90/// `lines` as rows, each branch shut until opened: `path` the steps down to
91/// them, `open` the branches opened, by their paths.
92fn branches(
93    lines: &[Line],
94    (path, open): (&mut Vec<String>, &std::collections::BTreeSet<String>),
95    (seen, rows): (&Seen<'_>, &mut Vec<Row>),
96) {
97    let colors = seen.colors;
98    let indent = "  ".repeat(path.len());
99    for (at, line) in lines.iter().enumerate() {
100        match line {
101            Line::Leaf(text) => rows.push(vec![word(format!("{indent}  {text}"), colors.canvas())]),
102            Line::Branch(label, inside) => {
103                path.push(step(label, at));
104                let key = path.join("\u{1f}");
105                let opened = open.contains(&key);
106                rows.push(vec![
107                    word(indent.clone(), colors.canvas()),
108                    button(
109                        format!("{} {label}", if opened { "▾" } else { "▸" }),
110                        colors.text(Token::Accent),
111                        format!("toggle:{key}"),
112                    ),
113                ]);
114                if opened {
115                    branches(inside, (path, open), (seen, rows));
116                }
117                path.pop();
118            }
119        }
120    }
121}

123/// The Field widget: the value at `path` alone, as plainly as the print
124/// allows.
125pub(super) fn field(
126    buf: &mut Buffer,
127    area: Rect,
128    seen: &Seen<'_>,
129    (path, print): (&[String], Option<Arc<Printed>>),
130) {
131    let said = match print {
132        Some(print) => value(top(&print.tree), path).unwrap_or_else(|| "not found".to_owned()),
133        None => "—".to_owned(),
134    };
135    line(
136        buf,
137        area,
138        &[Piece::new(seen.named(&said), seen.colors.canvas())],
139    );
140}

142#[cfg(test)]
143mod tests {
144    use hydra_ui::Story;
145    use hydra_ui::widget::Widget;
146
147    use super::*;
148    use crate::theme::Colors;
149    use crate::widgets::tests::{drawn, seen_of};
150
151    /// The game state's top is drawn shut, a branch a button; a field is
152    /// found by its path.
153    #[test]
154    fn the_game_state_prints_shut_and_a_field_is_found() {
155        let story = Story::default();
156        let colors = Colors::default();
157        let snapshot = hydra_ui::fixture::snapshot();
158        let seen = seen_of(&story, Some(&snapshot), &colors);
159        let (text, _) = drawn(&Widget::GameState, &seen, (60, 6));
160        assert_eq!(text[0], "[Find…]");
161        assert!(
162            text[1..].iter().all(|row| !row.starts_with("    ")),
163            "shut: {text:?}"
164        );
165        let mut prints = Prints::default();
166        let print = print(&seen, &mut prints).expect("printed");
167        let path: Vec<String> = ["room", "title"].map(str::to_owned).to_vec();
168        assert_eq!(
169            value(top(&print.tree), &path).as_deref(),
170            Some("Rawknuckle's, Watering Hole")
171        );
172        let (text, _) = drawn(&Widget::Field(path), &seen, (40, 1));
173        assert_eq!(text[0], "Rawknuckle's, Watering Hole");
174        assert_eq!(prints.len(), 1, "one print a character");
175    }
176
177    /// Narrowed, the lines holding the words are listed with where they
178    /// are.
179    #[test]
180    fn find_narrows_to_where_the_words_are() {
181        let story = Story::default();
182        let colors = Colors::default();
183        let snapshot = hydra_ui::fixture::snapshot();
184        let seen = seen_of(&story, Some(&snapshot), &colors);
185        let mut state = State::default();
186        *state.typed.lock().expect("held") = Some("watering".to_owned());
187        let mut prints = Prints::default();
188        let print = print(&seen, &mut prints);
189        let rows = game_state(&seen, &mut state, print);
190        let text: Vec<String> = rows
191            .iter()
192            .map(|row| row.iter().map(|bit| bit.piece.text.as_str()).collect())
193            .collect();
194        assert!(text[0].ends_with("watering"), "{text:?}");
195        assert!(
196            text.iter().any(|row| row.contains("room › title")),
197            "{text:?}"
198        );
199    }

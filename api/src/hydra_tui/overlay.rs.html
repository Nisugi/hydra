1//! What shows over a terminal's screen and takes its keys while it does: a
2//! form, an editor, a list to choose from, the main menu. They stack: the
3//! one on top has the keys and the mouse, and closing it shows the one
4//! beneath. Escape closes the one on top unless it says otherwise.
5//!
6//! The main menu (F10, or the `≡` at the foot) opens Hydra's own:
7//! the settings, the highlights, the hotbars, the theme, the keys, the web
8//! page, the performance log, and Shut down. Over a character's screen it
9//! holds that screen's too: *Windows* (show or hide each, add a widget,
10//! edit one, arrange) and *Layouts* (load one kept, save this one, fit it
11//! to the terminal).
12

27/// What an overlay asks after a key or a click.
28#[derive(Debug)]
29pub enum Next {
30    /// It stays, as it is.
31    Stay,
32    /// It closes.
33    Close,
34    /// It stays, and this opens over it.
35    Open(Box<dyn Overlay>),
36    /// It closes, and this opens in its place.
37    Instead(Box<dyn Overlay>),
38    /// It stays, and the player is told this at the foot.
39    Tell(String),
40    /// It closes, and the player is told this.
41    CloseTelling(String),
42    /// Copy this to the terminal's clipboard.
43    Copy(String),
44    /// It closes, and this is done on seat `seat`'s screen.
45    Screen {
46        /// The seat whose screen.
47        seat: u32,
48        /// What is done there.
49        does: ScreenDoes,
50    },
51    /// It stays, and a window of seat `seat`'s layout, named `was`, is
52    /// this now: the layout kept.
53    Window {
54        /// The seat whose layout holds it.
55        seat: u32,
56        /// Its name before.
57        was: String,
58        /// The window now.
59        def: Box<crate::layout::vellum_schema::WindowDef>,
60    },
61}

63/// What shows over the screen while it is open.
64pub trait Overlay: Debug + Send {
65    /// What it is called, for its frame.
66    fn title(&self) -> String;
67
68    /// A key pressed while it is on top.
69    fn key(&mut self, key: &Key, context: &Context) -> Next;
70
71    /// Text pasted while it is on top.
72    fn paste(&mut self, _text: &str, _context: &Context) -> Next {
73        Next::Stay
74    }
75
76    /// A click at `col`, `row` while it is on top, inside it.
77    fn click(&mut self, _col: u16, _row: u16, _context: &Context) -> Next {
78        Next::Stay
79    }
80
81    /// The wheel turned over it, up toward what came before.
82    fn wheel(&mut self, _up: bool) {}
83
84    /// The mouse moved to `col`, `row`.
85    fn hover(&mut self, _col: u16, _row: u16) {}
86
87    /// Whether `col`, `row` is its own though outside its frame: a menu it
88    /// opened beside it.
89    fn holds(&self, _col: u16, _row: u16) -> bool {
90        false
91    }
92
93    /// Draw it in `area`, the inside of its frame; where the cursor goes,
94    /// when it has one.
95    fn draw(
96        &mut self,
97        buf: &mut Buffer,
98        area: Rect,
99        seen: (&Context, &Colors, &mut Hits),
100    ) -> Option<(u16, u16)>;
101
102    /// How much of the screen it takes at the most: a part of its width
103    /// and height, out of 100 each.
104    fn size(&self) -> (u16, u16) {
105        (80, 80)
106    }
107
108    /// Whether it is drawn only as large as what it draws: measured within
109    /// [`Self::size`] and fitted to that, a line to spare. Not for one that
110    /// fills whatever it is given, as a log does.
111    fn fits(&self) -> bool {
112        true
113    }
114
115    /// Its inside's size in cells, when it knows what it needs: taken in
116    /// place of [`Self::size`], never more than the screen.
117    fn cells(&self) -> Option<(u16, u16)> {
118        None
119    }
120
121    /// How long until it needs drawing again with nothing new.
122    fn due(&self) -> Option<Duration> {
123        None
124    }
125}

127/// The place an overlay asking `size` takes on `screen`: centred.
128#[must_use]
129pub fn place(screen: Rect, (wide, tall): (u16, u16)) -> Rect {
130    let part = |of: u16, part: u16| {
131        u16::try_from(u32::from(of) * u32::from(part.min(100)) / 100).unwrap_or(of)
132    };
133    let (width, height) = (part(screen.width, wide), part(screen.height, tall));
134    let width = width.clamp(20.min(screen.width), screen.width);
135    let height = height.clamp(6.min(screen.height), screen.height);
136    Rect::new(
137        screen.x + (screen.width - width) / 2,
138        screen.y + (screen.height - height) / 2,
139        width,
140        height,
141    )
142}

144/// What the foot of an overlay's frame says: clicked, it closes.
145const CLOSES: &str = " Esc closes ";

147/// Whether a click at `col`, `row` is on the "Esc closes" of an overlay
148/// framed at `place`.
149#[must_use]
150pub fn closes_at(place: Rect, col: u16, row: u16) -> bool {
151    let width = u16::try_from(CLOSES.chars().count()).unwrap_or(0);
152    row == place.bottom().saturating_sub(1) && col > place.x && col <= place.x.saturating_add(width)
153}

155/// The place an overlay whose inside is `cols` by `rows` takes on
156/// `screen`: centred, its frame round it, no bigger than the screen.
157#[must_use]
158pub fn fitted(screen: Rect, (cols, rows): (u16, u16)) -> Rect {
159    let width = cols.saturating_add(2).min(screen.width);
160    let height = rows.saturating_add(2).min(screen.height);
161    Rect::new(
162        screen.x + (screen.width - width) / 2,
163        screen.y + (screen.height - height) / 2,
164        width,
165        height,
166    )
167}

169/// Draw `overlay` framed over `screen`: where the cursor goes, and its
170/// frame's place, for a click.
171pub fn draw(
172    overlay: &mut dyn Overlay,
173    buf: &mut Buffer,
174    screen: Rect,
175    (context, colors, hits): (&Context, &Colors, &mut Hits),
176) -> (Option<(u16, u16)>, Rect) {
177    let area = match overlay.cells() {
178        Some(cells) => fitted(screen, cells),
179        None if overlay.fits() => measured(overlay, screen, (context, colors)),
180        None => place(screen, overlay.size()),
181    };
182    Clear.render(area, buf);
183    Block::new()
184        .borders(Borders::ALL)
185        .border_type(BorderType::Rounded)
186        .border_style(colors.border(true))
187        .style(colors.canvas())
188        .title(format!(" {} ", overlay.title()))
189        .title_bottom(CLOSES)
190        .render(area, buf);
191    hits.add(area, crate::hits::Target::Button("overlay".to_owned()));
192    let inside = Rect::new(
193        area.x + 1,
194        area.y + 1,
195        area.width.saturating_sub(2),
196        area.height.saturating_sub(2),
197    );
198    let cursor = overlay.draw(buf, inside, (context, colors, hits));
199    (cursor, area)
200}

202/// Where `overlay` goes on `screen`: drawn aside in the most it may take,
203/// then fitted to how far its words and cursor reached, a column and a line
204/// to spare, its title and "Esc closes" never cut.
205fn measured(
206    overlay: &mut dyn Overlay,
207    screen: Rect,
208    (context, colors): (&Context, &Colors),
209) -> Rect {
210    let most = place(screen, overlay.size());
211    let inside = Rect::new(
212        0,
213        0,
214        most.width.saturating_sub(2),
215        most.height.saturating_sub(2),
216    );
217    let mut aside = Buffer::empty(inside);
218    let cursor = overlay.draw(&mut aside, inside, (context, colors, &mut Hits::default()));
219    let mut wide = cursor.map_or(0, |(x, _)| x + 1);
220    let mut written = vec![false; usize::from(inside.height)];
221    for y in 0..inside.height {
222        for x in 0..inside.width {
223            if aside.cell((x, y)).is_some_and(|cell| cell.symbol() != " ") {
224                wide = wide.max(x + 1);
225                written[usize::from(y)] = true;
226            }
227        }
228    }
229    if let Some((_, y)) = cursor
230        && let Some(row) = written.get_mut(usize::from(y))
231    {
232        *row = true;
233    }
234    let tall = high(&written);
235    let named = |text: &str| u16::try_from(text.chars().count()).unwrap_or(u16::MAX);
236    let wide = (wide + 1)
237        .max(named(&overlay.title()) + 2)
238        .max(named(CLOSES))
239        .min(inside.width);
240    let tall = tall.max(3).min(inside.height);
241    fitted(screen, (wide, tall))
242}

244/// How many lines an overlay needs whose lines with words, top to bottom,
245/// are `written`: its body, a line to spare, and its foot -- the lines with
246/// words held to the bottom, as a hint is, below a gap -- kept under it.
247fn high(written: &[bool]) -> u16 {
248    let lines = |count: usize| u16::try_from(count).unwrap_or(u16::MAX);
249    let count = written.len();
250    // The foot: a run with words ending within the bottom three lines.
251    let last = written.iter().rposition(|row| *row);
252    let foot = last.filter(|last| last + 3 >= count).map(|last| {
253        let start = written[..=last]
254            .iter()
255            .rposition(|row| !*row)
256            .map_or(0, |gap| gap + 1);
257        (start, count - start)
258    });
259    match foot {
260        Some((start, under)) => {
261            let body = written[..start].iter().rposition(|row| *row);
262            match body {
263                Some(body) => lines(body + 2 + under),
264                // Words all the way down, no gap: all of it.
265                None if start == 0 => lines(count),
266                None => lines(under + 1),
267            }
268        }
269        None => lines(last.map_or(0, |last| last + 2)),
270    }
271}

335#[derive(Clone, Debug, PartialEq, Eq)]

338    Opens(Opens),
339    /// It does this on the character's screen.
340    Screen(ScreenDoes),

343/// What the main menu offers of a character's screen.
344#[derive(Clone, Debug, Default)]
345pub struct Screened {
346    /// The command symbol the screen's commands are typed with.
347    pub symbol: char,
348    /// The layout's windows, each its name and whether it is hidden.
349    pub windows: Vec<(String, bool)>,
350    /// The layouts kept by name.
351    pub layouts: Vec<String>,
352    /// The widget catalog.
353    pub catalog: Vec<Entry<crate::arrange::Pick>>,
354}
355
356impl Screened {
357    /// Its *Windows* and *Layouts* menus.
358    fn entries(self) -> Vec<Entry<Chosen>> {
359        let line = |line: String| Chosen::Screen(ScreenDoes::Line(line));
360        let symbol = self.symbol;
361        let shown = self
362            .windows
363            .iter()
364            .map(|(name, hidden)| {
365                let label = if *hidden {
366                    format!("{name} (hidden)")
367                } else {
368                    name.clone()
369                };
370                Entry::new(label, line(format!("{symbol}hidewindow {name}")))
371            })
372            .collect();
373        let edited = self
374            .windows
375            .iter()
376            .map(|(name, _)| Entry::new(name.clone(), line(format!("{symbol}editwindow {name}"))))
377            .collect();
378        let windows = vec![
379            Entry::sub("Show/Hide windows", shown),
380            Entry::sub("Add a widget", adds(self.catalog)),
381            Entry::sub("Edit window", edited),
382            Entry::new("Arrange", line(format!("{symbol}arrange"))),
383        ];
384        let mut layouts: Vec<Entry<Chosen>> = self
385            .layouts
386            .iter()
387            .map(|name| Entry::new(name.clone(), line(format!("{symbol}loadlayout {name}"))))
388            .collect();
389        if layouts.is_empty() {
390            layouts.push(Entry::dim("None kept yet"));
391        }
392        layouts.push(Entry::new(
393            "Save as...",
394            Chosen::Screen(ScreenDoes::Type(format!("{symbol}savelayout "))),
395        ));
396        layouts.push(Entry::new(
397            "Fit to this terminal",
398            line(format!("{symbol}resize")),
399        ));
400        vec![
401            Entry::sub("Windows", windows),
402            Entry::sub("Layouts", layouts),
403        ]
404    }
405}
406
407/// The widget catalog's entries, each adding its widget.
408fn adds(catalog: Vec<Entry<crate::arrange::Pick>>) -> Vec<Entry<Chosen>> {
409    catalog
410        .into_iter()
411        .map(|entry| Entry {
412            label: entry.label,
413            does: entry.does.map(|pick| Chosen::Screen(ScreenDoes::Add(pick))),
414            heading: entry.heading,
415            sub: adds(entry.sub),
416        })
417        .collect()
418}

420/// The main menu, as an overlay: a list of what it opens.
421#[derive(Debug)]
422pub struct MainMenu {
423    menu: Menu<Chosen>,
424    /// The character whose screen it was opened over, for what is a
425    /// character's: its settings, its keys, its hotbars.
426    pub seat: Option<u32>,
427}
428
429impl MainMenu {
430    /// The main menu, opened over seat `seat`'s screen or the hub's.
431    #[must_use]
432    pub fn new(seat: Option<u32>) -> Self {
433        Self::with_screen(seat, None)
434    }
435
436    /// The main menu over seat `seat`'s screen, with what it offers of it,
437    /// `screened`, before Hydra's own.
438    #[must_use]
439    pub fn with_screen(seat: Option<u32>, screened: Option<Screened>) -> Self {
440        let mut entries = screened.map(Screened::entries).unwrap_or_default();
441        entries.extend(
442            Opens::ALL
443                .into_iter()
444                .map(|opens| Entry::new(opens.name(), Chosen::Opens(opens))),
445        );
446        Self {
447            menu: Menu::new("Hydra", entries, None),
448            seat,
449        }
450    }
451
452    /// What choosing `chosen` does.
453    fn chose(&self, chosen: Chosen, context: &Context) -> Next {
454        match (chosen, self.seat) {
455            (Chosen::Opens(opens), seat) => crate::opens::open(opens, seat, context),
456            (Chosen::Screen(does), Some(seat)) => Next::Screen { seat, does },
457            (Chosen::Screen(_), None) => Next::Close,
458        }
459    }
460}

462impl Overlay for MainMenu {
463    fn title(&self) -> String {
464        "Hydra".to_owned()
465    }
466
467    fn key(&mut self, key: &Key, context: &Context) -> Next {
468        match self.menu.key(key) {
469            Done::Chose(opens) => self.chose(opens, context),
470            Done::Closed => Next::Close,
471            Done::Open => Next::Stay,
472        }
473    }
474
475    fn click(&mut self, col: u16, row: u16, context: &Context) -> Next {
476        match self.menu.click(col, row) {
477            Done::Chose(opens) => self.chose(opens, context),
478            Done::Closed => Next::Close,
479            Done::Open => Next::Stay,
480        }
481    }
482
483    fn cells(&self) -> Option<(u16, u16)> {
484        let widest = self
485            .menu
486            .entries
487            .iter()
488            .map(|entry| entry.label.chars().count() + if entry.sub.is_empty() { 0 } else { 2 })
489            .chain([CLOSES.chars().count(), self.title().chars().count() + 2])
490            .max()
491            .unwrap_or(0);
492        let wide = u16::try_from(widest + 2).unwrap_or(u16::MAX);
493        let tall = u16::try_from(self.menu.entries.len()).unwrap_or(u16::MAX);
494        Some((wide, tall))
495    }
496
497    fn hover(&mut self, col: u16, row: u16) {
498        self.menu.hover(col, row);
499    }
500
501    fn holds(&self, col: u16, row: u16) -> bool {
502        self.menu.contains(col, row)
503    }
504
505    fn draw(
506        &mut self,
507        buf: &mut Buffer,
508        area: Rect,
509        (_, colors, hits): (&Context, &Colors, &mut Hits),
510    ) -> Option<(u16, u16)> {
511        self.menu.draw_list(buf, area, colors, hits);
512        if let Some(open) = &mut self.menu.open {
513            open.draw(buf, buf.area, colors, hits);
514        }
515        None
516    }
517}

519/// A line of words on an overlay: a hint, what was done.
520pub fn say(buf: &mut Buffer, area: Rect, row: u16, text: &str, colors: &Colors, token: Token) {
521    if row < area.height {
522        buf.set_stringn(
523            area.x,
524            area.y + row,
525            text,
526            usize::from(area.width),
527            colors.text(token),
528        );
529    }

1//! The keys, as the windows have them: Hydra's defaults, every character's
2//! keybinds file over them, and a character's own file over that, in the
3//! macro set it has chosen (`hydra_board::keys`). A key pressed on a
4//! character's screen does what its binding says before anything else sees
5//! it; one bound to nothing types, or moves the line, as keys do.
6//!
7//! The files are read again when they change, looked at no oftener than
8//! every `LOOK_EVERY`: a key changed on the windows' Keys page, or by
9//! hand, works here a moment later.
10//!
11//! A terminal names the key that typed a character, not always the key: a
12//! numpad key is told apart only in the Windows console and in the
13//! terminals that speak the kitty keyboard protocol. A numpad key bound
14//! does its binding; unbound, it types its digit.
15//!
16//! The files are changed here too ([`Files`]): by the Keys editor, and by
17//! `.keys import`, a Wrayth settings file's key sets, and `.keys vellum`,
18//! a Vellum keybinds file's keys ([`vellum`]), each into the character's
19//! own keys or, with `global`, every character's.
20

32/// How often the files are looked at again.
33const LOOK_EVERY: Duration = Duration::from_secs(2);

35/// A file as last read: when it was changed, and what it says.
36#[derive(Debug)]
37struct Read<T> {
38    modified: Option<SystemTime>,
39    held: T,
40}
41
42/// The keys every screen of one terminal presses.
43#[derive(Debug)]
44pub struct Keys {
45    data: PathBuf,
46    every: Read<Keybinds>,
47    /// What is wrong in the keybinds file, not yet told.
48    pub problems: Vec<String>,
49    mine: HashMap<PathBuf, Read<KeyFile>>,
50    looked: Option<Instant>,
51}
52
53/// When the file at `path` was last changed; `None` for none there.
54fn modified(path: &Path) -> Option<SystemTime> {
55    std::fs::metadata(path)
56        .and_then(|meta| meta.modified())
57        .ok()
58}
59
60impl Keys {
61    /// The keys of the data folder `data`, read now.
62    #[must_use]
63    pub fn read(data: &Path) -> Self {
64        let mut keys = Self {
65            data: data.to_path_buf(),
66            every: Read {
67                modified: None,
68                held: Keybinds::default(),
69            },
70            problems: Vec::new(),
71            mine: HashMap::new(),
72            looked: None,
73        };
74        keys.read_every();
75        keys
76    }
77
78    /// Every character's file read again.
79    fn read_every(&mut self) {
80        let path = hydra_board::keys::path(&self.data);
81        let text = std::fs::read_to_string(&path).unwrap_or_default();
82        let (binds, problems) = Keybinds::read(&text);
83        self.every = Read {
84            modified: modified(&path),
85            held: binds,
86        };
87        self.problems = problems
88            .into_iter()
89            .map(|why| format!("{}: {why}", hydra_board::keys::FILE))
90            .collect();
91    }
92
93    /// The files read again where they have changed, once [`LOOK_EVERY`]
94    /// has passed since they were last looked at.
95    fn look(&mut self, now: Instant) {
96        if self
97            .looked
98            .is_some_and(|at| now.saturating_duration_since(at) < LOOK_EVERY)
99        {
100            return;
101        }
102        self.looked = Some(now);
103        if modified(&hydra_board::keys::path(&self.data)) != self.every.modified {
104            self.read_every();
105        }
106        self.mine
107            .retain(|path, read| modified(path) == read.modified);
108    }
109
110    /// `seat`'s own keys file.
111    fn own(&mut self, seat: &Seat) -> &KeyFile {
112        let path = hydra_board::keys::character_path(
113            &self.data.join("layouts"),
114            seat.instance.as_deref(),
115            &seat.name,
116        );
117        let read = self.mine.entry(path.clone()).or_insert_with(|| {
118            let text = std::fs::read_to_string(&path).unwrap_or_default();
119            Read {
120                modified: modified(&path),
121                held: KeyFile::read(&text, Whose::Character).0,
122            }
123        });
124        &read.held
125    }
126
127    /// What `key` does on `seat`'s screen, if it is bound.
128    pub fn does(&mut self, key: &Key, seat: &Seat) -> Option<Macro> {
129        self.look(Instant::now());
130        let chord = Chord::new(&key.name, key.mods.0);
131        // A key that types is never bound; Shift held with one types too.
132        if chord.refused().is_some() {
133            return None;
134        }
135        let mine = self.own(seat).clone();
136        self.every.held.of(Some(&mine)).does(&chord).cloned()
137    }
138
139    /// The macro set `seat` has chosen.
140    pub fn chosen(&mut self, seat: &Seat) -> u8 {
141        self.own(seat).chosen
142    }
143}

145/// What `.keys` says when it is not given a word it knows and a file.
146const USAGE: &str = "keys import <Wrayth settings file>: its key sets into this character's \
147                     keys; keys import global <file>: into every character's. keys vellum \
148                     <keybinds.toml>: a Vellum keybinds file's [user] keys into this \
149                     character's; keys vellum global <file>: into every character's.";

151/// Where keys are imported from.
152#[derive(Clone, Copy, Debug, PartialEq, Eq)]
153enum Source {
154    /// A Wrayth settings file's key sets.
155    Wrayth,
156    /// A Vellum keybinds file's `[user]` keys.
157    Vellum,
158}
159
160impl Source {
161    /// What the player calls it.
162    const fn name(self) -> &'static str {
163        match self {
164            Self::Wrayth => "Wrayth",
165            Self::Vellum => "Vellum",
166        }
167    }
168
169    /// The keys in `text`, a file of this kind, and what could not be
170    /// bound.
171    fn read(self, text: &str) -> (Vec<Imported>, Vec<String>) {
172        match self {
173            Self::Wrayth => hydra_board::keys::wrayth::read(text),
174            Self::Vellum => vellum::read(text),
175        }
176    }
177}

179/// `.keys`, its words after it `rest`, typed on `seat`'s screen with the
180/// data folder `data`: what was done, a line at a time.
181#[must_use]
182pub fn command(rest: &str, seat: &Seat, data: &Path) -> Vec<String> {
183    let rest = rest.trim();
184    let (verb, after) = rest.split_once(char::is_whitespace).unwrap_or((rest, ""));
185    let source = match verb.to_ascii_lowercase().as_str() {
186        "import" => Source::Wrayth,
187        "vellum" => Source::Vellum,
188        _ => return vec![USAGE.to_owned()],
189    };
190    let after = after.trim();
191    let (every, file) = match after.split_once(char::is_whitespace) {
192        Some((word, file)) if word.eq_ignore_ascii_case("global") => (true, file.trim()),
193        _ => (false, after),
194    };
195    if file.is_empty() || file.eq_ignore_ascii_case("global") {
196        return vec![USAGE.to_owned()];
197    }
198    let mut files = Files::read(data, Some((seat.instance.as_deref(), &seat.name)));
199    files.import(Path::new(file.trim_matches('"')), every, source)
200}

217/// The keys files as they are, read to be changed: every character's, and
218/// a character's own. A change is made to the file as it then stands, with
219/// the batch of saves (`hydra_board::store`), so one made by the windows at
220/// the same time is kept too; the batch is written at once, so the keys
221/// pressed next do it, and the files are read again.
222#[derive(Debug)]
223pub struct Files {
224    /// Every character's file.
225    every_path: PathBuf,
226    /// Its text.
227    every_text: String,
228    /// Every character's keys over Hydra's.
229    pub every: Keybinds,
230    /// What is wrong in every character's file.
231    pub problems: Vec<String>,
232    /// The character's own, when they are a character's.
233    mine: Option<Mine>,
234}
235
236impl Files {
237    /// The keys files of the data folder `data`, the character on the game
238    /// `instance` called `name`'s own among them when there is one.
239    #[must_use]
240    pub fn read(data: &Path, character: Option<(Option<&str>, &str)>) -> Self {
241        let every_path = hydra_board::keys::path(data);
242        let mut files = Self {
243            every_path,
244            every_text: String::new(),
245            every: Keybinds::default(),
246            problems: Vec::new(),
247            mine: character.map(|(instance, name)| Mine {
248                path: hydra_board::keys::character_path(&data.join("layouts"), instance, name),
249                name: name.to_owned(),
250                text: String::new(),
251                file: KeyFile::default(),
252                problems: Vec::new(),
253            }),
254        };
255        files.read_again();
256        files
257    }
258
259    /// Every file read again from the disk.
260    fn read_again(&mut self) {
261        self.every_text = std::fs::read_to_string(&self.every_path).unwrap_or_default();
262        let (every, problems) = Keybinds::read(&self.every_text);
263        self.every = every;
264        self.problems = problems
265            .into_iter()
266            .map(|why| format!("{}: {why}", hydra_board::keys::FILE))
267            .collect();
268        if let Some(mine) = &mut self.mine {
269            mine.text = std::fs::read_to_string(&mine.path).unwrap_or_default();
270            let (file, problems) = KeyFile::read(&mine.text, Whose::Character);
271            mine.file = file;
272            mine.problems = problems;
273        }
274    }

276    /// The character's own file as read, when the keys are a character's.
277    #[must_use]
278    pub fn mine(&self) -> Option<&KeyFile> {
279        self.mine.as_ref().map(|mine| &mine.file)
280    }
281
282    /// The character's name, when the keys are a character's.
283    #[must_use]
284    pub fn name(&self) -> Option<&str> {
285        self.mine.as_ref().map(|mine| mine.name.as_str())
286    }

288    /// What is wrong in the files: every character's, then the
289    /// character's.
290    #[must_use]
291    pub fn wrong(&self) -> Vec<String> {
292        let mut wrong = self.problems.clone();
293        if let Some(mine) = &self.mine {
294            let name = name_of(&mine.path);
295            wrong.extend(mine.problems.iter().map(|why| format!("{name}: {why}")));
296        }
297        wrong
298    }

307    /// The file a change goes to: every character's (`every`), or the
308    /// character's own; whose it is, and the words for whose keys they are.
309    ///
310    /// # Errors
311    ///
312    /// The keys are no character's, and a character's own was asked.
313    fn file(&self, every: bool) -> Result<(PathBuf, Whose, String), String> {
314        match (every, &self.mine) {
315            (true, _) => Ok((
316                self.every_path.clone(),
317                Whose::Every,
318                "Every character's keys".to_owned(),
319            )),
320            (false, Some(mine)) => Ok((
321                mine.path.clone(),
322                Whose::Character,
323                format!("{}'s keys", mine.name),
324            )),
325            (false, None) => Err("Pick a character to change its own keys.".to_owned()),
326        }
327    }

329    /// The text of the file `whose` it is.
330    fn text(&self, whose: Whose) -> &str {
331        match (whose, &self.mine) {
332            (Whose::Character, Some(mine)) => &mine.text,
333            _ => &self.every_text,
334        }
335    }

337    /// `edit` made to the file at `path` with the batch, written now, and
338    /// the files read again.
339    ///
340    /// # Errors
341    ///
342    /// The file was not written.
343    fn write(&mut self, path: &Path, edit: hydra_board::store::Edit) -> Result<(), String> {
344        let dir = path.parent().unwrap_or(Path::new(".")).to_owned();
345        hydra_board::store::edit(&dir, path, edit);
346        hydra_board::store::flush();
347        let failed = hydra_board::store::failed(path);
348        self.read_again();
349        failed.map_or(Ok(()), |why| {
350            Err(format!("{} was not saved: {why}", name_of(path)))
351        })
352    }

354    /// Make `change`, asked on the Keys page, to the file it is for: what
355    /// was done, in words for the player.
356    ///
357    /// # Errors
358    ///
359    /// Why nothing was changed.
360    pub fn change(&mut self, change: &KeyChange) -> Result<String, String> {
361        let every = match change {
362            KeyChange::Bind { place, .. }
363            | KeyChange::Unbind { place, .. }
364            | KeyChange::Restore { place, .. } => place.every,
365            KeyChange::NumpadAlways(_) => true,
366            KeyChange::Choose(_) => false,
367            KeyChange::Share {
368                key,
369                does,
370                set,
371                every,
372            } => return self.share(key, does, *set, *every),
373        };
374        let (path, whose, said) = self.file(every)?;
375        let name = name_of(&path);
376        let (_, done) = write::made(
377            self.text(whose),
378            (&name, whose),
379            change,
380            (&self.every, &said),
381        )?;
382        let (keys, made) = (self.every.clone(), change.clone());
383        self.write(
384            &path,
385            Box::new(move |now: &str| {
386                write::made(now, (&name, whose), &made, (&keys, &said)).map(|(text, _)| text)
387            }),
388        )?;
389        Ok(done)
390    }

392    /// `key`, doing `does` in set `set`, moved to every character's file
393    /// (`to_every`) or to the character's own, and taken out of the other.
394    ///
395    /// # Errors
396    ///
397    /// Why nothing was changed.
398    fn share(
399        &mut self,
400        key: &str,
401        does: &Macro,
402        set: u8,
403        to_every: bool,
404    ) -> Result<String, String> {
405        let (to, to_whose, said) = self.file(to_every)?;
406        let (from, from_whose, _) = self.file(!to_every)?;
407        let place = |every| Place { set, every };
408        let bound = KeyChange::Bind {
409            key: key.to_owned(),
410            does: does.clone(),
411            was: None,
412            place: place(to_every),
413        };
414        let taken = KeyChange::Restore {
415            key: key.to_owned(),
416            place: place(!to_every),
417        };
418        let (to_name, from_name) = (name_of(&to), name_of(&from));
419        write::made_pair(
420            ((&to_name, self.text(to_whose)), to_whose, &bound),
421            ((&from_name, self.text(from_whose)), from_whose, &taken),
422            (&self.every, &said),
423        )?;
424        let keys = self.every.clone();
425        for (path, name, whose, made) in [
426            (to, to_name, to_whose, bound),
427            (from, from_name, from_whose, taken),
428        ] {
429            let (keys, said) = (keys.clone(), said.clone());
430            self.write(
431                &path,
432                Box::new(move |now: &str| {
433                    write::made(now, (&name, whose), &made, (&keys, &said)).map(|(text, _)| text)
434                }),
435            )?;
436        }
437        Ok(format!(
438            "{said}: {key} is {} now.",
439            if to_every {
440                "every character's"
441            } else {
442                "only this character's"
443            }
444        ))
445    }

447    /// The keys in `file`, a file of `source`'s, into the character's own
448    /// keys, or every character's (`every`): what was done, a line at a
449    /// time. A key already doing what it would, in that file as it is now,
450    /// is left as it is.
451    fn import(&mut self, file: &Path, every: bool, source: Source) -> Vec<String> {
452        let text = match std::fs::read_to_string(file) {
453            Ok(text) => text,
454            Err(why) => return vec![format!("Keys: {} could not be read: {why}", file.display())],
455        };
456        let (imported, mut said) = source.read(&text);
457        let (path, whose, whom) = match self.file(every) {
458            Ok((path, Whose::Every, _)) => {
459                (path, Whose::Every, "every character's keys".to_owned())
460            }
461            Ok((path, whose, whom)) => (path, whose, whom),
462            Err(why) => return vec![why],
463        };
464        let mine = if every { None } else { self.mine() };
465        let (mut changes, mut already, mut moved) = (Vec::new(), 0, Vec::new());
466        for key in &imported {
467            if key.set == 0 && self.every.in_file(whose, mine, &key.chord) == Some(&key.does) {
468                already += 1;
469                continue;
470            }
471            let beneath = self.every.beneath(whose, &key.chord);
472            if key.set == 0 && beneath.is_some_and(|beneath| *beneath != key.does) {
473                moved.push(key.chord.written());
474            }
475            changes.push(KeyChange::Bind {
476                key: key.chord.written(),
477                does: key.does.clone(),
478                was: None,
479                place: Place {
480                    set: key.set,
481                    every,
482                },
483            });
484        }
485        let name = name_of(&path);
486        let (_, made, mut refused) = match write::made_all(
487            self.text(whose),
488            (&name, whose),
489            &changes,
490            (&self.every, "Keys"),
491        ) {
492            Ok(done) => done,
493            Err(why) => return vec![why],
494        };
495        let keys = self.every.clone();
496        let written = self.write(
497            &path,
498            Box::new(move |now: &str| {
499                write::made_all(now, (&name, whose), &changes, (&keys, "Keys"))
500                    .map(|(text, _, _)| text)
501            }),
502        );
503        if let Err(why) = written {
504            return vec![why];
505        }
506        let mut lines = vec![format!(
507            "Keys: {made} imported from {} into {whom}; {already} already so.",
508            name_of(file)
509        )];
510        if !moved.is_empty() {
511            lines.push(format!(
512                "These now do what {} had them do, not Hydra's: {}. Restore one on the Keys \
513                 page for Hydra's.",
514                source.name(),
515                moved.join(", ")
516            ));
517        }
518        said.append(&mut refused);
519        if !said.is_empty() {
520            lines.push("Not imported:".to_owned());
521            lines.append(&mut said);
522        }
523        lines
524    }
525}

527/// One of Vellum's keybinds files, `file`, into the keys of the character
528/// `name` on `instance`, or every character's (`every`), in the data folder
529/// `data`: what was done, a line at a time.
530#[must_use]
531pub fn vellum_into(
532    data: &Path,
533    (instance, name): (Option<&str>, &str),
534    file: &Path,
535    every: bool,
536) -> Vec<String> {
537    let mut files = Files::read(data, Some((instance, name)));
538    files.import(file, every, Source::Vellum)
539}

541/// A file's name, for what is said of it.
542fn name_of(path: &Path) -> String {
543    path.file_name()
544        .map_or_else(String::new, |name| name.to_string_lossy().into_owned())
545}

652        let dir = folder("change");
653        let mut files = Files::read(&dir, Some((None, "Ashryn")));

655            set: 0,
656            every: false,
657        };

661            was: None,
662            place: mine,

667                .rows(0)
668                .into_iter()
669                .find(|row| row.key == key)
670                .expect("a row")
671        };

674            key: "F5".to_owned(),
675            does: Macro::Send("hide".to_owned()),
676            set: 0,
677            every: true,
678        });
679        assert!(shared.is_ok(), "{shared:?}");

681        let gone = files.change(&KeyChange::Unbind {
682            key: "F5".to_owned(),
683            place: Place {
684                set: 0,
685                every: true,
686            },
687        });
688        assert!(gone.is_ok(), "{gone:?}");

1//! The keys editor, over a terminal's screen: every key a macro set binds,
2//! Hydra's own among them in set 0, each with what it does, whose file says
3//! so, and Hydra's default. Opened over a character's screen it is that
4//! character's keys, its own file over every character's; over the hub's,
5//! every character's alone.
6//!
7//! On the list, Enter changes what the key marked does: commands sent, `\r`
8//! between two and `s1.5` a wait; the input filled; or one of Hydra's
9//! actions, Tab stepping between the three. `n` binds a new key, pressed
10//! rather than named, and `m` moves the marked key's macro to another key
11//! pressed; a key that types is refused, as the file refuses it, and Escape
12//! stops the wait. Delete unbinds the key; `r` takes its line out of its
13//! file, so what is beneath does again. `0` to `9` show a macro set. Over a
14//! character, `g` moves a key between its own file and every character's,
15//! and `s` uses the set shown over set 0; over the hub, `p` switches
16//! whether the numpad sends its keys with `NumLock` on too.
17//!
18//! Each change is written at once ([`crate::keys::Files`]), and the keys
19//! pressed next do it.
20

43/// What a key's macro is, as the editor offers it.
44#[derive(Clone, Copy, Debug, PartialEq, Eq)]
45enum Kind {
46    /// Commands sent.
47    Send,
48    /// The input filled.
49    Fill,
50    /// One of Hydra's actions.
51    Act,
52}
53
54impl Kind {
55    /// What the player calls it.
56    const fn label(self) -> &'static str {
57        match self {
58            Self::Send => "sends",
59            Self::Fill => "fills the input with",
60            Self::Act => "does",
61        }
62    }
63
64    /// The kind after it, Tab's.
65    const fn next(self) -> Self {
66        match self {
67            Self::Send => Self::Fill,
68            Self::Fill => Self::Act,
69            Self::Act => Self::Send,
70        }
71    }
72
73    /// The macro of this kind holding `text`, as the player typed it.
74    fn with(self, text: &str) -> Macro {
75        match self {
76            Self::Send | Self::Act => Macro::Send(binding::typed(text.trim())),
77            Self::Fill => Macro::Fill(text.trim_start().to_owned()),
78        }
79    }
80}

82/// What the editor is doing.
83#[derive(Debug)]
84enum Doing {
85    /// Showing the list.
86    Looking,
87    /// Waiting for a key pressed: a new one, or one to move this key's
88    /// macro to.
89    Waiting {
90        /// The key whose macro moves.
91        moving: Option<String>,
92    },
93    /// What a key does, typed.
94    Typing {
95        /// The key.
96        key: String,
97        /// Where the change goes.
98        place: Place,
99        /// Sent, or filling the input.
100        kind: Kind,
101        /// What is typed.
102        line: LineEdit,
103    },
104    /// The action a key does, chosen.
105    Choosing {
106        /// The key.
107        key: String,
108        /// Where the change goes.
109        place: Place,
110        /// The actions.
111        menu: Menu<Action>,
112    },
113}

115/// What a key asked of the editor.
116#[derive(Debug, PartialEq, Eq)]
117enum Asked {
118    /// Nothing to be made.
119    Nothing,
120    /// This change made.
121    Change(KeyChange),
122    /// The editor closed.
123    Close,
124}

151    pub fn new(seat: Option<u32>, context: &Context) -> Self {

153        let character = seat.and_then(|id| crate::seat::find(&seats, id));

164            files,
165            set: 0,
166            marked: 0,
167            top: 0,
168            visible: 10,
169            doing: Doing::Looking,
170            note: None,

181    /// Where a change to `row` goes: its own file, or for Hydra's default,
182    /// the character's on a character's keys.
183    fn place(&self, row: Option<&KeyRow>) -> Place {
184        let every = match row.and_then(|row| row.from) {
185            Some(whose) => whose == Whose::Every,
186            None => !self.character(),
187        };
188        Place {
189            set: self.set,
190            every,
191        }
192    }

194    /// `key` pressed: what it asks.
195    fn pressed(&mut self, key: &Key) -> Asked {
196        match std::mem::replace(&mut self.doing, Doing::Looking) {
197            Doing::Looking => self.looking(key),
198            Doing::Waiting { moving } => self.waiting(key, moving),
199            Doing::Typing {
200                key: bound,
201                place,
202                kind,
203                line,
204            } => self.typing(key, (bound, place, kind), line),
205            Doing::Choosing {
206                key: bound,
207                place,
208                mut menu,
209            } => {
210                if key.name == "Tab" {
211                    self.doing = Doing::Typing {
212                        key: bound,
213                        place,
214                        kind: Kind::Send,
215                        line: LineEdit::default(),
216                    };
217                    return Asked::Nothing;
218                }
219                match menu.key(key) {
220                    Done::Chose(action) => Asked::Change(KeyChange::Bind {
221                        key: bound,
222                        does: Macro::Act(action),
223                        was: None,
224                        place,
225                    }),
226                    Done::Closed => Asked::Nothing,
227                    Done::Open => {
228                        self.doing = Doing::Choosing {
229                            key: bound,
230                            place,
231                            menu,
232                        };
233                        Asked::Nothing
234                    }
235                }
236            }
237        }
238    }

240    /// A key on the list.
241    fn looking(&mut self, key: &Key) -> Asked {
242        let rows = self.files.rows(self.set);
243        let last = rows.len().saturating_sub(1);
244        let row = rows.get(self.marked.min(last));
245        match key.name.as_str() {
246            "Escape" => return Asked::Close,
247            "ArrowUp" => self.marked = self.marked.saturating_sub(1),
248            "ArrowDown" => self.marked = (self.marked + 1).min(last),
249            "PageUp" => self.marked = self.marked.saturating_sub(self.visible.max(2) - 1),
250            "PageDown" => self.marked = (self.marked + self.visible.max(2) - 1).min(last),
251            "Home" => self.marked = 0,
252            "End" => self.marked = last,
253            "Enter" | "NumpadEnter" => {
254                if let Some(row) = row {
255                    self.edit(row);
256                }
257            }
258            "Delete" | "Backspace" => return self.unbind(row),
259            "Insert" => self.wait(None),
260            _ => {
261                let typed = key.text.filter(|_| !key.mods.commands());
262                return self.letter(typed, row);
263            }
264        }
265        Asked::Nothing
266    }

268    /// A letter typed on the list: what it does.
269    fn letter(&mut self, typed: Option<char>, row: Option<&KeyRow>) -> Asked {
270        match typed {
271            Some('n' | 'N') => self.wait(None),
272            Some('m' | 'M') => match row {
273                Some(row) if row.does.is_some() => self.wait(Some(row.key.clone())),
274                _ => self.note = Some("That key does nothing: n binds a new key.".to_owned()),
275            },
276            Some('r' | 'R') => match row {
277                Some(row) if row.from.is_some() => {
278                    return Asked::Change(KeyChange::Restore {
279                        key: row.key.clone(),
280                        place: self.place(Some(row)),
281                    });
282                }
283                _ => self.note = Some("That key is Hydra's own: nothing to restore.".to_owned()),
284            },
285            Some('g' | 'G') if self.character() => match row {
286                Some(row) if row.from.is_some() => {
287                    if let Some(does) = row.does.clone() {
288                        return Asked::Change(KeyChange::Share {
289                            key: row.key.clone(),
290                            does,
291                            set: self.set,
292                            every: row.from != Some(Whose::Every),
293                        });
294                    }
295                    self.note = Some("That key does nothing to share.".to_owned());
296                }
297                _ => {
298                    self.note = Some(
299                        "That key is Hydra's own: change it first, and it is this character's."
300                            .to_owned(),
301                    );
302                }
303            },
304            Some('s' | 'S') if self.character() => {
305                return Asked::Change(KeyChange::Choose(self.set));
306            }
307            Some('p' | 'P') if !self.character() => {
308                return Asked::Change(KeyChange::NumpadAlways(!self.files.every.numpad_always()));
309            }
310            Some(digit @ '0'..='9') => {
311                let set = u8::try_from(u32::from(digit) - u32::from('0')).unwrap_or(0);
312                if set < SETS {
313                    self.set = set;
314                    self.marked = 0;
315                    self.top = 0;
316                }
317            }
318            _ => {}
319        }
320        Asked::Nothing
321    }

330    /// `row`'s key unbound.
331    fn unbind(&mut self, row: Option<&KeyRow>) -> Asked {
332        match row {
333            Some(row) if row.does.is_some() => Asked::Change(KeyChange::Unbind {
334                key: row.key.clone(),
335                place: self.place(Some(row)),
336            }),
337            _ => {
338                self.note = Some("That key does nothing already.".to_owned());
339                Asked::Nothing
340            }
341        }
342    }

344    /// What `row`'s key does, to be changed: its text typed, or its action
345    /// chosen.
346    fn edit(&mut self, row: &KeyRow) {
347        let place = self.place(Some(row));
348        let key = row.key.clone();
349        self.doing = match &row.does {
350            Some(Macro::Act(action)) => {
351                let mut menu = actions(&key);
352                menu.marked = Action::ALL.iter().position(|each| each == action);
353                Doing::Choosing { key, place, menu }
354            }
355            Some(Macro::Fill(text)) => Doing::Typing {
356                key,
357                place,
358                kind: Kind::Fill,
359                line: LineEdit::of(text),
360            },
361            Some(Macro::Hotbar { bar, .. }) => {
362                self.note = Some(format!(
363                    "{key} presses a button of {bar}: change it on the hotbar."
364                ));
365                Doing::Looking
366            }
367            Some(Macro::Run(name)) => {
368                self.note = Some(format!(
369                    "{key} runs the macro {name}: change it in the windows' macro editor."
370                ));
371                Doing::Looking
372            }
373            Some(Macro::Send(text)) => Doing::Typing {
374                key,
375                place,
376                kind: Kind::Send,
377                line: LineEdit::of(&binding::shown(text)),
378            },
379            None => Doing::Typing {
380                key,
381                place,
382                kind: Kind::Send,
383                line: LineEdit::default(),
384            },
385        };
386    }

388    /// A key pressed while one is waited for: a new key's macro typed
389    /// next, or the moving key's macro moved to it.
390    fn waiting(&mut self, key: &Key, moving: Option<String>) -> Asked {
391        if key.name == "Escape" && key.mods == Mods::default() {
392            self.note = None;
393            return Asked::Nothing;
394        }
395        let pressed = Chord::new(&key.name, key.mods.0);
396        let chord = match pressed.refused() {
397            Some(why) => Err(why),
398            None => Chord::parse(&pressed.written()),
399        };
400        let chord = match chord {
401            Ok(chord) => chord,
402            Err(why) => {
403                self.note = Some(why);
404                self.doing = Doing::Waiting { moving };
405                return Asked::Nothing;
406            }
407        };
408        let written = chord.written();
409        let rows = self.files.rows(self.set);
410        let taken = rows
411            .iter()
412            .position(|row| row.key == written && row.does.is_some());
413        match (moving, taken) {
414            (Some(from), _) if from == written => {
415                self.note = None;
416                Asked::Nothing
417            }
418            (Some(_), Some(at)) => {
419                self.note = Some(format!("{written} already {}.", said(&rows[at])));
420                Asked::Nothing
421            }
422            (None, Some(at)) => {
423                self.marked = at;
424                self.note = Some(format!(
425                    "{written} already {}: change what it does.",
426                    said(&rows[at])
427                ));
428                self.edit(&rows[at]);
429                Asked::Nothing
430            }
431            (None, None) => {
432                self.note = None;
433                self.doing = Doing::Typing {
434                    key: written,
435                    place: self.place(None),
436                    kind: Kind::Send,
437                    line: LineEdit::default(),
438                };
439                Asked::Nothing
440            }
441            (Some(from), None) => {
442                let Some(row) = rows.iter().find(|row| row.key == from) else {
443                    return Asked::Nothing;
444                };
445                let Some(does) = row.does.clone() else {
446                    return Asked::Nothing;
447                };
448                self.note = None;
449                Asked::Change(KeyChange::Bind {
450                    key: written,
451                    does,
452                    was: Some(from),
453                    place: self.place(Some(row)),
454                })
455            }
456        }
457    }

459    /// A key while what `bound` does is typed: Enter binds it, Tab steps
460    /// to the next kind, Escape stops, the rest edit it.
461    fn typing(
462        &mut self,
463        key: &Key,
464        (bound, place, kind): (String, Place, Kind),
465        mut line: LineEdit,
466    ) -> Asked {
467        match key.name.as_str() {
468            "Escape" => {
469                self.note = None;
470                Asked::Nothing
471            }
472            "Tab" => {
473                self.doing = match kind.next() {
474                    Kind::Act => Doing::Choosing {
475                        menu: actions(&bound),
476                        key: bound,
477                        place,
478                    },
479                    next => Doing::Typing {
480                        key: bound,
481                        place,
482                        kind: next,
483                        line,
484                    },
485                };
486                Asked::Nothing
487            }
488            "Enter" | "NumpadEnter" => {
489                let text = line.text().to_owned();
490                let does = kind.with(&text);
491                let refused = if text.trim().is_empty() {
492                    Some(format!("{bound} does something: Delete unbinds it."))
493                } else {
494                    does.check().err().map(|why| format!("{bound}: {why}."))
495                };
496                if let Some(why) = refused {
497                    self.note = Some(why);
498                    self.doing = Doing::Typing {
499                        key: bound,
500                        place,
501                        kind,
502                        line,
503                    };
504                    return Asked::Nothing;
505                }
506                self.note = None;
507                let same = self
508                    .files
509                    .rows(self.set)
510                    .iter()
511                    .any(|row| row.key == bound && row.does.as_ref() == Some(&does));
512                if same {
513                    return Asked::Nothing;
514                }
515                Asked::Change(KeyChange::Bind {
516                    key: bound,
517                    does,
518                    was: None,
519                    place,
520                })
521            }
522            _ => {
523                line.key(key);
524                self.doing = Doing::Typing {
525                    key: bound,
526                    place,
527                    kind,
528                    line,
529                };
530                Asked::Nothing
531            }
532        }
533    }

535    /// What was asked, done: a change made to its file, and what it did
536    /// said.
537    fn made(&mut self, asked: Asked) -> Next {
538        match asked {
539            Asked::Nothing => Next::Stay,
540            Asked::Close => Next::Close,
541            Asked::Change(change) => {
542                self.said = Some(self.files.change(&change).unwrap_or_else(|why| why));
543                let last = self.files.rows(self.set).len().saturating_sub(1);
544                self.marked = self.marked.min(last);
545                Next::Stay
546            }
547        }
548    }

550    /// The editor's heading: whose keys, which set.
551    fn heading(&self) -> String {
552        let set = if self.set == 0 {
553            "set 0, always in use".to_owned()
554        } else {
555            format!("set {}", self.set)
556        };
557        if let (Some(name), Some(mine)) = (self.files.name(), self.files.mine()) {
558            let using = match mine.chosen {
559                0 => String::new(),
560                chosen => format!("; set {chosen} in use over set 0"),
561            };
562            return format!("{name}'s own keys over every character's: {set}{using}");
563        }
564        let numpad = if self.files.every.numpad_always() {
565            "; the numpad sends its keys with NumLock on too"
566        } else {
567            ""
568        };
569        format!("Every character's keys: {set}{numpad}")
570    }

585    /// The rows drawn in `area`, scrolled so the marked one shows.
586    fn draw_rows(&mut self, buf: &mut Buffer, area: Rect, rows: &[KeyRow], colors: &Colors) {
587        self.drawn.clear();
588        self.visible = usize::from(area.height).max(1);
589        if self.marked < self.top {
590            self.top = self.marked;
591        } else if self.marked >= self.top + self.visible {
592            self.top = self.marked + 1 - self.visible;
593        }
594        let widths = columns(area.width);
595        let character = self.character();
596        for (row_y, (at, row)) in rows
597            .iter()
598            .enumerate()
599            .skip(self.top)
600            .take(self.visible)
601            .enumerate()
602        {
603            let y = area.y + u16::try_from(row_y).unwrap_or(0);
604            self.drawn.push((y, at));
605            let marked = at == self.marked;
606            let base = if marked {
607                colors.chosen()
608            } else {
609                colors.canvas()
610            };
611            if marked {
612                buf.set_style(Rect::new(area.x, y, area.width, 1), base);
613            }
614            let muted = if marked { base } else { colors.muted() };
615            let cells = [
616                (format!(" {}", row.key), base.add_modifier(Modifier::BOLD)),
617                (said(row), if row.does.is_some() { base } else { muted }),
618                (whose(row, character).to_owned(), muted),
619                (
620                    row.default
621                        .as_ref()
622                        .map_or_else(|| "—".to_owned(), Macro::said),
623                    muted,
624                ),
625            ];
626            let mut x = area.x;
627            for ((text, style), width) in cells.into_iter().zip(widths) {
628                buf.set_stringn(x, y, text, usize::from(width.saturating_sub(1)), style);
629                x += width;
630            }
631        }
632    }

634    /// The line where a key is waited for or its macro typed: where the
635    /// cursor goes, while it is typed.
636    fn doing_line(
637        &self,
638        buf: &mut Buffer,
639        area: Rect,
640        y: u16,
641        colors: &Colors,
642    ) -> Option<(u16, u16)> {
643        let prompt = colors.text(Token::Accent);
644        match &self.doing {
645            Doing::Waiting { moving: None } => {
646                put(
647                    buf,
648                    area,
649                    y,
650                    "Press the key to bind (Escape stops)...",
651                    prompt,
652                );
653                None
654            }
655            Doing::Waiting { moving: Some(from) } => {
656                let text = format!("Press the key to move {from}'s macro to (Escape stops)...");
657                put(buf, area, y, &text, prompt);
658                None
659            }
660            Doing::Typing {
661                key, kind, line, ..
662            } => {
663                let label = format!("{key} {}: ", kind.label());
664                let took = u16::try_from(unicode_width::UnicodeWidthStr::width(label.as_str()))
665                    .unwrap_or(area.width)
666                    .min(area.width);
667                put(buf, area, y, &label, prompt);
668                let width = area.width - took;
669                let field = colors.text(Token::Text).bg(colors.color(Token::Inset));
670                let x = area.x + took;
671                buf.set_style(Rect::new(x, area.y + y, width, 1), field);
672                let (text, col) = line.shown(usize::from(width.saturating_sub(1)));
673                buf.set_stringn(x, area.y + y, &text, usize::from(width), field);
674                Some((x + u16::try_from(col).unwrap_or(0), area.y + y))
675            }
676            Doing::Looking | Doing::Choosing { .. } => None,
677        }
678    }

681/// The actions to choose among, for `key`.

697/// Whose `row`'s key is, in a word: on a character's keys (`character`),
698/// its own or everyone's; else Hydra's, the player's own, or Hydra's
699/// changed.
700fn whose(row: &KeyRow, character: bool) -> &'static str {
701    match (&row.does, row.from) {
702        (None, _) => "unbound",
703        (Some(_), None) => "Hydra's",
704        (Some(_), Some(Whose::Character)) => "own",
705        (Some(_), Some(Whose::Every)) if character => "everyone's",
706        (Some(does), Some(Whose::Every)) => match &row.default {
707            None => "yours",
708            Some(default) if default == does => "Hydra's",
709            Some(_) => "changed",
710        },
711    }
712}
713
714/// The columns' widths in `width`: the key, what it does, whose, and
715/// Hydra's default.
716fn columns(width: u16) -> [u16; 4] {
717    let key = 18.min(width / 4);
718    let whose = 12.min(width / 6);
719    let default = (width / 4).min(32);
720    let does = width.saturating_sub(key + whose + default);
721    [key, does, whose, default]
722}
723
724/// `text` on line `y` of `area`, cut at its edge.
725fn put(buf: &mut Buffer, area: Rect, y: u16, text: &str, style: Style) {
726    if y < area.height {
727        buf.set_stringn(area.x, area.y + y, text, usize::from(area.width), style);
728    }
729}

731impl Overlay for KeysEditor {
732    fn title(&self) -> String {
733        match self.files.name() {
734            Some(name) => format!("Keys: {name}'s"),
735            None => "Keys: every character's".to_owned(),
736        }
737    }
738
739    fn key(&mut self, key: &Key, _context: &Context) -> Next {
740        let asked = self.pressed(key);
741        self.made(asked)
742    }
743
744    fn paste(&mut self, text: &str, _context: &Context) -> Next {
745        if let Doing::Typing { line, .. } = &mut self.doing {
746            line.insert(text);
747        }
748        Next::Stay
749    }
750
751    fn click(&mut self, col: u16, row: u16, context: &Context) -> Next {
752        if let Doing::Choosing { menu, .. } = &mut self.doing {
753            return match menu.click(col, row) {
754                Done::Chose(action) => {
755                    let Doing::Choosing { key, place, .. } =
756                        std::mem::replace(&mut self.doing, Doing::Looking)
757                    else {
758                        return Next::Stay;
759                    };
760                    self.made(Asked::Change(KeyChange::Bind {
761                        key,
762                        does: Macro::Act(action),
763                        was: None,
764                        place,
765                    }))
766                }
767                Done::Closed => {
768                    self.doing = Doing::Looking;
769                    Next::Stay
770                }
771                Done::Open => Next::Stay,
772            };
773        }
774        if !matches!(self.doing, Doing::Looking) {
775            return Next::Stay;
776        }
777        match self.drawn.iter().find(|(y, _)| *y == row) {
778            Some(&(_, at)) if at == self.marked => self.key(&Key::named("Enter"), context),
779            Some(&(_, at)) => {
780                self.marked = at;
781                Next::Stay
782            }
783            None => Next::Stay,
784        }
785    }
786
787    fn wheel(&mut self, up: bool) {
788        if matches!(self.doing, Doing::Looking) {
789            let last = self.files.rows(self.set).len().saturating_sub(1);
790            self.marked = if up {
791                self.marked.saturating_sub(3)
792            } else {
793                (self.marked + 3).min(last)
794            };
795        }
796    }
797
798    fn size(&self) -> (u16, u16) {
799        (90, 90)
800    }
801
802    fn draw(
803        &mut self,
804        buf: &mut Buffer,
805        area: Rect,
806        (_, colors, hits): (&Context, &Colors, &mut Hits),
807    ) -> Option<(u16, u16)> {
808        if area.height < 6 {
809            return None;
810        }
811        put(buf, area, 0, &self.heading(), colors.text(Token::Accent));
812        put(buf, area, 1, &self.hint(), colors.text(Token::Link));
813        let widths = columns(area.width);
814        let mut x = area.x;
815        for (heading, width) in [" Key", "What it does", "Whose", "Hydra's default"]
816            .into_iter()
817            .zip(widths)
818        {
819            buf.set_stringn(
820                x,
821                area.y + 2,
822                heading,
823                usize::from(width.saturating_sub(1)),
824                colors.muted().add_modifier(Modifier::BOLD),
825            );
826            x += width;
827        }
828        let rows = self.files.rows(self.set);
829        self.marked = self.marked.min(rows.len().saturating_sub(1));
830        let rows_at = Rect::new(area.x, area.y + 3, area.width, area.height - 5);
831        self.draw_rows(buf, rows_at, &rows, colors);
832        let cursor = self.doing_line(buf, area, area.height - 2, colors);
833        let wrong = self.files.wrong();
834        let (last, style) = match (&self.note, &self.said, wrong.first()) {
835            (Some(note), _, _) => (note.clone(), colors.text(Token::Wrong)),
836            (None, Some(said), _) => (said.clone(), colors.text(Token::Hydra)),
837            (None, None, Some(wrong)) => (wrong.clone(), colors.text(Token::Wrong)),
838            (None, None, None) => match &self.doing {
839                Doing::Typing { .. } => (
840                    "Enter binds · Tab: send, fill or an action · Escape stops · \\r between \
841                     commands, s1.5 waits"
842                        .to_owned(),
843                    colors.muted(),
844                ),
845                _ => (String::new(), colors.muted()),
846            },
847        };
848        put(buf, area, area.height - 1, &last, style);
849        if let Doing::Choosing { menu, .. } = &mut self.doing {
850            menu.draw(buf, area, colors, hits);
851            return None;
852        }
853        cursor
854    }
855}

1//! The theme, over a terminal's screen: every theme there is, built in and
2//! in the themes folder, the one marked shown in its own colours -- a
3//! line of the game's text on its ground, and a swatch of each main colour
4//! -- before it is worn.
5//!
6//! Up and Down mark a theme. Enter makes it Hydra's: every window, the web
7//! page and each terminal wear it. Over a character's screen, `c` makes it
8//! that character's own, worn by its window over Hydra's, asked of the
9//! binary as the character's Theme page would ask it.
10

54/// The id of a character's *Theme* page, and its theme's key on it.

60/// The theme picker opened over seat `seat`'s screen or the hub's.
61#[must_use]
62pub fn open(seat: Option<u32>, context: &Context) -> Next {
63    Next::Instead(Box::new(ThemePicker::new(seat, context)))
64}

66/// The theme picker.
67#[derive(Debug)]
68pub struct ThemePicker {
69    themes: Themes,
70    /// Every theme's name, the built-ins first.
71    names: Vec<String>,
72    /// The theme marked.
73    marked: usize,
74    /// The first name in sight.
75    top: usize,
76    /// The character whose screen it was opened over, as the roster names
77    /// it (`GAME:Name`), and its name.
78    whose: Option<(String, String)>,
79    /// What the last choice did.
80    said: Option<String>,
81    /// When a character's theme was last asked of the binary.
82    asked: Option<Instant>,
83    /// Where each name was drawn last, for a click.
84    drawn: Vec<(u16, usize)>,
85}

87impl ThemePicker {
88    /// The themes in the data folder's themes folder, the one Hydra wears
89    /// marked.
90    #[must_use]
91    pub fn new(seat: Option<u32>, context: &Context) -> Self {
92        let themes = Themes::load(&context.data.join("themes"));
93        let names = themes.names();
94        let worn = context.sessions.own().theme().to_owned();
95        let marked = names
96            .iter()
97            .position(|name| name.eq_ignore_ascii_case(&worn))
98            .unwrap_or(0);
99        let seats = context.sessions.seats();
100        let whose = seat
101            .and_then(|id| crate::seat::find(&seats, id))
102            .map(|seat| (format!("{}:{}", seat.game, seat.name), seat.name.clone()));
103        Self {
104            themes,
105            names,
106            marked,
107            top: 0,
108            whose,
109            said: None,
110            asked: None,
111            drawn: Vec::new(),
112        }
113    }
114
115    /// The theme marked, made Hydra's.
116    fn wear(&mut self, context: &Context) {
117        let Some(name) = self.names.get(self.marked) else {
118            return;
119        };
120        let said = context.sessions.own().change("theme", Some(name));
121        self.said = Some(said.unwrap_or_else(|why| why));
122        self.asked = None;
123    }
124
125    /// The theme marked, made the character's own.
126    fn wear_theirs(&mut self, context: &Context) {
127        let (Some(name), Some((roster, character))) = (self.names.get(self.marked), &self.whose)
128        else {
129            self.said =
130                Some("Open the themes over a character's screen to give it one.".to_owned());
131            return;
132        };
133        let change = Change {
134            character: roster.clone(),
135            page: THEME_PAGE.to_owned(),
136            key: "theme".to_owned(),
137            to: Some(name.clone()),
138        };
139        context
140            .sessions
141            .ask(HubRequest::Change(change), &context.runtime);
142        self.said = Some(format!("{character}'s window is to wear {name}."));
143        self.asked = Some(Instant::now());
144    }
145
146    /// The themes' names drawn down the left of `area`, the one worn
147    /// marked `●`, scrolled so the one marked shows, a line at its right:
148    /// how wide it took.
149    fn draw_names(&mut self, buf: &mut Buffer, area: Rect, worn: &str, colors: &Colors) -> u16 {
150        let widest = self
151            .names
152            .iter()
153            .map(|name| name.width() + 4)
154            .max()
155            .unwrap_or(12);
156        let list_width = u16::try_from(widest)
157            .unwrap_or(u16::MAX)
158            .clamp(12, (area.width / 3).max(12));
159        let visible = usize::from(area.height).max(1);
160        if self.marked < self.top {
161            self.top = self.marked;
162        } else if self.marked >= self.top + visible {
163            self.top = self.marked + 1 - visible;
164        }
165        self.drawn.clear();
166        for (row, (at, name)) in self
167            .names
168            .iter()
169            .enumerate()
170            .skip(self.top)
171            .take(visible)
172            .enumerate()
173        {
174            let y = area.y + u16::try_from(row).unwrap_or(0);
175            let marked = at == self.marked;
176            let mut style = if marked {
177                colors.chosen()
178            } else {
179                colors.canvas()
180            };
181            if marked {
182                buf.set_style(Rect::new(area.x, y, list_width, 1), style);
183            }
184            let is_worn = name.eq_ignore_ascii_case(worn);
185            if is_worn {
186                style = style.add_modifier(Modifier::BOLD);
187            }
188            let mark = if is_worn { "● " } else { "  " };
189            buf.set_stringn(
190                area.x,
191                y,
192                format!("{mark}{name}"),
193                usize::from(list_width),
194                style,
195            );
196            self.drawn.push((y, at));
197        }
198        for y in area.y..area.y + area.height {
199            buf.set_string(area.x + list_width, y, "│", colors.border(false));
200        }
201        list_width
202    }
203
204    /// The theme marked's swatches and a line of text in its colours, in
205    /// `area`.
206    fn preview(&self, buf: &mut Buffer, area: Rect, colors: &Colors) {
207        let Some(name) = self.names.get(self.marked) else {
208            return;
209        };
210        let palette = match self.themes.outfit(name) {
211            Ok(outfit) => outfit.palette,
212            Err(why) => {
213                say(buf, area, 0, &why, colors, Token::Wrong);
214                return;
215            }
216        };
217        let on = |token: Token| {
218            Style::new()
219                .fg(rgb(palette.get(token)))
220                .bg(rgb(palette.get(Token::Canvas)))
221        };
222        let sample = Rect::new(area.x, area.y, area.width, 4.min(area.height));
223        buf.set_style(sample, Style::new().bg(rgb(palette.get(Token::Canvas))));
224        let lines: [&[(&str, Token)]; 4] = [
225            &[(" [Town Square, Central]", Token::RoomName)],
226            &[
227                (" You also see ", Token::Text),
228                ("a kobold", Token::Creature),
229                (" and ", Token::Text),
230                ("a lump of coal", Token::Object),
231                (".", Token::Text),
232            ],
233            &[
234                (" Ashryn", Token::Player),
235                (" says, ", Token::Text),
236                ("\"Well met.\"", Token::Speech),
237            ],
238            &[
239                (" Obvious paths: ", Token::Muted),
240                ("north", Token::Link),
241                (", ", Token::Muted),
242                ("out", Token::Link),
243            ],
244        ];
245        for (at, pieces) in lines.iter().enumerate().take(usize::from(sample.height)) {
246            let y = area.y + u16::try_from(at).unwrap_or(0);
247            let mut x = area.x;
248            for (text, token) in *pieces {
249                let left = usize::from((area.x + area.width).saturating_sub(x));
250                let (end, _) = buf.set_stringn(x, y, text, left, on(*token));
251                x = end;
252            }
253        }
254        swatches(
255            buf,
256            Rect::new(
257                area.x,
258                area.y + 5,
259                area.width,
260                area.height.saturating_sub(5),
261            ),
262            &palette,
263            colors,
264        );
265    }
266}

268/// A swatch of each colour [`SHOWN`], in as many columns as fit in `area`.
269fn swatches(buf: &mut Buffer, area: Rect, palette: &Palette, colors: &Colors) {
270    const WIDE: u16 = 26;
271    if area.height == 0 {
272        return;
273    }
274    let columns = (area.width / WIDE).max(1);
275    let per = SHOWN.len().div_ceil(usize::from(columns));
276    for (at, token) in SHOWN.into_iter().enumerate() {
277        let (column, row) = (at / per.max(1), at % per.max(1));
278        let Ok(row) = u16::try_from(row) else {
279            continue;
280        };
281        let x = area.x + u16::try_from(column).unwrap_or(0) * WIDE;
282        if row >= area.height || x + 4 > area.x + area.width {
283            continue;
284        }
285        let y = area.y + row;
286        let color = palette.get(token);
287        buf.set_string(x, y, "    ", Style::new().bg(rgb(color)));
288        let named = format!(" {} {}", token.name(), hex(color));
289        let left =
290            usize::from((area.x + area.width).saturating_sub(x + 4)).min(usize::from(WIDE - 4));
291        buf.set_stringn(x + 4, y, named, left, colors.muted());
292    }

295impl Overlay for ThemePicker {
296    fn title(&self) -> String {
297        "Theme".to_owned()
298    }
299
300    fn key(&mut self, key: &Key, context: &Context) -> Next {
301        let last = self.names.len().saturating_sub(1);
302        match key.name.as_str() {
303            "Escape" => return Next::Close,
304            "ArrowUp" => self.marked = self.marked.saturating_sub(1),
305            "ArrowDown" => self.marked = (self.marked + 1).min(last),
306            "Home" | "PageUp" => self.marked = 0,
307            "End" | "PageDown" => self.marked = last,
308            "Enter" | "NumpadEnter" => self.wear(context),
309            _ if matches!(key.text, Some('c' | 'C')) => self.wear_theirs(context),
310            _ => {}
311        }
312        Next::Stay
313    }
314
315    fn click(&mut self, _col: u16, row: u16, context: &Context) -> Next {
316        if let Some(&(_, at)) = self.drawn.iter().find(|(y, _)| *y == row) {
317            if at == self.marked {
318                self.wear(context);
319            } else {
320                self.marked = at;
321            }
322        }
323        Next::Stay
324    }
325
326    fn wheel(&mut self, up: bool) {
327        let last = self.names.len().saturating_sub(1);
328        self.marked = if up {
329            self.marked.saturating_sub(1)
330        } else {
331            (self.marked + 1).min(last)
332        };
333    }
334
335    fn size(&self) -> (u16, u16) {
336        (80, 70)
337    }
338
339    fn due(&self) -> Option<Duration> {
340        self.asked
341            .filter(|at| at.elapsed() < ANSWER)
342            .map(|_| Duration::from_millis(250))
343    }
344
345    fn draw(
346        &mut self,
347        buf: &mut Buffer,
348        area: Rect,
349        (context, colors, _): (&Context, &Colors, &mut Hits),
350    ) -> Option<(u16, u16)> {
351        if area.height < 4 || area.width < 20 {
352            return None;
353        }
354        let worn = context.sessions.own().theme().to_owned();
355        let body = area.height - 2;
356        let list_width = self.draw_names(
357            buf,
358            Rect::new(area.x, area.y, area.width, body),
359            &worn,
360            colors,
361        );
362        let preview = Rect::new(
363            area.x + list_width + 2,
364            area.y,
365            area.width.saturating_sub(list_width + 2),
366            body,
367        );
368        self.preview(buf, preview, colors);
369        let hint = if self.whose.is_some() {
370            "Enter: Hydra's theme · c: this character's own · ● worn"
371        } else {
372            "Enter: Hydra's theme, every window and terminal · ● worn"
373        };
374        say(buf, area, body, hint, colors, Token::Muted);
375        let foot = match (&self.said, self.themes.problems.first()) {
376            (Some(said), _) => {
377                let answer = self
378                    .asked
379                    .filter(|at| at.elapsed() < ANSWER)
380                    .and(context.sessions.glance(Instant::now()).said);
381                Some((answer.unwrap_or_else(|| said.clone()), Token::Hydra))
382            }
383            (None, Some(problem)) => Some((problem.clone(), Token::Wrong)),
384            (None, None) => None,
385        };
386        if let Some((text, token)) = foot {
387            say(buf, area, body + 1, &text, colors, token);
388        }
389        None
390    }
391}

398    /// The built-in themes and one in the themes folder are listed, the one
399    /// worn marked; Enter makes the one marked Hydra's.
400    #[test]
401    fn the_themes_are_listed_and_one_worn() {
402        let dir = std::env::temp_dir().join(format!("hydra-tui-themes-{}", std::process::id()));
403        let _ = std::fs::remove_dir_all(&dir);
404        std::fs::create_dir_all(dir.join("themes")).expect("made");
405        std::fs::write(
406            dir.join("themes").join("Ember.toml"),
407            Themes::built_in()
408                .get("Despana")
409                .expect("built in")
410                .to_toml()
411                .replace("Despana", "Ember"),
412        )
413        .expect("written");
414        let context = test_context(&dir);
415        let mut picker = ThemePicker::new(None, &context);
416        assert!(
417            picker
418                .names
419                .starts_with(&["Despana".to_owned(), "Light".to_owned()])
420        );
421        assert!(
422            picker.names.iter().any(|name| name == "Ember"),
423            "{:?}",
424            picker.names
425        );
426        assert_eq!(picker.names[picker.marked], "Despana", "the one worn");
427        let _ = picker.key(&Key::named("ArrowDown"), &context);
428        let _ = picker.key(&Key::named("Enter"), &context);
429        assert_eq!(context.sessions.own().theme(), "Light");
430        assert_eq!(picker.said.as_deref(), Some("The theme is Light."));
431        let area = Rect::new(0, 0, 90, 30);
432        let mut buf = Buffer::empty(area);
433        picker.draw(
434            &mut buf,
435            area,
436            (&context, &Colors::default(), &mut Hits::default()),
437        );
438        let row: String = (0..30).map(|x| buf[(x, 1)].symbol().to_owned()).collect();
439        assert!(row.contains("● Light"), "{row:?}");
440        let _ = std::fs::remove_dir_all(&dir);
441    }

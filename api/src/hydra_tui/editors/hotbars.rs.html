1//! The hotbars editor, over a terminal's screen: over a character's, its
2//! bars -- every character's with its own over them, each marked whose --
3//! and over the hub's, every character's alone. A bar's name, whose it is,
4//! its grid and its buttons; a button is a slot holding a macro, by its
5//! name, which says what the button shows and sends. A bar is saved whole
6//! to the file it is kept in, the rest of that file as it was, and written
7//! at once, so a key pressing it sends what was saved.
8//!
9//! In the bars the arrows choose; Enter opens one; `n` begins a new one;
10//! Delete takes one away once `y` says so.
11//!
12//! In a bar the arrows and Tab go between its rows -- its name, whose it
13//! is, its rows and columns, then each button and *Add a button*; Enter on
14//! a button opens it, on *Add a button* adds one; Delete takes the button
15//! chosen off; Space or the left and right arrows turn whose it is; Ctrl+S
16//! saves the bar; Escape goes back to the bars, what was not saved let go.
17//!
18//! In a button the arrows and Tab go between its macro and its slot, each
19//! typed into; Enter or Escape goes back to the bar, Ctrl+S saves it.
20

37/// The hotbars editor opened over seat `seat`'s screen, for its
38/// character's bars, or the hub's, for every character's.
39#[must_use]
40pub fn open(seat: Option<u32>, context: &Context) -> Next {
41    let seats = context.sessions.seats();
42    let found = seat.and_then(|id| crate::seat::find(&seats, id));
43    let editor = match found {
44        Some(seat) => {
45            let [every, own] = hotbars_paths(&context.data, seat.instance.as_deref(), &seat.name);
46            Editor::new(every, Some((seat.name.clone(), own)))
47        }
48        None => Editor::new(context.data.join(hydra_ui::hotbar::FILE), None),
49    };
50    Next::Instead(Box::new(editor))
51}

76/// A button as its form edits it.
77#[derive(Clone, Debug, PartialEq, Eq)]
78struct ButtonForm {
79    /// The button's place in the bar's list.
80    index: usize,
81    /// The row chosen.
82    at: usize,
83    /// Its macro and its slot.
84    fields: [LineEdit; 2],
85}
86
87impl ButtonForm {
88    /// The button `button`, at `index` in its bar, in the form.
89    fn of(index: usize, button: &Button) -> Self {
90        Self {
91            index,
92            at: 0,
93            fields: [
94                LineEdit::of(button.macro_name.as_deref().unwrap_or_default()),
95                LineEdit::of(
96                    &button
97                        .slot
98                        .map_or_else(String::new, |slot| slot.to_string()),
99                ),
100            ],
101        }
102    }
103
104    /// `button` as the form says.
105    fn apply(&self, button: &mut Button) {
106        let given = |edit: &LineEdit| {
107            let text = edit.text().trim();
108            (!text.is_empty()).then(|| text.to_owned())
109        };
110        let [named, slot] = &self.fields;
111        button.macro_name = given(named);
112        button.id = button.macro_name.clone().unwrap_or_default();
113        button.slot = slot.text().trim().parse().ok();
114    }
115}

117/// A bar as its form edits it.
118#[derive(Clone, Debug, PartialEq)]
119struct BarForm {
120    /// The bar as kept: its name, and whether it is the character's own;
121    /// `None` for a new one.
122    was: Option<(String, bool)>,
123    /// The bar as it was when chosen or last saved, and whose.
124    kept: (Bar, bool),
125    /// The bar, its buttons as the form has them.
126    draft: Bar,
127    name: LineEdit,
128    /// Kept in the character's own file.
129    own: bool,
130    rows: LineEdit,
131    columns: LineEdit,
132    /// The row chosen: the bar's own, then each button, then adding one.
133    at: usize,
134    /// The button open, while one is.
135    button: Option<ButtonForm>,
136}
137
138impl BarForm {
139    /// `bar`, the character's own when `own`, in the form; `kept` when it
140    /// is in a file.
141    fn of(bar: &Bar, own: bool, kept: bool) -> Self {
142        let count =
143            |count: Option<u16>| LineEdit::of(&count.map_or_else(String::new, |n| n.to_string()));
144        Self {
145            was: kept.then(|| (bar.name.clone(), own)),
146            kept: (if kept { bar.clone() } else { Bar::default() }, own),
147            draft: bar.clone(),
148            name: LineEdit::of(&bar.name),
149            own,
150            rows: count(bar.rows),
151            columns: count(bar.columns),
152            at: 0,
153            button: None,
154        }
155    }
156
157    /// How many rows it has.
158    const fn count(&self) -> usize {
159        BAR_ROWS.len() + self.draft.buttons.len() + 1
160    }
161
162    /// The button open written into the bar.
163    fn commit(&mut self) {
164        if let Some(form) = &self.button
165            && let Some(button) = self.draft.buttons.get_mut(form.index)
166        {
167            form.apply(button);
168        }
169    }
170
171    /// The bar as the form says.
172    ///
173    /// # Errors
174    ///
175    /// A count that is not one.
176    fn bar(&self) -> Result<Bar, String> {
177        let mut bar = self.draft.clone();
178        if let Some(form) = &self.button
179            && let Some(button) = bar.buttons.get_mut(form.index)
180        {
181            form.apply(button);
182        }
183        self.name.text().trim().clone_into(&mut bar.name);
184        bar.rows = count_of(&self.rows, "Rows")?;
185        bar.columns = count_of(&self.columns, "Columns")?;
186        Ok(bar)
187    }
188
189    /// Whether it holds changes not saved.
190    fn changed(&self) -> bool {
191        self.bar().map_or(true, |bar| (bar, self.own) != self.kept)
192    }
193
194    /// A new button added, and opened, its macro to be named.
195    fn add_button(&mut self) {
196        self.draft.buttons.push(Button::default());
197        let index = self.draft.buttons.len() - 1;
198        self.at = BAR_ROWS.len() + index;
199        self.button = Some(ButtonForm::of(index, &self.draft.buttons[index]));
200    }
201}
202
203/// A number of rows or columns as typed: none for nothing typed.
204fn count_of(edit: &LineEdit, what: &str) -> Result<Option<u16>, String> {
205    let text = edit.text().trim();
206    if text.is_empty() {
207        return Ok(None);
208    }
209    text.parse::<u16>()
210        .ok()
211        .filter(|count| (1..=40).contains(count))
212        .map(Some)
213        .ok_or_else(|| format!("{what} is a number from 1 to 40, or none."))
214}

216/// The hotbars editor.
217#[derive(Debug)]
218pub struct Editor {
219    /// Every character's file.
220    every: Kept,
221    /// The character's own file, and its name, over a character.
222    own: Option<(String, Kept)>,
223    /// The bar chosen in the list, by its place there.
224    marked: usize,
225    /// The bar open, while one is.
226    bar: Option<BarForm>,
227    /// The bar asked about before it is taken away, and whether it is the
228    /// character's own.
229    removing: Option<(String, bool)>,
230    /// What was done, or why not, in its colour.
231    notice: Option<(String, Token)>,
232}
233
234impl Editor {
235    /// The editor over every character's file at `every`, and a
236    /// character's own, by its name, when there is one.
237    fn new(every: PathBuf, own: Option<(String, PathBuf)>) -> Self {
238        let mut editor = Self {
239            every: Kept::read(every),
240            own: own.map(|(name, path)| (name, Kept::read(path))),
241            marked: 0,
242            bar: None,
243            removing: None,
244            notice: None,
245        };
246        let problems: Vec<String> = [
247            Some(&editor.every),
248            editor.own.as_ref().map(|(_, kept)| kept),
249        ]
250        .into_iter()
251        .flatten()
252        .filter_map(|kept| kept.bars.as_ref().err().cloned())
253        .collect();
254        if !problems.is_empty() {
255            editor.notice = Some((problems.join("; "), Token::Wrong));
256        }
257        editor
258    }
259
260    /// The bars, each with whether it is the character's own: every
261    /// character's, the character's own in place of one of its name, or
262    /// after.
263    fn bars(&self) -> Vec<(Bar, bool)> {
264        let mut bars: Vec<(Bar, bool)> = self
265            .every
266            .bars
267            .as_ref()
268            .map(|file| file.bars.iter().map(|bar| (bar.clone(), false)).collect())
269            .unwrap_or_default();
270        if let Some((_, Kept { bars: Ok(own), .. })) = &self.own {
271            for bar in &own.bars {
272                match bars.iter_mut().find(|(had, _)| had.name == bar.name) {
273                    Some(had) => *had = (bar.clone(), true),
274                    None => bars.push((bar.clone(), true)),
275                }
276            }
277        }
278        bars
279    }
280
281    /// The file kept for the character (`own`) or every character.
282    fn kept_mut(&mut self, own: bool) -> Option<&mut Kept> {
283        if own {
284            self.own.as_mut().map(|(_, kept)| kept)
285        } else {
286            Some(&mut self.every)
287        }
288    }
289
290    /// `bar` kept in the character's own file (`own`) or every
291    /// character's, in place of the bar `was`, and taken out of the other
292    /// file when it moved: here at once, and written now.
293    fn keep(&mut self, bar: &Bar, own: bool, was: Option<&(String, bool)>) -> Result<(), String> {
294        let replacing = was
295            .filter(|(_, was_own)| *was_own == own)
296            .map(|(name, _)| name.as_str());
297        let kept = self
298            .kept_mut(own)
299            .ok_or("There is no character to keep it for.")?;
300        let file = match &mut kept.bars {
301            Ok(file) => file,
302            Err(why) => return Err(why.clone()),
303        };
304        let mut changed = file.clone();
305        put_bar(&mut changed, bar.clone(), replacing)?;
306        *file = changed;
307        save_bar(&kept.path, bar, replacing);
308        if let Some((name, was_own)) = was.filter(|(_, was_own)| *was_own != own)
309            && let Some(other) = self.kept_mut(*was_own)
310        {
311            if let Ok(file) = &mut other.bars {
312                take_bar(file, name);
313            }
314            remove_bar(&other.path, name);
315        }
316        hydra_board::store::flush();
317        Ok(())
318    }
319
320    /// The bar open saved.
321    fn save(&mut self) {
322        let Some(form) = &mut self.bar else {
323            return;
324        };
325        form.commit();
326        let (own, was) = (form.own, form.was.clone());
327        let saved = form.bar().and_then(|bar| {
328            self.keep(&bar, own, was.as_ref())?;
329            Ok(bar)
330        });
331        match saved {
332            Ok(bar) => {
333                self.notice = Some((format!("{} saved.", bar.name), Token::Text));
334                if let Some(form) = &mut self.bar {
335                    form.was = Some((bar.name.clone(), own));
336                    form.kept = (bar, own);
337                }
338            }
339            Err(why) => self.notice = Some((format!("Not saved: {why}"), Token::Wrong)),
340        }
341    }
342
343    /// The bar `name`, the character's own when `own`, taken away.
344    fn remove(&mut self, name: &str, own: bool) {
345        let Some(kept) = self.kept_mut(own) else {
346            return;
347        };
348        let said = match &mut kept.bars {
349            Ok(file) => {
350                take_bar(file, name);
351                remove_bar(&kept.path, name);
352                hydra_board::store::flush();
353                (format!("{name} taken away."), Token::Text)
354            }
355            Err(why) => (format!("Not taken away: {why}"), Token::Wrong),
356        };
357        self.notice = Some(said);
358    }
359
360    /// `key` pressed: asked whether to take a bar away, in the bar open, or
361    /// in the list.
362    fn pressed(&mut self, key: &Key) -> Next {
363        if let Some((name, own)) = self.removing.take() {
364            match key.text {
365                Some('y' | 'Y') => self.remove(&name, own),
366                Some('n' | 'N') => {}
367                _ if key.name == "Escape" => {}
368                _ => self.removing = Some((name, own)),
369            }
370            return Next::Stay;
371        }
372        if self.bar.is_some() {
373            return self.bar_key(key);
374        }
375        self.list_key(key)
376    }
377
378    /// A key in the list of bars.
379    fn list_key(&mut self, key: &Key) -> Next {
380        let bars = self.bars();
381        let last = bars.len().saturating_sub(1);
382        let marked = bars.get(self.marked.min(last));
383        match key.name.as_str() {
384            "Escape" => return Next::Close,
385            "ArrowUp" => self.marked = self.marked.saturating_sub(1),
386            "ArrowDown" => self.marked = (self.marked + 1).min(last),
387            "Enter" | "NumpadEnter" => {
388                if let Some((bar, own)) = marked {
389                    self.bar = Some(BarForm::of(bar, *own, true));
390                    self.notice = None;
391                }
392            }
393            "Delete" => {
394                self.removing = marked.map(|(bar, own)| (bar.name.clone(), *own));
395            }
396            _ if matches!(key.text, Some('n' | 'N')) => {
397                let mut name = "bar".to_owned();
398                let mut n = 1;
399                while bars
400                    .iter()
401                    .any(|(bar, _)| bar.name.eq_ignore_ascii_case(&name))
402                {
403                    n += 1;
404                    name = format!("bar{n}");
405                }
406                let bar = Bar {
407                    name,
408                    ..Bar::default()
409                };
410                self.bar = Some(BarForm::of(&bar, false, false));
411                self.notice = None;
412            }
413            _ => {}
414        }
415        Next::Stay
416    }

418    /// A key in the bar open.
419    fn bar_key(&mut self, key: &Key) -> Next {
420        if key.name == "KeyS" && key.mods == Mods(Mods::CTRL) {
421            self.save();
422            return Next::Stay;
423        }
424        let character = self.own.is_some();
425        let Some(form) = &mut self.bar else {
426            return Next::Stay;
427        };
428        if let Some(button) = &mut form.button {
429            if button_key(button, key) {
430                form.commit();
431                form.button = None;
432            }
433            return Next::Stay;
434        }
435        let count = form.count();
436        let shift = key.mods.has(Mods::SHIFT);
437        let button = form.at.checked_sub(BAR_ROWS.len());
438        match key.name.as_str() {
439            "Escape" => {
440                self.bar = None;
441                self.notice = None;
442            }
443            "ArrowUp" => form.at = (form.at + count - 1) % count,
444            "Tab" if shift => form.at = (form.at + count - 1) % count,
445            "ArrowDown" | "Tab" => form.at = (form.at + 1) % count,
446            "Enter" | "NumpadEnter" => match button {
447                Some(index) if index < form.draft.buttons.len() => {
448                    form.button = Some(ButtonForm::of(index, &form.draft.buttons[index]));
449                }
450                Some(_) => form.add_button(),
451                None => form.at = (form.at + 1) % count,
452            },
453            "Delete" if button.is_some_and(|index| index < form.draft.buttons.len()) => {
454                if let Some(index) = button {
455                    form.draft.buttons.remove(index);
456                }
457            }
458            _ => match form.at {
459                0 => {
460                    form.name.key(key);
461                }
462                1 => {
463                    let turns = matches!(key.name.as_str(), "Space" | "ArrowLeft" | "ArrowRight")
464                        || key.text == Some(' ');
465                    if turns && character {
466                        form.own = !form.own;
467                    }
468                }
469                2 => {
470                    form.rows.key(key);
471                }
472                3 => {
473                    form.columns.key(key);
474                }
475                _ => {}
476            },
477        }
478        Next::Stay
479    }

481    /// The list of bars drawn in `area`.
482    fn draw_list(&mut self, buf: &mut Buffer, area: Rect, colors: &Colors) {
483        let bars = self.bars();
484        self.marked = self.marked.min(bars.len().saturating_sub(1));
485        let whose = self.own.as_ref().map_or_else(
486            || "Every character's hotbars.".to_owned(),
487            |(name, _)| format!("{name}'s hotbars: every character's, and {name}'s own over them."),
488        );
489        say(buf, area, 0, &whose, colors, Token::Muted);
490        if bars.is_empty() {
491            say(
492                buf,
493                area,
494                2,
495                "No bars yet: n begins one.",
496                colors,
497                Token::Muted,
498            );
499        }
500        let height = usize::from(area.height.saturating_sub(3));
501        let skip = (self.marked + 1).saturating_sub(height);
502        for (row, (at, (bar, own))) in bars.iter().enumerate().skip(skip).take(height).enumerate() {
503            let y = area.y + 2 + u16::try_from(row).unwrap_or(0);
504            let chosen = at == self.marked;
505            let style = if chosen {
506                colors.chosen()
507            } else {
508                colors.canvas()
509            };
510            if chosen {
511                buf.set_style(Rect::new(area.x, y, area.width, 1), style);
512            }
513            let badge = if *own { "own" } else { "every" };
514            let buttons = bar.buttons.len();
515            let plural = if buttons == 1 { "" } else { "s" };
516            let pieces = [
517                Piece::new(format!(" {:<20} ", bar.name), style),
518                Piece::new(
519                    format!("[{badge}]  "),
520                    style.fg(colors.color(Token::Accent)),
521                ),
522                Piece::new(
523                    format!("{buttons} button{plural}"),
524                    style.fg(colors.color(Token::Muted)),
525                ),
526            ];
527            draw_row(
528                buf,
529                (area.x, y, area.width),
530                &pieces,
531                &mut Hits::default(),
532                None,
533            );
534        }
535        let foot = area.height.saturating_sub(1);
536        match (&self.removing, &self.notice) {
537            (Some((name, _)), _) => {
538                let asked = format!("Take the bar {name} away? y: yes   n: no");
539                say(buf, area, foot, &asked, colors, Token::Warning);
540            }
541            (None, Some((said, token))) => say(buf, area, foot, said, colors, *token),
542            (None, None) => say(
543                buf,
544                area,
545                foot,
546                "Enter opens · n a new bar · Del takes one away · Esc closes",
547                colors,
548                Token::Muted,
549            ),
550        }
551    }

553    /// The bar open drawn in `area`: where the cursor goes, on a row of
554    /// words.
555    fn draw_bar(&mut self, buf: &mut Buffer, area: Rect, colors: &Colors) -> Option<(u16, u16)> {
556        let character = self.own.as_ref().map(|(name, _)| name.clone());
557        let notice = self.notice.clone();
558        let form = self.bar.as_ref()?;
559        if let Some(button) = &form.button {
560            let on = form.draft.buttons.get(button.index);
561            return draw_button(
562                buf,
563                area,
564                (button, on, &form.draft.name),
565                (notice.as_ref(), colors),
566            );
567        }
568        let mut cursor = None;
569        let room = area
570            .width
571            .saturating_sub(u16::try_from(LABEL).unwrap_or(0) + 1);
572        let lines = form.lines(character.as_deref(), room, colors);
573        let height = usize::from(area.height.saturating_sub(1));
574        let line_of = lines
575            .iter()
576            .position(|(row, _)| *row == Some(form.at))
577            .unwrap_or(0);
578        let skip = (line_of + 1).saturating_sub(height);
579        for (row, (at, pieces)) in lines.iter().skip(skip).take(height).enumerate() {
580            let y = area.y + u16::try_from(row).unwrap_or(0);
581            draw_row(
582                buf,
583                (area.x, y, area.width),
584                pieces,
585                &mut Hits::default(),
586                None,
587            );
588            let edit = match at {
589                Some(0) => Some(&form.name),
590                Some(2) => Some(&form.rows),
591                Some(3) => Some(&form.columns),
592                _ => None,
593            };
594            if *at == Some(form.at)
595                && let Some(edit) = edit
596            {
597                let (_, col) = edit.shown(usize::from(room.saturating_sub(2)));
598                let x = area.x
599                    + u16::try_from(LABEL + 1).unwrap_or(0)
600                    + u16::try_from(col).unwrap_or(0);
601                cursor = Some((x, y));
602            }
603        }
604        let foot = area.height.saturating_sub(1);
605        if let Some((said, token)) = &notice {
606            say(buf, area, foot, said, colors, *token);
607        } else {
608            let unsaved = if form.changed() { "unsaved · " } else { "" };
609            let hint = format!(
610                "{unsaved}Ctrl+S saves · Enter opens a button · Del takes it off · Esc back"
611            );
612            say(buf, area, foot, &hint, colors, Token::Muted);
613        }
614        cursor
615    }
616}

618impl BarForm {
619    /// Its lines as drawn, `room` columns left for a row's value, over the
620    /// character called `character` when there is one: each with the row it
621    /// is, when it is one.
622    fn lines(
623        &self,
624        character: Option<&str>,
625        room: u16,
626        colors: &Colors,
627    ) -> Vec<(Option<usize>, Vec<Piece>)> {
628        let form = self;
629        let mut lines: Vec<(Option<usize>, Vec<Piece>)> = Vec::new();
630        for (at, label) in BAR_ROWS.iter().enumerate() {
631            let chosen = form.at == at;
632            let label_style = if chosen {
633                colors.chosen()
634            } else {
635                colors.muted()
636            };
637            let mut pieces = vec![Piece::new(format!(" {label:<LABEL$}"), label_style)];
638            let field = colors.text(Token::Text).bg(colors.color(Token::Inset));
639            let text = |edit: &LineEdit, hint: &str| {
640                let (shown, _) = edit.shown(usize::from(room.saturating_sub(2)));
641                vec![
642                    Piece::new(
643                        format!("{shown:<width$}", width = usize::from(room) / 2),
644                        field,
645                    ),
646                    Piece::new(format!(" {hint}"), colors.muted()),
647                ]
648            };
649            match at {
650                0 => pieces.extend(text(&form.name, "")),
651                1 => {
652                    let whose = match (&character, form.own) {
653                        (Some(name), true) => format!("‹ {name} only ›"),
654                        (Some(_), false) => "‹ every character ›".to_owned(),
655                        (None, _) => "every character".to_owned(),
656                    };
657                    pieces.push(Piece::new(whose, colors.text(Token::Accent)));
658                }
659                2 => pieces.extend(text(&form.rows, "none: one, more as the buttons need")),
660                _ => pieces.extend(text(&form.columns, "none: as many as it has buttons")),
661            }
662            lines.push((Some(at), pieces));
663        }
664        lines.push((
665            None,
666            vec![Piece::new(" Buttons:", colors.text(Token::Accent))],
667        ));
668        for (index, button) in form.draft.buttons.iter().enumerate() {
669            let at = BAR_ROWS.len() + index;
670            let style = if form.at == at {
671                colors.chosen()
672            } else {
673                colors.canvas()
674            };
675            let slot = button.slot.map_or_else(
676                || "the first free slot".to_owned(),
677                |slot| format!("slot {slot}"),
678            );
679            let said = format!(
680                "   {} — {slot}",
681                button.macro_name.as_deref().unwrap_or("no macro yet")
682            );
683            lines.push((Some(at), vec![Piece::new(said, style)]));
684        }
685        let add = form.count() - 1;
686        let style = if form.at == add {
687            colors.chosen()
688        } else {
689            colors.text(Token::Link)
690        };
691        lines.push((Some(add), vec![Piece::new("   + Add a button", style)]));
692        lines
693    }
694}

696/// `key` in a button's form: whether it goes back to the bar.
697fn button_key(form: &mut ButtonForm, key: &Key) -> bool {
698    let count = BUTTON_ROWS.len();
699    match key.name.as_str() {
700        "Escape" | "Enter" | "NumpadEnter" => return true,
701        "ArrowUp" => form.at = (form.at + count - 1) % count,
702        "Tab" if key.mods.has(Mods::SHIFT) => form.at = (form.at + count - 1) % count,
703        "ArrowDown" | "Tab" => form.at = (form.at + 1) % count,
704        _ => {
705            form.fields[form.at].key(key);
706        }
707    }
708    false
709}

711/// A button's form drawn in `area`, `button` the button as the bar has
712/// it, on the bar `bar`: where the cursor goes.
713fn draw_button(
714    buf: &mut Buffer,
715    area: Rect,
716    (form, button, bar): (&ButtonForm, Option<&Button>, &str),
717    (notice, colors): (Option<&(String, Token)>, &Colors),
718) -> Option<(u16, u16)> {
719    let title = format!("A button of {bar}");
720    say(buf, area, 0, &title, colors, Token::Accent);
721    let label_width = u16::try_from(LABEL + 1).unwrap_or(0);
722    let room = area.width.saturating_sub(label_width + 1);
723    let field = colors.text(Token::Text).bg(colors.color(Token::Inset));
724    let mut cursor = None;
725    for (at, label) in BUTTON_ROWS.iter().enumerate() {
726        let y = area.y + 2 + u16::try_from(at).unwrap_or(0);
727        if y >= area.y + area.height {
728            break;
729        }
730        let chosen = at == form.at;
731        let label_style = if chosen {
732            colors.chosen()
733        } else {
734            colors.muted()
735        };
736        buf.set_stringn(
737            area.x,
738            y,
739            format!(" {label:<LABEL$}"),
740            usize::from(label_width),
741            label_style,
742        );
743        let (shown, col) = form.fields[at].shown(usize::from(room));
744        buf.set_style(Rect::new(area.x + label_width, y, room, 1), field);
745        buf.set_stringn(area.x + label_width, y, shown, usize::from(room), field);
746        if chosen {
747            cursor = Some((area.x + label_width + u16::try_from(col).unwrap_or(0), y));
748        }
749    }
750    let after = 3 + u16::try_from(BUTTON_ROWS.len()).unwrap_or(0);
751    if button.is_some() {
752        say(
753            buf,
754            area,
755            after,
756            " What it shows and sends is its macro's; a slot left empty, the first free.",
757            colors,
758            Token::Muted,
759        );
760    }
761    let foot = area.height.saturating_sub(1);
762    match notice {
763        Some((said, token)) => say(buf, area, foot, said, colors, *token),
764        None => say(
765            buf,
766            area,
767            foot,
768            "Enter or Esc back to the bar · Ctrl+S saves",
769            colors,
770            Token::Muted,
771        ),
772    }
773    cursor
774}

776impl Overlay for Editor {
777    fn title(&self) -> String {
778        match &self.bar {
779            Some(form) if form.name.text().trim().is_empty() => "A new hotbar".to_owned(),
780            Some(form) => format!("Hotbar: {}", form.name.text().trim()),
781            None => "Hotbars".to_owned(),
782        }
783    }
784
785    fn key(&mut self, key: &Key, _context: &Context) -> Next {
786        self.pressed(key)
787    }
788
789    fn paste(&mut self, text: &str, _context: &Context) -> Next {
790        if let Some(form) = &mut self.bar {
791            if let Some(button) = &mut form.button {
792                button.fields[button.at].insert(text);
793            } else if form.at == 0 {
794                form.name.insert(text);
795            }
796        }
797        Next::Stay
798    }
799
800    fn wheel(&mut self, up: bool) {
801        match &mut self.bar {
802            Some(form) if form.button.is_none() => {
803                let count = form.count();
804                form.at = if up {
805                    form.at.saturating_sub(1)
806                } else {
807                    (form.at + 1).min(count - 1)
808                };
809            }
810            Some(_) => {}
811            None if up => self.marked = self.marked.saturating_sub(1),
812            None => self.marked += 1,
813        }
814    }
815
816    fn size(&self) -> (u16, u16) {
817        (80, 80)
818    }
819
820    fn draw(
821        &mut self,
822        buf: &mut Buffer,
823        area: Rect,
824        (_, colors, _): (&Context, &Colors, &mut Hits),
825    ) -> Option<(u16, u16)> {
826        if self.bar.is_some() {
827            self.draw_bar(buf, area, colors)
828        } else {
829            self.draw_list(buf, area, colors);
830            None
831        }
832    }
833}

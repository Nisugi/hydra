1//! The terminal's side of `hydra-tui`: the terminal made raw and given to
2//! Hydra, which draws on it.
3//!
4//! Each key, paste, mouse event and resize goes to Hydra as it happens;
5//! each frame Hydra draws is written as it comes. Text Hydra asks copied is
6//! put on this computer's clipboard -- by the terminal itself (OSC 52) when
7//! it is reached over SSH, where this computer's clipboard is not the
8//! player's -- and a link Hydra asks opened is opened here. The terminal is
9//! given back as it was however this ends.
10

26/// How often the terminal is looked at for the end while nothing happens.
27const POLL: Duration = Duration::from_millis(100);
28
29/// The terminal made raw, alternate and listening to the mouse, until this
30/// is dropped.
31struct Raw {
32    kitty: bool,
33}
34
35impl Raw {
36    /// The terminal made ready; with the kitty keyboard protocol when
37    /// `kitty`.
38    fn new(kitty: bool) -> io::Result<Self> {
39        terminal::enable_raw_mode()?;
40        // Made at once, so a step below that fails gives the terminal back.
41        let mut raw = Self { kitty: false };
42        let mut out = io::stdout();
43        execute!(
44            out,
45            terminal::EnterAlternateScreen,
46            EnableMouseCapture,
47            EnableBracketedPaste,
48            EnableFocusChange,
49            cursor::Hide,
50        )?;
51        if kitty {
52            raw.kitty = true;
53            // Pushed on the alternate screen, so they go with it even on a
54            // crash: the numpad's keys and every shifted key told exactly.
55            let _ = execute!(
56                out,
57                PushKeyboardEnhancementFlags(
58                    KeyboardEnhancementFlags::DISAMBIGUATE_ESCAPE_CODES
59                        | KeyboardEnhancementFlags::REPORT_ALL_KEYS_AS_ESCAPE_CODES
60                        | KeyboardEnhancementFlags::REPORT_ALTERNATE_KEYS
61                )
62            );
63        }
64        Ok(raw)
65    }
66}
67
68impl Drop for Raw {
69    fn drop(&mut self) {
70        let mut out = io::stdout();
71        if self.kitty {
72            let _ = execute!(out, PopKeyboardEnhancementFlags);
73        }
74        let _ = execute!(
75            out,
76            crossterm::style::ResetColor,
77            cursor::Show,
78            DisableFocusChange,
79            DisableBracketedPaste,
80            DisableMouseCapture,
81            terminal::LeaveAlternateScreen,
82        );
83        let _ = terminal::disable_raw_mode();
84    }
85}

87/// Whether this terminal is reached from another computer, over SSH.
88fn remote() -> bool {
89    std::env::var_os("SSH_CONNECTION").is_some() || std::env::var_os("SSH_TTY").is_some()
90}

92/// Give this terminal to Hydra over `stream`, its keys exact by the kitty
93/// keyboard protocol when the terminal speaks it or `kitty` says to, until
94/// Hydra closes the screen or the connection ends: what Hydra said as it
95/// closed it.
96///
97/// # Errors
98///
99/// The terminal could not be made ready, or the connection failed at once.
100pub fn attach(stream: TcpStream, kitty: bool) -> Result<String, String> {
101    let kitty = kitty || terminal::supports_keyboard_enhancement().unwrap_or(false);
102    let (cols, rows) = terminal::size().map_err(|why| format!("The terminal's size: {why}"))?;
103    let mut sending = stream
104        .try_clone()
105        .map_err(|why| format!("The connection: {why}"))?;
106    let raw = Raw::new(kitty).map_err(|why| format!("The terminal cannot be made ready: {why}"))?;
107    let remote = remote();
108    wire::send_up(
109        &mut sending,
110        &Up::Hello {
111            cols,
112            rows,
113            remote,
114            kitty,
115        },
116    )
117    .map_err(|why| format!("Hydra: {why}"))?;
118    let over = Arc::new(AtomicBool::new(false));
119    let said = Arc::new(Mutex::new(None::<String>));
120    let drawing = {
121        let (over, said) = (Arc::clone(&over), Arc::clone(&said));
122        std::thread::spawn(move || draw(stream, remote, &over, &said))
123    };
124    while !over.load(Ordering::Relaxed) {
125        let up = match event::poll(POLL) {
126            Ok(true) => match event::read() {
127                Ok(event) => up(event),
128                Err(_) => break,
129            },
130            Ok(false) => None,
131            Err(_) => break,
132        };
133        if let Some(up) = up
134            && wire::send_up(&mut sending, &up).is_err()
135        {
136            break;
137        }
138    }
139    let _ = wire::send_up(&mut sending, &Up::Bye);
140    let _ = sending.shutdown(std::net::Shutdown::Both);
141    let _ = drawing.join();
142    drop(raw);
143    let said = said.lock().unwrap_or_else(PoisonError::into_inner).take();
144    Ok(said.unwrap_or_else(|| "Hydra closed the connection.".to_owned()))
145}

147/// Write what Hydra draws until it closes the screen or the connection
148/// ends; then say it is over, and why.
149fn draw(mut stream: TcpStream, remote: bool, over: &AtomicBool, said: &Mutex<Option<String>>) {
150    let mut out = io::stdout();
151    // Kept for the whole attachment: on Linux the text copied lives only as
152    // long as the clipboard that set it.
153    let mut clipboard = None;
154    loop {
155        match wire::read_down(&mut stream) {
156            Ok(Down::Draw(bytes)) => {
157                let mut locked = out.lock();
158                if locked
159                    .write_all(&bytes)
160                    .and_then(|()| locked.flush())
161                    .is_err()
162                {
163                    break;
164                }
165            }
166            Ok(Down::Copy(text)) => copy(&mut out, &mut clipboard, &text, remote),
167            Ok(Down::OpenUrl(url)) => open(&url),
168            Ok(Down::Bye(why)) => {
169                *said.lock().unwrap_or_else(PoisonError::into_inner) = Some(why);
170                break;
171            }
172            Err(_) => break,
173        }
174    }
175    over.store(true, Ordering::Relaxed);
176}

178/// `text` on the clipboard: this computer's, or, over SSH, the terminal's
179/// own through OSC 52, which also serves when this computer has none.
180/// `clipboard` is this computer's, opened the first time.
181fn copy(
182    out: &mut io::Stdout,
183    clipboard: &mut Option<arboard::Clipboard>,
184    text: &str,
185    remote: bool,
186) {
187    if !remote && clipboard.is_none() {
188        *clipboard = arboard::Clipboard::new().ok();
189    }
190    let here = !remote
191        && clipboard
192            .as_mut()
193            .is_some_and(|clipboard| clipboard.set_text(text.to_owned()).is_ok());
194    if !here {
195        let _ = execute!(
196            out,
197            crossterm::clipboard::CopyToClipboard::to_clipboard_from(text)
198        );
199    }
200}

202/// `url` opened in this computer's browser: a web address alone.
203fn open(url: &str) {
204    if !(url.starts_with("https://") || url.starts_with("http://")) {
205        return;
206    }
207    #[cfg(windows)]
208    let opened = std::process::Command::new("rundll32")
209        .args(["url.dll,FileProtocolHandler", url])
210        .spawn();
211    #[cfg(target_os = "macos")]
212    let opened = std::process::Command::new("open").arg(url).spawn();
213    #[cfg(all(unix, not(target_os = "macos")))]
214    let opened = std::process::Command::new("xdg-open").arg(url).spawn();
215    drop(opened);
216}

218/// A terminal event as Hydra hears it; `None` for one it does not.
219fn up(event: Event) -> Option<Up> {
220    match event {
221        Event::Key(key) => crate::client_keys::key(&key).map(Up::Key),
222        Event::Paste(text) => Some(Up::Paste(text)),
223        Event::Resize(cols, rows) => Some(Up::Resize { cols, rows }),
224        Event::FocusGained => Some(Up::Focus(true)),
225        Event::FocusLost => Some(Up::Focus(false)),
226        Event::Mouse(mouse) => {
227            let button = |button: MouseButton| match button {
228                MouseButton::Left => wire::Button::Left,
229                MouseButton::Right => wire::Button::Right,
230                MouseButton::Middle => wire::Button::Middle,
231            };
232            let kind = match mouse.kind {
233                MouseEventKind::Down(b) => MouseKind::Down(button(b)),
234                MouseEventKind::Up(b) => MouseKind::Up(button(b)),
235                MouseEventKind::Drag(b) => MouseKind::Drag(button(b)),
236                MouseEventKind::Moved => MouseKind::Moved,
237                MouseEventKind::ScrollUp => MouseKind::ScrollUp,
238                MouseEventKind::ScrollDown => MouseKind::ScrollDown,
239                _ => return None,
240            };
241            let mut held = 0;
242            if mouse.modifiers.contains(event::KeyModifiers::CONTROL) {
243                held |= Mods::CTRL;
244            }
245            if mouse.modifiers.contains(event::KeyModifiers::SHIFT) {
246                held |= Mods::SHIFT;
247            }
248            if mouse.modifiers.contains(event::KeyModifiers::ALT) {
249                held |= Mods::ALT;
250            }
251            Some(Up::Mouse(Mouse {
252                kind,
253                col: mouse.column,
254                row: mouse.row,
255                mods: Mods(held),
256            }))
257        }
258    }

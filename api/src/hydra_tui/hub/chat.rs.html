1//! The hub's chat, under the Live tab's cards: what every character hears
2//! on the shared streams, merged so a line two characters both heard shows
3//! once, tagged with who heard it -- by the nickname the roster gives it,
4//! else its name. The newest at the foot; a line's links are done by the
5//! first character seated that heard it.

24/// The streams the chat shows.
25pub(super) const STREAMS: [&str; 5] = ["thoughts", "speech", "logons", "death", "announcements"];

27/// Draw the chat in `area`: a rule naming it across the top, the lines
28/// under it.
29pub(super) fn draw(
30    buf: &mut Buffer,
31    area: Rect,
32    glance: &Glance,
33    merged: &Merged,
34    (colors, hits): (&Colors, &mut Hits),
35) {
36    if area.height < 2 || area.width == 0 {
37        return;
38    }
39    let title = format!("── Chat · {} ", STREAMS.join(", "));
40    let rule = "─".repeat(usize::from(area.width));
41    buf.set_stringn(
42        area.x,
43        area.y,
44        &rule,
45        usize::from(area.width),
46        colors.border(false),
47    );
48    buf.set_stringn(
49        area.x,
50        area.y,
51        &title,
52        usize::from(area.width),
53        colors.muted(),
54    );
55    let lines = Rect::new(area.x, area.y + 1, area.width, area.height - 1);
56    let rows = rows(glance, merged, (lines.width, lines.height), colors);
57    if rows.is_empty() {
58        say(buf, lines, 0, "Nothing yet.", colors, Token::Muted);
59        return;
60    }
61    let top = lines.y + lines.height - u16::try_from(rows.len()).unwrap_or(lines.height);
62    for (at, (seat, row)) in rows.iter().rev().enumerate() {
63        let y = top + u16::try_from(at).unwrap_or(0);
64        draw_row(buf, (lines.x, y, lines.width), row, hits, *seat);
65    }
66}

68/// The rows the chat shows, `width` wide and at most `height` of them,
69/// newest first: each with the seat whose links its are.
70fn rows(
71    glance: &Glance,
72    merged: &Merged,
73    (width, height): (u16, u16),
74    colors: &Colors,
75) -> Vec<(Option<u32>, Line)> {
76    let merged = merged.lock().unwrap_or_else(PoisonError::into_inner);
77    let shown: Vec<&MergedLine> = merged
78        .lines()
79        .filter(|line| STREAMS.contains(&line.stream.as_str()))
80        .collect();
81    let mut rows = Vec::new();
82    for line in shown.into_iter().rev() {
83        let seat = line.from.iter().find_map(|key| {
84            glance
85                .seats
86                .iter()
87                .find(|seat| roster_name(&seat.game, &seat.name).eq_ignore_ascii_case(key))
88                .map(|seat| seat.id)
89        });
90        for row in wrap(&tagged(line, &glance.roster, colors), width)
91            .into_iter()
92            .rev()
93        {
94            rows.push((seat, row));
95            if rows.len() >= usize::from(height) {
96                return rows;
97            }
98        }
99    }
100    rows
101}

103/// `line` as the chat shows it: who heard it, then what was heard.
104fn tagged(line: &MergedLine, roster: &[RosterCard], colors: &Colors) -> Line {
105    let who: Vec<String> = line.from.iter().map(|key| tag(roster, key)).collect();
106    let mut pieces = vec![Piece::new(
107        format!("[{}] ", who.join(", ")),
108        colors.text(Token::Accent).add_modifier(Modifier::BOLD),
109    )];
110    pieces.extend(runs_line(&line.runs, colors));
111    pieces
112}

114/// What `key`'s tag says: its roster nickname, else its name.
115fn tag(roster: &[RosterCard], key: &str) -> String {
116    let name = key.split_once(':').map_or(key, |(_, name)| name);
117    roster
118        .iter()
119        .find(|card| roster_name(&card.game, &card.character).eq_ignore_ascii_case(key))
120        .and_then(|card| card.nickname.clone())
121        .filter(|nickname| !nickname.is_empty())
122        .unwrap_or_else(|| name.to_owned())
123}

125#[cfg(test)]
126mod tests {
127    use std::sync::{Arc, Mutex};
128    use std::time::Instant;
129
130    use hydra_ui::{MergedHistory, StyledRun};
131
132    use super::super::tests::{card, glance};
133    use super::*;
134
135    /// A tag is the roster's nickname when there is one, else the name.
136    #[test]
137    fn a_tag_is_the_nickname_or_the_name() {
138        let mut nisugi = card("Nisugi", "one", true, false);
139        nisugi.nickname = Some("Nisu".to_owned());
140        let roster = [nisugi, card("Ashryn", "two", true, false)];
141        assert_eq!(tag(&roster, "gs3:nisugi"), "Nisu");
142        assert_eq!(tag(&roster, "GS3:Ashryn"), "Ashryn");
143        assert_eq!(tag(&roster, "GS3:Dicate"), "Dicate", "not on the roster");
144    }
145
146    /// A line heard by two shows once, tagged with both, the newest at the
147    /// foot; a stream the chat does not show is left out.
148    #[test]
149    fn a_line_heard_by_two_shows_once() {
150        let mut history = MergedHistory::default();
151        let at = Instant::now();
152        let said = |text: &str| vec![StyledRun::plain(text)];
153        let heard = |stream: &str, text: &str| vec![(stream.to_owned(), said(text))];
154        history.offer(
155            at,
156            (1, "GS3:Nisugi", "Nisugi"),
157            &heard("thoughts", "[General] Someone: hello"),
158        );
159        history.offer(
160            at,
161            (2, "GS3:Ashryn", "Ashryn"),
162            &heard("thoughts", "[General] Someone: hello"),
163        );
164        history.offer(at, (1, "GS3:Nisugi", "Nisugi"), &heard("room", "Not chat"));
165        history.offer(
166            at,
167            (1, "GS3:Nisugi", "Nisugi"),
168            &heard("speech", "Dicate says, \"hi\""),
169        );
170        let merged: Merged = Arc::new(Mutex::new(history));
171        let glance = glance(Vec::new(), Vec::new(), &[]);
172        let area = Rect::new(0, 0, 50, 4);
173        let mut buf = Buffer::empty(area);
174        draw(
175            &mut buf,
176            area,
177            &glance,
178            &merged,
179            (&Colors::default(), &mut Hits::default()),
180        );
181        let row = |y: u16| -> String {
182            (0..50)
183                .map(|x| buf[(x, y)].symbol().to_owned())
184                .collect::<String>()
185                .trim_end()
186                .to_owned()
187        };
188        assert!(row(0).starts_with("── Chat"), "{}", row(0));
189        assert_eq!(row(2), "[Nisugi, Ashryn] [General] Someone: hello");
190        assert_eq!(row(3), "[Nisugi] Dicate says, \"hi\"");
191    }

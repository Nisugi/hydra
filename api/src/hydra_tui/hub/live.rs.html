23/// The tab's keys, as the line under the tabs says them.
24pub(super) const HINT: &str = "arrows or h j k l: choose · Enter: its screen · o: its own terminal · q: quit it · c: chat · Tab or 1-4: tabs";
25
26/// The fewest rows the hub needs for the chat to show under the cards.
27const CHAT_NEEDS: u16 = 12;
28
29/// The characters the tab shows: those not closed, in the order seated.
30pub(super) fn seats(glance: &Glance) -> Vec<&Arc<Seat>> {
31    glance
32        .seats
33        .iter()
34        .filter(|seat| seat.lifecycle() != State::Closed)
35        .collect()
36}
37
38/// `key` pressed on the tab, the card chosen `chosen` of a grid `per_row`
39/// to a row.
40pub(super) fn key(
41    key: &Key,
42    (chosen, per_row): (&mut usize, usize),
43    glance: &Glance,
44    chat_hidden: &mut bool,
45) -> Option<Asked> {
46    if letter(key) == Some('c') {
47        *chat_hidden = !*chat_hidden;
48        return None;
49    }
50    let seats = seats(glance);
51    if moved(key, chosen, (seats.len(), per_row)) {
52        return None;
53    }
54    let seat = seats.get(*chosen)?;
55    if enter(key) || letter(key) == Some('w') {
56        return Some(Asked::Show(seat.id));
57    }
58    match letter(key) {
59        Some('q') => Some(quit(&seat.name, seat.id)),
60        Some('o') => Some(Asked::Terminal(seat.id)),
61        _ => None,
62    }
63}

65/// Draw the tab in `area`, the chat under the cards when `chat` is given:
66/// how many cards went to a row.
67pub(super) fn draw(
68    buf: &mut Buffer,
69    area: Rect,
70    (glance, design, chat, data, perf): (&Glance, &Custom, Option<&Merged>, &Path, bool),
71    chosen: &mut usize,
72    (colors, hits, kept, now): (&Colors, &mut Hits, &mut Kept, Instant),
73) -> usize {
74    let mut cards = area;
75    if let Some(merged) = chat
76        && area.height >= CHAT_NEEDS
77    {
78        let tall = (area.height / 3).max(5);
79        cards.height = area.height - tall;
80        let below = Rect::new(area.x, area.y + cards.height, area.width, tall);
81        super::chat::draw(buf, below, glance, merged, (colors, hits));
82    }
83    let seats = seats(glance);
84    if seats.is_empty() {
85        say(
86            buf,
87            Rect::new(cards.x + 1, cards.y, cards.width.saturating_sub(1), 1),
88            0,
89            "No character is playing. 2 shows the roster's to start; 4 a new login.",
90            colors,
91            Token::Muted,
92        );
93        return 1;
94    }
95    *chosen = (*chosen).min(seats.len() - 1);
96    let (mut width, mut height) = card::size(design);
97    // With Performance on, each card's stages under it.
98    let card_height = height;
99    if perf {
100        width = width.max(card::STAGES_SIZE.0);
101        height += card::STAGES_SIZE.1;
102    }
103    let tiles = tiles(cards, (width + 2, height + 2), seats.len(), *chosen);
104    for (at, rect) in tiles.places {
105        let Some(seat) = seats.get(at) else {
106            continue;
107        };
108        hits.add(rect, Target::Button(format!("hub:card:{}", seat.id)));
109        let inside = frame(buf, rect, at == *chosen, colors);
110        card::draw(
111            buf,
112            inside,
113            (seat, design, nickname(glance, seat), data),
114            (colors, hits, kept, now),
115        );
116        if perf {
117            let under = Rect::new(
118                inside.x,
119                inside.y + card_height,
120                inside.width,
121                inside.height.saturating_sub(card_height),
122            );
123            card::stages(buf, under, seat, colors);
124        }
125    }
126    tiles.per_row
127}

129#[cfg(test)]
130mod tests {
131    use hydra_board::card::hydras;
132
133    use super::super::tests::glance;
134    use super::*;
135
136    /// Each character playing has a card; a click on one shows its screen,
137    /// q asks to quit it, and a closed one is not here.
138    #[test]
139    fn each_character_playing_has_a_card() {
140        let (ashryn, _a) = Seat::idle(1, "Ashryn");
141        let (dicate, _d) = Seat::idle(2, "Dicate");
142        let (gone, _g) = Seat::idle(3, "Nerten");
143        gone.end();
144        let glance = glance(vec![ashryn, dicate, gone], Vec::new(), &[]);
145        let area = Rect::new(0, 0, 100, 12);
146        let mut buf = Buffer::empty(area);
147        let mut hits = Hits::default();
148        let mut chosen = 5;
149        let per_row = draw(
150            &mut buf,
151            area,
152            (&glance, &hydras(), None, &std::env::temp_dir(), false),
153            &mut chosen,
154            (
155                &Colors::default(),
156                &mut hits,
157                &mut Kept::default(),
158                Instant::now(),
159            ),
160        );
161        assert_eq!(per_row, 2, "two cards of 44 across 100");
162        assert_eq!(chosen, 1, "kept among the cards");
163        let text: String = (0..12)
164            .flat_map(|y| (0..100).map(move |x| (x, y)))
165            .map(|at| buf[at].symbol().to_owned())
166            .collect();
167        assert!(text.contains("Ashryn") && text.contains("Dicate"));
168        assert!(!text.contains("Nerten"), "closed: on the Offline tab");
169        assert_eq!(
170            hits.at(46, 0),
171            Some(&Target::Button("hub:card:2".to_owned()))
172        );
173        let mut hidden = false;
174        let asked = key(
175            &Key::typing('q'),
176            (&mut chosen, per_row),
177            &glance,
178            &mut hidden,
179        );
180        assert!(matches!(asked, Some(Asked::Open(_))));
181        let asked = key(
182            &Key::named("Enter"),
183            (&mut chosen, per_row),
184            &glance,
185            &mut hidden,
186        );
187        assert!(matches!(asked, Some(Asked::Show(2))));
188        key(
189            &Key::typing('c'),
190            (&mut chosen, per_row),
191            &glance,
192            &mut hidden,
193        );
194        assert!(hidden, "c hides the chat");
195    }

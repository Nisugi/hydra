16/// One row of the list: a bar with words on it.
17#[derive(Clone)]
18pub struct ScrollableItem {
19    /// The words on the bar.
20    pub text: String,
21    /// Other words for it, shown instead when the list is switched to them:
22    /// a spell's number for its name.
23    pub alternate_text: Option<String>,
24    /// How full the bar is.
25    pub value: u32,
26    /// The bar's most.
27    pub max: u32,

29    pub suffix: Option<String>,
30    /// The bar's colour, `#rrggbb`.
31    pub color: Option<String>,
32    /// The words' colour, `#rrggbb`.
33    pub text_color: Option<String>,
34}

101    fn max_scroll_offset(&self) -> usize {
102        if self.item_order.is_empty() {
103            return 0;
104        }
105
106        let visible = self
107            .visible_count
108            .unwrap_or(self.last_available_height)
109            .max(1);
110        self.item_order.len().saturating_sub(visible)
111    }
112
113    /// It scrolled back by `amount` lines.
114    pub fn scroll_up(&mut self, amount: usize) {
115        self.scroll_offset = self.scroll_offset.saturating_sub(amount);
116    }
117
118    /// It scrolled forward by `amount` lines.
119    pub fn scroll_down(&mut self, amount: usize) {
120        let max_scroll = self.max_scroll_offset();
121        self.scroll_offset = self.scroll_offset.saturating_add(amount).min(max_scroll);
122    }

132        if self.item_order.is_empty() {
133            self.scroll_offset = 0;
134            return;
135        }

137        let max_scroll = self.max_scroll_offset();
138        self.scroll_offset = desired_offset.min(max_scroll);

219    /// It drawn into `area`.
220    pub fn render(&mut self, area: Rect, buf: &mut Buffer) {
221        if area.width < 3 || area.height < 1 {
222            return;
223        }
224
225        // Clear the area to prevent bleed-through from windows behind
226        Clear.render(area, buf);
227
228        if !self.transparent_background
229            && let Some(bg_color) = self.background_color
230        {
231            crate::vellum::support::paint(area, buf, bg_color);
232        }
233
234        // Determine which borders to show
235        let borders = if self.show_border {
236            support::to_ratatui_borders(&self.border_sides)
237        } else {
238            Borders::NONE
239        };
240
241        // The window's border colour, set from its settings; else the
242        // terminal's own.
243        let border_color = self.border_color.unwrap_or(Color::Reset);
244
245        let inner_area: Rect;
246
247        if self.show_border {
248            // Use Block widget for borders
249            let mut block = Block::default().borders(borders);
250
251            if let Some(ref style) = self.border_style {
252                block = block.border_type(crate::chrome::border_type(style));
253            }
254
255            block = block.border_style(ratatui::style::Style::default().fg(border_color));
256            // Only set title if label is non-empty (avoids empty title affecting layout)
257            if !self.label.is_empty() {
258                block = block.title(self.label.as_str());
259            }
260
261            inner_area = block.inner(area);
262            block.render(area, buf);
263        } else {
264            inner_area = area;
265        }
266
267        if inner_area.width == 0 || inner_area.height == 0 {
268            return;
269        }
270
271        // Calculate how many items we can display
272        let available_height = inner_area.height as usize;
273        self.last_available_height = available_height; // Store for scroll calculations
274        let display_count = self
275            .visible_count
276            .unwrap_or(available_height)
277            .min(available_height);
278
279        let max_scroll = self.max_scroll_offset();
280        if self.scroll_offset > max_scroll {
281            self.scroll_offset = max_scroll;
282        }
283
284        // Get the slice of items to display
285        let start_idx = self.scroll_offset;
286        let end_idx = (start_idx + display_count).min(self.item_order.len());
287
288        // Render each visible item as a progress bar
289        for (i, item_id) in self.item_order[start_idx..end_idx].iter().enumerate() {
290            if let Some(item) = self.items.get(item_id) {
291                // Choose which text to display (primary or alternate)
292                let source_text = if self.show_alternate_text {
293                    item.alternate_text.as_ref().unwrap_or(&item.text)
294                } else {
295                    &item.text
296                };
297
298                let display_text = with_suffix(
299                    source_text,
300                    item.suffix.as_deref(),
301                    inner_area.width as usize,
302                );
303
304                // Create a progress bar for this item
305                let mut pb = ProgressBar::new("");
306                pb.set_value_with_text(item.value, item.max, Some(display_text));
307                pb.set_text_align_left(true);
308
309                // Use item-specific color if provided, otherwise default bar color
310                let bar_fill = item.color.clone().or_else(|| Some(self.bar_color.clone()));
311                pb.set_colors(bar_fill, None);
312                if self.transparent_background {
313                    pb.set_background_color(None);
314                } else {
315                    pb.set_background_color(self.background_color_hex.clone());
316                }
317
318                // Check for highlight match on the item text
319                let highlight_color = self.highlight_engine.get_first_match_color(source_text);
320
321                let row_text_color = highlight_color
322                    .or_else(|| item.text_color.clone())
323                    .or_else(|| self.text_color.clone());
324
325                pb.set_text_color(row_text_color);
326
327                pb.set_transparent_background(self.transparent_background);
328                pb.set_border_config(
329                    false,
330                    None,
331                    None,
332                    crate::vellum::data::BorderSides {
333                        top: false,
334                        bottom: false,
335                        left: false,
336                        right: false,
337                    },
338                ); // No borders on individual items
339
340                // Calculate the area for this item (single row)
341                let item_area = Rect {
342                    x: inner_area.x,
343                    y: inner_area.y + u16::try_from(i).unwrap_or(u16::MAX),
344                    width: inner_area.width,
345                    height: 1,
346                };
347
348                pb.render(item_area, buf);
349            }
350        }
351    }

353    /// Convert mouse position to text coordinates
354    #[must_use]
355    pub fn mouse_to_text_coords(
356        &self,
357        mouse_col: u16,
358        mouse_row: u16,
359        window_rect: Rect,
360    ) -> Option<(usize, usize)> {
361        let border_offset = u16::from(self.show_border);
362
363        // Bounds check within content area
364        if mouse_col < window_rect.x + border_offset
365            || mouse_col >= window_rect.x + window_rect.width - border_offset
366            || mouse_row < window_rect.y + border_offset
367            || mouse_row >= window_rect.y + window_rect.height - border_offset
368        {
369            return None;
370        }
371
372        let line_idx = self.scroll_offset + (mouse_row - window_rect.y - border_offset) as usize;
373        let col_offset = (mouse_col - window_rect.x - border_offset) as usize;
374
375        Some((line_idx, col_offset))
376    }

378    /// Extract text from a selection range
379    #[must_use]
380    pub fn extract_selection_text(
381        &self,
382        start_line: usize,
383        start_col: usize,
384        end_line: usize,
385        end_col: usize,
386    ) -> String {
387        let mut result = String::new();
388
389        for line_idx in start_line..=end_line.min(self.item_order.len().saturating_sub(1)) {
390            if line_idx >= self.item_order.len() {
391                break;
392            }
393
394            let item_id = &self.item_order[line_idx];
395            let Some(item) = self.items.get(item_id) else {
396                continue;
397            };
398
399            let line_text = if self.show_alternate_text {
400                item.alternate_text.as_deref().unwrap_or(&item.text)
401            } else {
402                &item.text
403            };
404            let line_len = line_text.chars().count();
405
406            if start_line == end_line {
407                // Single line selection
408                let start = start_col.min(line_len);
409                let end = end_col.min(line_len);
410                if start < end {
411                    result.push_str(
412                        &line_text
413                            .chars()
414                            .skip(start)
415                            .take(end - start)
416                            .collect::<String>(),
417                    );
418                }
419            } else if line_idx == start_line {
420                // First line of multi-line selection
421                let start = start_col.min(line_len);
422                result.push_str(&line_text.chars().skip(start).collect::<String>());
423                result.push('\n');
424            } else if line_idx == end_line {
425                // Last line of multi-line selection
426                let end = end_col.min(line_len);
427                result.push_str(&line_text.chars().take(end).collect::<String>());
428            } else {
429                // Middle lines - take all
430                result.push_str(line_text);
431                result.push('\n');
432            }
433        }
434
435        result
436    }

439/// `text` with `suffix` held at the right edge of `width` columns: the text
440/// cut short to leave a space before the suffix, and the suffix itself cut
441/// when even it does not fit.
442fn with_suffix(text: &str, suffix: Option<&str>, width: usize) -> String {
443    let Some(suffix) = suffix else {
444        return text.to_owned();
445    };
446    let suffix_len = suffix.chars().count();
447    if width <= suffix_len + 1 {
448        return suffix.chars().take(width).collect();
449    }
450    let room = width - suffix_len - 1;
451    let shown: String = text.chars().take(room).collect();
452    let padding = width - shown.chars().count() - suffix_len;
453    format!("{shown}{}{suffix}", " ".repeat(padding))

18/// A run of text as it is added: its words, its colours, whether bold, what
19/// it is, and where it links.
20#[derive(Clone)]
21pub struct StyledText {
22    /// The words.
23    pub content: String,
24    /// Their colour.
25    pub fg: Option<Color>,
26    /// The colour behind them.
27    pub bg: Option<Color>,
28    /// Whether they are bold.
29    pub bold: bool,
30    /// What they are, which decides whose colour wins.

36/// A run of a line: its text, its look, what it is, and where it links.
37type Run = (String, Style, SpanType, Option<LinkData>);

39// One display line (post-wrapping) with multiple styled spans
40#[derive(Clone)]
41struct WrappedLine {
42    spans: Vec<Run>,
43}

45// One logical line (before wrapping) - stores original styled content
46#[derive(Clone)]
47struct LogicalLine {
48    spans: Vec<Run>,
49    /// Number of wrapped lines this logical line produces (for scroll position tracking)
50    wrapped_count: usize,
51}

282    /// Unfreeze the buffer and apply all pending lines.
283    /// Call this when selection is complete (copy finished or selection cancelled).
284    pub fn unfreeze_and_apply_pending(&mut self) {
285        if !self.frozen_for_selection {
286            return;
287        }
288
289        self.frozen_for_selection = false;
290
291        // Apply pending logical lines
292
293        for logical_line in self.pending_logical_lines.drain(..) {
294            self.logical_lines.push_back(logical_line);
295
296            // Handle buffer overflow for logical lines
297            if self.logical_lines.len() > self.max_lines
298                && let Some(old_line) = self.logical_lines.pop_front()
299            {
300                let removed_count = old_line.wrapped_count;
301
302                // Remove corresponding wrapped lines from front
303                for _ in 0..removed_count {
304                    self.wrapped_lines.pop_front();
305                }
306
307                // Adjust scroll_position to maintain view on same content
308                if let Some(pos) = self.scroll_position {
309                    if pos < removed_count {
310                        self.scroll_position = None;
311                        self.scroll_offset = 0;
312                    } else {
313                        self.scroll_position = Some(pos - removed_count);
314                    }
315                }
316            }
317        }
318
319        // Apply pending wrapped lines
320        for wrapped_line in self.pending_wrapped_lines.drain(..) {
321            self.wrapped_lines.push_back(wrapped_line);
322        }
323    }

405    /// A run of text added to the line being built.
406    pub fn add_text(&mut self, styled: StyledText) {
407        let mut style = Style::default();
408        if let Some(fg) = styled.fg {
409            style = style.fg(fg);
410        }
411        if let Some(bg) = styled.bg {
412            style = style.bg(bg);
413        }
414        if styled.bold {
415            style = style.add_modifier(Modifier::BOLD);
416        }
417
418        // Only process links if links_enabled is true
419        let link_data = if self.links_enabled {
420            // Cache link data if present, accumulating text for the same exist_id
421            if let Some(ref link_data) = styled.link_data {
422                // Check if we already have this exist_id in the most recent entry
423                let should_append = if let Some(last) = self.recent_links.back_mut() {
424                    if last.exist_id == link_data.exist_id {
425                        // Append to existing text (no debug log for appends - too spammy)
426                        last.text.push_str(&styled.content);
427                        true
428                    } else {
429                        false
430                    }
431                } else {
432                    false
433                };
434
435                if !should_append {
436                    // New link - create new entry with this content as the text
437                    let mut new_link = link_data.clone();
438                    new_link.text.clone_from(&styled.content);
439                    self.recent_links.push_back(new_link);
440                    if self.recent_links.len() > self.max_recent_links {
441                        self.recent_links.pop_front();
442                    }
443                }
444            }
445            styled.link_data.clone()
446        } else {
447            // Links disabled - don't cache or include link data
448            None
449        };
450
451        // Add this styled chunk to current line with semantic type and link metadata
452        self.current_line_spans
453            .push((styled.content, style, styled.span_type, link_data));
454    }

456    /// The line being built finished and kept.
457    pub fn finish_line(&mut self, _width: u16) {
458        // Every line given is kept, a blank one too, and drawn as it came:
459        // which lines show and their highlights are settled before here.
460
461        // Add timestamp if enabled (before storing/wrapping)
462        if self.show_timestamps {
463            let timestamp_style = Style::default().fg(Color::DarkGray);
464            match self.timestamp_position {
465                TimestampPosition::Start => {
466                    let timestamp = self.format_timestamp_start();
467                    self.current_line_spans
468                        .insert(0, (timestamp, timestamp_style, SpanType::Normal, None));
469                }
470                TimestampPosition::End => {
471                    let timestamp = self.format_timestamp_end();
472                    self.current_line_spans.push((
473                        timestamp,
474                        timestamp_style,
475                        SpanType::Normal,
476                        None,
477                    ));
478                }
479            }
480        }
481
482        // Wrap this logical line first to get the count
483        let actual_width = if self.last_width > 0 {
484            self.last_width
485        } else {
486            80 // Fallback
487        };
488
489        let wrap_start = std::time::Instant::now();
490        let wrapped = self.wrap_styled_spans(&self.current_line_spans, actual_width as usize);
491        let wrap_duration = wrap_start.elapsed();
492        self.wrap_samples.push(wrap_duration);
493
494        let wrapped_count = wrapped.len();
495
496        // Store the original logical line with wrapped count
497        let logical_line = LogicalLine {
498            spans: self.current_line_spans.clone(),
499            wrapped_count,
500        };
501
502        // If frozen for selection, queue lines in pending buffers instead of main buffers.
503        // This prevents selection indices from drifting when new content arrives.
504        if self.frozen_for_selection {
505            self.pending_logical_lines.push_back(logical_line);
506            for line in wrapped {
507                self.pending_wrapped_lines.push_back(line);
508            }
509            // Cap pending growth: anything beyond max_lines would be evicted
510            // immediately on unfreeze anyway, so drop the oldest pending line
511            // now instead of growing without bound during a long freeze.
512            while self.pending_logical_lines.len() > self.max_lines {
513                if let Some(old_line) = self.pending_logical_lines.pop_front() {
514                    for _ in 0..old_line.wrapped_count {
515                        self.pending_wrapped_lines.pop_front();
516                    }
517                }
518            }
519            self.current_line_spans.clear();
520            return;
521        }
522
523        // Normal path: add to main buffers
524        self.logical_lines.push_back(logical_line);
525
526        // Remove oldest logical line AND its wrapped lines if we exceed buffer
527        if self.logical_lines.len() > self.max_lines
528            && let Some(old_line) = self.logical_lines.pop_front()
529        {
530            let removed_count = old_line.wrapped_count;
531
532            // Remove corresponding wrapped lines from front
533            for _ in 0..removed_count {
534                self.wrapped_lines.pop_front();
535            }
536
537            // Adjust scroll_position to maintain view on same content
538            if let Some(pos) = self.scroll_position {
539                if pos < removed_count {
540                    // Viewed content was purged - reset to live view
541                    self.scroll_position = None;
542                    self.scroll_offset = 0;
543                } else {
544                    // Shift position to maintain same content
545                    self.scroll_position = Some(pos - removed_count);
546                }
547            }
548        }
549
550        // Add new wrapped lines to the END
551        for line in wrapped {
552            self.wrapped_lines.push_back(line);
553        }
554
555        self.current_line_spans.clear();
556    }

558    /// A word placed: on the line when it fits there, else on a new line,
559    /// else, longer than a line, broken where the lines end.
560    fn place_word(
561        word: Vec<Run>,
562        (word_len, width): (usize, usize),
563        (line, line_len): (&mut Vec<Run>, &mut usize),
564        result: &mut Vec<WrappedLine>,
565    ) {
566        if *line_len + word_len <= width {
567            for (text, style, kind, link) in word {
568                Self::append_to_line(line, text, style, kind, link);
569            }
570            *line_len += word_len;
571        } else if word_len <= width {
572            if !line.is_empty() {
573                result.push(WrappedLine {
574                    spans: std::mem::take(line),
575                });
576                *line_len = 0;
577            }
578            for (text, style, kind, link) in word {
579                Self::append_to_line(line, text, style, kind, link);
580            }
581            *line_len += word_len;
582        } else {
583            for (text, style, kind, link) in word {
584                for ch in text.chars() {
585                    if *line_len >= width {
586                        result.push(WrappedLine {
587                            spans: std::mem::take(line),
588                        });
589                        *line_len = 0;
590                    }
591                    Self::append_char_to_line(line, ch, style, kind, link.as_ref());
592                    *line_len += 1;
593                }
594            }
595        }
596    }

598    /// Styled spans wrapped into the lines they take at `width`.
599    fn wrap_styled_spans(
600        &self,
601        spans: &[(String, Style, SpanType, Option<LinkData>)],
602        width: usize,
603    ) -> Vec<WrappedLine> {
604        if width == 0 {
605            return vec![];
606        }
607        if !self.wordwrap {
608            let mut line_spans = Vec::new();
609            for (text, style, span_type, link) in spans {
610                Self::append_to_line(
611                    &mut line_spans,
612                    text.clone(),
613                    *style,
614                    *span_type,
615                    link.clone(),
616                );
617            }
618            return vec![WrappedLine { spans: line_spans }];
619        }
620
621        let mut result = Vec::new();
622        let mut current_line_spans: Vec<Run> = Vec::new();
623        let mut current_line_len = 0;
624
625        // Track word buffer for smart wrapping
626        let mut word_buffer: Vec<Run> = Vec::new();
627        let mut word_buffer_len = 0;
628        let mut in_word = false;
629
630        for (text, style, span_type, link) in spans {
631            for ch in text.chars() {
632                let is_whitespace = ch.is_whitespace();
633
634                if is_whitespace {
635                    // Flush word buffer if we have one
636                    if in_word && !word_buffer.is_empty() {
637                        Self::place_word(
638                            std::mem::take(&mut word_buffer),
639                            (word_buffer_len, width),
640                            (&mut current_line_spans, &mut current_line_len),
641                            &mut result,
642                        );
643                        word_buffer_len = 0;
644                        in_word = false;
645                    }
646
647                    // Add whitespace immediately (don't buffer it)
648                    if current_line_len >= width {
649                        // Wrap before whitespace
650                        result.push(WrappedLine {
651                            spans: current_line_spans.clone(),
652                        });
653                        current_line_spans.clear();
654                        current_line_len = 0;
655                        // Don't add whitespace at start of new line
656                        continue;
657                    }
658                    Self::append_char_to_line(
659                        &mut current_line_spans,
660                        ch,
661                        *style,
662                        *span_type,
663                        link.as_ref(),
664                    );
665                    current_line_len += 1;
666                } else {
667                    // Non-whitespace character - add to word buffer
668                    in_word = true;
669                    Self::append_char_to_line(
670                        &mut word_buffer,
671                        ch,
672                        *style,
673                        *span_type,
674                        link.as_ref(),
675                    );
676                    word_buffer_len += 1;
677                }
678            }
679        }
680
681        if !word_buffer.is_empty() {
682            Self::place_word(
683                word_buffer,
684                (word_buffer_len, width),
685                (&mut current_line_spans, &mut current_line_len),
686                &mut result,
687            );
688        }
689
690        // Push any remaining content
691        if !current_line_spans.is_empty() {
692            result.push(WrappedLine {
693                spans: current_line_spans,
694            });
695        }
696
697        if result.is_empty() {
698            // Return at least one empty line
699            result.push(WrappedLine { spans: vec![] });
700        }
701
702        result
703    }

706    fn append_to_line(
707        spans: &mut Vec<Run>,
708        text: String,
709        style: Style,
710        span_type: SpanType,
711        link: Option<LinkData>,
712    ) {
713        if let Some((last_text, last_style, last_type, last_link)) = spans.last_mut() {
714            if *last_style == style && *last_type == span_type && *last_link == link {
715                last_text.push_str(&text);
716            } else {
717                spans.push((text, style, span_type, link));
718            }
719        } else {
720            spans.push((text, style, span_type, link));
721        }
722    }

751    /// It scrolled back by `amount` lines.
752    pub fn scroll_up(&mut self, amount: usize) {
753        // Scrolling up = viewing older lines
754        let total_lines = self.wrapped_lines.len();
755
756        if let Some(pos) = self.scroll_position {
757            // Already scrolled - move the absolute position up (to older lines)
758            self.scroll_position = Some(pos.saturating_sub(amount));
759        } else {
760            // First scroll up from live view - convert to absolute position
761            // We're currently viewing the last last_visible_height lines
762            // The view starts at (total_lines - visible_height)
763            let current_start = total_lines.saturating_sub(self.last_visible_height);
764            // Scroll up means move the start position back
765            self.scroll_position = Some(current_start.saturating_sub(amount));
766        }
767    }

769    /// It scrolled forward by `amount` lines.
770    pub fn scroll_down(&mut self, amount: usize) {
771        // Scrolling down = viewing newer lines
772        let total_lines = self.wrapped_lines.len();
773
774        if let Some(pos) = self.scroll_position {
775            let new_pos = pos.saturating_add(amount);
776
777            // Check if we've scrolled back to the bottom (within visible_height of end)
778            let bottom_threshold = total_lines.saturating_sub(self.last_visible_height);
779            if new_pos >= bottom_threshold {
780                // Return to live view mode
781                self.scroll_position = None;
782                self.scroll_offset = 0;
783            } else {
784                self.scroll_position = Some(new_pos);
785            }
786        } else {
787            // Already in live view, just decrease offset (shouldn't normally happen)
788            self.scroll_offset = self.scroll_offset.saturating_sub(amount);
789        }
790    }

816    /// A search begun for `pattern`, a regular expression: how many lines
817    /// match.
818    ///
819    /// # Errors
820    ///
821    /// `pattern` is not a regular expression.
822    pub fn start_search(&mut self, pattern: &str) -> Result<usize, regex::Error> {
823        let regex = Regex::new(pattern)?;
824
825        // Search through all wrapped lines
826        let mut matches = Vec::new();
827
828        for (line_idx, wrapped_line) in self.wrapped_lines.iter().enumerate() {
829            // Combine all spans into a single text string for searching
830            let line_text: String = wrapped_line
831                .spans
832                .iter()
833                .map(|(text, _, _, _)| text.as_str())
834                .collect();
835
836            // Find all matches in this line
837            for mat in regex.find_iter(&line_text) {
838                matches.push(SearchMatch {
839                    line_idx,
840                    start: mat.start(),
841                    end: mat.end(),
842                });
843            }
844        }
845
846        let match_count = matches.len();
847
848        if matches.is_empty() {
849            self.search_state = None;
850        } else {
851            self.search_state = Some(SearchState {
852                matches,
853                current_match_idx: 0,
854            });
855
856            // Scroll to first match
857            self.scroll_to_match(0);
858        }
859
860        Ok(match_count)
861    }

881        let new_idx = if let Some(state) = &mut self.search_state {
882            if state.matches.is_empty() {
883                None
884            } else {
885                state.current_match_idx = (state.current_match_idx + 1) % state.matches.len();
886                Some(state.current_match_idx)
887            }
888        } else {
889            None
890        };
891
892        if let Some(idx) = new_idx {
893            self.scroll_to_match(idx);
894            true
895        } else {
896            false
897        }
898    }

905            } else {

913        } else {
914            None
915        };
916
917        if let Some(idx) = new_idx {
918            self.scroll_to_match(idx);
919            true
920        } else {
921            false
922        }
923    }

948    /// Create spans for a line with highlighted search matches
949    fn create_highlighted_spans(
950        &self,
951        wrapped: &WrappedLine,
952        line_matches: &[&SearchMatch],
953        current_match: Option<&SearchMatch>,
954    ) -> Vec<Span<'_>> {
955        // Build the full line text to know character positions
956        let _full_text: String = wrapped
957            .spans
958            .iter()
959            .map(|(text, _, _, _)| text.as_str())
960            .collect();
961
962        // Collect all character positions that should be highlighted
963        let mut highlight_ranges: Vec<(usize, usize, bool)> = Vec::new(); // (start, end, is_current)
964
965        for m in line_matches {
966            let is_current = current_match.is_some_and(|cm| {
967                cm.line_idx == m.line_idx && cm.start == m.start && cm.end == m.end
968            });
969            highlight_ranges.push((m.start, m.end, is_current));
970        }
971
972        // Sort ranges by start position
973        highlight_ranges.sort_by_key(|(start, _, _)| *start);
974
975        // Reconstruct spans, splitting where highlights occur
976        let mut result_spans = Vec::new();
977        let mut char_pos = 0;
978        let mut highlight_idx = 0;
979
980        for (text, style, _span_type, _link) in &wrapped.spans {
981            let base_style = self.style_with_defaults(*style);
982            let text_len = text.len();
983            let span_start = char_pos;
984            let span_end = char_pos + text_len;
985
986            let mut current_pos = span_start;
987
988            // Check for highlights that overlap this span
989            while highlight_idx < highlight_ranges.len()
990                && highlight_ranges[highlight_idx].0 < span_end
991            {
992                let (hl_start, hl_end, is_current) = highlight_ranges[highlight_idx];
993
994                if hl_end <= span_start {
995                    // Highlight is before this span
996                    highlight_idx += 1;
997                    continue;
998                }
999
1000                // Add non-highlighted part before the match
1001                if current_pos < hl_start && hl_start >= span_start {
1002                    let offset = current_pos - span_start;
1003                    let length = hl_start - current_pos;
1004                    let substr = &text[offset..offset + length];
1005                    result_spans.push(Span::styled(substr.to_string(), base_style));
1006                    current_pos = hl_start;
1007                }
1008
1009                // Add highlighted part
1010                if current_pos < hl_end && current_pos >= span_start {
1011                    let offset = current_pos - span_start;
1012                    let end_pos = hl_end.min(span_end);
1013                    let length = end_pos - current_pos;
1014                    let substr = &text[offset..offset + length];
1015
1016                    // Use different colors for current match vs other matches
1017                    let highlight_style = if is_current {
1018                        Style::default()
1019                            .bg(Color::Yellow)
1020                            .fg(Color::Black)
1021                            .add_modifier(Modifier::BOLD)
1022                    } else {
1023                        Style::default().bg(Color::DarkGray).fg(Color::White)
1024                    };
1025
1026                    let highlight_style = self.style_with_defaults(highlight_style);
1027                    result_spans.push(Span::styled(substr.to_string(), highlight_style));
1028                    current_pos = end_pos;
1029                }
1030
1031                if hl_end <= span_end {
1032                    highlight_idx += 1;
1033                }
1034
1035                if current_pos >= span_end {
1036                    break;
1037                }
1038            }
1039
1040            // Add remaining non-highlighted part
1041            if current_pos < span_end {
1042                let offset = current_pos - span_start;
1043                let substr = &text[offset..];
1044                result_spans.push(Span::styled(substr.to_string(), base_style));
1045            }
1046
1047            char_pos = span_end;
1048        }
1049
1050        result_spans
1051    }

1053    /// Create spans for a line with text selection highlighting
1054    fn create_spans_with_selection(
1055        &self,
1056        wrapped: &WrappedLine,
1057        line_idx: usize,
1058        selection_state: Option<&super::selection::SelectionState>,
1059        selection_bg: Option<Color>,
1060        window_index: usize,
1061    ) -> Vec<Span<'_>> {
1062        // If no selection or no background color, render normally
1063        let selection = match (selection_state, selection_bg) {
1064            (Some(sel), Some(bg)) if sel.active => (sel, bg),
1065            _ => {
1066                return wrapped
1067                    .spans
1068                    .iter()
1069                    .map(|(text, style, _span_type, _link)| {
1070                        let resolved = self.style_with_defaults(*style);
1071                        Span::styled(text.clone(), resolved)
1072                    })
1073                    .collect();
1074            }
1075        };
1076
1077        let (sel, bg_color) = selection;
1078
1079        // Reconstruct spans, applying selection background where needed
1080        let mut result_spans = Vec::new();
1081        let mut char_pos = 0;
1082
1083        for (text, style, _span_type, _link) in &wrapped.spans {
1084            let text_chars: Vec<char> = text.chars().collect();
1085            let text_len = text_chars.len(); // Character count, not byte count
1086            let span_start = char_pos;
1087            let span_end = char_pos + text_len;
1088
1089            let mut current_pos = span_start;
1090            let mut char_idx = 0;
1091
1092            // Process each character to check if it's selected
1093            while char_idx < text_chars.len() {
1094                let char_col = current_pos;
1095                let is_selected = sel.contains(window_index, line_idx, char_col);
1096
1097                // Collect consecutive characters with same selection state
1098                let mut chunk = String::new();
1099                chunk.push(text_chars[char_idx]);
1100                char_idx += 1;
1101                current_pos += 1;
1102
1103                while char_idx < text_chars.len() {
1104                    let next_col = current_pos;
1105                    let next_selected = sel.contains(window_index, line_idx, next_col);
1106
1107                    if next_selected == is_selected {
1108                        chunk.push(text_chars[char_idx]);
1109                        char_idx += 1;
1110                        current_pos += 1;
1111                    } else {
1112                        break;
1113                    }
1114                }
1115
1116                // Create span with appropriate style
1117                let span_style = if is_selected {
1118                    style.bg(bg_color)
1119                } else {
1120                    *style
1121                };
1122
1123                let span_style = self.style_with_defaults(span_style);
1124                result_spans.push(Span::styled(chunk, span_style));
1125            }
1126
1127            char_pos = span_end;
1128        }
1129
1130        result_spans
1131    }

1133    /// Convert a relative row position (visible row in window) to absolute line index in `wrapped_lines` buffer
1134    /// This accounts for scroll position to map visible coordinates to buffer coordinates
1135    #[must_use]
1136    pub fn relative_row_to_absolute_line(&self, rel_row: usize, visible_height: usize) -> usize {
1137        let total_lines = self.wrapped_lines.len();
1138
1139        // Calculate start_line using same logic as render_with_focus
1140        let start_line = if let Some(pos) = self.scroll_position {
1141            // Scrolled back - use absolute position (frozen view)
1142            pos
1143        } else {
1144            // Live view mode - show the last visible_height lines
1145            let end = total_lines.saturating_sub(self.scroll_offset);

1147        };
1148

1151    }

1153    /// Re-wrap all logical lines with the current width
1154    fn rewrap_all(&mut self) {
1155        self.wrapped_lines.clear();
1156
1157        let width = if self.last_width > 0 {
1158            self.last_width as usize
1159        } else {
1160            80
1161        };
1162
1163        // First pass: collect spans for wrapping (to avoid borrow conflict)
1164        let spans_to_wrap: Vec<_> = self
1165            .logical_lines
1166            .iter()
1167            .map(|ll| ll.spans.clone())
1168            .collect();
1169
1170        // Second pass: wrap and update counts
1171        for (i, spans) in spans_to_wrap.iter().enumerate() {
1172            let wrapped = self.wrap_styled_spans(spans, width);
1173            let wrapped_count = wrapped.len();
1174
1175            // Update the wrapped_count in the logical line
1176            if let Some(ll) = self.logical_lines.get_mut(i) {
1177                ll.wrapped_count = wrapped_count;
1178            }
1179
1180            for line in wrapped {
1181                self.wrapped_lines.push_back(line);
1182            }
1183        }
1184
1185        // Validate scroll_position after rebuild
1186        let total = self.wrapped_lines.len();
1187        if let Some(pos) = self.scroll_position
1188            && pos >= total
1189        {
1190            // Position is now invalid - reset to live view
1191            self.scroll_position = None;
1192            self.scroll_offset = 0;
1193        }
1194
1195        self.needs_rewrap = false;
1196    }

1251    /// Convert mouse position to text coordinates (line index, column index)
1252    /// Returns None if position is outside content area
1253    #[must_use]
1254    pub fn mouse_to_text_coords(
1255        &self,
1256        mouse_col: u16,
1257        mouse_row: u16,
1258        window_rect: ratatui::layout::Rect,
1259    ) -> Option<(usize, usize)> {
1260        let border_offset = u16::from(self.has_border());
1261
1262        // Bounds check within content area
1263        if mouse_col < window_rect.x + border_offset
1264            || mouse_col >= window_rect.x + window_rect.width - border_offset
1265            || mouse_row < window_rect.y + border_offset
1266            || mouse_row >= window_rect.y + window_rect.height - border_offset
1267        {
1268            return None;
1269        }
1270
1271        let visible_height = (window_rect.height.saturating_sub(2 * border_offset)) as usize;
1272        let total_lines = self.wrapped_lines.len();
1273
1274        // Calculate start_line (same logic as render)
1275        let start_line = if let Some(pos) = self.scroll_position {
1276            pos
1277        } else {
1278            let end = total_lines.saturating_sub(self.scroll_offset);
1279            end.saturating_sub(visible_height)
1280        };
1281
1282        let line_idx = start_line + (mouse_row - window_rect.y - border_offset) as usize;
1283        let col_offset = (mouse_col - window_rect.x - border_offset) as usize;
1284
1285        // Line index might be beyond visible lines
1286        if line_idx >= total_lines {
1287            return None;
1288        }
1289
1290        Some((line_idx, col_offset))
1291    }

1293    /// Extract text from a selection range
1294    /// Returns the selected text as a String
1295    #[must_use]
1296    pub fn extract_selection_text(
1297        &self,
1298        start_line: usize,
1299        start_col: usize,
1300        end_line: usize,
1301        end_col: usize,
1302    ) -> String {
1303        let mut result = String::new();
1304
1305        if start_line == end_line {
1306            // Single line selection
1307            if let Some(wrapped) = self.wrapped_lines.get(start_line) {
1308                let line_text: String = wrapped
1309                    .spans
1310                    .iter()
1311                    .map(|(text, _, _, _)| text.as_str())
1312                    .collect();
1313                let chars: Vec<char> = line_text.chars().collect();
1314                let actual_end = end_col.min(chars.len());
1315                if start_col < chars.len() {
1316                    result.push_str(&chars[start_col..actual_end].iter().collect::<String>());
1317                }
1318            }
1319        } else {
1320            // Multi-line selection
1321            for line_idx in start_line..=end_line.min(self.wrapped_lines.len().saturating_sub(1)) {
1322                if let Some(wrapped) = self.wrapped_lines.get(line_idx) {
1323                    let line_text: String = wrapped
1324                        .spans
1325                        .iter()
1326                        .map(|(text, _, _, _)| text.as_str())
1327                        .collect();
1328                    let chars: Vec<char> = line_text.chars().collect();
1329
1330                    if line_idx == start_line {
1331                        // First line - from start_col to end
1332                        if start_col < chars.len() {
1333                            result.push_str(&chars[start_col..].iter().collect::<String>());
1334                        }
1335                    } else if line_idx == end_line {
1336                        // Last line - from beginning to end_col
1337                        let actual_end = end_col.min(chars.len());
1338                        result.push_str(&chars[..actual_end].iter().collect::<String>());
1339                    } else {
1340                        // Middle line - entire line
1341                        result.push_str(&line_text);
1342                    }
1343
1344                    // Add newline between lines (but not after last line)
1345                    if line_idx < end_line {
1346                        result.push('\n');
1347                    }
1348                }
1349            }
1350        }
1351
1352        result
1353    }

1356    /// Returns (`start_line_index`, `visible_lines`)
1357    #[must_use]
1358    pub fn get_visible_lines_info(&self, visible_height: usize) -> (usize, Vec<LineSegments>) {
1359        let total_lines = self.wrapped_lines.len();
1360
1361        // Calculate which lines are visible based on scroll mode
1362        let (start_line, end_line) = if let Some(pos) = self.scroll_position {
1363            // Scrolled back - use absolute position
1364            let start = pos;
1365            let end = (pos + visible_height).min(total_lines);
1366            (start, end)
1367        } else {
1368            // Live view mode - show the last visible_height lines
1369            let end = total_lines.saturating_sub(self.scroll_offset);
1370            let start = end.saturating_sub(visible_height);
1371            (start, end)
1372        };
1373
1374        // Collect visible lines
1375        let visible_lines: Vec<LineSegments> = (start_line..end_line)
1376            .filter_map(|idx| {
1377                self.wrapped_lines.get(idx).map(|line| LineSegments {
1378                    segments: line
1379                        .spans
1380                        .iter()
1381                        .map(|(text, style, span_type, link)| {
1382                            let resolved = self.style_with_defaults(*style);
1383                            TextSegment {
1384                                text: text.clone(),
1385                                fg: resolved.fg,
1386                                bg: resolved.bg,
1387                                bold: resolved.add_modifier.contains(Modifier::BOLD),
1388                                span_type: *span_type,
1389                                link_data: link.clone(),
1390                            }
1391                        })
1392                        .collect(),
1393                })
1394            })
1395            .collect();
1396
1397        (start_line, visible_lines)
1398    }
1399}

1537impl TextWindow {
1538    /// Render the window with optional focus indicator and selection highlighting
1539    /// Its title, with how far it is scrolled back when it is.
1540    fn scrolled_title(&self, total_lines: usize) -> String {
1541        if let Some(pos) = self.scroll_position {
1542            format!("{} [{}]", self.title, total_lines.saturating_sub(pos))
1543        } else if self.scroll_offset > 0 {
1544            format!("{} [{}]", self.title, self.scroll_offset)
1545        } else {
1546            self.title.clone()
1547        }
1548    }
1549
1550    /// Its border's look: its own colour, or, while it has the keyboard,
1551    /// `focused_border` in bold.
1552    fn border_look(&self, focused: bool, focused_border: Color) -> Style {
1553        let mut style = Style::default();
1554        if let Some(color) = self.border_color.as_deref().and_then(Self::parse_hex_color) {
1555            style = style.fg(color);
1556        }
1557        if focused {
1558            style = style
1559                .fg(support::to_ratatui_color(focused_border))
1560                .add_modifier(Modifier::BOLD);
1561        }
1562        style
1563    }
1564
1565    /// The wrapped lines shown, first and past the last: from where it is
1566    /// held when scrolled back, else the newest `height` lines.
1567    fn shown_lines(&self, total_lines: usize, height: usize) -> (usize, usize) {
1568        if let Some(pos) = self.scroll_position {
1569            (pos, (pos + height).min(total_lines))
1570        } else {
1571            let end = total_lines.saturating_sub(self.scroll_offset);
1572            (end.saturating_sub(height), end)
1573        }
1574    }
1575
1576    /// It drawn into `area` as `drawn` says.
1577    pub fn render_with_focus(&mut self, area: Rect, buf: &mut Buffer, drawn: Drawn<'_>) {
1578        let Drawn {
1579            focused,
1580            selection: selection_state,
1581            selection_bg: selection_bg_color,
1582            window: window_index,
1583            focused_border: focused_border_color,
1584        } = drawn;
1585        // Clear the area to prevent bleed-through from windows behind
1586        Clear.render(area, buf);
1587
1588        // Paint background before any content/borders so empty windows still show theme colors
1589        if let Some(bg_color) = self.background_color {
1590            crate::vellum::support::paint(area, buf, bg_color);
1591        }
1592
1593        // Pre-compute border geometry so we can rewrap to the actual inner width
1594        let borders = support::to_ratatui_borders(&self.border_sides);
1595        let border_type = crate::chrome::border_type(self.border_style.as_deref().unwrap_or(""));
1596        let inner_area = if self.show_border {
1597            Block::default()
1598                .borders(borders)
1599                .border_type(border_type)
1600                .inner(area)
1601        } else {
1602            area
1603        };
1604
1605        // Update width based on inner area and re-wrap if needed
1606        self.set_width(inner_area.width);
1607        if self.needs_rewrap {
1608            self.rewrap_all();
1609        }
1610
1611        let total_lines = self.wrapped_lines.len();
1612        let visible_height = inner_area.height as usize;
1613        self.last_visible_height = visible_height; // Save for scroll calculations
1614
1615        let title = self.scrolled_title(total_lines);
1616        let border_style = self.border_look(focused, focused_border_color);
1617
1618        // Draw border/title for the current window state
1619        let inner_area = crate::chrome::block_with_title(
1620            buf,
1621            area,
1622            (self.show_border, &self.border_sides),
1623            (border_type, border_style),
1624            (&title, self.title_position),
1625        );
1626
1627        if total_lines == 0 {
1628            return;
1629        }
1630
1631        // Calculate which lines to display based on scroll mode
1632        let (start_line, end_line) = self.shown_lines(total_lines, visible_height);
1633
1634        // Parse selection background color
1635        let selection_bg = Self::parse_hex_color(selection_bg_color);
1636
1637        // Collect lines from buffer (oldest to newest order)
1638        let mut display_lines: Vec<Line> = Vec::new();
1639        for idx in start_line..end_line {
1640            if let Some(wrapped) = self.wrapped_lines.get(idx) {
1641                // Check if this line has search matches
1642                let line_matches: Vec<&SearchMatch> = self
1643                    .search_state
1644                    .as_ref()
1645                    .map(|state| state.matches.iter().filter(|m| m.line_idx == idx).collect())
1646                    .unwrap_or_default();
1647
1648                let current_match = self
1649                    .search_state
1650                    .as_ref()
1651                    .and_then(|state| state.matches.get(state.current_match_idx));
1652
1653                let spans: Vec<Span> = if line_matches.is_empty() {
1654                    // No search matches - check for selection
1655                    self.create_spans_with_selection(
1656                        wrapped,
1657                        idx,
1658                        selection_state,
1659                        selection_bg,
1660                        window_index,
1661                    )
1662                } else {
1663                    // Has matches - need to highlight them
1664                    self.create_highlighted_spans(wrapped, &line_matches, current_match)
1665                };
1666
1667                display_lines.push(Line::from(spans));
1668            }
1669        }
1670
1671        // Lines are already in the correct order (oldest at top, newest at bottom)
1672        // No need to reverse!
1673
1674        // Calculate content alignment offset if specified
1675        // Only apply centering when content is LESS than window height
1676        // Once content fills the window, behave normally (top-aligned scrolling)
1677        let row_offset = if let Some(ref align_str) = self.content_align {
1678            let content_height = u16::try_from(display_lines.len()).unwrap_or(u16::MAX);
1679            // Only apply alignment if content is shorter than window
1680            if content_height < inner_area.height {
1681                let align = crate::vellum::support::ContentAlign::parse(align_str);
1682                let (offset, _) = align.calculate_offset(
1683                    inner_area.width,
1684                    content_height,
1685                    inner_area.width,
1686                    inner_area.height,
1687                );
1688                offset
1689            } else {
1690                // Content fills or exceeds window - use default top alignment
1691                0
1692            }
1693        } else {
1694            0
1695        };
1696
1697        // Apply row offset by padding top with empty lines
1698        let mut padded_lines = display_lines;
1699        if row_offset > 0 {
1700            let empty_lines: Vec<Line> = (0..row_offset).map(|_| Line::from("")).collect();
1701            padded_lines.splice(0..0, empty_lines);
1702        }
1703
1704        // Render content inside the inner area (border already drawn above)
1705        let mut paragraph = Paragraph::new(padded_lines);
1706
1707        // Apply horizontal alignment based on content_align setting
1708        if let Some(ref align) = self.content_align
1709            && align.contains("center")
1710        {
1711            paragraph = paragraph.alignment(Alignment::Center);
1712        }
1713
1714        paragraph.render(inner_area, buf);
1715    }

1717    fn fallback_text_color(&self) -> Color {

1721    fn style_with_defaults(&self, mut style: Style) -> Style {
1722        if matches!(style.fg, None | Some(Color::Reset)) {
1723            style = style.fg(self.fallback_text_color());
1724        }
1725
1726        if matches!(style.bg, None | Some(Color::Reset))
1727            && let Some(bg) = self.background_color
1728        {
1729            style = style.bg(bg);
1730        }
1731
1732        style
1733    }

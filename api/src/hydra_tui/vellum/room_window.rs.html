173    /// A word placed: on the line when it fits there, else on a new line,
174    /// else, longer than a line, broken where the lines end.
175    fn place_word(
176        &self,
177        word: Vec<TextSegment>,
178        word_len: usize,
179        (line, width): (&mut Vec<TextSegment>, &mut usize),
180        lines: &mut Vec<Vec<TextSegment>>,
181    ) {
182        if *width + word_len <= self.inner_width {
183            for part in word {
184                Self::append_to_line(line, part);
185            }
186            *width += word_len;
187        } else if word_len <= self.inner_width {
188            if !line.is_empty() {
189                lines.push(std::mem::take(line));
190                *width = 0;
191            }
192            for part in word {
193                Self::append_to_line(line, part);
194            }
195            *width += word_len;
196        } else {
197            for part in word {
198                for ch in part.text.chars() {
199                    if *width >= self.inner_width {
200                        lines.push(std::mem::take(line));
201                        *width = 0;
202                    }
203                    Self::append_to_line(line, Self::one_char(ch, &part));
204                    *width += 1;
205                }
206            }
207        }
208    }

210    /// A line of segments wrapped to the window's width, a word kept whole
211    /// where it fits, taken a character at a time so a segment splits where
212    /// the line ends.
213    fn wrap_line(&self, segments: Vec<TextSegment>) -> Vec<Vec<TextSegment>> {
214        if self.inner_width == 0 {
215            return vec![Vec::new()];
216        }
217
218        let mut wrapped_lines = Vec::new();
219        let mut current_line = Vec::new();
220        let mut current_width = 0;
221
222        // Track word buffer for smart wrapping
223        let mut word_buffer: Vec<TextSegment> = Vec::new();
224        let mut word_buffer_len = 0;
225        let mut in_word = false;
226
227        for segment in segments {
228            for ch in segment.text.chars() {
229                let is_whitespace = ch.is_whitespace();
230
231                if is_whitespace {
232                    // Flush word buffer if we have one
233                    if in_word && !word_buffer.is_empty() {
234                        self.place_word(
235                            std::mem::take(&mut word_buffer),
236                            word_buffer_len,
237                            (&mut current_line, &mut current_width),
238                            &mut wrapped_lines,
239                        );
240                        word_buffer_len = 0;
241                        in_word = false;
242                    }
243
244                    // Add whitespace immediately (don't buffer it)
245                    if current_width >= self.inner_width {
246                        // Wrap before whitespace
247                        wrapped_lines.push(std::mem::take(&mut current_line));
248                        current_width = 0;
249                        // Don't add whitespace at start of new line
250                        continue;
251                    }
252                    Self::append_to_line(&mut current_line, Self::one_char(ch, &segment));
253                    current_width += 1;
254                } else {
255                    // Non-whitespace character - add to word buffer
256                    in_word = true;
257                    Self::append_to_buffer(&mut word_buffer, Self::one_char(ch, &segment));
258                    word_buffer_len += 1;
259                }
260            }
261        }
262
263        if !word_buffer.is_empty() {
264            self.place_word(
265                word_buffer,
266                word_buffer_len,
267                (&mut current_line, &mut current_width),
268                &mut wrapped_lines,
269            );
270        }
271
272        // Add remaining line if not empty
273        if !current_line.is_empty() {
274            wrapped_lines.push(current_line);
275        }
276
277        // Return at least one empty line if nothing was added
278        if wrapped_lines.is_empty() {
279            wrapped_lines.push(Vec::new());
280        }
281
282        wrapped_lines
283    }

285    /// Helper to append a segment to a line, merging with last segment if style matches
286    fn append_to_line(line: &mut Vec<TextSegment>, segment: TextSegment) {
287        if let Some(last_seg) = line.last_mut() {
288            // Check if all properties match (including link_data for proper link boundaries)
289            let styles_match = last_seg.fg == segment.fg
290                && last_seg.bg == segment.bg
291                && last_seg.bold == segment.bold
292                && last_seg.span_type == segment.span_type
293                && last_seg.link_data == segment.link_data;
294
295            if styles_match {
296                last_seg.text.push_str(&segment.text);
297            } else {
298                line.push(segment);
299            }
300        } else {
301            line.push(segment);
302        }
303    }

325    /// Update inner dimensions based on window size
326    pub fn update_inner_size(&mut self, width: u16, height: u16) {
327        let new_width = if self.show_border {
328            (width.saturating_sub(2)) as usize
329        } else {
330            width as usize
331        };
332        let new_height = if self.show_border {
333            (height.saturating_sub(2)) as usize
334        } else {
335            height as usize
336        };
337
338        if new_width != self.inner_width || new_height != self.inner_height {
339            self.needs_rewrap = true;
340        }
341        self.inner_width = new_width;
342        self.inner_height = new_height;
343    }

358    /// Get all lines combined for wrapping
359    /// Layout: desc+objs on one line, players on next line, exits on next line
360    fn get_combined_lines(&self) -> Vec<Vec<TextSegment>> {
361        let mut all_lines: Vec<Vec<TextSegment>> = Vec::new();
362
363        if self.show_inline_title && !self.title.is_empty() {
364            let title_segment = vec![TextSegment {
365                text: self.title.clone(),
366                fg: self.title_fg_hex.clone(),
367                bg: self.title_bg_hex.clone(),
368                bold: true,
369                mono: false,
370                span_type: SpanType::Normal,
371                link_data: None,
372                custom_emoji: None,
373                inline_image: None,
374            }];
375            // Apply highlights to title
376            let highlighted = self
377                .highlight_engine
378                .apply_highlights_to_segments(&title_segment, "room")
379                .unwrap_or(title_segment);
380            all_lines.push(highlighted);
381        }
382
383        // Combine desc + objs on same line (only if visible)
384        let mut desc_and_objs_line = Vec::new();
385
386        // Add room desc segments (if visible)
387        if self.is_component_visible("room desc")
388            && let Some(desc_lines) = self.components.get("room desc")
389        {
390            for line in desc_lines {
391                desc_and_objs_line.extend(line.clone());
392            }
393        }
394
395        // Append room objs segments to same line (if visible)
396        if self.is_component_visible("room objs")
397            && let Some(objs_lines) = self.components.get("room objs")
398        {
399            for line in objs_lines {
400                desc_and_objs_line.extend(line.clone());
401            }
402        }
403
404        // Only add the combined line if it's not empty
405        if !desc_and_objs_line.is_empty() {
406            // Apply highlights to desc+objs line
407            let highlighted = self
408                .highlight_engine
409                .apply_highlights_to_segments(&desc_and_objs_line, "room")
410                .unwrap_or(desc_and_objs_line);
411            all_lines.push(highlighted);
412        }
413
414        // Add room players on own line (skip if empty or not visible)
415        if self.is_component_visible("room players")
416            && let Some(players_lines) = self.components.get("room players")
417            && !players_lines.is_empty()
418            && !players_lines.iter().all(std::vec::Vec::is_empty)
419        {
420            for line in players_lines {
421                if !line.is_empty() {
422                    // Apply highlights to players line
423                    let highlighted = self
424                        .highlight_engine
425                        .apply_highlights_to_segments(line, "room")
426                        .unwrap_or_else(|| line.clone());
427                    all_lines.push(highlighted);
428                }
429            }
430        }
431
432        // Add room exits on own line (skip if empty or not visible)
433        if self.is_component_visible("room exits")
434            && let Some(exits_lines) = self.components.get("room exits")
435            && !exits_lines.is_empty()
436            && !exits_lines.iter().all(std::vec::Vec::is_empty)
437        {
438            for line in exits_lines {
439                if !line.is_empty() {
440                    // Apply highlights to exits line
441                    let highlighted = self
442                        .highlight_engine
443                        .apply_highlights_to_segments(line, "room")
444                        .unwrap_or_else(|| line.clone());
445                    all_lines.push(highlighted);
446                }
447            }
448        }
449
450        all_lines
451    }

453    /// Rewrap all combined lines and cache them
454    fn rewrap_all(&mut self) {
455        self.wrapped_lines.clear();
456
457        let combined = self.get_combined_lines();
458        for line in combined {
459            let wrapped = self.wrap_line(line);
460            self.wrapped_lines.extend(wrapped);
461        }
462
463        self.needs_rewrap = false;
464    }

556    /// Render the room window
557    pub fn render(&mut self, area: Rect, buf: &mut Buffer) {
558        use ratatui::widgets::Widget;
559        Clear.render(area, buf);
560
561        if let Some(bg_color) = self.background_color {
562            crate::vellum::support::paint(area, buf, bg_color);
563        }
564
565        // Update inner size
566        self.update_inner_size(area.width, area.height);
567
568        // Rewrap if needed
569        if self.needs_rewrap {
570            self.rewrap_all();
571        }
572
573        // Create border block
574        let mut block = if self.show_border {
575            let borders = crate::vellum::support::to_ratatui_borders(&self.border_sides);
576
577            // Use styled title with roomName preset if provided
578            let title_line = if self.title_fg_hex.is_some() || self.title_bg_hex.is_some() {
579                let mut style = Style::default();
580                if let Some(ref fg) = self.title_fg_hex
581                    && let Some(color) = parse_color_to_ratatui(fg)
582                {
583                    style = style.fg(color);
584                }
585                if let Some(ref bg) = self.title_bg_hex
586                    && let Some(color) = parse_color_to_ratatui(bg)
587                {
588                    style = style.bg(color);
589                }
590                Line::from(Span::styled(
591                    self.title.clone(),
592                    style.add_modifier(Modifier::BOLD),
593                ))
594            } else {
595                Line::from(self.title.as_str())
596            };
597
598            Block::default().title(title_line).borders(borders)
599        } else {
600            Block::default()
601        };
602
603        if self.show_border {
604            if let Some(ref style_name) = self.border_style {
605                block = block.border_type(crate::chrome::border_type(style_name));
606            }
607
608            let border_color = self.border_color.unwrap_or(Color::White);
609            block = block.border_style(Style::default().fg(border_color));
610        }
611
612        // Calculate visible range
613        let total_lines = self.wrapped_lines.len();
614        let start_line = if total_lines > self.inner_height {
615            total_lines
616                .saturating_sub(self.inner_height)
617                .saturating_sub(self.scroll_offset)
618        } else {
619            0
620        };
621        let end_line = start_line + self.inner_height.min(total_lines);
622
623        // Get visible lines
624        let visible_lines: Vec<Line> = self.wrapped_lines[start_line..end_line.min(total_lines)]
625            .iter()
626            .map(|segments| {
627                let spans: Vec<Span> = segments
628                    .iter()
629                    .map(|seg| Span::styled(seg.text.clone(), self.segment_style(seg)))
630                    .collect();
631                Line::from(spans)
632            })
633            .collect();
634
635        let paragraph = Paragraph::new(visible_lines).block(block);
636        paragraph.render(area, buf);
637    }

639    fn segment_style(&self, segment: &TextSegment) -> Style {
640        let mut style = Style::default();
641
642        if let Some(ref fg) = segment.fg {
643            if let Some(color) = parse_color_to_ratatui(fg) {
644                style = style.fg(color);
645            }
646        } else if let Some(default_fg) = self.default_text_color {
647            style = style.fg(default_fg);
648        }
649
650        if let Some(ref bg) = segment.bg {
651            if let Some(color) = parse_color_to_ratatui(bg) {
652                style = style.bg(color);
653            }
654        } else if let Some(bg_color) = self.background_color {
655            style = style.bg(bg_color);
656        }
657
658        if segment.bold {
659            style = style.add_modifier(Modifier::BOLD);
660        }
661
662        style
663    }

665    fn parse_color_setting(color: Option<String>) -> Option<Color> {
666        color.and_then(|value| {
667            let trimmed = value.trim();
668            if trimmed.is_empty() || trimmed == "-" {
669                None
670            } else {
671                parse_color_to_ratatui(trimmed)
672            }
673        })

676    /// Convert mouse position to text coordinates
677    /// Returns (`line_index`, `column_index`) if position is within content area
678    #[must_use]
679    pub fn mouse_to_text_coords(
680        &self,
681        mouse_col: u16,
682        mouse_row: u16,
683        window_rect: Rect,
684    ) -> Option<(usize, usize)> {
685        let border_offset = u16::from(self.show_border);
686
687        // Bounds check within content area
688        if mouse_col < window_rect.x + border_offset
689            || mouse_col >= window_rect.x + window_rect.width - border_offset
690            || mouse_row < window_rect.y + border_offset
691            || mouse_row >= window_rect.y + window_rect.height - border_offset
692        {
693            return None;
694        }
695
696        let visible_height = (window_rect.height.saturating_sub(2 * border_offset)) as usize;
697        let total_lines = self.wrapped_lines.len();
698        let start_line = if total_lines > visible_height + self.scroll_offset {
699            total_lines - visible_height - self.scroll_offset
700        } else {
701            0
702        };
703
704        let line_idx = start_line + (mouse_row - window_rect.y - border_offset) as usize;
705        let col_offset = (mouse_col - window_rect.x - border_offset) as usize;
706
707        Some((line_idx, col_offset))
708    }
709
710    /// Extract text from a selection range
711    #[must_use]
712    pub fn extract_selection_text(
713        &self,
714        start_line: usize,
715        start_col: usize,
716        end_line: usize,
717        end_col: usize,
718    ) -> String {
719        if self.wrapped_lines.is_empty() {
720            return String::new();
721        }
722
723        let mut result = String::new();
724
725        for line_idx in start_line..=end_line.min(self.wrapped_lines.len().saturating_sub(1)) {
726            if line_idx >= self.wrapped_lines.len() {
727                break;
728            }
729
730            let line = &self.wrapped_lines[line_idx];
731            let line_text: String = line.iter().map(|seg| seg.text.as_str()).collect();
732            let line_len = line_text.chars().count();
733
734            if start_line == end_line {
735                // Single line selection
736                let start = start_col.min(line_len);
737                let end = end_col.min(line_len);
738                if start < end {
739                    result.push_str(
740                        &line_text
741                            .chars()
742                            .skip(start)
743                            .take(end - start)
744                            .collect::<String>(),
745                    );
746                }
747            } else if line_idx == start_line {
748                // First line of multi-line selection
749                let start = start_col.min(line_len);
750                result.push_str(&line_text.chars().skip(start).collect::<String>());
751                result.push('\n');
752            } else if line_idx == end_line {
753                // Last line of multi-line selection
754                let end = end_col.min(line_len);
755                result.push_str(&line_text.chars().take(end).collect::<String>());
756            } else {
757                // Middle lines - take all
758                result.push_str(&line_text);
759                result.push('\n');
760            }
761        }
762
763        result
764    }

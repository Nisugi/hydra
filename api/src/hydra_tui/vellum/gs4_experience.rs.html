1//! `GemStone` IV experience widget.
2//!
3//! Displays GS4 experience data from the `expr` dialog:
4//! - Level text (yourLvl label)
5//! - Mind state progress bar (mindState)
6//! - Experience progress bar (nextLvlPB)
7//!
8//! Reads data from `GameState.gs4_experience` (populated from dialogData updates).

171    /// Update the widget from `GS4ExperienceState`.
172    /// Returns true if the display changed.
173    pub fn update_from_state(&mut self, state: &GS4ExperienceState) -> bool {
174        // Quick check: if generation matches, no update needed
175        if self.generation == state.generation {
176            return false;
177        }
178
179        self.generation = state.generation;
180        self.level_text.clone_from(&state.level_text);
181        self.mind_value = state.mind_state_value;
182        self.mind_text.clone_from(&state.mind_state_text);
183        self.next_level_value = state.next_level_value;
184        self.next_level_text.clone_from(&state.next_level_text);
185        self.exp_total = state.exp;
186        self.ascension_exp = state.ascension_exp;
187
188        true
189    }

191    /// Group digits in threes: 1234567 -> "1,234,567".
192    fn format_thousands(value: u64) -> String {
193        let digits = value.to_string();
194        let mut out = String::with_capacity(digits.len() + digits.len() / 3);
195        for (i, ch) in digits.chars().enumerate() {
196            if i > 0 && (digits.len() - i).is_multiple_of(3) {
197                out.push(',');
198            }
199            out.push(ch);
200        }
201        out
202    }

204    /// Render one plain text line at `y`, returning the next row.
205    fn render_text_row(&self, inner: Rect, buf: &mut Buffer, y: u16, text: String) -> u16 {
206        let line = Line::from(Span::styled(text, Style::default().fg(self.text_color)));
207        let line_area = Rect {
208            x: inner.x,
209            y,
210            width: inner.width,
211            height: 1,
212        };
213        let para = ratatui::widgets::Paragraph::new(line).alignment(self.align);
214        para.render(line_area, buf);
215        y + 1
216    }

218    /// Render a simple progress bar within a single line
219    /// `fill_color`: Some(color) = use that color for filled portion, None = use theme background
220    fn render_bar(
221        &self,
222        area: Rect,
223        buf: &mut Buffer,
224        value: u32,
225        text: &str,
226        fill_color: Option<Color>,
227    ) {
228        if area.width == 0 || area.height == 0 {
229            return;
230        }
231
232        let bar_width = area.width as usize;
233        let filled_width =
234            (u32::try_from(bar_width).unwrap_or(u32::MAX) * value.min(100) / 100) as usize;
235
236        // Prepare display text, truncate if needed
237        let display_text = if text.len() > bar_width {
238            &text[..bar_width]
239        } else {
240            text
241        };
242
243        // Center the text
244        let text_start = (bar_width.saturating_sub(display_text.len())) / 2;
245
246        for col in 0..bar_width {
247            let x = area.x + u16::try_from(col).unwrap_or(u16::MAX);
248            let y = area.y;
249
250            if x >= buf.area().width || y >= buf.area().height {
251                continue;
252            }
253
254            let is_filled = col < filled_width;
255
256            // Determine character at this position
257            let ch = if col >= text_start && col < text_start + display_text.len() {
258                display_text.chars().nth(col - text_start).unwrap_or(' ')
259            } else {
260                ' '
261            };
262
263            if is_filled {
264                buf[(x, y)].set_char(ch);
265                if let Some(fc) = fill_color {
266                    // Has fill color - use it with black text for contrast
267                    buf[(x, y)].set_fg(Color::Black);
268                    buf[(x, y)].set_bg(fc);
269                } else {
270                    // No fill color (e.g., exp bar for max-level users) - use theme background
271                    buf[(x, y)].set_fg(self.text_color);
272                    if let Some(bg) = self.background_color {
273                        buf[(x, y)].set_bg(bg);
274                    }
275                }
276            } else {
277                buf[(x, y)].set_char(ch);
278                buf[(x, y)].set_fg(self.text_color);
279                // Only set bg if we have a theme background, otherwise leave transparent
280                if let Some(bg) = self.background_color {
281                    buf[(x, y)].set_bg(bg);
282                }
283            }
284        }
285    }

287    /// Render the experience widget
288    /// The window's border and title drawn, as its settings ask: the inside.
289    fn frame(&self, area: Rect, buf: &mut Buffer) -> Rect {
290        if !(self.show_border && self.border_sides.any()) {
291            return area;
292        }
293        let borders = crate::vellum::support::to_ratatui_borders(&self.border_sides);
294        let mut block = Block::default()
295            .borders(borders)
296            .border_style(Style::default().fg(self.border_color));
297        if self.show_title {
298            block = block.title(self.title.as_str());
299        }
300        let inner = block.inner(area);
301        block.render(area, buf);
302        inner
303    }
304
305    /// It drawn into `area`.
306    pub fn render(&self, area: Rect, buf: &mut Buffer) {
307        if let Some(bg_color) = self.background_color {
308            crate::vellum::support::paint(area, buf, bg_color);
309        }
310        let inner = self.frame(area, buf);
311
312        if inner.width == 0 || inner.height == 0 {
313            return;
314        }
315
316        // If no data, show placeholder
317        if self.level_text.is_empty()
318            && self.mind_text.is_empty()
319            && self.next_level_text.is_empty()
320        {
321            let placeholder = Line::from(Span::styled(
322                "(No experience data)",
323                Style::default().fg(Color::DarkGray),
324            ));
325            let placeholder_text =
326                ratatui::widgets::Paragraph::new(placeholder).alignment(self.align);
327            placeholder_text.render(inner, buf);
328            return;
329        }
330
331        let mut current_y = inner.y;
332
333        // Row 1: Level text (if show_level enabled)
334        if self.show_level && inner.height > 0 && !self.level_text.is_empty() {
335            let level_line = Line::from(Span::styled(
336                self.level_text.clone(),
337                Style::default().fg(self.text_color),
338            ));
339            let line_area = Rect {
340                x: inner.x,
341                y: current_y,
342                width: inner.width,
343                height: 1,
344            };
345            let para = ratatui::widgets::Paragraph::new(level_line).alignment(self.align);
346            para.render(line_area, buf);
347            current_y += 1;
348        }
349
350        // Row 2: Mind state bar (if enabled)
351        if self.show_mind_bar && current_y < inner.y + inner.height && !self.mind_text.is_empty() {
352            let bar_area = Rect {
353                x: inner.x,
354                y: current_y,
355                width: inner.width,
356                height: 1,
357            };
358            self.render_bar(
359                bar_area,
360                buf,
361                self.mind_value,
362                &self.mind_text,
363                Some(self.mind_bar_color),
364            );
365            current_y += 1;
366        }
367
368        // Row 3: Exp progress bar (if show_exp_bar enabled)
369        if self.show_exp_bar
370            && current_y < inner.y + inner.height
371            && !self.next_level_text.is_empty()
372        {
373            let bar_area = Rect {
374                x: inner.x,
375                y: current_y,
376                width: inner.width,
377                height: 1,
378            };
379            self.render_bar(
380                bar_area,
381                buf,
382                self.next_level_value,
383                &self.next_level_text,
384                self.exp_bar_color,
385            );
386            current_y += 1;
387        }
388
389        // Row 4: Total absorbed experience (if enabled and the feed sent it)
390        if self.show_total_exp
391            && current_y < inner.y + inner.height
392            && let Some(total) = self.exp_total
393        {
394            current_y = self.render_text_row(
395                inner,
396                buf,
397                current_y,
398                format!("Exp: {}", Self::format_thousands(total)),
399            );
400        }
401
402        // Row 5: Total ascension experience (if enabled and the feed sent it)
403        if self.show_ascension_exp
404            && current_y < inner.y + inner.height
405            && let Some(ascension) = self.ascension_exp
406        {
407            self.render_text_row(
408                inner,
409                buf,
410                current_y,
411                format!("Ascension: {}", Self::format_thousands(ascension)),
412            );
413        }
414    }

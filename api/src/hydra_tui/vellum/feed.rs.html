27/// A colour as the widgets take it, `#rrggbb`.
28fn hex(color: Color) -> String {
29    match color {
30        Color::Rgb(red, green, blue) => format!("#{red:02x}{green:02x}{blue:02x}"),
31        _ => "#ffffff".to_owned(),
32    }
33}

42/// Draw window `def` into `area`, the inside of its frame, when it is a
43/// kind the widgets here draw. Whether it was.
44pub fn draw(
45    def: &WindowDef,
46    buf: &mut Buffer,
47    area: Rect,
48    (seen, hits): (&Seen<'_>, &mut Hits),
49) -> bool {
50    if area.width == 0 || area.height == 0 {
51        return true;
52    }
53    match def {
54        WindowDef::Progress { base, data } => progress((base, data), buf, area, seen),
55        WindowDef::Countdown { base, data } => countdown_of((base, data), buf, area, seen),
56        WindowDef::Hand { base, data } => hand_of((base, data), buf, area, (seen, hits)),
57        WindowDef::Compass { base, data } => compass_of((base, data), buf, area, seen),
58        WindowDef::Indicator { base, data } => indicator_of((base, data), buf, area, seen),
59        WindowDef::Dashboard { base, data } => dashboard_of((base, data), buf, area, seen),
60        WindowDef::InjuryDoll { base, data } => injuries_of((base, data), buf, area, seen),
61        WindowDef::MiniVitals { base, data } => minivitals_of((base, data), buf, area, seen),
62        WindowDef::Encumbrance { base, data } => encumbrance_of((base, data), buf, area, seen),
63        WindowDef::Spacer { .. } => true,
64        WindowDef::Targets { base, data } => {
65            super::lists::targets((base, data), buf, area, (seen, hits));
66            true
67        }
68        WindowDef::Players { base, .. } => {
69            super::lists::players(base, buf, area, (seen, hits));
70            true
71        }
72        WindowDef::Items { base, .. } => {
73            super::lists::items(base, buf, area, seen);
74            true
75        }
76        WindowDef::Room { base, data } => {
77            super::lists::room((base, data), buf, area, (seen, hits));
78            true
79        }
80        WindowDef::GS4Experience { base, data } => {
81            super::lists::experience((base, data), buf, area, seen);
82            true
83        }
84        WindowDef::Quickbar { base, .. } => {
85            super::lists::quick(base, buf, area, (seen, hits));
86            true
87        }
88        WindowDef::MissingSpells { .. } => {
89            super::panels::missing(buf, area, (seen, hits));
90            true
91        }
92        WindowDef::Quests { base } => {
93            super::panels::quests(base, buf, area, (seen, hits));
94            true
95        }
96        WindowDef::Spells { base, .. } => {
97            super::panels::spells(base, buf, area, (seen, hits));
98            true
99        }
100        WindowDef::Inventory { base, data } => {
101            super::panels::inventory((base, data), buf, area, (seen, hits));
102            true
103        }
104        WindowDef::Container { base, data } => {
105            super::panels::container((base, data), buf, area, (seen, hits));
106            true
107        }
108        WindowDef::Containers { base, .. } => {
109            super::panels::containers(base, buf, area, (seen, hits));
110            true
111        }
112        WindowDef::Betrayer { base, data } => {
113            super::panels::betrayer((base, data), buf, area, seen);
114            true
115        }
116        WindowDef::DialogPanel { data, .. } => {
117            super::panels::dialog_panel(data, buf, area, seen);
118            true
119        }
120        _ => false,
121    }
122}

129/// The indicator a layout names `id`.
130fn lamp(id: &str) -> Option<Lamp> {
131    let id = id.to_ascii_lowercase();
132    let id = id.strip_prefix("icon").unwrap_or(&id);
133    Lamp::ALL
134        .into_iter()
135        .find(|lamp| lamp.id() == id || lamp.name().eq_ignore_ascii_case(id))
136}

138/// The game's bar `id` as a bar shows it: how full, out of 100, and the
139/// words the game puts on it (`health 348/400`, `defensive`).
140fn progress_of(state: &GameState, id: &str) -> Option<(u32, String)> {
141    let vital = |name: &str, vital: Option<hydra_model::state::vitals::Vital>| {
142        let vital = vital?;
143        let said = match (vital.current, vital.max) {
144            (Some(current), Some(max)) => format!("{name} {current}/{max}"),
145            _ => format!("{name} {}%", vital.percent),
146        };
147        Some((vital.percent, said))
148    };
149    let character = &state.character;
150    let experience = &character.experience;
151    let worded = |words: Option<&String>, percent: Option<u32>| {
152        Some((
153            percent.unwrap_or(0),
154            words.map(|words| hydra_ui::sheet::worded(words).to_owned())?,
155        ))
156    };
157    match id {
158        "health" => vital("health", state.health()),
159        "mana" | "concentration" => vital("mana", state.mana()),
160        "stamina" => vital("stamina", state.stamina()),
161        "spirit" => vital("spirit", state.spirit()),
162        "pbarStance" | "stance" => worded(character.stance.as_ref(), character.stance_percent),
163        "encumlevel" | "encumbrance" => worded(
164            character.encumbrance.level.as_ref(),
165            character.encumbrance.percent,
166        ),
167        "mindState" | "mind" => worded(experience.mind_state.as_ref(), experience.mind_percent),
168        "nextLvlPB" => worded(
169            experience.next_level.as_ref(),
170            experience.next_level_percent,
171        ),
172        _ => None,
173    }
174}

176/// The numbers in a bar's words, `348/400` in `health 348/400`.
177fn numbers(said: &str) -> Option<(String, String)> {
178    let (current, max) = said.split_whitespace().last()?.split_once('/')?;
179    Some((current.to_owned(), max.to_owned()))
180}

197/// What a hand of `kind` holds, as words, and the thing held to link to:
198/// its id and noun.
199fn held(state: Option<&GameState>, kind: hand::HandType) -> (String, Option<(String, String)>) {
200    let Some(state) = state else {
201        return (String::new(), None);
202    };
203    match kind {
204        hand::HandType::Spell => (
205            state
206                .prepared
207                .clone()
208                .filter(|spell| !spell.eq_ignore_ascii_case("none"))
209                .unwrap_or_default(),
210            state
211                .prepared_id
212                .clone()
213                .map(|exist| (exist, String::new())),
214        ),
215        hand::HandType::Left | hand::HandType::Right => {
216            let hand = if matches!(kind, hand::HandType::Left) {
217                &state.left_hand
218            } else {
219                &state.right_hand
220            };
221            match hand {
222                Hand::Holding { id, noun, name } => (
223                    name.clone(),
224                    id.clone().map(|id| (id, noun.clone().unwrap_or_default())),
225                ),
226                Hand::Empty | Hand::Unknown => (String::new(), None),
227            }
228        }
229    }
230}

232/// A progress window drawn: whether it was.
233fn progress(
234    (base, data): (&WindowBase, &ProgressWidgetData),
235    buf: &mut Buffer,
236    area: Rect,
237    seen: &Seen<'_>,
238) -> bool {
239    let state = seen.state();
240    let id = data.id.as_deref().unwrap_or(&base.name);
241    // Nothing known yet: an empty bar.
242    let (percent, said) = state
243        .and_then(|state| progress_of(state, id))
244        .unwrap_or_default();
245    let mut bar = progress_bar::ProgressBar::new("");
246    bar.set_border_config(false, None, None, BorderSides::default());
247    let text = if data.current_only || data.numbers_only {
248        // The numbers alone, from what the game said.
249        numbers(&said).map_or_else(
250            || said.clone(),
251            |(current, max)| {
252                if data.current_only {
253                    current
254                } else {
255                    format!("{current}/{max}")
256                }
257            },
258        )
259    } else {
260        said
261    };
262    bar.set_value_with_text(percent, 100, Some(text));
263    let fill = data
264        .color
265        .clone()
266        .or_else(|| bar_color(id).map(str::to_owned));
267    bar.set_colors(fill, None);
268    bar.set_text_color(Some(text_color(base, seen)));
269    bar.set_transparent_background(true);
270    bar.render(area, buf);
271    true
272}

274/// A countdown window drawn: whether it was.
275fn countdown_of(
276    (base, data): (&WindowBase, &CountdownWidgetData),
277    buf: &mut Buffer,
278    area: Rect,
279    seen: &Seen<'_>,
280) -> bool {
281    let state = seen.state();
282    let id = data.id.as_deref().unwrap_or(&base.name);
283    let left = state.and_then(|state| match id {
284        "casttime" => state.casttime_remaining(),
285        "stun" | "stuntime" => state.stun_remaining(),
286        _ => state.roundtime_remaining(),
287    });
288    let mut clock = countdown::Countdown::new("");
289    clock.set_border_config(false, None, None);
290    if let Some(icon) = data.icon {
291        clock.set_icon(icon);
292    }
293    clock.set_show_when_zero(data.show_when_zero.unwrap_or(false));
294    clock.set_count_past_zero(data.count_past_zero.unwrap_or(false));
295    clock.set_text_color(Some(
296        data.color.clone().unwrap_or_else(|| text_color(base, seen)),
297    ));
298    clock.set_background_color(data.countdown_background_color.clone());
299    clock.set_transparent_background(data.countdown_background_color.is_none());
300    // The end as a moment on the computer's clock, the game's seconds
301    // left from now.
302    let now = std::time::SystemTime::now()
303        .duration_since(std::time::UNIX_EPOCH)
304        .map_or(0, |since| {
305            i64::try_from(since.as_secs()).unwrap_or(i64::MAX)
306        });
307    clock.set_end_time(now + i64::from(left.unwrap_or(0)));
308    clock.render(area, buf, 0, &super::support::AppTheme::default());
309    true
310}

312/// A hand window drawn: whether it was.
313fn hand_of(
314    (base, data): (&WindowBase, &HandWidgetData),
315    buf: &mut Buffer,
316    area: Rect,
317    (seen, hits): (&Seen<'_>, &mut Hits),
318) -> bool {
319    let state = seen.state();
320    let which = data
321        .hand
322        .as_deref()
323        .unwrap_or(&base.name)
324        .to_ascii_lowercase();
325    let kind = match which.as_str() {
326        "left" | "left_hand" => hand::HandType::Left,
327        "spell" | "spell_hand" => hand::HandType::Spell,
328        _ => hand::HandType::Right,
329    };
330    let mut widget = hand::Hand::new("", kind);
331    widget.set_border_config(false, None, None);
332    if let Some(icon) = &data.icon {
333        widget.set_icon(icon.clone());
334    }
335    widget.set_icon_color(data.icon_color.clone());
336    widget.set_text_color(Some(text_color(base, seen)));
337    let (content, link) = held(state, kind);
338    widget.set_content(content);
339    let linked = link.is_some();
340    widget.set_link_data(link.clone().map(|link| LinkData {
341        exist_id: link.0,
342        noun: link.1,
343        text: String::new(),
344        coord: None,
345    }));
346    // A held thing reads as a link: the hand's own colour, else the
347    // links'.
348    let highlight = data
349        .hand_text_color
350        .clone()
351        .or_else(|| linked.then(|| hex(seen.colors.color(Token::Link))));
352    widget.set_content_highlight_color(highlight);
353    widget.set_transparent_background(true);
354    widget.render(area, buf);
355    if let (Some((exist, noun)), Some(seat)) = (link, seen.own.then_some(seen.seat)) {
356        hits.add(
357            area,
358            Target::Link {
359                seat,
360                link: RunLink::Object {
361                    exist,
362                    noun,
363                    coord: None,
364                },
365            },
366        );
367    }
368    true
369}

393/// An indicator window drawn: whether it was.
394fn indicator_of(
395    (base, data): (&WindowBase, &IndicatorWidgetData),
396    buf: &mut Buffer,
397    area: Rect,
398    seen: &Seen<'_>,
399) -> bool {
400    let state = seen.state();
401    let id = data.indicator_id.as_deref().unwrap_or(&base.name);
402    let Some(lamp) = lamp(id) else {
403        return false;
404    };
405    let mut widget = indicator::Indicator::new("");
406    widget.set_border_config(false, None, None);
407    widget.set_title(data.icon.clone().unwrap_or_else(|| lamp.name().to_owned()));
408    widget.set_active(state.is_some_and(|state| state.status.get(lamp.id())));
409    let on = data
410        .active_color
411        .clone()
412        .unwrap_or_else(|| hex(seen.colors.color(lamp.token())));
413    let off = data
414        .inactive_color
415        .clone()
416        .unwrap_or_else(|| "#555555".to_owned());
417    widget.set_colors(off, on);
418    widget.set_transparent_background(true);
419    widget.render(area, buf);
420    true
421}

423/// A dashboard window drawn: whether it was.
424fn dashboard_of(
425    (base, data): (&WindowBase, &DashboardWidgetData),
426    buf: &mut Buffer,
427    area: Rect,
428    seen: &Seen<'_>,
429) -> bool {
430    let state = seen.state();
431    let mut widget = dashboard::Dashboard::new("", DashboardLayout::parse(&data.layout));
432    widget.set_border_config(false, None, None);
433    widget.set_spacing(data.spacing);
434    widget.set_hide_inactive(data.hide_inactive);
435    widget.set_content_align(base.content_align.clone());
436    for each in &data.indicators {
437        let colors = if each.colors.is_empty() {
438            vec!["#ffffff".to_owned()]
439        } else {
440            each.colors.clone()
441        };
442        widget.add_indicator(each.id.clone(), each.icon.clone(), colors);
443        let on = lamp(&each.id)
444            .is_some_and(|lamp| state.is_some_and(|state| state.status.get(lamp.id())));
445        widget.set_indicator_value(&each.id, u8::from(on));
446    }
447    widget.set_transparent_background(true);
448    widget.render(area, buf);
449    true
450}

452/// An injury doll window drawn: whether it was.
453fn injuries_of(
454    (base, data): (&WindowBase, &InjuryDollWidgetData),
455    buf: &mut Buffer,
456    area: Rect,
457    seen: &Seen<'_>,
458) -> bool {
459    let state = seen.state();
460    let mut widget = injury_doll::InjuryDoll::new("");
461    widget.set_border_config(false, None, None);
462    widget.set_content_align(base.content_align.clone());
463    widget.set_colors(data.resolved_colors().to_vec());
464    if let Some(state) = state {
465        for (part, injury) in &state.character.injuries {
466            let level = if injury.wound > 0 {
467                injury.wound
468            } else if injury.scar > 0 {
469                3 + injury.scar
470            } else {
471                0
472            };
473            widget.set_injury(part.clone(), level);
474        }
475    }
476    widget.set_transparent_background(true);
477    widget.render(area, buf);
478    true

481/// A minivitals window drawn: whether it was.
482fn minivitals_of(
483    (base, data): (&WindowBase, &MiniVitalsWidgetData),
484    buf: &mut Buffer,
485    area: Rect,
486    seen: &Seen<'_>,
487) -> bool {
488    let state = seen.state();
489    let mut widget = minivitals::MiniVitals::new("", false);
490    widget.set_show_border(false);
491    widget.set_show_title(false);
492    widget.set_display_mode(data.numbers_only, data.current_only);
493    widget.set_bar_order(data.bar_order.clone());
494    widget.set_depleted_color(data.depleted_color.clone());
495    if let Some(color) = color_of(base.text_color.as_deref()) {
496        widget.set_text_color(color);
497    }
498    for (set, color) in [
499        (
500            minivitals::MiniVitals::set_health_color as fn(&mut minivitals::MiniVitals, Color),
501            &data.health_color,
502        ),
503        (minivitals::MiniVitals::set_mana_color, &data.mana_color),
504        (
505            minivitals::MiniVitals::set_stamina_color,
506            &data.stamina_color,
507        ),
508        (minivitals::MiniVitals::set_spirit_color, &data.spirit_color),
509    ] {
510        if let Some(color) = color_of(color.as_deref()) {
511            set(&mut widget, color);
512        }
513    }
514    if let Some(state) = state {
515        let entry = |id: &str| {
516            progress_of(state, id).map_or_else(VitalEntry::default, |(value, text)| VitalEntry {
517                value,
518                max: 100,
519                text,
520            })
521        };
522        widget.update_from_state(&MiniVitalsState {
523            health: entry("health"),
524            mana: entry("mana"),
525            stamina: entry("stamina"),
526            spirit: entry("spirit"),
527            generation: 1,
528        });
529    }
530    widget.render(area, buf);
531    true
532}

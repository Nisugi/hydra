17/// Injury doll widget showing body part injuries/scars
18///
19/// Three parts have no place on the figure, so they ride as two-letter
20/// labels down the right side: neck, back, then nervous system.
21/// Layout:
22///  👁   👁
23///     0    nk
24///    /|\
25///   o | o  bk
26///    / \
27///   o   o  ns
28pub struct InjuryDoll {
29    label: String,
30    // Map body part name to injury level (0=none, 1-3=injury, 4-6=scar)
31    injuries: HashMap<String, u8>,
32    show_border: bool,
33    border_style: Option<String>,
34    border_color: Option<Color>,
35    border_sides: crate::layout::vellum_schema::BorderSides,
36    // The colours for none, each wound and each scar, in that order.
37    colors: Vec<String>,
38    background_color: Option<Color>,
39    content_align: Option<String>,
40    transparent_background: bool,
41}
42
43impl InjuryDoll {
44    /// A doll titled `label`, with no wounds.
45    #[must_use]
46    pub fn new(label: &str) -> Self {
47        Self {
48            label: label.to_string(),
49            injuries: HashMap::new(),
50            show_border: false,
51            border_style: None,
52            border_color: None,
53            border_sides: crate::layout::vellum_schema::BorderSides::default(),
54            // Shared default palette (single source of truth); overridden per
55            // widget via set_colors from the resolved config.
56            colors: crate::layout::vellum_schema::DEFAULT_INJURY_PALETTE
57                .iter()
58                .map(std::string::ToString::to_string)
59                .collect(),
60            background_color: None,
61            content_align: None,
62            transparent_background: false, // Default to transparent
63        }
64    }
65
66    /// Whether its border is drawn, and its style and colour.
67    pub fn set_border_config(
68        &mut self,
69        show_border: bool,
70        border_style: Option<String>,
71        border_color: Option<String>,
72    ) {
73        self.show_border = show_border;
74        self.border_style = border_style;
75        self.border_color = border_color.and_then(|value| Self::parse_color(&value));
76    }
77
78    /// Which of its border's edges are drawn.
79    pub fn set_border_sides(&mut self, border_sides: crate::layout::vellum_schema::BorderSides) {
80        self.border_sides = border_sides;
81    }
82
83    /// Its title.
84    pub fn set_title(&mut self, title: String) {
85        self.label = title;
86    }
87
88    /// The wound or scar at `body_part`: 0 none, 1 to 3 a wound, 4 to 6 a
89    /// scar.
90    pub fn set_injury(&mut self, body_part: String, level: u8) {
91        self.injuries.insert(body_part, level.min(6));
92    }
93
94    /// The colours for none, each wound and each scar, in that order,
95    /// `#rrggbb`.
96    pub fn set_colors(&mut self, colors: Vec<String>) {
97        if colors.len() == 7 {
98            self.colors = colors;
99        }
100    }
101
102    /// The colour behind it, `#rrggbb`.
103    pub fn set_background_color(&mut self, color: Option<String>) {
104        self.background_color = match color {
105            Some(ref s) if s.trim() == "-" => None,
106            Some(value) => Self::parse_color(&value),
107            None => None,
108        };
109    }
110
111    /// Where its content sits in the window, as `ContentAlign` names it.
112    pub fn set_content_align(&mut self, align: Option<String>) {
113        self.content_align = align;
114    }
115
116    /// Whether what is behind it shows through.
117    pub fn set_transparent_background(&mut self, transparent: bool) {
118        self.transparent_background = transparent;
119    }
120
121    fn parse_color(input: &str) -> Option<Color> {
122        parse_color_to_ratatui(input)
123    }
124
125    fn get_injury_color(&self, body_part: &str) -> Color {
126        let level = self.injuries.get(body_part).copied().unwrap_or(0);
127        self.colors
128            .get(usize::from(level))
129            .and_then(|color| Self::parse_color(color))
130            .unwrap_or(Color::White)
131    }

133    /// The window's frame drawn when it has one: the inside.
134    fn frame(&self, area: Rect, buf: &mut Buffer) -> Rect {
135        let mut block = Block::default();
136        if !self.transparent_background
137            && let Some(bg_color) = self.background_color
138        {
139            block = block.style(Style::default().bg(bg_color));
140        }
141
142        if self.show_border {
143            let borders = support::to_ratatui_borders(&self.border_sides);
144            block = block.borders(borders);
145
146            if let Some(ref style) = self.border_style {
147                block = block.border_type(crate::chrome::border_type(style));
148            }
149
150            if let Some(color) = self.border_color {
151                block = block.border_style(Style::default().fg(color));
152            }
153
154            // Only set title if label is non-empty (avoids empty title affecting layout)
155            if !self.label.is_empty() {
156                block = block.title(self.label.as_str());
157            }
158        }
159
160        if !self.show_border {
161            return area;
162        }
163        let inner_area = block.inner(area);
164        block.render(area, buf);
165        inner_area
166    }

168    /// It drawn into `area`.
169    pub fn render(&self, area: Rect, buf: &mut Buffer) {
170        if !self.transparent_background {
171            Clear.render(area, buf);
172            if let Some(bg_color) = self.background_color {
173                crate::vellum::support::paint(area, buf, bg_color);
174            }
175        }
176
177        let inner_area = self.frame(area, buf);
178
179        if inner_area.width == 0 || inner_area.height == 0 {
180            return;
181        }
182
183        let bg_color = self.background_color;
184
185        let (content_width, content_height) = doll_size();
186
187        let (row_offset, col_offset) = if let Some(ref align_str) = self.content_align {
188            let align = crate::vellum::support::ContentAlign::parse(align_str);
189            align.calculate_offset(
190                content_width,
191                content_height,
192                inner_area.width,
193                inner_area.height,
194            )
195        } else {
196            (0, 0) // Default to top-left
197        };
198
199        // Render body parts
200        for (col, row, ch, body_part) in &PARTS {
201            let x = inner_area.x + col + col_offset;
202            let y = inner_area.y + row + row_offset;
203
204            // Bounds check - must be within inner_area AND buffer
205            if x >= inner_area.x
206                && x < inner_area.x + inner_area.width
207                && y >= inner_area.y
208                && y < inner_area.y + inner_area.height
209                && x < buf.area().width
210                && y < buf.area().height
211            {
212                let color = self.get_injury_color(body_part);
213                buf[(x, y)].set_char(*ch);
214                buf[(x, y)].set_fg(color);
215                if !self.transparent_background
216                    && let Some(bg) = bg_color
217                {
218                    buf[(x, y)].set_bg(bg);
219                }
220            }
221        }
222
223        for (start_col, row, text, body_part) in &MARKS {
224            let color = self.get_injury_color(body_part);
225
226            for (i, ch) in text.chars().enumerate() {
227                let x =
228                    inner_area.x + start_col + u16::try_from(i).unwrap_or(u16::MAX) + col_offset;
229                let y = inner_area.y + row + row_offset;
230
231                // Bounds check - must be within inner_area AND buffer
232                if x >= inner_area.x
233                    && x < inner_area.x + inner_area.width
234                    && y >= inner_area.y
235                    && y < inner_area.y + inner_area.height
236                    && x < buf.area().width
237                    && y < buf.area().height
238                {
239                    buf[(x, y)].set_char(ch);
240                    buf[(x, y)].set_fg(color);
241                    if !self.transparent_background
242                        && let Some(bg) = bg_color
243                    {
244                        buf[(x, y)].set_bg(bg);
245                    }
246                }
247            }
248        }
249    }
250}

252/// Render an injuries popup for viewing another player's injuries
253/// Returns the bounding rect of the popup for click detection
254pub fn render_injuries_popup(
255    popup: &crate::vellum::support::InjuriesPopupState,
256    screen: Rect,
257    buf: &mut Buffer,
258    theme: &crate::vellum::support::AppTheme,
259) -> Rect {
260    // Popup dimensions: injury doll needs 8 cols, 6 rows content + border + title
261    // Layout: title bar (1 row) + injury doll (6 rows) + close hint (1 row)
262    let popup_width: u16 = 22; // 8 cols content + 2 border + padding
263    let popup_height: u16 = 10; // 1 title + 6 content + 1 hint + 2 border
264
265    // Center the popup on screen
266    let popup_x = screen.x + (screen.width.saturating_sub(popup_width)) / 2;
267    let popup_y = screen.y + (screen.height.saturating_sub(popup_height)) / 2;
268    let popup_area = Rect::new(popup_x, popup_y, popup_width, popup_height);
269
270    // Clear the popup area
271    Clear.render(popup_area, buf);
272
273    // Draw border and background - convert theme colors to ratatui colors
274    let bg_color = support::to_ratatui_color(theme.window_background);
275    let border_color = support::to_ratatui_color(theme.window_border);
276    let title_color = support::to_ratatui_color(theme.window_title);
277
278    // Fill background
279    for row in popup_area.y..popup_area.y + popup_area.height {
280        for col in popup_area.x..popup_area.x + popup_area.width {
281            if col < buf.area().width && row < buf.area().height {
282                buf[(col, row)].set_bg(bg_color);
283            }
284        }
285    }
286
287    // Draw border
288    let block = Block::default()
289        .borders(ratatui::widgets::Borders::ALL)
290        .border_type(BorderType::Rounded)
291        .border_style(Style::default().fg(border_color))
292        .title(format!(" {}'s Injuries ", popup.player_name))
293        .title_style(
294            Style::default()
295                .fg(title_color)
296                .add_modifier(ratatui::style::Modifier::BOLD),
297        )
298        .style(Style::default().bg(bg_color));
299
300    let inner_area = block.inner(popup_area);
301    block.render(popup_area, buf);
302
303    // Render the injury doll in the inner area
304    let mut doll = InjuryDoll::new("");
305    doll.set_background_color(Some(format!(
306        "#{:02x}{:02x}{:02x}",
307        bg_color.to_string().as_bytes()[0],
308        0,
309        0
310    )));
311
312    // Copy injuries from popup state to doll
313    for (body_part, level) in &popup.injuries {
314        doll.set_injury(body_part.clone(), *level);
315    }
316
317    // Render doll in upper portion, leaving room for close hint
318    let doll_area = Rect::new(
319        inner_area.x,
320        inner_area.y,
321        inner_area.width,
322        inner_area.height.saturating_sub(1),
323    );
324
325    // Center the injury doll content
326    doll.set_content_align(Some("center".to_string()));
327    doll.set_transparent_background(false);
328    if let Some((r, g, b)) = color_to_rgb(bg_color) {
329        doll.set_background_color(Some(format!("#{r:02x}{g:02x}{b:02x}")));
330    }
331    doll.render(doll_area, buf);
332
333    // Render close hint at bottom
334    let hint_y = inner_area.y + inner_area.height.saturating_sub(1);
335    let hint_text = "[Esc to close]";
336    let hint_x = inner_area.x
337        + (inner_area
338            .width
339            .saturating_sub(u16::try_from(hint_text.len()).unwrap_or(u16::MAX)))
340            / 2;
341    let hint_style = Style::default().fg(Color::DarkGray);
342
343    for (i, ch) in hint_text.chars().enumerate() {
344        let x = hint_x + u16::try_from(i).unwrap_or(u16::MAX);
345        if x < buf.area().width && hint_y < buf.area().height {
346            buf[(x, hint_y)].set_char(ch);
347            buf[(x, hint_y)].set_style(hint_style);
348        }
349    }
350
351    popup_area
352}

354/// Convert ratatui Color to RGB tuple if possible
355fn color_to_rgb(color: Color) -> Option<(u8, u8, u8)> {
356    match color {
357        Color::Rgb(r, g, b) => Some((r, g, b)),
358        Color::Black => Some((0, 0, 0)),
359        Color::White => Some((255, 255, 255)),
360        Color::Red => Some((255, 0, 0)),
361        Color::Green => Some((0, 255, 0)),
362        Color::Blue => Some((0, 0, 255)),
363        Color::Yellow => Some((255, 255, 0)),
364        Color::Magenta => Some((255, 0, 255)),
365        Color::Cyan => Some((0, 255, 255)),
366        Color::Gray => Some((128, 128, 128)),
367        Color::DarkGray => Some((64, 64, 64)),
368        Color::LightRed => Some((255, 128, 128)),
369        Color::LightGreen => Some((128, 255, 128)),
370        Color::LightBlue => Some((128, 128, 255)),
371        Color::LightYellow => Some((255, 255, 128)),
372        Color::LightMagenta => Some((255, 128, 255)),
373        Color::LightCyan => Some((128, 255, 255)),
374        _ => None,
375    }
376}

378/// Each part of the figure: its column and row, its glyph, and the part of
379/// the body it shows.
380const PARTS: [(u16, u16, char, &str); 13] = [
381    (0, 0, '\u{f06e}', "leftEye"),
382    (4, 0, '\u{f06e}', "rightEye"),
383    (2, 1, '0', "head"),
384    (1, 2, '/', "leftArm"),
385    (2, 2, '|', "chest"),
386    (3, 2, '\\', "rightArm"),
387    (0, 3, 'o', "leftHand"),
388    (2, 3, '|', "abdomen"),
389    (4, 3, 'o', "rightHand"),
390    (1, 4, '/', "leftLeg"),
391    (3, 4, '\\', "rightLeg"),
392    (0, 5, 'o', "leftLeg"),
393    (4, 5, 'o', "rightLeg"),
394];
395
396/// The parts named in words at the figure's right: the neck, the back and
397/// the nerves, each at its column and row.
398const MARKS: [(u16, u16, &str, &str); 3] = [
399    (6, 1, "nk", "neck"),
400    (6, 3, "bk", "back"),
401    (6, 5, "ns", "nsys"),
402];
403
404/// The columns and rows the figure and its words take.
405fn doll_size() -> (u16, u16) {
406    let mut width = 0;
407    let mut height = 0;
408    for (col, row, _, _) in &PARTS {
409        width = width.max(col + 1);
410        height = height.max(row + 1);
411    }
412    for (col, row, text, _) in &MARKS {
413        width = width.max(col + u16::try_from(text.chars().count()).unwrap_or(u16::MAX));
414        height = height.max(row + 1);
415    }
416    (width, height)
417}

424    #[test]
425    fn content_stays_within_inner_area_with_left_bottom_borders() {
426        // Test case: 7 total rows, 10 total cols
427        // Borders: left + bottom (no top, no right)
428        // Expected: content should NOT overlap bottom border at row 6
429
430        let mut doll = InjuryDoll::new("Injuries");
431        doll.set_border_config(
432            true,
433            Some("single".to_string()),
434            Some("#ffffff".to_string()),
435        );
436        doll.set_border_sides(BorderSides {
437            left: true,
438            right: false,
439            top: false,
440            bottom: true,
441        });
442
443        let area = Rect::new(0, 0, 10, 7);
444        let mut buf = Buffer::empty(area);
445        doll.render(area, &mut buf);
446
447        // Check that the bottom border row (row 6) contains border character on left
448        // and the content (eyes at row 0) is rendered in the top row of inner area
449        assert_eq!(
450            buf[(0, 6)].symbol(),
451            "└",
452            "Bottom-left corner should have border character"
453        );
454
455        // Row 0 should have content (left eye at col 1 within inner area)
456        // Inner area starts at x=1 (after left border), so left eye at inner col 0 = buf col 1
457        let eye_char = buf[(1, 0)].symbol();
458        assert!(
459            eye_char != " ",
460            "Row 0 should have content (eye), got: '{eye_char}'"
461        );

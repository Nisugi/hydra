1//! Items list widget for showing room objects (non-creature items).
2//!
3//! Parses item data from room objs component (non-bold links) and displays
4//! them with clickable links for popup menus and drag-to-inventory.
5//! Uses `ListWidget` for proper text rendering with clickable links.
6
7use crate::vellum::support::LinkData;
8use ratatui::{buffer::Buffer, layout::Rect};
9
10/// What else is in the room, a line each, each a link.
11pub struct Items {
12    widget: super::list_widget::ListWidget,
13    count: u32,
14    base_title: String,
15    /// Generation counter for change detection
16    generation: u64,
17    /// Cached object IDs for change detection (comma-joined)
18    object_ids_cache: String,
19}
20
21impl Items {
22    /// A list titled `title`, empty.
23    #[must_use]
24    pub fn new(title: &str) -> Self {
25        Self {
26            widget: super::list_widget::ListWidget::new(title),
27            count: 0,
28            base_title: title.to_string(),
29            generation: 0,
30            object_ids_cache: String::new(),
31        }
32    }
33
34    /// Update the widget from room objects.
35    /// Returns true if the display changed.
36    pub fn update_from_state(
37        &mut self,
38        room_objects: &[crate::vellum::support::RoomObject],
39    ) -> bool {
40        // Build cache string for comparison
41        let new_object_ids: String = room_objects
42            .iter()
43            .map(|o| o.id.as_str())
44            .collect::<Vec<_>>()
45            .join(",");
46        let new_count = u32::try_from(room_objects.len()).unwrap_or(u32::MAX);
47
48        // Quick check: if cache matches, no change needed
49        if self.object_ids_cache == new_object_ids && self.count == new_count {
50            return false;
51        }
52
53        // Update cache
54        self.object_ids_cache = new_object_ids;
55
56        self.widget.clear();
57        self.count = 0;
58
59        for obj in room_objects {
60            // Build LinkData for clickable interaction
61            // - exist_id: ID (e.g., "123456789")
62            // - noun: Use parsed noun or fallback to last word
63            // - text: full object name
64            let link_noun = obj
65                .noun
66                .clone()
67                .or_else(|| {
68                    obj.name
69                        .split_whitespace()
70                        .last()
71                        .map(std::string::ToString::to_string)
72                })
73                .unwrap_or_else(|| obj.name.clone());
74
75            let link_data = Some(LinkData {
76                exist_id: obj.id.clone(),
77                noun: link_noun,
78                text: obj.name.clone(),
79                coord: None,
80            });
81
82            // Add to widget with link data for click handling
83            self.widget
84                .add_simple_line(obj.name.clone(), None, link_data);
85
86            self.count += 1;
87        }
88
89        self.generation += 1;
90        self.update_title();
91        true
92    }
93
94    fn update_title(&mut self) {
95        if self.base_title.is_empty() {
96            self.widget.set_title(String::new());
97        } else {
98            let title = format!("{} [{:02}]", self.base_title, self.count);
99            self.widget.set_title(title);
100        }
101    }
102
103    /// Its title.
104    pub fn set_title(&mut self, title: &str) {
105        self.base_title = title.to_string();
106        self.update_title();
107    }

194#[cfg(test)]
195mod tests {
196    use super::*;
197    use crate::vellum::support::RoomObject;
198
199    #[test]
200    fn test_new_defaults() {
201        let items = Items::new("Items");
202        assert_eq!(items.base_title, "Items");
203        assert_eq!(items.count, 0);
204        assert_eq!(items.generation, 0);
205    }
206
207    #[test]
208    fn test_update_from_state_empty() {
209        let mut items = Items::new("Items");
210        let objects: Vec<RoomObject> = vec![];
211
212        // First update with empty state matches initial widget state, so no change
213        let changed = items.update_from_state(&objects);
214        assert!(!changed);
215        assert_eq!(items.count, 0);
216    }
217
218    #[test]
219    fn test_update_from_state_with_objects() {
220        let mut items = Items::new("Items");
221        let objects = vec![
222            RoomObject {
223                id: "123".to_string(),
224                name: "a silver ring".to_string(),
225                noun: Some("ring".to_string()),
226            },
227            RoomObject {
228                id: "456".to_string(),
229                name: "some gold coins".to_string(),
230                noun: Some("coins".to_string()),
231            },
232        ];
233
234        let changed = items.update_from_state(&objects);
235        assert!(changed);
236        assert_eq!(items.count, 2);
237    }
238
239    #[test]
240    fn test_update_from_state_no_change() {
241        let mut items = Items::new("Items");
242        let objects = vec![RoomObject {
243            id: "123".to_string(),
244            name: "a silver ring".to_string(),
245            noun: Some("ring".to_string()),
246        }];
247
248        items.update_from_state(&objects);
249
250        // Same state again

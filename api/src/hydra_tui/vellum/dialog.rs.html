13/// Where one of a dialog's rows sits, for finding what a click is on: one
14/// for each row the dialog draws.
15pub enum DialogRowArea {
16    /// A label, which a click does nothing on.
17    DisplayLabel {
18        /// Its row.
19        area: Rect,
20    },
21    /// A bar, which a click does nothing on.
22    ProgressBar {
23        /// Its row.
24        area: Rect,
25    },
26    /// A field and the button that sends it, on one row.
27    FieldWithButton {
28        /// The field's input box.
29        field_area: Rect,
30        /// The button.
31        button_area: Rect,
32        /// The field, by its place in `fields`.
33        field_idx: usize,
34        /// The button, by its place in `buttons`.
35        button_idx: usize,
36    },
37    /// A field with no button of its own.
38    FieldOnly {
39        /// Its input box.
40        field_area: Rect,
41        /// The field, by its place in `fields`.
42        field_idx: usize,
43    },
44    /// Several links on one row.
45    LinkGroup {
46        /// Each link's place, and the link by its place in `buttons`.
47        button_areas: Vec<(Rect, usize)>,
48    },
49    /// A button alone on its row, centred.
50    SingleButton {
51        /// Its place.
52        area: Rect,
53        /// The button, by its place in `buttons`.
54        button_idx: usize,
55    },
56    /// A row of controls the game placed.
57    ControlBand {
58        /// Each control's place, and which control it is.
59        cells: Vec<(Rect, BandHit)>,
60    },
61}

91/// One band's cells as a line's spans: a button in the picked style when it
92/// is the one picked, everything else in the normal style.
93fn band_spans(
94    band: &[BandCell],
95    dialog: &DialogState,
96    (normal, selected): (Style, Style),
97) -> Vec<Span<'static>> {
98    let mut spans = vec![Span::raw(" ")];
99    for (i, cell) in band.iter().enumerate() {
100        if i > 0 {
101            spans.push(Span::raw("  "));
102        }
103        match cell {
104            BandCell::Button { index, text } => {
105                let style = if *index == dialog.selected {
106                    selected
107                } else {
108                    normal
109                };
110                spans.push(Span::styled(text.clone(), style));
111            }
112            BandCell::DropDown { text, .. } | BandCell::Label { text } => {
113                spans.push(Span::styled(text.clone(), normal));
114            }
115            BandCell::Progress { index } => {
116                if let Some(pb) = dialog.progress_bars.get(*index) {
117                    spans.push(Span::styled(pb.text.clone(), normal));
118                }
119            }
120        }
121    }
122    spans
123}

125/// Group the anchor-grid resolution into horizontal bands for the TUI:
126/// controls sorted by resolved y then x, with a new band whenever the
127/// vertical gap exceeds half a control height. Pixel positions become
128/// row order + left-to-right cell order — the TUI keeps the structure,
129/// not the pixels. None when the dialog carries no layout data.
130fn positioned_band_cells(dialog: &DialogState) -> Option<Vec<Vec<BandCell>>> {
131    let (mut controls, _) = dialog.positioned_controls()?;
132    controls.sort_by(|a, b| {
133        (a.rect.1, a.rect.0)
134            .partial_cmp(&(b.rect.1, b.rect.0))
135            .unwrap_or(std::cmp::Ordering::Equal)
136    });
137
138    let mut bands: Vec<Vec<BandCell>> = Vec::new();
139    let mut band_y = f32::MIN;
140    for control in controls {
141        use crate::vellum::data::PositionedControlKind;
142        let cell = match control.kind {
143            PositionedControlKind::Button(index) => {
144                dialog.buttons.get(index).map(|b| BandCell::Button {
145                    index,
146                    text: display_label(b),
147                })
148            }
149            PositionedControlKind::DropDown(index) => dialog.dropdowns.get(index).map(|d| {
150                let shown = d
151                    .options
152                    .iter()
153                    .find(|(_, value)| *value == d.value)
154                    .map_or(d.value.as_str(), |(text, _)| text.as_str());
155                BandCell::DropDown {
156                    index,
157                    text: format!("[{shown} ▼]"),
158                }
159            }),
160            PositionedControlKind::ProgressBar(index) => dialog
161                .progress_bars
162                .get(index)
163                .map(|_| BandCell::Progress { index }),
164            PositionedControlKind::Label(index) => {
165                dialog.display_labels.get(index).map(|l| BandCell::Label {
166                    text: l.value.clone(),
167                })
168            }
169            // Links and spinners are shown, read only, in the banded view.
170            PositionedControlKind::Link(index) => {
171                dialog.links.get(index).map(|l| BandCell::Label {
172                    text: format!("[{}]", l.label),
173                })
174            }
175            PositionedControlKind::SpinBox(index) => {
176                dialog.spinboxes.get(index).map(|s| BandCell::Label {
177                    text: format!("<{}>", s.value),
178                })
179            }
180            // Skins + anchor-only images have no TUI representation.
181            PositionedControlKind::Skin(_) | PositionedControlKind::Image(_) => None,
182        };
183        let Some(cell) = cell else { continue };
184        match bands.last_mut() {
185            Some(band) if (control.rect.1 - band_y).abs() <= 12.0 => band.push(cell),
186            _ => {
187                bands.push(vec![cell]);
188                band_y = control.rect.1;
189            }
190        }
191    }
192    (!bands.is_empty()).then_some(bands)
193}

195fn band_cell_text_len(cell: &BandCell, dialog: &DialogState) -> usize {
196    match cell {
197        BandCell::Button { text, .. }
198        | BandCell::DropDown { text, .. }
199        | BandCell::Label { text } => text.chars().count(),
200        BandCell::Progress { index } => dialog
201            .progress_bars
202            .get(*index)
203            .map_or(20, |pb| pb.text.chars().count().max(20)),
204    }
205}

207fn band_width(band: &[BandCell], dialog: &DialogState) -> usize {
208    let cells: usize = band.iter().map(|c| band_cell_text_len(c, dialog)).sum();
209    cells + band.len().saturating_sub(1) * 2
210}

212fn display_label(button: &crate::vellum::data::DialogButton) -> String {
213    if button.is_radio {
214        let marker = if button.selected { "x" } else { " " };
215        format!("[{}] {}", marker, button.label)
216    } else {
217        button.label.clone()
218    }

221/// Format a number with commas for display (e.g., 1000 → 1,000)
222fn format_number_with_commas(value: &str) -> String {
223    // Only format if it's a pure number (digits only)
224    if value.is_empty() || !value.chars().all(|c| c.is_ascii_digit()) {
225        return value.to_string();
226    }
227
228    let mut result = String::new();
229    let chars: Vec<char> = value.chars().collect();
230    let len = chars.len();
231
232    for (i, ch) in chars.iter().enumerate() {
233        result.push(*ch);
234        let remaining = len - i - 1;
235        if remaining > 0 && remaining.is_multiple_of(3) {
236            result.push(',');
237        }
238    }
239    result
240}

242/// Get formatted value for a dialog field (with commas, no cursor)
243fn field_formatted_value(dialog: &DialogState, index: usize) -> String {
244    let field = &dialog.fields[index];

248/// Create a fixed-width input box display: "[     value]"
249/// The `box_width` is the total width including brackets
250const FIELD_BOX_WIDTH: usize = 22; // 20 chars + 2 brackets
251
252fn field_input_box(value: &str) -> String {
253    let content_width = FIELD_BOX_WIDTH - 2; // Space for content inside brackets
254    if value.len() >= content_width {
255        format!("[{}]", &value[value.len().saturating_sub(content_width)..])
256    } else {
257        // Right-align value in the box
258        format!("[{value:>content_width$}]")
259    }
260}

262/// Strip common suffixes from field IDs for button matching
263/// e.g., "depositSB" → "deposit", "withdrawAmount" → "withdraw"
264fn strip_field_suffix(field_id: &str) -> String {
265    // Common suffixes in dialog field IDs (case-insensitive, already lowercase input)
266    const SUFFIXES: &[&str] = &[
267        "sb",     // SpinBox
268        "amount", // Amount field
269        "input",  // Input field
270        "field",  // Generic field
271        "value",  // Value field
272        "text",   // Text field
273        "edit",   // Edit field
274        "box",    // Box suffix
275        "entry",  // Entry field
276    ];
277
278    let mut result = field_id.to_string();
279    for suffix in SUFFIXES {
280        if result.len() > suffix.len() && result.ends_with(suffix) {
281            result.truncate(result.len() - suffix.len());
282            break; // Only strip one suffix
283        }
284    }
285    result
286}

288/// The rows the dialog takes: its labels, bands and fields when its
289/// controls are placed, else the rows `build_dialog_rows` makes.
290fn count_dialog_rows(dialog: &DialogState) -> usize {
291    if let Some(bands) = positioned_band_cells(dialog) {
292        return (dialog.display_labels.len() + bands.len() + dialog.fields.len()).max(1);
293    }
294    build_dialog_rows(dialog).len().max(1)
295}

297/// The button each field is paired with, by its place in `buttons`: the
298/// one its `enter_button` names, else the first unpaired button with a
299/// command whose id its own id names (`depositSB` and `deposit`).
300fn field_buttons(dialog: &DialogState) -> Vec<Option<usize>> {
301    let mut used: std::collections::HashSet<usize> = std::collections::HashSet::new();
302    let mut pairs = Vec::with_capacity(dialog.fields.len());
303    for field in &dialog.fields {
304        let named = field
305            .enter_button
306            .as_ref()
307            .and_then(|wanted| dialog.buttons.iter().position(|b| &b.id == wanted));
308        let paired = named.or_else(|| {
309            let field_id = field.id.to_lowercase();
310            let base = strip_field_suffix(&field_id);
311            dialog.buttons.iter().enumerate().position(|(at, button)| {
312                let button_id = button.id.to_lowercase();
313                !used.contains(&at)
314                    && !button.is_close
315                    && !button.command.is_empty()
316                    && (base == button_id
317                        || field_id.starts_with(&button_id)
318                        || button_id.starts_with(&base))
319            })
320        });
321        if let Some(at) = paired {
322            used.insert(at);
323        }
324        pairs.push(paired);
325    }
326    pairs
327}

329/// The columns the dialog's content takes: its widest row, its title, and
330/// never under 20.
331fn content_width(dialog: &DialogState, title: &str, bands: Option<&[Vec<BandCell>]>) -> usize {
332    let widest = |lens: &mut dyn Iterator<Item = usize>| lens.max().unwrap_or(0);
333    let labels = widest(&mut dialog.display_labels.iter().map(|label| label.value.len()));
334    if let Some(bands) = bands {
335        let band = widest(&mut bands.iter().map(|band| band_width(band, dialog)));
336        return band.max(labels).max(title.len()).max(20);
337    }
338    let buttons = dialog
339        .buttons
340        .iter()
341        .map(|button| display_label(button).len())
342        .max()
343        .unwrap_or(4);
344    // A field's input box is 22 columns, then two before its button.
345    let fields = widest(
346        &mut field_buttons(dialog)
347            .into_iter()
348            .flatten()
349            .map(|at| 22 + 2 + display_label(&dialog.buttons[at]).len()),
350    );
351    let bars = widest(&mut dialog.progress_bars.iter().map(|bar| bar.text.len()));
352    let links: usize = dialog
353        .buttons
354        .iter()
355        .filter(|b| !b.is_close && !b.command.is_empty() && !b.is_radio)
356        .map(|b| display_label(b).len() + 2)
357        .sum();
358    buttons
359        .max(fields)
360        .max(bars)
361        .max(labels)
362        .max(links)
363        .max(title.len())
364        .max(20)
365}

367/// Where the dialog sits on `screen`: where it was put and its size as it
368/// was left, else its content's size, centred.
369fn dialog_area(screen: Rect, dialog: &DialogState, content_width: usize) -> Rect {
370    let (width, height) = dialog.size.unwrap_or_else(|| {
371        let mut width = u16::try_from(content_width + 4).unwrap_or(u16::MAX);
372        if screen.width > 0 {
373            width = width.min(screen.width);
374        }
375        let mut height = u16::try_from(count_dialog_rows(dialog) + 2).unwrap_or(u16::MAX);
376        if screen.height > 0 {
377            height = height.min(screen.height);
378        }
379        (width, height)
380    });
381    let (x, y) = dialog.position.unwrap_or_else(|| {
382        (
383            screen
384                .x
385                .saturating_add(screen.width.saturating_sub(width) / 2),
386            screen
387                .y
388                .saturating_add(screen.height.saturating_sub(height) / 2),
389        )
390    });
391    Rect {
392        x,
393        y,
394        width,
395        height,
396    }
397}

399/// Where the dialog and each of its rows sit on `screen`.
400#[must_use]
401pub fn compute_dialog_layout(screen: Rect, dialog: &DialogState) -> DialogLayout {
402    let title = dialog.title.as_deref().unwrap_or("Dialog");
403    let bands = positioned_band_cells(dialog);
404    let area = dialog_area(
405        screen,
406        dialog,
407        content_width(dialog, title, bands.as_deref()),
408    );
409    let title_bar_area = Rect { height: 1, ..area };
410    let inner = (area.x.saturating_add(1), area.width.saturating_sub(2));
411    let rows = (
412        area.y.saturating_add(1),
413        area.y.saturating_add(area.height.saturating_sub(1)),
414    );
415    let row_areas = match &bands {
416        Some(bands) => banded_row_areas(dialog, bands, inner, rows),
417        None => plain_row_areas(dialog, inner, rows),
418    };
419    DialogLayout {
420        area,
421        title_bar_area,
422        row_areas,
423    }
424}

437/// The rows of a dialog whose controls are placed: its labels, its bands,
438/// then its fields, as many as fit between `top` and `bottom`.
439fn banded_row_areas(
440    dialog: &DialogState,
441    bands: &[Vec<BandCell>],
442    (inner_x, inner_width): (u16, u16),
443    (top, bottom): (u16, u16),
444) -> Vec<DialogRowArea> {
445    let mut areas = Vec::new();
446    let mut y = top;
447    let row = |y: u16| Rect {
448        x: inner_x,
449        y,
450        width: inner_width,
451        height: 1,
452    };
453    for _ in &dialog.display_labels {
454        if y < bottom {
455            areas.push(DialogRowArea::DisplayLabel { area: row(y) });
456            y += 1;
457        }
458    }
459    for band in bands {
460        if y >= bottom {
461            break;
462        }
463        let mut cell_x = inner_x.saturating_add(1);
464        let mut cells = Vec::new();
465        for cell in band {
466            let width = u16::try_from(band_cell_text_len(cell, dialog)).unwrap_or(u16::MAX);
467            let area = Rect {
468                x: cell_x,
469                y,
470                width,
471                height: 1,
472            };
473            match cell {
474                BandCell::Button { index, .. } => cells.push((area, BandHit::Button(*index))),
475                BandCell::DropDown { index, .. } => cells.push((area, BandHit::DropDown(*index))),
476                // Bars and labels are not clicked.
477                BandCell::Progress { .. } | BandCell::Label { .. } => {}
478            }
479            cell_x = cell_x.saturating_add(width + 2);
480        }
481        areas.push(DialogRowArea::ControlBand { cells });
482        y += 1;
483    }
484    for field_idx in 0..dialog.fields.len() {
485        if y >= bottom {
486            break;
487        }
488        areas.push(DialogRowArea::FieldOnly {
489            field_area: field_area(dialog, field_idx, inner_x, y),
490            field_idx,
491        });
492        y += 1;
493    }
494    areas
495}

497/// The rows of a dialog laid out in rows, as `build_dialog_rows` makes
498/// them, as many as fit between `top` and `bottom`.
499fn plain_row_areas(
500    dialog: &DialogState,
501    (inner_x, inner_width): (u16, u16),
502    (top, bottom): (u16, u16),
503) -> Vec<DialogRowArea> {
504    let row = |y: u16| Rect {
505        x: inner_x,
506        y,
507        width: inner_width,
508        height: 1,
509    };
510    build_dialog_rows(dialog)
511        .into_iter()
512        .zip(top..bottom)
513        .map(|(shown, y)| match shown {
514            DialogRow::DisplayLabel(_) => DialogRowArea::DisplayLabel { area: row(y) },
515            DialogRow::ProgressBar(_) => DialogRowArea::ProgressBar { area: row(y) },
516            DialogRow::FieldWithButton {
517                field_idx,
518                button_idx,
519            } => {
520                let field_area = field_area(dialog, field_idx, inner_x, y);
521                let label = display_label(&dialog.buttons[button_idx]);
522                DialogRowArea::FieldWithButton {
523                    field_area,
524                    button_area: Rect {
525                        x: field_area.x + field_area.width + 2,
526                        y,
527                        width: u16::try_from(label.len()).unwrap_or(u16::MAX),
528                        height: 1,
529                    },
530                    field_idx,
531                    button_idx,
532                }
533            }
534            DialogRow::FieldOnly(field_idx) => DialogRowArea::FieldOnly {
535                field_area: field_area(dialog, field_idx, inner_x, y),
536                field_idx,
537            },
538            DialogRow::LinkGroup(buttons) => DialogRowArea::LinkGroup {
539                button_areas: compute_link_group_areas(dialog, &buttons, inner_x, y, inner_width),
540            },
541            DialogRow::SingleButton(button_idx) => {
542                let width = u16::try_from(display_label(&dialog.buttons[button_idx]).len())
543                    .unwrap_or(u16::MAX);
544                DialogRowArea::SingleButton {
545                    area: Rect {
546                        x: inner_x + inner_width.saturating_sub(width) / 2,
547                        y,
548                        width,
549                        height: 1,
550                    },
551                    button_idx,
552                }
553            }
554        })
555        .collect()
556}

558/// Compute hit-test areas for a group of link buttons on the same row
559fn compute_link_group_areas(
560    dialog: &DialogState,
561    button_indices: &[usize],
562    inner_x: u16,
563    row_y: u16,
564    inner_width: u16,
565) -> Vec<(Rect, usize)> {
566    let mut areas = Vec::new();
567    let num_buttons = button_indices.len();
568    if num_buttons == 0 {
569        return areas;
570    }
571
572    let total_button_len: u16 = button_indices
573        .iter()
574        .map(|idx| u16::try_from(display_label(&dialog.buttons[*idx]).len()).unwrap_or(u16::MAX))
575        .sum();
576
577    let spacing = if num_buttons > 1 {
578        inner_width.saturating_sub(total_button_len)
579            / (u16::try_from(num_buttons).unwrap_or(u16::MAX) + 1)
580    } else {
581        inner_width.saturating_sub(total_button_len) / 2
582    };
583
584    let mut current_x = inner_x + spacing;
585
586    for &btn_idx in button_indices {
587        let button_text = display_label(&dialog.buttons[btn_idx]);
588        let button_len = u16::try_from(button_text.len()).unwrap_or(u16::MAX);
589
590        areas.push((
591            Rect {
592                x: current_x,
593                y: row_y,
594                width: button_len,
595                height: 1,
596            },
597            btn_idx,
598        ));
599
600        current_x += button_len + spacing.max(2);
601    }
602
603    areas
604}

606/// The button at `x`, `y`, by its place in `buttons`.
607#[must_use]
608pub fn hit_test_button(layout: &DialogLayout, x: u16, y: u16) -> Option<usize> {
609    for row_area in &layout.row_areas {
610        match row_area {
611            DialogRowArea::FieldWithButton {
612                button_area,
613                button_idx,
614                ..
615            } => {
616                if x >= button_area.x
617                    && x < button_area.x.saturating_add(button_area.width)
618                    && y >= button_area.y
619                    && y < button_area.y.saturating_add(button_area.height)
620                {
621                    return Some(*button_idx);
622                }
623            }
624            DialogRowArea::LinkGroup { button_areas } => {
625                for (area, btn_idx) in button_areas {
626                    if x >= area.x
627                        && x < area.x.saturating_add(area.width)
628                        && y >= area.y
629                        && y < area.y.saturating_add(area.height)
630                    {
631                        return Some(*btn_idx);
632                    }
633                }
634            }
635            DialogRowArea::SingleButton { area, button_idx } => {
636                if x >= area.x
637                    && x < area.x.saturating_add(area.width)
638                    && y >= area.y
639                    && y < area.y.saturating_add(area.height)
640                {
641                    return Some(*button_idx);
642                }
643            }
644            DialogRowArea::ControlBand { cells } => {
645                for (area, hit) in cells {
646                    if let BandHit::Button(button_idx) = hit
647                        && x >= area.x
648                        && x < area.x.saturating_add(area.width)
649                        && y >= area.y
650                        && y < area.y.saturating_add(area.height)
651                    {
652                        return Some(*button_idx);
653                    }
654                }
655            }
656            _ => {}
657        }
658    }
659    None
660}

662/// Hit-test positioned control bands for a dropdown cell.
663#[must_use]
664pub fn hit_test_dropdown(layout: &DialogLayout, x: u16, y: u16) -> Option<usize> {
665    for row_area in &layout.row_areas {
666        if let DialogRowArea::ControlBand { cells } = row_area {
667            for (area, hit) in cells {
668                if let BandHit::DropDown(index) = hit
669                    && x >= area.x
670                    && x < area.x.saturating_add(area.width)
671                    && y >= area.y
672                    && y < area.y.saturating_add(area.height)
673                {
674                    return Some(*index);
675                }
676            }
677        }
678    }
679    None
680}

723/// Test if a mouse position is on a resize handle.
724/// Returns the resize operation type if on a handle, None otherwise.
725/// Resize handles are the edges and corners of the dialog.
726#[must_use]
727pub fn hit_test_resize_handle(
728    layout: &DialogLayout,
729    x: u16,
730    y: u16,
731) -> Option<DialogDragOperation> {
732    let area = &layout.area;
733    let left = area.x;
734    let right = area.x.saturating_add(area.width.saturating_sub(1));
735    let top = area.y;
736    let bottom = area.y.saturating_add(area.height.saturating_sub(1));
737
738    // Corner detection (2 cells from corner)
739    let corner_size = 2u16;
740    let is_near_left = x <= left.saturating_add(corner_size);
741    let is_near_right = x >= right.saturating_sub(corner_size);
742    let is_on_left = x == left;
743    let is_on_right = x == right;
744    let is_on_top = y == top;
745    let is_on_bottom = y == bottom;
746
747    // Corners take priority
748    if is_on_top && is_near_left && is_on_left {
749        return Some(DialogDragOperation::ResizeTopLeft);
750    }
751    if is_on_top && is_near_right && is_on_right {
752        return Some(DialogDragOperation::ResizeTopRight);
753    }
754    if is_on_bottom && is_near_left && is_on_left {
755        return Some(DialogDragOperation::ResizeBottomLeft);
756    }
757    if is_on_bottom && is_near_right && is_on_right {
758        return Some(DialogDragOperation::ResizeBottomRight);
759    }
760
761    // Edges
762    if is_on_left && y > top && y < bottom {
763        return Some(DialogDragOperation::ResizeLeft);
764    }
765    if is_on_right && y > top && y < bottom {
766        return Some(DialogDragOperation::ResizeRight);
767    }
768    if is_on_bottom && x > left && x < right {
769        return Some(DialogDragOperation::ResizeBottom);
770    }
771    // Note: top edge is title bar for move, not resize
772
773    None
774}

776/// Represents a row in the dialog layout
777enum DialogRow<'a> {
778    /// Centered display label (e.g., "Balance: 144573897")
779    DisplayLabel(&'a crate::vellum::data::DialogLabel),
780    /// Progress bar
781    ProgressBar(&'a crate::vellum::data::DialogProgressBar),
782    /// Field with its paired button on the same row
783    FieldWithButton { field_idx: usize, button_idx: usize },
784    /// Field without a paired button (rendered alone)
785    FieldOnly(usize),
786    /// Multiple links grouped on the same row
787    LinkGroup(Vec<usize>), // button indices
788    /// Single button (e.g., Close) - centered
789    SingleButton(usize),
790}
791
792/// Build dialog rows from dialog state
793fn build_dialog_rows(dialog: &DialogState) -> Vec<DialogRow<'_>> {
794    let mut rows = Vec::new();
795
796    // 1. Display labels first (centered)
797    for label in &dialog.display_labels {
798        rows.push(DialogRow::DisplayLabel(label));
799    }
800
801    // 2. Progress bars
802    for pb in &dialog.progress_bars {
803        rows.push(DialogRow::ProgressBar(pb));
804    }
805
806    // 3. Each field, with the button it is paired with when it is.
807    let pairs = field_buttons(dialog);
808    let used_buttons: std::collections::HashSet<usize> = pairs.iter().flatten().copied().collect();
809    for (field_idx, paired) in pairs.into_iter().enumerate() {
810        rows.push(match paired {
811            Some(button_idx) => DialogRow::FieldWithButton {
812                field_idx,
813                button_idx,
814            },
815            None => DialogRow::FieldOnly(field_idx),
816        });
817    }
818
819    // 4. Group remaining buttons
820    // - Links (non-close buttons with commands) get grouped together
821    // - Close buttons are rendered individually
822    let mut link_group: Vec<usize> = Vec::new();
823
824    for (idx, button) in dialog.buttons.iter().enumerate() {
825        if used_buttons.contains(&idx) {
826            continue;
827        }
828
829        // Check if this is a "link" type button (has command, not a close button)
830        let is_link = !button.is_close && !button.command.is_empty() && !button.is_radio;
831
832        if is_link {
833            link_group.push(idx);
834        } else {
835            // Flush any pending link group
836            if !link_group.is_empty() {
837                rows.push(DialogRow::LinkGroup(std::mem::take(&mut link_group)));
838            }
839            // Add this button as a single button
840            rows.push(DialogRow::SingleButton(idx));
841        }
842    }
843
844    // Flush remaining link group
845    if !link_group.is_empty() {
846        rows.push(DialogRow::LinkGroup(link_group));
847    }
848
849    rows
850}

852/// The lines of a dialog whose controls are placed: its labels centred,
853/// its bands, a lone bar drawn whole, then its fields.
854fn banded_lines(
855    dialog: &DialogState,
856    bands: &[Vec<BandCell>],
857    inner_width: usize,
858    (normal, selected, theme): (Style, Style, &crate::vellum::support::AppTheme),
859) -> Vec<Line<'static>> {
860    let mut lines: Vec<Line> = Vec::new();
861    for label in &dialog.display_labels {
862        let padding = inner_width.saturating_sub(label.value.len()) / 2;
863        lines.push(Line::from(vec![
864            Span::raw(" "),
865            Span::styled(format!("{:>padding$}{}", "", label.value), normal),
866        ]));
867    }
868    for band in bands {
869        if let [BandCell::Progress { index }] = band.as_slice()
870            && let Some(pb) = dialog.progress_bars.get(*index)
871        {
872            lines.push(render_progress_bar_line(pb, inner_width, theme));
873            continue;
874        }
875        lines.push(Line::from(band_spans(band, dialog, (normal, selected))));
876    }
877    for field_idx in 0..dialog.fields.len() {
878        let style = if dialog.focused_field == Some(field_idx) {
879            selected
880        } else {
881            normal
882        };
883        let value = field_formatted_value(dialog, field_idx);
884        lines.push(Line::from(vec![
885            Span::raw("  "),
886            Span::styled(field_input_box(&value), style),
887        ]));
888    }
889    lines
890}

892/// The links of a row of links, spaced evenly across it.
893fn link_group_line(
894    dialog: &DialogState,
895    buttons: &[usize],
896    inner_width: usize,
897    (normal, selected): (Style, Style),
898) -> Line<'static> {
899    let total: usize = buttons
900        .iter()
901        .map(|at| display_label(&dialog.buttons[*at]).len())
902        .sum();
903    let count = buttons.len();
904    let spacing = if count > 1 {
905        inner_width.saturating_sub(total) / (count + 1)
906    } else {
907        inner_width.saturating_sub(total) / 2
908    };
909    let mut spans = vec![Span::raw(" ")];
910    for (i, &at) in buttons.iter().enumerate() {
911        if i == 0 || count == 1 {
912            spans.push(Span::raw(" ".repeat(spacing)));
913        }
914        let style = if at == dialog.selected {
915            selected
916        } else {
917            normal
918        };
919        spans.push(Span::styled(display_label(&dialog.buttons[at]), style));
920        if i < count - 1 {
921            spans.push(Span::raw(" ".repeat(spacing.max(2))));
922        }
923    }
924    spans.push(Span::raw(" "));
925    Line::from(spans)
926}

928/// One of a plain dialog's rows as its line: a label centred, a bar, a
929/// field with its button padded to `button_width` so the rows line up, a
930/// row of links, or a button centred.
931fn row_line(
932    dialog: &DialogState,
933    row: &DialogRow<'_>,
934    (inner_width, button_width): (usize, usize),
935    (normal, selected, theme): (Style, Style, &crate::vellum::support::AppTheme),
936) -> Line<'static> {
937    let pick = |picked: bool| if picked { selected } else { normal };
938    let field = |field_idx: usize| {
939        let value = field_formatted_value(dialog, field_idx);
940        Span::styled(
941            field_input_box(&value),
942            pick(dialog.focused_field == Some(field_idx)),
943        )
944    };
945    match row {
946        DialogRow::DisplayLabel(label) => {
947            let padding = inner_width.saturating_sub(label.value.len()) / 2;
948            Line::from(vec![
949                Span::raw(" "),
950                Span::styled(format!("{:>padding$}{}", "", label.value), normal),
951                Span::raw(" "),
952            ])
953        }
954        DialogRow::ProgressBar(pb) => render_progress_bar_line(pb, inner_width, theme),
955        DialogRow::FieldWithButton {
956            field_idx,
957            button_idx,
958        } => {
959            let label = display_label(&dialog.buttons[*button_idx]);
960            Line::from(vec![
961                Span::raw("  "),
962                field(*field_idx),
963                Span::raw("  "),
964                Span::styled(
965                    format!("{label:<button_width$}"),
966                    pick(*button_idx == dialog.selected),
967                ),
968            ])
969        }
970        DialogRow::LinkGroup(buttons) => {
971            link_group_line(dialog, buttons, inner_width, (normal, selected))
972        }
973        DialogRow::SingleButton(button_idx) => {
974            let label = display_label(&dialog.buttons[*button_idx]);
975            let padding = inner_width.saturating_sub(label.len()) / 2;
976            Line::from(vec![
977                Span::raw(" "),
978                Span::raw(" ".repeat(padding)),
979                Span::styled(label, pick(*button_idx == dialog.selected)),
980                Span::raw(" "),
981            ])
982        }
983        DialogRow::FieldOnly(field_idx) => Line::from(vec![Span::raw("  "), field(*field_idx)]),
984    }
985}

987/// The dialog drawn over `screen` as a box of its own, where its layout
988/// puts it.
989pub fn render_dialog(
990    dialog: &DialogState,
991    screen: Rect,
992    buf: &mut Buffer,
993    theme: &crate::vellum::support::AppTheme,
994) {
995    let layout = compute_dialog_layout(screen, dialog);
996    let title = dialog.title.as_deref().unwrap_or("Dialog");
997    Clear.render(layout.area, buf);
998    let normal = Style::default()
999        .fg(support::to_ratatui_color(theme.menu_item_normal))
1000        .bg(support::to_ratatui_color(theme.menu_background));
1001    let selected = Style::default()
1002        .fg(support::to_ratatui_color(theme.menu_item_selected))
1003        .bg(support::to_ratatui_color(theme.menu_item_focused));
1004    // Borders and a column of padding each side.
1005    let inner_width = layout.area.width.saturating_sub(4) as usize;
1006
1007    let mut lines = if let Some(bands) = positioned_band_cells(dialog) {
1008        banded_lines(dialog, &bands, inner_width, (normal, selected, theme))
1009    } else {
1010        let rows = build_dialog_rows(dialog);
1011        let button_width = rows
1012            .iter()
1013            .filter_map(|row| match row {
1014                DialogRow::FieldWithButton { button_idx, .. } => {
1015                    Some(display_label(&dialog.buttons[*button_idx]).len())
1016                }
1017                _ => None,
1018            })
1019            .max()
1020            .unwrap_or(0);
1021        rows.iter()
1022            .map(|row| {
1023                row_line(
1024                    dialog,
1025                    row,
1026                    (inner_width, button_width),
1027                    (normal, selected, theme),
1028                )
1029            })
1030            .collect()
1031    };
1032    if lines.is_empty() {
1033        lines.push(Line::from(Span::raw(" ")));
1034    }
1035    let block = Block::default()
1036        .borders(Borders::ALL)
1037        .title(title)
1038        .border_style(Style::default().fg(support::to_ratatui_color(theme.menu_border)))
1039        .style(Style::default().bg(support::to_ratatui_color(theme.menu_background)));
1040    Paragraph::new(lines).block(block).render(layout.area, buf);
1041}

1043/// Render a resident dialog as a DOCKED panel filling `area` (no popup
1044/// centering/sizing): title border plus the banded control rows. Reuses
1045/// the banded cell layout so combat's rows read defense|stance ▼|offense.
1046pub fn render_dialog_panel(
1047    dialog: &DialogState,
1048    area: Rect,
1049    buf: &mut Buffer,
1050    theme: &crate::vellum::support::AppTheme,
1051) {
1052    let title = dialog.title.as_deref().unwrap_or("Panel");
1053    let normal = Style::default()
1054        .fg(support::to_ratatui_color(theme.menu_item_normal))
1055        .bg(support::to_ratatui_color(theme.menu_background));
1056    let selected = Style::default()
1057        .fg(support::to_ratatui_color(theme.menu_item_selected))
1058        .bg(support::to_ratatui_color(theme.menu_item_focused));
1059
1060    let inner_width = area.width.saturating_sub(2) as usize;
1061    let mut lines: Vec<Line> = Vec::new();
1062
1063    if let Some(bands) = positioned_band_cells(dialog) {
1064        for band in &bands {
1065            if let [BandCell::Progress { index }] = band.as_slice()
1066                && let Some(pb) = dialog.progress_bars.get(*index)
1067            {
1068                lines.push(render_progress_bar_line(pb, inner_width, theme));
1069                continue;
1070            }
1071            lines.push(Line::from(band_spans(band, dialog, (normal, selected))));
1072        }
1073    }
1074    // Footer links (skin/search/grip/...).
1075    if !dialog.links.is_empty() {
1076        let mut spans = vec![Span::raw(" ")];
1077        for (i, link) in dialog.links.iter().enumerate() {
1078            if i > 0 {
1079                spans.push(Span::raw("  "));
1080            }
1081            spans.push(Span::styled(link.label.clone(), normal));
1082        }
1083        lines.push(Line::from(spans));
1084    }
1085    if lines.is_empty() {
1086        lines.push(Line::from(Span::raw(" waiting for panel data… ")));
1087    }
1088
1089    let block = Block::default()
1090        .borders(Borders::ALL)
1091        .title(title)
1092        .border_style(Style::default().fg(support::to_ratatui_color(theme.menu_border)))
1093        .style(Style::default().bg(support::to_ratatui_color(theme.menu_background)));
1094    Paragraph::new(lines).block(block).render(area, buf);
1095}

1097/// Render a progress bar as a styled line for dialog display
1098fn render_progress_bar_line(
1099    pb: &crate::vellum::data::DialogProgressBar,
1100    width: usize,
1101    theme: &crate::vellum::support::AppTheme,
1102) -> Line<'static> {
1103    let value = pb.value.min(100) as usize;
1104    let text = &pb.text;
1105
1106    // Calculate filled vs unfilled portions
1107    let bar_width = width.saturating_sub(2); // Leave space for borders
1108    let filled_width = (bar_width * value) / 100;
1109
1110    // Create the progress bar characters
1111    let filled_char = '█';
1112    let unfilled_char = '░';
1113
1114    // Use theme colors
1115    let filled_style = Style::default().fg(support::to_ratatui_color(theme.background_selected));
1116    let unfilled_style = Style::default().fg(support::to_ratatui_color(theme.background_secondary));
1117
1118    // Build the line with text overlaid on the bar
1119    // Center the text over the bar
1120    let text_start = if bar_width > text.len() {
1121        (bar_width - text.len()) / 2
1122    } else {
1123        0
1124    };
1125    let text_end = text_start + text.len().min(bar_width);
1126
1127    let mut spans = Vec::new();
1128    spans.push(Span::raw(" ")); // Left padding
1129
1130    // Build the bar with text overlay
1131    let mut pos = 0;
1132    while pos < bar_width {
1133        if pos >= text_start && pos < text_end {
1134            // Render text character with appropriate background
1135            let char_idx = pos - text_start;
1136            let ch = text.chars().nth(char_idx).unwrap_or(' ');
1137            let is_filled = pos < filled_width;
1138            let style = if is_filled {
1139                Style::default()
1140                    .fg(support::to_ratatui_color(theme.text_primary))
1141                    .bg(support::to_ratatui_color(theme.background_selected))
1142            } else {
1143                Style::default()
1144                    .fg(support::to_ratatui_color(theme.text_primary))
1145                    .bg(support::to_ratatui_color(theme.menu_background))
1146            };
1147            spans.push(Span::styled(ch.to_string(), style));
1148            pos += 1;
1149        } else {
1150            // Render bar character
1151            let is_filled = pos < filled_width;
1152            if is_filled {
1153                // Collect consecutive filled chars
1154                let count = (text_start.saturating_sub(pos)).min(filled_width.saturating_sub(pos));
1155                if count > 0 {
1156                    let s: String = std::iter::repeat_n(filled_char, count).collect();
1157                    spans.push(Span::styled(s, filled_style));
1158                    pos += count;
1159                } else {
1160                    spans.push(Span::styled(filled_char.to_string(), filled_style));
1161                    pos += 1;
1162                }
1163            } else {
1164                // Collect consecutive unfilled chars
1165                let count = text_start
1166                    .saturating_sub(pos)
1167                    .max(1)
1168                    .min(bar_width.saturating_sub(pos));
1169                if count > 0 && pos < text_start {
1170                    let s: String = std::iter::repeat_n(unfilled_char, count).collect();
1171                    spans.push(Span::styled(s, unfilled_style));
1172                    pos += count;
1173                } else {
1174                    spans.push(Span::styled(unfilled_char.to_string(), unfilled_style));
1175                    pos += 1;
1176                }
1177            }
1178        }
1179    }
1180
1181    spans.push(Span::raw(" ")); // Right padding
1182
1183    Line::from(spans)
1184}

27/// A colour as the widgets take it, `#rrggbb`.
28fn hex(color: ratatui::style::Color) -> Option<String> {
29    match color {
30        ratatui::style::Color::Rgb(red, green, blue) => {
31            Some(format!("#{red:02x}{green:02x}{blue:02x}"))
32        }
33        _ => None,
34    }
35}

44/// One of Hydra's pieces of a line as a widget's segment.
45fn segment(piece: &crate::text::Piece) -> TextSegment {
46    let link = piece
47        .link
48        .as_ref()
49        .map(|link| super::text::link_data(link, &piece.text));
50    TextSegment {
51        text: piece.text.clone(),
52        fg: piece.style.fg.and_then(hex),
53        bg: piece.style.bg.and_then(hex),
54        bold: piece.style.add_modifier.contains(Modifier::BOLD),
55        mono: false,
56        span_type: if link.is_some() {
57            SpanType::Link
58        } else {
59            SpanType::Normal
60        },
61        link_data: link,
62        custom_emoji: None,
63        inline_image: None,
64    }
65}

67/// The lines of the first of `streams` the story keeps, each as a widget's
68/// segments.
69fn stream_lines(streams: &[&str], seen: &Seen<'_>) -> Vec<Vec<TextSegment>> {
70    streams
71        .iter()
72        .find_map(|stream| {
73            crate::widgets::text_lines(&Widget::Stream((*stream).to_owned()), seen)
74                .filter(|lines| !lines.is_empty())
75        })
76        .unwrap_or_default()
77        .into_iter()
78        .map(|(_, line)| line.iter().map(segment).collect())
79        .collect()
80}

82/// Each cell's link in `area`, found by `link_at`, added as a click on the
83/// window's own character.
84fn cell_links(
85    area: Rect,
86    (seen, hits): (&Seen<'_>, &mut Hits),
87    link_at: impl Fn(u16, u16) -> Option<super::data::LinkData>,
88) {
89    if !seen.own {
90        return;
91    }
92    for y in area.y..area.y + area.height {
93        for x in area.x..area.x + area.width {
94            if let Some(link) = link_at(x, y) {
95                hits.add(
96                    Rect::new(x, y, 1, 1),
97                    Target::Link {
98                        seat: seen.seat,
99                        link: super::text::run_link(&link),
100                    },
101                );
102            }
103        }
104    }
105}

107/// A missing window drawn: every tracked effect that is not up, a click on
108/// one typing what puts it up.
109pub fn missing(buf: &mut Buffer, area: Rect, (seen, hits): (&Seen<'_>, &mut Hits)) {
110    let (gone, tracked) = crate::widgets::gone(seen);
111    let mut widget = missing_spells::MissingSpells::new("");
112    let listed: Vec<MissingSpell> = gone
113        .iter()
114        .map(|(name, _)| MissingSpell {
115            number: None,
116            name: name.clone(),
117        })
118        .collect();
119    widget.update_from_state(&listed, tracked);
120    widget.render(area, buf);
121    for (row, (_, command)) in gone.iter().enumerate() {
122        let (Ok(row), Some(command)) = (u16::try_from(row), command) else {
123            continue;
124        };
125        if row >= area.height {
126            break;
127        }
128        hits.add(
129            Rect::new(area.x, area.y + row, area.width, 1),
130            Target::Button(format!("{PRESS}typed:{command}")),
131        );
132    }
133}

135/// The objectives as the quest panel takes them.
136fn objectives(state: &GameState) -> ObjectivesContent {
137    ObjectivesContent {
138        objectives: state
139            .objectives
140            .all()
141            .map(|objective| Objective {
142                id: objective.id.clone(),
143                kind: objective.kind.clone(),
144                state: objective.state.clone().unwrap_or_default(),
145                name: objective.name.clone().unwrap_or_default(),
146                description: objective.description.clone().unwrap_or_default(),
147                location: objective.location.clone(),
148                cadence: objective.cadence.clone(),
149                rewards: objective
150                    .rewards
151                    .iter()
152                    .map(|reward| ObjectiveReward {
153                        reward_type: reward.kind.clone(),
154                        amount: reward.amount.unwrap_or(0),
155                    })
156                    .collect(),
157                actions: objective
158                    .actions
159                    .iter()
160                    .map(|action| ObjectiveAction {
161                        action_type: action.kind.clone(),
162                        cmd: action.command.clone(),
163                    })
164                    .collect(),
165            })
166            .collect(),
167        generation: 1,
168    }
169}

171/// A quests window drawn: the objectives the game lists, a click on an
172/// action typing it.
173pub fn quests(
174    base: &WindowBase,
175    buf: &mut Buffer,
176    area: Rect,
177    (seen, hits): (&Seen<'_>, &mut Hits),
178) {
179    let mut widget = quests::QuestsWindow::new("");
180    widget.set_border_config(false, None, None);
181    widget.set_text_color(text_color(base, seen));
182    widget.set_transparent_background(true);
183    if let Some(state) = seen.state() {
184        widget.update_from_state(&objectives(state));
185    }
186    widget.render(area, buf);
187    if !seen.own {
188        return;
189    }
190    for y in area.y..area.y + area.height {
191        if let Some(command) = widget.handle_click(y, area) {
192            hits.add(
193                Rect::new(area.x, y, area.width, 1),
194                Target::Button(format!("{PRESS}typed:{command}")),
195            );
196        }
197    }
198}

210    widget.set_transparent_background(true);

222        widget.finish_line();

224    widget.render(area, buf);
225    cell_links(area, (seen, hits), |x, y| widget.handle_click(x, y, area));

290/// The whole-inventory snapshot as the tree takes it; none before one
291/// came.
292fn snapshot(state: &GameState) -> Option<ManagedInventoryState> {
293    let snapshot = &state.inventory_snapshot;
294    let items: Vec<ManagedInventoryItem> = snapshot
295        .all()
296        .map(|item| ManagedInventoryItem {
297            id: item.id.clone(),
298            relation: item.relation.clone(),
299            parent: item.parent.clone(),
300            name: item.name.clone(),
301            article: item.article.clone(),
302            adjective: item.adjective.clone(),
303            noun: item.noun.clone(),
304            long: item.long.clone(),
305            weight: item.weight.unwrap_or(-1),
306            encum: item.encum,
307            in_max: item.in_max,
308            on_max: item.on_max,
309            in_encum: item.in_encum,
310            in_selector: item.in_selector.clone(),
311            locker: item.locker,
312            familyvault: item.familyvault,
313            flags: item.flags.clone(),
314        })
315        .collect();
316    if items.is_empty() && snapshot.room().is_none() {
317        return None;
318    }
319    Some(ManagedInventoryState {
320        token: String::new(),
321        room: snapshot.room().unwrap_or_default().to_owned(),
322        items,
323        complete: snapshot.is_complete(),
324        generation: 1,
325    })
326}

348/// The Betrayer's blood and items, as its panel's labels say them.
349fn blood(dialog: &Dialog) -> BetrayerState {
350    let label = |wanted: &str| {
351        dialog.parts.iter().find_map(|(id, part)| match part {
352            Part::Label { text, .. } if id == wanted => Some(text.clone()),
353            _ => None,
354        })
355    };
356    let mut state = BetrayerState::default();
357    if let Some(text) = label("lblBPs") {
358        state.update_blood_points(&text);
359    }
360    let items: Vec<String> = (1..=20)
361        .map_while(|at| label(&format!("lblitem{at}")))
362        .collect();
363    state.update_items(items);
364    state
365}

399/// An attribute of a dialog control, by name.
400fn attr<'a>(attrs: &'a [(String, String)], name: &str) -> Option<&'a str> {
401    attrs
402        .iter()
403        .find(|(key, _)| key == name)
404        .map(|(_, value)| value.as_str())
405}

407/// A control's place, as its attributes give it; none when they give none.
408fn placed(attrs: &[(String, String)]) -> Option<DialogControlLayout> {
409    DialogControlLayout::from_attrs(attrs)
410}
411
412/// Whether an attribute says yes: `1`, `t` or `true`.
413fn yes(value: Option<&str>) -> bool {
414    value.is_some_and(|value| matches!(value, "1" | "t") || value.eq_ignore_ascii_case("true"))
415}

417/// A button, a close button or a radio button, read from its attributes.
418fn button(id: &str, kind: &str, attrs: &[(String, String)]) -> DialogButton {
419    let radio = kind == "radio";
420    let command = attr(attrs, "cmd").unwrap_or_default().to_owned();
421    let label = attr(attrs, if radio { "text" } else { "value" })
422        .map_or_else(|| id.to_owned(), str::to_owned);
423    let autosend = attr(attrs, "autosend").is_some_and(|value| {
424        let value = value.trim();
425        value.is_empty() || !(matches!(value, "0" | "f") || value.eq_ignore_ascii_case("false"))
426    });
427    DialogButton {
428        id: id.to_owned(),
429        label,
430        is_close: kind == "closeButton" || command.trim().is_empty(),
431        command,
432        is_radio: radio,
433        selected: radio && yes(attr(attrs, "value")),
434        autosend: radio && autosend,
435        group: if radio {
436            attr(attrs, "group").map(str::to_owned)
437        } else {
438            None
439        },
440        layout: placed(attrs),
441    }
442}

444/// A picker, read from its attributes: its choices paired with the values
445/// they send, a choice with no value sending its own words.
446fn picker(id: &str, attrs: &[(String, String)]) -> DialogDropDown {
447    let text = |name: &str| attr(attrs, name).unwrap_or_default().to_owned();
448    let texts = text("content_text");
449    let values = text("content_value");
450    let values: Vec<&str> = values
451        .split(',')
452        .filter(|value| !value.is_empty())
453        .collect();
454    let options = texts
455        .split(',')
456        .filter(|shown| !shown.is_empty())
457        .enumerate()
458        .map(|(at, shown)| {
459            let value = values.get(at).copied().unwrap_or(shown);
460            (shown.to_owned(), value.to_owned())
461        })
462        .collect();
463    DialogDropDown {
464        id: id.to_owned(),
465        value: text("value"),
466        options,
467        command: text("cmd"),
468        tooltip: attr(attrs, "tooltip")
469            .filter(|value| !value.is_empty())
470            .map(str::to_owned),
471        layout: placed(attrs),
472    }
473}

475/// One of a dialog's widgets added to the dialog drawn: buttons, links,
476/// images, skins, spinners, pickers and fields, each read from its
477/// attributes.
478fn add_control(shown: &mut DialogState, id: &str, (kind, attrs): (&str, &[(String, String)])) {
479    let text = |name: &str| attr(attrs, name).unwrap_or_default().to_owned();
480    let layout = placed(attrs);
481    match kind {
482        "cmdButton" | "closeButton" | "radio" => shown.buttons.push(button(id, kind, attrs)),
483        "link" => shown.links.push(DialogLink {
484            id: id.to_owned(),
485            label: text("value"),
486            command: text("cmd"),
487            layout,
488        }),
489        "image" => shown.images.push(DialogImage {
490            id: id.to_owned(),
491            name: text("name"),
492            command: text("cmd"),
493            tooltip: attr(attrs, "tooltip")
494                .filter(|value| !value.is_empty())
495                .map(str::to_owned),
496            layout,
497        }),
498        "skin" => shown.skins.push(DialogSkin {
499            id: id.to_owned(),
500            name: text("name"),
501            controls: text("controls")
502                .split(',')
503                .map(str::trim)
504                .filter(|control| !control.is_empty())
505                .map(str::to_owned)
506                .collect(),
507            layout,
508        }),
509        "upDownEditBox" => shown.spinboxes.push(DialogSpinBox {
510            id: id.to_owned(),
511            value: attr(attrs, "value")
512                .and_then(|value| value.parse().ok())
513                .unwrap_or(0),
514            min: attr(attrs, "min")
515                .and_then(|value| value.parse().ok())
516                .unwrap_or(i32::MIN),
517            max: attr(attrs, "max")
518                .and_then(|value| value.parse().ok())
519                .unwrap_or(i32::MAX),
520            layout,
521        }),
522        "dropDownBox" => shown.dropdowns.push(picker(id, attrs)),
523        "editBox" => {
524            let value = text("value");
525            shown.fields.push(DialogField {
526                id: id.to_owned(),
527                cursor: value.chars().count(),
528                value,
529                enter_button: attr(attrs, "enterButton").map(str::to_owned),
530                focused: false,
531            });
532        }
533        _ => {}
534    }
535}

537/// One of the game's dialogs as the dialog drawn: its labels, bars and
538/// controls in the order they came.
539fn dialog_of(id: &str, kept: &Dialog) -> DialogState {
540    let mut shown = DialogState::empty(id.to_owned(), kept.title.clone());
541    for (part_id, part) in &kept.parts {
542        match part {
543            Part::Label { text, attrs } => shown.display_labels.push(DialogLabel {
544                id: part_id.clone(),
545                value: text.clone(),
546                layout: placed(attrs),
547                justify: attr(attrs, "justify").and_then(|value| value.parse().ok()),
548            }),
549            Part::Bar {
550                percent,
551                text,
552                attrs,
553            } => shown.progress_bars.push(DialogProgressBar {
554                id: part_id.clone(),
555                value: *percent,
556                text: text.clone(),
557                layout: placed(attrs),
558            }),
559            Part::Widget { kind, attrs } => add_control(&mut shown, part_id, (kind, attrs)),
560        }
561    }
562    shown
563}

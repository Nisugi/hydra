88    /// Add a line of styled text segments (for SpellsWindow/InventoryWindow style usage)
89    pub fn add_line(&mut self, segments: Vec<TextSegment>) {
90        if segments.is_empty() {
91            // Add empty line as-is (preserves spacing)
92            self.lines.push(Vec::new());
93        } else {
94            // Cache any links in the segments
95            for segment in &segments {
96                if let Some(ref link_data) = segment.link_data {
97                    // Check if we already have this exist_id in the most recent entry
98                    let should_append = if let Some(last) = self.recent_links.back_mut() {
99                        if last.exist_id == link_data.exist_id {
100                            // Append to existing text
101                            last.text.push_str(&segment.text);
102                            true
103                        } else {
104                            false
105                        }
106                    } else {
107                        false
108                    };
109
110                    if !should_append {
111                        // New link - create new entry
112                        let mut new_link = link_data.clone();
113                        new_link.text.clone_from(&segment.text);
114                        self.recent_links.push_back(new_link);
115                        if self.recent_links.len() > self.max_recent_links {
116                            self.recent_links.pop_front();
117                        }
118                    }
119                }
120            }
121
122            // Apply highlights before wrapping/adding to buffer
123            let highlighted = self
124                .highlight_engine
125                .apply_highlights_to_segments(&segments, "")
126                .unwrap_or(segments);
127
128            // Apply word wrap if enabled
129            if self.word_wrap && self.wrap_width > 0 {
130                let wrapped_lines = self.wrap_line(highlighted);
131                for line in wrapped_lines {
132                    self.lines.push(line);
133                }
134            } else {
135                self.lines.push(highlighted);
136            }
137        }
138    }

142    pub fn add_simple_line(&mut self, text: String, color: Option<String>, link: Option<LinkData>) {
143        if text.is_empty() {
144            return;
145        }
146
147        // Cache link if present
148        if let Some(ref link_data) = link {
149            // Check for deduplication
150            let should_append = if let Some(last) = self.recent_links.back_mut() {
151                if last.exist_id == link_data.exist_id {
152                    last.text.push_str(&text);
153                    true
154                } else {
155                    false
156                }
157            } else {
158                false
159            };
160
161            if !should_append {
162                let mut new_link = link_data.clone();
163                new_link.text.clone_from(&text);
164                self.recent_links.push_back(new_link);
165                if self.recent_links.len() > self.max_recent_links {
166                    self.recent_links.pop_front();
167                }
168            }
169        }
170
171        let segment = TextSegment {
172            text,
173            fg: color,
174            bg: None,
175            bold: false,
176            mono: false,
177            span_type: crate::vellum::support::SpanType::Normal,
178            link_data: link,
179            custom_emoji: None,
180            inline_image: None,
181        };
182
183        // Apply highlights
184        let segments = vec![segment];
185        let highlighted = self
186            .highlight_engine
187            .apply_highlights_to_segments(&segments, "")
188            .unwrap_or(segments);
189
190        self.lines.push(highlighted);
191    }

376    /// Wrap a line of styled segments to window width (word-boundary aware)
377    /// Copied from `InventoryWindow` for DRY
378    fn wrap_line(&self, segments: Vec<TextSegment>) -> Vec<Vec<TextSegment>> {
379        if self.wrap_width == 0 {
380            return vec![Vec::new()];
381        }
382
383        let mut wrapped_lines = Vec::new();
384        let mut current_line = Vec::new();
385        let mut current_width = 0;
386
387        // Track word buffer for smart wrapping
388        let mut word_buffer: Vec<TextSegment> = Vec::new();
389        let mut word_buffer_len = 0;
390        let mut in_word = false;
391
392        for segment in segments {
393            for ch in segment.text.chars() {
394                let is_whitespace = ch.is_whitespace();
395
396                if is_whitespace {
397                    // Flush word buffer if we have one
398                    if in_word && !word_buffer.is_empty() {
399                        // Check if word fits on current line
400                        if current_width + word_buffer_len <= self.wrap_width {
401                            // Word fits - add it to current line
402                            for word_seg in word_buffer.drain(..) {
403                                Self::append_to_line(&mut current_line, word_seg);
404                            }
405                            current_width += word_buffer_len;
406                        } else if word_buffer_len <= self.wrap_width {
407                            // Word doesn't fit on current line, but fits on new line - wrap
408                            if !current_line.is_empty() {
409                                wrapped_lines.push(std::mem::take(&mut current_line));
410                                current_width = 0;
411                            }
412                            // Add word to new line
413                            for word_seg in word_buffer.drain(..) {
414                                Self::append_to_line(&mut current_line, word_seg);
415                            }
416                            current_width += word_buffer_len;
417                        } else {
418                            // Word is longer than width - must break it mid-word
419                            for word_seg in word_buffer.drain(..) {
420                                for word_ch in word_seg.text.chars() {
421                                    if current_width >= self.wrap_width {
422                                        wrapped_lines.push(std::mem::take(&mut current_line));
423                                        current_width = 0;
424                                    }
425                                    Self::append_char_to_line(
426                                        &mut current_line,
427                                        word_ch,
428                                        &word_seg,
429                                    );
430                                    current_width += 1;
431                                }
432                            }
433                        }
434                        word_buffer_len = 0;
435                        in_word = false;
436                    }
437
438                    // Add whitespace immediately (don't buffer it)
439                    if current_width >= self.wrap_width {
440                        // Wrap before whitespace
441                        wrapped_lines.push(std::mem::take(&mut current_line));
442                        current_width = 0;
443                        // Don't add whitespace at start of new line
444                        continue;
445                    }
446                    Self::append_char_to_line(&mut current_line, ch, &segment);
447                    current_width += 1;
448                } else {
449                    // Non-whitespace character - add to word buffer
450                    in_word = true;
451                    Self::append_char_to_line(&mut word_buffer, ch, &segment);
452                    word_buffer_len += 1;
453                }
454            }
455        }
456
457        // Flush remaining word buffer
458        if !word_buffer.is_empty() {
459            if current_width + word_buffer_len <= self.wrap_width {
460                // Word fits on current line
461                for word_seg in word_buffer {
462                    Self::append_to_line(&mut current_line, word_seg);
463                }
464            } else if word_buffer_len <= self.wrap_width {
465                // Word needs new line
466                if !current_line.is_empty() {
467                    wrapped_lines.push(std::mem::take(&mut current_line));
468                }
469                for word_seg in word_buffer {
470                    Self::append_to_line(&mut current_line, word_seg);
471                }
472            } else {
473                // Word is too long - must break it
474                for word_seg in word_buffer {
475                    for word_ch in word_seg.text.chars() {
476                        if current_width >= self.wrap_width {
477                            wrapped_lines.push(std::mem::take(&mut current_line));
478                            current_width = 0;
479                        }
480                        Self::append_char_to_line(&mut current_line, word_ch, &word_seg);
481                        current_width += 1;
482                    }
483                }
484            }
485        }
486
487        // Add remaining line if not empty
488        if !current_line.is_empty() {
489            wrapped_lines.push(current_line);
490        }
491
492        // Return at least one empty line if nothing was added
493        if wrapped_lines.is_empty() {
494            wrapped_lines.push(Vec::new());
495        }
496
497        wrapped_lines
498    }

500    /// Helper to append a segment to a line, merging with last segment if style matches
501    fn append_to_line(line: &mut Vec<TextSegment>, segment: TextSegment) {
502        if let Some(last_seg) = line.last_mut() {
503            // Check if all properties match (including link_data for proper link boundaries)
504            if last_seg.fg == segment.fg
505                && last_seg.bg == segment.bg
506                && last_seg.bold == segment.bold
507                && last_seg.span_type == segment.span_type
508                && last_seg.link_data == segment.link_data
509            {
510                // Merge text into existing segment
511                last_seg.text.push_str(&segment.text);
512                return;
513            }
514        }
515        // Can't merge - append as new segment
516        line.push(segment);
517    }

519    /// Append one character, merging with the last segment when its style
520    /// matches `template`. The merge path (overwhelmingly common) performs no
521    /// allocation; only a new segment clones the template's colors and link.
522    fn append_char_to_line(line: &mut Vec<TextSegment>, ch: char, template: &TextSegment) {
523        if let Some(last_seg) = line.last_mut()
524            && last_seg.fg == template.fg
525            && last_seg.bg == template.bg
526            && last_seg.bold == template.bold
527            && last_seg.span_type == template.span_type
528            && last_seg.link_data == template.link_data
529        {
530            last_seg.text.push(ch);
531            return;
532        }
533        line.push(TextSegment {
534            text: ch.to_string(),
535            fg: template.fg.clone(),
536            bg: template.bg.clone(),
537            bold: template.bold,
538            mono: false,
539            span_type: template.span_type,
540            link_data: template.link_data.clone(),
541            custom_emoji: None,
542            inline_image: None,
543        });
544    }

556    /// Render the widget with optional focus indicator
557    pub fn render_with_focus(&mut self, area: Rect, buf: &mut Buffer, _focused: bool) {
558        // Update inner size based on area
559        self.update_inner_size(area.width, area.height);
560
561        // Clear the area to prevent bleed-through from windows behind
562        Clear.render(area, buf);
563
564        // Fill background if not transparent
565        if !self.transparent_background
566            && let Some(bg_color) = self.background_color
567        {
568            crate::vellum::support::paint(area, buf, bg_color);
569        }
570
571        // Create border block
572        let mut block = Block::default();
573
574        if self.show_border {
575            let border_color = self.border_color.unwrap_or(Color::White);
576
577            // Apply border sides configuration
578            let mut borders = Borders::empty();
579            if self.border_sides.top {
580                borders |= Borders::TOP;
581            }
582            if self.border_sides.bottom {
583                borders |= Borders::BOTTOM;
584            }
585            if self.border_sides.left {
586                borders |= Borders::LEFT;
587            }
588            if self.border_sides.right {
589                borders |= Borders::RIGHT;
590            }
591
592            block = block
593                .borders(borders)
594                .border_style(Style::default().fg(border_color));
595
596            // Apply border type
597            if let Some(ref style) = self.border_style {
598                block = block.border_type(crate::chrome::border_type(style));
599            }
600
601            if !self.title.is_empty() {
602                block = block.title(self.title.as_str());
603            }
604        }
605
606        let inner = block.inner(area);
607
608        // Calculate visible range (scroll from bottom, like SpellsWindow)
609        let total_lines = self.lines.len();
610        let visible_start = total_lines.saturating_sub(self.scroll_offset + inner.height as usize);
611        let visible_end = total_lines.saturating_sub(self.scroll_offset);
612
613        // Build visible lines
614        let mut display_lines = Vec::new();
615        for line in &self.lines[visible_start..visible_end] {
616            let spans: Vec<Span> = line
617                .iter()
618                .map(|segment| Span::styled(segment.text.clone(), self.apply_style(segment)))
619                .collect();
620            display_lines.push(Line::from(spans));
621        }
622
623        let paragraph = Paragraph::new(display_lines).block(block);
624
625        ratatui::widgets::Widget::render(paragraph, area, buf);
626    }

628    /// Apply styling to a text segment
629    fn apply_style(&self, segment: &TextSegment) -> Style {
630        let mut style = Style::default();
631
632        if let Some(ref fg) = segment.fg {
633            if let Some(color) = Self::parse_color(fg) {
634                style = style.fg(color);
635            }
636        } else if let Some(default_fg) = self.default_text_color {
637            style = style.fg(default_fg);
638        }
639
640        if let Some(ref bg) = segment.bg {
641            if let Some(color) = Self::parse_color(bg) {
642                style = style.bg(color);
643            }
644        } else if !self.transparent_background
645            && let Some(bg_color) = self.background_color
646        {
647            style = style.bg(bg_color);
648        }
649
650        if segment.bold {
651            style = style.add_modifier(ratatui::style::Modifier::BOLD);
652        }
653
654        style
655    }

657    /// Handle a click at the given coordinates (for targets style usage)
658    /// Returns the `LinkData` if a link was clicked at that position
659    /// Note: This is a simple implementation - coordinate-based click handling
660    /// is complex and may need refinement based on actual usage
661    #[must_use]
662    pub fn handle_click(&self, _x: u16, y: u16, area: Rect) -> Option<LinkData> {
663        // Calculate which line was clicked
664        let inner_y = if self.show_border {
665            y.saturating_sub(area.y + 1)
666        } else {
667            y.saturating_sub(area.y)
668        };
669
670        let total_lines = self.lines.len();
671        let visible_start = total_lines.saturating_sub(self.scroll_offset + self.inner_height);
672        let clicked_line_idx = visible_start + inner_y as usize;
673
674        if clicked_line_idx >= self.lines.len() {
675            return None;
676        }
677
678        // Get the clicked line
679        let line = &self.lines[clicked_line_idx];
680
681        // Find first link in this line
682        for segment in line {
683            if let Some(ref link) = segment.link_data {
684                return Some(link.clone());
685            }
686        }
687
688        None
689    }

691    /// Convert mouse position to text coordinates
692    #[must_use]
693    pub fn mouse_to_text_coords(
694        &self,
695        mouse_col: u16,
696        mouse_row: u16,
697        window_rect: Rect,
698    ) -> Option<(usize, usize)> {
699        let border_offset = u16::from(self.show_border);
700
701        // Bounds check within content area
702        if mouse_col < window_rect.x + border_offset
703            || mouse_col >= window_rect.x + window_rect.width - border_offset
704            || mouse_row < window_rect.y + border_offset
705            || mouse_row >= window_rect.y + window_rect.height - border_offset
706        {
707            return None;
708        }
709
710        let visible_height = self.inner_height;
711        let total_lines = self.lines.len();
712        let start_line = total_lines.saturating_sub(self.scroll_offset + visible_height);
713
714        let line_idx = start_line + (mouse_row - window_rect.y - border_offset) as usize;
715        let col_offset = (mouse_col - window_rect.x - border_offset) as usize;
716
717        Some((line_idx, col_offset))
718    }
719
720    /// Extract text from a selection range
721    #[must_use]
722    pub fn extract_selection_text(
723        &self,
724        start_line: usize,
725        start_col: usize,
726        end_line: usize,
727        end_col: usize,
728    ) -> String {
729        if self.lines.is_empty() {
730            return String::new();
731        }
732
733        let mut result = String::new();
734
735        for line_idx in start_line..=end_line.min(self.lines.len().saturating_sub(1)) {
736            if line_idx >= self.lines.len() {
737                break;
738            }
739
740            let line = &self.lines[line_idx];
741            let line_text: String = line.iter().map(|seg| seg.text.as_str()).collect();
742            let line_len = line_text.chars().count();
743
744            if start_line == end_line {
745                // Single line selection
746                let start = start_col.min(line_len);
747                let end = end_col.min(line_len);
748                if start < end {
749                    result.push_str(
750                        &line_text
751                            .chars()
752                            .skip(start)
753                            .take(end - start)
754                            .collect::<String>(),
755                    );
756                }
757            } else if line_idx == start_line {
758                // First line of multi-line selection
759                let start = start_col.min(line_len);
760                result.push_str(&line_text.chars().skip(start).collect::<String>());
761                result.push('\n');
762            } else if line_idx == end_line {
763                // Last line of multi-line selection
764                let end = end_col.min(line_len);
765                result.push_str(&line_text.chars().take(end).collect::<String>());
766            } else {
767                // Middle lines - take all
768                result.push_str(&line_text);
769                result.push('\n');
770            }
771        }
772
773        result
774    }

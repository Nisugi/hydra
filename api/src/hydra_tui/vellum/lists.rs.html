25/// A colour as the widgets take it, `#rrggbb`.
26fn hex(color: ratatui::style::Color) -> String {
27    match color {
28        ratatui::style::Color::Rgb(red, green, blue) => {
29            format!("#{red:02x}{green:02x}{blue:02x}")
30        }
31        _ => "#ffffff".to_owned(),
32    }
33}

35/// The colour a window's settings give its text, else the theme's.
36fn text_color(base: &WindowBase, seen: &Seen<'_>) -> String {
37    base.text_color
38        .clone()
39        .unwrap_or_else(|| hex(seen.colors.color(Token::Text)))
40}

42/// The creatures in the room, as the targets list takes them.
43fn creatures(state: &GameState) -> Vec<Creature> {
44    use hydra_model::state::creature::status::Classification as C;
45    let now = state.game_time_now();
46    state
47        .creatures()
48        .in_room()
49        .map(|creature| {
50            let flag = |class: C| creature.flag(class);
51            Creature {
52                name: creature.name.clone(),
53                noun: creature.noun.clone(),
54                id: format!("#{}", creature.id),
55                status: None,
56                flags: Some(CreatureFlags {
57                    statuses: creature
58                        .statuses(now)
59                        .into_iter()
60                        .map(hydra_ui::room::status_word)
61                        .collect(),
62                    hostile: creature.hostile_when_first_seen() == Some(true),
63                    disengaged: flag(C::Disengaged),
64                    dead: creature.dead(),
65                    sympathetic: flag(C::Sympathetic),
66                    ascended: flag(C::Ascended),
67                    inferior: flag(C::Inferior),
68                    ascension_boss: flag(C::AscensionBoss),
69                    mini_boss: flag(C::MiniBoss),
70                    challenging: flag(C::Challenging),
71                    rider: flag(C::Rider),
72                    mount: flag(C::Mount),
73                    ..CreatureFlags::default()
74                }),
75            }
76        })
77        .collect()
78}

80/// A targets window drawn: the room's creatures, the one targeted marked.
81pub fn targets(
82    (base, data): (&WindowBase, &TargetsWidgetData),
83    buf: &mut Buffer,
84    area: Rect,
85    (seen, hits): (&Seen<'_>, &mut Hits),
86) {
87    let mut widget = targets::Targets::new("");
88    widget.set_border_config(false, None, None);
89    widget.set_show_body_part_count(data.show_body_part_count);
90    widget.set_text_color(Some(text_color(base, seen)));
91    widget.set_transparent_background(true);
92    if let Some(state) = seen.state() {
93        let current = state
94            .targeting
95            .current()
96            .map(|id| format!("#{id}"))
97            .unwrap_or_default();
98        let listed: Vec<String> = state
99            .targeting
100            .ids()
101            .iter()
102            .map(|id| format!("#{id}"))
103            .collect();
104        widget.update_from_state(
105            &creatures(state),
106            &current,
107            &listed,
108            &TargetListConfig::default(),
109            area.width,
110            data.status_position.as_deref(),
111        );
112    }
113    widget.render(area, buf);
114    if seen.own {
115        for y in area.y..area.y + area.height {
116            if let Some(command) = widget.handle_click(y, area) {
117                hits.add(
118                    Rect::new(area.x, y, area.width, 1),
119                    Target::Button(format!("{PRESS}typed:{command}")),
120                );
121            }
122        }
123    }
124}

126/// A players window drawn: who else is in the room.
127pub fn players(
128    base: &WindowBase,
129    buf: &mut Buffer,
130    area: Rect,
131    (seen, hits): (&Seen<'_>, &mut Hits),
132) {
133    let mut widget = players::Players::new("");
134    widget.set_border_config(false, None, None);
135    widget.set_text_color(Some(text_color(base, seen)));
136    widget.set_transparent_background(true);
137    if let Some(state) = seen.state() {
138        let listed: Vec<Player> = state
139            .room
140            .players
141            .iter()
142            .map(|player| Player {
143                name: player.text.clone(),
144                id: player.id.clone(),
145                primary_status: player.before.clone(),
146                secondary_status: player.after.clone(),
147                dead: hydra_ui::room::appears_dead(player),
148            })
149            .collect();
150        widget.update_from_state(&listed, &TargetListConfig::default());
151    }
152    widget.render(area, buf);
153    if seen.own {
154        for y in area.y..area.y + area.height {
155            if let Some(link) = widget.handle_click(y, area) {
156                hits.add(
157                    Rect::new(area.x, y, area.width, 1),
158                    Target::Link {
159                        seat: seen.seat,
160                        link: super::text::run_link(&link),
161                    },
162                );
163            }
164        }
165    }
166}

190/// One of Hydra's runs as a widget's segment, coloured as the theme colours
191/// it.
192fn segment(run: &hydra_ui::StyledRun, seen: &Seen<'_>) -> TextSegment {
193    let style = seen.colors.run(run);
194    let link = run
195        .link
196        .as_ref()
197        .map(|link| super::text::link_data(link, &run.text));
198    TextSegment {
199        text: run.text.clone(),
200        fg: style.fg.map(hex),
201        bg: None,
202        bold: run.bold,
203        mono: run.monospace,
204        span_type: if link.is_some() {
205            SpanType::Link
206        } else {
207            SpanType::Normal
208        },
209        link_data: link,
210        custom_emoji: None,
211        inline_image: None,
212    }
213}

215/// A room window drawn: its name over the game's components, each shown
216/// or not as the window says.
217pub fn room(
218    (base, data): (&WindowBase, &RoomWidgetData),
219    buf: &mut Buffer,
220    area: Rect,
221    (seen, hits): (&Seen<'_>, &mut Hits),
222) {
223    let mut widget = room_window::RoomWindow::new(String::new());
224    widget.set_border_config(false, None, None);
225    widget.set_text_color(Some(text_color(base, seen)));
226    widget.set_component_visible("room desc", data.show_desc);
227    widget.set_component_visible("room objs", data.show_objs);
228    widget.set_component_visible("room players", data.show_players);
229    widget.set_component_visible("room exits", data.show_exits);
230    widget.set_show_name(data.show_name);
231    if let Some(state) = seen.state() {
232        let room = &state.room;
233        widget.set_title(hydra_ui::room::heading(room));
234        let player = seen.colors.palette().get(Token::Player);
235        let components: [(&str, Vec<hydra_ui::StyledRun>); 4] = [
236            (
237                "room desc",
238                room.description
239                    .as_ref()
240                    .map(|runs| hydra_ui::room::component(runs, false))
241                    .unwrap_or_default(),
242            ),
243            (
244                "room objs",
245                room.component("room objs")
246                    .map(|runs| hydra_ui::room::component(runs, true))
247                    .unwrap_or_default(),
248            ),
249            ("room players", hydra_ui::room::players(room, player)),
250            (
251                "room exits",
252                match hydra_ui::room::ways_out(Some(room)) {
253                    hydra_ui::room::WaysOut::Runs(runs) => runs,
254                    _ => Vec::new(),
255                },
256            ),
257        ];
258        for (id, runs) in components {
259            widget.start_component(id.to_owned());
260            if !runs.is_empty() {
261                for run in &runs {
262                    widget.add_segment(segment(run, seen));
263                }
264                widget.finish_line();
265            }
266            widget.finish_component();
267        }
268    }
269    widget.update_inner_size(area.width, area.height);
270    widget.render(area, buf);
271    if seen.own {
272        room_links(&widget, area, (seen, hits));
273    }
274}

276/// The links of the room window's lines as drawn, each found by a click.
277fn room_links(widget: &room_window::RoomWindow, area: Rect, (seen, hits): (&Seen<'_>, &mut Hits)) {
278    let lines = widget.get_wrapped_lines();
279    let shown = lines.iter().skip(widget.get_start_line());
280    for (y, line) in (area.y..area.bottom()).zip(shown) {
281        let mut x = area.x;
282        for segment in line {
283            let width = u16::try_from(unicode_width::UnicodeWidthStr::width(segment.text.as_str()))
284                .unwrap_or(u16::MAX);
285            let fits = width.min(area.right().saturating_sub(x));
286            if let Some(link) = segment.link_data.as_ref().filter(|_| fits > 0) {
287                hits.add(
288                    Rect::new(x, y, fits, 1),
289                    Target::Link {
290                        seat: seen.seat,
291                        link: super::text::run_link(link),
292                    },
293                );
294            }
295            x = x.saturating_add(width);
296        }
297    }
298}

300/// An experience window drawn: the level, the mind and the experience.
301pub fn experience(
302    (base, data): (&WindowBase, &GS4ExperienceWidgetData),
303    buf: &mut Buffer,
304    area: Rect,
305    seen: &Seen<'_>,
306) {
307    let mut widget = gs4_experience::GS4Experience::new("", &data.align);
308    widget.set_show_border(false);
309    widget.set_show_title(false);
310    widget.set_show_level(data.show_level);
311    widget.set_show_exp_bar(data.show_exp_bar);
312    widget.set_show_mind_bar(data.show_mind_bar);
313    if let Some(color) = base
314        .text_color
315        .as_deref()
316        .and_then(super::support::parse_color_to_ratatui)
317    {
318        widget.set_text_color(color);
319    }
320    if let Some(state) = seen.state() {
321        let experience = &state.character.experience;
322        widget.update_from_state(&GS4ExperienceState {
323            level_text: experience
324                .level
325                .as_deref()
326                .map(|level| format!("Level {level}"))
327                .unwrap_or_default(),
328            mind_state_value: experience.mind_percent.unwrap_or(0),
329            mind_state_text: experience.mind_state.clone().unwrap_or_default(),
330            next_level_value: experience.next_level_percent.unwrap_or(0),
331            next_level_text: experience.next_level.clone().unwrap_or_default(),
332            field_exp: experience.field_experience.map(u64::from),
333            max_field_exp: experience.field_experience_max.map(u64::from),
334            exp: experience.experience,
335            ascension_exp: experience.ascension_experience,
336            until_next: experience.until_next.map(u64::from),
337            generation: 1,
338            ..GS4ExperienceState::default()
339        });
340    }
341    widget.render(area, buf);
342}

344/// The quick bar's entries for the page shown, as the bar takes them.
345fn quick_entries(state: &GameState) -> Vec<QuickbarEntry> {
346    let shown = hydra_ui::quick::showing(Some(state), &QuickLook::default());
347    let Some(dialog) = state.dialogs.get(&shown) else {
348        return Vec::new();
349    };
350    hydra_ui::quick::parts(dialog)
351        .into_iter()
352        .enumerate()
353        .map(|(at, part)| match part {
354            BarPart::Label(words) => QuickbarEntry::Label {
355                id: at.to_string(),
356                value: words,
357            },
358            BarPart::Separator(_) => QuickbarEntry::Separator,
359            BarPart::Link { words, does } => match does {
360                BarLink::Typed(command) | BarLink::Game(command) => QuickbarEntry::Link {
361                    id: at.to_string(),
362                    value: words,
363                    cmd: command,
364                    echo: None,
365                },
366                BarLink::Url(href) => QuickbarEntry::Link {
367                    id: at.to_string(),
368                    value: words,
369                    cmd: href,
370                    echo: None,
371                },
372            },
373        })
374        .collect()
375}

377/// A quick bar drawn: the game's bar shown, each link a button.
378pub fn quick(
379    base: &WindowBase,
380    buf: &mut Buffer,
381    area: Rect,
382    (seen, hits): (&Seen<'_>, &mut Hits),
383) {
384    let mut widget = quickbar::Quickbar::new("");
385    widget.set_border_config(false, None, None);
386    widget.set_text_color(Some(text_color(base, seen)));
387    widget.set_transparent_background(true);
388    widget.set_entries(seen.state().map(quick_entries).unwrap_or_default());
389    widget.render(area, buf, false);
390    if !seen.own {
391        return;
392    }
393    // Each cell of the bar's row, asked of the bar as it was drawn.
394    for x in area.x..area.x + area.width {
395        let target = match widget.handle_click(x, area.y, area) {
396            Some(quickbar::QuickbarAction::ExecuteCommand(command)) => {
397                Target::Button(format!("{PRESS}typed:{command}"))
398            }
399            Some(quickbar::QuickbarAction::MenuRequest { exist, noun }) => Target::Link {
400                seat: seen.seat,
401                link: RunLink::Object {
402                    exist,
403                    noun,
404                    coord: None,
405                },
406            },
407            _ => continue,
408        };
409        hits.add(Rect::new(x, area.y, 1, 1), target);
410    }

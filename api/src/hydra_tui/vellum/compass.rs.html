1//! ASCII compass widget that highlights available exits and respects theme colors.
2//!
3//! The compass is a simple 7×3 grid of characters. Each cell can be styled with
4//! active/inactive colors, and the surrounding chrome inherits border/background
5//! settings from the window definition or theme fallback.
6
7use ratatui::{
8    buffer::Buffer,
9    layout::Rect,
10    style::{Color, Style},
11    widgets::{Block, Clear, Widget as RatatuiWidget},
12};
13use std::collections::HashSet;
14
15use crate::vellum::support;
16use crate::vellum::support::parse_color_to_ratatui;
17
18const CONTENT_WIDTH: u16 = 7;
19const CONTENT_HEIGHT: u16 = 3;
20
21const POSITIONS: [(u16, u16, &str, &str, &str); 11] = [
22    (0, 0, "↑", "up", "up"),
23    (2, 0, "↖", "nw", "northwest"),
24    (4, 0, "▲", "n", "north"),
25    (6, 0, "↗", "ne", "northeast"),
26    (2, 1, "◀", "w", "west"),
27    (4, 1, "o", "out", "out"),
28    (6, 1, "▶", "e", "east"),
29    (0, 2, "↓", "down", "down"),
30    (2, 2, "↙", "sw", "southwest"),
31    (4, 2, "▼", "s", "south"),
32    (6, 2, "↘", "se", "southeast"),
33];
34
35/// The ways out, as a compass: each direction lit when there is a way
36/// that way.
37pub struct Compass {
38    label: String,
39    directions: HashSet<String>, // normalized to lowercase
40    show_border: bool,
41    border_style: Option<String>,
42    border_color: Option<Color>,
43    border_sides: crate::layout::vellum_schema::BorderSides,
44    active_color: Option<Color>,
45    inactive_color: Option<Color>,
46    content_align: Option<String>,
47    background_color: Option<Color>,
48    transparent_background: bool,
49}
50
51impl Compass {
52    /// A compass titled `label`, with no ways out yet.
53    #[must_use]
54    pub fn new(label: &str) -> Self {
55        Self {
56            label: label.to_string(),
57            directions: HashSet::new(),
58            show_border: false,
59            border_style: None,
60            border_color: None,
61            border_sides: crate::layout::vellum_schema::BorderSides::default(),
62            active_color: Some(Color::Green),
63            inactive_color: Some(Color::DarkGray),
64            content_align: None,
65            background_color: None,
66            transparent_background: false,
67        }
68    }
69
70    /// Whether its border is drawn, and its style and colour.
71    pub fn set_border_config(
72        &mut self,
73        show_border: bool,
74        border_style: Option<String>,
75        border_color: Option<String>,
76    ) {
77        self.show_border = show_border;
78        self.border_style = border_style;
79        self.border_color = border_color.and_then(|c| Self::parse_color(&c));
80    }
81
82    /// Which of its border's edges are drawn.
83    pub fn set_border_sides(&mut self, border_sides: crate::layout::vellum_schema::BorderSides) {
84        self.border_sides = border_sides;
85    }
86
87    /// Its title.
88    pub fn set_title(&mut self, title: String) {
89        self.label = title;
90    }
91
92    /// The ways out, by the game's names for them: `n`, `up`, `out`.
93    pub fn set_directions(&mut self, directions: Vec<String>) {
94        self.directions = directions.into_iter().map(|d| d.to_lowercase()).collect();
95    }
96
97    /// The colours of a way there is and a way there is not, `#rrggbb`.
98    pub fn set_colors(&mut self, active_color: Option<String>, inactive_color: Option<String>) {
99        if let Some(color) = active_color.and_then(|c| Self::parse_color(&c)) {
100            self.active_color = Some(color);
101        }
102        if let Some(color) = inactive_color.and_then(|c| Self::parse_color(&c)) {
103            self.inactive_color = Some(color);
104        }
105    }
106
107    /// Where its content sits in the window, as `ContentAlign` names it.
108    pub fn set_content_align(&mut self, content_align: Option<String>) {
109        self.content_align = content_align;
110    }
111
112    /// The colour behind it, `#rrggbb`.
113    pub fn set_background_color(&mut self, color: Option<String>) {
114        self.background_color = match color {
115            Some(ref s) if s.trim() == "-" => None,
116            Some(value) => Self::parse_color(value.trim()),
117            None => None,
118        };
119    }
120
121    /// Whether what is behind it shows through.
122    pub fn set_transparent_background(&mut self, transparent: bool) {
123        self.transparent_background = transparent;
124    }
125
126    fn parse_color(input: &str) -> Option<Color> {
127        parse_color_to_ratatui(input)
128    }
129
130    /// It drawn into `area`.
131    pub fn render(&self, area: Rect, buf: &mut Buffer) {
132        if !self.transparent_background {
133            Clear.render(area, buf);
134            if let Some(bg_color) = self.background_color {
135                crate::vellum::support::paint(area, buf, bg_color);
136            }
137        }
138
139        let mut block = Block::default();
140        if !self.transparent_background
141            && let Some(bg_color) = self.background_color
142        {
143            block = block.style(Style::default().bg(bg_color));
144        }
145
146        if self.show_border {
147            let borders = support::to_ratatui_borders(&self.border_sides);
148            block = block.borders(borders);
149
150            if let Some(ref style) = self.border_style {
151                block = block.border_type(crate::chrome::border_type(style));
152            }
153
154            if let Some(color) = self.border_color {
155                block = block.border_style(Style::default().fg(color));
156            }
157
158            // Only set title if label is non-empty (avoids empty title affecting layout)
159            if !self.label.is_empty() {
160                block = block.title(self.label.as_str());
161            }
162        }
163
164        let inner_area = if self.show_border {
165            block.inner(area)
166        } else {
167            area
168        };
169
170        if self.show_border {
171            block.render(area, buf);
172        }
173
174        if inner_area.width == 0 || inner_area.height == 0 {
175            return;
176        }
177
178        let (row_offset, col_offset) = if let Some(ref align_str) = self.content_align {
179            let align = crate::vellum::support::ContentAlign::parse(align_str);
180            align.calculate_offset(
181                CONTENT_WIDTH,
182                CONTENT_HEIGHT,
183                inner_area.width,
184                inner_area.height,
185            )
186        } else {
187            (0, 0)
188        };
189
190        let active_color = self.active_color.unwrap_or(Color::Green);
191        let inactive_color = self.inactive_color.unwrap_or(Color::DarkGray);
192
193        for (col, row, glyph, short, long) in &POSITIONS {
194            let x = inner_area.x + col + col_offset;
195            let y = inner_area.y + row + row_offset;
196
197            if x >= buf.area().width || y >= buf.area().height {
198                continue;
199            }
200
201            let short_lower = short.to_lowercase();
202            let long_lower = long.to_lowercase();
203            let is_active =
204                self.directions.contains(&short_lower) || self.directions.contains(&long_lower);
205
206            let color = if is_active {
207                active_color
208            } else {
209                inactive_color
210            };
211
212            for (i, ch) in glyph.chars().enumerate() {
213                let char_x = x + u16::try_from(i).unwrap_or(u16::MAX);
214                if char_x < inner_area.x + inner_area.width && y < inner_area.y + inner_area.height
215                {
216                    buf[(char_x, y)].set_char(ch).set_fg(color);
217                    if !self.transparent_background
218                        && let Some(bg_color) = self.background_color
219                    {
220                        buf[(char_x, y)].set_bg(bg_color);
221                    }
222                }
223            }
224        }
225    }
226}

228#[cfg(test)]
229mod tests {
230    use super::*;
231
232    #[test]
233    fn test_active_and_inactive_colors() {
234        let mut compass = Compass::new("Compass");
235        compass.set_border_config(false, None, None);
236        compass.set_transparent_background(true);
237        compass.set_colors(Some("#ff0000".to_string()), Some("#00ff00".to_string()));
238        compass.set_directions(vec!["north".to_string(), "out".to_string()]);
239
240        let area = Rect::new(0, 0, 7, 3);
241        let mut buf = Buffer::empty(area);
242        compass.render(area, &mut buf);
243
244        assert_eq!(buf[(4, 0)].fg, Color::Rgb(255, 0, 0));
245        assert_eq!(buf[(4, 1)].fg, Color::Rgb(255, 0, 0));
246        assert_eq!(buf[(6, 1)].fg, Color::Rgb(0, 255, 0));
247    }
248
249    #[test]
250    fn test_background_fill_when_not_transparent() {
251        let mut compass = Compass::new("Compass");
252        compass.set_border_config(false, None, None);
253        compass.set_background_color(Some("#0000ff".to_string()));
254
255        let area = Rect::new(0, 0, 7, 3);
256        let mut buf = Buffer::empty(area);
257        compass.render(area, &mut buf);
258
259        assert_eq!(buf[(0, 0)].bg, Color::Rgb(0, 0, 255));
260    }
261
262    #[test]
263    fn test_content_alignment_offsets_positions() {
264        let mut compass = Compass::new("Compass");
265        compass.set_border_config(false, None, None);
266        compass.set_content_align(Some("center".to_string()));
267        compass.set_directions(vec!["north".to_string()]);
268
269        let area = Rect::new(0, 0, 9, 5);
270        let mut buf = Buffer::empty(area);
271        compass.render(area, &mut buf);
272
273        assert_eq!(buf[(5, 1)].fg, Color::Green);
274        assert_eq!(buf[(4, 0)].fg, Color::Reset);
275    }

1//! The survey's log: JSON Lines, append-only, one file per run, every
2//! record stamped with the run, the time, the server and the character.
3//!
4//! Four kinds of record. A `room` is written once, on the first arrival:
5//! what the game states about it, and what the verbs asked there said in
6//! their own fields. A `verb` is one line asked and its reply verbatim. An
7//! `edge` is one exit walked: where it led, or how it was refused. An
8//! `event` is a death, a heal, a resume or a stop. The log is the survey's
9//! only memory: a run resumes by reading it back.
10
11use std::collections::BTreeMap;
12use std::fs::{File, OpenOptions};
13use std::io::{BufRead, BufReader, Write};
14use std::path::{Path, PathBuf};
15
16use serde::{Deserialize, Serialize};

18/// A survey's own id for a room: `s1`, `s2`, ...
19pub type Sid = String;
20
21/// One record, with what every record carries.
22#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
23pub struct Record {
24    /// The run's stamp, the file's.
25    pub run: String,
26    /// When, as the wall clock reads, RFC 3339.
27    pub t: String,
28    /// The game instance: `GST`, `GS3`.
29    pub server: String,
30    /// The character.
31    #[serde(rename = "char")]
32    pub character: String,
33    /// The record itself.
34    #[serde(flatten)]
35    pub body: Body,
36}
37
38/// What a record says, by its kind.
39#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
40#[serde(tag = "kind", rename_all = "lowercase")]
41pub enum Body {
42    /// A room on its first arrival.
43    Room(Box<RoomRecord>),
44    /// A verb asked in a room, and its reply.
45    Verb(VerbRecord),
46    /// An exit walked from a room.
47    Edge(EdgeRecord),
48    /// Something that happened to the walk.
49    Event(EventRecord),
50}
51
52/// A room as first seen.
53#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
54pub struct RoomRecord {
55    /// The survey's id.
56    pub sid: Sid,
57    /// `<nav rm=>` exactly as given; `None` for a bare `<nav/>`.
58    pub uid: Option<String>,
59    /// The room's title, as the window subtitle gives it.
60    pub title: String,
61    /// The `room desc` component only.
62    pub desc: String,
63    /// The "Obvious exits/paths" line.
64    pub paths: String,
65    /// The compass's direction tokens.
66    pub compass: Vec<String>,
67    /// Every command the survey will try from here: the compass, then
68    /// `go <noun>` for what `anal room`, SENSE and the room's objects
69    /// name as ways out.
70    pub exits: Vec<String>,
71    /// `<roommeta>` as a raw attribute map.
72    pub roommeta: BTreeMap<String, String>,
73    /// How the room was reached; `None` for the starting room.
74    pub via: Option<Via>,
75    /// A first guess at where it lies on the survey's own grid.
76    pub pos: Option<[i32; 3]>,
77    /// From a Ranger's SENSE: the climate as the game words it.
78    pub climate: Option<String>,
79    /// From a Ranger's SENSE: the terrain as the game words it.
80    pub terrain: Option<String>,
81    /// From a Ranger's SENSE: the animals whose signs are about.
82    pub animals: Vec<String>,
83    /// From a Ranger's SENSE: what circles overhead.
84    pub overhead: Option<String>,
85    /// From a Ranger's SENSE: the exit it tacks on.
86    pub sensed_exit: Option<String>,
87    /// From a Ranger's TRACK AREA: the creatures that have been about.
88    #[serde(default)]
89    pub tracked: Vec<String>,
90    /// From LOCATION: the name the game guesses.
91    pub location: Option<String>,
92}

120/// An exit walked.
121#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
122pub struct EdgeRecord {
123    /// The room left.
124    pub from: Sid,
125    /// The command sent.
126    pub cmd: String,
127    /// The room arrived in, the same room when nothing happened, or `None`

129    pub to: Option<Sid>,

131    pub rt: Option<u32>,
132    /// The compass token when `cmd` is a direction.
133    pub dir: Option<String>,
134    /// How it went.
135    pub result: EdgeResult,
136    /// The game's lines: the refusal, or what came with the arrival.
137    pub lines: Vec<String>,
138}

140/// How an exit went.
141#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
142#[serde(rename_all = "lowercase")]
143pub enum EdgeResult {
144    /// The character arrived somewhere.
145    #[default]
146    Arrived,
147    /// The game refused it.
148    Failed,
149    /// Nothing happened and nothing was refused.
150    Stayed,
151}

153/// Something that happened to the walk.
154#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
155pub struct EventRecord {
156    /// Where.
157    pub sid: Sid,
158    /// What.
159    pub what: Happening,

162/// What can happen to the walk.
163#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
164#[serde(rename_all = "lowercase")]
165pub enum Happening {
166    /// The character died.
167    Died,
168    /// `healme` brought it back.
169    Healed,

172    /// The walk stopped, told to or done.

174}

176/// The file name of a run: `survey-<character>-<stamp>.jsonl`.
177#[must_use]
178pub fn file_name(server: &str, character: &str, stamp: &str) -> String {
179    format!(
180        "survey-{}-{}-{stamp}.jsonl",
181        server.to_lowercase(),
182        character.to_lowercase()
183    )
184}

186/// A stamp for a run starting now: `YYYYMMDD-HHMMSS`.
187#[must_use]
188pub fn stamp_now() -> String {
189    let now = jiff::Zoned::now();
190    format!(
191        "{:04}{:02}{:02}-{:02}{:02}{:02}",
192        now.year(),
193        now.month(),
194        now.day(),
195        now.hour(),
196        now.minute(),
197        now.second()
198    )
199}

207/// The newest run file of `character` on `server` in `dir`, if any: one
208/// named for both, or one named for the character alone, as runs were
209/// named before the game was in the name. The newest is by the stamp,
210/// which ends either name.
211#[must_use]
212pub fn newest(dir: &Path, server: &str, character: &str) -> Option<PathBuf> {
213    let character = character.to_lowercase();
214    let prefixes = [
215        format!("survey-{}-{character}-", server.to_lowercase()),
216        format!("survey-{character}-"),
217    ];
218    let stamp_of = |path: &Path| -> Option<String> {
219        let name = path.file_name()?.to_str()?;
220        let stem = name.strip_suffix(".jsonl")?;
221        prefixes
222            .iter()
223            .find_map(|prefix| stem.strip_prefix(prefix.as_str()))
224            .filter(|stamp| stamp.len() == 15)
225            .map(str::to_owned)
226    };
227    let mut found: Vec<(String, PathBuf)> = std::fs::read_dir(dir)
228        .ok()?
229        .filter_map(Result::ok)
230        .map(|entry| entry.path())
231        .filter_map(|path| stamp_of(&path).map(|stamp| (stamp, path)))
232        .collect();
233    found.sort();
234    found.pop().map(|(_, path)| path)
235}

237/// The records of a run file, in order. A line that does not parse is
238/// skipped: the log is append-only, and a cut-off last line is expected.
239///
240/// # Errors
241///
242/// The file cannot be read.
243pub fn read(path: &Path) -> Result<Vec<Record>, String> {
244    let file = File::open(path).map_err(|why| format!("{}: {why}", path.display()))?;
245    Ok(BufReader::new(file)
246        .lines()
247        .map_while(Result::ok)
248        .filter_map(|line| serde_json::from_str(&line).ok())
249        .collect())
250}

252/// A run's file, written to as the walk goes.
253pub struct Log {
254    file: File,
255    path: PathBuf,
256    run: String,
257    server: String,
258    character: String,
259}
260
261impl Log {
262    /// Open `path` for appending, or create it, for a run stamped `run`.
263    ///
264    /// # Errors
265    ///
266    /// The file cannot be opened.
267    pub fn open(
268        path: PathBuf,
269        run: String,
270        server: String,
271        character: String,
272    ) -> Result<Self, String> {
273        // A file cut off mid-record ends without a newline: the next record
274        // would join the fragment and be lost with it.
275        let ends_open = std::fs::read(&path)
276            .ok()
277            .is_some_and(|bytes| bytes.last().is_some_and(|&last| last != b'\n'));
278        let mut file = OpenOptions::new()
279            .create(true)
280            .append(true)
281            .open(&path)
282            .map_err(|why| format!("{}: {why}", path.display()))?;
283        if ends_open {
284            file.write_all(b"\n")
285                .map_err(|why| format!("{}: {why}", path.display()))?;
286        }
287        Ok(Self {
288            file,
289            path,
290            run,
291            server,
292            character,
293        })
294    }
295
296    /// Where it is written.
297    #[must_use]
298    pub fn path(&self) -> &Path {
299        &self.path
300    }
301
302    /// The run's stamp.
303    #[must_use]
304    pub fn run(&self) -> &str {
305        &self.run
306    }
307
308    /// Append one record, flushed at once: a stop or a crash loses nothing
309    /// before it.
310    ///
311    /// # Errors
312    ///
313    /// The file cannot be written.
314    pub fn write(&mut self, body: Body) -> Result<(), String> {
315        let record = Record {
316            run: self.run.clone(),
317            t: time_now(),
318            server: self.server.clone(),
319            character: self.character.clone(),
320            body,
321        };
322        let mut line = serde_json::to_string(&record).map_err(|why| why.to_string())?;
323        line.push('\n');
324        self.file
325            .write_all(line.as_bytes())
326            .and_then(|()| self.file.flush())
327            .map_err(|why| format!("{}: {why}", self.path.display()))
328    }
329}

331#[cfg(test)]
332mod tests {
333    use super::*;
334
335    /// Every kind of record round-trips through a file, and a cut-off last
336    /// line is skipped rather than failing the read.
337    #[test]
338    fn records_round_trip_and_a_cut_line_is_skipped() {
339        let dir = std::env::temp_dir().join(format!("hydra-survey-log-{}", std::process::id()));
340        let _ = std::fs::remove_dir_all(&dir);
341        std::fs::create_dir_all(&dir).expect("made");
342        let path = dir.join(file_name("GST", "Nisugi", "20261008-120000"));
343        let mut log = Log::open(
344            path.clone(),
345            "20261008-120000".to_owned(),
346            "GST".to_owned(),
347            "Nisugi".to_owned(),
348        )
349        .expect("opened");
350        let room = RoomRecord {
351            sid: "s1".to_owned(),
352            uid: Some("2022626".to_owned()),
353            title: "[Emberthorn Refuge, Glade]".to_owned(),
354            desc: "Sunlight filters through.".to_owned(),
355            paths: "Obvious paths: out".to_owned(),
356            compass: vec!["out".to_owned()],
357            roommeta: BTreeMap::from([("inside".to_owned(), "0".to_owned())]),
358            via: None,
359            pos: Some([0, 0, 0]),
360            animals: vec!["lop-eared sable tigrina".to_owned()],
361            ..RoomRecord::default()
362        };
363        log.write(Body::Room(Box::new(room.clone())))
364            .expect("written");
365        log.write(Body::Verb(VerbRecord {
366            sid: "s1".to_owned(),
367            verb: "splash".to_owned(),
368            lines: vec!["You just splashed yourself!".to_owned()],
369            rt: None,
370            cast_rt: None,
371            prompt_closed: true,
372        }))
373        .expect("written");
374        log.write(Body::Edge(EdgeRecord {
375            from: "s1".to_owned(),
376            cmd: "out".to_owned(),
377            to: Some("s2".to_owned()),
378            rt: None,
379            dir: Some("out".to_owned()),
380            result: EdgeResult::Arrived,
381            lines: Vec::new(),
382        }))
383        .expect("written");
384        log.write(Body::Event(EventRecord {
385            sid: "s2".to_owned(),
386            what: Happening::Stopped,
387        }))
388        .expect("written");
389        drop(log);
390        let mut file = OpenOptions::new().append(true).open(&path).expect("opened");
391        file.write_all(b"{\"kind\":\"room\",\"run\":\"x")
392            .expect("cut");
393        drop(file);
394        let records = read(&path).expect("read");
395        assert_eq!(records.len(), 4, "{records:#?}");
396        assert_eq!(records[0].body, Body::Room(Box::new(room)));
397        assert_eq!(records[0].server, "GST");
398        assert!(matches!(&records[2].body, Body::Edge(edge) if edge.result == EdgeResult::Arrived));
399        let text = std::fs::read_to_string(&path).expect("text");
400        assert!(text.contains("\"kind\":\"edge\""), "{text}");
401        assert!(text.contains("\"char\":\"Nisugi\""), "{text}");
402        assert_eq!(newest(&dir, "GST", "nisugi"), Some(path.clone()));
403        // An older run, named for the character alone, is found too; the
404        // newest of either naming is by its stamp.
405        let older = dir.join("survey-nisugi-20261001-090000.jsonl");
406        std::fs::write(&older, "").expect("older");
407        assert_eq!(newest(&dir, "GST", "nisugi"), Some(path.clone()));
408        let newer = dir.join("survey-nisugi-20261009-090000.jsonl");
409        std::fs::write(&newer, "").expect("newer");
410        assert_eq!(newest(&dir, "GST", "Nisugi"), Some(newer.clone()));
411        // Another game's run, named for its game, is not this game's; one
412        // named for the character alone is anyone's, as runs once were.
413        let other = dir.join("survey-gs3-nisugi-20261010-090000.jsonl");
414        std::fs::write(&other, "").expect("other");
415        assert_eq!(newest(&dir, "GST", "nisugi"), Some(newer));
416        assert_eq!(newest(&dir, "GS3", "nisugi"), Some(other));
417
418        // Opened again after the cut, the next record starts on a line of
419        // its own, so it is read back.
420        let mut log = Log::open(
421            path.clone(),
422            "20261008-120000".to_owned(),
423            "GST".to_owned(),
424            "Nisugi".to_owned(),
425        )
426        .expect("opened again");
427        log.write(Body::Event(EventRecord {
428            sid: "s3".to_owned(),
429            what: Happening::Stopped,
430        }))
431        .expect("written");
432        drop(log);
433        let records = read(&path).expect("read");
434        assert_eq!(records.len(), 5, "{records:#?}");
435        let _ = std::fs::remove_dir_all(&dir);
436    }

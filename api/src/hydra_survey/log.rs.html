1//! The survey's log: JSON Lines, append-only, one file per run, every
2//! record stamped with the run, the time, the server and the character.
3//!
4//! Four kinds of record. A `room` is written once, on the first arrival:
5//! what the game states about it, and what the verbs asked there said in
6//! their own fields. A `verb` is one line asked and its reply verbatim. An
7//! `edge` is one exit walked: where it led, or how it was refused. An
8//! `event` is a death, a heal, a resume or a stop. The log is the survey's
9//! only memory: a run resumes by reading it back.
10
11use std::collections::BTreeMap;
12use std::fs::{File, OpenOptions};
13use std::io::{BufRead, BufReader, Write};
14use std::path::{Path, PathBuf};
15
16use serde::{Deserialize, Serialize};
17
18/// A survey's own id for a room: `s1`, `s2`, ...
19pub type Sid = String;
20
21/// One record, with what every record carries.
22#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
23pub struct Record {
24    /// The run's stamp, the file's.
25    pub run: String,
26    /// When, as the wall clock reads, RFC 3339.
27    pub t: String,
28    /// The game instance: `GST`, `GS3`.
29    pub server: String,
30    /// The character.
31    #[serde(rename = "char")]
32    pub character: String,
33    /// The record itself.
34    #[serde(flatten)]
35    pub body: Body,
36}
37
38/// What a record says, by its kind.
39#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
40#[serde(tag = "kind", rename_all = "lowercase")]
41pub enum Body {
42    /// A room on its first arrival.
43    Room(Box<RoomRecord>),
44    /// A verb asked in a room, and its reply.
45    Verb(VerbRecord),
46    /// An exit walked from a room.
47    Edge(EdgeRecord),
48    /// Something that happened to the walk.
49    Event(EventRecord),
50}
51
52/// A room as first seen.
53#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
54pub struct RoomRecord {
55    /// The survey's id.
56    pub sid: Sid,
57    /// `<nav rm=>` exactly as given; `None` for a bare `<nav/>`.
58    pub uid: Option<String>,
59    /// The room's title, as the window subtitle gives it.
60    pub title: String,
61    /// The `room desc` component only.
62    pub desc: String,
63    /// The "Obvious exits/paths" line.
64    pub paths: String,
65    /// The compass's direction tokens.
66    pub compass: Vec<String>,
67    /// Every command the survey will try from here: the compass, then
68    /// `go <noun>` for what `anal room`, SENSE and the room's objects
69    /// name as ways out.
70    pub exits: Vec<String>,
71    /// `<roommeta>` as a raw attribute map.
72    pub roommeta: BTreeMap<String, String>,
73    /// How the room was reached; `None` for the starting room.
74    pub via: Option<Via>,
75    /// A first guess at where it lies on the survey's own grid.
76    pub pos: Option<[i32; 3]>,
77    /// From a Ranger's SENSE: the climate as the game words it.
78    pub climate: Option<String>,
79    /// From a Ranger's SENSE: the terrain as the game words it.
80    pub terrain: Option<String>,
81    /// From a Ranger's SENSE: the animals whose signs are about.
82    pub animals: Vec<String>,
83    /// From a Ranger's SENSE: what circles overhead.
84    pub overhead: Option<String>,
85    /// From a Ranger's SENSE: the exit it tacks on.
86    pub sensed_exit: Option<String>,
87    /// From a Ranger's TRACK AREA: the creatures that have been about.
88    #[serde(default)]
89    pub tracked: Vec<String>,
90    /// From LOCATION: the name the game guesses.
91    pub location: Option<String>,
92}

94/// The exit a room was reached by.

96pub struct Via {

103/// A verb asked, and its reply.
104#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
105pub struct VerbRecord {
106    /// Where.
107    pub sid: Sid,
108    /// The line as sent.
109    pub verb: String,
110    /// The reply, verbatim.
111    pub lines: Vec<String>,
112    /// Seconds of roundtime it cost.
113    pub rt: Option<u32>,
114    /// Seconds of cast roundtime it cost.
115    pub cast_rt: Option<u32>,
116    /// A prompt closed the reply; `false` when the wait ran out first.
117    pub prompt_closed: bool,

120/// An exit walked.
121#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
122pub struct EdgeRecord {
123    /// The room left.
124    pub from: Sid,
125    /// The command sent.
126    pub cmd: String,
127    /// The room arrived in, the same room when nothing happened, or `None`
128    /// when refused.
129    pub to: Option<Sid>,
130    /// Seconds of roundtime it cost.
131    pub rt: Option<u32>,
132    /// The compass token when `cmd` is a direction.
133    pub dir: Option<String>,
134    /// How it went.
135    pub result: EdgeResult,
136    /// The game's lines: the refusal, or what came with the arrival.
137    pub lines: Vec<String>,
138}

140/// How an exit went.
141#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
142#[serde(rename_all = "lowercase")]
143pub enum EdgeResult {
144    /// The character arrived somewhere.
145    #[default]
146    Arrived,
147    /// The game refused it.
148    Failed,
149    /// Nothing happened and nothing was refused.
150    Stayed,
151}

153/// Something that happened to the walk.
154#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
155pub struct EventRecord {
156    /// Where.
157    pub sid: Sid,
158    /// What.
159    pub what: Happening,
160}

162/// What can happen to the walk.
163#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
164#[serde(rename_all = "lowercase")]
165pub enum Happening {
166    /// The character died.
167    Died,
168    /// `healme` brought it back.
169    Healed,
170    /// A run was resumed from this log.
171    Resumed,
172    /// The walk stopped, told to or done.
173    Stopped,
174}

176/// The file name of a run: `survey-<character>-<stamp>.jsonl`.
177#[must_use]
178pub fn file_name(character: &str, stamp: &str) -> String {
179    format!("survey-{}-{stamp}.jsonl", character.to_lowercase())
180}

182/// A stamp for a run starting now: `YYYYMMDD-HHMMSS`.
183#[must_use]
184pub fn stamp_now() -> String {
185    let now = jiff::Zoned::now();
186    format!(
187        "{:04}{:02}{:02}-{:02}{:02}{:02}",
188        now.year(),
189        now.month(),
190        now.day(),
191        now.hour(),
192        now.minute(),
193        now.second()
194    )
195}

197/// The time on a record, now.
198#[must_use]
199pub fn time_now() -> String {
200    jiff::Timestamp::now().to_string()
201}

203/// The newest run file of `character` in `dir`, if any.
204#[must_use]
205pub fn newest(dir: &Path, character: &str) -> Option<PathBuf> {
206    let prefix = format!("survey-{}-", character.to_lowercase());
207    let mut found: Vec<PathBuf> = std::fs::read_dir(dir)
208        .ok()?
209        .filter_map(Result::ok)
210        .map(|entry| entry.path())
211        .filter(|path| {
212            path.file_name()
213                .and_then(|name| name.to_str())
214                .is_some_and(|name| {
215                    name.starts_with(&prefix)
216                        && std::path::Path::new(name)
217                            .extension()
218                            .is_some_and(|ext| ext.eq_ignore_ascii_case("jsonl"))
219                })
220        })
221        .collect();
222    found.sort();
223    found.pop()
224}

226/// The records of a run file, in order. A line that does not parse is
227/// skipped: the log is append-only, and a cut-off last line is expected.
228///
229/// # Errors
230///
231/// The file cannot be read.
232pub fn read(path: &Path) -> Result<Vec<Record>, String> {
233    let file = File::open(path).map_err(|why| format!("{}: {why}", path.display()))?;
234    Ok(BufReader::new(file)
235        .lines()
236        .map_while(Result::ok)
237        .filter_map(|line| serde_json::from_str(&line).ok())
238        .collect())
239}

241/// A run's file, written to as the walk goes.
242pub struct Log {
243    file: File,
244    path: PathBuf,
245    run: String,
246    server: String,
247    character: String,
248}
249
250impl Log {
251    /// Open `path` for appending, or create it, for a run stamped `run`.
252    ///
253    /// # Errors
254    ///
255    /// The file cannot be opened.
256    pub fn open(
257        path: PathBuf,
258        run: String,
259        server: String,
260        character: String,
261    ) -> Result<Self, String> {
262        let file = OpenOptions::new()
263            .create(true)
264            .append(true)
265            .open(&path)
266            .map_err(|why| format!("{}: {why}", path.display()))?;
267        Ok(Self {
268            file,
269            path,
270            run,
271            server,
272            character,
273        })
274    }
275
276    /// Where it is written.
277    #[must_use]
278    pub fn path(&self) -> &Path {
279        &self.path
280    }
281
282    /// The run's stamp.
283    #[must_use]
284    pub fn run(&self) -> &str {
285        &self.run
286    }
287
288    /// Append one record, flushed at once: a stop or a crash loses nothing
289    /// before it.
290    ///
291    /// # Errors
292    ///
293    /// The file cannot be written.
294    pub fn write(&mut self, body: Body) -> Result<(), String> {
295        let record = Record {
296            run: self.run.clone(),
297            t: time_now(),
298            server: self.server.clone(),
299            character: self.character.clone(),
300            body,
301        };
302        let mut line = serde_json::to_string(&record).map_err(|why| why.to_string())?;
303        line.push('\n');
304        self.file
305            .write_all(line.as_bytes())
306            .and_then(|()| self.file.flush())
307            .map_err(|why| format!("{}: {why}", self.path.display()))
308    }
309}

311#[cfg(test)]
312mod tests {
313    use super::*;
314
315    /// Every kind of record round-trips through a file, and a cut-off last
316    /// line is skipped rather than failing the read.
317    #[test]
318    fn records_round_trip_and_a_cut_line_is_skipped() {
319        let dir = std::env::temp_dir().join(format!("hydra-survey-log-{}", std::process::id()));
320        let _ = std::fs::remove_dir_all(&dir);
321        std::fs::create_dir_all(&dir).expect("made");
322        let path = dir.join(file_name("Nisugi", "20261008-120000"));
323        let mut log = Log::open(
324            path.clone(),
325            "20261008-120000".to_owned(),
326            "GST".to_owned(),
327            "Nisugi".to_owned(),
328        )
329        .expect("opened");
330        let room = RoomRecord {
331            sid: "s1".to_owned(),
332            uid: Some("2022626".to_owned()),
333            title: "[Emberthorn Refuge, Glade]".to_owned(),
334            desc: "Sunlight filters through.".to_owned(),
335            paths: "Obvious paths: out".to_owned(),
336            compass: vec!["out".to_owned()],
337            roommeta: BTreeMap::from([("inside".to_owned(), "0".to_owned())]),
338            via: None,
339            pos: Some([0, 0, 0]),
340            animals: vec!["lop-eared sable tigrina".to_owned()],
341            ..RoomRecord::default()
342        };
343        log.write(Body::Room(Box::new(room.clone())))
344            .expect("written");
345        log.write(Body::Verb(VerbRecord {
346            sid: "s1".to_owned(),
347            verb: "splash".to_owned(),
348            lines: vec!["You just splashed yourself!".to_owned()],
349            rt: None,
350            cast_rt: None,
351            prompt_closed: true,
352        }))
353        .expect("written");
354        log.write(Body::Edge(EdgeRecord {
355            from: "s1".to_owned(),
356            cmd: "out".to_owned(),
357            to: Some("s2".to_owned()),
358            rt: None,
359            dir: Some("out".to_owned()),
360            result: EdgeResult::Arrived,
361            lines: Vec::new(),
362        }))
363        .expect("written");
364        log.write(Body::Event(EventRecord {
365            sid: "s2".to_owned(),
366            what: Happening::Stopped,
367        }))
368        .expect("written");
369        drop(log);
370        let mut file = OpenOptions::new().append(true).open(&path).expect("opened");
371        file.write_all(b"{\"kind\":\"room\",\"run\":\"x")
372            .expect("cut");
373        drop(file);
374        let records = read(&path).expect("read");
375        assert_eq!(records.len(), 4, "{records:#?}");
376        assert_eq!(records[0].body, Body::Room(Box::new(room)));
377        assert_eq!(records[0].server, "GST");
378        assert!(matches!(&records[2].body, Body::Edge(edge) if edge.result == EdgeResult::Arrived));
379        let text = std::fs::read_to_string(&path).expect("text");
380        assert!(text.contains("\"kind\":\"edge\""), "{text}");
381        assert!(text.contains("\"char\":\"Nisugi\""), "{text}");
382        assert_eq!(newest(&dir, "nisugi"), Some(path));
383        let _ = std::fs::remove_dir_all(&dir);
384    }

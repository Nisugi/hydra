1//! What Hydra's walker can cross, and the count of what it cannot yet: the
2//! finish line for travel, which is done when the walker can cross every
3//! exit the map has.
4//!
5//! Every exit is crossed by one kind of crossing -- a plain command, a pass
6//! through an urchin hub, a list of steps (each step one kind of action), or
7//! a named routine -- and priced by one kind of cost. [`kinds`] names them,
8//! by the map's own names for its variants, so a kind a later map adds is
9//! counted the day it appears, unrun. [`RUNS`] is what the walker runs, and
10//! an exit is walked only when every kind it needs is in it.
11//!
12//! An exit the map gives **no cost** is not a route anywhere -- no walker
13//! is meant to take it -- and is not counted.
14//!
15//! The test at the foot of this file holds the built-in map's count of
16//! exits not yet walked **exactly**: a change that makes the walker run more
17//! must lower it, and nothing may raise it. Travel is done at zero.
18

106/// The kinds `exit` needs: its crossing's (each step's action, for a list of
107/// steps), and its cost's.
108#[must_use]
109pub fn kinds(exit: &Exit) -> Vec<String> {
110    let mut kinds = match &exit.crossing {
111        Crossing::Command(_) => vec!["command".to_owned()],
112        Crossing::PassThrough(_) => vec!["pass".to_owned()],
113        Crossing::Steps(steps) => steps
114            .iter()
115            .map(|step| format!("step {}", variant(&step.action)))
116            .collect(),
117        Crossing::Routine(routine) => vec![format!("routine {}", routine_name(routine))],
118        Crossing::Unported(_) => vec!["unported".to_owned()],
119        Crossing::Unknown(name) => vec![format!("unknown {name}")],
120    };
121    kinds.push(match &exit.cost {
122        Some(cost) => format!("cost {}", variant(cost)),
123        None => "no cost".to_owned(),
124    });
125    kinds
126}
127
128/// Whether the walker can cross `exit` now.
129#[must_use]
130pub fn walked(exit: &Exit) -> bool {
131    kinds(exit).iter().all(|kind| RUNS.contains(&kind.as_str()))
132}
133
134/// A routine's name; a puzzle's and an errand's are their own.
135fn routine_name(routine: &Routine) -> String {
136    match routine {
137        Routine::Puzzle { puzzle } => format!("Puzzle {}", variant(puzzle)),
138        Routine::Errand { errand } => format!("Errand {}", variant(errand)),
139        other => variant(other),
140    }
141}

143/// The name of `value`'s variant, as its `Debug` begins. Formatting stops
144/// where the name does: a route search asks this of every exit it looks
145/// at, and a step's whole `Debug` -- its commands and its strings -- would be
146/// made only to be thrown away.
147fn variant(value: &impl std::fmt::Debug) -> String {
148    /// Keeps what is written while it is a name, and stops the formatting
149    /// at the first character that is not.
150    struct Name(String);
151    impl std::fmt::Write for Name {
152        fn write_str(&mut self, text: &str) -> std::fmt::Result {
153            for c in text.chars() {
154                if !c.is_ascii_alphanumeric() {
155                    return Err(std::fmt::Error);
156                }
157                self.0.push(c);
158            }
159            Ok(())
160        }
161    }
162    let mut name = Name(String::new());
163    // Stopped on purpose at the name's end: the error is the stop.
164    let _ = std::fmt::Write::write_fmt(&mut name, format_args!("{value:?}"));
165    name.0
166}

168/// Every exit of `map` that is a route somewhere: one with a cost.
169fn routes(map: &Map) -> impl Iterator<Item = &Exit> {
170    map.rooms()
171        .iter()
172        .flat_map(|room| &room.exits)
173        .filter(|exit| exit.cost.is_some())
174}

176/// Every kind the map uses: how many exits need it, and whether it is run.
177#[must_use]
178pub fn census(map: &Map) -> BTreeMap<String, (usize, bool)> {
179    let mut census = BTreeMap::new();
180    for exit in routes(map) {
181        let mut needs = kinds(exit);
182        needs.sort_unstable();
183        needs.dedup();
184        for kind in needs {
185            let runs = RUNS.contains(&kind.as_str());
186            census.entry(kind).or_insert((0, runs)).0 += 1;
187        }
188    }
189    census
190}
191
192/// How many of the map's exits the walker cannot cross yet, of how many.
193#[must_use]
194pub fn unwalked(map: &Map) -> (usize, usize) {
195    let (mut not, mut all) = (0, 0);
196    for exit in routes(map) {
197        all += 1;
198        if !walked(exit) {
199            not += 1;
200        }
201    }
202    (not, all)
203}
204
205/// The census as lines a person reads: what is left, most-used first.
206#[must_use]
207pub fn report(map: &Map) -> Vec<String> {
208    let (not, all) = unwalked(map);
209    let mut left: Vec<(String, usize)> = census(map)
210        .into_iter()
211        .filter(|(_, (_, runs))| !runs)
212        .map(|(kind, (exits, _))| (kind, exits))
213        .collect();
214    left.sort_by(|a, b| b.1.cmp(&a.1).then_with(|| a.0.cmp(&b.0)));
215    let mut lines = vec![format!(
216        "Travel walks {} of the map's {all} exits; {not} are left, needing {} kinds:",
217        all - not,
218        left.len()
219    )];
220    lines.extend(
221        left.into_iter()
222            .map(|(kind, exits)| format!("  {exits:>6}  {kind}")),
223    );
224    lines
225}

228mod tests {

233    /// The ratchet: the built-in map's exits not yet walked, held exactly.
234    /// Lower it when the walker runs more; it may never rise.
235    const UNWALKED: usize = 0;
236
237    #[test]
238    fn the_count_of_what_is_not_walked_yet() {
239        let map = cena_map::binary::decode(cena_gs_map::GS_MAP).expect("the built-in map");
240        let (not, all) = unwalked(&map);
241        assert_eq!(not, UNWALKED, "{}", report(&map).join("\n"));
242        assert!(all >= not);
243    }
244
245    /// Nothing in the map is beyond the vocabulary: every exit is a kind the
246    /// walker can be taught.
247    #[test]
248    fn the_map_holds_nothing_unported() {
249        let map = cena_map::binary::decode(cena_gs_map::GS_MAP).expect("the built-in map");
250        let strays: Vec<String> = census(&map)
251            .into_keys()
252            .filter(|kind| kind.starts_with("unported") || kind.starts_with("unknown"))
253            .collect();
254        assert_eq!(strays, Vec::<String>::new());
255    }
256
257    #[test]
258    fn an_exit_is_walked_only_when_all_it_needs_is_run() {
259        let exit = |crossing| Exit {
260            to: cena_map::RoomId(1),
261            kind: cena_map::ExitKind::Cardinal,
262            crossing,
263            cost: Some(Cost::Fixed(0.2)),
264            dirto: None,
265        };
266        let north = exit(Crossing::Command("north".into()));
267        assert_eq!(kinds(&north), ["command", "cost Fixed"]);
268        assert!(walked(&north), "a plain exit is walked");
269        let mut free = north.clone();
270        free.cost = None;
271        assert_eq!(kinds(&free), ["command", "no cost"]);
272        assert!(!walked(&free));
273    }

1//! What Hydra's walker can cross, and the count of what it cannot yet: the
2//! finish line for travel, which is done when the walker can cross every
3//! exit the map has.
4//!
5//! Every exit is crossed by one kind of crossing -- a plain command, a pass
6//! through an urchin hub, a list of steps (each step one kind of action), or
7//! a named routine -- and priced by one kind of cost. [`kinds`] names them,
8//! by the map's own names for its variants, so a kind a later map adds is
9//! counted the day it appears, unrun. [`RUNS`] is what the walker runs, and
10//! an exit is walked only when every kind it needs is in it.
11//!
12//! An exit the map gives **no cost** is not a route anywhere -- no walker
13//! is meant to take it -- and is not counted.
14//!
15//! The test at the foot of this file holds the built-in map's count of
16//! exits not yet walked **exactly**: a change that makes the walker run more
17//! must lower it, and nothing may raise it. Travel is done at zero.

106/// The kinds `exit` needs: its crossing's (each step's action, for a list of
107/// steps), and its cost's.
108#[must_use]
109pub fn kinds(exit: &Exit) -> Vec<String> {
110    let mut kinds = match &exit.crossing {
111        Crossing::Command(_) => vec!["command".to_owned()],
112        Crossing::PassThrough(_) => vec!["pass".to_owned()],
113        Crossing::Steps(steps) => steps
114            .iter()
115            .map(|step| format!("step {}", variant(&step.action)))
116            .collect(),
117        Crossing::Routine(routine) => vec![format!("routine {}", routine_name(routine))],
118        Crossing::Unported(_) => vec!["unported".to_owned()],
119        Crossing::Unknown(name) => vec![format!("unknown {name}")],
120    };
121    kinds.push(match &exit.cost {
122        Some(cost) => format!("cost {}", variant(cost)),
123        None => "no cost".to_owned(),
124    });
125    kinds
126}

128/// Whether the walker can cross `exit` now: each kind it needs is one of
129/// [`RUNS`]. The same answer as [`kinds`] checked against [`RUNS`], without
130/// a string made: a route search asks this of every exit it prices.
131#[must_use]
132pub fn walked(exit: &Exit) -> bool {
133    let crossing = match &exit.crossing {
134        Crossing::Command(_) => RUNS.contains(&"command"),
135        Crossing::PassThrough(_) => RUNS.contains(&"pass"),
136        Crossing::Steps(steps) => steps.iter().all(|step| runs("step ", &step.action)),
137        Crossing::Routine(Routine::Puzzle { puzzle }) => runs("routine Puzzle ", puzzle),
138        Crossing::Routine(Routine::Errand { errand }) => runs("routine Errand ", errand),
139        Crossing::Routine(other) => runs("routine ", other),
140        Crossing::Unported(_) | Crossing::Unknown(_) => false,
141    };
142    crossing
143        && match &exit.cost {
144            Some(cost) => runs("cost ", cost),
145            None => RUNS.contains(&"no cost"),
146        }
147}

149/// Whether `prefix` and the name of `value`'s variant make one of [`RUNS`].
150fn runs(prefix: &str, value: &impl std::fmt::Debug) -> bool {
151    RUNS.iter()
152        .filter_map(|run| run.strip_prefix(prefix))
153        .any(|name| variant_is(value, name))
154}
155
156/// Whether the name of `value`'s variant is `name`: its `Debug` read as far
157/// as the name goes and compared as it is written, nothing kept.
158fn variant_is(value: &impl std::fmt::Debug, name: &str) -> bool {
159    /// Compares what is written to the name left to match, and stops the
160    /// formatting at the name's end or the first mismatch.
161    struct Matching<'a> {
162        left: &'a str,
163        matched: bool,
164    }
165    impl std::fmt::Write for Matching<'_> {
166        fn write_str(&mut self, text: &str) -> std::fmt::Result {
167            for c in text.chars() {
168                if !c.is_ascii_alphanumeric() {
169                    // The name's end: a match when nothing of it is left.
170                    self.matched = self.left.is_empty();
171                    return Err(std::fmt::Error);
172                }
173                match self.left.strip_prefix(c) {
174                    Some(rest) => self.left = rest,
175                    None => return Err(std::fmt::Error),
176                }
177            }
178            Ok(())
179        }
180    }
181    let mut matching = Matching {
182        left: name,
183        matched: false,
184    };
185    // Stopped on purpose at the name's end: the error is the stop. A value
186    // whose Debug is the bare name alone ends with nothing written after
187    // it, and the name left decides.
188    match std::fmt::Write::write_fmt(&mut matching, format_args!("{value:?}")) {
189        Ok(()) => matching.left.is_empty(),
190        Err(_) => matching.matched,
191    }
192}

194/// A routine's name; a puzzle's and an errand's are their own.
195fn routine_name(routine: &Routine) -> String {
196    match routine {
197        Routine::Puzzle { puzzle } => format!("Puzzle {}", variant(puzzle)),
198        Routine::Errand { errand } => format!("Errand {}", variant(errand)),
199        other => variant(other),
200    }
201}

203/// The name of `value`'s variant, as its `Debug` begins. Formatting stops
204/// where the name does: a route search asks this of every exit it looks
205/// at, and a step's whole `Debug` -- its commands and its strings -- would be
206/// made only to be thrown away.
207fn variant(value: &impl std::fmt::Debug) -> String {
208    /// Keeps what is written while it is a name, and stops the formatting
209    /// at the first character that is not.
210    struct Name(String);
211    impl std::fmt::Write for Name {
212        fn write_str(&mut self, text: &str) -> std::fmt::Result {
213            for c in text.chars() {
214                if !c.is_ascii_alphanumeric() {
215                    return Err(std::fmt::Error);
216                }
217                self.0.push(c);
218            }
219            Ok(())
220        }
221    }
222    let mut name = Name(String::new());
223    // Stopped on purpose at the name's end: the error is the stop.
224    let _ = std::fmt::Write::write_fmt(&mut name, format_args!("{value:?}"));
225    name.0
226}

228/// Every exit of `map` that is a route somewhere: one with a cost.
229fn routes(map: &Map) -> impl Iterator<Item = &Exit> {
230    map.rooms()
231        .iter()
232        .flat_map(|room| &room.exits)
233        .filter(|exit| exit.cost.is_some())
234}

236/// Every kind the map uses: how many exits need it, and whether it is run.
237#[must_use]
238pub fn census(map: &Map) -> BTreeMap<String, (usize, bool)> {
239    let mut census = BTreeMap::new();
240    for exit in routes(map) {
241        let mut needs = kinds(exit);
242        needs.sort_unstable();
243        needs.dedup();
244        for kind in needs {
245            let runs = RUNS.contains(&kind.as_str());
246            census.entry(kind).or_insert((0, runs)).0 += 1;
247        }
248    }
249    census
250}

252/// How many of the map's exits the walker cannot cross yet, of how many.
253#[must_use]
254pub fn unwalked(map: &Map) -> (usize, usize) {
255    let (mut not, mut all) = (0, 0);
256    for exit in routes(map) {
257        all += 1;
258        if !walked(exit) {
259            not += 1;
260        }
261    }
262    (not, all)
263}
264
265/// The census as lines a person reads: what is left, most-used first.
266#[must_use]
267pub fn report(map: &Map) -> Vec<String> {
268    let (not, all) = unwalked(map);
269    let mut left: Vec<(String, usize)> = census(map)
270        .into_iter()
271        .filter(|(_, (_, runs))| !runs)
272        .map(|(kind, (exits, _))| (kind, exits))
273        .collect();
274    left.sort_by(|a, b| b.1.cmp(&a.1).then_with(|| a.0.cmp(&b.0)));
275    let mut lines = vec![format!(
276        "Travel walks {} of the map's {all} exits; {not} are left, needing {} kinds:",
277        all - not,
278        left.len()
279    )];
280    lines.extend(
281        left.into_iter()
282            .map(|(kind, exits)| format!("  {exits:>6}  {kind}")),
283    );
284    lines
285}

293    /// The ratchet: the built-in map's exits not yet walked, held exactly.
294    /// Lower it when the walker runs more; it may never rise.
295    const UNWALKED: usize = 0;
296
297    #[test]
298    fn the_count_of_what_is_not_walked_yet() {
299        let map = cena_map::binary::decode(cena_gs_map::GS_MAP).expect("the built-in map");
300        let (not, all) = unwalked(&map);
301        assert_eq!(not, UNWALKED, "{}", report(&map).join("\n"));
302        assert!(all >= not);
303    }
304
305    /// `walked`, which makes no string, agrees with `kinds` checked
306    /// against `RUNS` for every exit of the built-in map, and for one of
307    /// each kind the map does not have.
308    #[test]
309    fn walked_agrees_with_the_kinds_named() {
310        let map = cena_map::binary::decode(cena_gs_map::GS_MAP).expect("the built-in map");
311        let by_kinds = |exit: &Exit| kinds(exit).iter().all(|kind| RUNS.contains(&kind.as_str()));
312        for exit in map.rooms().iter().flat_map(|room| &room.exits) {
313            assert_eq!(walked(exit), by_kinds(exit), "{:?}", kinds(exit));
314        }
315        let made = |crossing, cost| Exit {
316            to: cena_map::RoomId(1),
317            kind: cena_map::ExitKind::Cardinal,
318            crossing,
319            cost,
320            dirto: None,
321        };
322        for exit in [
323            made(
324                Crossing::Unknown("ghost".to_owned()),
325                Some(Cost::Fixed(0.2)),
326            ),
327            made(Crossing::Command("north".to_owned()), None),
328        ] {
329            assert_eq!(walked(&exit), by_kinds(&exit), "{:?}", kinds(&exit));
330        }
331    }
332
333    /// Nothing in the map is beyond the vocabulary: every exit is a kind the
334    /// walker can be taught.
335    #[test]
336    fn the_map_holds_nothing_unported() {
337        let map = cena_map::binary::decode(cena_gs_map::GS_MAP).expect("the built-in map");
338        let strays: Vec<String> = census(&map)
339            .into_keys()
340            .filter(|kind| kind.starts_with("unported") || kind.starts_with("unknown"))
341            .collect();
342        assert_eq!(strays, Vec::<String>::new());
343    }

345    #[test]
346    fn an_exit_is_walked_only_when_all_it_needs_is_run() {
347        let exit = |crossing| Exit {
348            to: cena_map::RoomId(1),
349            kind: cena_map::ExitKind::Cardinal,
350            crossing,
351            cost: Some(Cost::Fixed(0.2)),
352            dirto: None,
353        };
354        let north = exit(Crossing::Command("north".into()));
355        assert_eq!(kinds(&north), ["command", "cost Fixed"]);
356        assert!(walked(&north), "a plain exit is walked");
357        let mut free = north.clone();
358        free.cost = None;
359        assert_eq!(kinds(&free), ["command", "no cost"]);
360        assert!(!walked(&free));
361    }

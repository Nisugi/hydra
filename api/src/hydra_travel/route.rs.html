1//! The route, shown and not walked (`.route2`), and where a target is.
2//!
3//! Showing the way rather than walking it is how a wrong or missing exit in
4//! the map is found: the table says, room by room, what would be taken and
5//! what was not.
6//!
7//! # One pricing for every route
8//!
9//! The way shown, the way walked and "the nearest bank" are all priced by
10//! [`pricing`]: an exit is open only when **this character can take it** --
11//! the map's own questions about it, a soulstone, a profession, a setting --
12//! **and Hydra walks it** ([`crate::coverage::walked`]). Finding the nearest
13//! anything, or any route at all, tries only paths this character can
14//! actually take. What Hydra does not walk yet is shut, and said so under
15//! the room it leaves.
16//!
17//! # What the table shows
18//!
19//! A scripted exit is shown as its steps, and a scripted cost is counted:
20//! a crossing is a list of steps and a cost is a question asked of the
21//! walker, so both are shown and the total is the real one. And it says
22//! **what it did not take**: at each room on the way, every exit shut to
23//! *this* walker, with the reason -- a fact the game has not told yet, a
24//! fact that says no, or no cost in the map.
25//!
26//! **Pure**: a map and a walker in, rows out. [`table`] lays them out as
27//! text for a notice; the minimap takes the rooms.
28

37/// One exit taken.
38#[derive(Debug, Clone, PartialEq)]
39pub struct Leg {
40    /// The room this exit arrives in.
41    pub to: RoomId,
42    /// What crosses it: a command, or the steps in order.
43    pub how: String,
44    /// What this walker pays for it, in seconds.
45    pub seconds: f64,
46    /// And for everything up to and including it.
47    pub so_far: f64,
48    /// The arrival room's first title in the map.
49    pub title: Option<String>,
50    /// The arrival room's location name in the map.
51    pub location: Option<String>,
52    /// The exits out of the room this leg leaves that are shut to this
53    /// walker: what explains a long way round.
54    pub shut: Vec<Shut>,
55}

57/// An exit this walker cannot take.
58#[derive(Debug, Clone, PartialEq, Eq)]
59pub struct Shut {
60    /// The room it leads to.
61    pub to: RoomId,
62    /// What would cross it.
63    pub how: String,
64    /// Why this walker cannot.
65    pub why: ShutWhy,
66}

68/// Why an exit is shut to this walker.
69#[derive(Debug, Clone, PartialEq, Eq)]
70pub enum ShutWhy {
71    /// Its price hangs on a fact nobody has said yet: the question, as the
72    /// map asks it.
73    NotKnown(String),
74    /// Every fact is known, and they say no.
75    SaysNo,
76    /// This character could take it; Hydra does not walk it yet.
77    NotBuiltYet,
78    /// The map gives it no cost: not a route for anyone.
79    NoCost,
80}

82/// How every route is priced for `walker`: what it costs *them*, of the
83/// exits they can take and Hydra walks (module docs). The trip adds only
84/// what it has banned on the way.
85pub fn pricing(walker: &Walker) -> impl Fn(&Room, &Exit) -> Option<f64> + '_ {
86    let theirs = priced_for(walker);
87    move |room, exit| walked(exit).then(|| theirs(room, exit)).flatten()
88}

90/// The room `state` puts the character in, as the minimap finds it, with
91/// the travel file's last room offered to break a tie between rooms that
92/// read alike -- a hint, never a fact; `None` when the map cannot tell.
93#[must_use]
94pub fn here(map: &Map, state: &hydra_model::GameState, last_room: Option<u32>) -> Option<RoomId> {
95    let whence = last_room.map_or(cena_map::Origin::Nowhere, |room| {
96        cena_map::Origin::Still(RoomId(room))
97    });
98    hydra_map::room_of(map, state, whence)
99}

101/// What a player may mean by `what`, standing in `from`: a room, or rooms
102/// to choose between.
103#[derive(Debug, Clone, PartialEq, Eq)]
104pub enum Meant {
105    /// This room.
106    Room(RoomId),
107    /// Words that fit several rooms: these, listed for the player to choose.
108    Several(Vec<RoomId>),
109    /// Nowhere the map knows, or nowhere this walker can reach.
110    Nowhere,
111}
112
113/// The room a player means by `what`, standing in `from`, tried in this
114/// order, the most exact first:
115///
116/// 1. **A number**: the map's own id, or the game's with a `u` (`u7120`)
117///    when it names one room.
118/// 2. **A name the player saved** (`targets`), exactly, not minding case.
119///    Several rooms mean the nearest.
120/// 3. **`guild` and `guild shop`** mean this character's: `wizard guild`.
121/// 4. **A tag** (`bank`, `town`): the nearest room that has it.
122/// 5. **A name the player saved that the words begin**: `bank a` for
123///    `bank alley`, after the tag, so a saved `bank alley` does not take
124///    `bank` from the banks.
125/// 6. **Words from a room's title or description** ([`described`]).
126///
127/// "Nearest" is by what *this walker* would pay to get there, so a bard's
128/// nearest bank may not be a warrior's.
129#[must_use]
130pub fn destination(
131    map: &Map,
132    walker: &Walker,
133    from: RoomId,
134    what: &str,
135    targets: &BTreeMap<String, Vec<u32>>,
136) -> Meant {
137    let what = what.trim();
138    let one = |room: Option<RoomId>| room.map_or(Meant::Nowhere, Meant::Room);
139    if let Some(uid) = what
140        .strip_prefix(['u', 'U'])
141        .and_then(|digits| digits.parse().ok())
142    {
143        return match map.ids_for_uid(Uid(uid)) {
144            [room] => Meant::Room(*room),
145            [] => Meant::Nowhere,
146            several => Meant::Several(several.to_vec()),
147        };
148    }
149    if let Ok(id) = what.parse() {
150        return one(map.room(RoomId(id)).map(|room| room.id));
151    }
152    let nearest = |rooms: &[RoomId]| {
153        map.routes(from, Target::Nearest(rooms), pricing(walker))
154            .reached()
155    };
156    let lower = what.to_lowercase();
157    // A saved name's rooms: those the map has, the nearest meant.
158    let saved = |(_, rooms): (&String, &Vec<u32>)| {
159        let rooms: Vec<RoomId> = rooms
160            .iter()
161            .filter_map(|id| map.room(RoomId(*id)).map(|room| room.id))
162            .collect();
163        one(nearest(&rooms))
164    };
165    if let Some(named) = targets
166        .iter()
167        .find(|(name, _)| name.to_lowercase() == lower)
168    {
169        return saved(named);
170    }
171    let tag = match (lower.as_str(), walker.profession.as_deref()) {
172        ("guild" | "guild shop", Some(profession)) => {
173            format!("{} {lower}", profession.to_lowercase())
174        }
175        _ => lower.clone(),
176    };
177    let tagged: Vec<RoomId> = map
178        .rooms()
179        .iter()
180        .filter(|room| room.tags.iter().any(|has| has.eq_ignore_ascii_case(&tag)))
181        .map(|room| room.id)
182        .collect();
183    if !tagged.is_empty() {
184        return one(nearest(&tagged));
185    }
186    if let Some(named) = targets
187        .iter()
188        .find(|(name, _)| name.to_lowercase().starts_with(&lower))
189    {
190        return saved(named);
191    }
192    match described(map, what)[..] {
193        [] => Meant::Nowhere,
194        [only] => Meant::Room(only),
195        ref several => Meant::Several(several.to_vec()),
196    }
197}

199/// The rooms a player may mean by words from a title or a description:
200/// any title or description that holds the text, without regard to case --
201/// or, of a description, any one of the pieces its full stops divide it
202/// into, so a description pasted with its `...` still finds its room.
203#[must_use]
204pub fn described(map: &Map, what: &str) -> Vec<RoomId> {
205    let whole = what.trim().to_lowercase();
206    if whole.is_empty() {
207        return Vec::new();
208    }
209    let pieces: Vec<&str> = whole
210        .trim_end_matches('.')
211        .split('.')
212        .map(str::trim)
213        .filter(|piece| !piece.is_empty())
214        .collect();
215    let holds = |texts: &[String], wanted: &str| {
216        texts
217            .iter()
218            .any(|text| text.to_lowercase().contains(wanted))
219    };
220    map.rooms()
221        .iter()
222        .filter(|room| {
223            holds(&room.title, &whole)
224                || holds(&room.description, &whole)
225                || pieces.iter().any(|piece| holds(&room.description, piece))
226        })
227        .map(|room| room.id)
228        .collect()
229}

231/// The route from `from` to `goal` as this walker would walk it now; `None`
232/// when there is no way.
233#[must_use]
234pub fn itinerary(map: &Map, walker: &Walker, from: RoomId, goal: RoomId) -> Option<Vec<Leg>> {
235    let price = pricing(walker);
236    let routes = map.routes(from, Target::Room(goal), &price);
237    let exits = routes.exits_to(goal)?;
238    let mut legs = Vec::with_capacity(exits.len());
239    let mut so_far = 0.0;
240    for (here, index, to) in exits {
241        let room = map.room(here)?;
242        let exit = room.exits.get(index)?;
243        let seconds = price(room, exit)?;
244        so_far += seconds;
245        let arrived = map.room(to);
246        legs.push(Leg {
247            to,
248            how: how(&exit.crossing),
249            seconds,
250            so_far,
251            title: arrived.and_then(|room| room.title.first().cloned()),
252            location: arrived.and_then(|room| room.location.clone()),
253            shut: shut_from(room, walker, &price),
254        });
255    }
256    Some(legs)
257}

259fn shut_from(
260    room: &Room,
261    walker: &Walker,
262    price: impl Fn(&Room, &Exit) -> Option<f64>,
263) -> Vec<Shut> {
264    room.exits
265        .iter()
266        .filter(|exit| price(room, exit).is_none())
267        .map(|exit| Shut {
268            to: exit.to,
269            how: how(&exit.crossing),
270            why: why_shut(exit, walker),
271        })
272        .collect()
273}
274
275fn why_shut(exit: &Exit, walker: &Walker) -> ShutWhy {
276    let Some(cost) = &exit.cost else {
277        return ShutWhy::NoCost;
278    };
279    if cost.price(walker).is_some() {
280        // Payable, and still shut: Hydra does not walk it yet.
281        return ShutWhy::NotBuiltYet;
282    }
283    let asked: Vec<&Cond> = match cost {
284        Cost::Gated { when, .. } => vec![when],
285        Cost::Ladder { ladder, .. } => ladder.iter().map(|rung| &rung.when).collect(),
286        _ => Vec::new(),
287    };
288    // Only a question with no answer has an unknown part worth naming: one
289    // whose answer is no is no, whatever else it would ask.
290    let unanswered = asked.into_iter().filter(|cond| cond.ask(walker).is_none());
291    match unanswered
292        .into_iter()
293        .find_map(|cond| not_known(cond, walker))
294    {
295        Some(question) => ShutWhy::NotKnown(format!("{question:?}")),
296        None if matches!(cost, Cost::Table { .. }) => ShutWhy::NotKnown(format!("{cost:?}")),
297        None => ShutWhy::SaysNo,
298    }
299}
300
301/// The first question inside `cond` that has no answer yet.
302fn not_known<'c>(cond: &'c Cond, walker: &Walker) -> Option<&'c Cond> {
303    match cond {
304        Cond::All(parts) | Cond::Any(parts) => {
305            parts.iter().find_map(|part| not_known(part, walker))
306        }
307        Cond::Not(inner) | Cond::Otherwise(inner) => not_known(inner, walker),
308        leaf => leaf.ask(walker).is_none().then_some(leaf),
309    }
310}

312fn how(crossing: &Crossing) -> String {
313    match crossing {
314        Crossing::Command(command) => command.clone(),
315        Crossing::Steps(steps) => steps
316            .iter()
317            .map(|step| format!("{:?}", step.action))
318            .collect::<Vec<_>>()
319            .join("; "),
320        Crossing::Routine(routine) => format!("({})", named(routine)),
321        Crossing::PassThrough(_) => "(through the urchins)".to_owned(),
322        other => format!("{other:?}"),
323    }
324}

326/// A routine, as a person would say it: `MinotaurMaze` is "minotaur maze".
327fn named(routine: &Routine) -> String {
328    let name = match routine {
329        Routine::Puzzle { puzzle } => format!("{puzzle:?}"),
330        Routine::Errand { errand } => format!("{errand:?}"),
331        other => format!("{other:?}"),
332    };
333    let mut said = String::new();
334    for letter in name.chars().take_while(char::is_ascii_alphabetic) {
335        if letter.is_ascii_uppercase() && !said.is_empty() {
336            said.push(' ');
337        }
338        said.push(letter.to_ascii_lowercase());
339    }
340    said
341}

343/// The rows as a table of text, in columns: step, trip time, this step's
344/// time, the move, the room, its name and where it is. A rule every ten
345/// rows; and the shut exits under the row that left them behind.
346#[must_use]
347pub fn table(map: &Map, from: RoomId, legs: &[Leg]) -> Vec<String> {
348    const MOVE_WIDTH: usize = 28;
349    let clip = |text: &str| {
350        if text.chars().count() > MOVE_WIDTH {
351            let kept: String = text.chars().take(MOVE_WIDTH - 3).collect();
352            format!("{kept}...")
353        } else {
354            text.to_owned()
355        }
356    };
357    let row = |step: usize, trip: &str, time: &str, how: &str, room: RoomId, name: &str| {
358        format!(
359            "{step:>4}  {trip:>8}  {time:>6}  {:<MOVE_WIDTH$}  {:>6}  {name}",
360            clip(how),
361            room.0
362        )
363    };
364    let name_of = |title: Option<&String>, location: Option<&String>| match (title, location) {
365        (Some(title), Some(location)) => format!("{title} ({location})"),
366        (Some(title), None) => title.clone(),
367        (None, _) => String::new(),
368    };
369    let start = map.room(from);
370    let mut lines = vec![
371        format!(
372            "STEP      TRIP    TIME  {:<MOVE_WIDTH$}    ROOM  NAME",
373            "MOVE"
374        ),
375        row(
376            0,
377            "",
378            "",
379            "",
380            from,
381            &name_of(
382                start.and_then(|room| room.title.first()),
383                start.and_then(|room| room.location.as_ref()),
384            ),
385        ),
386    ];
387    for (index, leg) in legs.iter().enumerate() {
388        let step = index + 1;
389        if step % 10 == 0 {
390            lines.push("-".repeat(72));
391        }
392        for shut in &leg.shut {
393            let why = match &shut.why {
394                ShutWhy::NotKnown(question) => format!("not known yet: {question}"),
395                ShutWhy::SaysNo => "not for this character".to_owned(),
396                ShutWhy::NotBuiltYet => "Hydra does not walk this yet".to_owned(),
397                ShutWhy::NoCost => "the map gives it no cost".to_owned(),
398            };
399            lines.push(format!(
400                "      shut: {} -> {}: {why}",
401                clip(&shut.how),
402                shut.to.0
403            ));
404        }
405        lines.push(row(
406            step,
407            &format!("{:.1}", leg.so_far),
408            &format!("{:.1}", leg.seconds),
409            &leg.how,
410            leg.to,
411            &name_of(leg.title.as_ref(), leg.location.as_ref()),
412        ));
413    }
414    lines
415}

466/// The places to go (`.go2 targets`): town by town, the places the map
467/// names near each -- **which are targets already**, with nothing saved.
468/// Under each town, one line for each of [`PLACES`] that has a room whose
469/// nearest town it is. Then the names the player chose (`named`).
470///
471/// "Nearest" is by what this walker would pay. A town tagged `closed` is
472/// not listed.
473#[must_use]
474pub fn places(map: &Map, walker: &Walker, named: &BTreeMap<String, Vec<u32>>) -> Vec<String> {
475    let has = |room: &Room, tag: &str| room.tags.iter().any(|is| is == tag);
476    let towns: Vec<&Room> = map
477        .rooms()
478        .iter()
479        .filter(|room| has(room, "town") && !has(room, "closed"))
480        .collect();
481    let town_ids: Vec<RoomId> = towns.iter().map(|town| town.id).collect();
482    let mut near: BTreeMap<RoomId, BTreeMap<&str, String>> = BTreeMap::new();
483    for tag in PLACES {
484        for room in map.rooms().iter().filter(|room| has(room, tag)) {
485            let nearest = map
486                .routes(room.id, Target::Nearest(&town_ids), pricing(walker))
487                .reached();
488            let Some(town) = nearest else { continue };
489            // The first room of a kind stands for it.
490            near.entry(town).or_default().entry(tag).or_insert_with(|| {
491                let title = room.title.first().map_or("", String::as_str);
492                let title = title.trim_start_matches('[').trim_end_matches(']');
493                format!(" - {tag:<17} {title:<34} - {:>5}", room.id.0)
494            });
495        }
496    }
497    let rule = "-".repeat(63);
498    let mut lines = Vec::new();
499    for town in towns {
500        // `the free port of Solhaven` is Solhaven.
501        let at = town.location.as_deref().unwrap_or("");
502        let at = at.find(char::is_uppercase).map_or(at, |from| &at[from..]);
503        lines.push(rule.clone());
504        lines.push(format!(" - {:<17} {at:<34} - {:>5}", "town", town.id.0));
505        lines.push(rule.clone());
506        lines.extend(
507            near.remove(&town.id)
508                .into_iter()
509                .flat_map(BTreeMap::into_values),
510        );
511        lines.push(String::new());
512    }
513    if !named.is_empty() {
514        lines.push("names you chose:".to_owned());
515        lines.extend(
516            named
517                .iter()
518                .map(|(name, rooms)| format!("   {name:<15} = {rooms:?}")),
519        );
520    }
521    lines

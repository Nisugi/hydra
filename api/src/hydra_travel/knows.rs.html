1//! Which spells the walker knows, and which it can pay for now, read from
2//! the model. **Pure.**
3//!
4//! # Known
5//!
6//! A spell's circle is its number less its last two digits, and it is known
7//! when the ranks in that circle -- **capped at the character's level** --
8//! reach those last two digits. A society's powers (97, 98, 99) go by rank in
9//! *that* society. 1700 is known to the five pure professions and nothing
10//! else in its circle is known to anyone.
11//!
12//! # Affordable
13//!
14//! Enough stamina, mana and spirit for what the table says it costs, and
15//! spirit must be left with **one to spare**. Not counted, and said so rather
16//! than guessed at: the Council of Light signs that raise the spirit to
17//! spare, Mental Acuity (a Monk feat that pays for 1200s in stamina), and the
18//! `Overexerted` debuff. Each makes a spell *less* affordable than this says,
19//! so the cost of the gap is a cast the game refuses, which the trip already
20//! recovers from.
21
22use std::collections::HashSet;
23
24use hydra_model::spells::{self, Spell};
25use hydra_model::{GameState, Society};

27/// The spell circles as `skills` prints them, by circle number, less the
28/// circles no character trains.
29const CIRCLES: [(u16, &str); 12] = [
30    (1, "Minor Spiritual"),
31    (2, "Major Spiritual"),
32    (3, "Cleric"),
33    (4, "Minor Elemental"),
34    (5, "Major Elemental"),
35    (6, "Ranger"),
36    (7, "Sorcerer"),
37    (9, "Wizard"),
38    (10, "Bard"),
39    (11, "Empath"),
40    (12, "Minor Mental"),
41    (16, "Paladin"),
42];
43
44/// The professions to whom 1700 is known.
45const PURES: [&str; 5] = ["Wizard", "Cleric", "Empath", "Sorcerer", "Savant"];
46
47/// Whether `skills` has ever been read: a skill row or a circle row.
48pub(crate) fn skills_listed(state: &GameState) -> bool {
49    let skills = &state.character.skills;
50    skills.known_count() > 0 || skills.circles().next().is_some()
51}
52
53/// The circle names as the map's questions spell them, with the ranks in
54/// each: `major elemental` -> 25.
55pub(crate) fn circle_ranks(state: &GameState) -> impl Iterator<Item = (String, u32)> + '_ {
56    let skills = &state.character.skills;
57    CIRCLES.iter().map(move |(_, name)| {
58        let ranks = skills.circle(name).unwrap_or(0);
59        (name.to_lowercase(), u32::from(ranks))
60    })
61}
62
63/// The ranks that decide whether `spell` is known, or `None` when nothing
64/// could make it so.
65fn deciding_ranks(state: &GameState, spell: &Spell, level: u32) -> Option<u32> {
66    let circle = spell.number / 100;
67    let society = |which: Society| {
68        let standing = &state.character.standing;
69        (standing.society == Some(Some(which))).then(|| standing.society_rank.map_or(0, u32::from))
70    };
71    match circle {
72        97 => society(Society::GuardiansOfSunfist),
73        98 => society(Society::OrderOfVoln),
74        99 => society(Society::CouncilOfLight),
75        _ => {
76            let (_, name) = CIRCLES.iter().find(|(number, _)| *number == circle)?;
77            let ranks = state.character.skills.circle(name).unwrap_or(0);
78            Some(u32::from(ranks).min(level))
79        }
80    }
81}

83/// Every spell and society power the character knows, by name. `None` until
84/// the skills have been listed: before that, nothing is known *about* them.
85pub(crate) fn known_spells(state: &GameState, level: Option<u32>) -> Option<HashSet<String>> {
86    if !skills_listed(state) {
87        return None;
88    }
89    let level = level?;
90    let pure = state
91        .character
92        .identity
93        .profession
94        .as_deref()
95        .is_some_and(|is| PURES.contains(&is));
96    Some(
97        spells::all()
98            .filter(|spell| {
99                if spell.number == 1700 {
100                    return pure;
101                }
102                deciding_ranks(state, spell, level)
103                    .is_some_and(|ranks| u32::from(spell.number % 100) <= ranks)
104            })
105            .map(|spell| spell.name.clone())
106            .collect(),
107    )
108}

110/// Of `known`, the ones the character can pay for now. `None` until all three
111/// gauges have been seen.
112pub(crate) fn affordable_spells(
113    state: &GameState,
114    known: &HashSet<String>,
115) -> Option<HashSet<String>> {
116    let have = |id: &str| state.vital(id)?.current;
117    let (mana, stamina, spirit) = (have("mana")?, have("stamina")?, have("spirit")?);
118    Some(
119        spells::all()
120            .filter(|spell| known.contains(&spell.name))
121            .filter(|spell| {
122                let costs = |cost: Option<u16>| i32::from(cost.unwrap_or(0));
123                let spirit_cost = costs(spell.spirit);
124                costs(spell.stamina) <= stamina
125                    && costs(spell.mana) <= mana
126                    // One to spare: a spell that takes the last spirit kills.
127                    && (spirit_cost == 0 || spirit_cost < spirit)
128            })
129            .map(|spell| spell.name.clone())
130            .collect(),
131    )
132}

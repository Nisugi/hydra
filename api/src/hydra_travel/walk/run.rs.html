1//! Crossing one exit: a flat list of guarded steps, run one at a time.
2//!
3//! A plain exit is the one-step list `[move "north"]`, so there is a single
4//! way across and move recovery (`super::recovery`) lives inside it.
5//!
6//! # Two ways of sending
7//!
8//! - A **move** ([`Mover`]) must change the room, and has the whole ladder
9//!   of remedies behind it.
10//! - An **exchange** ([`Exchange`]) is a command and the game's answer to it:
11//!   `open door`, one `search` of many. It is answered by the next prompt or
12//!   by a room change, waits out a `...wait N`, and is otherwise not second
13//!   guessed. Every loop in the vocabulary -- *until the room changes*,
14//!   *while there is an exit east*, *until the game says so* -- is exchanges
15//!   and a question between them.
16//!
17//! # A guard is asked when its step is reached
18//!
19//! Not when the walk is planned (`cena_map::Step`): `go north` *when Water
20//! Walking is up* must see what the `cast` before it did. The trip lends the
21//! walker two facts only it knows -- the room, and whether this crossing has
22//! moved it yet -- and asks again at every step.
23//!
24//! # What is left to whoever drives
25//!
26//! A [`Deed`] is something the trip cannot spell as a command without facts
27//! it should not hold: what is in each hand and where it stows, how this
28//! character casts a spell, where memories are kept. The driver does the
29//! deed, then ticks again.
30

38/// Turns of any loop: a stop for loops the map leaves unbounded, and a cap
39/// on those it bounds.
40pub const MAX_TURNS: u32 = 50;
41/// How long a step will wait for a line, an arrival or a fact. A ship's
42/// voyage is the longest thing waited for; this is a stop, not an estimate.
43pub const MAX_WAIT_MS: u64 = 30 * 60 * 1000;
44
45/// How long a spell cast to carry the walker ([`Deed::CastAt`]) is given to
46/// do it, from the moment the cast is done. A move's own
47/// [`STEP_TIMEOUT_MS`]: Phase moves the caster as part of the cast's answer.
48///
49/// Not the generic arrival wait, bounded only by [`MAX_WAIT_MS`] -- thirty
50/// minutes, sized for a ship's voyage -- or a cast that fizzled, or was never
51/// made for want of mana, would stand the walker in place for half an hour
52/// before the exit was given up.
53pub const CARRIED_WITHIN_MS: u64 = STEP_TIMEOUT_MS;

55/// Something the driver must do for the trip, to completion, before it ticks
56/// again. See the module docs.
57#[derive(Debug, Clone, PartialEq, Eq)]
58pub enum Deed {
59    /// Put away what is in the hands, remembering what went where.
60    EmptyHands,
61    /// Take back what [`Deed::EmptyHands`] put away, last first.
62    FillHands,
63    /// Cast this, by name, and wait for it to land.
64    Cast(String),
65    /// Cast `.0` at `.1`, waiting for the mana and casting again if hindered.
66    CastAt(String, String),
67    /// Take this stance, remembering the one held.
68    Stance(String),
69    /// Go back to the stance [`Deed::Stance`] replaced.
70    RestoreStance,
71    /// Write a memory down, in the character's travel file.
72    Remember(String, String),
73    /// Strike one out.
74    Forget(String),
75    /// Wait for whoever follows the walker to arrive.
76    AwaitFollowers,
77    /// Speak this language, remembering the one spoken.
78    Speak(String),
79    /// Go back to the language [`Deed::Speak`] replaced.
80    RestoreSpeech,
81    /// `get my <this>`, remembering what came out of where. A walker with
82    /// none says so with [`super::Trip::could_not`].
83    TakeOut(String),
84    /// Put what [`Deed::TakeOut`] took back where it came from.
85    PutBack,
86}

88/// What one tick of a crossing comes to.
89#[derive(Debug, Clone, PartialEq, Eq)]
90pub(super) enum Out {
91    Send(String),
92    Deed(Deed),
93    Hold,
94    /// Every step is done.
95    Done,
96    /// A step asked for the walk to be planned again from here.
97    Replan,
98    /// This exit cannot be crossed on this trip. `ban`: never offer it again.
99    /// `wrong`: the game said there is no such way.
100    GiveUp {
101        ban: bool,
102        wrong: bool,
103    },
104    /// The game answered the move with lines no rule knows, and the walker
105    /// never moved. Perhaps a refusal of a kind not yet named; perhaps the
106    /// room's own talk, while the move is only slow. Given up, and the exit
107    /// banned the second time.
108    Doubted,
109    /// The game said a move landed where no room change could show it.
110    Believed(RoomId),
111}

132/// When a loop of exchanges stops.
133#[derive(Debug, Clone, PartialEq, Eq)]
134enum Until {
135    /// The walker is no longer in the room the loop began in.
136    RoomChanges,
137    /// The walker is at the exit's destination.
138    There,
139    /// The question no longer holds.
140    NoLonger(Cond),
141    /// The game has said one of these, in answer to the round's last command.
142    Said(Vec<String>),
143}

145/// Which command a turn of a loop sends.

147enum Pick {
148    /// Each in order, round and round.

157#[derive(Debug, Clone, PartialEq, Eq)]

169/// What a step is doing, when it takes more than one tick.
170#[derive(Debug, Clone, PartialEq, Eq)]
171enum Job {
172    Moving(Mover),
173    /// A move that may well not work: answered, not insisted on.
174    Trying(Exchange),
175    /// Commands still to exchange, in order, then the step is done.
176    Round(Vec<String>, Option<Exchange>),
177    /// Moves still to make, in order.
178    Moves(Vec<String>, Option<Mover>),
179    Loop(Looping),
180    /// Nothing sent: waiting on the clock, a line, the room, or a fact.
181    Pause(u64),
182    Line(Vec<String>, u64),
183    Arrival(RoomId, u64),
184    /// A spell cast to carry the walker ([`Deed::CastAt`]): the room it was
185    /// cast in, and when the cast was done -- `None` until the tick after the
186    /// deed, because the deed's own mana wait is not the carrying's.
187    Carried(RoomId, Option<u64>),
188    Fact(Cond, u64),
189    /// A deed was handed to the driver; the step is done when it ticks again.
190    Deeded,
191    /// `look trail`, sent: listening for the word that follows `.1`.
192    Asking(Exchange, String),
193    /// `inquire`, sent: listening for the numbered line that names `.1`.
194    Inquiring(Exchange, String),
195}

197/// One exit being crossed.
198#[derive(Debug, Clone, PartialEq, Eq)]
199pub(super) struct Run {
200    pub from: RoomId,
201    /// The room the exit leaves in the map: `from`, unless the path passed

203    pub leaving: RoomId,

231    pub fn new(from: RoomId, leaving: RoomId, expected: RoomId, steps: Vec<Step>) -> Run {
232        Run {
233            from,
234            leaving,
235            expected,
236            exit: None,
237            steps,
238            at: 0,
239            job: None,
240            moved: false,
241            owes: Owes::default(),
242            told: None,
243        }

246    pub fn tick(&mut self, tick: &Tick<'_>) -> Out {
247        if tick.here != self.from {
248            self.moved = true;
249        }
250        // The two facts only the trip knows, lent for the guards.
251        let mut walker = tick.walker.clone();
252        walker.room = Some(tick.here.0);
253        walker.still_here = Some(!self.moved);
254        loop {
255            if let Some(job) = self.job.take() {
256                match self.work(job, tick, &walker) {
257                    Some(out) => return out,
258                    None => self.at += 1,
259                }
260            }
261            let Some(step) = self.steps.get(self.at).cloned() else {
262                return Out::Done;
263            };
264            if step.when.as_ref().is_some_and(|when| !when.holds(&walker)) {
265                self.at += 1;
266                continue;
267            }
268            match self.begin(step.action, tick, &walker) {
269                Begun::Out(out) => return out,
270                Begun::Skip => self.at += 1,
271            }
272        }
273    }

275    /// Carry a job on. `None` when it has finished.
276    fn work(&mut self, job: Job, tick: &Tick<'_>, walker: &Walker) -> Option<Out> {
277        match job {
278            Job::Moving(mover) => self.moving(mover, tick),
279            Job::Trying(mut exchange) => {
280                let out = exchange.tick(tick, tick.here != self.from)?;
281                self.job = Some(Job::Trying(exchange));
282                Some(out)
283            }
284            Job::Round(mut rest, exchange) => {
285                if let Some(mut exchange) = exchange
286                    && let Some(out) = exchange.tick(tick, false)
287                {
288                    self.job = Some(Job::Round(rest, Some(exchange)));
289                    return Some(out);
290                }
291                if rest.is_empty() {
292                    return None;
293                }
294                let command = rest.remove(0);
295                self.job = Some(Job::Round(rest, Some(exchange_of(&command, tick.ms))));
296                Some(Out::Send(command))
297            }
298            Job::Moves(mut rest, mover) => {
299                if let Some(mut mover) = mover {
300                    match mover.tick(tick) {
301                        Moved::Not(out) => {
302                            self.job = Some(Job::Moves(rest, Some(mover)));
303                            return Some(out);
304                        }
305                        Moved::Landed { .. } => self.moved = true,
306                    }
307                }
308                if rest.is_empty() {
309                    return None;
310                }
311                let command = rest.remove(0);
312                let mover = Mover::new(tick.here, &command, tick.ms);
313                self.job = Some(Job::Moves(rest, Some(mover)));
314                Some(Out::Send(command))
315            }
316            Job::Loop(looped) => self.turn(looped, tick, walker),
317            Job::Pause(until) => (tick.ms < until).then(|| {
318                self.job = Some(Job::Pause(until));
319                Out::Hold
320            }),
321            Job::Line(lines, since) => (!said(tick.lines, &lines))
322                .then(|| self.wait(Job::Line(lines, since), since, tick.ms)),
323            Job::Arrival(began_in, since) => (tick.here == began_in && tick.here != self.expected)
324                .then(|| self.wait(Job::Arrival(began_in, since), since, tick.ms)),
325            Job::Carried(began_in, since) => {
326                // The driver could not cast it: no mana in ten minutes, or
327                // hindered every time. Nothing is coming to carry the walker.
328                if tick.could_not {
329                    return Some(Out::GiveUp {
330                        ban: true,
331                        wrong: false,
332                    });
333                }
334                if tick.here != began_in || tick.here == self.expected {
335                    return None;
336                }
337                let since = since.unwrap_or(tick.ms);
338                if tick.ms.saturating_sub(since) >= CARRIED_WITHIN_MS {
339                    return Some(Out::GiveUp {
340                        ban: true,
341                        wrong: false,
342                    });
343                }
344                self.job = Some(Job::Carried(began_in, Some(since)));
345                Some(Out::Hold)
346            }
347            Job::Fact(cond, since) => {
348                (!cond.holds(walker)).then(|| self.wait(Job::Fact(cond, since), since, tick.ms))
349            }
350            // A deed the driver could not do: a walker with no key. The
351            // crossing fails here, before anything is unlocked.
352            Job::Deeded => tick.could_not.then_some(Out::GiveUp {
353                ban: true,
354                wrong: false,
355            }),
356            Job::Asking(mut exchange, after) => {
357                if let Some(word) = replies::word_after(tick.lines, &after) {
358                    self.told = Some(word);
359                    return None;
360                }
361                let out = exchange.tick(tick, false)?;
362                self.job = Some(Job::Asking(exchange, after));
363                Some(out)
364            }
365            Job::Inquiring(mut exchange, named) => {
366                if let Some(number) = replies::numbered(tick.lines, &named) {
367                    let command = format!("order {number}");
368                    self.job = Some(Job::Round(Vec::new(), Some(exchange_of(&command, tick.ms))));
369                    return Some(Out::Send(command));
370                }
371                // Answered, and the list does not name it: no caravan goes there.
372                let Some(out) = exchange.tick(tick, false) else {
373                    return Some(Out::GiveUp {
374                        ban: true,
375                        wrong: false,
376                    });
377                };
378                self.job = Some(Job::Inquiring(exchange, named));
379                Some(out)
380            }
381        }
382    }

384    fn moving(&mut self, mut mover: Mover, tick: &Tick<'_>) -> Option<Out> {
385        match mover.tick(tick) {
386            Moved::Not(out) => {
387                self.owes.hands |= mover.hands_emptied;
388                self.job = Some(Job::Moving(mover));
389                Some(out)
390            }
391            Moved::Landed { believed } => {
392                self.moved = true;
393                if believed {
394                    return Some(Out::Believed(self.expected));
395                }
396                if mover.hands_emptied {
397                    // Hands emptied for the move are filled once it lands.
398                    self.owes.hands = false;
399                    self.job = Some(Job::Deeded);
400                    return Some(Out::Deed(Deed::FillHands));
401                }
402                None
403            }
404        }
405    }

407    /// One turn of a loop: is it over, and if not, what is sent next.
408    fn turn(&mut self, mut looped: Looping, tick: &Tick<'_>, walker: &Walker) -> Option<Out> {
409        let mut heard = false;
410        if let Some(mut exchange) = looped.exchange.take() {
411            if let Until::Said(lines) = &looped.until {
412                heard = said(tick.lines, lines);
413            }
414            if !heard && let Some(out) = exchange.tick(tick, tick.here != looped.began_in) {
415                looped.exchange = Some(exchange);
416                self.job = Some(Job::Loop(looped));
417                return Some(out);
418            }
419        }
420        let over = match &looped.until {
421            Until::RoomChanges => tick.here != looped.began_in,
422            Until::There => tick.here == self.expected,
423            Until::NoLonger(cond) => !cond.holds(walker),
424            Until::Said(_) => heard,
425        };
426        if over {
427            return None;
428        }
429        if looped.turns >= looped.tries.unwrap_or(MAX_TURNS).min(MAX_TURNS) {
430            // A bounded try that ran out is no failure: the walk goes
431            // on to the next step. An unbounded one that hit the walker's own
432            // stop has met something it cannot do.
433            return looped.tries.is_none().then_some(Out::GiveUp {
434                ban: true,
435                wrong: false,
436            });
437        }
438        if nothing_to_pick(&looped.pick, walker) {
439            // A list left empty, or a room with no way out: the step cannot
440            // be done, and the exit is given up.
441            return Some(Out::GiveUp {
442                ban: true,
443                wrong: false,
444            });
445        }
446        // The room's ways out not known yet -- just arrived -- is a wait,
447        // counted as a turn so it is not for ever; skipping the step would
448        // go on from the wrong room.
449        if looped.pick == Pick::AnyExit && walker.exits.is_none() {
450            looped.turns += 1;
451            self.job = Some(Job::Loop(looped));
452            return Some(Out::Hold);
453        }
454        let command = choose(&looped, tick, walker)?;
455        if looped.pick == Pick::AnyExit {
456            looped.last_way = Some(command.clone());
457        }
458        looped.turns += 1;
459        looped.exchange = Some(exchange_of(&command, tick.ms));
460        self.job = Some(Job::Loop(looped));
461        Some(Out::Send(command))
462    }

464    /// Keep waiting, unless it has gone on too long.
465    fn wait(&mut self, job: Job, since: u64, ms: u64) -> Out {
466        if ms.saturating_sub(since) >= MAX_WAIT_MS {
467            return Out::GiveUp {
468                ban: true,
469                wrong: false,
470            };
471        }
472        self.job = Some(job);
473        Out::Hold
474    }

476    fn begin(&mut self, action: Action, tick: &Tick<'_>, walker: &Walker) -> Begun {
477        let ms = tick.ms;
478        if let Some(deed) = self.deed_for(&action) {
479            self.job = Some(Job::Deeded);
480            return Begun::Out(Out::Deed(deed));
481        }
482        let job = if let Some((pick, until, tries)) = loop_for(&action) {
483            Job::Loop(Looping {
484                pick,
485                until,
486                began_in: tick.here,
487                turns: 0,
488                tries,
489                exchange: None,
490                last_way: None,
491            })
492        } else {
493            match action {
494                Action::Move(command) => {
495                    let Some(command) = filled(&command, walker, self.told.as_deref()) else {
496                        return Begun::cannot();
497                    };
498                    self.job = Some(Job::Moving(Mover::new(tick.here, &command, ms)));
499                    return Begun::Out(Out::Send(command));
500                }
501                Action::TryMove(command) | Action::Put(command) => {
502                    let Some(command) = filled(&command, walker, self.told.as_deref()) else {
503                        return Begun::cannot();
504                    };
505                    self.job = Some(Job::Trying(exchange_of(&command, ms)));
506                    return Begun::Out(Out::Send(command));
507                }
508                Action::MoveByAnyExitBut(not_this) => {
509                    let way = walker
510                        .exits
511                        .as_ref()
512                        .and_then(|exits| exits.iter().find(|exit| **exit != not_this));
513                    let Some(way) = way.cloned() else {
514                        return Begun::cannot();
515                    };
516                    self.job = Some(Job::Moving(Mover::new(tick.here, &way, ms)));
517                    return Begun::Out(Out::Send(way));
518                }
519                Action::Moves(commands) => Job::Moves(commands, None),
520                Action::Round(commands) => Job::Round(commands, None),
521                Action::MovesFromSetting(name) => {
522                    let Some(listed) = walker.settings.get(&name) else {
523                        return Begun::cannot();
524                    };
525                    let commands = listed.split(',').map(|c| c.trim().to_owned()).collect();
526                    Job::Round(commands, None)
527                }
528                Action::Ask(command, after) => {
529                    self.job = Some(Job::Asking(exchange_of(&command, ms), after));
530                    return Begun::Out(Out::Send(command));
531                }
532                Action::OrderByName(named) => {
533                    self.job = Some(Job::Inquiring(exchange_of("inquire", ms), named));
534                    return Begun::Out(Out::Send("inquire".to_owned()));
535                }
536                Action::Pause(length) => Job::Pause(ms + u64::from(length)),
537                Action::Await(line) => Job::Line(vec![line], ms),
538                Action::AwaitAny(lines) => Job::Line(lines, ms),
539                Action::AwaitArrival => Job::Arrival(tick.here, ms),
540                Action::WaitUntil(cond) => Job::Fact(cond, ms),
541                Action::Replan => {
542                    return if tick.here == self.expected {
543                        Begun::Skip
544                    } else {
545                        Begun::Out(Out::Replan)
546                    };
547                }
548                Action::CastAt(spell, target) => {
549                    // It carries the walker, so what follows the deed is the
550                    // wait to land -- a short one, since the spell has gone
551                    // off by the time the deed is done.
552                    self.job = Some(Job::Carried(tick.here, None));
553                    return Begun::Out(Out::Deed(Deed::CastAt(spell, target)));
554                }
555                // The loops and the deeds were taken above, so nothing is left.
556                _ => return Begun::cannot(),
557            }
558        };
559        // The job starts on this same tick.
560        match self.work(job, tick, walker) {
561            Some(out) => Begun::Out(out),
562            None => Begun::Skip,
563        }
564    }

566    /// The deed a step is, if it is one -- noting what it will owe back.
567    fn deed_for(&mut self, action: &Action) -> Option<Deed> {
568        Some(match action {
569            Action::EmptyHands => {
570                self.owes.hands = true;
571                Deed::EmptyHands
572            }
573            Action::FillHands => {
574                self.owes.hands = false;
575                Deed::FillHands
576            }
577            Action::Stance(stance) => {
578                self.owes.stance = true;
579                Deed::Stance(stance.clone())
580            }
581            Action::RestoreStance => {
582                self.owes.stance = false;
583                Deed::RestoreStance
584            }
585            Action::Cast(spell) => Deed::Cast(spell.clone()),
586            Action::Remember(name, value) => Deed::Remember(name.clone(), value.clone()),
587            Action::Forget(name) => Deed::Forget(name.clone()),
588            Action::AwaitFollowers => Deed::AwaitFollowers,
589            Action::Speak(language) => {
590                self.owes.speech = true;
591                Deed::Speak(language.clone())
592            }
593            Action::RestoreSpeech => {
594                self.owes.speech = false;
595                Deed::RestoreSpeech
596            }
597            Action::TakeOut(thing) => {
598                self.owes.taken = true;
599                Deed::TakeOut(thing.clone())
600            }
601            Action::PutBack => {
602                self.owes.taken = false;
603                Deed::PutBack
604            }
605            _ => return None,
606        })
607    }
608}

610/// The loop a step is, if it is one: what is sent each turn, when it stops,
611/// and the step's own bound on it.
612fn loop_for(action: &Action) -> Option<(Pick, Until, Option<u32>)> {
613    let one = |command: &String| Pick::InTurn(vec![command.clone()]);
614    Some(match action {
615        Action::KeepMoving(command) => (one(command), Until::RoomChanges, None),
616        Action::KeepMovingAny(commands) => {
617            (Pick::InTurn(commands.clone()), Until::RoomChanges, None)
618        }
619        Action::MoveUntilThere(command) => (one(command), Until::There, None),
620        Action::MoveWhile(command, cond) | Action::PutWhile(command, cond) => {
621            (one(command), Until::NoLonger(cond.clone()), None)
622        }
623        Action::RoundWhile(commands, cond) => (
624            Pick::InTurn(commands.clone()),
625            Until::NoLonger(cond.clone()),
626            None,
627        ),
628        Action::MoveAnyWhile(commands, cond) => (
629            Pick::AnyOf(commands.clone()),
630            Until::NoLonger(cond.clone()),
631            None,
632        ),
633        Action::WanderWhile(cond) => (Pick::AnyExit, Until::NoLonger(cond.clone()), None),
634        Action::PutUntil {
635            command,
636            until,
637            tries,
638        } => (one(command), Until::Said(until.clone()), *tries),
639        Action::RoundUntil {
640            commands,
641            until,
642            tries,
643        } => (
644            Pick::InTurn(commands.clone()),
645            Until::Said(until.clone()),
646            *tries,
647        ),
648        _ => return None,
649    })
650}

673/// Whether `pick` has nothing to choose from: no commands, or no way out of
674/// a room whose ways out are known.
675fn nothing_to_pick(pick: &Pick, walker: &Walker) -> bool {
676    match pick {
677        Pick::InTurn(commands) | Pick::AnyOf(commands) => commands.is_empty(),
678        Pick::AnyExit => walker.exits.as_ref().is_some_and(Vec::is_empty),
679    }
680}
681
682fn choose(looped: &Looping, tick: &Tick<'_>, walker: &Walker) -> Option<String> {
683    let at = |len: usize, n: u64| usize::try_from(n.checked_rem(u64::try_from(len).ok()?)?).ok();
684    match &looped.pick {
685        Pick::InTurn(commands) => {
686            let command = commands.get(at(commands.len(), u64::from(looped.turns))?)?;
687            filled(command, walker, None)
688        }
689        Pick::AnyOf(commands) => commands.get(at(commands.len(), tick.random)?).cloned(),
690        Pick::AnyExit => {
691            let exits = walker.exits.as_ref()?;
692            // A random walk does not turn straight back when it can help it.
693            let back = looped.last_way.as_deref().and_then(opposite);
694            let ways: Vec<&String> = exits
695                .iter()
696                .filter(|exit| exits.len() < 2 || Some(exit.as_str()) != back)
697                .collect();
698            ways.get(at(ways.len(), tick.random)?)
699                .map(|way| (*way).clone())
700        }
701    }
702}

704fn opposite(way: &str) -> Option<&'static str> {
705    Some(match way {
706        "n" => "s",
707        "s" => "n",
708        "e" => "w",
709        "w" => "e",
710        "ne" => "sw",
711        "sw" => "ne",
712        "nw" => "se",
713        "se" => "nw",
714        "up" => "down",
715        "down" => "up",
716        "out" => "out",
717        _ => return None,
718    })

721/// A command with its placeholders filled in (`cena_map::Action`). `None`
722/// when one cannot be: a setting the profile does not carry, or a `{told}`
723/// nothing was told. **`{item:…}` is left as it is**: the game's id for a
724/// thing is the driver's to look up when it sends (`drive`), since the trip
725/// holds no ids.
726pub(super) fn filled(command: &str, walker: &Walker, told: Option<&str>) -> Option<String> {
727    let mut out = String::new();
728    let mut rest = command;
729    while let Some((before, after)) = rest.split_once('{') {
730        let (name, tail) = after.split_once('}')?;
731        out.push_str(before);
732        if name == "told" {
733            out.push_str(told?);
734        } else if name.starts_with("item:") {
735            out.push('{');
736            out.push_str(name);
737            out.push('}');
738        } else {
739            let value = walker
740                .settings
741                .get(name.strip_prefix("setting:")?)
742                .filter(|value| !value.is_empty())?;
743            out.push_str(value);
744        }
745        rest = tail;
746    }
747    out.push_str(rest);
748    Some(out)
749}

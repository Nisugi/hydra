1//! What a crossing reads out of the game's answer: a word, a number.
2//!
3//! The lines read are the model's, already whole, not raw text matched as it
4//! arrives.
5
6/// The word that follows `after` in any of these lines: `look trail`, and
7/// `...the trail heads off to the northeast.`
8pub(super) fn word_after(lines: &[String], after: &str) -> Option<String> {
9    lines.iter().find_map(|line| {
10        let (_, rest) = line.split_once(after)?;
11        let word: String = rest
12            .trim_start()
13            .chars()
14            .take_while(char::is_ascii_alphabetic)
15            .collect();
16        (!word.is_empty()).then_some(word)
17    })
18}
19
20/// The number of the line that names this: `2) the Sea of Fire`. A caravan's
21/// list is renumbered between visits, and the name is what stays.
22pub(super) fn numbered(lines: &[String], named: &str) -> Option<u32> {
23    let wanted = format!(") {named}");
24    lines.iter().find_map(|line| {
25        let (before, _) = line.split_once(&wanted)?;
26        let digits: String = before
27            .chars()
28            .rev()
29            .take_while(char::is_ascii_digit)
30            .collect::<Vec<_>>()
31            .into_iter()
32            .rev()
33            .collect();
34        digits.parse().ok()
35    })
36}

38#[cfg(test)]
39mod tests {
40    use super::*;
41
42    fn lines(text: &[&str]) -> Vec<String> {
43        text.iter().map(|line| (*line).to_owned()).collect()
44    }
45
46    #[test]
47    fn the_word_is_what_follows_and_stops_at_the_full_stop() {
48        let heard =
49            lines(&["You peer into the mist and see that the trail heads off to the northeast."]);
50        assert_eq!(
51            word_after(&heard, "the trail heads off to the ").as_deref(),
52            Some("northeast")
53        );
54        assert_eq!(word_after(&heard, "the path heads"), None);
55    }
56
57    #[test]
58    fn a_destination_is_found_by_name_whatever_its_number() {
59        let heard = lines(&["  1) Wehnimer's Landing", "  12) the Sea of Fire"]);
60        assert_eq!(numbered(&heard, "the Sea of Fire"), Some(12));
61        assert_eq!(numbered(&heard, "Wehnimer's Landing"), Some(1));
62        assert_eq!(numbered(&heard, "Vornavis"), None);
63    }

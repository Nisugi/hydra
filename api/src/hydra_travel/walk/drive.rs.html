1//! The thin `async` layer round [`Trip`]: what has the socket and the
2//! clock. It joins the session (`SessionHandle::join`), keeps its own
3//! [`GameState`] by folding the frames that follow, says where the character
4//! is, sends what the trip says to send, does its [`Deed`]s (`deeds`), and
5//! stops when told.
6//!
7//! # When it stops
8//!
9//! - **Stopped**: `.stop`, `.go2 stop`, or another `.go2`.
10//! - **Handed to Lich**: one of the player's Lich scripts sent a move while
11//!   Hydra walked. Hydra hands over control rather than fight it: there is no
12//!   telling whether that script will go on moving the character, and two
13//!   walkers pulling one character apart get neither where they mean to go.
14//! - **Lost**: the connection dropped or the session ended. A walk does not
15//!   outlive the connection it set out on, and sends nothing after.
16//! - **Halted**: a rule of the walk stopped it, such as `stop_for_dead`
17//!   beside someone dead.
18//!
19//! A walk that cannot tell where it is for [`LOST_WAIT`] ends, rather than
20//! wait for ever.
21//!
22//! # Nothing is sent into roundtime
23//!
24//! A search is followed by roundtime, and so are climbs and many other
25//! steps: a command sent into it is only told to wait. So after any single
26//! command -- a search, a climb, a step of a crossing -- nothing more is
27//! sent until the game's prompt has closed its answer ([`ANSWER_WAIT`] at
28//! most), which is when its roundtime is known; and then nothing until the
29//! roundtime is over. Plain moves sent ahead in a batch cause none, and a
30//! batch begins only out of roundtime.
31//!
32//! # What a stop sends
33//!
34//! A stop is not a reason to send more, with **one exception**: what a
35//! crossing put away is asked back, **one command each and nothing else**,
36//! since where the walker stands it may not even be able to hold a shield;
37//! and a key a crossing took out goes back first, so the walk finishes up
38//! and leaves the character safe. Nothing is waited for. What was not seen
39//! back is in [`Walked`], for the player to be told. A stance or a language
40//! is not put back on a stop. A walk that lost its connection sends nothing
41//! at all.

158/// Walk to `goal` on `handle`'s character, over `map`, until it arrives,
159/// fails or `stop` is cancelled. `shown` is told the route still ahead,
160/// from the room the walker is in, each time it changes; `wrote` the
161/// memories, each time a crossing changes them, to be kept at once -- a
162/// memory lost strands the character.
163pub async fn walk(
164    handle: &SessionHandle,
165    map: &Map,
166    goal: RoomId,
167    setting: Setting,
168    stop: &CancellationToken,
169    shown: impl FnMut(&[u32]),
170    wrote: impl FnMut(&BTreeMap<String, String>),
171) -> Walked {
172    let mut trip = Trip::to(goal);
173    let Some((snapshot, events)) = handle.join().await else {
174        return gone(&trip);
175    };
176    if matches!(snapshot.lifecycle, State::Reconnecting | State::Closed) {
177        return gone(&trip);
178    }
179    let mut trip_seeded = Trip::seeded(goal, seed_for(&snapshot.state, goal));
180    std::mem::swap(&mut trip, &mut trip_seeded);
181    let heard = heard_so_far(&snapshot.state);
182    let mut driver = Driver {
183        handle,
184        stop,
185        state: snapshot.state,
186        events,
187        heard,
188        began: Instant::now(),
189        was: None,
190        hint: setting.last_room.map(RoomId),
191        lost_since: None,
192        typeahead: None,
193        shown,
194        last_shown: Vec::new(),
195        wrote,
196        memories: setting.memories.clone(),
197        answer: Vec::new(),
198        stored: Vec::new(),
199        stance_before: None,
200        speech_before: None,
201        taken: Vec::new(),
202        behind: Vec::new(),
203        found: HashMap::new(),
204        unanswered: None,
205        kept: super::routines::Kept::default(),
206        prompts: 0,
207        arrived_at_prompt: None,
208        instabilities: setting.instabilities.clone(),
209        instabilities_changed: false,
210        now: setting.now,
211        table: None,
212    };
213    let ended = match setting.into {
214        Some(into) => match driver.enter_the_plane(&mut trip, map, &setting, into).await {
215            // Inside, and on to a room of the plane.
216            Ok(Some(room)) => {
217                let mut inside = Trip::seeded(room, trip.draw());
218                let ended = driver.walk(&mut inside, map, &setting).await;
219                trip = inside;
220                ended
221            }
222            Ok(None) => Ended::Arrived,
223            Err(ended) => ended,
224        },
225        None => match driver.preflight(&mut trip, map, &setting).await {
226            Ok(()) => driver.walk(&mut trip, map, &setting).await,
227            Err(ended) => ended,
228        },
229    };
230    if matches!(ended, Ended::Stopped | Ended::Halted(_)) {
231        driver.take_back_once().await;
232    }
233    Walked {
234        ended,
235        last_room: driver.was.map(|(_, room)| room),
236        typeahead: driver.typeahead,
237        wrong_for_the_map: trip.wrong_for_the_map().to_vec(),
238        // Still put away: what no hand holds as the walk ends, whatever the
239        // record says of when it was taken back.
240        still_stored: driver
241            .stored
242            .into_iter()
243            .filter(|stored| driver.state.hand_holding(&stored.id).is_none())
244            .collect(),
245        stance_before: driver.stance_before,
246        speech_before: driver.speech_before,
247        still_out: driver
248            .taken
249            .into_iter()
250            .map(|taken| (taken.name, taken.sent_back))
251            .collect(),
252        instabilities: driver.instabilities_changed.then_some(driver.instabilities),
253    }
254}

271/// The seed for the choices a crossing asks of this trip: a wander, a
272/// random way out. **Made of what the wire said** -- the game's clock at the
273/// last prompt, the room, the goal -- so a replay chooses the same; mixed
274/// with `splitmix64`'s finaliser so clocks a second apart are not seeds a
275/// bit apart.
276#[must_use]
277pub fn seed_for(state: &GameState, goal: RoomId) -> u64 {
278    let room = state.room.id.as_deref().unwrap_or("");
279    let mut seed = u64::from(state.game_time().unwrap_or(0));
280    seed = mix(seed ^ u64::from(goal.0).rotate_left(32));
281    for byte in room.bytes() {
282        seed = mix(seed ^ u64::from(byte));
283    }
284    seed
285}
286
287const fn mix(mut x: u64) -> u64 {
288    x = x.wrapping_add(0x9E37_79B9_7F4A_7C15);
289    x = (x ^ (x >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
290    x = (x ^ (x >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
291    x ^ (x >> 31)
292}

294/// How many lines of the open chunk are already old news: counted with what
295/// a long chunk dropped from its front, so a line is heard once however the
296/// buffer slides.
297fn heard_so_far(state: &GameState) -> usize {
298    let chunk = state.open_chunk();
299    chunk.dropped() + chunk.lines().len()
300}

375impl<F: FnMut(&[u32]), W: FnMut(&BTreeMap<String, String>)> Driver<'_, F, W> {
376    async fn walk(&mut self, trip: &mut Trip, map: &Map, setting: &Setting) -> Ended {
377        loop {
378            match self.turn(trip, map, setting).await {
379                Ok(Turn::Arrived) => return Ended::Arrived,
380                Ok(_) => {}
381                Err(ended) => return ended,
382            }
383        }
384    }

386    /// One tick of the trip, and what it asked done.
387    async fn turn(&mut self, trip: &mut Trip, map: &Map, setting: &Setting) -> Result<Turn, Ended> {
388        if self.stop.is_cancelled() {
389            return Err(Ended::Stopped);
390        }
391        self.drain(trip).await?;
392        // Dead already when the walk began: nothing new was heard to say so.
393        if self.dead() {
394            return Err(Ended::Halted(
395                "you are dead, so the walk has stopped.".to_owned(),
396            ));
397        }
398        // A single command's answer first, and then its roundtime.
399        if self
400            .unanswered
401            .is_some_and(|sent| sent.elapsed() < ANSWER_WAIT)
402        {
403            self.hold(trip, BEAT).await?;
404            return Ok(Turn::On);
405        }
406        self.unanswered = None;
407        self.out_of_roundtime(trip).await?;
408        let before = self.was;
409        let here = self.locate(map);
410        if before.is_some() && before != self.was {
411            self.arrived(&setting.settings).await?;
412        }
413        if here.is_some() {
414            self.lost_since = None;
415        } else if self.lost_since.get_or_insert_with(Instant::now).elapsed() >= LOST_WAIT {
416            return Err(Ended::Failed(Why::OffTheMap));
417        }
418        let now = Now {
419            here,
420            ms: u64::try_from(self.began.elapsed().as_millis()).unwrap_or(u64::MAX),
421        };
422        self.price_the_way_out(map, setting);
423        let walker = self.walker(setting);
424        let said = trip.tick(map, &walker, now);
425        if let Some(typeahead) = trip.typeahead_learned() {
426            self.typeahead = Some(typeahead);
427        }
428        if let Some(here) = here {
429            self.show(here, &trip.ahead());
430        }
431        match said {
432            Said::Arrived => return Ok(Turn::Arrived),
433            Said::Failed(why) => return Err(Ended::Failed(why)),
434            Said::Aside(worked) => return Ok(Turn::Aside(worked)),
435            Said::Routine(routine) => {
436                // A routine reads the room it is in: not before the room is
437                // told, which ends with the prompt after the arrival. A move
438                // sent ahead lands at its `<nav>`, before the room's things.
439                self.until_the_room_is_told(trip).await?;
440                // A routine moves by asking the trip to, which comes back
441                // through here: boxed, as recursion must be.
442                if !Box::pin(self.solve(trip, map, setting, &routine)).await? {
443                    // Said, not only gone round: the player sees why the walk
444                    // turned aside.
445                    self.handle.say(
446                        hydra_session::Notice::line(
447                            hydra_session::NoticeKind::Warn,
448                            format!(
449                                "Travel: the {} could not be crossed; going another way.",
450                                routine_name(&routine)
451                            ),
452                        )
453                        .answering(),
454                    );
455                    trip.could_not();
456                }
457            }
458            Said::Hold => self.hold(trip, BEAT).await?,
459            Said::Send(line) => match kept::with_items(&line, &self.state) {
460                Some(line) => {
461                    self.send(&line).await?;
462                    self.unanswered = Some(Instant::now());
463                }
464                // A thing nobody has: the crossing cannot be made.
465                None => trip.cannot_send(),
466            },
467            Said::SendAll(lines, gap) => {
468                for line in &lines {
469                    self.send(line).await?;
470                }
471                let until = Instant::now() + Duration::from_millis(gap);
472                while Instant::now() < until {
473                    self.hold(trip, until - Instant::now()).await?;
474                }
475            }
476            Said::Do(Deed::Remember(name, value)) => {
477                self.memories.insert(name, value);
478                (self.wrote)(&self.memories);
479            }
480            Said::Do(Deed::Forget(name)) => {
481                self.memories.remove(&name);
482                (self.wrote)(&self.memories);
483            }
484            Said::Do(deed) => self.deed(trip, deed, setting).await?,
485        }
486        Ok(Turn::On)
487    }

489    /// Until a prompt has followed the last arrival, so what the room holds
490    /// has been told; no longer than [`DEED_DEADLINE`].
491    pub(super) async fn until_the_room_is_told(&mut self, trip: &mut Trip) -> Result<(), Ended> {
492        let until = Instant::now() + DEED_DEADLINE;
493        while self.arrived_at_prompt.is_some_and(|at| self.prompts <= at) && Instant::now() < until
494        {
495            self.hold(trip, until - Instant::now()).await?;
496        }
497        Ok(())
498    }

500    /// The walker's facts now: the model's, the Travel page's, the
501    /// memories, and what the pre-flight found out.
502    fn walker(&self, setting: &Setting) -> cena_map::Walker {
503        let mut walker = walker_from(&self.state, &setting.settings, &self.memories);
504        walker.flags.extend(self.found.clone());
505        if let Some((_, table)) = &self.table {
506            walker
507                .tables
508                .insert("instability".to_owned(), table.clone());
509        }
510        walker
511    }

513    /// The map's `instability` table, worked out once for the instability
514    /// the walker last entered the plane by ([`LAST_INSTABILITY`]): leaving
515    /// the plane puts it back there, so the way out to each town costs the
516    /// walk from it. Unknown, and so shut, for a walker Hydra did not see
517    /// go in.
518    fn price_the_way_out(&mut self, map: &Map, setting: &Setting) {
519        let Some(entered) = self
520            .memories
521            .get(LAST_INSTABILITY)
522            .and_then(|room| room.parse::<u32>().ok())
523        else {
524            self.table = None;
525            return;
526        };
527        if self.table.as_ref().is_some_and(|(at, _)| *at == entered) {
528            return;
529        }
530        let walker = walker_from(&self.state, &setting.settings, &self.memories);
531        let table = super::instability::instability_table(
532            map,
533            RoomId(entered),
534            crate::route::pricing(&walker),
535        );
536        self.table = Some((entered, table));
537    }

539    /// Tell `shown` the route from `here` on, when it is not what it last
540    /// was.
541    fn show(&mut self, here: RoomId, ahead: &[RoomId]) {
542        let route: Vec<u32> = std::iter::once(here.0)
543            .chain(ahead.iter().map(|room| room.0))
544            .collect();
545        if route != self.last_shown {
546            (self.shown)(&route);
547            self.last_shown = route;
548        }
549    }

551    /// The two rules for a room just walked into:
552    /// `delay` seconds are waited, and `stop_for_dead` halts the walk where
553    /// someone lies dead. Not asked of the room the walk began in, so
554    /// starting again beside the body walks on.
555    async fn arrived(&mut self, settings: &TravelSettings) -> Result<(), Ended> {
556        let dead = |player: &hydra_model::RoomItem| {
557            player
558                .status
559                .as_ref()
560                .is_some_and(|status| status.as_str().contains("dead"))
561        };
562        if settings.on("stop_for_dead") && self.state.room.players.iter().any(dead) {
563            return Err(Ended::Halted(
564                "someone here is dead, and Stop beside the dead is on.".to_owned(),
565            ));
566        }
567        let delay = settings
568            .0
569            .get("delay")
570            .and_then(|delay| delay.parse::<f64>().ok())
571            .filter(|delay| delay.is_finite() && *delay > 0.0);
572        if let Some(delay) = delay {
573            let pause = tokio::time::sleep(Duration::from_secs_f64(delay.min(600.0)));
574            tokio::select! {
575                biased;
576                () = self.stop.cancelled() => return Err(Ended::Stopped),
577                () = pause => {}
578            }
579        }
580        Ok(())
581    }

583    /// Which room of the map the model's room is, remembering the last one
584    /// so that two rooms alike in everything are told apart by the way in.
585    /// "Has it moved" is the model's arrival count, which answers for a room
586    /// with no number too.
587    fn locate(&mut self, map: &Map) -> Option<RoomId> {
588        let arrivals = self.state.arrivals;
589        let whence = match self.was {
590            Some((then, at)) if then == arrivals => Whence::Still(at),
591            Some((_, at)) => Whence::Left(at),
592            None => self.hint.map_or(Whence::Nowhere, Whence::Still),
593        };
594        let here = hydra_map::room_of(map, &self.state, whence)?;
595        self.was = Some((arrivals, here));
596        Some(here)
597    }

599    /// Fold every event already waiting, without waiting for more.
600    async fn drain(&mut self, trip: &mut Trip) -> Result<(), Ended> {
601        loop {
602            match self.events.try_recv() {
603                Ok(event) => self.fold(trip, &event)?,
604                Err(TryRecvError::Lagged(_)) => self.rejoin().await?,
605                Err(TryRecvError::Empty) => return Ok(()),
606                Err(TryRecvError::Closed) => return Err(Ended::Lost),
607            }
608        }
609    }
610
611    /// Events were lost: the state is taken afresh, with the stream from just
612    /// after it, rather than decided from the holes.
613    async fn rejoin(&mut self) -> Result<(), Ended> {
614        let (snapshot, events) = self.handle.join().await.ok_or(Ended::Lost)?;
615        if matches!(snapshot.lifecycle, State::Reconnecting | State::Closed) {
616            return Err(Ended::Lost);
617        }
618        self.heard = heard_so_far(&snapshot.state);
619        self.state = snapshot.state;
620        self.events = events;
621        self.was = None;
622        Ok(())
623    }
624
625    fn fold(&mut self, trip: &mut Trip, event: &Event) -> Result<(), Ended> {
626        match event {
627            Event::Frame(frame) => {
628                let arrivals = self.state.arrivals;
629                self.state.apply(frame);
630                if self.state.arrivals != arrivals {
631                    self.arrived_at_prompt = Some(self.prompts);
632                    trip.arrived();
633                }
634                // Dead, the walker goes nowhere: a move, a routine's command
635                // or a wait would go on and on. Heard here, every one of them
636                // stops at once.
637                if self.dead() {
638                    return Err(Ended::Halted(
639                        "you are dead, so the walk has stopped.".to_owned(),
640                    ));
641                }
642                match &**frame {
643                    // The chunk closed, and took its lines with it.
644                    Frame::Prompt { .. } => {
645                        self.heard = 0;
646                        self.prompts += 1;
647                        self.unanswered = None;
648                        trip.prompted();
649                    }
650                    Frame::Text(_) => self.hear(trip),
651                    _ => {}
652                }
653                Ok(())
654            }
655            Event::Sent {
656                line,
657                origin: Origin::Lich,
658            } if is_a_move(line) => Err(Ended::HandedToLich(line.clone())),
659            Event::StateChanged(State::Reconnecting | State::Closed) => Err(Ended::Lost),
660            _ => Ok(()),
661        }
662    }
663
664    /// Tell the trip the whole lines the model has finished since it last
665    /// heard any, keep them as the answer to a deed's command, and strike
666    /// off whoever they say has rejoined.
667    fn hear(&mut self, trip: &mut Trip) {
668        let chunk = self.state.open_chunk();
669        let lines = chunk.lines();
670        let total = chunk.dropped() + lines.len();
671        let fresh = total.saturating_sub(self.heard).min(lines.len());
672        for line in &lines[lines.len() - fresh..] {
673            let text = line.text();
674            trip.heard(&text);
675            // The one way the game tells of a mount: by refusing.
676            if text.ends_with("while mounted.") {
677                self.found.insert("mounted".to_owned(), true);
678            }
679            self.answer.push(line.clone());
680            if let Some(GroupEvent::Joined(member) | GroupEvent::Added(member)) =
681                group::classify(line)
682            {
683                self.behind.retain(|id| *id != member.id);
684            }
685        }
686        self.heard = total;
687    }
688
689    /// Until the roundtime the game stated is over, or
690    /// [`LONGEST_ROUNDTIME`]: folding what comes meanwhile.
691    async fn out_of_roundtime(&mut self, trip: &mut Trip) -> Result<(), Ended> {
692        let until = Instant::now() + LONGEST_ROUNDTIME;
693        while self.state.in_roundtime() == Some(true) && Instant::now() < until {
694            self.hold(trip, BEAT).await?;
695        }
696        Ok(())
697    }
698
699    /// Wait for the game's next word, or `most`, whichever is first.
700    async fn hold(&mut self, trip: &mut Trip, most: Duration) -> Result<(), Ended> {
701        let event = tokio::select! {
702            biased;
703            () = self.stop.cancelled() => return Err(Ended::Stopped),
704            event = self.events.recv() => event,
705            () = tokio::time::sleep(most) => return Ok(()),
706        };
707        match event {
708            Ok(event) => self.fold(trip, &event),
709            Err(RecvError::Lagged(_)) => self.rejoin().await,
710            Err(RecvError::Closed) => Err(Ended::Lost),
711        }
712    }
713
714    /// Write `line` to the game at once: the trip checks every move by the
715    /// room it lands in, so nothing waits for an answer here.
716    async fn send(&mut self, line: &str) -> Result<(), Ended> {
717        if self.stop.is_cancelled() {
718            return Err(Ended::Stopped);
719        }
720        if self.handle.write(line, Origin::Travel).await {
721            Ok(())
722        } else {
723            Err(Ended::Lost)
724        }
725    }
726
727    /// Whether the game says the character is dead.
728    fn dead(&self) -> bool {
729        self.state.status.known().dead() == Some(true)
730    }
731
732    /// The exception to "a stop sends nothing": the key a crossing
733    /// took out put back, then one command per stored thing, none waited
734    /// for (module docs). They stay on the record, since nothing saw them go.
735    async fn take_back_once(&mut self) {
736        // A dead character can take nothing back.
737        if self.dead() {
738            return;
739        }
740        for taken in self.taken.iter_mut().rev() {
741            let command = format!("put #{} in #{}", taken.thing, taken.container);
742            taken.sent_back = self.handle.write(&command, Origin::Travel).await;
743        }
744        for stored in self.stored.iter().rev() {
745            let command = take_back(&self.state, stored);
746            let _ = self.handle.write(&command, Origin::Travel).await;
747        }
748    }
749}

751/// Whether `line` moves the character: a direction, or a verb that walks
752/// it through something. What a Lich script sends when it walks.
753#[must_use]
754pub fn is_a_move(line: &str) -> bool {
755    const WAYS: [&str; 26] = [
756        "n",
757        "s",
758        "e",
759        "w",
760        "ne",
761        "nw",
762        "se",
763        "sw",
764        "u",
765        "d",
766        "north",
767        "south",
768        "east",
769        "west",
770        "northeast",
771        "northwest",
772        "southeast",
773        "southwest",
774        "up",
775        "down",
776        "out",
777        "go",
778        "climb",
779        "swim",
780        "crawl",
781        "jump",
782    ];
783    line.split_whitespace()
784        .next()
785        .is_some_and(|word| WAYS.iter().any(|way| way.eq_ignore_ascii_case(word)))
786}

788/// A routine as the player is told of it: its name in the map, in words
789/// (`minotaur maze`).
790fn routine_name(routine: &cena_map::Routine) -> String {
791    serde_json::to_value(routine)
792        .ok()
793        .and_then(|value| {
794            value
795                .get("name")?
796                .as_str()
797                .map(|name| name.replace('_', " "))
798        })
799        .unwrap_or_else(|| "way".to_owned())
800}

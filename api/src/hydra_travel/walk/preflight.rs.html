1//! What must be known before a trip is priced: **a cost never acts**.
2//! A cost only reads a fact, and never sends a command while the pathfinder
3//! runs, so whatever a fact needs asked is asked once, before the first plan
4//! ([`super::drive`]'s pre-flight). **Pure**: the map and the game's
5//! answers in, facts out.

7use cena_map::{Map, RoomId, Uid};

9/// The silver a walk asks for: each room's `silver-cost:<to>:<silver>` tag
10/// for the room that follows it. `path` begins where the walker stands. A
11/// cost that does not parse costs nothing here rather than a guess
12/// (MEASURED: all 88 of the map's parse).
13#[must_use]
14pub fn silver_for(map: &Map, path: &[RoomId]) -> u64 {
15    path.windows(2)
16        .filter_map(|pair| {
17            let (from, to) = (map.room(*pair.first()?)?, *pair.get(1)?);
18            from.tags.iter().find_map(|tag| {
19                let (room, silver) = tag.strip_prefix("silver-cost:")?.split_once(':')?;
20                (room.parse() == Ok(to.0)).then_some(())?;
21                silver.parse::<u64>().ok()
22            })
23        })
24        .sum()
25}

27/// What to say to a banker for this much: Pinefar's has no `withdraw`, and
28/// hands out no less than twenty. Never for nothing: the least asked is one.
29#[must_use]
30pub fn withdraw_command(room_title: Option<&str>, silver: u64) -> String {
31    if room_title.is_some_and(|title| title.contains("Pinefar, Depository")) {
32        format!("ask banker for {} silvers", silver.max(20))
33    } else {
34        format!("withdraw {} silvers", silver.max(1))
35    }
36}

38/// What a player shop's room title ends with, as the game names one: no map
39/// holds a player's shop.
40const PLAYER_SHOPS: [&str; 7] = [
41    "Outfitting",
42    "Magic Shoppe",
43    "Weaponry",
44    "General Store",
45    "Armory",
46    "Combat Gear",
47    "Lockpicks",
48];
49
50/// The thousands a player shop's rooms are numbered in.
51const PLAYER_SHOP_ROOMS: std::ops::RangeInclusive<u64> = 631..=646;
52
53/// The way out of a player's shop, which no map holds, for a walker standing
54/// in `room`, a room the map does not know: `out` -- after a step further
55/// in by its first exit, when the room has no `out`. `None` when `room` is
56/// no player's shop by its title or its number.
57#[must_use]
58pub fn out_of_a_shop(room: &hydra_model::state::room::Room) -> Option<Vec<String>> {
59    let titled = room
60        .title
61        .as_deref()
62        .is_some_and(|title| PLAYER_SHOPS.iter().any(|shop| title.contains(shop)));
63    let numbered = room
64        .id
65        .as_deref()
66        .and_then(|id| id.parse::<u64>().ok())
67        .is_some_and(|id| PLAYER_SHOP_ROOMS.contains(&(id / 1000)));
68    if !(titled || numbered) {
69        return None;
70    }
71    let exits = room.exits.clone().unwrap_or_default();
72    let mut commands = Vec::new();
73    if !exits.iter().any(|exit| exit == "out")
74        && let Some(first) = exits.first()
75    {
76        commands.push(first.clone());
77    }
78    commands.push("out".to_owned());
79    Some(commands)
80}

82/// The Long Snow's encampment and Cairnfang Manor's attic: the two ends of
83/// the long way to the Hinterwilds.
84const HINTERWILDS_WAYS: [u32; 2] = [29860, 22154];
85
86/// Sparkfinger's workroom in Cold River, where `order 3` takes a walker
87/// out of the Hinterwilds by gigas fragments, and where the teleporters
88/// bring one in.
89pub const WORKROOM: i64 = 7_503_253;
90
91/// The way by gigas fragments in place of the long way to or from the
92/// Hinterwilds: the teleporter walked to, and what is said there.
93#[derive(Clone, Copy, Debug, PartialEq, Eq)]
94pub struct GigasWay {
95    /// The room of the teleporter: Ta'Illistim's or the Abbey's, coming;
96    /// Sparkfinger's workroom, going.
97    pub teleporter: RoomId,
98    /// What is said there.
99    pub commands: &'static [&'static str],
100    /// The side the Hinterwilds are entered from, `EN` or `IM`, remembered
101    /// while the walker is there; none, leaving.
102    pub side: Option<&'static str>,
103    /// Whether it goes into the Hinterwilds.
104    pub arriving: bool,
105}
106
107/// The gigas fragments the Travel page keeps back: a walk spends them only
108/// with at least this many carried. Read from `settings`'s
109/// `gigas_min_number`, four until set. One reading, so `.go2` spends and
110/// `.route2` says by the same count.
111#[must_use]
112pub fn gigas_least(settings: &crate::settings::TravelSettings) -> u64 {
113    settings
114        .0
115        .get("gigas_min_number")
116        .and_then(|least| least.parse().ok())
117        .unwrap_or(4)
118}

120/// The way by gigas fragments a walk from `here` to `goal` along `path`
121/// (from `here`) would take instead, when `path` takes the long way: into
122/// the Hinterwilds by the teleporter -- Ta'Illistim's when `path` passes
123/// Seethe Naedal, the Abbey's otherwise -- or out by Sparkfinger's workroom.
124/// Arriving or leaving is the goal's alone: a goal in the Hinterwilds is
125/// arriving, any other leaving, wherever the walk sets out from. A walk from
126/// Mist Harbor's bank back through Cold River and out by the caravan starts
127/// outside the Hinterwilds, and still leaves by the workroom. A walk with
128/// both ends inside takes no fragments. Whether the fragments are there to
129/// spend is the caller's to ask.
130#[must_use]
131pub fn gigas_way(map: &Map, path: &[RoomId], here: RoomId, goal: RoomId) -> Option<GigasWay> {
132    if !path.iter().any(|room| HINTERWILDS_WAYS.contains(&room.0)) {
133        return None;
134    }
135    let wild = |room: RoomId| {
136        map.room(room).and_then(|room| room.location.as_deref()) == Some("the Hinterwilds")
137    };
138    let arriving = wild(goal);
139    if wild(here) && arriving {
140        return None;
141    }
142    let elven = path.iter().any(|room| {
143        map.room(*room).is_some_and(|room| {
144            room.title
145                .first()
146                .is_some_and(|title| title.contains("Seethe Naedal"))
147        })
148    });
149    let (uid, side, commands): (i64, _, &'static [&'static str]) = match (arriving, elven) {
150        (true, true) => (13_205_202, Some("EN"), &["go sliver", "go sliver"]),
151        (true, false) => (4_132_054, Some("IM"), &["go sliver", "go sliver"]),
152        (false, _) => (WORKROOM, None, &["order 3", "order confirm"]),
153    };
154    let teleporter = map.ids_for_uid(Uid(uid)).first().copied()?;
155    Some(GigasWay {
156        teleporter,
157        commands,
158        side,
159        arriving,
160    })
161}

169    /// A player's shop, by its title or its number, is left by `out`, a step
170    /// further in first when it has none; any other room is no shop.
171    #[test]
172    fn a_players_shop_is_left_by_out() {
173        let room = |id: &str, title: &str, exits: &[&str]| {
174            let mut room = hydra_model::state::room::Room::default();
175            room.id = Some(id.to_owned());
176            room.title = Some(title.to_owned());
177            room.exits = Some(exits.iter().map(|exit| (*exit).to_owned()).collect());
178            room
179        };
180        assert_eq!(
181            out_of_a_shop(&room("12", "[Lirdena's Outfitting]", &["out"])),
182            Some(vec!["out".to_owned()])
183        );
184        assert_eq!(
185            out_of_a_shop(&room("631005", "[Back Room]", &["w"])),
186            Some(vec!["w".to_owned(), "out".to_owned()])
187        );
188        assert_eq!(out_of_a_shop(&room("12", "[Town Square]", &["n"])), None);
189    }
190
191    #[test]
192    fn a_walk_costs_what_its_own_exits_ask_and_no_others() {
193        let rooms: Vec<Room> = serde_json::from_str(
194            r#"[{"id":1,"tags":["bank","silver-cost:2:50","silver-cost:9:7000"]},
195                {"id":2,"tags":["silver-cost:3:25"]},{"id":3},{"id":9}]"#,
196        )
197        .unwrap();
198        let map = Map::from_rooms(rooms).unwrap();
199        let path = [RoomId(1), RoomId(2), RoomId(3)];
200        assert_eq!(silver_for(&map, &path), 75);
201        assert_eq!(silver_for(&map, &path[1..]), 25);
202        assert_eq!(silver_for(&map, &[RoomId(3)]), 0);
203    }
204
205    #[test]
206    fn pinefars_banker_is_asked_and_for_no_less_than_twenty() {

208            withdraw_command(Some("[Bank, Teller]"), 5),

211        assert_eq!(
212            withdraw_command(Some("[Pinefar, Depository]"), 5),
213            "ask banker for 20 silvers"
214        );

216        assert_eq!(withdraw_command(None, 0), "withdraw 1 silvers");

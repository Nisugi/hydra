1//! `Puzzle::AltarLevers`: the altar of room 18893.
2//!
3//! `look altar` and take the line that begins "There is a strange grid on
4//! the altar."; in it `Yellow 3`, `Blue 1` and so on give each coloured
5//! lever its row. Then, lever by lever -- yellow, blue, brown, red, green --
6//! `pull <colour> lever` until the "You pull" line names that row: `first`
7//! to `fifth`. Then `go step`, `go passage`, `go opening`.
8//!
9//! A grid that cannot be read is [`Next::Failed`]. Every pull counts toward
10//! [`MAX_TURNS`], so a lever that never says its row cannot hold the walker
11//! for ever. A move that fails ends the routine, and the trip plans again
12//! from wherever the walker is.
13
14use hydra_model::ChunkLine;
15
16use super::{MAX_TURNS, Next, Seen, Solver};
17
18const GRID: &str = "There is a strange grid on the altar.";
19/// The levers as the grid labels them and as they are pulled, in the order
20/// they are pulled.
21const LEVERS: &[(&str, &str)] = &[
22    ("Yellow", "yellow"),
23    ("Blue", "blue"),
24    ("Brown", "brown"),
25    ("Red", "red"),
26    ("Green", "green"),
27];
28/// The grid's rows as a pull names them.
29const ROWS: &[(char, &str)] = &[
30    ('1', "first"),
31    ('2', "second"),
32    ('3', "third"),
33    ('4', "fourth"),
34    ('5', "fifth"),
35];
36const WAY_OUT: &[&str] = &["go step", "go passage", "go opening"];
37
38/// The row the grid gives this lever: the digit after `Yellow `.
39fn row_of(grid: &str, label: &str) -> Option<&'static str> {
40    let label = format!("{label} ");
41    grid.match_indices(&label).find_map(|(at, _)| {
42        let digit = grid[at + label.len()..].chars().next()?;
43        ROWS.iter()
44            .find(|(number, _)| *number == digit)
45            .map(|(_, row)| *row)
46    })
47}
48
49#[derive(Default)]
50pub(super) struct AltarLevers {
51    /// Where each of [`LEVERS`] must stand, once the grid is read.
52    rows: Vec<&'static str>,
53    turns: u32,
54    at: At,
55}
56
57#[derive(Default)]
58enum At {
59    #[default]
60    Start,
61    Looked,
62    /// Pulling this one of [`LEVERS`].
63    Pulling(usize),
64    /// How many of [`WAY_OUT`] have been sent.
65    Leaving(usize),
66}
67
68impl AltarLevers {
69    fn read(seen: &Seen<'_>) -> Option<Vec<&'static str>> {
70        let grid = seen
71            .answer
72            .iter()
73            .map(ChunkLine::text)
74            .find(|text| text.starts_with(GRID))?;
75        LEVERS
76            .iter()
77            .map(|(label, _)| row_of(&grid, label))
78            .collect()
79    }
80
81    fn pull(&mut self, seen: &Seen<'_>, lever: usize, pulled: bool) -> Next {
82        let (Some((_, colour)), Some(row)) = (LEVERS.get(lever), self.rows.get(lever)) else {
83            self.at = At::Leaving(0);
84            return self.next(seen);
85        };
86        let set = pulled
87            && seen
88                .answer
89                .iter()
90                .map(ChunkLine::text)
91                .any(|text| text.starts_with("You pull") && text.contains(row));
92        if set {
93            return self.pull(seen, lever + 1, false);
94        }
95        if self.turns >= MAX_TURNS {
96            return Next::Failed;
97        }
98        self.turns += 1;
99        self.at = At::Pulling(lever);
100        Next::Put(format!("pull {colour} lever"))
101    }
102}
103
104impl Solver for AltarLevers {
105    fn next(&mut self, seen: &Seen<'_>) -> Next {
106        match self.at {
107            At::Start => {
108                self.at = At::Looked;
109                Next::Put("look altar".to_owned())
110            }
111            At::Looked => {
112                let Some(rows) = Self::read(seen) else {
113                    return Next::Failed;
114                };
115                self.rows = rows;
116                self.pull(seen, 0, false)
117            }
118            At::Pulling(lever) => self.pull(seen, lever, true),
119            At::Leaving(_) if !seen.ok => Next::Done,
120            At::Leaving(sent) => match WAY_OUT.get(sent) {
121                Some(command) => {
122                    self.at = At::Leaving(sent + 1);
123                    Next::Go((*command).to_owned())
124                }
125                None => Next::Done,
126            },
127        }
128    }
129}

131#[cfg(test)]
132mod tests {
133    use super::super::testing::Scene;
134    use super::*;
135
136    const ALTAR: &str = "There is a strange grid on the altar.  Five rows are labeled with \
137        numbers and five columns are labeled with colors.  A red glow illuminates the \
138        following squares: Yellow 3, Blue 1, Brown 5, Red 2, Green 4.";
139
140    fn put(command: &str) -> Next {
141        Next::Put(command.to_owned())
142    }
143
144    fn hears(altar: &mut AltarLevers, line: &str) -> Next {
145        Scene::at(1, 2).answered(&[line]).ask(altar)
146    }
147
148    fn pulled(row: &str) -> String {
149        format!("You pull the lever and it clicks into the {row} position.")
150    }
151
152    #[test]
153    fn each_lever_is_pulled_to_its_row_and_then_the_way_out_is_walked() {
154        let mut altar = AltarLevers::default();
155        assert_eq!(Scene::at(1, 2).ask(&mut altar), put("look altar"));
156        assert_eq!(hears(&mut altar, ALTAR), put("pull yellow lever"));
157        assert_eq!(
158            hears(&mut altar, &pulled("second")),
159            put("pull yellow lever")
160        );
161        assert_eq!(hears(&mut altar, &pulled("third")), put("pull blue lever"));
162        assert_eq!(hears(&mut altar, &pulled("first")), put("pull brown lever"));
163        assert_eq!(hears(&mut altar, &pulled("fifth")), put("pull red lever"));
164        assert_eq!(
165            hears(&mut altar, &pulled("second")),
166            put("pull green lever")
167        );
168        assert_eq!(
169            hears(&mut altar, &pulled("fourth")),
170            Next::Go("go step".into())
171        );
172        assert_eq!(
173            Scene::at(3, 2).ask(&mut altar),
174            Next::Go("go passage".into())
175        );
176        assert_eq!(
177            Scene::at(4, 2).ask(&mut altar),
178            Next::Go("go opening".into())
179        );
180        assert_eq!(Scene::at(2, 2).ask(&mut altar), Next::Done);
181    }
182
183    #[test]
184    fn only_a_pull_names_the_row() {
185        let mut altar = AltarLevers::default();
186        Scene::at(1, 2).ask(&mut altar);
187        hears(&mut altar, ALTAR);
188        let other = "Someone says, \"the third one\"";
189        assert_eq!(hears(&mut altar, other), put("pull yellow lever"));
190    }
191
192    #[test]
193    fn every_row_of_every_lever_is_read_off_the_grid() {
194        for (at, (label, _)) in LEVERS.iter().enumerate() {
195            for (digit, row) in ROWS {
196                let grid = format!("{GRID}  A red glow illuminates Mauve 9, {label} {digit}.");
197                assert_eq!(row_of(&grid, label), Some(*row));
198                // And it is what that lever is pulled to.
199                let mut altar = AltarLevers {
200                    rows: vec![row; LEVERS.len()],
201                    turns: 0,
202                    at: At::Pulling(at),
203                };
204                let next = LEVERS.get(at + 1).map_or_else(
205                    || Next::Go("go step".into()),
206                    |(_, colour)| put(&format!("pull {colour} lever")),
207                );
208                assert_eq!(hears(&mut altar, &pulled(row)), next);
209            }
210        }
211    }
212
213    #[test]
214    fn a_grid_that_cannot_be_read_cannot_be_crossed() {
215        let mut altar = AltarLevers::default();
216        Scene::at(1, 2).ask(&mut altar);
217        assert_eq!(hears(&mut altar, "I could not find that."), Next::Failed);
218
219        let mut altar = AltarLevers::default();
220        Scene::at(1, 2).ask(&mut altar);
221        let short = format!("{GRID}  A red glow illuminates Yellow 3, Blue 1, Brown 5, Red 2.");
222        assert_eq!(hears(&mut altar, &short), Next::Failed);
223    }
224
225    #[test]
226    fn a_lever_that_never_says_its_row_is_given_up() {
227        let mut altar = AltarLevers::default();
228        Scene::at(1, 2).ask(&mut altar);
229        for _ in 0..MAX_TURNS {
230            assert_eq!(hears(&mut altar, ALTAR), put("pull yellow lever"));
231        }
232        assert_eq!(hears(&mut altar, ALTAR), Next::Failed);
233    }
234
235    #[test]
236    fn a_move_that_fails_ends_it() {
237        let mut altar = AltarLevers {
238            rows: Vec::new(),
239            turns: 0,
240            at: At::Leaving(1),
241        };
242        let mut stuck = Scene::at(1, 2);
243        stuck.failed = true;
244        assert_eq!(stuck.ask(&mut altar), Next::Done);
245    }

1//! `Routine::DayPass`: a Chronomage day pass between two towns (6 exits).
2//!
3//! Find the `day_pass_sack`, opening it if it is shut; empty the hands; drop
4//! every pass that has expired; `get` one valid for both towns. With none,
5//! and `buy_day_pass` saying `yes`, `true` or this route, step to the clerk,
6//! unhide, and ask twice -- the first asking is quoted a price, the second
7//! pays it. Short of silver, and allowed to fetch it, walk to the bank once,
8//! `withdraw 5000`, walk back and ask again. Look at the new pass, step back,
9//! `raise` the pass, drag it into the sack -- `put` it in, when the sack is
10//! known only by its name -- fill the hands, shut the sack if this opened
11//! it. The clerk's word, the step in and out, and the walk to the
12//! bank are the town's ([`TOWNS`]); what to ask for is the destination's.
13//!
14//! # How it behaves, and why
15//!
16//! - **A pass read is remembered**, by its id and the expiry it states,
17//!   under the flag of its two towns ([`memory_of`]): a pass bought, or
18//!   one the sack showed. While it is good, the routine `get`s it by id
19//!   from wherever it is, with no look in the sack, and the pre-flight
20//!   takes it as held. A `get` that leaves the hands empty -- the id
21//!   changed with a login, or the pass is gone -- forgets it, and the sack
22//!   is looked in as before, which remembers the pass under its new id.
23//!   [`read_pass`] and [`flag_for`] are the reading as pure functions, for
24//!   the planner's pre-flight.
25//! - **"Now" is the game's.** The expiry is compared against
26//!   `GameState::game_time` -- the last prompt's server time, so a replay
27//!   reads the same. With no prompt seen yet, a pass not stamped `EXPIRED` is
28//!   taken as good: the game refuses a dead one when raised, and nothing is
29//!   dropped on a guess.
30//! - **Both hands are emptied**, not one: `Action::EmptyHands` is what the
31//!   trip can owe back, and there is no one-handed form.
32//! - **A move that fails ends it.** A failed `Next::Go` gives the hands back
33//!   and ends, and the trip looks at where it landed.
34//! - **Too poor** answers `Next::Failed`, which gives the exit up for this
35//!   trip. A solver cannot write the profile, so `buy_day_pass` is left as it
36//!   is.
37//! - **No pass and not buying** is `Next::Failed`: the planner's flag should
38//!   have priced this exit shut.
39//! - Fetching silver is the setting `get_silvers` (`yes`/`true`).
40//! - The sack is sent as `#id` when the inventory names it ([`sack_in`]), and
41//!   as `my <sack>` when it does not.
42//! - Every list is bounded: [`MAX_PASSES`] passes looked at, one bank trip.
43//! - **Each look and each asking is answered by its own line** ([`Next::Ask`]).
44//!   A reply read up to the first prompt was whatever came first: a stranger
45//!   walking in could end the paying `ask`'s window, so the routine took it as
46//!   done and the replan bought a second pass; one late `look #pass` could put
47//!   an expired pass's text in the next pass's answer, and the valid one was
48//!   dropped; a late `look in` read as an empty sack. A paying asking nobody
49//!   answered is decided by the hands, which were emptied before it: a pass in
50//!   one is the purchase, and none is a failure, which bans the exit rather
51//!   than buy again.

222/// When a pass stops working, as far as looking at it says.
223#[derive(Debug, Clone, Copy, PartialEq, Eq)]
224pub(in crate::walk) enum Expiry {
225    /// Stamped `EXPIRED`.
226    Stamped,
227    /// Server epoch seconds: the stated time read at UTC-5, daylight saving
228    /// or not.
229    At(i64),
230    /// The pass did not say.
231    Unsaid,
232}

234/// What looking at one pass said.
235#[derive(Debug, Clone, PartialEq, Eq)]
236pub(in crate::walk) struct Pass {
237    /// The two towns, as the pass spells them. `None` on a stamped pass,
238    /// which no longer says.
239    pub towns: Option<(String, String)>,
240    pub expiry: Expiry,
241}

243impl Pass {
244    /// Expired more than [`MARGIN`] seconds before `now`. Not knowing the
245    /// time, only the stamp says so.
246    pub fn lapsed(&self, now: Option<i64>) -> bool {
247        match (self.expiry, now) {
248            (Expiry::Stamped, _) => true,
249            (Expiry::At(at), Some(now)) => at < now - MARGIN,
250            _ => false,
251        }
252    }
253
254    /// Names these two towns, in either order, and is good for more than
255    /// [`MARGIN`] seconds past `now`.
256    pub fn serves(&self, one: &str, other: &str, now: Option<i64>) -> bool {
257        let named = self
258            .towns
259            .as_ref()
260            .is_some_and(|(a, b)| (a == one && b == other) || (a == other && b == one));
261        let live = match (self.expiry, now) {
262            (Expiry::At(at), Some(now)) => at > now + MARGIN,
263            (Expiry::At(_), None) => true,
264            _ => false,
265        };
266        named && live
267    }
268}

270/// Read the answer to `look #<pass>`.
271/// `None` when the lines are not a day pass's.
272pub(in crate::walk) fn read_pass(lines: &[ChunkLine]) -> Option<Pass> {
273    let mut pass: Option<Pass> = None;
274    for line in lines {
275        let text = line.text();
276        if text.starts_with("Bold red block letters spelling out \"EXPIRED\"") {
277            return Some(Pass {
278                towns: None,
279                expiry: Expiry::Stamped,
280            });
281        }
282        if text.starts_with("Bold calligraphy states simply") {
283            let (_, rest) = text.split_once("between the towns of ")?;
284            let (towns, _) = rest.split_once(", commencing")?;
285            let (one, other) = towns.split_once(" and ")?;
286            pass = Some(Pass {
287                towns: Some((one.to_owned(), other.to_owned())),
288                expiry: Expiry::Unsaid,
289            });
290        } else if let (Some(read), Some(rest)) = (
291            pass.as_mut(),
292            text.strip_prefix("[Your pass will expire on "),
293        ) {
294            read.expiry = expiry_of(rest).map_or(Expiry::Unsaid, Expiry::At);
295        }
296    }
297    pass
298}

300/// `Sun Sep 21 14:03:22 ET 2026.` as epoch seconds at UTC-5.
301fn expiry_of(stated: &str) -> Option<i64> {
302    const MONTHS: [&str; 12] = [
303        "Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec",
304    ];
305    let mut words = stated.split_whitespace();
306    let (_weekday, month, day, clock, zone, year) = (
307        words.next()?,
308        words.next()?,
309        words.next()?,
310        words.next()?,
311        words.next()?,
312        words.next()?,
313    );
314    (zone == "ET").then_some(())?;
315    let month = i64::try_from(MONTHS.iter().position(|is| *is == month)?).ok()? + 1;
316    let day: i64 = day.parse().ok()?;
317    let year: i64 = year.split('.').next()?.parse().ok()?;
318    let mut clock = clock.split(':').map(|part| part.parse::<i64>().ok());
319    let (hour, minute, second) = (clock.next()??, clock.next()??, clock.next()??);
320    // Days from the civil date (Howard Hinnant's algorithm).
321    let y = if month <= 2 { year - 1 } else { year };
322    let era = y.div_euclid(400);
323    let yoe = y.rem_euclid(400);
324    let doy = (153 * ((month + 9) % 12) + 2) / 5 + day - 1;
325    let doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
326    let days = era * 146_097 + doe - 719_468;
327    Some(days * 86_400 + hour * 3_600 + minute * 60 + second + 5 * 3_600)
328}

330/// The planner's flag for a pass between these two towns, named as a pass
331/// spells them: `day_pass:imt,wl`, the codes in alphabetical order.
332pub(in crate::walk) fn flag_for(one: &str, other: &str) -> Option<String> {
333    let code = |name: &str| TOWNS.iter().find(|town| town.name == name).map(|t| t.code);
334    let (mut first, mut second) = (code(one)?, code(other)?);
335    if first > second {
336        std::mem::swap(&mut first, &mut second);
337    }
338    (first != second).then(|| format!("{FLAG}{first},{second}"))
339}

341/// What is remembered of a pass under its towns' flag: its id and the
342/// epoch second it ends, `"1124908932 1793230734"`.
343pub(in crate::walk) fn memory_of(id: &str, expiry: i64) -> String {
344    format!("{id} {expiry}")
345}

347/// The id a pass's memory names, while the pass is good for more than
348/// [`MARGIN`] seconds past `now`; with no time known, while it is kept.
349pub(in crate::walk) fn remembered(memory: &str, now: Option<i64>) -> Option<&str> {
350    let (id, at) = memory.split_once(' ')?;
351    let at: i64 = at.parse().ok()?;
352    now.is_none_or(|now| at > now + MARGIN).then_some(id)
353}

355/// Whether either hand holds the thing `id`.
356fn in_hand(seen: &Seen<'_>, id: Option<&str>) -> bool {
357    id.is_some_and(|id| {
358        [&seen.state.right_hand, &seen.state.left_hand]
359            .into_iter()
360            .any(|hand| hand.id() == Some(id))

364/// The ids of the day passes a `look in` answer shows.
365pub(in crate::walk) fn passes_in(lines: &[ChunkLine]) -> Vec<String> {
366    let mut ids = Vec::new();
367    for link in lines.iter().flat_map(ChunkLine::objects) {
368        if let LinkKind::Exist { id, noun } = &link.kind
369            && noun == "pass"
370            && link.text.contains("day pass")
371            && !ids.contains(id)
372        {
373            ids.push(id.clone());
374        }
375    }
376    ids.truncate(MAX_PASSES);
377    ids
378}

380#[derive(Clone, Copy, PartialEq, Eq, Debug)]
381enum At {
382    Start,
383    /// The answer is to `look in`; whether opening it has been tried.
384    LookedIn(bool),
385    Opened,
386    Emptied,
387    /// The answer is to `look` at pass `.0`.
388    Read(usize),
389    /// Pass `.0` was dropped; on to the next.
390    Dropped(usize),
391    Got,
392    Raised,
393    /// Stepped to the clerk, or walked back from the bank.
394    Clerk,
395    Unhidden,
396    /// Asked this many times since arriving.
397    Asked(u32),
398    ToBank(usize),
399    Withdrew,
400    FromBank(usize),
401    LookedAtNew,
402    SteppedBack,
403    /// The pass was put away; the answer says whether the sack took it.
404    PutBack,
405    /// The shut sack was opened, to put the pass in again.
406    Reopened,
407    /// Putting things back, then answering `.0` (`true` is `Done`).
408    Ending(bool),
409}

432impl DayPass {
433    pub fn new(route: &str) -> Self {
434        let (from, to) = route.split_once(',').unwrap_or_default();
435        DayPass {
436            from: town(from),
437            to: town(to),
438            route: route.to_owned(),
439            at: At::Start,
440            sack: String::new(),
441            opened_sack: false,
442            hands_emptied: false,
443            banked: false,
444            passes: Vec::new(),
445            pass: None,
446            remembered: None,
447            notes: Vec::new(),
448        }
449    }
450
451    /// The flag of this route's two towns, which a pass is remembered under.
452    fn flag(&self) -> Option<String> {
453        let (from, to) = (self.from?, self.to?);
454        flag_for(from.name, to.name)
455    }
456
457    /// Remember the pass `id` as `looked` described it, when it says its
458    /// towns and when it ends, and is good past `now`.
459    fn note(&mut self, id: &str, looked: &[ChunkLine], now: Option<i64>) {
460        let Some(pass) = read_pass(looked) else {
461            return;
462        };
463        let (Some((one, other)), Expiry::At(at)) = (&pass.towns, pass.expiry) else {
464            return;
465        };
466        if !pass.serves(one, other, now) {
467            return;
468        }
469        if let Some(flag) = flag_for(one, other) {
470            self.notes.push(Action::Remember(flag, memory_of(id, at)));
471        }
472    }
473
474    /// The command that puts the pass away: dragging takes two ids; a sack
475    /// known only by its name is `put ... in`.
476    fn put_away(&self) -> String {
477        let pass = self.pass.as_deref().unwrap_or_default();
478        if self.sack.starts_with('#') {
479            format!("_drag #{pass} {}", self.sack)
480        } else {
481            format!("put #{pass} in {}", self.sack)
482        }
483    }
484
485    /// Give back the hands, shut the sack if this opened it, and answer.
486    fn end(&mut self, done: bool) -> Next {
487        self.at = At::Ending(done);
488        if !self.notes.is_empty() {
489            return Next::Steps(self.notes.drain(..).map(step).collect());
490        }
491        if std::mem::take(&mut self.hands_emptied) {
492            return Next::Steps(vec![step(Action::FillHands)]);
493        }
494        if std::mem::take(&mut self.opened_sack) {
495            return Next::Put(format!("close {}", self.sack));
496        }
497        if done { Next::Done } else { Next::Failed }
498    }
499
500    fn look_at(&mut self, index: usize, seen: &Seen<'_>) -> Next {
501        if let Some(id) = self.passes.get(index) {
502            self.at = At::Read(index);
503            return Next::Ask(format!("look #{id}"), DESCRIBED);
504        }
505        if let Some(id) = &self.pass {
506            self.at = At::Got;
507            return Next::Put(format!("get #{id}"));
508        }
509        let buying = seen
510            .walker
511            .settings
512            .get(crate::settings::BUY_DAY_PASS)
513            .is_some_and(|is| {
514                let is = is.to_lowercase();
515                is == "yes" || is == "true" || is.split([' ', ';']).any(|word| word == self.route)
516            });
517        match (buying, self.from) {
518            (true, Some(from)) => {
519                self.at = At::Clerk;
520                Next::Go(from.step_in.to_owned())
521            }
522            _ => self.end(false),
523        }
524    }
525
526    fn read(&mut self, index: usize, seen: &Seen<'_>) -> Next {
527        let now = seen.state.game_time().map(i64::from);
528        let (Some(id), Some(from), Some(to)) = (self.passes.get(index), self.from, self.to) else {
529            return self.end(false);
530        };
531        // Unanswered, the look says nothing of this pass: it is not passed
532        // over for another bought.
533        if !super::holds(seen.answer, DESCRIBED) {
534            return self.end(false);
535        }
536        if let Some(pass) = read_pass(seen.answer) {
537            if pass.lapsed(now) {
538                self.at = At::Dropped(index);
539                return Next::Put(format!("_drag #{id} drop"));
540            }
541            if self.pass.is_none() && pass.serves(from.name, to.name, now) {
542                self.pass = Some(id.clone());
543            }
544            let id = id.clone();
545            self.note(&id, seen.answer, now);
546        }
547        self.look_at(index + 1, seen)
548    }
549
550    fn ask(&mut self, asked: u32) -> Next {
551        let (Some(from), Some(to)) = (self.from, self.to) else {
552            return self.end(false);
553        };
554        self.at = At::Asked(asked + 1);
555        Next::Ask(format!("ask {} for {}", from.clerk, to.ask_for), CLERK)
556    }
557
558    /// The clerk's answer to the second asking, or a later one.
559    fn asked(&mut self, asked: u32, seen: &Seen<'_>) -> Next {
560        let held = [&seen.state.right_hand, &seen.state.left_hand]
561            .into_iter()
562            .find(|hand| hand.noun() == Some("pass"))
563            .and_then(|hand| hand.id());
564        // The hands were emptied before the asking, so a pass in one is the
565        // purchase, whether or not the clerk's line was heard.
566        if seen.answered("quickly hands you") || held.is_some() {
567            let Some(id) = held else {
568                return self.end(true);
569            };
570            self.pass = Some(id.to_owned());
571            self.at = At::LookedAtNew;
572            return Next::Ask(format!("look #{id}"), DESCRIBED);
573        }
574        if seen.answered("don't have enough") {
575            let fetch = crate::settings::on(&seen.walker.settings, crate::settings::GET_SILVERS);
576            if fetch && !std::mem::replace(&mut self.banked, true) {
577                self.at = At::ToBank(0);
578                return self.bank(seen);
579            }
580            return self.end(false);
581        }
582        // Quoted a price: ask again, but only the once.
583        if seen.answered("says to you") && asked < 2 {
584            return self.ask(asked);
585        }
586        // Nobody answered the asking: not done, or the replan buys again.
587        self.end(!seen.answer.is_empty())
588    }
589
590    /// The next move of a walk, or `None` at its end.
591    fn walk(&mut self, dirs: &[&str], index: usize, at: fn(usize) -> At) -> Option<Next> {
592        let dir = dirs.get(index)?;
593        self.at = at(index + 1);
594        Some(Next::Go((*dir).to_owned()))
595    }
596
597    /// To the bank, the withdrawal, and back to the clerk.
598    fn bank(&mut self, seen: &Seen<'_>) -> Next {
599        let Some(from) = self.from else {
600            return self.end(false);
601        };
602        let walked = match self.at {
603            At::ToBank(index) => self.walk(from.to_bank, index, At::ToBank),
604            At::FromBank(index) => self.walk(from.from_bank, index, At::FromBank),
605            _ => self.walk(from.from_bank, 0, At::FromBank),
606        };
607        match (walked, self.at) {
608            (Some(next), _) => next,
609            (None, At::ToBank(_)) => {
610                self.at = At::Withdrew;
611                Next::Put("withdraw 5000".to_owned())
612            }
613            (None, _) => self.arrive(seen),
614        }
615    }
616
617    fn arrive(&mut self, seen: &Seen<'_>) -> Next {
618        let hiding = ["hidden", "invisible"]
619            .iter()
620            .any(|flag| seen.walker.flags.get(*flag) == Some(&true));
621        if hiding {
622            self.at = At::Unhidden;
623            return Next::Put("unhide".to_owned());
624        }
625        self.ask(0)
626    }
627
628    fn start(&mut self, seen: &Seen<'_>) -> Next {
629        let (Some(name), Some(_), Some(_)) = (
630            seen.walker.settings.get(crate::settings::DAY_PASS_SACK),
631            self.from,
632            self.to,
633        ) else {
634            return Next::Failed;
635        };
636        self.sack =
637            sack_in(seen.state, name).map_or_else(|| format!("my {name}"), |id| format!("#{id}"));
638        // A pass remembered and still good is got by its id, from wherever
639        // it is: no look in the sack.
640        let now = seen.state.game_time().map(i64::from);
641        let known = self.flag().and_then(|flag| {
642            let memory = seen.walker.memories.get(&flag)?;
643            remembered(memory, now).map(str::to_owned)
644        });
645        if let Some(id) = known {
646            self.pass = Some(id.clone());
647            self.remembered = Some(id);
648            self.hands_emptied = true;
649            self.at = At::Emptied;
650            return Next::Steps(vec![step(Action::EmptyHands)]);
651        }
652        self.at = At::LookedIn(false);
653        Next::Ask(format!("look in {}", self.sack), LOOKED_IN)
654    }
655}
656
657/// Find the sack among what is worn, four ways, in order: by noun, by whole
658/// name, by the end of the name, and by its words in order within the name.
659pub(in crate::walk) fn sack_in(state: &GameState, named: &str) -> Option<String> {
660    let lower = named.to_lowercase();
661    let worn = || state.inventory_snapshot.on_person();
662    let in_order = |name: &str| {
663        let name = name.to_lowercase();
664        let mut from = 0;
665        lower.split(' ').all(|word| {
666            name.get(from..)
667                .and_then(|rest| rest.find(word))
668                .is_some_and(|found| {
669                    from += found + word.len();
670                    true
671                })
672        })
673    };
674    worn()
675        .find(|item| item.noun == named)
676        .or_else(|| worn().find(|item| item.name == named))
677        .or_else(|| worn().find(|item| item.name.to_lowercase().ends_with(&lower)))
678        .or_else(|| worn().find(|item| in_order(&item.name)))
679        .map(|item| item.id.clone())
680}
681
682impl Solver for DayPass {
683    fn next(&mut self, seen: &Seen<'_>) -> Next {
684        let from = self.from;
685        match self.at {
686            At::Start => self.start(seen),
687            At::LookedIn(tried) => {
688                // Unanswered, the look says nothing of what is in the sack:
689                // a pass is not bought on the strength of it.
690                if !super::holds(seen.answer, LOOKED_IN) {
691                    return self.end(false);
692                }
693                if seen.answered("closed") {
694                    if tried {
695                        // Still shut: what is in it is not known.
696                        return self.end(false);
697                    }
698                    self.at = At::Opened;
699                    return Next::Put(format!("open {}", self.sack));
700                }
701                self.passes = passes_in(seen.answer);
702                self.at = At::Emptied;
703                // Emptied once only: a second emptying would owe back
704                // nothing.
705                if std::mem::replace(&mut self.hands_emptied, true) {
706                    return self.look_at(0, seen);
707                }
708                Next::Steps(vec![step(Action::EmptyHands)])
709            }
710            At::Opened => {
711                self.opened_sack = seen.answered("You open");
712                self.at = At::LookedIn(true);
713                Next::Ask(format!("look in {}", self.sack), LOOKED_IN)
714            }
715            At::Emptied => self.look_at(0, seen),
716            At::Read(index) => self.read(index, seen),
717            At::Dropped(index) => self.look_at(index + 1, seen),
718            // The remembered pass did not come to hand: forgotten, and the
719            // sack looked in as before.
720            At::Got if self.remembered.is_some() && !in_hand(seen, self.remembered.as_deref()) => {
721                self.remembered = None;
722                self.pass = None;
723                if let Some(flag) = self.flag() {
724                    self.notes.push(Action::Forget(flag));
725                }
726                self.at = At::LookedIn(false);
727                Next::Ask(format!("look in {}", self.sack), LOOKED_IN)
728            }
729            At::Got | At::SteppedBack => match &self.pass {
730                Some(id) => {
731                    self.at = At::Raised;
732                    Next::Put(format!("raise #{id}"))
733                }
734                None => self.end(true),
735            },
736            At::Raised => {
737                self.at = At::PutBack;
738                Next::Put(self.put_away())
739            }
740            // A remembered pass skips the look that would have found the
741            // sack shut: shut, it is opened, the pass put in, and the sack
742            // shut again at the end.
743            At::PutBack
744                if !self.opened_sack
745                    && seen.answered("closed")
746                    && in_hand(seen, self.pass.as_deref()) =>
747            {
748                self.opened_sack = true;
749                self.at = At::Reopened;
750                Next::Put(format!("open {}", self.sack))
751            }
752            At::Reopened => {
753                self.at = At::Ending(true);
754                Next::Put(self.put_away())
755            }
756            At::PutBack => self.end(true),
757            // A move of the bank walk failed: the walker is on the way,
758            // at neither end of the exit. Failed, so the exit is banned,
759            // rather than Done, which would only plan the trip again.
760            At::Clerk | At::ToBank(_) | At::FromBank(_) if !seen.ok => self.end(false),
761            At::Clerk => self.arrive(seen),
762            At::Unhidden => self.ask(0),
763            // The first asking is only quoted a price.
764            At::Asked(1) if !self.banked => self.ask(1),
765            At::Asked(asked) => self.asked(asked, seen),
766            At::ToBank(_) | At::Withdrew | At::FromBank(_) => self.bank(seen),
767            At::LookedAtNew => {
768                if let Some(id) = self.pass.clone() {
769                    let now = seen.state.game_time().map(i64::from);
770                    self.note(&id, seen.answer, now);
771                }
772                self.at = At::SteppedBack;
773                Next::Go(from.map_or("", |town| town.step_back).to_owned())
774            }
775            At::Ending(done) => self.end(done),
776        }
777    }
778}

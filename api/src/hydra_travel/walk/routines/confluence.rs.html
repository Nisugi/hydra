1//! `Routine::Confluence`: the Elemental Confluence (3,234 exits).
2//!
3//! Every exit of the plane is this one routine. The plane's rooms keep their
4//! numbers and **reshuffle their exits**, so no map of it stays true; the
5//! routine explores instead:
6//!
7//! - it writes down where each exit of each room led ([`Learned::wayto`]),
8//!   and **forgets all of it** the moment a room's exits are not the ones it
9//!   wrote down -- the plane has shifted;
10//! - to reach a room it searches what it has learned backwards from the
11//!   target, thirty rooms deep ([`Learned::dir_to`]); knowing no way, it
12//!   heads for the nearest exit it has never taken; with none of those, for
13//!   the neighbour it visited longest ago ([`Learned::wander`]);
14//! - the plane is two halves, hot and cold, joined only by a *gaping
15//!   bottomless pit* that wanders, and left only by a *point of elemental
16//!   tranquility* that wanders too. Each is remembered where last seen and
17//!   forgotten when it is seen to be gone.
18//!
19//! A move that fails is answered with a random exit; the exits are already
20//! known, so there is no need to look first.
21//!
22//! What is learned is [`Learned`], which the driver keeps for the trip and
23//! lends to each crossing in turn: one walk through the plane is many of
24//! these exits, and each would otherwise start knowing nothing.
25//!
26//! Not done here: waiting for an escorted child to catch up, which is the
27//! bounty behaviour's business when there is one.
28

35/// Moves one crossing may make. The plane has 53 rooms, and this is room to
36/// learn it several times over.
37const MAX_MOVES: u32 = 500;
38/// How deep the search back from a target goes.
39const MAX_DEPTH: usize = 30;
40
41const TRANQUILITY: &str = "point of elemental tranquility";
42const PIT: &str = "gaping bottomless pit";
43
44/// Which half of the plane a room is in, by runs of ids.
45fn is_hot(room: RoomId) -> Option<bool> {
46    match room.0 {
47        23282..=23303 | 23329..=23334 => Some(true),
48        23304..=23328 => Some(false),
49        _ => None,
50    }
51}

53/// What has been learned of the plane. Kept between crossings.
54#[derive(Debug, Clone, Default, PartialEq, Eq)]
55pub(in crate::walk) struct Learned {
56    /// Each known room's exits, in the game's order, and where each led.
57    wayto: HashMap<RoomId, Vec<(String, Option<RoomId>)>>,
58    /// Rooms visited, the most recent last.
59    wander: Vec<RoomId>,
60    /// Where the way out and the way across were last seen: `[cold, hot]`.
61    tranquility: [Option<RoomId>; 2],
62    pit: [Option<RoomId>; 2],
63}
64
65impl Learned {
66    /// An exit of `from` that leads towards any of `targets`, searching
67    /// backwards through what is known. `None` among the targets means "an
68    /// exit never taken".
69    fn dir_to(&self, from: RoomId, targets: &[Option<RoomId>]) -> Option<String> {
70        let exits = self.wayto.get(&from)?;
71        let mut targets = targets.to_vec();
72        let mut tried: Vec<Option<RoomId>> = Vec::new();
73        for _ in 0..MAX_DEPTH {
74            if let Some((dir, _)) = exits.iter().find(|(_, to)| targets.contains(to)) {
75                return Some(dir.clone());
76            }
77            for target in &targets {
78                if !tried.contains(target) {
79                    tried.push(*target);
80                }
81            }
82            // The rooms known to lead to a target are the next targets.
83            let mut nearer: Vec<Option<RoomId>> = self
84                .wayto
85                .iter()
86                .filter(|(room, exits)| {
87                    exits.iter().any(|(_, to)| targets.contains(to))
88                        && !tried.contains(&Some(**room))
89                })
90                .map(|(room, _)| Some(*room))
91                .collect();
92            if nearer.is_empty() {
93                return None;
94            }
95            // A `HashMap` has no order, and a replay must choose the same.
96            nearer.sort_unstable();
97            targets = nearer;
98        }
99        None
100    }
101
102    /// Note whether `thing` is here, or is gone from where it was.
103    fn note(seen_at: &mut Option<RoomId>, here: RoomId, is_here: bool) {
104        if is_here {
105            *seen_at = Some(here);
106        } else if *seen_at == Some(here) {
107            *seen_at = None;
108        }
109    }
110
111    /// Make sure this room's exits are written down, forgetting everything
112    /// if they are not the ones that were.
113    fn look_at(&mut self, here: RoomId, exits: &[String]) {
114        let same = self
115            .wayto
116            .get(&here)
117            .is_none_or(|known| known.iter().map(|(dir, _)| dir).eq(exits.iter()));
118        if !same {
119            self.wayto.clear();
120        }
121        self.wayto
122            .entry(here)
123            .or_insert_with(|| exits.iter().map(|dir| (dir.clone(), None)).collect());
124    }
125
126    fn landed(&mut self, from: RoomId, dir: &str, at: RoomId) {
127        if let Some(exit) = self
128            .wayto
129            .get_mut(&from)
130            .and_then(|exits| exits.iter_mut().find(|(known, _)| known == dir))
131        {
132            exit.1 = Some(at);
133        }
134        self.wander.retain(|room| *room != at);
135        self.wander.push(at);
136    }
137
138    /// The last two resorts: an exit to somewhere not yet wandered
139    /// through, and then the neighbour visited longest ago.
140    fn wander_from(&self, here: RoomId) -> Option<String> {
141        let exits = self.wayto.get(&here)?;
142        let fresh = exits
143            .iter()
144            .find(|(_, to)| to.is_none_or(|to| !self.wander.contains(&to)));
145        let stale = || {
146            let oldest = self
147                .wander
148                .iter()
149                .find(|room| exits.iter().any(|(_, to)| *to == Some(**room)))?;
150            exits.iter().find(|(_, to)| *to == Some(*oldest))
151        };
152        fresh.or_else(stale).map(|(dir, _)| dir.clone())
153    }
154}

156pub(in crate::walk) struct Confluence {
157    /// The goal is to leave the plane, by the point of tranquility.
158    leave: bool,
159    learned: Learned,
160    /// The move under way: the room it left, and by which exit.
161    moving: Option<(RoomId, String)>,
162    moves: u32,
163    /// `go tranquility` has been sent.
164    left: bool,
165}
166
167impl Confluence {
168    pub fn new(leave: bool, learned: Learned) -> Self {
169        Confluence {
170            leave,
171            learned,
172            moving: None,
173            moves: 0,
174            left: false,
175        }
176    }
177
178    fn go(&mut self, from: RoomId, dir: String) -> Next {
179        self.moves += 1;
180        self.moving = Some((from, dir.clone()));
181        Next::Go(dir)
182    }
183
184    /// The exit to take from `here`, by everything there is to try, in turn.
185    fn choose(&self, here: RoomId, hot: bool, goal: RoomId) -> Option<String> {
186        let learned = &self.learned;
187        let half = usize::from(hot);
188        let known = if self.leave {
189            learned.tranquility[half].and_then(|at| learned.dir_to(here, &[Some(at)]))
190        } else if is_hot(goal).is_some_and(|goal_hot| goal_hot != hot) {
191            learned.pit[half].and_then(|at| learned.dir_to(here, &[Some(at)]))
192        } else {
193            learned.dir_to(here, &[Some(goal)])
194        };
195        known
196            .or_else(|| learned.dir_to(here, &[None]))
197            .or_else(|| learned.wander_from(here))
198    }
199}
200
201impl Solver for Confluence {
202    fn keep(self: Box<Self>, kept: &mut super::Kept) {
203        kept.confluence = self.learned;
204    }
205
206    fn next(&mut self, seen: &Seen<'_>) -> Next {
207        let (Some(here), false) = (seen.here, self.left) else {
208            return Next::Done;
209        };
210        let exits = seen.walker.exits.as_deref().unwrap_or(&[]);
211        if let Some((from, dir)) = self.moving.take() {
212            if !seen.ok {
213                // Any exit will do; the exits are known without a look.
214                let Some(any) = pick(exits, seen.random) else {
215                    return Next::Failed;
216                };
217                self.moves += 1;
218                return Next::Go(any.clone());
219            }
220            if here != from {
221                self.learned.landed(from, &dir, here);
222            }
223        }
224        if !self.leave && here == seen.goal {
225            return Next::Done;
226        }
227        // Out of the plane: fallen, carried, or never in it. Plan again.
228        let Some(hot) = is_hot(here) else {
229            return Next::Done;
230        };
231        if self.moves >= MAX_MOVES || exits.is_empty() {
232            return Next::Failed;
233        }
234        let shows = |name: &str| {
235            seen.state
236                .room
237                .objects
238                .iter()
239                .any(|thing| thing.text == name)
240        };
241        let half = usize::from(hot);
242        Learned::note(
243            &mut self.learned.tranquility[half],
244            here,
245            shows(TRANQUILITY),
246        );
247        Learned::note(&mut self.learned.pit[half], here, shows(PIT));
248        self.learned.look_at(here, exits);
249
250        if self.leave && shows(TRANQUILITY) {
251            self.left = true;
252            return Next::Go("go tranquility".to_owned());
253        }
254        let across = !self.leave && is_hot(seen.goal).is_some_and(|goal_hot| goal_hot != hot);
255        if across && shows(PIT) {
256            self.moves += 1;
257            return Next::Go("go pit".to_owned());
258        }
259        match self.choose(here, hot, seen.goal) {
260            Some(dir) => self.go(here, dir),
261            None => Next::Failed,
262        }
263    }
264}
265
266fn pick(exits: &[String], random: u64) -> Option<&String> {
267    let len = u64::try_from(exits.len()).ok().filter(|len| *len > 0)?;
268    exits.get(usize::try_from(random % len).ok()?)
269}

283    fn room(here: u32, goal: u32, exits: &[&str]) -> Scene {
284        let mut scene = Scene::at(here, goal);
285        scene.walker.exits = Some(exits.iter().map(|dir| (*dir).to_owned()).collect());
286        scene
287    }

289    fn showing(mut scene: Scene, name: &str) -> Scene {
290        scene.state.room.objects.push(RoomItem {
291            id: "9".into(),
292            noun: "x".into(),
293            text: name.into(),
294            before: None,
295            after: None,
296            status: None,
297        });
298        scene

301    #[test]
302    fn the_halves_are_the_known_runs_of_rooms() {
303        let hot = (23000..24000).filter(|id| is_hot(RoomId(*id)) == Some(true));
304        let cold = (23000..24000).filter(|id| is_hot(RoomId(*id)) == Some(false));
305        assert_eq!((hot.count(), cold.count()), (28, 25));
306        assert_eq!(is_hot(RoomId(23328)), Some(false));
307        assert_eq!(is_hot(RoomId(23329)), Some(true));
308        assert_eq!(is_hot(RoomId(1)), None);
309    }

311    #[test]
312    fn it_explores_what_it_has_never_taken_and_then_knows_the_way() {
313        let mut plane = Confluence::new(false, Learned::default());
314        // Nothing known: the first exit never taken.
315        assert_eq!(
316            room(A, C, &["n", "e"]).ask(&mut plane),
317            Next::Go("n".into())
318        );
319        // `n` led to B, which is a dead end but for the way back.
320        assert_eq!(room(B, C, &["s"]).ask(&mut plane), Next::Go("s".into()));
321        // Back in A: `n` is known now, so the exit never taken is `e`.
322        assert_eq!(
323            room(A, C, &["n", "e"]).ask(&mut plane),
324            Next::Go("e".into())
325        );
326        assert_eq!(room(C, C, &["w"]).ask(&mut plane), Next::Done);
327
328        // The next crossing is lent what this one learned, and goes straight.
329        let mut again = Confluence::new(false, plane.learned);
330        assert_eq!(
331            room(A, C, &["n", "e"]).ask(&mut again),
332            Next::Go("e".into())
333        );
334    }

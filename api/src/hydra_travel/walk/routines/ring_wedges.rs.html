1//! `Routine::RingWedges`: the stone ring of room 8373.
2//!
3//! For each wedge in turn --
4//! northern, eastern, western, southern -- `look <which> wedge`, and while it
5//! is not "currently pointed at a sigil of" the one that is its own, `turn
6//! ring` and look again; then `push <which> wedge`. Nothing more: the trip
7//! looks at where the last push left the walker.
8//!
9//! An answer that names no sigil is looked at again rather than turned on:
10//! the look has not been answered yet. Every command sent counts toward
11//! [`MAX_TURNS`], so a ring that never comes round does not turn for ever.
12
13use hydra_model::ChunkLine;
14
15use super::{MAX_TURNS, Next, Seen, Solver};
16
17/// Each wedge and the sigil it must point at, in the order they are set.
18const WEDGES: &[(&str, &str)] = &[
19    (
20        "northern",
21        "a large thick torus surrounded by nine tiny circles",
22    ),
23    (
24        "eastern",
25        "two crossed upside-down hammers circumscribed by a rounded arc",
26    ),
27    (
28        "western",
29        "a jagged triangular arch bisected by a vertical line",
30    ),
31    (
32        "southern",
33        "three pairs of obliquely intersecting parallel lines",
34    ),
35];
36
37#[derive(Default)]
38pub(super) struct RingWedges {
39    /// The position in [`WEDGES`] of the wedge being set.
40    wedge: usize,
41    /// Whether the last thing sent was the look, so its answer is the wedge's.
42    looked: bool,
43    turns: u32,
44}
45
46impl RingWedges {
47    fn put(&mut self, looked: bool, command: String) -> Next {
48        if self.turns >= MAX_TURNS {
49            return Next::Failed;
50        }
51        self.turns += 1;
52        self.looked = looked;
53        Next::Put(command)
54    }
55}
56
57impl Solver for RingWedges {
58    fn next(&mut self, seen: &Seen<'_>) -> Next {
59        let Some((which, sigil)) = WEDGES.get(self.wedge) else {
60            return Next::Done;
61        };
62        let points = format!("The stone ring's {which} wedge is currently pointed at a sigil of ");
63        let pointed = seen
64            .answer
65            .iter()
66            .map(ChunkLine::text)
67            .find(|text| text.contains(&points));
68        match pointed {
69            Some(text) if self.looked && text.contains(sigil) => {
70                self.wedge += 1;
71                self.put(false, format!("push {which} wedge"))
72            }
73            Some(_) if self.looked => self.put(false, "turn ring".to_owned()),
74            _ => self.put(true, format!("look {which} wedge")),
75        }
76    }
77}

79#[cfg(test)]
80mod tests {
81    use super::super::testing::Scene;
82    use super::*;
83
84    fn put(command: &str) -> Next {
85        Next::Put(command.to_owned())
86    }
87
88    fn pointed(which: &str, sigil: &str) -> String {
89        format!("The stone ring's {which} wedge is currently pointed at a sigil of {sigil}.")
90    }
91
92    fn hears(ring: &mut RingWedges, line: &str) -> Next {
93        Scene::at(1, 2).answered(&[line]).ask(ring)
94    }
95
96    #[test]
97    fn the_ring_is_turned_until_the_wedge_points_home_and_then_pushed() {
98        let mut ring = RingWedges::default();
99        assert_eq!(Scene::at(1, 2).ask(&mut ring), put("look northern wedge"));
100        let wrong = pointed("northern", WEDGES[3].1);
101        assert_eq!(hears(&mut ring, &wrong), put("turn ring"));
102        // The answer to the turn is not the wedge's, whatever it holds.
103        let right = pointed("northern", WEDGES[0].1);
104        assert_eq!(hears(&mut ring, &right), put("look northern wedge"));
105        assert_eq!(hears(&mut ring, &right), put("push northern wedge"));
106        assert_eq!(hears(&mut ring, "Click."), put("look eastern wedge"));
107    }
108
109    #[test]
110    fn every_wedge_is_set_in_order_and_that_is_all() {
111        let mut ring = RingWedges::default();
112        for (which, sigil) in WEDGES {
113            assert_eq!(
114                Scene::at(1, 2).ask(&mut ring),
115                put(&format!("look {which} wedge"))
116            );
117            assert_eq!(
118                hears(&mut ring, &pointed(which, sigil)),
119                put(&format!("push {which} wedge"))
120            );
121        }
122        assert_eq!(Scene::at(1, 2).ask(&mut ring), Next::Done);
123    }
124
125    #[test]
126    fn the_table_pairs_each_wedge_with_its_sigil() {
127        let expected = [
128            "'northern' => 'a large thick torus surrounded by nine tiny circles'",
129            "'eastern' => 'two crossed upside-down hammers circumscribed by a rounded arc'",
130            "'western' => 'a jagged triangular arch bisected by a vertical line'",
131            "'southern' => 'three pairs of obliquely intersecting parallel lines'",
132        ];
133        let ours: Vec<String> = WEDGES
134            .iter()
135            .map(|(which, sigil)| format!("'{which}' => '{sigil}'"))
136            .collect();
137        assert_eq!(ours, expected);
138    }
139
140    #[test]
141    fn another_wedges_sigil_is_never_home() {
142        for (at, (which, _)) in WEDGES.iter().enumerate() {
143            for (other, (_, sigil)) in WEDGES.iter().enumerate() {
144                if at == other {
145                    continue;
146                }
147                let mut ring = RingWedges {
148                    wedge: at,
149                    looked: true,
150                    turns: 0,
151                };
152                assert_eq!(hears(&mut ring, &pointed(which, sigil)), put("turn ring"));
153            }
154        }
155    }
156
157    #[test]
158    fn a_look_the_game_did_not_answer_is_looked_again() {
159        let mut ring = RingWedges::default();
160        Scene::at(1, 2).ask(&mut ring);
161        assert_eq!(
162            hears(&mut ring, "...wait 1 seconds."),
163            put("look northern wedge")
164        );
165        // Another wedge's line is not this one's answer either.
166        let eastern = pointed("eastern", WEDGES[0].1);
167        assert_eq!(hears(&mut ring, &eastern), put("look northern wedge"));
168    }
169
170    #[test]
171    fn a_ring_that_never_comes_round_is_given_up() {
172        let mut ring = RingWedges::default();
173        let wrong = pointed("northern", WEDGES[1].1);
174        for _ in 0..MAX_TURNS {
175            assert!(matches!(hears(&mut ring, &wrong), Next::Put(_)));
176        }
177        assert_eq!(hears(&mut ring, &wrong), Next::Failed);
178    }

1//! Into the Elemental Confluence: find an instability of the walker's
2//! element and push it with a soulstone. The reading is pure
3//! ([`super::super::instability`]); this asks, walks and looks.
4//!
5//! In this order:
6//!
7//! 1. The walker's element: kept as the memory `element`, else `attune`.
8//! 2. The instability `.go2` was told (`--instability=<room>`), if it was.
9//! 3. An instability of that element seen in the last six hours, within ten
10//!    seconds' walk.
11//! 4. Else `attune sense`, up to twice -- walking to the nearest room it
12//!    rules out between -- which narrows the map to where one may be.
13//! 5. Each of those, nearest first: walk there, and `look instability` if
14//!    the room shows one. Every one seen is kept, whatever its element.
15//! 6. Found: `push instability with my soulstone`, and the room it was in
16//!    kept as the memory `last_instability`, which prices the way out.
17//!
18//! Nothing turns room descriptions on around `attune sense`: Hydra's story
19//! always has the room's description.
20

32/// The rooms one search may look in: a stop, so a search that keeps finding
33/// rooms to look in does not walk for ever.
34const MAX_LOOKS: usize = 200;
35
36/// The memory the walker's element is kept under.
37pub const ELEMENT: &str = "element";
38/// The memory the instability the walker entered by is kept under.
39pub const LAST_INSTABILITY: &str = "last_instability";

41impl<F: FnMut(&[u32]), W: FnMut(&BTreeMap<String, String>)> Driver<'_, F, W> {
42    /// Into the plane as `into` asks. `Some(room)`: walk on to there from
43    /// inside; `None`: there is nothing more to walk.
44    pub(super) async fn enter_the_plane(
45        &mut self,
46        trip: &mut Trip,
47        map: &Map,
48        setting: &Setting,
49        into: Into,
50    ) -> Result<Option<RoomId>, Ended> {
51        let element = self.element(trip).await?;
52        let found = self.find_instability(trip, map, setting, &element).await?;
53        if !found {
54            return Err(Ended::Halted(
55                "I could not find an instability of your element.".to_owned(),
56            ));
57        }
58        if into == Into::Instability {
59            return Ok(None);
60        }
61        let entered = self.locate(map);
62        let unseen = ["hidden", "invisible"]
63            .iter()
64            .any(|flag| self.walker(setting).flags.get(*flag) == Some(&true));
65        if unseen {
66            self.put(trip, "unhide").await?;
67        }
68        let push = Step {
69            action: Action::Move("push instability with my soulstone".to_owned()),
70            when: None,
71        };
72        if !self.aside(trip, map, setting, vec![push]).await? {
73            return Err(Ended::Halted(
74                "the instability would not let me through: is there a soulstone on you?".to_owned(),
75            ));
76        }
77        if let Some(entered) = entered {
78            self.memories
79                .insert(LAST_INSTABILITY.to_owned(), entered.0.to_string());
80            (self.wrote)(&self.memories);
81        }
82        Ok(match into {
83            Into::Room(room) => Some(room),
84            Into::Hot | Into::Cold => self.nearest_half(trip, map, setting, into == Into::Hot),
85            Into::Plane | Into::Instability => None,
86        })
87    }

89    /// The walker's element: kept, or asked.
90    async fn element(&mut self, trip: &mut Trip) -> Result<String, Ended> {
91        if let Some(element) = self.memories.get(ELEMENT) {
92            return Ok(element.clone());
93        }
94        let answer = self.put(trip, "attune").await?;
95        let lines: Vec<String> = answer.iter().map(hydra_model::ChunkLine::text).collect();
96        match element_in(&lines) {
97            Some(Ok(element)) => {
98                self.memories.insert(ELEMENT.to_owned(), element.clone());
99                (self.wrote)(&self.memories);
100                Ok(element)
101            }
102            Some(Err(())) => Err(Ended::Halted(
103                "you are not attuned to an element, so I do not know how to find an instability."
104                    .to_owned(),
105            )),
106            None => Err(Ended::Halted(
107                "I could not tell which element you are attuned to.".to_owned(),
108            )),
109        }
110    }

112    /// The search, to a room with an instability of `element`. `false`:
113    /// none was found.
114    async fn find_instability(
115        &mut self,
116        trip: &mut Trip,
117        map: &Map,
118        setting: &Setting,
119        element: &str,
120    ) -> Result<bool, Ended> {
121        if let Some(told) = setting.instability_at {
122            return Ok(self.walk_to(trip, map, setting, RoomId(told)).await?
123                && self.look_for(trip, map, element).await?);
124        }
125        // Seen lately, near enough to look before sensing.
126        let now = setting.now;
127        self.instabilities
128            .retain(|seen| seen.seen + RECENT_SECONDS >= now);
129        let recent: Vec<RoomId> = self
130            .instabilities
131            .iter()
132            .filter(|seen| seen.element == element)
133            .map(|seen| RoomId(seen.room))
134            .collect();
135        for room in recent {
136            let Some(here) = self.locate(map) else { break };
137            let walker = self.walker(setting);
138            let routes = map.routes(here, Target::Room(room), trip.pricing(&walker));
139            if routes
140                .seconds_to(room)
141                .is_some_and(|seconds| seconds <= NEAR_SECONDS)
142                && self.walk_to(trip, map, setting, room).await?
143                && self.look_for(trip, map, element).await?
144            {
145                return Ok(true);
146            }
147        }
148        // Sensed: where one may be, and where not.
149        let (mut good, mut bad) = (Vec::new(), Vec::new());
150        for _ in 0..2 {
151            let Some(here) = self.locate(map) else { break };
152            let answer = self.put(trip, "attune sense").await?;
153            let lines: Vec<String> = answer.iter().map(hydra_model::ChunkLine::text).collect();
154            let sensed = senses(&lines);
155            let pulled = sensed
156                .iter()
157                .any(|sense| matches!(sense, super::super::instability::Sensed::Pulled { .. }));
158            let Some((more_good, more_bad)) = narrow(map, here, &sensed) else {
159                return Ok(false);
160            };
161            for room in more_good {
162                if !good.contains(&room) {
163                    good.push(room);
164                }
165            }
166            for room in more_bad {
167                if !bad.contains(&room) && !good.contains(&room) {
168                    bad.push(room);
169                }
170            }
171            if pulled || !good.is_empty() {
172                break;
173            }
174            let Some(nearer) = self.nearest(trip, map, setting, &bad) else {
175                break;
176            };
177            self.walk_to(trip, map, setting, nearer).await?;
178        }
179        // Each, nearest first: where one should be, then where one of the
180        // element was seen lately, then the rest.
181        for _ in 0..MAX_LOOKS {
182            let seen_of_element: Vec<RoomId> = bad
183                .iter()
184                .copied()
185                .filter(|room| {
186                    self.instabilities
187                        .iter()
188                        .any(|seen| seen.room == room.0 && seen.element == element)
189                })
190                .collect();
191            let next = self
192                .nearest(trip, map, setting, &good)
193                .or_else(|| self.nearest(trip, map, setting, &seen_of_element))
194                .or_else(|| self.nearest(trip, map, setting, &bad));
195            let Some(next) = next else {
196                return Ok(false);
197            };
198            good.retain(|room| *room != next);
199            bad.retain(|room| *room != next);
200            if self.walk_to(trip, map, setting, next).await?
201                && self.look_for(trip, map, element).await?
202            {
203                return Ok(true);
204            }
205        }
206        Ok(false)
207    }

209    /// The nearest of `rooms` the walker can be walked to.
210    fn nearest(
211        &self,
212        trip: &Trip,
213        map: &Map,
214        setting: &Setting,
215        rooms: &[RoomId],
216    ) -> Option<RoomId> {
217        if rooms.is_empty() {
218            return None;
219        }
220        let here = self.was.map(|(_, room)| room)?;
221        let walker = self.walker(setting);
222        map.routes(here, Target::Nearest(rooms), trip.pricing(&walker))
223            .reached()
224    }

226    /// Whether the room shows an instability of `element`. Each one seen is kept, whatever its element;
227    /// a room that shows none is forgotten as one.
228    async fn look_for(&mut self, trip: &mut Trip, map: &Map, element: &str) -> Result<bool, Ended> {
229        let Some(here) = self.locate(map) else {
230            return Ok(false);
231        };
232        let shows = self
233            .state
234            .room
235            .objects
236            .iter()
237            .any(|thing| thing.noun == "instability");
238        if !shows {
239            self.instabilities.retain(|seen| seen.room != here.0);
240            self.instabilities_changed = true;
241            return Ok(false);
242        }
243        let answer = self.put(trip, "look instability").await?;
244        let lines: Vec<String> = answer.iter().map(hydra_model::ChunkLine::text).collect();
245        let Some(given) = element_given_off(&lines) else {
246            return Ok(false);
247        };
248        let known = self
249            .instabilities
250            .iter()
251            .any(|seen| seen.room == here.0 && seen.element == given);
252        if !known {
253            self.instabilities.push(Seen {
254                room: here.0,
255                element: given.to_owned(),
256                seen: self.now,
257            });
258            self.instabilities_changed = true;
259        }
260        Ok(given == element)
261    }

263    /// The nearest room of the hot half, or the cold, from inside: the rooms
264    /// of its huge elementals ([`half_tag`]).
265    fn nearest_half(&self, trip: &Trip, map: &Map, setting: &Setting, hot: bool) -> Option<RoomId> {
266        let tagged: Vec<RoomId> = map
267            .rooms()
268            .iter()
269            .filter(|room| room.tags.iter().any(|tag| half_tag(hot, tag)))
270            .map(|room| room.id)
271            .collect();
272        self.nearest(trip, map, setting, &tagged)
273    }

1//! Asking what a trip must know before it is priced, and fetching the
2//! silver it will be asked for. The reading is pure
3//! ([`super::super::preflight`], the day pass's readers); this sends,
4//! walks and stops.
5//!
6//! # The silver detour
7//!
8//! Add up what the route's exits charge -- and the way back, if the Travel
9//! page says so -- and compare it with `wealth`. Short, and allowed the bank
10//! (`get_silvers`): walk to the nearest bank the silver in hand can reach,
11//! price the walk again from there, withdraw the difference, and look again.
12//! Short and not allowed: say so, and go; the ferryman will say no soon
13//! enough, and the trip plans round him. **The walk halts** when the walker
14//! is too poor to reach a bank, or the bank has not enough -- both for the
15//! player to settle.
16//!
17//! **Silver nobody has stated is not no silver**: if `wealth quiet` leaves
18//! the model unable to say what is carried, this says so and goes, rather
19//! than walk a purse of platinum to the bank.

32    /// Before the first plan. `goal` is where the trip is going.
33    pub(super) async fn preflight(
34        &mut self,
35        trip: &mut Trip,
36        map: &cena_map::Map,
37        setting: &Setting,
38    ) -> Result<(), Ended> {
39        self.out_of_a_shop(trip, map).await?;
40        let goal = trip.goal();
41        // Asked once, and only while nobody has said: the model reads the
42        // answer and keeps it, so a walk does not ask at every room.
43        let urchins = self
44            .state
45            .character
46            .standing
47            .urchins(self.state.game_time_now());
48        if setting.settings.on("use_urchins") && urchins.is_none() {
49            self.put(trip, "urchin status").await?;
50        }
51        self.day_passes(trip, setting).await?;
52        // Silver is priced from a room; an unplaced walker is the walk's to
53        // wait for, and the ferryman's to refuse.
54        let Some(here) = self.locate(map) else {
55            return Ok(());
56        };
57        self.by_gigas(trip, map, setting, here, goal).await?;
58        let Some(here) = self.locate(map) else {
59            return Ok(());
60        };
61        let needed = self.silver_needed(trip, map, setting, here, goal);
62        if needed == 0 {
63            return Ok(());
64        }
65        let Some(have) = self.silver(trip).await? else {
66            self.warn(&format!(
67                "this route asks for {needed} silver, and I could not tell how much you carry, \
68                 so I am going anyway."
69            ));
70            return Ok(());
71        };
72        if have >= needed {
73            return Ok(());
74        }
75        if !setting.settings.on("get_silvers") {
76            self.warn(&format!(
77                "this route asks for {needed} silver and you carry {have}. Fetch silver from \
78                 the bank is off, so I am going anyway."
79            ));
80            return Ok(());
81        }
82        self.by_way_of_the_bank(trip, map, setting, (here, goal), have)
83            .await
84    }

91    /// Which day passes the walker holds, asked once of a walker that uses
92    /// passes and names the sack they are kept in; each that still serves
93    /// becomes the flag the map asks for.
94    async fn day_passes(&mut self, trip: &mut Trip, setting: &Setting) -> Result<(), Ended> {
95        let sack = setting.settings.0.get("day_pass_sack");
96        let Some(sack) =
97            sack.filter(|sack| !sack.is_empty() && setting.settings.on("use_day_pass"))
98        else {
99            return Ok(());
100        };
101        let sack = day_pass::sack_in(&self.state, sack)
102            .map_or_else(|| format!("my {sack}"), |id| format!("#{id}"));
103        let mut inside = self.put(trip, &format!("look in {sack}")).await?;
104        let shut = inside.iter().any(|line| line.text().contains("closed"));
105        if shut {
106            self.put(trip, &format!("open {sack}")).await?;
107            inside = self.put(trip, &format!("look in {sack}")).await?;
108        }
109        let now = self.state.game_time().map(i64::from);
110        for id in day_pass::passes_in(&inside) {
111            let looked = self.put(trip, &format!("look #{id}")).await?;
112            let Some(pass) = day_pass::read_pass(&looked) else {
113                continue;
114            };
115            let flag = pass.towns.as_ref().and_then(|(one, other)| {
116                pass.serves(one, other, now)
117                    .then(|| day_pass::flag_for(one, other))?
118            });
119            if let Some(flag) = flag {
120                self.found.insert(flag, true);
121            }
122        }
123        if shut {
124            self.put(trip, &format!("close {sack}")).await?;
125        }
126        Ok(())
127    }

129    /// Out of a player's shop, which no map holds, when the walk sets out in
130    /// a room the map does not know ([`out_of_a_shop`]): then on the map.
131    async fn out_of_a_shop(&mut self, trip: &mut Trip, map: &cena_map::Map) -> Result<(), Ended> {
132        if self.locate(map).is_some() {
133            return Ok(());
134        }
135        let Some(commands) = out_of_a_shop(&self.state.room) else {
136            return Ok(());
137        };
138        for command in commands {
139            self.put(trip, &command).await?;
140        }
141        Ok(())
142    }

144    /// To or from the Hinterwilds by gigas fragments, where the route takes
145    /// the long way ([`gigas_way`]), the Travel page says so, and `wealth
146    /// gigas` counts at least the fragments kept back: the teleporter
147    /// walked to and used. Which side was entered by is remembered
148    /// (`hinterwilds_location`) while the walker is there. The trip then
149    /// plans from wherever that landed.
150    async fn by_gigas(
151        &mut self,
152        trip: &mut Trip,
153        map: &cena_map::Map,
154        setting: &Setting,
155        here: RoomId,
156        goal: RoomId,
157    ) -> Result<(), Ended> {
158        if !setting.settings.on("use_gigas_hwtravel") {
159            return Ok(());
160        }
161        let walker = self.walker(setting);
162        let Some(path) = trip.path_from(map, &walker, here, goal) else {
163            return Ok(());
164        };
165        let Some(way) = gigas_way(map, &path, here, goal) else {
166            return Ok(());
167        };
168        self.put(trip, "wealth gigas").await?;
169        let fragments = self
170            .state
171            .character
172            .currency
173            .gigas_artifact_fragments
174            .unwrap_or(0);
175        if fragments < gigas_least(&setting.settings) {
176            return Ok(());
177        }
178        let (teleporter, commands, side) = (way.teleporter, way.commands, way.side);
179        if !self.walk_to(trip, map, setting, teleporter).await? {
180            return Ok(());
181        }
182        for command in commands {
183            self.put(trip, command).await?;
184        }
185        // The side entered by is remembered only when the walker landed in
186        // the workroom; anywhere else it is forgotten.
187        let landed = self.locate(map);
188        let there = landed.is_some_and(|room| map.ids_for_uid(Uid(WORKROOM)).contains(&room));
189        match side.filter(|_| there) {
190            Some(side) => self
191                .memories
192                .insert("hinterwilds_location".to_owned(), side.to_owned()),
193            None => self.memories.remove("hinterwilds_location"),
194        };
195        (self.wrote)(&self.memories);
196        Ok(())
197    }

199    /// What the route there -- and back, if the Travel page asks -- charges.
200    fn silver_needed(
201        &self,
202        trip: &Trip,
203        map: &cena_map::Map,
204        setting: &Setting,
205        from: RoomId,
206        goal: RoomId,
207    ) -> u64 {
208        let walker = self.walker(setting);
209        let leg = |from, to| {
210            trip.path_from(map, &walker, from, to)
211                .map_or(0, |path| silver_for(map, &path))
212        };
213        let back = if setting.settings.on("get_return_trip_silvers") {
214            leg(goal, from)
215        } else {
216            0
217        };
218        leg(from, goal) + back
219    }

221    /// `wealth`, as the model read it. `None`: nobody has said, which is not
222    /// zero.
223    async fn silver(&mut self, trip: &mut Trip) -> Result<Option<u64>, Ended> {
224        self.put(trip, "wealth quiet").await?;
225        Ok(self.state.character.currency.silver)
226    }

228    async fn by_way_of_the_bank(
229        &mut self,
230        trip: &mut Trip,
231        map: &cena_map::Map,
232        setting: &Setting,
233        (here, goal): (RoomId, RoomId),
234        have: u64,
235    ) -> Result<(), Ended> {
236        let walker = self.walker(setting);
237        let mut banks: Vec<(f64, RoomId)> = map
238            .rooms()
239            .iter()
240            .filter(|room| room.tags.iter().any(|tag| tag == "bank"))
241            .filter_map(|room| {
242                let path = trip.path_from(map, &walker, here, room.id)?;
243                (silver_for(map, &path) <= have).then_some(())?;
244                let routes = map.routes(here, Target::Room(room.id), trip.pricing(&walker));
245                Some((routes.seconds_to(room.id)?, room.id))
246            })
247            .collect();
248        banks.sort_by(|(a, _), (b, _)| a.total_cmp(b));
249        let Some((_, bank)) = banks.first().copied() else {
250            return Err(halted("you are too poor to reach a bank."));
251        };
252        if !self.walk_to(trip, map, setting, bank).await? {
253            return Err(halted("I could not get to the bank."));
254        }
255        let needed = self.silver_needed(trip, map, setting, bank, goal);
256        // Known before the walk here, and the model keeps a stated amount;
257        // were it to forget, the amount known before withdraws the most.
258        let have = self.silver(trip).await?.unwrap_or(have);
259        // Priced again from the bank, the walk may ask for nothing: then
260        // nothing is asked of the teller.
261        let short = needed.saturating_sub(have);
262        if short > 0 {
263            let walker = self.walker(setting);
264            let unseen = ["hidden", "invisible"]
265                .iter()
266                .any(|flag| walker.flags.get(*flag) == Some(&true));
267            if unseen {
268                self.put(trip, "unhide").await?;
269            }
270            let command = withdraw_command(self.state.room.title.as_deref(), short);
271            self.put(trip, &command).await?;
272            // Only a shortfall the game stated halts the trip.
273            if self.silver(trip).await?.is_some_and(|have| have < needed) {
274                return Err(halted(
275                    "there is not enough silver in this bank for the trip.",
276                ));
277            }
278        }
279        Ok(())
280    }

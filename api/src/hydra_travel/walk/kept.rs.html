1//! What the driver reads for the deeds that keep something to give back: the
2//! language spoken, the key taken out, and the game's id for a named thing.
3//! **Pure**: the model and the game's answer in, a command or a fact out.
4
5use hydra_model::{ChunkLine, GameState};
6use hydra_protocol::frame::LinkKind;
7
8/// What the game calls the tongue `speak <this>` selects, where the two
9/// differ. A guild door's crossing sends `speak wizard` unless the answer
10/// to `speak` was `Guildspeak`.
11const TONGUES: [(&str, &str); 1] = [("wizard", "Guildspeak")];
12
13/// Whether `speaking`, as the game names it, is what `speak <language>` sets.
14pub(super) fn is_spoken(language: &str, speaking: &str) -> bool {
15    let named = TONGUES
16        .iter()
17        .find(|(word, _)| word.eq_ignore_ascii_case(language))
18        .map_or(language, |(_, named)| named);
19    named.eq_ignore_ascii_case(speaking)
20}
21
22/// The language in `You are currently speaking Common.`
23pub(super) fn language_in(answer: &[ChunkLine]) -> Option<String> {
24    answer.iter().find_map(|line| {
25        let text = line.text();
26        let (_, rest) = text.split_once("You are currently speaking ")?;
27        let language = rest.trim_end().trim_end_matches('.');
28        (!language.is_empty()).then(|| language.to_owned())
29    })
30}

32/// What `get my heavy key` took, and out of what, as the game's ids: the
33/// first two things linked in `You remove a heavy iron key from in your
34/// cloak.`, as the crossings that borrow a key read it. `None` for
35/// `Get what?`.
36pub(super) fn taken_from(answer: &[ChunkLine]) -> Option<(String, String)> {
37    answer.iter().find_map(|line| {
38        let text = line.text();
39        (text.starts_with("You remove") || text.contains("detach")).then_some(())?;
40        let mut ids = line.objects().filter_map(|link| match &link.kind {
41            LinkKind::Exist { id, .. } => Some(id.clone()),
42            _ => None,
43        });
44        Some((ids.next()?, ids.next()?))
45    })
46}
47
48/// Whether the game's answer says a thing was put away: *"You put a heavy
49/// iron key in your cloak."*
50pub(super) fn put_back(answer: &[ChunkLine]) -> bool {
51    answer
52        .iter()
53        .any(|line| line.text().starts_with("You put "))
54}

56/// `#id` for the thing with exactly this name: on the walker first -- the
57/// inventory snapshot, then the worn list -- then in
58/// the room -- its objects, and what its description links (a building with
59/// a long name). Names are as the game links them: no article.
60pub(super) fn item_id(state: &GameState, name: &str) -> Option<String> {
61    let carried = state.inventory_snapshot.on_person().find_map(|item| {
62        let named = format!("{} {}", item.adjective, item.noun);
63        (named.trim() == name || item.name == name).then(|| item.id.clone())
64    });
65    // Worn, by the list the login sends: its whole name, or its end, as
66    // the map names a thing by its last adjective and noun.
67    let worn = || {
68        state.worn.items()?.iter().find_map(|item| {
69            (item.text == name || item.text.ends_with(&format!(" {name}"))).then(|| item.id.clone())
70        })
71    };
72    let room = &state.room;
73    let shown = || {
74        room.objects
75            .iter()
76            .find(|thing| thing.text == name)
77            .map(|thing| thing.id.clone())
78    };
79    let described = || {
80        room.description
81            .as_ref()?
82            .objects()
83            .find_map(|link| match &link.kind {
84                LinkKind::Exist { id, .. } if link.text == name => Some(id.clone()),
85                _ => None,
86            })
87    };
88    carried.or_else(worn).or_else(shown).or_else(described)
89}

91/// A command with every `{item:name}` made `#id`. `None` when the walker has
92/// no such thing and the room shows none.
93pub(super) fn with_items(command: &str, state: &GameState) -> Option<String> {
94    let mut out = String::new();
95    let mut rest = command;
96    while let Some((before, after)) = rest.split_once("{item:") {
97        let (name, tail) = after.split_once('}')?;
98        out.push_str(before);
99        out.push('#');
100        out.push_str(&item_id(state, name)?);
101        rest = tail;
102    }
103    out.push_str(rest);
104    Some(out)
105}

107#[cfg(test)]
108mod tests {
109    use super::*;
110
111    /// A worn thing, named as a crossing names it, is found by the worn
112    /// list the login sends: play has no inventory snapshot.
113    #[test]
114    fn a_worn_thing_is_found_by_the_worn_list() {
115        let mut state = GameState::default();
116        let mut parser = hydra_protocol::Parser::new();
117        let wire = concat!(
118            "<clearStream id='inv' ifClosed=''/><pushStream id='inv'/>Your worn items are:
119",
120            "  a <a exist=\"2376165\" noun=\"key\">heavy iron key</a>
121",
122            "<popStream/><prompt time=\"1\">&gt;</prompt>
123",
124        );
125        for frame in parser.push_bytes(wire.as_bytes()) {
126            state.apply(&frame);
127        }
128        assert_eq!(item_id(&state, "iron key").as_deref(), Some("2376165"));
129        assert_eq!(
130            with_items("turn {item:heavy iron key}", &state).as_deref(),
131            Some("turn #2376165")
132        );
133    }
134
135    #[test]
136    fn guildspeak_is_what_speak_wizard_sets() {
137        assert!(is_spoken("wizard", "Guildspeak"));
138        assert!(!is_spoken("wizard", "Common"));
139        assert!(is_spoken("common", "Common"));
140    }
141
142    #[test]
143    fn the_language_is_read_without_its_full_stop() {
144        let answer = [ChunkLine::plain("You are currently speaking Common.")];
145        assert_eq!(language_in(&answer).as_deref(), Some("Common"));
146        assert_eq!(language_in(&[ChunkLine::plain("Get what?")]), None);
147    }
148
149    #[test]
150    fn a_command_with_no_item_in_it_is_sent_as_written() {
151        let state = GameState::default();
152        assert_eq!(with_items("go door", &state).as_deref(), Some("go door"));
153        assert_eq!(with_items("remove {item:brass key}", &state), None);
154    }

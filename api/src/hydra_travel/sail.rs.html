1//! Sailing the character's own ship: making it ready and putting to sea.
2//!
3//! # Making ready is a list read off the ship, not a script
4//!
5//! [`MakeReady`] is asked what to do next, shown the model as it stands,
6//! and answers with one line to send. It keeps no place in a list: each
7//! time it looks at what the game has said of the ship -- the flag flown,
8//! the gangplank, the sail, the anchor, the deck the character is on -- and
9//! does the first thing still undone. So it can be started anywhere aboard
10//! or on the pier beside the ship, with any of it done already, and a step
11//! the game refuses is seen in the state and tried again or passed over.
12//! It only counts what it has sent, so nothing is sent for ever.
13//!
14//! The order is the one a captain keeps: the white flag, which keeps other
15//! ships off and can only be raised from the crow's nest; the gangplank in;
16//! the sail let out, and wind called into it by a character who can; the
17//! anchor up; and `DEPART`, said twice.
18//!
19//! [`Voyage`] then sails it to the port, and [`sail`] is the loop that runs
20//! both against a session: send, wait for the game's prompt and for the
21//! roundtime to pass, ask again.
22//!
23//! # Or the player sails
24//!
25//! With the player at the wheel ([`Helm::Player`]) nothing is sent at all:
26//! the course is worked out from where the ship is each time it moves, and
27//! shown, until the ship lies at the port.
28

42/// The spells that fill a sail with wind, the first known of them cast:
43/// Breeze, and Call Wind.
44pub const WINDS: [u32; 2] = [612, 912];

46/// The wind spell `state`'s character knows, if any.
47#[must_use]
48pub fn wind_known(state: &GameState) -> Option<u32> {
49    WINDS
50        .into_iter()
51        .find(|spell| state.known_spells.knows(*spell) == Some(true))
52}

71/// How long the game's answer to a line is waited for, and the longest a
72/// roundtime is.
73const ANSWER_WAIT: Duration = Duration::from_secs(4);
74const LONGEST_ROUNDTIME: Duration = Duration::from_secs(20);
75const BEAT: Duration = Duration::from_millis(250);

80/// How long a voyage with nothing to do waits for the game's next word,
81/// and the most times it is asked what is next: some hours at sea.

85/// What making ready asks for next.
86#[derive(Clone, Debug, PartialEq, Eq)]
87pub enum Act {
88    /// Send this, and ask again when the game has answered.
89    Send(String),
90    /// Nothing to do yet: ask again when the game has said something.
91    Wait,
92    /// The ship is at sea.
93    Done,
94    /// It cannot go on, and why, in words for the player.
95    Stop(String),
96}

98/// How many times each thing has been sent.
99#[derive(Clone, Copy, Debug, Default)]
100struct Sent {
101    look: u8,
102    watch: u8,
103    board: u8,
104    flag_asked: bool,
105    flag: u8,
106    gangplank: u8,
107    sail: u8,
108    breeze: u8,
109    capstan: u8,
110    depart: u8,

113/// A ship being made ready and put to sea.
114#[derive(Clone, Copy, Debug, Default)]
115pub struct MakeReady {
116    /// The spell the character calls wind into the sail with, if any.
117    breeze: Option<u32>,
118    sent: Sent,
119    flag_settled: bool,
120}
121
122impl MakeReady {
123    /// For a character who knows `breeze`, one of [`WINDS`], or none.
124    #[must_use]
125    pub fn new(breeze: Option<u32>) -> Self {
126        Self {
127            breeze,
128            ..Self::default()
129        }
130    }
131
132    /// The next thing to do, by what `state` has been told.
133    pub fn next(&mut self, state: &GameState) -> Act {
134        let gear = &state.ship.gear;
135        // A room told by its number and not yet by its title is waited for.
136        if state.room.id.is_some() && state.room.title.is_none() {
137            return Act::Wait;
138        }
139        let Some(deck) = aboard(&state.room) else {
140            return self.board(state);
141        };
142        if deck.enemy {
143            return Act::Stop("this is not your ship.".to_owned());
144        }
145        // Where the ship lies is not known until the game has shown it: a
146        // look at the ocean says, in port or out.
147        let unknown = state.ship.waters.is_none()
148            && state.ship.berth.is_none()
149            && state.ship.boarded.is_none();
150        if unknown && self.sent.look == 0 {
151            self.sent.look += 1;
152            return Act::Send("look ocean".to_owned());
153        }
154        // The ocean watched, so every move states the room: a voyage is
155        // steered by it.
156        if !state.ship.watched_ocean && self.sent.watch == 0 {
157            self.sent.watch += 1;
158            return Act::Send("watch ocean".to_owned());
159        }
160        // At sea already: the game has shown open water and no pier since.
161        let at_sea = state.ship.berth.is_none()
162            && state
163                .ship
164                .waters
165                .as_ref()
166                .is_some_and(|waters| waters.at_sea);
167        if let Some(act) = self.flag(&deck, state) {
168            return act;
169        }
170        // At sea, at anchor: the rest is the voyage's, in its order -- the
171        // wheel to the course, the anchor up, and then wind in the sail --
172        // so the ship does not pull at its anchor the wrong way.
173        if at_sea {
174            return Act::Done;
175        }
176        if gear.gangplank_out != Some(false) && self.sent.gangplank < 2 {
177            return go(&deck, "Main Deck").unwrap_or_else(|| {
178                self.sent.gangplank += 1;
179                Act::Send("pull gangplank".to_owned())
180            });
181        }
182        if gear.sails.get("Main Deck") != Some(&Sail::Lowered) && self.sent.sail < 4 {
183            return go(&deck, "Main Deck").unwrap_or_else(|| {
184                self.sent.sail += 1;
185                Act::Send("lower sail".to_owned())
186            });
187        }
188        if self.breeze.is_some() && self.sent.breeze < 2 {
189            return go(&deck, "Main Deck").unwrap_or_else(|| {
190                self.sent.breeze += 1;
191                Act::Send(if self.sent.breeze == 1 {
192                    format!("prep {}", self.breeze.unwrap_or(WINDS[0]))
193                } else {
194                    "cast sail".to_owned()
195                })
196            });
197        }
198        if gear.anchor_up != Some(true) && self.sent.capstan < 10 {
199            return go(&deck, "Helm").unwrap_or_else(|| {
200                self.sent.capstan += 1;
201                Act::Send("push capstan".to_owned())
202            });
203        }
204        // Refused for the gangplank: draw it in, and say it again.
205        if gear.gangplank_out == Some(true) && self.sent.depart > 0 && self.sent.gangplank < 4 {
206            self.sent.gangplank += 1;
207            return go(&deck, "Main Deck")
208                .unwrap_or_else(|| Act::Send("pull gangplank".to_owned()));
209        }
210        if self.sent.depart >= 6 {
211            return Act::Stop("the ship would not depart.".to_owned());
212        }
213        go(&deck, "Helm").unwrap_or_else(|| {
214            self.sent.depart += 1;
215            Act::Send("depart".to_owned())
216        })
217    }
218
219    /// From the pier: up the gangplank when it is out, else toward the ship,
220    /// which puts it out. A gangplank the room shows is out, whatever the
221    /// gear has been told: a ship left at a pier in an earlier session has
222    /// its gangplank down and nothing has said so this one.
223    fn board(&mut self, state: &GameState) -> Act {
224        if self.sent.board >= 4 {
225            return Act::Stop("I could not get aboard.".to_owned());
226        }
227        self.sent.board += 1;
228        if state.ship.gear.gangplank_out != Some(false)
229            && state
230                .room
231                .objects
232                .iter()
233                .any(|thing| thing.noun == "gangplank")
234        {
235            return Act::Send("go gangplank".to_owned());
236        }
237        let ship = state
238            .room
239            .objects
240            .iter()
241            .find(|thing| SHIPS.contains(&thing.noun.as_str()));
242        match ship {
243            Some(ship) => Act::Send(format!("go {}", ship.noun)),
244            None => Act::Stop("there is no ship here to board.".to_owned()),
245        }
246    }
247
248    /// The white flag, raised from the crow's nest: asked after first, tried
249    /// twice, and let be when the game says it cannot be changed yet.
250    fn flag(&mut self, deck: &Aboard, state: &GameState) -> Option<Act> {
251        if self.flag_settled {
252            return None;
253        }
254        let gear = &state.ship.gear;
255        if gear.flag.is_none() && !self.sent.flag_asked {
256            self.sent.flag_asked = true;
257            return Some(Act::Send("ship flag".to_owned()));
258        }
259        let refused = self.sent.flag > 0 && gear.flag_wait.is_some();
260        if gear.flag.as_deref() == Some(WHITE) || refused || self.sent.flag >= 2 {
261            self.flag_settled = true;
262            return None;
263        }
264        Some(go(deck, "Crow's Nest").unwrap_or_else(|| {
265            self.sent.flag += 1;
266            Act::Send("ship flag white".to_owned())
267        }))
268    }
269}

271/// The move from `deck` toward the deck `wanted`, or `None` when the
272/// character is on it. The main deck is the middle of every ship: the helm
273/// west of it, the crow's nest up, the hold down.
274fn go(deck: &Aboard, wanted: &str) -> Option<Act> {
275    if deck.deck == wanted {
276        return None;
277    }
278    let way = match (deck.deck.as_str(), wanted) {
279        ("Main Deck", "Helm") => "west",
280        ("Main Deck", "Crow's Nest") | ("Cargo Hold", _) => "up",
281        ("Main Deck", "Cargo Hold") | ("Crow's Nest", _) => "down",
282        ("Helm", _) => "east",
283        ("Captain's Quarters", _) => "out",
284        (from, _) => {
285            return Some(Act::Stop(format!(
286                "I do not know the way from the {from}: go to the main deck."
287            )));
288        }
289    };
290    Some(Act::Send(way.to_owned()))
291}

293/// How a sailing ended.
294#[derive(Clone, Debug, PartialEq, Eq)]
295pub enum Sailed {
296    /// The ship lies at the port, its gangplank out.
297    Arrived,
298    /// The player stopped it.
299    Stopped,
300    /// The connection dropped, or the session ended.
301    Lost,
302    /// It could not go on, and why.
303    Halted(String),
304}

306/// A session being sailed: its model kept up from the game's own frames.
307struct Runner<'a> {
308    handle: &'a SessionHandle,
309    stop: &'a CancellationToken,
310    state: GameState,
311    events: Receiver<Event>,
312}
313
314/// What a wait heard.
315#[derive(Clone, Debug, PartialEq, Eq)]
316enum Heard {
317    /// Nothing, in the time given, or nothing that matters.
318    Nothing,
319    /// A prompt: a chunk of the game's closed.
320    Prompt,
321    /// A line of travel's own went out: this one.
322    Sent(String),
323}
324
325impl Runner<'_> {
326    /// One of the game's events folded in, or `most` gone by with none.
327    async fn hold(&mut self, most: Duration) -> Result<Heard, Sailed> {
328        let event = tokio::select! {
329            biased;
330            () = self.stop.cancelled() => return Err(Sailed::Stopped),
331            event = self.events.recv() => event,
332            () = tokio::time::sleep(most) => return Ok(Heard::Nothing),
333        };
334        match event {
335            Ok(Event::Frame(frame)) => {
336                self.state.apply(&frame);
337                Ok(if matches!(*frame, Frame::Prompt { .. }) {
338                    Heard::Prompt
339                } else {
340                    Heard::Nothing
341                })
342            }
343            Ok(Event::Sent {
344                line,
345                origin: Origin::Travel,
346            }) => Ok(Heard::Sent(line)),
347            Ok(Event::StateChanged(State::Reconnecting | State::Closed))
348            | Err(RecvError::Closed) => Err(Sailed::Lost),
349            // Events were lost: the state is taken afresh.
350            Err(RecvError::Lagged(_)) => {
351                let (snapshot, fresh) = self.handle.join().await.ok_or(Sailed::Lost)?;
352                self.state = snapshot.state;
353                self.events = fresh;
354                Ok(Heard::Prompt)
355            }
356            Ok(_) => Ok(Heard::Nothing),
357        }
358    }
359
360    /// Until the roundtime the game stated is over.
361    async fn out_of_roundtime(&mut self) -> Result<(), Sailed> {
362        let by = Instant::now() + LONGEST_ROUNDTIME;
363        while self.state.in_roundtime() == Some(true) && Instant::now() < by {
364            self.hold(BEAT).await?;
365        }
366        Ok(())
367    }
368
369    /// Send `line` once any roundtime is over, and wait for its own prompt
370    /// and then its roundtime.
371    ///
372    /// **Its own prompt is the first after the line is seen to go out.** A
373    /// line is often sent on the word of a line of the game's, whose prompt
374    /// is still to come: taken for the answer, the model would be read
375    /// before the game had answered, and the line sent again.
376    async fn send(&mut self, line: &str) -> Result<(), Sailed> {
377        self.out_of_roundtime().await?;
378        if self.stop.is_cancelled() {
379            return Err(Sailed::Stopped);
380        }
381        if !self.handle.write(line, Origin::Travel).await {
382            return Err(Sailed::Lost);
383        }
384        let answer_by = Instant::now() + ANSWER_WAIT;
385        let mut gone_out = false;
386        while Instant::now() < answer_by {
387            match self.hold(BEAT).await? {
388                Heard::Sent(sent) if sent.trim() == line.trim() => gone_out = true,
389                Heard::Prompt if gone_out => break,
390                _ => self.wheel_already_turned(line),
391            }
392        }
393        if self.not_the_captain() {
394            return Err(Sailed::Halted(
395                "this is not your ship: only its captain can sail it.".to_owned(),
396            ));
397        }
398        self.out_of_roundtime().await
399    }
400
401    /// `[Only the Captain can begin your voyage.]`: the ship stood on is
402    /// another's, whatever it looks like.
403    fn not_the_captain(&self) -> bool {
404        self.state.open_chunk().lines().iter().any(|heard| {
405            heard
406                .text()
407                .contains("Only the Captain can begin your voyage")
408        })
409    }
410
411    /// `turn wheel <way>` answered with `The wooden wheel is already turned
412    /// that way!`: the game names no way, so the model cannot read where
413    /// the wheel is from it. The way is the one just asked for, and the
414    /// model is told so, else the wheel would be turned again and again.
415    fn wheel_already_turned(&mut self, line: &str) {
416        let Some(way) = line.trim().strip_prefix("turn wheel ") else {
417            return;
418        };
419        let already = self
420            .state
421            .open_chunk()
422            .lines()
423            .iter()
424            .any(|heard| heard.text().ends_with("wheel is already turned that way!"));
425        if already {
426            self.state.ship.gear.wheel = Some(match way {
427                "port" => "toward a port".to_owned(),
428                way => format!("to the {way}"),
429            });
430        }
431    }
432}

434/// Who has the wheel.
435#[derive(Clone, Copy, Debug, PartialEq, Eq)]
436pub enum Helm {
437    /// Hydra: it makes the ship ready and sails it, calling wind into the
438    /// sail on the way when `wind` is asked and the character knows how.
439    Hydra {
440        /// Whether to call wind into the sail (Breeze).
441        wind: bool,
442    },
443    /// The player: Hydra sends nothing, and shows the course.
444    Player,
445}
446
447/// Sail the ship of `handle`'s character to the port whose room of `ocean`
448/// is `goal`, until it lies there, it cannot go on, or `stop` is cancelled:
449/// made ready and sailed by Hydra, or sailed by the player with the course
450/// kept before them, as `helm` says. `shown` is told the rooms still to
451/// sail, the ship's first, each time they change, and none at the end;
452/// `told` a word for the player when Hydra's voyage is under way.
453pub async fn sail(
454    handle: &SessionHandle,
455    stop: &CancellationToken,
456    ocean: &Ocean,
457    goal: i64,
458    helm: Helm,
459    mut shown: impl FnMut(Option<&[i64]>),
460    mut told: impl FnMut(String),
461) -> Sailed {
462    let Some((snapshot, events)) = handle.join().await else {
463        return Sailed::Lost;
464    };
465    if matches!(snapshot.lifecycle, State::Reconnecting | State::Closed) {
466        return Sailed::Lost;
467    }
468    let mut runner = Runner {
469        handle,
470        stop,
471        state: snapshot.state,
472        events,
473    };
474    let sailed = match helm {
475        Helm::Hydra { wind } => voyage(&mut runner, ocean, goal, wind, &mut shown, &mut told).await,
476        Helm::Player => guide(&mut runner, ocean, goal, &mut shown).await,
477    };
478    shown(None);
479    match sailed {
480        Ok(()) => Sailed::Arrived,
481        Err(sailed) => sailed,
482    }
483}

485/// What a player at the wheel is shown.
486#[derive(Clone, Debug, PartialEq, Eq)]
487enum Guided {
488    /// The rooms still to sail, the ship's first.
489    Course(Vec<i64>),
490    /// Nothing says where the ship is: what was shown stays.
491    Unknown,
492    /// The ship lies at the port.
493    Arrived,
494    /// There is nothing to guide, and why.
495    Halted(String),
496}
497
498/// The course to `goal` for a ship the player sails, from where it is: at
499/// sea, from its room and the heading its wheel holds; in another port, or
500/// on that port's pier before boarding, from the port's room.
501fn guided(ocean: &Ocean, state: &GameState, at: Sailing, goal: i64, sailed: bool) -> Guided {
502    let held = state
503        .ship
504        .gear
505        .wheel
506        .as_deref()
507        .and_then(|wheel| wheel.strip_prefix("to the "))
508        .and_then(hydra_map::Heading::named);
509    let (from, held) = match at {
510        Sailing::At { room, .. } if room == goal => return Guided::Arrived,
511        Sailing::At { room, .. } => (room, held),
512        Sailing::Docked { port: Some(port) } if port == goal => return Guided::Arrived,
513        Sailing::Docked { port: Some(port) } => (port, None),
514        Sailing::Docked { port: None } | Sailing::Lost => return Guided::Unknown,
515        Sailing::Ashore if sailed => {
516            return Guided::Halted("you have left the ship.".to_owned());
517        }
518        Sailing::Ashore => {
519            let pier = state.room.id.as_deref().and_then(|id| id.parse().ok());
520            match pier.and_then(|pier| ocean.port_of_pier(pier)) {
521                Some(port) if port.uid == goal => return Guided::Arrived,
522                Some(port) => (port.uid, None),
523                None => {
524                    return Guided::Halted(
525                        "you are not aboard a ship, nor on a pier beside one.".to_owned(),
526                    );
527                }
528            }
529        }
530    };
531    match ocean.course_holding(from, held, goal) {
532        Some(course) => Guided::Course(std::iter::once(from).chain(course.rooms).collect()),
533        None => Guided::Halted("I find no course from here.".to_owned()),
534    }
535}

537/// The course kept before a player who sails: nothing is sent.
538async fn guide(
539    runner: &mut Runner<'_>,
540    ocean: &Ocean,
541    goal: i64,
542    shown: &mut impl FnMut(Option<&[i64]>),
543) -> Result<(), Sailed> {
544    let mut navigator = Navigator::default();
545    let mut last_shown: Vec<i64> = Vec::new();
546    let mut sailed = false;
547    for _ in 0..MOST_TURNS {
548        let at = navigator.fix(ocean, &runner.state);
549        sailed |= matches!(at, Sailing::At { .. });
550        match guided(ocean, &runner.state, at, goal, sailed) {
551            Guided::Course(rooms) => {
552                if rooms != last_shown {
553                    shown(Some(&rooms));
554                    last_shown = rooms;
555                }
556            }
557            Guided::Unknown => {}
558            Guided::Arrived => return Ok(()),
559            Guided::Halted(why) => return Err(Sailed::Halted(why)),
560        }
561        runner.hold(IDLE).await?;
562    }
563    Err(Sailed::Halted(
564        "the voyage has gone on too long, so the course is off the chart.".to_owned(),
565    ))
566}

568async fn voyage(
569    runner: &mut Runner<'_>,
570    ocean: &Ocean,
571    goal: i64,
572    wind: bool,
573    shown: &mut impl FnMut(Option<&[i64]>),
574    told: &mut impl FnMut(String),
575) -> Result<(), Sailed> {
576    let breeze = wind_known(&runner.state).filter(|_| wind);
577    let mut ready = MakeReady::new(breeze);
578    // The course from the port, shown while the ship is made ready.
579    let mut in_port = Navigator::default();
580    let mut ready_shown: Vec<i64> = Vec::new();
581    for _ in 0..MOST_LINES {
582        let at = in_port.fix(ocean, &runner.state);
583        if let Guided::Course(rooms) = guided(ocean, &runner.state, at, goal, false)
584            && rooms != ready_shown
585        {
586            shown(Some(&rooms));
587            ready_shown = rooms;
588        }
589        match ready.next(&runner.state) {
590            Act::Send(line) => runner.send(&line).await?,
591            Act::Wait => {
592                runner.hold(BEAT).await?;
593            }
594            Act::Done => break,
595            Act::Stop(why) => return Err(Sailed::Halted(why)),
596        }
597    }
598    let began = Instant::now();
599    let mut navigator = Navigator::default();
600    let mut voyage = Voyage::new(goal, breeze, 0);
601    // Wind not called while making ready is called once under way.
602    if ready.sent.breeze == 0 {
603        voyage.wind_now();
604    }
605    let mut last_shown: Vec<i64> = Vec::new();
606    for _ in 0..MOST_TURNS {
607        let at = navigator.fix(ocean, &runner.state);
608        let now = began.elapsed().as_secs();
609        let act = voyage.next(ocean, &runner.state, at, now);
610        for note in voyage.notes() {
611            told(note);
612        }
613        if voyage.course() != last_shown.as_slice() {
614            // Said once, at sea: a ship is slow, and a silence reads as a fault.
615            if last_shown.is_empty() && voyage.course().len() > 1 {
616                told(format!(
617                    "at sea, {} rooms to go. The first takes a minute, the rest some twenty seconds each; I am at the wheel.",
618                    voyage.course().len() - 1
619                ));
620            }
621            last_shown = voyage.course().to_vec();
622            shown(Some(&last_shown));
623        }
624        match act {
625            Act::Send(line) => runner.send(&line).await?,
626            // Until the game says something, or a moment passes.
627            Act::Wait => {
628                runner.hold(IDLE).await?;
629            }
630            Act::Done => return Ok(()),
631            Act::Stop(why) => return Err(Sailed::Halted(why)),
632        }
633    }
634    Err(Sailed::Halted(
635        "the voyage has gone on too long, so I have let go of the wheel.".to_owned(),
636    ))
637}

645    /// A state told these wire lines, then a prompt.
646    fn told(state: &mut GameState, lines: &[&str]) {
647        let mut parser = Parser::new();
648        for line in lines
649            .iter()
650            .copied()
651            .chain(["<prompt time=\"1\">&gt;</prompt>"])
652        {
653            for frame in parser.parse_line(line) {
654                state.apply(&frame);
655            }
656        }
657    }

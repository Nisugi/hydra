1//! A recording's rewards: the wire's bytes through the parser, the lines
2//! through a watcher, each reward stamped by the prompt that closed its chunk.
3
4use hydra_ledger::{Reward, Stamp, Who};
5use hydra_protocol::{Frame, Parser};
6
7use crate::watcher::{Heard, Watcher};
8
9/// The words a recording must hold for it to hold a reward at all; a
10/// recording without them is not parsed.
11const MARK: &[u8] = b"earned the following reward";
12
13/// Every login reward `bytes` holds, as `who`'s, in the order told. A reward
14/// told before any prompt has no time and is left out.
15#[must_use]
16pub fn rewards(bytes: &[u8], who: &Who) -> Vec<Reward> {
17    if !bytes.windows(MARK.len()).any(|w| w == MARK) {
18        return Vec::new();
19    }
20    let mut parser = Parser::new();
21    let mut watcher = Watcher::new();
22    let mut waiting: Vec<Heard> = Vec::new();
23    let mut found = Vec::new();
24    // The same reads a live session made, so lines split across them take
25    // the same path they did.
26    for chunk in bytes.chunks(4096).map(Some).chain(std::iter::once(None)) {
27        let frames = match chunk {
28            Some(chunk) => parser.push_bytes(chunk),
29            None => parser.finish(),
30        };
31        for frame in frames {
32            match &frame {
33                Frame::Text(text) => waiting.extend(watcher.hear(text)),
34                Frame::Prompt { time, .. } => {
35                    if let Ok(at) = time.parse::<i64>() {
36                        let stamp = Stamp {
37                            at,
38                            game_time: None,
39                        };
40                        found.extend(waiting.drain(..).map(|heard| heard.at(who, stamp)));
41                    }
42                }
43                _ => {}
44            }
45        }
46    }
47    found
48}

50#[cfg(test)]
51mod tests {
52    use super::*;
53
54    #[test]
55    fn a_recording_s_reward_is_stamped_by_its_prompt() {
56        let bytes = b"<prompt time=\"1760000000\">&gt;</prompt>\n\
57Thank you for logging into GemStone IV!\n\
58You have earned the following reward: 500 fame!\n\
59<prompt time=\"1760000001\">&gt;</prompt>\n";
60        let who = Who::new("prime", "Peteriath");
61        let found = rewards(bytes, &who);
62        assert_eq!(found.len(), 1);
63        assert_eq!(found[0].stamp.at, 1_760_000_001);
64        assert_eq!(found[0].reward, "500 fame");
65        assert_eq!(found[0].streak, None);
66        // 1760000001 is 2025-10-09 08:53:21Z, the 9th Eastern.
67        assert_eq!(found[0].day, "2025-10-09");
68    }
69
70    #[test]
71    fn a_recording_without_one_yields_none() {
72        let who = Who::new("prime", "Peteriath");
73        assert!(rewards(b"<prompt time=\"1\">&gt;</prompt>\nhello\n", &who).is_empty());
74    }
75}

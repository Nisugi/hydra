1//! The two lines the game says on a day's first login, read.
2//!
3//! ```text
4//! Thank you for logging into GemStone IV for 2 consecutive days! [1 login bridge expended.]
5//! You have earned the following reward: 1 Random Magical Crystal!  You may use the BOOST command to access this reward!
6//! ```
7//!
8//! The first day of a streak has no count (`Thank you for logging into
9//! GemStone IV!`), and the bracketed bridge note comes only when a missed
10//! day was covered.
11
12/// What the thank-you line told.
13#[derive(Debug, Clone, Copy, PartialEq, Eq)]
14pub struct Thanks {
15    /// The consecutive days the game counted; none when it counted none.
16    pub streak: Option<i64>,
17    /// The login bridges it said it used.
18    pub bridges: Option<i64>,
19}
20
21const THANKS: &str = "Thank you for logging into ";
22const REWARD: &str = "You have earned the following reward: ";
23
24/// The thank-you line, read; `None` for any other line.
25#[must_use]
26pub fn thanks(line: &str) -> Option<Thanks> {
27    let rest = line.trim().strip_prefix(THANKS)?;
28    Some(Thanks {
29        streak: number_before(rest, " consecutive day"),
30        bridges: number_before(rest, " login bridge"),
31    })
32}
33
34/// The reward the reward line names, without the game's hint about how to
35/// use it; `None` for any other line.
36#[must_use]
37pub fn reward(line: &str) -> Option<String> {
38    let rest = line.trim().strip_prefix(REWARD)?;
39    let named = rest.split_once('!').map_or(rest, |(named, _)| named);
40    // A bounty boost's line says it twice: `You have earned the following
41    // reward: You have been awarded 2 Bounty Boosts!`
42    let named = named.trim();
43    let named = named
44        .strip_prefix("You have been awarded ")
45        .unwrap_or(named);
46    (!named.is_empty()).then(|| named.to_owned())
47}
48
49/// The whole number just before `marker` in `text`.
50fn number_before(text: &str, marker: &str) -> Option<i64> {
51    let before = text.get(..text.find(marker)?)?;
52    let digits = before.rsplit(|c: char| !c.is_ascii_digit()).next()?;
53    digits.parse().ok()
54}

56#[cfg(test)]
57mod tests {
58    use super::*;
59
60    #[test]
61    fn a_streak_is_counted() {
62        assert_eq!(
63            thanks("Thank you for logging into GemStone IV for 2 consecutive days!"),
64            Some(Thanks {
65                streak: Some(2),
66                bridges: None
67            })
68        );
69        assert_eq!(
70            thanks("Thank you for logging into GemStone IV for 20 consecutive days!")
71                .and_then(|t| t.streak),
72            Some(20)
73        );
74    }
75
76    #[test]
77    fn the_first_day_has_no_count() {
78        assert_eq!(
79            thanks("Thank you for logging into GemStone IV!"),
80            Some(Thanks {
81                streak: None,
82                bridges: None
83            })
84        );
85    }
86
87    #[test]
88    fn a_bridge_is_noted() {
89        assert_eq!(
90            thanks(
91                "Thank you for logging into GemStone IV for 2 consecutive days! [1 login bridge expended.]"
92            ),
93            Some(Thanks {
94                streak: Some(2),
95                bridges: Some(1)
96            })
97        );
98    }
99
100    #[test]
101    fn the_reward_is_named_without_its_hint() {
102        assert_eq!(
103            reward(
104                "You have earned the following reward: 1 Random Magical Crystal!  You may use the BOOST command to access this reward!"
105            )
106            .as_deref(),
107            Some("1 Random Magical Crystal")
108        );
109        assert_eq!(
110            reward("You have earned the following reward: 500 fame!").as_deref(),
111            Some("500 fame")
112        );
113        assert_eq!(
114            reward(
115                "You have earned the following reward: 2 activations of 15 minute Urchin Guide Access!  You may use the BOOST command to access this reward!"
116            )
117            .as_deref(),
118            Some("2 activations of 15 minute Urchin Guide Access")
119        );
120    }
121
122    #[test]
123    fn other_lines_are_neither() {
124        for line in [
125            "You have outstanding SimuCoin purchases to receive.",
126            "Obvious exits: none",
127            "",
128        ] {
129            assert_eq!(thanks(line), None, "{line}");
130            assert_eq!(reward(line), None, "{line}");
131        }
132    }

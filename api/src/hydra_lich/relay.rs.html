1//! The Lich relay: the player's own Lich, for one character, with Hydra
2//! keeping the game's connection.
3//!
4//! [`run`] starts Lich in pipe mode, pointed at a port it holds:
5//!
6//! ```text
7//! ruby lich.rbw --pipe --stormfront -g 127.0.0.1:<port>
8//! ```
9//!
10//! Lich reads a key and a version line from its standard input, as a
11//! frontend sends them, and writes both to that port before anything else.
12//! The key is a new token, and only a connection that says it first is
13//! taken: until Lich connects, anything on
14//! this machine could reach the port, and what comes through it goes to the
15//! game. It is not the game's key, which Hydra has already given. **Lich
16//! never holds a credential**, nor does its environment ([`crate::child`]).
17//!
18//! From then on Lich takes the port as its game:
19//!
20//! - the game's bytes, as they came, go to it, after a login built from
21//!   what the session knows as it connects ([`LichDoor::claim`]);
22//! - what it writes back goes to the game ([`LichDoor::send`]). A line with
23//!   Lich's `<c>` is one of its scripts'; one without is what
24//!   the player typed, passed on after Lich's own hooks;
25//! - what the player types that Hydra does not take goes to its standard
26//!   input, as a frontend's typing does, and Hydra's own lines that start
27//!   with Lich's `;` (`hydra_session::lich`);
28//! - what it would show a frontend comes out of its standard output, always
29//!   read, so Lich never waits to write it, and is shown
30//!   ([`LichDoor::show`]). Lich is asked to mark the lines it forwards from
31//!   the game (`--sentinel`), so a window can tell them from its own; a Lich
32//!   release without that flag ignores it, as it ignores any flag it does
33//!   not know.
34//!
35//! Closing its standard input stops it.
36//!
37//! **A Lich that stops reading is let go, never waited on** (as the
38//! session's copy is): one whose script's hook never returns reads neither
39//! its game nor its standard input. Each write to it is a task of its own
40//! (`feed`), so a stop, its exit, and the [`WIRE_CHUNKS`] chunks it has
41//! left unread are always seen, and end it.
42//!
43//! # Hydra is `stormfront`
44//!
45//! Pipe mode tells Lich its frontend is `unknown`, so a script's text would
46//! come out unescaped and the many scripts that ask for `stormfront` would
47//! take another branch. Hydra speaks Wrayth's protocol, which Lich calls
48//! `stormfront`: Lich is started with `--stormfront`, and, for a Lich release
49//! that drops that flag in pipe mode, `STORMFRONT` puts `unknown` back to
50//! `stormfront` whenever Lich sets it.
51

72/// How long Lich has to connect once started. It takes about 3 to 5 s
73/// without its windows; with them, and on a slower machine, it takes
74/// longer. Lich allows itself 30 s to reach a game.
75pub const CONNECT_DEADLINE: Duration = Duration::from_mins(1);
76
77/// How long a connection to the port has to say the key before it is let
78/// go. Lich says it as it connects.
79const KEY_DEADLINE: Duration = Duration::from_secs(5);
80
81/// How long Lich has to exit once its standard input closes before it is
82/// killed. It usually takes under a second.
83const STOP_DEADLINE: Duration = Duration::from_secs(5);
84
85/// How many of Lich's last lines of standard error are kept to say why it
86/// ended.
87const ERRORS_KEPT: usize = 20;
88
89/// How long Lich's last words on standard error are waited for once it has
90/// ended. They are read by a task of their own, which Lich's exit can
91/// outrun; a process Lich started can hold the pipe open, hence a limit.
92const ERRORS_DEADLINE: Duration = Duration::from_secs(1);
93
94/// The version line a frontend sends after the key. Lich passes it on, and
95/// the relay reads it and lets it go: the game has had Hydra's.
96const VERSION: &str = "/FE:WRAYTH /VERSION:1.0.1.28 /P:WIN_UNKNOWN /XML";
97
98/// Puts Lich's `$frontend` back to `stormfront` whenever pipe mode sets it
99/// to `unknown`. A Lich that keeps `--stormfront` in pipe mode never sets
100/// it, and this does nothing.
101const STORMFRONT: &str =
102    r#"trace_var(:$frontend) { |name| $frontend = "stormfront" if name == "unknown" }"#;
103
104/// Runs `lich.rbw` as Ruby would run it, after [`STORMFRONT`]: `$0` names
105/// it, as Lich finds its folders by it.
106const LOAD: &str = "$0 = ARGV.shift; load $0";

111    /// Ruby: [`crate::find_ruby`].
112    pub ruby: PathBuf,
113    /// The player's `lich.rbw`. Lich runs in its folder.

115    /// More of Lich's own flags, after Hydra's: a test's `--home`.
116    pub args: Vec<OsString>,

122/// How a relay ended.
123#[derive(Clone, Debug, PartialEq, Eq)]
124pub enum Ended {
125    /// Stopped as asked: its standard input closed, and it exited.
126    Stopped,
127    /// Lich exited on its own.
128    Exited {
129        /// How, in words.
130        status: String,
131        /// Its last lines of standard error.
132        errors: Vec<String>,
133    },
134    /// Lich did not connect within [`CONNECT_DEADLINE`], and was stopped.
135    NeverConnected {
136        /// Its last lines of standard error.
137        errors: Vec<String>,
138    },
139    /// Lich had [`WIRE_CHUNKS`] of the game's chunks unread, and was stopped.
140    FellBehind,
141    /// The character's session ended, and Lich was stopped.
142    SessionGone,
143    /// Another Lich already runs for this character.
144    Busy,
145    /// Lich could not be started: why.
146    Failed(String),
147}
148
149impl Ended {
150    /// What the player is told, if anything: a session that is gone has
151    /// nobody to tell.
152    #[must_use]
153    pub fn notice(&self) -> Option<Notice> {
154        let (kind, first, errors) = match self {
155            Self::Stopped => (NoticeKind::Info, "Lich stopped.".to_owned(), &[][..]),
156            Self::Exited { status, errors } => (
157                NoticeKind::Warn,
158                format!("Lich exited ({status})."),
159                &errors[..],
160            ),
161            Self::NeverConnected { errors } => (
162                NoticeKind::Error,
163                format!(
164                    "Lich did not connect within {} seconds, and was stopped.",
165                    CONNECT_DEADLINE.as_secs()
166                ),
167                &errors[..],
168            ),
169            Self::FellBehind => (
170                NoticeKind::Error,
171                format!(
172                    "Lich fell {WIRE_CHUNKS} chunks behind the game, and was stopped: \
173                     a stream with a hole in it would mislead its scripts."
174                ),
175                &[][..],
176            ),
177            Self::SessionGone => return None,
178            Self::Busy => (
179                NoticeKind::Warn,
180                "Lich already runs for this character.".to_owned(),
181                &[][..],
182            ),
183            Self::Failed(why) => (
184                NoticeKind::Error,
185                format!("Lich could not start: {why}"),
186                &[][..],
187            ),
188        };
189        let mut lines = vec![first];
190        lines.extend(errors.iter().cloned());
191        Some(Notice::table(kind, lines))
192    }
193}

195/// Run the player's Lich for the character `door` reaches, until `stop`, or
196/// until Lich or the session ends; then tell the player how it ended, and
197/// say so.
198///
199/// Lich is claimed from the moment this is called, not when the future is
200/// first polled: what the player types for it from now on waits for it.
201/// It is handed the game once it has connected, and only then do the
202/// windows turn to its text: while Ruby and Lich start, which can take half
203/// a minute, they show the game's own.
204pub fn run(
205    door: LichDoor,
206    launch: Launch,
207    stop: CancellationToken,
208) -> impl Future<Output = Ended> + Send {
209    let attached = door.claim();
210    async move {
211        let ended = match attached {
212            Some(Attached { wire, mut typing }) => {
213                if let Some(notice) = shared_symbol(door.symbol(), door.lich_symbol()) {
214                    door.say(notice);
215                }
216                let mut unsent = Vec::new();
217                let ended = relay(&door, &launch, wire, &mut typing, &stop, &mut unsent).await;
218                // What was typed for it and never given -- left unread by a
219                // Lich that hung, or never handed on -- said, not lost.
220                unsent.extend(typing.waiting());
221                if !unsent.is_empty() && ended != Ended::SessionGone {
222                    let mut lines = vec!["Typed for Lich and never given to it:".to_owned()];
223                    lines.extend(unsent.into_iter().map(|line| format!("  {line}")));
224                    door.say(Notice::table(NoticeKind::Warn, lines));
225                }
226                ended
227            }
228            None if door.session_over() => {
229                Ended::Failed("the character's session has ended".to_owned())
230            }
231            None => Ended::Busy,
232        };
233        if let Some(notice) = ended.notice() {
234            door.say(notice);
235        }
236        ended
237    }
238}

240/// What the player is told when Hydra's command symbol is Lich's: what they
241/// type with it is Hydra's, so Lich's commands cannot be reached. The
242/// settings menu refuses it; a file written by hand may still say it.
243fn shared_symbol(symbol: char, lich: char) -> Option<Notice> {
244    (symbol == lich).then(|| {
245        Notice::line(
246            NoticeKind::Warn,
247            format!(
248                "Hydra's commands start with {symbol}, as Lich's do, so a line typed with \
249                 {symbol} is Hydra's and Lich's commands can't be reached. Give Hydra \
250                 another command symbol, such as ., on the settings menu's General page."
251            ),
252        )
253    })
254}

256async fn relay(
257    door: &LichDoor,
258    launch: &Launch,
259    mut wire: Wire,
260    typing: &mut Typing,
261    stop: &CancellationToken,
262    unsent: &mut Vec<String>,
263) -> Ended {
264    let key = match crate::new_token() {
265        Ok(key) => key,
266        Err(why) => return Ended::Failed(why),
267    };
268    let listener = match TcpListener::bind(("127.0.0.1", 0)).await {
269        Ok(listener) => listener,
270        Err(why) => return Ended::Failed(why.to_string()),
271    };
272    let port = match listener.local_addr() {
273        Ok(address) => address.port(),
274        Err(why) => return Ended::Failed(why.to_string()),
275    };
276    let mut child = match spawn(launch, port) {
277        Ok(child) => child,
278        Err(why) => return Ended::Failed(format!("{}: {why}", launch.ruby.display())),
279    };
280    let (Some(mut stdin), Some(stdout), Some(stderr)) =
281        (child.stdin.take(), child.stdout.take(), child.stderr.take())
282    else {
283        return Ended::Failed("its standard streams were not piped".to_owned());
284    };
285    show(stdout, door.clone());
286    let mut errors = keep_errors(stderr);
287    if let Err(why) = stdin
288        .write_all(format!("{key}\n{VERSION}\n").as_bytes())
289        .await
290    {
291        finish(child, stdin).await;
292        return Ended::Failed(why.to_string());
293    }
294
295    let taken = tokio::select! {
296        taken = accept_keyed(&listener, &key) => taken,
297        () = tokio::time::sleep(CONNECT_DEADLINE) => {
298            finish(child, stdin).await;
299            return Ended::NeverConnected { errors: errors.tail().await };
300        }
301        status = child.wait() => return exited(status, &mut errors).await,
302        () = stop.cancelled() => {
303            finish(child, stdin).await;
304            return Ended::Stopped;
305        }
306    };
307    let (reader, writer) = match taken {
308        Ok(taken) => taken,
309        Err(why) => {
310            finish(child, stdin).await;
311            return Ended::Failed(why.to_string());
312        }
313    };
314    drop(listener);
315    // Connected: handed its login and the game from the session's next turn.
316    wire.begin();
317    // Every write to Lich is its own task's, so nothing Lich leaves unread
318    // holds up this loop: its stop, its exit and its falling behind are
319    // always seen. Each ends with `quit`, however this returns.
320    let quit = CancellationToken::new();
321    let _quit = quit.clone().drop_guard();
322    let (game, _) = feed(writer, WIRE_CHUNKS, quit.clone());
323    let (typed_in, unwritten) = feed(stdin, TYPED_LINES, quit.clone());
324    let mut lines = read_lines(reader);
325    let (mut lines_open, mut typing_open) = (true, true);
326    let ended = loop {
327        tokio::select! {
328            chunk = wire.next() => {
329                let Some(chunk) = chunk else {
330                    let ended = if wire.fell_behind() { Ended::FellBehind } else { Ended::SessionGone };
331                    break let_go(child, &quit, ended).await;
332                };
333                // Closed: a Lich that hung up is going, and its exit says why.
334                if let Err(TrySendError::Full(_)) = game.try_send(chunk) {
335                    break let_go(child, &quit, Ended::FellBehind).await;
336                }
337            }
338            line = lines.recv(), if lines_open => match line {
339                Some(line) => pass_on(door, &line).await,
340                None => lines_open = false,
341            },
342            typed = typing.next(), if typing_open => match typed {
343                Some(line) => {
344                    if let Err(TrySendError::Full(_)) = typed_in.try_send(format!("{line}\n").into_bytes()) {
345                        door.say(Notice::line(
346                            NoticeKind::Warn,
347                            format!("Lich is not reading what is typed for it ({TYPED_LINES} lines wait): not handed: {line}"),
348                        ));
349                    }
350                }
351                None => typing_open = false,
352            },
353            status = child.wait() => break exited(status, &mut errors).await,
354            () = stop.cancelled() => break let_go(child, &quit, Ended::Stopped).await,
355        }
356    };
357    quit.cancel();
358    if let Ok(left) = unwritten.await {
359        unsent.extend(left.iter().map(|line| {
360            String::from_utf8_lossy(line)
361                .trim_end_matches(['\r', '\n'])
362                .to_owned()
363        }));
364    }
365    ended
366}

368/// Writes each chunk sent to `out`, in order, from a task of its own, until
369/// the sender goes or `quit`: a write Lich never reads ends with `quit`
370/// rather than holding anything up. `waiting` chunks wait for it; past them
371/// the sender's `try_send` is refused, and Lich has stopped reading.
372///
373/// The task ends with the chunks it never wrote whole: the one a write was
374/// stopped in, if any, and those still waiting.
375fn feed<W>(
376    mut out: W,
377    waiting: usize,
378    quit: CancellationToken,
379) -> (mpsc::Sender<Vec<u8>>, JoinHandle<Vec<Vec<u8>>>)
380where
381    W: AsyncWrite + Unpin + Send + 'static,
382{
383    let (chunks, mut unwritten) = mpsc::channel::<Vec<u8>>(waiting);
384    let task = tokio::spawn(hydra_mem::task(hydra_mem::Part::Lich, async move {
385        let mut left = Vec::new();
386        loop {
387            let chunk = tokio::select! {
388                chunk = unwritten.recv() => chunk,
389                () = quit.cancelled() => None,
390            };
391            let Some(chunk) = chunk else { break };
392            let written = tokio::select! {
393                written = out.write_all(&chunk) => written.is_ok(),
394                () = quit.cancelled() => false,
395            };
396            if !written {
397                left.push(chunk);
398                break;
399            }
400        }
401        unwritten.close();
402        while let Ok(chunk) = unwritten.try_recv() {
403            left.push(chunk);
404        }
405        left
406    }));
407    (chunks, task)
408}

410/// Stop a Lich the relay had connected: its feeds end, which closes its
411/// standard input, and it is killed if it has not exited within
412/// [`STOP_DEADLINE`]. Says `ended`.
413async fn let_go(child: Child, quit: &CancellationToken, ended: Ended) -> Ended {
414    quit.cancel();
415    reap(child).await;
416    ended
417}

419fn spawn(launch: &Launch, port: u16) -> std::io::Result<Child> {
420    // Whole: Lich runs in its own folder, where a path relative to here
421    // would be looked for again from there.
422    let lich = std::path::absolute(&launch.lich).unwrap_or_else(|_| launch.lich.clone());
423    let mut command = crate::child::command(&launch.ruby);
424    command
425        .arg("-e")
426        .arg(STORMFRONT)
427        .arg("-e")
428        .arg(LOAD)
429        .arg(&lich)
430        .args(["--pipe", "--stormfront", "--sentinel", "-g"])
431        .arg(format!("127.0.0.1:{port}"))
432        .args(&launch.args)
433        .envs(launch.env.iter().map(|(name, value)| (name, value)))
434        .stdin(Stdio::piped())
435        .stdout(Stdio::piped())
436        .stderr(Stdio::piped())
437        .kill_on_drop(true);
438    if let Some(folder) = lich.parent().filter(|f| !f.as_os_str().is_empty()) {
439        command.current_dir(folder);
440    }
441    command.spawn()
442}

444/// Takes the first connection to `listener` that says `key` on its first
445/// line, and reads its second, the version line, and lets it go. Any other
446/// is dropped.
447async fn accept_keyed(
448    listener: &TcpListener,
449    key: &str,
450) -> std::io::Result<(BufReader<OwnedReadHalf>, OwnedWriteHalf)> {
451    loop {
452        let (stream, _) = listener.accept().await?;
453        // Each chunk of the game goes to Lich as it came, not held back for
454        // the next (Nagle's algorithm, which waits on Lich's acknowledgement).
455        stream.set_nodelay(true)?;
456        let (read, write) = stream.into_split();
457        let mut reader = BufReader::new(read);
458        if let Ok(Ok(true)) = tokio::time::timeout(KEY_DEADLINE, said_key(&mut reader, key)).await {
459            return Ok((reader, write));
460        }
461    }
462}

464async fn said_key(reader: &mut BufReader<OwnedReadHalf>, key: &str) -> std::io::Result<bool> {
465    let mut line = String::new();
466    reader.read_line(&mut line).await?;
467    if line.trim_end() != key {
468        return Ok(false);
469    }
470    line.clear();
471    reader.read_line(&mut line).await?;
472    Ok(true)
473}

475/// Each line Lich writes to its game, without its line ending.
476fn read_lines(mut reader: BufReader<OwnedReadHalf>) -> mpsc::Receiver<String> {
477    let (lines, receiver) = mpsc::channel(64);
478    tokio::spawn(hydra_mem::task(hydra_mem::Part::Lich, async move {
479        let mut buf = Vec::new();
480        loop {
481            buf.clear();
482            match reader.read_until(b'\n', &mut buf).await {
483                Ok(0) | Err(_) => return,
484                Ok(_) => {
485                    let line = String::from_utf8_lossy(&buf);
486                    if lines
487                        .send(line.trim_end_matches(['\r', '\n']).to_owned())
488                        .await
489                        .is_err()
490                    {
491                        return;
492                    }
493                }
494            }
495        }
496    }));
497    receiver
498}

500/// Send a line Lich wrote to its game, and tell the player when it did not
501/// go.
502async fn pass_on(door: &LichDoor, line: &str) {
503    let (line, from) = line
504        .strip_prefix("<c>")
505        .map_or((line, LineFrom::Player), |line| (line, LineFrom::Lich));
506    let why = match door.send(line, from).await {
507        Sending::Sent | Sending::Ran => return,
508        Sending::Unknown => "Hydra has no such command",
509        Sending::Lost => "not connected to the game",
510    };
511    door.say(Notice::line(
512        NoticeKind::Warn,
513        format!("Lich's line was not sent ({why}): {line}"),
514    ));
515}

517/// Shows what Lich writes to its standard output, as it writes it, until
518/// it ends: read always, so Lich never waits to write it.
519fn show(mut stdout: ChildStdout, door: LichDoor) {
520    tokio::spawn(hydra_mem::task(hydra_mem::Part::Lich, async move {
521        let mut buf = vec![0; 8 * 1024];
522        while let Ok(n @ 1..) = stdout.read(&mut buf).await {
523            door.show(&buf[..n]);
524        }
525    }));
526}

528/// Lich's last lines of standard error, and the task reading them.
529struct Errors {
530    kept: Arc<Mutex<VecDeque<String>>>,
531    /// Ends when Lich's standard error does; `None` once waited for.
532    reading: Option<JoinHandle<()>>,
533}
534
535impl Errors {
536    /// The lines kept, once Lich has ended: its last are waited for, no
537    /// longer than [`ERRORS_DEADLINE`], so what it died saying is among them.
538    async fn tail(&mut self) -> Vec<String> {
539        if let Some(reading) = self.reading.take() {
540            let _ = tokio::time::timeout(ERRORS_DEADLINE, reading).await;
541        }
542        self.kept
543            .lock()
544            .unwrap_or_else(PoisonError::into_inner)
545            .iter()
546            .cloned()
547            .collect()
548    }
549}
550
551/// Keeps Lich's last [`ERRORS_KEPT`] lines of standard error.
552fn keep_errors(stderr: ChildStderr) -> Errors {
553    let kept = Arc::new(Mutex::new(VecDeque::new()));
554    let keeping = Arc::clone(&kept);
555    let reading = tokio::spawn(hydra_mem::task(hydra_mem::Part::Lich, async move {
556        let mut lines = BufReader::new(stderr).lines();
557        while let Ok(Some(line)) = lines.next_line().await {
558            let mut kept = keeping.lock().unwrap_or_else(PoisonError::into_inner);
559            if kept.len() == ERRORS_KEPT {
560                kept.pop_front();
561            }
562            kept.push_back(line);
563        }
564    }));
565    Errors {
566        kept,
567        reading: Some(reading),
568    }
569}

571async fn exited(status: std::io::Result<ExitStatus>, errors: &mut Errors) -> Ended {
572    let status = match status {
573        Ok(status) => status.to_string(),
574        Err(why) => why.to_string(),
575    };
576    Ended::Exited {
577        status,
578        errors: errors.tail().await,
579    }
580}

582/// Stop Lich: close its standard input, and kill it if it has not exited
583/// within [`STOP_DEADLINE`].
584async fn finish(child: Child, stdin: ChildStdin) {
585    drop(stdin);
586    reap(child).await;
587}
588
589/// Wait [`STOP_DEADLINE`] for Lich to exit, its standard input closed, and
590/// kill it if it has not; then wait as long again, and no longer. On Windows
591/// a killed Ruby that had stopped reading was seen to take more than ten
592/// seconds to be reported gone, and a stop that waits on it for ever is the
593/// wedge [`feed`] exists to prevent.
594async fn reap(mut child: Child) {
595    if tokio::time::timeout(STOP_DEADLINE, child.wait())
596        .await
597        .is_err()
598    {
599        let _ = child.start_kill();
600        let _ = tokio::time::timeout(STOP_DEADLINE, child.wait()).await;
601    }
602}

606    use super::{accept_keyed, feed, shared_symbol};
607    use tokio::io::AsyncWriteExt;

609    use tokio::sync::mpsc::error::TrySendError;

611
612    /// A reader that never reads: the feed's queue fills, so the relay sees
613    /// it, and the write it is stuck in ends with `quit`.
614    #[tokio::test]
615    async fn a_feed_nobody_reads_is_refused_and_let_go() {
616        let (out, _unread) = tokio::io::duplex(64);
617        let quit = CancellationToken::new();
618        let (chunks, _) = feed(out, 4, quit.clone());
619        let mut sent = 0;
620        let refused = loop {
621            match chunks.try_send(vec![0; 64]) {
622                Ok(()) => sent += 1,
623                Err(why) => break why,
624            }
625            tokio::task::yield_now().await;
626            assert!(sent < 100, "the queue never filled");
627        };
628        assert!(matches!(refused, TrySendError::Full(_)));
629        quit.cancel();
630        tokio::time::timeout(std::time::Duration::from_secs(5), chunks.closed())
631            .await
632            .expect("the stuck write ended with quit");
633    }
634
635    /// What a reader never read is given back once the feed is let go: the
636    /// chunk its write was stopped in, and those that waited behind it.
637    #[tokio::test]
638    async fn a_feed_gives_back_what_it_never_wrote() {
639        let (out, _unread) = tokio::io::duplex(64);
640        let quit = CancellationToken::new();
641        let (chunks, unwritten) = feed(out, 4, quit.clone());
642        for n in 0..4u8 {
643            chunks.try_send(vec![n; 64]).unwrap();
644            tokio::task::yield_now().await;
645        }
646        quit.cancel();
647        let left = unwritten.await.unwrap();
648        assert_eq!(
649            left.iter().map(|chunk| chunk[0]).collect::<Vec<_>>(),
650            [1, 2, 3],
651            "the first fit the pipe; the rest were never written"
652        );
653    }
654
655    /// Hydra's default, `.`, leaves Lich's `;` alone; a player who gives
656    /// Hydra `;` is told Lich's commands are out of reach.
657    #[test]
658    fn only_a_shared_symbol_is_warned_of() {
659        assert!(shared_symbol(hydra_session::claimant::DEFAULT_SYMBOL, ';').is_none());
660        assert!(shared_symbol(';', ';').is_some());
661    }
662
663    /// Something on this machine that reaches the port first, without the
664    /// key, is let go, and Lich, with it, is taken.
665    #[tokio::test]
666    async fn only_a_connection_that_says_the_key_is_taken() {
667        let listener = TcpListener::bind(("127.0.0.1", 0)).await.unwrap();
668        let address = listener.local_addr().unwrap();
669        let clients = tokio::spawn(async move {
670            let mut stranger = TcpStream::connect(address).await.unwrap();
671            stranger.write_all(b"guess\nlook\n").await.unwrap();
672            let mut lich = TcpStream::connect(address).await.unwrap();
673            lich.write_all(b"the-key\n/FE:WRAYTH\n<c>look\n")
674                .await
675                .unwrap();
676            (stranger, lich)
677        });
678        let (mut reader, _writer) = accept_keyed(&listener, "the-key").await.unwrap();
679        let mut first = String::new();
680        tokio::io::AsyncBufReadExt::read_line(&mut reader, &mut first)
681            .await
682            .unwrap();
683        assert_eq!(
684            first, "<c>look\n",
685            "the key and version are read, and nothing after"
686        );
687        drop(clients.await.unwrap());
688    }

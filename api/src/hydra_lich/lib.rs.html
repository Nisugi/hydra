17/// The player's Ruby: `ruby` on the `PATH`, else the newest under Lich's
18/// Windows install, `C:\Ruby4Lich5`. `None` when there is neither.
19#[must_use]
20pub fn find_ruby() -> Option<PathBuf> {
21    let name = if cfg!(windows) { "ruby.exe" } else { "ruby" };
22    let on_path = std::env::var_os("PATH").and_then(|paths| {
23        std::env::split_paths(&paths)
24            .map(|dir| dir.join(name))
25            .find(|candidate| candidate.is_file())
26    });
27    on_path.or_else(|| lich_install(Path::new(r"C:\Ruby4Lich5"), name))
28}

30/// `<root>\<version>\bin\<name>`, the highest version first.
31fn lich_install(root: &Path, name: &str) -> Option<PathBuf> {
32    let mut versions: Vec<(Vec<u32>, PathBuf)> = std::fs::read_dir(root)
33        .ok()?
34        .filter_map(Result::ok)
35        .map(|entry| entry.path())
36        .filter(|version| version.join("bin").join(name).is_file())
37        .map(|version| {
38            let number = version
39                .file_name()
40                .and_then(|n| n.to_str())
41                .map(|n| {
42                    n.split('.')
43                        .map(|part| part.parse::<u32>().unwrap_or(0))
44                        .collect()
45                })
46                .unwrap_or_default();
47            (number, version)
48        })
49        .collect();
50    versions.sort();
51    versions
52        .pop()
53        .map(|(_, version)| version.join("bin").join(name))
54}

56/// A new secret: 32 random bytes, as hex.
57///
58/// # Errors
59///
60/// When the system gives no randomness.
61pub fn new_token() -> Result<String, String> {
62    const HEX: &[u8; 16] = b"0123456789abcdef";
63    let mut bytes = [0_u8; 32];
64    getrandom::fill(&mut bytes).map_err(|e| e.to_string())?;
65    Ok(bytes.iter().fold(String::with_capacity(64), |mut hex, b| {
66        hex.push(char::from(HEX[usize::from(b >> 4)]));
67        hex.push(char::from(HEX[usize::from(b & 0xf)]));
68        hex
69    }))
70}

72#[cfg(test)]
73mod tests {
74    use super::*;
75
76    #[test]
77    fn the_newest_ruby_lich_installed_is_found() {
78        let root = std::env::temp_dir().join(format!("hydra-ruby-{}", std::process::id()));
79        for version in ["3.4.9", "4.0.3", "4.0.10"] {
80            let bin = root.join(version).join("bin");
81            std::fs::create_dir_all(&bin).unwrap();
82            std::fs::write(bin.join("ruby.exe"), "").unwrap();
83        }
84        assert_eq!(
85            lich_install(&root, "ruby.exe"),
86            Some(root.join("4.0.10").join("bin").join("ruby.exe"))
87        );
88        assert_eq!(lich_install(&root.join("none"), "ruby.exe"), None);
89        let _ = std::fs::remove_dir_all(&root);
90    }
91
92    #[test]
93    fn a_token_is_new_each_time() {
94        let (one, two) = (new_token().unwrap(), new_token().unwrap());
95        assert_eq!(one.len(), 64);
96        assert!(one.bytes().all(|b| b.is_ascii_hexdigit()));
97        assert_ne!(one, two);
98    }

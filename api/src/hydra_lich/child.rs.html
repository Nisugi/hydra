1//! The player's own Lich, started as a child of Hydra.
2//!
3//! **It is never handed a password.** It runs code Hydra did not write: any
4//! script can read its environment (`ENV[...]` in Ruby). A password given
5//! through the environment, `HYDRA_PASSWORD_<ACCOUNT>` (the binary's
6//! `account::env_name`), lives in Hydra's own environment, which a child
7//! inherits unless told otherwise; [`command`] removes every variable of that
8//! family before anything else is set. The rest of the environment (`PATH`,
9//! Ruby's and the system's own variables) is inherited as before.
10//!
11//! **On Windows it has no console window.** Ruby is a console program, and
12//! one started by Hydra with no console of its own -- resting in the
13//! background, or started fresh -- would be given a window of its own,
14//! empty, since Lich talks to Hydra through pipes and sockets.
15
16use std::ffi::OsStr;
17
18/// What every password variable's name starts with: `HYDRA_PASSWORD_` and
19/// the account (the binary's `account::env_name`).
20pub const PASSWORD_PREFIX: &str = "HYDRA_PASSWORD_";
21
22/// A command for `program` whose environment is Hydra's without a password
23/// in it: every variable named [`PASSWORD_PREFIX`] and more, matched without
24/// regard to case, as Windows matches a variable's name. On Windows it
25/// opens no console window.
26#[must_use]
27pub fn command(program: impl AsRef<OsStr>) -> tokio::process::Command {
28    let mut command = tokio::process::Command::new(program);
29    for (name, _) in std::env::vars_os() {
30        if is_password(&name) {
31            command.env_remove(name);
32        }
33    }
34    #[cfg(windows)]
35    {
36        const CREATE_NO_WINDOW: u32 = 0x0800_0000;
37        command.creation_flags(CREATE_NO_WINDOW);
38    }
39    command
40}
41
42/// Whether the variable `name` holds a password.
43fn is_password(name: &OsStr) -> bool {
44    name.to_str().is_some_and(|name| {
45        name.get(..PASSWORD_PREFIX.len())
46            .is_some_and(|start| start.eq_ignore_ascii_case(PASSWORD_PREFIX))
47    })
48}

50#[cfg(test)]
51mod tests {
52    use super::*;
53
54    /// A variable set in the process that runs this test again; never in the
55    /// one `cargo test` started, which has none.
56    const PROBE: &str = "HYDRA_PASSWORD_PROBE";
57    /// Not a password: inherited, to show the child has Hydra's environment.
58    const KEPT: &str = "HYDRA_PROBE_KEPT";
59
60    #[test]
61    fn a_password_variable_is_known_by_its_name() {
62        assert!(is_password(OsStr::new("HYDRA_PASSWORD_NISUGI")));
63        assert!(is_password(OsStr::new("hydra_password_nisugi")));
64        assert!(!is_password(OsStr::new("HYDRA_MAP")));
65        assert!(!is_password(OsStr::new("PATH")));
66    }
67
68    /// A child started through [`command`] has Hydra's environment, less its
69    /// password. Rust 2024 makes setting a variable in this process unsafe,
70    /// so this test runs itself again in a process that has one, and that
71    /// process starts Ruby and asks what it sees.
72    #[tokio::test]
73    async fn a_child_is_never_handed_a_password() {
74        if std::env::var_os(PROBE).is_none() {
75            let me = std::env::current_exe().unwrap();
76            let ran = std::process::Command::new(me)
77                .args([
78                    "--exact",
79                    "child::tests::a_child_is_never_handed_a_password",
80                    "--nocapture",
81                ])
82                .env(PROBE, "hunter2")
83                .env(KEPT, "yes")
84                .output()
85                .unwrap();
86            let said = String::from_utf8_lossy(&ran.stdout);
87            let why = String::from_utf8_lossy(&ran.stderr);
88            assert!(ran.status.success(), "{said}{why}");
89            assert!(said.contains("1 passed"), "the inner run ran: {said}");
90            return;
91        }
92        let ruby = crate::find_ruby().expect("Ruby, which the CI runners have");
93        let seen = command(ruby)
94            .args(["-e", "print ENV.keys.join(\"\\n\")"])
95            .output()
96            .await
97            .unwrap();
98        let names = String::from_utf8_lossy(&seen.stdout);
99        let names: Vec<&str> = names.lines().collect();
100        assert!(names.contains(&KEPT), "the rest is inherited: {names:?}");
101        assert!(
102            !names.iter().any(|name| is_password(OsStr::new(name))),
103            "{names:?}"
104        );
105    }
106}

1//! Fetching the skill manager and applying goals to it, over HTTPS.
2//!
3//! One [`Trainer`] keeps the website's session from the page to the goals
4//! applied: the one-time link sets the session's cookie, and the form's
5//! post is answered only within it. The website answers a request without a
6//! browser's `User-Agent` with a 500, and a form post without a browser's
7//! headers may be turned away, so both are sent as a browser sends them.
8//! Nothing here is tested against the website: what it reads is
9//! ([`super::page`]), and where a fetch landed ([`landed`]).
10
11use std::time::Duration;
12
13use super::goals::SkillGoals;
14use super::page::{answer, form, read_page, resolved};
15
16/// The website.
17const BASE: &str = "https://www.play.net";
18
19/// A browser's `User-Agent`: without one the link is answered with a 500.
20const USER_AGENT: &str = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 \
21     (KHTML, like Gecko) Chrome/126.0.0.0 Safari/537.36";
22
23/// How long a request may take.
24const TIMEOUT: Duration = Duration::from_secs(20);
25
26/// What applying goals came to.
27#[derive(Clone, Copy, Debug, PartialEq, Eq)]
28pub enum Applied {
29    /// Saved, and the game told.
30    Saved,
31    /// Saved on the website, but the game could not be told.
32    NotTold,
33}
34
35/// Whether a fetch of the one-time link that landed at `url` reached the
36/// skill manager: the loader means the session never started; the home
37/// page, that the link was used or old.
38///
39/// # Errors
40///
41/// What went wrong, and what to do.
42pub fn landed(url: &str) -> Result<(), String> {
43    if url.contains("loader.asp") {
44        return Err("the website did not start its session: send GOALS again".to_owned());
45    }
46    if url.contains("home.asp") {
47        return Err("the link was used or old: send GOALS again for a new one".to_owned());
48    }
49    Ok(())
50}
51
52/// The website's session, from the skill manager page to goals applied.
53#[derive(Clone, Debug)]
54pub struct Trainer {
55    client: reqwest::Client,
56}
57
58impl Trainer {
59    /// A fresh session.
60    ///
61    /// # Errors
62    ///
63    /// Why the HTTP client could not be built.
64    pub fn new() -> Result<Self, String> {
65        let client = reqwest::Client::builder()
66            .cookie_store(true)
67            .redirect(reqwest::redirect::Policy::limited(8))
68            .user_agent(USER_AGENT)
69            .timeout(TIMEOUT)
70            .build()
71            .map_err(|why| why.to_string())?;
72        Ok(Self { client })
73    }
74
75    /// The skill manager the one-time `link` opens, read.
76    ///
77    /// # Errors
78    ///
79    /// Why it could not be fetched or read.
80    pub async fn fetch(&self, link: &str) -> Result<SkillGoals, String> {
81        let url = if link.starts_with("http") {
82            link.to_owned()
83        } else {
84            format!("{BASE}{link}")
85        };
86        let response = self.client.get(&url).send().await.map_err(|why| {
87            format!(
88                "the skill manager could not be fetched: {}",
89                why.without_url()
90            )
91        })?;
92        let at = response.url().to_string();
93        let page = response
94            .text()
95            .await
96            .map_err(|why| format!("the skill manager could not be read: {}", why.without_url()))?;
97        landed(&at)?;
98        read_page(&page, &at)
99    }
100
101    /// `goals` applied: the page's form posted with them, and the frame
102    /// that tells the game loaded, as a browser would.
103    ///
104    /// # Errors
105    ///
106    /// Why they were not saved.
107    pub async fn apply(&self, goals: &SkillGoals) -> Result<Applied, String> {
108        let action = resolved(&goals.page_url, &goals.action);
109        let response = self
110            .client
111            .post(&action)
112            .header("Referer", &goals.page_url)
113            .header("Origin", BASE)
114            .header(
115                "Accept",
116                "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
117            )
118            .header("Accept-Language", "en-US,en;q=0.9")
119            .header("Sec-Fetch-Site", "same-origin")
120            .header("Sec-Fetch-Mode", "navigate")
121            .header("Sec-Fetch-User", "?1")
122            .header("Sec-Fetch-Dest", "document")
123            .header("Upgrade-Insecure-Requests", "1")
124            .form(&form(goals))
125            .send()
126            .await
127            .map_err(|why| format!("the goals could not be sent: {}", why.without_url()))?;
128        let at = response.url().to_string();
129        let page = response.text().await.map_err(|why| {
130            format!(
131                "the website's answer could not be read: {}",
132                why.without_url()
133            )
134        })?;
135        let said = answer(&page);
136        let mut told = false;
137        if let Some(frame) = &said.delivery {
138            let url = resolved(&at, frame);
139            if let Ok(response) = self
140                .client
141                .get(&url)
142                .header("Referer", &at)
143                .header("Sec-Fetch-Dest", "iframe")
144                .send()
145                .await
146            {
147                let page = response.text().await.unwrap_or_default();
148                told = !page.to_lowercase().contains("unavailable");
149            }
150        }
151        if !said.saved {
152            return Err(
153                "the website did not say the goals were saved: send GOALS to see what it holds"
154                    .to_owned(),
155            );
156        }
157        Ok(if said.game_unavailable && !told {
158            Applied::NotTold
159        } else {
160            Applied::Saved
161        })
162    }
163}

165#[cfg(test)]
166mod tests {
167    use super::*;
168
169    #[test]
170    fn a_fetch_that_landed_elsewhere_says_why() {
171        assert_eq!(
172            landed("https://www.play.net/gs4/play/cm/skills.asp"),
173            Ok(())
174        );
175        assert!(landed("https://www.play.net/gs4/play/cm/loader.asp?x=1").is_err());
176        let old = landed("https://www.play.net/gs4/home.asp").unwrap_err();
177        assert!(old.contains("used or old"), "{old}");
178    }

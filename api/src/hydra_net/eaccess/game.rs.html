1//! The game socket: the second connection, and the handshake that makes it
2//! usable.
3//!
4//! **This is not the eaccess socket.** [`super::handshake`] finishes with a
5//! host, a port and a one-shot key over TLS; this opens a *plain* TCP
6//! connection to that host and hands the key over in the clear. It is a module
7//! of its own because nothing below speaks the tab-delimited `EAccess`
8//! protocol at all.
9

12use crate::gemstone::endpoint::other_spelling;

16/// Open the game socket and send the three-part handshake.
17///
18/// This is the second connection: **plain TCP, not TLS**, to the host and port
19/// the [`LaunchPayload`] names. The eaccess socket is already closed by the
20/// time this is called.
21///
22/// The handshake is key, then banner, then two `<c>` ready signals once the
23/// game is ready for them:
24///
25/// 1. `{key}\n` -- the one-shot key from `L`.
26/// 2. [`CLIENT_BANNER`] -- which selects the **extended feed**. See that
27///    constant: it is not cosmetic.
28/// 3. Wait for the game's `<settingsInfo>`, which says it is ready.
29/// 4. Two `<c>` signals, ~300ms apart.
30///
31/// What the game sent while the handshake waited is not lost: the
32/// [`GameSource`] returned reads it first, as if it had just arrived.
33///
34/// # The `<c>` signals, and why they are sent
35///
36/// The game server expects two `<c>` signals with a delay between them. An
37/// A/B run against the live server settled whether they matter:
38///
39/// | run | result |
40/// |---|---|
41/// | with the signals | **2,347 bytes** -- full login burst: room, exits, inventory, `exposeContainer` |
42/// | without | **163 bytes** -- stops at `<settingsInfo>` |
43///
44/// So they are **not required to log in, and required to be usable.** Both
45/// runs "connect"; only one produces a session worth having. Connected is not
46/// working, which is why the pass criterion is game text, not a TCP accept.
47///
48/// # Waiting for `<settingsInfo>` before sending them
49///
50/// Sent on a timer from the banner, the signals can arrive before the game
51/// is listening for them, and are lost: the game stops at `<settingsInfo>`
52/// as if they were never sent, and waits for the player's first command.
53/// Recorded: a team of four logged in within one second, and three stopped
54/// there; on another day, two of three. Every team whose logins were a
55/// second or more apart got through. The game takes longer to validate
56/// logins that arrive together, and `<settingsInfo>` is its word that it is
57/// ready. Should it never come, the signals go after 30 seconds
58/// (`READY_WAIT`), as they did before.
59///
60/// # The endpoint fallback
61///
62/// The connect is tried twice when the endpoint has a known second spelling:
63/// `connect_with_fallback` below has the detail, and `gemstone::endpoint`
64/// records why the table is a fallback rather than a pre-emptive rewrite.
65///
66/// # Errors
67///
68/// [`EaccessError`] with stage `game_connect` or `game_handshake`.
69pub async fn connect_game(payload: &LaunchPayload) -> Result<GameSource, EaccessError> {
70    let mut sock = connect_with_fallback(&payload.gamehost, payload.gameport)
71        .await
72        .map_err(|e| err("game_connect", e))?;
73
74    // One write per message throughout -- `ByteSource::write_all`'s contract.
75    let mut key_line = Vec::with_capacity(payload.key.len() + 1);
76    key_line.extend_from_slice(payload.key.as_bytes());
77    key_line.push(b'\n');
78    sock.write_all(&key_line)
79        .await
80        .map_err(|e| err("game_handshake", e))?;
81
82    let mut banner = Vec::with_capacity(CLIENT_BANNER.len() + 1);
83    banner.extend_from_slice(CLIENT_BANNER.as_bytes());
84    banner.push(b'\n');
85    sock.write_all(&banner)
86        .await
87        .map_err(|e| err("game_handshake", e))?;
88
89    let early = until_ready(&mut sock)
90        .await
91        .map_err(|e| err("game_handshake", e))?;
92
93    for _ in 0..2 {
94        sock.write_all(b"<c>\n")
95            .await
96            .map_err(|e| err("game_handshake", e))?;
97        tokio::time::sleep(READY_SIGNAL_GAP).await;
98    }
99
100    Ok(GameSource { early, sock })
101}

103/// What the game says when it is ready for the client's `<c>` signals.
104const READY: &[u8] = b"<settingsInfo";
105
106/// The longest the handshake waits to hear [`READY`] before sending the
107/// signals anyway: well past a slow validation, and only ever waited out by
108/// a game that never says it.
109const READY_WAIT: std::time::Duration = std::time::Duration::from_secs(30);
110
111/// Read `sock` until the game says it is ready ([`READY`]), it closes, or
112/// [`READY_WAIT`] passes: what it sent meanwhile.
113///
114/// # Errors
115///
116/// A read that failed.
117async fn until_ready(sock: &mut LiveSource) -> std::io::Result<Vec<u8>> {
118    let deadline = tokio::time::Instant::now() + READY_WAIT;
119    let mut early = Vec::new();
120    let mut buf = [0u8; 4096];
121    while !early.windows(READY.len()).any(|window| window == READY) {
122        match tokio::time::timeout_at(deadline, sock.read(&mut buf)).await {
123            // Waited long enough, or closed: the session finds the rest out.
124            Err(_) | Ok(Ok(0)) => break,
125            Ok(Ok(read)) => early.extend_from_slice(&buf[..read]),
126            Ok(Err(e)) => return Err(e),
127        }
128    }
129    Ok(early)
130}

132/// The game socket as the handshake leaves it: what the game sent while
133/// the handshake waited for it to be ready is read first, as if it had just
134/// arrived, then the socket.
135pub struct GameSource {
136    /// What the game sent before it said it was ready, not yet read.
137    early: Vec<u8>,
138    /// The socket.
139    sock: LiveSource,
140}
141
142impl std::fmt::Debug for GameSource {
143    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
144        f.debug_struct("GameSource")
145            .field("early", &self.early.len())
146            .finish_non_exhaustive()
147    }
148}
149
150impl ByteSource for GameSource {
151    async fn read(&mut self, buf: &mut [u8]) -> std::io::Result<usize> {
152        if self.early.is_empty() {
153            return self.sock.read(buf).await;
154        }
155        let read = self.early.len().min(buf.len());
156        buf[..read].copy_from_slice(&self.early[..read]);
157        self.early.drain(..read);
158        Ok(read)
159    }
160
161    async fn write_all(&mut self, message: &[u8]) -> std::io::Result<()> {
162        self.sock.write_all(message).await
163    }
164
165    async fn shutdown(&mut self) -> std::io::Result<()> {
166        self.sock.shutdown().await
167    }
168}

170/// The gap between the two `<c>` ready signals.
171///
172/// 300ms. Whether the *gap* matters, as against the two signals existing at
173/// all, is **UNVERIFIED** -- the live A/B varied their presence, not their
174/// spacing. It is `tokio::time::sleep`, so it is virtual under
175/// `tokio::time::pause()` and costs a test nothing.
176const READY_SIGNAL_GAP: std::time::Duration = std::time::Duration::from_millis(300);

178/// Connect to the game, retrying once against the endpoint's other spelling.
179///
180/// A connect failure is caught, the inverse host/port mapping applied, and
181/// the connect tried once more before giving up.
182///
183/// # Why the retry is scoped to the connect and no further
184///
185/// A connect failure is the only failure the other spelling could fix. Once
186/// bytes have been exchanged, a failure means something about *this* session --
187/// a rejected key, a closed socket mid-handshake -- and retrying against a
188/// different host would send the one-shot key to a second server. The key is
189/// single-use, so a handshake retry would burn it.
190///
191/// # Which error is reported when both fail
192///
193/// **The first one.** `L` named the first host, so its failure is the answer to
194/// "why could I not reach the server the login told me to use". The fallback's
195/// failure is a footnote: it is expected to fail whenever the table is simply
196/// not relevant to this endpoint.
197///
198/// # Then the WebSocket shim
199///
200/// When neither spelling connects, the same stream is tried through play.net's
201/// WebSocket shim on 443 (`crate::gemstone::shim`). That is for the network
202/// that blocks the game port and leaves 443 open -- a school, a hotel, a
203/// carrier -- and the fallback is automatic rather than a flag.
204///
205/// **Every connect failure triggers it**, not a chosen list of error kinds.
206/// A list would spare a delay on an error the shim would hit identically; but
207/// nothing has been exchanged at this point and the key is unspent, so the
208/// cost of trying is one bounded connect, and an allowlist of error KINDS is
209/// fragile: a DNS failure's kind differs by platform. A shim failure after a
210/// direct one reports both.
211async fn connect_with_fallback(host: &str, port: u16) -> std::io::Result<LiveSource> {
212    let first = match LiveSource::connect(host, port).await {
213        Ok(sock) => return Ok(sock),
214        Err(e) => e,
215    };
216
217    // No counterpart is the common case -- an endpoint the table does not know
218    // -- and it is not an error. There is simply nothing else to try on TCP.
219    if let Some((alt_host, alt_port)) = other_spelling(host, port)
220        && let Ok(sock) = LiveSource::connect(alt_host, alt_port).await
221    {
222        return Ok(sock);
223    }
224
225    match LiveSource::connect_shim(host, port).await {
226        Ok(sock) => {
227            // Said once, where the operator will see it: a session on the
228            // fallback behaves the same, and nothing else would tell them the
229            // game port is blocked where they are.
230            eprintln!(
231                "[game] {host}:{port} unreachable ({first}); connected through \
232                 the WebSocket shim at wss://{}/shim/{port}",
233                shim::shim_host(host)
234            );
235            Ok(sock)
236        }
237        Err(shim) => Err(std::io::Error::new(
238            first.kind(),
239            format!("{first}; the WebSocket shim fallback failed too: {shim}"),
240        )),
241    }

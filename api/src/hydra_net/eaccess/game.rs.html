1//! The game socket: the second connection, and the handshake that makes it
2//! usable.
3//!
4//! **This is not the eaccess socket.** [`super::handshake`] finishes with a
5//! host, a port and a one-shot key over TLS; this opens a *plain* TCP
6//! connection to that host and hands the key over in the clear. It is a module
7//! of its own because nothing below speaks the tab-delimited `EAccess`
8//! protocol at all.
9
10use super::wire::{CLIENT_BANNER, EaccessError, LaunchPayload, err};
11use crate::bytes::ByteSource;
12use crate::gemstone::endpoint::other_spelling;
13use crate::gemstone::shim;
14use crate::live::LiveSource;

16/// Open the game socket and send the three-part handshake.
17///
18/// This is the second connection: **plain TCP, not TLS**, to the host and port
19/// the [`LaunchPayload`] names. The eaccess socket is already closed by the
20/// time this is called.
21///
22/// The handshake is key, then banner, then two `<c>` ready signals:
23///
24/// 1. `{key}\n` -- the one-shot key from `L`.
25/// 2. [`CLIENT_BANNER`] -- which selects the **extended feed**. See that
26///    constant: it is not cosmetic.
27/// 3. Two `<c>` signals, ~300ms apart.
28///
29/// # The `<c>` signals, and why they are sent
30///
31/// The game server expects two `<c>` signals with a delay between them. An
32/// A/B run against the live server settled whether they matter:
33///
34/// | run | result |
35/// |---|---|
36/// | with the signals | **2,347 bytes** -- full login burst: room, exits, inventory, `exposeContainer` |
37/// | without | **163 bytes** -- stops at `<settingsInfo>` |
38///
39/// So they are **not required to log in, and required to be usable.** Both
40/// runs "connect"; only one produces a session worth having. Connected is not
41/// working, which is why the pass criterion is game text, not a TCP accept.
42///
43/// # The endpoint fallback
44///
45/// The connect is tried twice when the endpoint has a known second spelling:
46/// `connect_with_fallback` below has the detail, and `gemstone::endpoint`
47/// records why the table is a fallback rather than a pre-emptive rewrite.
48///
49/// # Errors
50///
51/// [`EaccessError`] with stage `game_connect` or `game_handshake`.
52pub async fn connect_game(payload: &LaunchPayload) -> Result<LiveSource, EaccessError> {
53    let mut sock = connect_with_fallback(&payload.gamehost, payload.gameport)
54        .await
55        .map_err(|e| err("game_connect", e))?;
56
57    // One write per message throughout -- `ByteSource::write_all`'s contract.
58    let mut key_line = Vec::with_capacity(payload.key.len() + 1);
59    key_line.extend_from_slice(payload.key.as_bytes());
60    key_line.push(b'\n');
61    sock.write_all(&key_line)
62        .await
63        .map_err(|e| err("game_handshake", e))?;
64
65    let mut banner = Vec::with_capacity(CLIENT_BANNER.len() + 1);
66    banner.extend_from_slice(CLIENT_BANNER.as_bytes());
67    banner.push(b'\n');
68    sock.write_all(&banner)
69        .await
70        .map_err(|e| err("game_handshake", e))?;
71
72    for _ in 0..2 {
73        sock.write_all(b"<c>\n")
74            .await
75            .map_err(|e| err("game_handshake", e))?;
76        tokio::time::sleep(READY_SIGNAL_GAP).await;
77    }
78
79    Ok(sock)
80}
81
82/// The gap between the two `<c>` ready signals.
83///
84/// 300ms. Whether the *gap* matters, as against the two signals existing at
85/// all, is **UNVERIFIED** -- the live A/B varied their presence, not their
86/// spacing. It is `tokio::time::sleep`, so it is virtual under
87/// `tokio::time::pause()` and costs a test nothing.
88const READY_SIGNAL_GAP: std::time::Duration = std::time::Duration::from_millis(300);

90/// Connect to the game, retrying once against the endpoint's other spelling.
91///
92/// A connect failure is caught, the inverse host/port mapping applied, and
93/// the connect tried once more before giving up.
94///
95/// # Why the retry is scoped to the connect and no further
96///
97/// A connect failure is the only failure the other spelling could fix. Once
98/// bytes have been exchanged, a failure means something about *this* session --
99/// a rejected key, a closed socket mid-handshake -- and retrying against a
100/// different host would send the one-shot key to a second server. The key is
101/// single-use, so a handshake retry would burn it.
102///
103/// # Which error is reported when both fail
104///
105/// **The first one.** `L` named the first host, so its failure is the answer to
106/// "why could I not reach the server the login told me to use". The fallback's
107/// failure is a footnote: it is expected to fail whenever the table is simply
108/// not relevant to this endpoint.
109///
110/// # Then the WebSocket shim
111///
112/// When neither spelling connects, the same stream is tried through play.net's
113/// WebSocket shim on 443 (`crate::gemstone::shim`). That is for the network
114/// that blocks the game port and leaves 443 open -- a school, a hotel, a
115/// carrier -- and the fallback is automatic rather than a flag.
116///
117/// **Every connect failure triggers it**, not a chosen list of error kinds.
118/// A list would spare a delay on an error the shim would hit identically; but
119/// nothing has been exchanged at this point and the key is unspent, so the
120/// cost of trying is one bounded connect, and an allowlist of error KINDS is
121/// fragile: a DNS failure's kind differs by platform. A shim failure after a
122/// direct one reports both.
123async fn connect_with_fallback(host: &str, port: u16) -> std::io::Result<LiveSource> {
124    let first = match LiveSource::connect(host, port).await {
125        Ok(sock) => return Ok(sock),
126        Err(e) => e,
127    };
128
129    // No counterpart is the common case -- an endpoint the table does not know
130    // -- and it is not an error. There is simply nothing else to try on TCP.
131    if let Some((alt_host, alt_port)) = other_spelling(host, port)
132        && let Ok(sock) = LiveSource::connect(alt_host, alt_port).await
133    {
134        return Ok(sock);
135    }
136
137    match LiveSource::connect_shim(host, port).await {
138        Ok(sock) => {
139            // Said once, where the operator will see it: a session on the
140            // fallback behaves the same, and nothing else would tell them the
141            // game port is blocked where they are.
142            eprintln!(
143                "[game] {host}:{port} unreachable ({first}); connected through \
144                 the WebSocket shim at wss://{}/shim/{port}",
145                shim::shim_host(host)
146            );
147            Ok(sock)
148        }
149        Err(shim) => Err(std::io::Error::new(
150            first.kind(),
151            format!("{first}; the WebSocket shim fallback failed too: {shim}"),
152        )),
153    }
154}

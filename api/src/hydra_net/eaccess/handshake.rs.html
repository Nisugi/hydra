1//! The handshake itself: the `K A M F G P C L` sequence over TLS.
2//!
3//! Everything here touches the **eaccess** socket. The pure half -- the types,
4//! the hash and the field parsers -- is in [`super::wire`], and that split is
5//! what makes the parsers testable at all. The *game* socket that this
6//! sequence's [`LaunchPayload`] points at is in [`super::game`].
7//!
8//! Read [`super`] for why `EAccess` lives in `hydra-net` and for the rule that
9//! no test logs in, which governs every function below.
10

22/// How long one stage may wait for its answer.
23///
24/// **Every read has its own deadline.** A whole-exchange watchdog alone would
25/// leave the TLS handshake and every protocol read unbounded underneath it.
26///
27/// 10s, the value a working login was measured against.
28///
29/// # This is a deadline, and a deadline is not `set_read_timeout`
30///
31/// Every wait must be cancellable, and a socket option is not. But every wait
32/// must also have a deadline: no unbounded `await`. Dropping the socket
33/// option without a replacement would leave every read unbounded.
34///
35/// `tokio::time::timeout` satisfies both: it is a deadline *and* it is
36/// cancellable, because dropping the future cancels the read with it. The two
37/// goals are not in conflict.
38const STAGE_DEADLINE: std::time::Duration = std::time::Duration::from_secs(10);
39
40/// How long the server's hash key is: 32 bytes on every login captured, read
41/// whole in one record. A reply of any other length is refused as this
42/// attempt's, before it is hashed with.
43const KEY_LEN: usize = 32;
44
45/// One read, bounded by [`STAGE_DEADLINE`].
46///
47/// **No response carries a terminator** -- not `\n`, not `\r\n` -- so there
48/// is nothing to scan for. We take what the socket gives and dispatch on the
49/// leading command letter.
50///
51/// The deadline names the stage it expired in. Without it a server that
52/// accepts a command and never answers leaves the program sitting with the
53/// *previous* stage's banner on screen, and the user cannot tell a hang from a
54/// slow link. A bad login must fail cleanly, naming the stage -- not hang.
55async fn read_response(
56    conn: &mut impl ByteSource,
57    stage: &'static str,
58) -> Result<String, EaccessError> {
59    let mut buf = vec![0u8; READ_BUF];
60    let read = tokio::time::timeout(STAGE_DEADLINE, conn.read(&mut buf))
61        .await
62        .map_err(|_elapsed| {
63            err(
64                stage,
65                format!(
66                    "no response within {STAGE_DEADLINE:?}. The command was \
67                     accepted but never answered, or the link stalled."
68                ),
69            )
70        })?;
71    let n = read.map_err(|e| err(stage, e))?;
72    if n == 0 {
73        return Err(err(stage, "connection closed by peer (0 bytes)"));
74    }
75    Ok(String::from_utf8_lossy(&buf[..n]).into_owned())
76}

78/// Send one line, as **one** write.
79///
80/// [`ByteSource::write_all`] carries the single-write contract; this just adds
81/// the newline to the same buffer rather than writing it separately.
82async fn send(
83    conn: &mut impl ByteSource,
84    line: &str,
85    stage: &'static str,
86) -> Result<(), EaccessError> {
87    let mut message = Vec::with_capacity(line.len() + 1);
88    message.extend_from_slice(line.as_bytes());
89    message.push(b'\n');
90    conn.write_all(&message).await.map_err(|e| err(stage, e))
91}

93/// Run the full `K A M F G P C L` handshake and return where the game is.
94///
95/// Each step's diagnostics go to `progress`, which receives already-redacted
96/// lines. A caller that wants them on stderr passes a closure that prints; a
97/// caller that wants them nowhere passes one that drops them. Taking a sink
98/// rather than printing directly keeps this module free of a policy about
99/// where a program's output goes -- and it is what lets the tests below exist
100/// at all.
101///
102/// `pin` is the certificate pin file (`<data dir>/`[`PIN_FILENAME`]): the
103/// server's certificate is recorded there on first use and must match it on
104/// every later login. See `pin.rs` for why a mismatch is fatal rather than
105/// silently re-pinned.
106///
107/// # Errors
108///
109/// [`EaccessError`], naming the stage. The failure paths are deliberately
110/// specific: a game code the server does not offer, a launch refusal, and a
111/// character that is not on the account each produce their own message rather
112/// than a generic rejection. A certificate that does not match the pin is a
113/// fatal `cert_pin` error naming the file.
114///
115/// [`PIN_FILENAME`]: super::PIN_FILENAME
116///
117/// # Panics
118///
119/// Does not panic.
120pub async fn authenticate(
121    creds: Credentials<'_>,
122    pin: &Path,
123    mut progress: impl FnMut(&str),
124) -> Result<LaunchPayload, EaccessError> {
125    // Three weakenings, all required, all documented on `connect_tls`: no SNI,
126    // no cert verification, no hostname check. The cert is self-signed with no
127    // chain, so there is nothing to verify against -- the PIN replaces that
128    // verification, and is checked before `converse` sends a byte.
129    progress(&format!(
130        "[stage: tls_handshake] {EACCESS_HOST}:{EACCESS_PORT}, no SNI"
131    ));
132    let mut conn = open_pinned(EACCESS_HOST, EACCESS_PORT, pin, &mut progress).await?;
133
134    let launch = converse(&mut conn, creds, &mut progress).await;
135
136    // The eaccess socket has done its job -- or failed at it. Close it either
137    // way: the game socket is a different connection, and leaving this one
138    // open is a leaked socket for the whole life of the session.
139    let _ = conn.shutdown().await;
140
141    launch
142}

144/// The `K A M F G P C L` conversation, over any [`ByteSource`].
145///
146/// # Why this is split from [`authenticate`]
147///
148/// So that a test can hold the other end. Every decision below -- the key used
149/// whole, a code `M` does not list refused fatally, a strange `L` redacted --
150/// would otherwise be reachable only through `connect_tls`, which no test may
151/// call. A test that pins "the key is used whole" by calling
152/// [`hash_password`] directly stays green with a trim added at the one call
153/// site that matters. Driving the conversation over
154/// [`crate::AnsweringSource`] closes that: `handshake_tests.rs` scripts each
155/// server reply and reads back the bytes this function actually wrote.
156///
157/// Generic rather than `&mut dyn`: `ByteSource`'s methods return `impl Future`,
158/// so it is not object-safe, and the generic costs one monomorphisation per
159/// caller -- `LiveSource` in production, `AnsweringSource` under test.
160pub(super) async fn converse(
161    conn: &mut impl ByteSource,
162    creds: Credentials<'_>,
163    progress: &mut impl FnMut(&str),
164) -> Result<LaunchPayload, EaccessError> {
165    prove_identity(conn, creds, progress).await?;
166    select_instance(conn, creds, progress).await?;
167    let char_code = resolve_character(conn, creds, progress).await?;
168    launch_character(conn, &char_code, progress).await
169}

171/// Ask which characters an account has on one game: the handshake's `K A M F
172/// G P C`, stopping before `L` launches anything.
173/// The window's login lists them; playing one logs in afresh, as a reconnect
174/// must. `creds.character` is not read.
175///
176/// # Errors
177///
178/// As [`authenticate`]'s, as far as `C`: a wrong password is `A`'s refusal,
179/// and counts against the account as any login's does.
180pub async fn list_characters(
181    creds: Credentials<'_>,
182    pin: &Path,
183    mut progress: impl FnMut(&str),
184) -> Result<Vec<String>, EaccessError> {
185    progress(&format!(
186        "[stage: tls_handshake] {EACCESS_HOST}:{EACCESS_PORT}, no SNI"
187    ));
188    let mut conn = open_pinned(EACCESS_HOST, EACCESS_PORT, pin, &mut progress).await?;
189    let listed = list(&mut conn, creds, &mut progress).await;
190    // Closed once the list is in: nothing waits on this connection.
191    let _ = conn.shutdown().await;
192    listed
193}

195/// The `K A M F G P C` conversation over any [`ByteSource`]: what
196/// [`list_characters`] runs, without the connect, as [`converse`] is
197/// [`authenticate`]'s.
198pub(super) async fn list(
199    conn: &mut impl ByteSource,
200    creds: Credentials<'_>,
201    progress: &mut impl FnMut(&str),
202) -> Result<Vec<String>, EaccessError> {
203    prove_identity(conn, creds, progress).await?;
204    select_instance(conn, creds, progress).await?;
205    send(conn, "C", "c_request").await?;
206    let c = read_response(conn, "c_response").await?;
207    expect_echo(&c, 'C', "c_response")?;
208    Ok(character_names(&c).into_iter().map(str::to_owned).collect())
209}

211/// `K` then `A`: get the server's hash key, and prove we know the password.
212///
213/// One of the steps [`converse`] runs. Kept apart so that no scope holds more
214/// than a few single-letter bindings (`a`, `m`, `f`, `g`, `c`, `l` -- one per
215/// protocol letter). The letters stay: the response to `A` being called `a` is
216/// the clearest name available, once only one of them is in scope at a time.
217async fn prove_identity(
218    conn: &mut impl ByteSource,
219    creds: Credentials<'_>,
220    progress: &mut impl FnMut(&str),
221) -> Result<(), EaccessError> {
222    send(conn, "K", "k_request").await?;
223    // Read as BYTES, not through `read_response`: the key is key material and
224    // need not be valid UTF-8 (a captured key contained 0x7F), so a
225    // `from_utf8_lossy` round trip could alter it before the XOR. Bounded by
226    // the same [`STAGE_DEADLINE`] as every other read.
227    let key_raw = {
228        let mut buf = vec![0u8; READ_BUF];
229        let n = tokio::time::timeout(STAGE_DEADLINE, conn.read(&mut buf))
230            .await
231            .map_err(|_elapsed| {
232                err(
233                    "k_response",
234                    format!("no hash key within {STAGE_DEADLINE:?}"),
235                )
236            })?
237            .map_err(|e| err("k_response", e))?;
238        if n == 0 {
239            return Err(err("k_response", "connection closed reading the hash key"));
240        }
241        buf.truncate(n);
242        buf
243    };
244    // **The key is used WHOLE. Nothing is trimmed off it.**
245    //
246    // A capture settled it: `K` answers 32 random bytes, **not printable
247    // ASCII** (one contained DEL), with no trailing newline, so the hash loop
248    // consumes exactly the bytes sent. The key is 32 bytes of RANDOM BINARY
249    // with no terminator, so a leading `0x09`, `0x0A`, `0x0D` or `0x20` is an
250    // ordinary key byte and not framing.
251    //
252    // Trimming one shifts every XOR index by one. Nothing errors: a WRONG
253    // PASSWORD goes to the auth server, which answers `PASSWORD`, which is a
254    // FATAL stop plus one bad-password strike against the account.
255    //
256    // 4 of 256 first bytes are whitespace, so a trim would break roughly one
257    // login in 64 -- not an exotic case -- and buys nothing in either
258    // direction.
259    //
260    // `handshake_tests.rs` pins it HERE, at the call site, by reading back the
261    // `A` line written for a key that starts with 0x20. The vectors in
262    // `tests/eaccess_mandated_vectors.rs` pin only the hash's arithmetic, and
263    // would stay green with a trim on this line.
264    let key = key_raw.as_slice();
265    if key.len() != KEY_LEN {
266        // Transient: a read that came up short is this attempt's, not the
267        // account's. Unchecked, a short key would reach `hash_password`, which
268        // would hash against it (a wrong password sent, one strike) or refuse
269        // a password longer than it as FATAL, which stops the retries and
270        // skips the web login.
271        return Err(err(
272            "k_response",
273            format!(
274                "MALFORMED_K_RESPONSE ({} bytes, {KEY_LEN} expected)",
275                key.len()
276            ),
277        ));
278    }
279    progress(&format!(
280        "[stage: k_response] {} bytes, used whole (not printed -- key material)",
281        key.len()
282    ));
283
284    // The hashed password is ARBITRARY BYTES, not UTF-8. Routing it through a
285    // String mangles every byte above 0x7F into U+FFFD, silently corrupting
286    // the credential. Build the request as bytes.
287    let hashed = hash_password(creds.password.as_bytes(), key)?;
288    let mut a_req = Vec::with_capacity(creds.account.len() + hashed.len() + 8);
289    a_req.extend_from_slice(b"A\t");
290    a_req.extend_from_slice(creds.account.as_bytes());
291    a_req.push(b'\t');
292    a_req.extend_from_slice(&hashed);
293    a_req.push(b'\n');
294    conn.write_all(&a_req)
295        .await
296        .map_err(|e| err("a_request", e))?;
297
298    let a = read_response(conn, "a_response").await?;
299    if !a.contains("\tKEY\t") {
300        // **Two cases, and they do NOT share a verdict**, and neither echoes
301        // the raw body. See `super::reject` for the full argument.
302        //
303        // In short: a recognised rejection token is the credentials being wrong,
304        // which retrying cannot fix and which costs a strike. Anything else is
305        // "we do not know what that was", where stopping strands a session whose
306        // credentials are good -- the direction `wire.rs` itself calls unsafe.
307        let rejection = super::reject::classify_a_rejection(&a);
308        let detail = match &rejection {
309            super::reject::Rejection::Credentials(token) => {
310                format!("authentication rejected by the server: {token}")
311            }
312            // The body is deliberately absent: an unrecognised reply is the case
313            // most likely to be a WAF intercept page or a truncated fragment,
314            // and a length is all a diagnostic needs.
315            super::reject::Rejection::Divergence { bytes } => format!(
316                "unrecognised response to A ({bytes} bytes, body not logged -- it \
317                 may be an intercept page). Treated as TRANSIENT: this is not \
318                 evidence the credentials are wrong, and stopping here would \
319                 strand a session a retry could recover."
320            ),
321        };
322        let error = err("a_response", detail);
323        return Err(if rejection.is_fatal() {
324            error.fatal()
325        } else {
326            error
327        });
328    }
329    progress(&format!("[stage: a_response] OK: {}", redact(a.trim())));
330    Ok(())
331}

333/// `M`, `F`, `G`, `P`: check the instance exists, that the account may enter
334/// it, and select it.
335///
336/// **`M` is used as a guard, not for information.** A login can skip it and
337/// go from `A` straight to `F`; we send it because it is the only place the
338/// server will tell us a game code is wrong.
339///
340/// REFUSE, don't warn. A code missing from `M` is unrecognised, and the server
341/// does **not** say so at the point of use: `F\t{bad}` still answers -- about
342/// the account's DEFAULT instance -- so the session silently proceeds pointed
343/// at the wrong game and fails four commands later at `L` with PROBLEM 3.
344/// MEASURED: `gs3` lowercase produced exactly that chain (F=PREMIUM, C=16
345/// slots, L=PROBLEM 3).
346async fn select_instance(
347    conn: &mut impl ByteSource,
348    creds: Credentials<'_>,
349    progress: &mut impl FnMut(&str),
350) -> Result<(), EaccessError> {
351    send(conn, "M", "m_request").await?;
352    let m = read_response(conn, "m_response").await?;
353    expect_echo(&m, 'M', "m_response")?;
354    let codes = offered_game_codes(&m);
355    if !codes.contains(&creds.game_code) {
356        // FATAL: the server's own list of instances does not contain this code.
357        // No number of retries adds one.
358        //
359        // Without the `.fatal()` the error would default to transient -- and a
360        // transient eaccess error falls back to web login, which refuses the
361        // same code as `UnsupportedGameCode`, which `carry_forward`
362        // deliberately does not treat as a verdict. So a mistyped `GS4` would
363        // become a supervisor retrying forever, each attempt a FULL login that
364        // sends the password to `A` every 30 seconds: the exact account-lock
365        // pattern the mistyped CHARACTER name below is fatal to stop.
366        return Err(err(
367            "m_response",
368            format!(
369                "game code {:?} is not offered by the server. Codes are \
370                 CASE-SENSITIVE; the server offers: {}. An unrecognised code \
371                 does not fail here -- F answers about the account's default \
372                 instance instead, and the login fails later at L with an \
373                 unrelated-looking PROBLEM.",
374                creds.game_code,
375                codes.join(", ")
376            ),
377        )
378        .fatal());
379    }
380    // The WHOLE line, not just the parsed codes. `offered_game_codes` keeps
381    // every other field, which drops the human-readable instance names -- and
382    // those are what tell the user at a glance whether GST is the instance
383    // they meant. A login is one-shot, so a diagnostic dropped here cannot be
384    // recovered without a second live login.
385    progress(&format!("[stage: m_response] {:?}", m.trim()));
386    progress(&format!("[stage: m_response] codes: {}", codes.join(", ")));
387
388    // F: entitlement.
389    send(conn, &format!("F\t{}", creds.game_code), "f_request").await?;
390    let f = read_response(conn, "f_response").await?;
391    expect_echo(&f, 'F', "f_response")?;
392    if !["NORMAL", "PREMIUM", "TRIAL", "INTERNAL", "FREE"]
393        .iter()
394        .any(|t| f.contains(t))
395    {
396        // FATAL: an entitlement is account state. The fix is a subscription,
397        // not another login -- the same reasoning `refusal.rs` gives for
398        // PROBLEM 1.
399        return Err(err(
400            "f_response",
401            format!("no entitlement for {}: {}", creds.game_code, f.trim()),
402        )
403        .fatal());
404    }
405    // The RAW response, not a parsed field: parsing before seeing it fail
406    // turned the server's bare "?" into an innocuous `tier="?"`.
407    progress(&format!("[stage: f_response] {:?}", f.trim()));
408
409    // G SELECTS the game; it is not merely an info query.
410    // Everything after it -- C's character list, and therefore L -- depends on
411    // it having succeeded.
412    send(conn, &format!("G\t{}", creds.game_code), "g_request").await?;
413    let g = read_response(conn, "g_response").await?;
414    expect_echo(&g, 'G', "g_response")?;
415    progress(&format!("[stage: g_response] {:?}", g.trim()));
416
417    // P: pricing. TESTED AND DISPROVEN that it is what moves the session: a
418    // run with P removed still produced 16 slots and PROBLEM 3. Sent anyway,
419    // because a known-working login sends it.
420    send(conn, &format!("P\t{}", creds.game_code), "p_request").await?;
421    let p = read_response(conn, "p_response").await?;
422    // PRINTED, not discarded. Only this response shows an open anomaly: in
423    // one capture, `P` asked about **GST** and answered
424    // `P\tGSX\t1000\tGSX.EC\t-1\tGS3.P\t2500` -- GST's documented pricing
425    // under a **GSX** code, still unexplained. Discarding it here would remove
426    // the one place a reader could ever see the echo recur.
427    progress(&format!("[stage: p_response] {:?}", p.trim()));
428    Ok(())
429}

431/// `C`: list the account's characters on the selected instance and find the
432/// one asked for.
433async fn resolve_character(
434    conn: &mut impl ByteSource,
435    creds: Credentials<'_>,
436    progress: &mut impl FnMut(&str),
437) -> Result<String, EaccessError> {
438    send(conn, "C", "c_request").await?;
439    let c = read_response(conn, "c_response").await?;
440    expect_echo(&c, 'C', "c_response")?;
441    // The second field is the account's MAX CHARACTER SLOTS.
442    //
443    // **It reflects the account's ENTITLEMENT, not which instance was
444    // selected.** Reading 100 as GST and 16 as premium is wrong on any
445    // account holding more than one entitlement: such an account may be
446    // offered ten codes by `M`, and 16 is simply what its tier grants, on
447    // several of them.
448    //
449    // What a login can show is a *contrast*: 16 where a working login showed
450    // 100, for the same account and character. The signal is the
451    // disagreement, never the value. A bare count identifies nothing.
452    //
453    // Context for a human reading a failure, not a check.
454    //
455    // Nor is the instance confirmed by F, G and P echoing the code that was
456    // sent:
457    //
458    // * `expect_echo` checks the command LETTER only -- `response.starts_with("G\t")`
459    //   -- and never the code.
460    // * `F` does not echo the code at all. It answers an entitlement word
461    //   (`NORMAL`, `PREMIUM`, `TRIAL`, `INTERNAL`, `FREE`), which the arm
462    //   above checks for by name.
463    // * `P` is not passed to `expect_echo` anywhere; its response is only
464    //   printed, as the comment at the `P` send records.
465    //
466    // So nothing verifies that the instance the server selected is the
467    // instance that was asked for. `refusal.rs`'s PROBLEM 3 advice says a
468    // rejected `G` produces exactly that failure, which is the case this
469    // would catch -- but catching it means comparing `G`'s echoed field to
470    // `creds.game_code`, which is a check nobody has written. Left unwritten
471    // rather than half-claimed: an unverified assumption stated as a
472    // guarantee is worse than one stated as an assumption.
473    if let Some(slots) = c.trim().split('\t').nth(2) {
474        progress(&format!(
475            "[stage: c_response] max slots={slots} for {} (entitlement, not \
476             instance -- a count that DISAGREES with a previous login on the \
477             same code is the signal, not any particular number)",
478            creds.game_code
479        ));
480    }
481
482    let code = resolve_char_code(&c, creds.character).ok_or_else(|| {
483        // FATAL: a MISTYPED CHARACTER NAME retried every 30 seconds sends the
484        // password to eaccess indefinitely, which risks locking the account.
485        err(
486            "resolve_char",
487            format!(
488                "character {:?} is not on this account's {} list",
489                creds.character, creds.game_code
490            ),
491        )
492        .fatal()
493    })?;
494    // {:?} so a stray control byte in the parsed code is visible rather than
495    // invisibly breaking the L request -- and REDACTED, because a character code
496    // is `W_<ACCOUNT>_<SLOT>`, and unredacted this line would print the
497    // account on every login. The same exposure `redact` covers for the `A`
498    // response, but here the account is inside a field rather than a field of
499    // its own.
500    progress(&format!(
501        "[stage: resolve_char] code={:?}",
502        redact_char_code(code)
503    ));
504    Ok(code.to_owned())
505}

507/// `L`: launch, and read back where the game is.
508///
509/// The success guard MUST be `L\tOK`, not `^L\t`. A refusal is
510/// `L\tPROBLEM\t<n>`, which also starts with `L\t`; a loose guard accepts it
511/// and then parses a garbage launch payload.
512async fn launch_character(
513    conn: &mut impl ByteSource,
514    char_code: &str,
515    progress: &mut impl FnMut(&str),
516) -> Result<LaunchPayload, EaccessError> {
517    send(conn, &format!("L\t{char_code}\tSTORM"), "l_request").await?;
518    let l = read_response(conn, "l_response").await?;
519    if !is_launch_ok(&l) {
520        // Fatal for sub-codes 1-3 and NOT for 4: `launch_refusal_is_fatal`
521        // reads them one at a time, because "the account service failed while
522        // assigning the character" is a hiccup and the other three are not.
523        let error = err("l_response", describe_launch_refusal(&l));
524        return Err(if launch_refusal_is_fatal(&l) {
525            error.fatal()
526        } else {
527            error
528        });
529    }
530    progress(&format!("[stage: l_response] {}", redact(l.trim())));
531    parse_launch(&l)
532}

1//! What an `L\tPROBLEM\t<n>` refusal means, and whether to retry it.
2//!
3//! Apart from [`super::wire`] because it is a lookup table over one wire
4//! value, with no dependency on the rest of the vocabulary.
5//!
6//! # All four codes are VERIFIED, and three of them must not be retried
7//!
8//! Their meanings are the ones **Simutronics' own client shows the user**,
9//! not inferred from control flow. The retry verdict is the operational half:
10//! a blanket retry is wrong for **2** and **3**, which are server-side
11//! configuration facts that will not change between attempts, so retrying
12//! burns logins to reach the same refusal and delays the real message to the
13//! user.
14
15use super::wire::redact;
16
17/// Explain an `L` refusal.
18///
19/// A function of its own so the explanations are testable without a live
20/// login: a reading that cannot be re-checked is one that quietly becomes
21/// folklore.
22///
23/// # The reply is REDACTED before it is quoted
24///
25/// Both arms quote the raw `L` line, and this string becomes an
26/// [`EaccessError`](super::EaccessError) detail -- which `main` prints to
27/// stderr and the session writes to its `.log`. The arm that matters is the
28/// one for a reply that is **neither** `L\tOK` nor `L\tPROBLEM`: nothing here
29/// knows what it carries, and an `L` reply is the one response in the
30/// sequence built to carry a `KEY=`. The success line in `handshake.rs` is
31/// printed through [`redact`](super::redact), and so is this one, on the path
32/// whose premise is that the reply is not the shape expected -- the same
33/// reasoning `expect_echo` records for the same exposure.
34#[must_use]
35pub fn describe_launch_refusal(l: &str) -> String {
36    let quoted = redact(l.trim());
37    let Some(rest) = l.trim().strip_prefix("L\tPROBLEM") else {
38        return format!("launch refused: {quoted}");
39    };
40    // All four codes are VERIFIED against the meanings Simutronics' own client
41    // shows the user. 3 is server-side, not "the character code is not valid
42    // on the selected instance"; and the slot count is no help here, since it
43    // reflects the account's ENTITLEMENT rather than the selected instance --
44    // which any account holding several entitlements makes plain.
45    //
46    // The retry verdict is the operational point: a blanket retry is wrong
47    // for 2 and 3, which are server-side facts that will not change between
48    // attempts.
49    let detail = match rest.trim_start().split('\t').next().map(str::trim) {
50        Some("1") => {
51            "the account's access level does not permit playing this instance \
52             -- a lapsed or missing subscription -- or the account service \
53             timed out. DO NOT RETRY: this is account state, and the fix is a \
54             subscription, not another login."
55        }
56        Some("2") => {
57            "the server has no STORM launch entry for this game's \
58             configuration. SERVER-SIDE, not an account problem. DO NOT RETRY: \
59             it will not change between attempts."
60        }
61        Some("3") => {
62            "the server has no configuration for the selected game. \
63             SERVER-SIDE. DO NOT RETRY. Check first that the game code was \
64             listed by M, and that G selected the instance you meant -- a \
65             rejected G leaves the session pointed at an instance the server \
66             cannot launch, which produces exactly this, and NOTHING IN THIS \
67             CLIENT VERIFIES IT: the handshake checks each response's command \
68             letter, not the code it carries. Byte-diffing the L request is a \
69             dead end: its bytes are correct in this failure."
70        }
71        Some("4") => "the account service failed while assigning the character. RETRY: transient.",
72        _ => {
73            "unrecognised PROBLEM sub-code. Codes 1-4 are known; a fifth \
74             means the server grew one."
75        }
76    };
77    format!("launch refused ({quoted}): {detail}")
78}

80/// Whether an `L PROBLEM` refusal is worth another attempt.
81///
82/// # Three of four are fatal, and the fourth is NOT
83///
84/// "Every launch refusal is fatal" would be wrong. **Not all four sub-codes
85/// advise against retrying**, as the messages themselves show:
86///
87/// | Code | Verdict | Why |
88/// |---|---|---|
89/// | 1 | fatal | account state -- the fix is a subscription, not a login |
90/// | 2 | fatal | server-side, will not change between attempts |
91/// | 3 | fatal | server-side, no configuration for the game |
92/// | 4 | **transient** | *"the account service failed while assigning the character"* |
93///
94/// 4 is a service hiccup. Treating it as fatal would strand a session on a
95/// failure that a second attempt fixes.
96///
97/// An unrecognised sub-code is **transient**, the same fail-safe default
98/// `EaccessError::fatal` takes: a fifth code the server grew is not something
99/// to give up on without evidence.
100#[must_use]
101pub fn launch_refusal_is_fatal(l: &str) -> bool {
102    let Some(rest) = l.trim().strip_prefix("L\tPROBLEM") else {
103        // Not a PROBLEM at all -- an unparseable L. Retryable, because nothing
104        // here says the account is at fault.
105        return false;
106    };
107    matches!(
108        rest.trim_start().split('\t').next().map(str::trim),
109        Some("1" | "2" | "3")
110    )
111}

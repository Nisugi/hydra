1//! The game stream over play.net's WebSocket shim: `wss://<host>/shim/<port>`.
2//!
3//! # What this is for
4//!
5//! A network that allows only 80 and 443 -- a school, a hotel, some carriers
6//! and VPN egress rules -- cannot reach the game port at all. play.net's own
7//! browser client gets through such a network with a WebSocket-to-TCP shim on
8//! 443, and once the WebSocket is open **the bytes are the same stream**: the
9//! same key, the same banner, the same newline-delimited XML. Nothing above
10//! the transport changes.
11//!
12//! The behaviour here follows the browser client's connection code
13//! (`SimuSocket.tryWebSocket` in play.net's `all_web_fe_min.js`). No test
14//! reaches the real shim; live logins through it are run by hand.
15//!
16//! # The shim is not dialled at GAMEHOST
17//!
18//! The browser client remaps the host it was given to one of two fixed names,
19//! by substring, `GemStone` first ([`shim_host`]). Both are `*.play.net` names
20//! on the shared edge certificate, so **ordinary, unweakened TLS verification
21//! passes** -- unlike the eaccess socket (`live.rs`), which needs three
22//! weakenings. This is the first connection in this crate that verifies its
23//! peer.
24//!
25//! # Framing
26//!
27//! Each [`ByteSource::write_all`](crate::bytes::ByteSource::write_all) is one
28//! text message, which is what the browser sends per command. Inbound text and binary messages are both game bytes and
29//! are handed up unchanged: a message boundary is a read boundary, no more,
30//! and the parser already reassembles lines across reads.
31

44/// The shim's port.
45pub const SHIM_PORT: u16 = 443;
46
47/// `Sec-WebSocket-Protocol`, as the browser client requests it.
48const SUBPROTOCOL: &str = "websocket_shim-protocol";
49
50/// A desktop Chrome user agent. play.net's WAF has refused requests that did
51/// not look like the browser client they share this endpoint with.
52const USER_AGENT: &str = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 \
53                          (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36";
54
55/// Bound on the TLS handshake plus the HTTP upgrade. The TCP connect before it
56/// has its own bound (`live.rs`'s `CONNECT_TIMEOUT`); this matches the eaccess
57/// socket's TLS bound, for the same reason -- neither is covered by the other.
58const HANDSHAKE_TIMEOUT: Duration = Duration::from_secs(10);
59
60/// Bound on the closing handshake.
61///
62/// The plain game socket's `shutdown` cannot hang, and the session relies on
63/// that: the session's shutdown awaits it with no deadline of its own. A
64/// WebSocket close is a round trip over TLS, and a peer that never answers must
65/// not turn an orderly stop into a stuck one, so the bound lives here, in the
66/// one source that needs it.
67const CLOSE_TIMEOUT: Duration = Duration::from_secs(2);
68
69/// The host the shim is dialled at, for the game host the login named.
70///
71/// The browser client's own rule (`all_web_fe_min.js`), checked in its order,
72/// first match wins:
73///
74/// | GAMEHOST contains | dial |
75/// |---|---|
76/// | `gs` or `chimera` | `chimera.play.net` |
77/// | `dr` or `hydra` | `hydra.play.net` |
78/// | neither | GAMEHOST unchanged |
79///
80/// The last row is dead in practice -- every known instance matches one of
81/// the first two -- and is kept because it is what the browser client does.
82/// The match is case-insensitive.
83#[must_use]
84pub fn shim_host(gamehost: &str) -> &str {
85    let lower = gamehost.to_ascii_lowercase();
86    if lower.contains("gs") || lower.contains("chimera") {
87        "chimera.play.net"
88    } else if lower.contains("dr") || lower.contains("hydra") {
89        "hydra.play.net"
90    } else {
91        gamehost
92    }
93}

95/// The upgrade request: `wss://<host>/shim/<gameport>`, with the headers the
96/// browser client sends.
97///
98/// The game port rides in the PATH, not the connection: the socket goes to
99/// 443 and the shim forwards to the port named here.
100///
101/// # Errors
102///
103/// A host that does not form a valid URI.
104pub fn shim_request(host: &str, gameport: u16) -> io::Result<Request> {
105    let mut request = format!("wss://{host}/shim/{gameport}")
106        .into_client_request()
107        .map_err(io::Error::other)?;
108    let origin = HeaderValue::from_str(&format!("https://{host}")).map_err(io::Error::other)?;
109    let headers = request.headers_mut();
110    headers.insert(
111        "Sec-WebSocket-Protocol",
112        HeaderValue::from_static(SUBPROTOCOL),
113    );
114    headers.insert("Origin", origin);
115    headers.insert("User-Agent", HeaderValue::from_static(USER_AGENT));
116    Ok(request)
117}

119/// The TLS stream the shim runs over in production.
120type TlsTcp = tokio_native_tls::TlsStream<TcpStream>;
121
122/// A WebSocket carrying the game stream.
123///
124/// Generic over the transport only so the framing can be tested over an
125/// in-memory pipe; production is always [`TlsTcp`].
126#[derive(Debug)]
127pub struct ShimStream<S = TlsTcp> {
128    ws: WebSocketStream<S>,
129    /// The rest of a message longer than the caller's buffer.
130    unread: Bytes,
131}
132
133impl ShimStream {
134    /// TLS to `host` over an already-connected `tcp`, then the upgrade.
135    ///
136    /// **Verification is ON**, with SNI: `TlsConnector::new()`, none of the
137    /// eaccess socket's weakenings. See the module docs for why it passes.
138    ///
139    /// # Errors
140    ///
141    /// TLS or upgrade failure, or [`HANDSHAKE_TIMEOUT`] passing.
142    pub async fn open(tcp: TcpStream, host: &str, gameport: u16) -> io::Result<Self> {
143        let request = shim_request(host, gameport)?;
144        let connector = native_tls::TlsConnector::new().map_err(io::Error::other)?;
145        let connector = tokio_native_tls::TlsConnector::from(connector);
146        let handshake = async {
147            let tls = connector
148                .connect(host, tcp)
149                .await
150                .map_err(|e| io::Error::other(format!("tls handshake with {host}: {e}")))?;
151            Self::upgrade(request, tls).await
152        };
153        tokio::time::timeout(HANDSHAKE_TIMEOUT, handshake)
154            .await
155            .map_err(|_elapsed| {
156                io::Error::new(
157                    io::ErrorKind::TimedOut,
158                    format!("websocket shim at {host} did not open within {HANDSHAKE_TIMEOUT:?}"),
159                )
160            })?
161    }
162}
163
164impl<S: AsyncRead + AsyncWrite + Unpin> ShimStream<S> {
165    /// The HTTP upgrade over an established stream.
166    ///
167    /// tungstenite checks the `101`, `Sec-WebSocket-Accept`, and that the
168    /// server chose the subprotocol requested.
169    ///
170    /// # Errors
171    ///
172    /// Any upgrade failure, as an [`io::Error`].
173    pub async fn upgrade(request: Request, stream: S) -> io::Result<Self> {
174        let (ws, _response) = tokio_tungstenite::client_async(request, stream)
175            .await
176            .map_err(|e| io::Error::other(format!("websocket upgrade: {e}")))?;
177        Ok(Self {
178            ws,
179            unread: Bytes::new(),
180        })
181    }
182
183    /// Read game bytes. `Ok(0)` is the end of the stream, including a close.
184    ///
185    /// Cancel-safe, which the session's read deadline requires: the only
186    /// await is the stream's `next`, which loses nothing when dropped, and
187    /// the copy out of `unread` happens after it returns.
188    ///
189    /// # Errors
190    ///
191    /// A transport or protocol error.
192    pub async fn read(&mut self, buf: &mut [u8]) -> io::Result<usize> {
193        loop {
194            if !self.unread.is_empty() {
195                let n = buf.len().min(self.unread.len());
196                buf[..n].copy_from_slice(&self.unread.split_to(n));
197                return Ok(n);
198            }
199            let Some(message) = self.ws.next().await else {
200                return Ok(0);
201            };
202            match message.map_err(ws_to_io) {
203                // An empty message leaves `unread` empty and the loop reads
204                // on: returning `Ok(0)` for it would claim the stream ended.
205                Ok(message @ (Message::Text(_) | Message::Binary(_))) => {
206                    self.unread = message.into_data();
207                }
208                Ok(Message::Close(_)) => return Ok(0),
209                // tungstenite answers pings itself; neither carries game bytes.
210                Ok(Message::Ping(_) | Message::Pong(_) | Message::Frame(_)) => {}
211                Err(e) if e.kind() == io::ErrorKind::NotConnected => return Ok(0),
212                Err(e) => return Err(e),
213            }
214        }
215    }
216
217    /// Send one whole message as one text frame.
218    ///
219    /// # Errors
220    ///
221    /// A message that is not UTF-8 (a text frame must be), or a send failure.
222    pub async fn write_all(&mut self, message: &[u8]) -> io::Result<()> {
223        let text = std::str::from_utf8(message)
224            .map_err(|e| io::Error::new(io::ErrorKind::InvalidData, e))?;
225        self.ws.send(Message::text(text)).await.map_err(ws_to_io)
226    }
227
228    /// Close the WebSocket, bounded by [`CLOSE_TIMEOUT`]. Idempotent.
229    ///
230    /// # Errors
231    ///
232    /// A transport error other than the connection already being closed.
233    pub async fn shutdown(&mut self) -> io::Result<()> {
234        match tokio::time::timeout(CLOSE_TIMEOUT, self.ws.close(None)).await {
235            // A peer that never answers the close: the stream is dropped with
236            // the source, which is the end either way.
237            Err(_elapsed) => Ok(()),
238            Ok(result) => match result.map_err(ws_to_io) {
239                Err(e) if e.kind() == io::ErrorKind::NotConnected => Ok(()),
240                other => other,
241            },
242        }
243    }
244}
245
246/// A WebSocket error as the [`io::Error`] every `ByteSource` speaks.
247///
248/// A connection that is already closed maps to `NotConnected`, the kind the
249/// plain socket's idempotent `shutdown` already treats as success.
250fn ws_to_io(error: WsError) -> io::Error {
251    match error {
252        WsError::Io(e) => e,
253        // `SendAfterClosing` is a close sent after the peer's: the handshake
254        // is already over, which is closed by another name (exercised by
255        // `a_close_is_the_end_of_the_stream_and_shutdown_is_idempotent`).
256        WsError::ConnectionClosed
257        | WsError::AlreadyClosed
258        | WsError::Protocol(
259            tokio_tungstenite::tungstenite::error::ProtocolError::SendAfterClosing,
260        ) => io::Error::new(io::ErrorKind::NotConnected, "websocket closed"),
261        other => io::Error::other(other),
262    }
263}

319    /// A client over one end of an in-memory pipe, and the raw server end.
320    #[allow(clippy::result_large_err)] // the callback's `Err` type is tungstenite's, not ours
321    async fn pair() -> (ShimStream<DuplexStream>, WebSocketStream<DuplexStream>) {
322        let (client, server) = tokio::io::duplex(64 * 1024);
323        let request = shim_request("chimera.play.net", 10024).expect("valid host");
324        let accept = tokio_tungstenite::accept_hdr_async(
325            server,
326            |_req: &tokio_tungstenite::tungstenite::handshake::server::Request,
327             mut response: tokio_tungstenite::tungstenite::handshake::server::Response| {
328                // The shim echoes the subprotocol; without it the client
329                // refuses the upgrade, which is the check doing its job.
330                response.headers_mut().insert(
331                    "Sec-WebSocket-Protocol",
332                    HeaderValue::from_static(SUBPROTOCOL),
333                );
334                Ok(response)
335            },
336        );
337        let (client, server) = tokio::join!(ShimStream::upgrade(request, client), accept);
338        (
339            client.expect("client upgrade"),
340            server.expect("server accept"),
341        )
342    }

344    async fn read_all(client: &mut ShimStream<DuplexStream>, buf_len: usize) -> Vec<u8> {
345        let mut out = Vec::new();
346        let mut buf = vec![0; buf_len];
347        loop {
348            let n = client.read(&mut buf).await.expect("read");
349            if n == 0 {
350                return out;
351            }
352            out.extend_from_slice(&buf[..n]);
353        }
354    }

356    #[tokio::test]
357    async fn messages_become_one_byte_stream_whatever_their_boundaries() {
358        let (mut client, mut server) = pair().await;
359        // A line split across a text and a binary message, a ping between,
360        // then the close.
361        server
362            .send(Message::text("<prompt ti"))
363            .await
364            .expect("send");
365        server
366            .send(Message::Ping(Bytes::from_static(b"p")))
367            .await
368            .expect("send");
369        server
370            .send(Message::binary(b"me='1'>&gt;</prompt>\n".to_vec()))
371            .await
372            .expect("send");
373        server.close(None).await.expect("close");
374
375        // A buffer smaller than either message: the remainder is kept.
376        assert_eq!(
377            read_all(&mut client, 4).await,
378            b"<prompt time='1'>&gt;</prompt>\n"
379        );
380    }

1//! Reading the three answers out of play.net's HTML and URLs.
2//!
3//! Everything here is **pure**: a response's status, headers and body in, a
4//! verdict out. That is the same seam [`reject`](crate::eaccess) draws, and it
5//! is what lets the fragile half of this protocol be tested without a network
6//! -- which matters more here than anywhere else in the login, because no
7//! test may reach a live service and the markup can change under us at any
8//! time.
9//!
10//! # The character scrape is the most fragile thing in this module
11//!
12//! The reason is worth keeping in view: there is no enumeration endpoint.
13//! Unlike eaccess's tab-delimited `C` response, the
14//! character list is **server-rendered HTML**, so a markup change does not
15//! error -- it silently matches nothing, which looks exactly like "no such
16//! character". That is why the response status is checked **before** the body
17//! is ever scraped: a transient 503 would otherwise be reported as a missing
18//! character, and a missing character is fatal.
19

23/// Where a login POST's redirect landed.
24#[derive(Clone, Copy, Debug, PartialEq, Eq)]
25pub enum LoginRedirect {
26    /// Authenticated. The session cookie is set.
27    Authenticated,
28    /// The account or password was rejected.
29    Rejected,
30    /// Somewhere the confirmed protocol does not describe.
31    Unexpected,
32}
33
34/// The path an account that has never set a security question is sent to.
35///
36/// **This is a success, not a failure**, and it is the kind of thing only a
37/// live run finds. Confirmed with a second test account: the session is
38/// already fully authenticated at this point -- the account name renders in the
39/// page banner and the real session cookie is set -- and navigating straight to
40/// the play page works. Treating it as a failure would lock out every account
41/// that has never set a security question.
42const SECURITY_QA_PATH: &str = "/playdotnet/account/security_qa.asp";
43
44/// Classify the `Location` of a login POST's redirect.
45///
46/// The **target path is the signal**: the `Location` header's path is
47/// compared against the two page params sent, and the body is never parsed
48/// for control flow.
49///
50/// A query string is ignored, because the two confirmed failure shapes differ
51/// only there: a wrong password gives `?error=&returnto=/dr/` and a nonexistent
52/// account `?error=&returnto=/dr/play/home.asp`. Both mean the same thing.
53#[must_use]
54pub fn classify_login_redirect(location: Option<&str>, instance: &Instance) -> LoginRedirect {
55    let Some(location) = location else {
56        return LoginRedirect::Unexpected;
57    };
58    let path = location.split(['?', '#']).next().unwrap_or(location);
59
60    if path == instance.error_page() {
61        LoginRedirect::Rejected
62    } else if path == instance.okay_page() || path == SECURITY_QA_PATH {
63        LoginRedirect::Authenticated
64    } else {
65        LoginRedirect::Unexpected
66    }
67}

69/// Whether a redirect away from the character list means "no subscription".
70///
71/// Matched by **shape, not by an exact string**. The path is instance-specific
72/// (confirmed live) -- `DragonRealms` Prime's is plain
73/// `subscription_needed.asp`, but Platinum's is `subscription_to_plat_needed.asp`
74/// and Fallen's is `subscription_to_fall_needed.asp` -- so an exact match would
75/// let an unseen variant fall through to a generic error and lose the real
76/// cause.
77///
78/// No `GemStone` instance has been observed with a `_to_` variant. That is
79/// exactly why the loose match is kept rather than tightened: assuming `GemStone`
80/// will never have one is the guess this avoids making.
81#[must_use]
82pub fn is_subscription_needed(location: &str, instance: &Instance) -> bool {
83    let path = location.split(['?', '#']).next().unwrap_or(location);
84    let prefix = format!("/{}/play/subscription", instance.family);
85    let Some(tail) = path.strip_prefix(prefix.as_str()) else {
86        return false;
87    };
88
89    // A path separator in the variable middle would mean this is a DIFFERENT
90    // page under a similar prefix, not an instance-specific variant.
91    if tail.contains('/') {
92        return false;
93    }
94    if tail == "_needed.asp" {
95        return true;
96    }
97    tail.strip_prefix("_to_")
98        .and_then(|rest| rest.strip_suffix("_needed.asp"))
99        .is_some_and(|middle| !middle.is_empty())
100}

102/// Find a character's `charID` in a character-list page.
103///
104/// The markup, from a live capture:
105///
106/// ```html
107/// <input type=radio name="charID" id="W_TESTACCOUNT_000" value="W_TESTACCOUNT_000" checked  >
108/// <label for="W_TESTACCOUNT_000"><span class="normS1">Raiyen</span></label><br>
109/// ```
110///
111/// The id and the display name are in **two different elements**, tied by the
112/// `for` attribute. Matching is case-insensitive on a trimmed name.
113///
114/// Written as a scan rather than with a regex crate: this crate has no regex
115/// dependency, the simplest thing that works is preferred, and the shape being
116/// matched is fixed and small.
117#[must_use]
118pub fn find_char_code(body: &str, character: &str) -> Option<String> {
119    let wanted = character.trim();
120    character_entries(body)
121        .into_iter()
122        .find(|(_, name)| name.trim().eq_ignore_ascii_case(wanted))
123        .map(|(code, _)| code)
124}
125
126/// Every `(charID, display name)` pair the page declares.
127///
128/// Public so a diagnostic can report *which* characters were found when the one
129/// asked for was not. That is the difference between "that character is not on
130/// this instance" and "the markup changed and we matched nothing at all" --
131/// precisely the failure that is otherwise silent.
132#[must_use]
133pub fn character_entries(body: &str) -> Vec<(String, String)> {
134    let mut found = Vec::new();
135    for tail in body.split("id=\"W_").skip(1) {
136        let Some((code_rest, after)) = tail.split_once('"') else {
137            continue;
138        };
139        if !code_rest
140            .bytes()
141            .all(|b| b.is_ascii_alphanumeric() || b == b'_')
142        {
143            continue;
144        }
145        let code = format!("W_{code_rest}");
146
147        // The label for THIS code, not merely the next label on the page: an
148        // input with no matching label must not borrow the following
149        // character's name.
150        let marker = format!("<label for=\"{code}\">");
151        let Some(label) = after
152            .find(marker.as_str())
153            .map(|at| &after[at + marker.len()..])
154        else {
155            continue;
156        };
157        // Skip the opening `<span ...>` and take its text.
158        let Some((_, span)) = label.split_once('>') else {
159            continue;
160        };
161        if let Some((name, _)) = span.split_once('<') {
162            found.push((code, name.to_owned()));
163        }
164    }
165    found
166}

168/// The host, port and key from the final launch URL.
169///
170/// `Debug` is hand-written to redact the key; see the impl below.
171#[derive(Clone, PartialEq, Eq)]
172pub struct Launch {
173    /// The game host to connect to.
174    pub host: String,
175    /// The game port.
176    pub port: u16,
177    /// The one-shot launch key.
178    pub key: String,
179}
180
181impl std::fmt::Debug for Launch {
182    /// Redacts the key, as [`LaunchPayload`](crate::eaccess::LaunchPayload)
183    /// does: it is the live game credential.
184    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
185        f.debug_struct("Launch")
186            .field("host", &self.host)
187            .field("port", &self.port)
188            .field("key", &"<REDACTED>")
189            .finish()
190    }
191}

193/// Parse and **verify** the final launch URL.
194///
195/// # This is a security boundary, not a parse
196///
197/// By the time this runs the URL has come through a redirect chain. It decides
198/// where the client opens a socket and hands over a launch key, so every field
199/// is checked against the instance table rather than trusted:
200///
201/// - a **duplicate** `host`, `port` or `key` is refused outright. Last-value-
202///   wins would otherwise let a repeated parameter override the real one --
203///   the subtlest of these checks.
204/// - the host and port must **equal** the instance's pinned pair. Every
205///   `GemStone` row is confirmed live, so there is no looser suffix path here.
206/// - a blank value is a missing value.
207///
208/// # Errors
209///
210/// [`WebLoginFailure::UnexpectedResponse`] with a label naming which check
211/// failed. **The URL itself is never carried**: it holds the launch key.
212pub fn parse_launch(url: &str, instance: &Instance) -> Result<Launch, WebLoginFailure> {
213    let query = url
214        .split_once('?')
215        .map(|(_, query)| query)
216        .ok_or(WebLoginFailure::UnexpectedResponse("launch url: no query"))?;
217    let query = query.split('#').next().unwrap_or(query);
218
219    let mut host = None;
220    let mut port = None;
221    let mut key = None;
222    for pair in query.split('&').filter(|pair| !pair.is_empty()) {
223        let (name, value) = pair.split_once('=').unwrap_or((pair, ""));
224        let slot = match name {
225            "host" => &mut host,
226            "port" => &mut port,
227            "key" => &mut key,
228            _ => continue,
229        };
230        if slot.is_some() {
231            return Err(WebLoginFailure::UnexpectedResponse(
232                "launch url: duplicate host, port or key",
233            ));
234        }
235        *slot = Some(percent_decode(value));
236    }
237
238    let (Some(host), Some(port), Some(key)) = (host, port, key) else {
239        return Err(WebLoginFailure::UnexpectedResponse(
240            "launch url: missing host, port or key",
241        ));
242    };
243    if host.is_empty() || port.is_empty() || key.is_empty() {
244        return Err(WebLoginFailure::UnexpectedResponse(
245            "launch url: blank host, port or key",
246        ));
247    }
248
249    let port: u16 = port.parse().map_err(|_error| {
250        WebLoginFailure::UnexpectedResponse("launch url: port is not a number")
251    })?;
252
253    if host != instance.expected_host || port != instance.expected_port {
254        return Err(WebLoginFailure::UnexpectedResponse(
255            "launch url: host or port is not this instance's pinned pair",
256        ));
257    }
258
259    Ok(Launch { host, port, key })
260}

262/// Minimal `application/x-www-form-urlencoded` value decoding.
263///
264/// Only what a query value needs: `+` for a space, and `%XX`. A malformed
265/// escape is left verbatim rather than dropped, so a value that fails to decode
266/// still fails the pinned-host comparison rather than silently becoming
267/// something else.
268fn percent_decode(value: &str) -> String {
269    let bytes = value.as_bytes();
270    let mut out = Vec::with_capacity(bytes.len());
271    let mut index = 0;
272    while index < bytes.len() {
273        match bytes[index] {
274            b'+' => {
275                out.push(b' ');
276                index += 1;
277            }
278            b'%' if index + 2 < bytes.len() => {
279                // The two bytes after `%`, as bytes: sliced as a `str`, a
280                // multibyte character there would panic on its boundary; it is
281                // not hex either way.
282                let hex = std::str::from_utf8(&bytes[index + 1..index + 3]);
283                if let Some(byte) = hex.ok().and_then(|hex| u8::from_str_radix(hex, 16).ok()) {
284                    out.push(byte);
285                    index += 3;
286                } else {
287                    out.push(b'%');
288                    index += 1;
289                }
290            }
291            byte => {
292                out.push(byte);
293                index += 1;
294            }
295        }
296    }
297    String::from_utf8_lossy(&out).into_owned()
298}

300/// The one host every hop of the selection chain may point at.
301pub const TRUSTED_HOST: &str = "www.play.net";
302
303/// The path the chain's final, absolute redirect must name. Nothing else is
304/// accepted.
305const FINAL_PATH: &str = "/play/home.asp";
306
307/// What to do with one `Location` from the character-selection chain.
308#[derive(Clone, Debug, PartialEq, Eq)]
309pub enum Hop {
310    /// The chain has ended: hand this to [`parse_launch`]. Carries the launch
311    /// key, so it is never printed.
312    Launch(String),
313    /// Request this absolute URL next. Always `https://www.play.net/...`.
314    Follow(String),
315}
316
317/// Decide where one redirect in the selection chain goes -- **and refuse any
318/// that leaves play.net**.
319///
320/// # This is a security boundary
321///
322/// Following any `Location` that starts with `http` would fetch a redirect to
323/// `http://anything/` -- a downgrade, an off-site host, an intercepted hop --
324/// with the session's cookie jar, and the chain it led to could end on a URL
325/// whose `host=`/`key=` then reached [`parse_launch`]. `parse_launch` pins
326/// host and port against the instance table, which keeps the game socket
327/// safe; this keeps the REQUESTS safe. The rule:
328///
329/// - **An absolute `Location` ends the chain**, and must be exactly
330///   `https://www.play.net/play/home.asp` -- scheme `https`, default port,
331///   no userinfo, that path. Every absolute URL is checked, `http://`
332///   included, because of the downgrade: a plain-http target must be
333///   recognised and REJECTED, not requested.
334/// - **A relative one is followed**, resolved against `https://www.play.net/`,
335///   and the resolved URL is checked again. That catches the one relative
336///   form that is not relative: `//evil.example/x` is a network-path
337///   reference, and resolving it lands on another host.
338/// - A RELATIVE location already carrying `host=` and `key=` is taken as the
339///   launch URL. It is same-origin by construction and `parse_launch` still
340///   pins what it names.
341///
342/// # Errors
343///
344/// [`WebLoginFailure::UnexpectedResponse`] -- transient, as every "the server
345/// did something the confirmed shape does not cover" is. The URL is never
346/// carried: on the last hop it holds the launch key.
347pub fn next_hop(location: &str) -> Result<Hop, WebLoginFailure> {
348    const UNTRUSTED: WebLoginFailure =
349        WebLoginFailure::UnexpectedResponse("character selection: untrusted redirect");
350
351    let has_prefix = |prefix: &str| {
352        location
353            .get(..prefix.len())
354            .is_some_and(|head| head.eq_ignore_ascii_case(prefix))
355    };
356    if has_prefix("https://") || has_prefix("http://") {
357        let url = reqwest::Url::parse(location).map_err(|_| UNTRUSTED)?;
358        return if is_play_net(&url) && url.path() == FINAL_PATH {
359            Ok(Hop::Launch(location.to_owned()))
360        } else {
361            Err(UNTRUSTED)
362        };
363    }
364
365    if location.contains("host=") && location.contains("key=") {
366        return Ok(Hop::Launch(location.to_owned()));
367    }
368
369    let base = reqwest::Url::parse(&format!("https://{TRUSTED_HOST}/")).map_err(|_| UNTRUSTED)?;
370    let next = base.join(location).map_err(|_| UNTRUSTED)?;
371    if !is_play_net(&next) {
372        return Err(UNTRUSTED);
373    }
374    Ok(Hop::Follow(next.into()))
375}

377/// Scheme `https`, host `www.play.net`, port 443, no userinfo -- the origin
378/// half of the final-URL check.
379fn is_play_net(url: &reqwest::Url) -> bool {
380    url.scheme() == "https"
381        && url.host_str() == Some(TRUSTED_HOST)
382        && url.port_or_known_default() == Some(443)
383        && url.username().is_empty()
384        && url.password().is_none()

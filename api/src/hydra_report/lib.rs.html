1//! A report as blocks. A report crate says what it shows: a heading, a
2//! card of facts, a table with typed columns, a chart of bars or lines, a
3//! note, or plain lines. A front end draws each block its own way: the
4//! Reports window as real tables and painted charts, a terminal or an
5//! export as the text [`Report::text`] renders. A link is a command, as a
6//! cell or inside a line, and a front end runs it when clicked.
7//!
8//! Text lines carry a link as `\u{1}text\u{2}command\u{3}`, the markup the
9//! combat reports write; [`Report::text`] keeps it, so whatever reads the
10//! text rendering finds the links where they were.
11
12use std::fmt::Write as _;
13
14/// A report: its blocks, in order.
15#[derive(Debug, Clone, PartialEq, Default)]
16pub struct Report {
17    /// The blocks, top to bottom.
18    pub blocks: Vec<Block>,
19}
20
21/// One block of a report.
22#[derive(Debug, Clone, PartialEq)]
23pub enum Block {
24    /// A heading over what follows.
25    Heading(String),
26    /// Facts, a name and its value each, drawn as a card.
27    Facts(Vec<(String, Cell)>),
28    /// A table.
29    Table(Table),
30    /// A chart.
31    Chart(Chart),
32    /// A note under a block, in smaller or weaker text.
33    Note(String),
34    /// Lines as a text report says them, links in their markup.
35    Text(Vec<String>),
36}

38/// A cell of a table or a fact's value.
39#[derive(Debug, Clone, PartialEq)]
40pub enum Cell {
41    /// Nothing.
42    Empty,
43    /// Words.
44    Text(String),
45    /// A whole number, grouped by thousands when drawn.
46    Number(i64),
47    /// A number with a point.
48    Float(f64, u8),
49    /// A length of time in seconds: `2h 05m`, `41m`, `30s`.
50    Duration(i64),
51    /// A share, 0 to 100.
52    Percent(f64),
53    /// A link: its text and the command it runs.
54    Link(String, String),
55    /// Words with links among them, each in its markup, and any leading
56    /// spaces kept as an indent: drawn with the links live.
57    Linked(String),
58    /// A moment, as the report formats it, kept as words with its seconds
59    /// for sorting.
60    Time(String, i64),
61    /// A bar inside the cell, `value` against `max`, with its words.
62    Bar(f64, f64, String),
63}

160impl Report {
161    /// A report of lines, as a text report says them.
162    #[must_use]
163    pub fn from_lines(lines: Vec<String>) -> Self {
164        Self {
165            blocks: vec![Block::Text(lines)],
166        }
167    }
168
169    /// Add a block.
170    pub fn push(&mut self, block: Block) -> &mut Self {
171        self.blocks.push(block);
172        self
173    }
174
175    /// A heading.
176    pub fn heading(&mut self, text: impl Into<String>) -> &mut Self {
177        self.push(Block::Heading(text.into()))
178    }
179
180    /// A note.
181    pub fn note(&mut self, text: impl Into<String>) -> &mut Self {
182        self.push(Block::Note(text.into()))
183    }
184
185    /// The report as lines of text, links in their markup.
186    #[must_use]
187    pub fn text(&self) -> Vec<String> {
188        let mut lines = Vec::new();
189        for (i, block) in self.blocks.iter().enumerate() {
190            // A blank line sets a heading, a table or a chart apart from
191            // what came before, unless that was its heading.
192            let after_heading = i > 0 && matches!(self.blocks[i - 1], Block::Heading(_));
193            if i > 0
194                && !after_heading
195                && matches!(block, Block::Heading(_) | Block::Table(_) | Block::Chart(_))
196            {
197                lines.push(String::new());
198            }
199            match block {
200                Block::Heading(text) => lines.push(format!("{text}:")),
201                Block::Facts(facts) => {
202                    let wide = facts
203                        .iter()
204                        .map(|(n, _)| n.chars().count())
205                        .max()
206                        .unwrap_or(0);
207                    for (name, value) in facts {
208                        lines.push(format!("  {name:<wide$}  {}", value.words()));
209                    }
210                }
211                Block::Table(table) => lines.extend(table.text()),
212                Block::Chart(chart) => lines.extend(chart.text()),
213                Block::Note(text) => lines.push(text.clone()),
214                Block::Text(text) => lines.extend(text.iter().cloned()),
215            }
216        }
217        lines
218    }
219}

221/// Whether `text` holds a link's markup.
222#[must_use]
223pub fn has_link(text: &str) -> bool {
224    text.contains(['\u{1}', '\u{2}', '\u{3}'])
225}
226
227/// `text` as it is seen: each link's words, its command and its markers
228/// left out.
229#[must_use]
230pub fn seen(text: &str) -> String {
231    let mut out = String::new();
232    let mut hidden = false;
233    for c in text.chars() {
234        match c {
235            '\u{1}' | '\u{3}' => hidden = false,
236            '\u{2}' => hidden = true,
237            _ if !hidden => out.push(c),
238            _ => {}
239        }
240    }
241    out
242}

244/// `1,234,567`.
245#[must_use]
246pub fn grouped(n: i64) -> String {
247    let digits = n.abs().to_string();
248    let mut out = String::new();
249    for (i, c) in digits.chars().enumerate() {
250        if i > 0 && (digits.len() - i).is_multiple_of(3) {
251            out.push(',');
252        }
253        out.push(c);
254    }
255    if n < 0 { format!("-{out}") } else { out }
256}

258/// `2h 05m`, or `41m`, or `30s`.
259#[must_use]
260pub fn duration(seconds: i64) -> String {
261    if seconds >= 3600 {
262        format!("{}h {:02}m", seconds / 3600, seconds % 3600 / 60)
263    } else if seconds >= 60 {
264        format!("{}m", seconds / 60)
265    } else {
266        format!("{seconds}s")
267    }
268}

270impl Cell {
271    /// The cell as words, a link in its markup.
272    #[must_use]
273    pub fn words(&self) -> String {
274        match self {
275            Self::Empty => String::new(),
276            Self::Text(text) | Self::Linked(text) | Self::Time(text, _) | Self::Bar(_, _, text) => {
277                text.clone()
278            }
279            Self::Number(n) => grouped(*n),
280            Self::Float(x, places) => format!("{x:.*}", usize::from(*places)),
281            Self::Duration(secs) => duration(*secs),
282            Self::Percent(share) => format!("{share:.0}%"),
283            Self::Link(text, command) => format!("\u{1}{text}\u{2}{command}\u{3}"),
284        }
285    }
286
287    /// The cell as seen: a link's text alone.
288    #[must_use]
289    pub fn shown(&self) -> String {
290        match self {
291            Self::Link(text, _) => text.clone(),
292            Self::Linked(text) => seen(text),
293            other => other.words(),
294        }
295    }
296
297    /// What the cell sorts by: a number when it is one.
298    #[must_use]
299    pub fn key(&self) -> Key {
300        match self {
301            Self::Empty => Key::Nothing,
302            #[allow(clippy::cast_precision_loss)]
303            Self::Number(n) | Self::Duration(n) | Self::Time(_, n) => Key::Number(*n as f64),
304            Self::Float(x, _) | Self::Percent(x) | Self::Bar(x, _, _) => Key::Number(*x),
305            // A link whose words are a number sorts as one: `1,234`, `#7`.
306            Self::Link(text, _) => text
307                .trim()
308                .trim_start_matches('#')
309                .replace(',', "")
310                .parse::<f64>()
311                .map_or_else(|_| Key::Words(text.trim().to_lowercase()), Key::Number),
312            Self::Text(text) => Key::Words(text.trim().to_lowercase()),
313            Self::Linked(text) => Key::Words(seen(text).trim().to_lowercase()),
314        }
315    }
316}

318/// What a cell sorts by.
319#[derive(Debug, Clone, PartialEq)]
320pub enum Key {
321    /// An empty cell, last either way.
322    Nothing,
323    /// A number.
324    Number(f64),
325    /// Words, case aside.
326    Words(String),
327}
328
329impl Key {
330    /// Numbers before words, empties last.
331    #[must_use]
332    pub fn order(&self, other: &Self) -> std::cmp::Ordering {
333        use std::cmp::Ordering;
334        match (self, other) {
335            (Self::Nothing, Self::Nothing) => Ordering::Equal,
336            (Self::Nothing, _) | (Self::Words(_), Self::Number(_)) => Ordering::Greater,
337            (_, Self::Nothing) | (Self::Number(_), Self::Words(_)) => Ordering::Less,
338            (Self::Number(a), Self::Number(b)) => a.total_cmp(b),
339            (Self::Words(a), Self::Words(b)) => a.cmp(b),
340        }
341    }
342}

364impl Table {
365    /// A table with these columns and no rows yet.
366    #[must_use]
367    pub fn new(columns: Vec<Column>) -> Self {
368        Self {
369            columns,
370            ..Self::default()
371        }
372    }
373
374    /// The rows sorted by column `by`, descending when `desc`; a stable
375    /// sort, so ties keep the report's order.
376    #[must_use]
377    pub fn sorted(&self, by: usize, desc: bool) -> Vec<&Vec<Cell>> {
378        let mut rows: Vec<&Vec<Cell>> = self.rows.iter().collect();
379        rows.sort_by(|a, b| {
380            let (ka, kb) = (
381                a.get(by).map_or(Key::Nothing, Cell::key),
382                b.get(by).map_or(Key::Nothing, Cell::key),
383            );
384            if desc && ka != Key::Nothing && kb != Key::Nothing {
385                kb.order(&ka)
386            } else {
387                ka.order(&kb)
388            }
389        });
390        rows
391    }
392
393    /// The table as aligned lines: the header, the rows, a rule and the
394    /// footer. A link's markup is not counted in its width.
395    #[must_use]
396    pub fn text(&self) -> Vec<String> {
397        let mut widths: Vec<usize> = self
398            .columns
399            .iter()
400            .map(|c| c.name.chars().count())
401            .collect();
402        let all: Vec<&Vec<Cell>> = self.rows.iter().chain(self.footer.iter()).collect();
403        for row in &all {
404            for (i, cell) in row.iter().enumerate() {
405                if let Some(w) = widths.get_mut(i) {
406                    *w = (*w).max(cell.shown().chars().count());
407                }
408            }
409        }
410        let line = |cells: Vec<(String, usize)>| -> String {
411            let mut out = String::new();
412            for (i, (text, seen)) in cells.into_iter().enumerate() {
413                if i > 0 {
414                    out.push_str("  ");
415                }
416                let pad = widths.get(i).copied().unwrap_or(0).saturating_sub(seen);
417                match self.columns.get(i).map_or(Align::Left, |c| c.align) {
418                    Align::Left => {
419                        out.push_str(&text);
420                        out.push_str(&" ".repeat(pad));
421                    }
422                    Align::Right => {
423                        out.push_str(&" ".repeat(pad));
424                        out.push_str(&text);
425                    }
426                }
427            }
428            out.trim_end().to_owned()
429        };
430        let mut lines = vec![line(
431            self.columns
432                .iter()
433                .map(|c| (c.name.clone(), c.name.chars().count()))
434                .collect(),
435        )];
436        let cells = |row: &Vec<Cell>| -> Vec<(String, usize)> {
437            row.iter()
438                .map(|cell| (cell.words(), cell.shown().chars().count()))
439                .collect()
440        };
441        for row in &self.rows {
442            lines.push(line(cells(row)));
443        }
444        if let Some(footer) = &self.footer {
445            lines.push(
446                "-".repeat(widths.iter().sum::<usize>() + 2 * widths.len().saturating_sub(1)),
447            );
448            lines.push(line(cells(footer)));
449        }
450        lines
451    }
452}

454/// How wide a bar is drawn in text.
455const BAR: usize = 24;
456
457/// A text bar `value` long against `max`.
458fn text_bar(value: f64, max: f64) -> String {
459    #[allow(
460        clippy::cast_possible_truncation,
461        clippy::cast_sign_loss,
462        clippy::cast_precision_loss
463    )]
464    let cells = if max > 0.0 && value > 0.0 {
465        ((value * BAR as f64 / max).ceil() as usize).min(BAR)
466    } else {
467        0
468    };
469    format!("{:<BAR$}", "█".repeat(cells))
470}
471
472/// A value as words: whole when it is.
473fn value_words(x: f64) -> String {
474    if x.fract() == 0.0 && x.abs() < 1e15 {
475        #[allow(clippy::cast_possible_truncation)]
476        grouped(x as i64)
477    } else {
478        format!("{x:.1}")
479    }
480}

482impl Chart {
483    /// A chart of one series.
484    #[must_use]
485    pub fn one(
486        kind: ChartKind,
487        title: impl Into<String>,
488        unit: impl Into<String>,
489        categories: Vec<String>,
490        values: Vec<f64>,
491        hue: Hue,
492    ) -> Self {
493        Self {
494            kind,
495            title: title.into(),
496            unit: unit.into(),
497            categories,
498            series: vec![Series {
499                name: String::new(),
500                values,
501                hue,
502                hues: None,
503            }],
504            notes: Vec::new(),
505        }
506    }
507
508    /// The widest value across the series.
509    #[must_use]
510    pub fn max(&self) -> f64 {
511        self.series
512            .iter()
513            .flat_map(|s| s.values.iter().copied())
514            .fold(0.0, f64::max)
515    }
516
517    /// The chart as lines: a title, then a row a category with a bar per
518    /// series; a stacked chart as a row a series with its share.
519    #[must_use]
520    pub fn text(&self) -> Vec<String> {
521        let mut lines = vec![format!("{}:", self.title)];
522        if self.kind == ChartKind::Stacked {
523            let whole: f64 = self
524                .series
525                .iter()
526                .map(|s| s.values.iter().sum::<f64>())
527                .sum();
528            let wide = self
529                .series
530                .iter()
531                .map(|s| s.name.chars().count())
532                .max()
533                .unwrap_or(0);
534            let max = self
535                .series
536                .iter()
537                .map(|s| s.values.iter().sum::<f64>())
538                .fold(0.0, f64::max);
539            for s in &self.series {
540                let value: f64 = s.values.iter().sum();
541                lines.push(format!(
542                    "  {:<wide$}  {} {:>4.0}%  {}",
543                    s.name,
544                    text_bar(value, max),
545                    if whole > 0.0 {
546                        value * 100.0 / whole
547                    } else {
548                        0.0
549                    },
550                    value_words(value)
551                ));
552            }
553            return lines;
554        }
555        let max = self.max();
556        let wide = self
557            .categories
558            .iter()
559            .map(|c| c.chars().count())
560            .max()
561            .unwrap_or(0);
562        for (i, category) in self.categories.iter().enumerate() {
563            let mut line = format!("  {category:<wide$}");
564            for s in &self.series {
565                let value = s.values.get(i).copied().unwrap_or(0.0);
566                let _ = write!(line, "  {} {:>9}", text_bar(value, max), value_words(value));
567            }
568            if let Some(note) = self.notes.get(i).filter(|n| !n.is_empty()) {
569                let _ = write!(line, "  {note}");
570            }
571            lines.push(line.trim_end().to_owned());
572        }
573        if self.series.len() > 1 {
574            lines.push(format!(
575                "  (bars left to right: {})",
576                self.series
577                    .iter()
578                    .map(|s| s.name.as_str())
579                    .collect::<Vec<_>>()
580                    .join(", ")
581            ));
582        }
583        lines
584    }
585}

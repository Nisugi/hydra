1//! `.go2` across the sea: a walk to the ship, a voyage, and a walk on.
2//!
3//! When no way by land leads where `.go2` was asked to go, the character's
4//! own ship is the way: [`passage`] finds the port nearest by land, and the
5//! port from which the place can be walked to with the shortest voyage
6//! between the two. [`set_out`] then makes the whole journey in the walk's
7//! place, so one `.stop` stops whichever part of it is under way:
8//!
9//! 1. To the ship: to the port's dock handler, who is asked for it and
10//!    names its pier, whether the ship is put there now or is out already.
11//!    Without the handler's fee, to the bank for it and back. A pier is
12//!    several rooms long and the ship lies in one of them, so each is
13//!    walked to until the ship is seen.
14//! 2. The voyage (`hydra_travel::sail`), which boards the ship itself.
15//!    The ship is left at the far pier, not stored: it lies there a good
16//!    while, and may be wanted again.
17//! 3. Down the gangplank, and the walk on, found afresh from the far pier.
18//!
19//! The places of Sailor's Grief have open water between them, which the
20//! map does not cross: the way on, there, is a swim ([`swim`]) and then a
21//! walk, and a journey between two of them is that alone, with no ship.
22

45/// The way by sea, for a walk with none by land: `passage` when there is
46/// one and the character lets its ship be taken (`own_ship`); else `none`
47/// said, with the ship offered when it would have served.
48fn way(
49    passage: Option<Passage>,
50    (own_ship, sure): (bool, bool),
51    none: String,
52) -> Result<super::Way, super::Said> {
53    match passage {
54        // A room of the open sea, asked for from land: said back first, so
55        // a number mistyped does not put to sea.
56        Some(Passage {
57            sail: Some((from, to)),
58            what,
59            named,
60        }) if own_ship && !to.port && !sure => Err(super::Said::Lines(vec![format!(
61            "{named} is a room of the open sea, {} by ship from {}. To sail there and drop \
62             anchor: .go2 {what} --no-confirm",
63            rooms_word(to.uid, from),
64            from.name
65        )])),
66        Some(passage) if own_ship || passage.sail.is_none() => Ok(super::Way::Sea(passage)),
67        Some(Passage {
68            sail: Some((from, to)),
69            ..
70        }) => Err(super::Said::Lines(vec![format!(
71            "{none} Your own ship could take you, from {} to {}: turn on \"Use your own ship\" \
72             on the Travel page.",
73            from.name, to.name
74        )])),
75        Some(_) | None => Err(super::Said::Lines(vec![none])),
76    }
77}

79/// What the way by sea is found with, as the character is now: the ocean,
80/// whether its own ship may be taken, and its state -- SHIP INFO asked
81/// first, when `ask`, if the ship may be taken and nothing has said where
82/// it lies, since that decides the port sailed from. `.route2` never asks:
83/// it says what is known, and writes nothing to the game.
84async fn at_sea(
85    handle: &SessionHandle,
86    dir: &Path,
87    roster_name: &str,
88    ask: bool,
89) -> (Option<&'static Ocean>, bool, Option<GameState>) {
90    let ocean = hydra_map::ocean().as_ref().ok();
91    let own_ship = crate::pages::travel_settings(dir, roster_name).on(OWN_SHIP);
92    if ask
93        && own_ship
94        && handle
95            .snapshot()
96            .await
97            .is_some_and(|snapshot| snapshot.state.ship.owned.is_empty())
98        && handle.write_quietly("ship info", Origin::Travel).await
99    {
100        let _ = until(
101            handle,
102            &CancellationToken::new(),
103            Duration::from_secs(3),
104            |state| !state.ship.owned.is_empty(),
105        )
106        .await;
107    }
108    let state = handle.snapshot().await.map(|snapshot| snapshot.state);
109    (ocean, own_ship, state)
110}

112/// What finds the way by sea for a walk with none by land, as the
113/// character is now: asked with where it is, what it asked for, and what to
114/// say when the sea is no way either.
115pub(super) async fn asker(
116    handle: &SessionHandle,
117    dir: &Path,
118    roster_name: &str,
119) -> impl Fn(&Here, &str, String) -> Result<super::Way, super::Said> + Send + 'static {
120    let (ocean, own_ship, state) = at_sea(handle, dir, roster_name, true).await;
121    move |here: &Here, what: &str, none: String| {
122        let (what, sure) = match what.strip_suffix(" --no-confirm") {
123            Some(what) => (what, true),
124            None => (what, false),
125        };
126        let passage = ocean
127            .zip(state.as_ref())
128            .and_then(|(ocean, state)| passage(here, ocean, state, what));
129        way(passage, (own_ship, sure), none)
130    }
131}

144/// What tells `.route2` of the way by sea for a place with none by land,
145/// as the character is now: asked with where it is and what it asked for;
146/// `None` when the sea is no way there either.
147pub(super) async fn teller(
148    handle: &SessionHandle,
149    dir: &Path,
150    roster_name: &str,
151) -> impl Fn(&Here, &str) -> Option<BySea> + Send + 'static {
152    let (ocean, own_ship, state) = at_sea(handle, dir, roster_name, false).await;
153    move |here: &Here, what: &str| {
154        let (ocean, state) = ocean.zip(state.as_ref())?;
155        let passage = passage(here, ocean, state, what)?;
156        Some(told(here, ocean, &passage, own_ship))
157    }
158}

160/// `passage` as `.route2` says it, from `here`, `own_ship` whether the
161/// Travel page lets `.go2` take the character's ship.
162fn told(here: &Here, ocean: &'static Ocean, passage: &Passage, own_ship: bool) -> BySea {
163    let walked = |legs: &[route::Leg]| {
164        let seconds = legs.last().map_or(0.0, |leg| leg.so_far);
165        format!(
166            "{} step{}, about {seconds:.0} seconds",
167            legs.len(),
168            if legs.len() == 1 { "" } else { "s" }
169        )
170    };
171    let Some((from, to)) = &passage.sail else {
172        return BySea {
173            lines: vec![format!(
174                "No way by land to {}: .go2 swims there.",
175                passage.named
176            )],
177            walk: None,
178        };
179    };
180    if !own_ship {
181        return BySea {
182            lines: vec![format!(
183                "No way by land to {}. Your own ship could take you, from {} to {}: turn on \
184                 \"Use your own ship\" on the Travel page.",
185                passage.named, from.name, to.name
186            )],
187            walk: None,
188        };
189    }
190    let mut lines = vec![format!(
191        "No way by land to {}: .go2 takes your ship.",
192        passage.named
193    )];
194    let walk = from
195        .handler
196        .and_then(|uid| room_of(&here.map, uid))
197        .and_then(|handler| {
198            let legs = route::itinerary(&here.map, &here.walker, here.room, handler)?;
199            Some((handler, legs))
200        });
201    lines.push(match &walk {
202        Some((_, legs)) => format!("1. To {}'s dock handler: {}.", from.name, walked(legs)),
203        None => format!(
204            "1. To {}'s dock handler: no way there is known from here.",
205            from.name
206        ),
207    });
208    lines.push(match ocean.course(from.uid, to.uid) {
209        Some(course) => format!(
210            "2. Sail from {} to {}: {} rooms of sea.",
211            from.name,
212            to.name,
213            course.rooms.len()
214        ),
215        None => format!("2. Sail from {} to {}.", from.name, to.name),
216    });
217    if to.port {
218        let on = ocean
219            .ports()
220            .iter()
221            .find(|port| port.uid == to.uid)
222            .and_then(|port| a_pier(&here.map, ocean, port))
223            .and_then(|pier| {
224                let Meant::Room(goal) = route::destination(
225                    &here.map,
226                    &here.walker,
227                    pier,
228                    &passage.what,
229                    &here.notes.targets,
230                ) else {
231                    return None;
232                };
233                route::itinerary(&here.map, &here.walker, pier, goal)
234            });
235        lines.push(match on {
236            Some(legs) => format!(
237                "3. From {}'s pier to {}: {}.",
238                to.name,
239                passage.named,
240                walked(&legs)
241            ),
242            None => format!("3. From {}'s pier to {}.", to.name, passage.named),
243        });
244    } else {
245        lines.push("3. The anchor dropped there.".to_owned());
246    }
247    BySea { lines, walk }
248}

250/// How far `room` is from `port` by ship, in words.
251fn rooms_word(room: i64, port: &Port) -> String {
252    let ocean = hydra_map::ocean().as_ref().ok();
253    match ocean.and_then(|ocean| ocean.course(port.uid, room)) {
254        Some(course) => format!("{} rooms", course.rooms.len()),
255        None => "no known course".to_owned(),
256    }

262    /// The port the ship sails from -- the nearest by land, or the one the

293/// A pier of `port`, as a room of the map.
294fn a_pier(map: &Map, ocean: &Ocean, port: &Port) -> Option<RoomId> {
295    let mut piers: Vec<i64> = ocean.piers_of(port).collect();
296    piers.sort_unstable();
297    piers.into_iter().find_map(|pier| room_of(map, pier))
298}

311/// The way to `what` by sea, for a character with no way there by land:
312/// `None` when no port can be walked to from here, or the place can be
313/// walked to from no other port.
314fn passage(here: &Here, ocean: &'static Ocean, state: &GameState, what: &str) -> Option<Passage> {
315    let steps = |from: RoomId, to: RoomId| {
316        route::itinerary(&here.map, &here.walker, from, to).map(|legs| legs.len())
317    };
318    let named = |goal: RoomId| {
319        let room = here.map.room(goal);
320        room.and_then(|room| room.title.first().cloned())
321            .unwrap_or_default()
322    };
323    // A swim alone, between two places of Sailor's Grief.
324    if aboard(&state.room).is_none()
325        && let Some(crossing) = swim::crossing(here, here.room, what)
326    {
327        return Some(Passage {
328            sail: None,
329            what: what.to_owned(),
330            named: named(crossing.goal),
331        });
332    }
333    // From the ship stood on, the port it lies in; else the port SHIP INFO
334    // says the ship is docked at; else the nearest by land.
335    let from = if aboard(&state.room).is_some() {
336        match hydra_map::Navigator::default().fix(ocean, state) {
337            hydra_map::Sailing::Docked { port: Some(port) }
338            | hydra_map::Sailing::At { room: port, .. } => ocean.port_at(port),
339            _ => ocean.port_of_pier_titled(&state.ship.berth.as_ref()?.pier),
340        }?
341    } else if let Some(port) = docked_at(ocean, state) {
342        port
343    } else {
344        ocean
345            .ports()
346            .iter()
347            .filter_map(|port| {
348                let pier = a_pier(&here.map, ocean, port)?;
349                Some((steps(here.room, pier)?, port))
350            })
351            .min_by_key(|(far, _)| *far)?
352            .1
353    };
354    // A room of the open sea: sailed to from the port, and the anchor dropped.
355    if let Some(bound) = ocean.bound(what).filter(|bound| !bound.port) {
356        let sea = ocean.room(bound.uid).map_or("", |room| room.sea.as_str());
357        let named = format!("{} ({sea})", bound.name);
358        return Some(Passage {
359            sail: Some((from, bound)),
360            what: what.to_owned(),
361            named,
362        });
363    }
364    let (to, goal) = ocean
365        .ports()
366        .iter()
367        .filter(|port| port.uid != from.uid)
368        .filter_map(|port| {
369            let pier = a_pier(&here.map, ocean, port)?;
370            let Meant::Room(goal) =
371                route::destination(&here.map, &here.walker, pier, what, &here.notes.targets)
372            else {
373                return None;
374            };
375            // Walked to from the pier, or swum to from it.
376            let goal = match steps(pier, goal) {
377                Some(_) => goal,
378                None => swim::crossing(here, pier, what)?.goal,
379            };
380            let voyage = ocean.course(from.uid, port.uid)?.rooms.len();
381            Some((voyage, port, goal))
382        })
383        .min_by_key(|(voyage, ..)| *voyage)
384        .map(|(_, port, goal)| (port, goal))?;
385    Some(Passage {
386        sail: Some((
387            from,
388            Bound {
389                uid: to.uid,
390                name: to.name.clone(),
391                port: true,
392            },
393        )),
394        what: what.to_owned(),
395        named: named(goal),
396    })
397}

399/// Make the journey `passage` describes, in a task of its own, in the place
400/// `claim` took: what is said of it as it sets out.
401pub(super) fn set_out(
402    handle: SessionHandle,
403    roster_name: String,
404    passage: Passage,
405    mut claim: Claim,
406) -> Result<super::Said, String> {
407    let ocean = hydra_map::ocean().as_ref().map_err(Clone::clone)?;
408    if claim.stopped() {
409        return Ok(super::Said::Lines(vec![
410            "Stopped before setting out.".to_owned(),
411        ]));
412    }
413    let said = match &passage.sail {
414        Some((_, to)) if !to.port => format!("To {}, by sea, to drop anchor there.", passage.named),
415        Some((from, to)) => format!(
416            "To {} by sea: from {} to {}, and on from there.",
417            passage.named, from.name, to.name
418        ),
419        None => format!("To {}, by a swim.", passage.named),
420    };
421    claim.set_out();
422    let (key, number, stop) = (claim.key.clone(), claim.number, claim.stop.clone());
423    crate::commands::spawn(async move {
424        let dir = crate::account::data_dir();
425        let halted = Box::pin(journey(&handle, &dir, &roster_name, ocean, &passage, &stop)).await;
426        if let Ok(mut walks) = walks().lock()
427            && let Some(walking) = walks.get_mut(&key)
428            && walking
429                .under_way
430                .as_ref()
431                .is_some_and(|(at, _)| *at == number)
432        {
433            walking.under_way = None;
434        }
435        hydra_map::show_course(&roster_name, None);
436        if let Err(Halt::Because(why)) = halted {
437            handle.say(Notice::line(NoticeKind::Error, format!("Travel: {why}")).answering());
438        }
439    });
440    Ok(super::Said::Lines(vec![said]))
441}

443async fn journey(
444    handle: &SessionHandle,
445    dir: &Path,
446    roster_name: &str,
447    ocean: &'static Ocean,
448    passage: &Passage,
449    stop: &CancellationToken,
450) -> Result<(), Halt> {
451    if let Some((from, to)) = &passage.sail {
452        Box::pin(fund(handle, dir, roster_name, ocean, passage, stop)).await?;
453        Box::pin(voyage(handle, dir, roster_name, ocean, (from, to), stop)).await?;
454        // At a room of the sea the journey ends, at anchor.
455        if !to.port {
456            return Ok(());
457        }
458    }
459    Box::pin(onward(handle, dir, roster_name, stop, &passage.what)).await
460}

462/// The way on to `what` from where the character stands: a walk; or, with
463/// open water between, a swim and then a walk.
464async fn onward(
465    handle: &SessionHandle,
466    dir: &Path,
467    roster_name: &str,
468    stop: &CancellationToken,
469    what: &str,
470) -> Result<(), Halt> {
471    let none = || Halt::Because(format!("I know no way on to {what} from here."));
472    let mut from = here(handle, dir, roster_name).await?;
473    let goal = |from: &Here| {
474        let meant = route::destination(
475            &from.map,
476            &from.walker,
477            from.room,
478            what,
479            &from.notes.targets,
480        );
481        match meant {
482            Meant::Room(goal) => Some(goal),
483            _ => None,
484        }
485    };
486    let walked_to = |from: &Here, goal: RoomId| {
487        goal == from.room || route::itinerary(&from.map, &from.walker, from.room, goal).is_some()
488    };
489    if !goal(&from).is_some_and(|goal| walked_to(&from, goal)) {
490        let crossing = swim::crossing(&from, from.room, what).ok_or_else(none)?;
491        Box::pin(swim::cross(handle, dir, roster_name, stop, &crossing)).await?;
492        from = here(handle, dir, roster_name).await?;
493    }
494    let goal = goal(&from).ok_or_else(none)?;
495    Box::pin(walk(handle, dir, roster_name, stop, from, goal)).await
496}

498/// What a dock handler asks to launch a stored ship, when none has said.
499const LAUNCH_FEE: u64 = 5_000;
500
501/// How long an answer to `bank account` is waited for.
502const ANSWER: Duration = Duration::from_secs(3);
503
504/// What a journey by sea charges, and in words.
505struct Charges {
506    /// The whole of it.
507    need: u64,
508    /// Each part, for the player: `5000 to launch the sloop, 0 in tolls ...`.
509    said: String,
510}
511
512/// The silver `path` charges, where `path` begins at `from`.
513fn tolls(here: &Here, from: RoomId, to: RoomId) -> Option<u64> {
514    let routes = here
515        .map
516        .routes(from, Target::Room(to), route::pricing(&here.walker));
517    routes.path_to(to).map(|mut path| {
518        path.insert(0, from);
519        silver_for(&here.map, &path)
520    })
521}

523/// What the journey will charge: the handler's fee when the ship is stored,
524/// the tolls on the walk to the handler, and the tolls on from the far
525/// pier. Pure: nothing is sent.
526fn charges(
527    here: &Here,
528    state: &GameState,
529    ocean: &'static Ocean,
530    passage: &Passage,
531    (from, to): (&'static Port, &Bound),
532) -> Charges {
533    let hull = state
534        .ship
535        .owned
536        .keys()
537        .next()
538        .map_or_else(|| "ship".to_owned(), |hull| hull.to_lowercase());
539    let fee = if state.ship.berth.is_some() {
540        0
541    } else {
542        state.ship.launch.fee.map_or(LAUNCH_FEE, u64::from)
543    };
544    let there = from
545        .handler
546        .and_then(|uid| room_of(&here.map, uid))
547        .and_then(|handler| tolls(here, here.room, handler))
548        .unwrap_or(0);
549    let on = ocean
550        .port_at(to.uid)
551        .and_then(|port| a_pier(&here.map, ocean, port))
552        .and_then(|pier| {
553            let Meant::Room(goal) = route::destination(
554                &here.map,
555                &here.walker,
556                pier,
557                &passage.what,
558                &here.notes.targets,
559            ) else {
560                return None;
561            };
562            tolls(here, pier, goal)
563        })
564        .unwrap_or(0);
565    Charges {
566        need: fee + there + on,
567        said: format!(
568            "{fee} to launch the {hull}, {there} in tolls to the handler, {on} on from {}",
569            to.name
570        ),
571    }
572}

574/// The nearest bank that holds `short` and can be reached on the `have` in
575/// hand, by the listing the model holds; `Err` says what each bank holds
576/// when none will do. Mist Harbor's counts, by the trinket, when the map
577/// lets this character through.
578fn bank_holding(
579    here: &Here,
580    state: &GameState,
581    (have, short): (u64, u64),
582) -> Result<RoomId, String> {
583    let mut banks: Vec<(f64, RoomId, Option<u64>)> = here
584        .map
585        .rooms()
586        .iter()
587        .filter(|room| room.tags.iter().any(|tag| tag == "bank"))
588        .filter_map(|room| {
589            let routes = here.map.routes(
590                here.room,
591                Target::Room(room.id),
592                route::pricing(&here.walker),
593            );
594            let seconds = routes.seconds_to(room.id)?;
595            let mut path = routes.path_to(room.id)?;
596            path.insert(0, here.room);
597            (silver_for(&here.map, &path) <= have).then_some(())?;
598            let held = room
599                .location
600                .as_deref()
601                .and_then(|location| state.bank.local(location))
602                .map(|balance| balance.silver);
603            Some((seconds, room.id, held))
604        })
605        .collect();
606    banks.sort_by(|(a, ..), (b, ..)| a.total_cmp(b));
607    banks
608        .iter()
609        .find(|(_, _, held)| held.is_some_and(|held| held >= short))
610        .map(|(_, bank, _)| *bank)
611        .ok_or_else(|| {
612            let held: Vec<String> = state
613                .bank
614                .balances()
615                .iter()
616                .map(|balance| format!("{} {}", balance.bank, balance.silver))
617                .collect();
618            if held.is_empty() {
619                "the bank listing names no account".to_owned()
620            } else {
621                held.join(", ")
622            }
623        })
624}

626/// What the journey charges, before a step is taken ([`charges`]). Short
627/// of it, the bank that holds the difference ([`bank_holding`]) is walked
628/// to first and the difference withdrawn. No such bank, the journey halts
629/// here, saying what it needs and what each bank holds.
630async fn fund(
631    handle: &SessionHandle,
632    dir: &Path,
633    roster_name: &str,
634    ocean: &'static Ocean,
635    passage: &Passage,
636    stop: &CancellationToken,
637) -> Result<(), Halt> {
638    let Some((from, to)) = &passage.sail else {
639        return Ok(());
640    };
641    let state = state_of(handle).await?;
642    if aboard(&state.room).is_some() {
643        return Ok(());
644    }
645    let here = here(handle, dir, roster_name).await?;
646    let Charges { need, said } = charges(&here, &state, ocean, passage, (from, to));
647    if need == 0 {
648        return Ok(());
649    }
650    let say = |said: String| {
651        handle.say(Notice::line(NoticeKind::Info, format!("Travel: {said}")).answering());
652    };
653    let Some(have) = wealth(handle, stop).await? else {
654        say(format!(
655            "this journey asks for {need} silver ({said}), and I could not tell how much you \
656             carry, so I am going anyway."
657        ));
658        return Ok(());
659    };
660    let short = need.saturating_sub(have);
661    if short == 0 {
662        return Ok(());
663    }
664    let settings = crate::pages::travel_settings(dir, roster_name);
665    if !settings.on(hydra_travel::settings::GET_SILVERS) {
666        say(format!(
667            "this journey asks for {need} silver ({said}) and you carry {have}. Fetch silver \
668             from the bank is off, so I am going anyway."
669        ));
670        return Ok(());
671    }
672    if !state.bank.is_known() {
673        send(handle, stop, "bank account").await?;
674        until(handle, stop, ANSWER, |state| state.bank.is_known()).await?;
675    }
676    let state = state_of(handle).await?;
677    let bank = bank_holding(&here, &state, (have, short)).map_err(|held| {
678        Halt::Because(format!(
679            "this journey asks for {need} silver ({said}) and you carry {have}. No bank I can \
680             reach from here holds the {short} more: {held}."
681        ))
682    })?;
683    let bank_title = here
684        .map
685        .room(bank)
686        .and_then(|room| room.title.first().cloned())
687        .unwrap_or_else(|| "the bank".to_owned());
688    say(format!(
689        "to {bank_title} first, for {short} silver: this journey asks for {need} ({said}) and \
690         you carry {have}."
691    ));
692    Box::pin(walk(handle, dir, roster_name, stop, here, bank)).await?;
693    let room_title = state_of(handle).await?.room.title;
694    send(
695        handle,
696        stop,
697        &withdraw_command(room_title.as_deref(), short),
698    )
699    .await?;
700    until(handle, stop, Duration::from_secs(2), |_| false).await?;
701    if wealth(handle, stop).await?.is_some_and(|have| have < need) {
702        return Err(Halt::Because(format!(
703            "{bank_title} did not give me the {short} silver: the teller says the account has \
704             not got that much."
705        )));
706    }
707    Ok(())
708}

717/// To the ship at `from`, the voyage to `to`, and down its gangplank.
718async fn voyage(
719    handle: &SessionHandle,
720    dir: &Path,
721    roster_name: &str,
722    ocean: &'static Ocean,
723    (from, to): (&'static Port, &Bound),
724    stop: &CancellationToken,
725) -> Result<(), Halt> {
726    let state = state_of(handle).await?;
727    if aboard(&state.room).is_none() {
728        Box::pin(to_the_ship(handle, dir, roster_name, ocean, from, stop)).await?;
729    }
730    let teller = handle.clone();
731    let sailed = hydra_travel::sail::sail(
732        handle,
733        stop,
734        ocean,
735        to.uid,
736        Helm::Hydra {
737            wind: crate::pages::travel_settings(dir, roster_name)
738                .on(hydra_travel::settings::CALL_WIND),
739        },
740        |rooms| {
741            hydra_map::show_course(
742                roster_name,
743                rooms.map(|rooms| hydra_map::ShownCourse {
744                    rooms: rooms.to_vec(),
745                    port: to.name.clone(),
746                }),
747            );
748        },
749        |said| {
750            teller.say(Notice::line(NoticeKind::Info, format!("Sailing: {said}")).answering());
751        },
752    )
753    .await;
754    match sailed {
755        Sailed::Arrived if !to.port => {
756            handle.say(
757                Notice::line(
758                    NoticeKind::Info,
759                    format!(
760                        "Travel: at {}, and the anchor is down. The wheel is yours.",
761                        to.name
762                    ),
763                )
764                .answering(),
765            );
766            return Ok(());
767        }
768        Sailed::Arrived => {}
769        Sailed::Stopped => return Err(Halt::Because("stopped. The wheel is yours.".to_owned())),
770        Sailed::Lost => {
771            return Err(Halt::Because(
772                "stopped, because the connection went away.".to_owned(),
773            ));
774        }
775        Sailed::Halted(why) => return Err(Halt::Because(format!("{why} The wheel is yours."))),
776    }
777    // Down the gangplank, and on by land from wherever that is.
778    send(handle, stop, "go gangplank").await?;
779    let ashore = until(handle, stop, Duration::from_secs(8), |state| {
780        state.room.title.is_some() && aboard(&state.room).is_none()
781    })
782    .await?;
783    if !ashore {
784        return Err(Halt::Because(format!(
785            "in port at {}, and I could not get ashore.",
786            to.name
787        )));
788    }
789    Ok(())
790}

792/// What a dock handler answered, asked for the ship.
793enum Asked {
794    /// The pier it is at: put there now, or out already.
795    Pier(String),
796    /// The character has not the fee with it.
797    Refused,
798    /// Nothing Hydra knows.
799    Nothing,
800}

802/// The nouns a port's dock NPC goes by: Solhaven's `dock handler`,
803/// Icemule's `gaunt dock master`.
804const DOCK_NOUNS: [&str; 3] = ["handler", "master", "warden"];
805
806/// The dock NPC in the room, by exist id, so the ask reaches it whatever
807/// the port calls it: a dock handler, a dock master, or Sailor's Grief's
808/// warden, who has no dock in his name; `None` when none is in sight.
809fn dock_npc(state: &GameState) -> Option<String> {
810    // Among the creatures, where the room puts a bold NPC, and the objects,
811    // where it puts a plain one: ports differ.
812    state
813        .room
814        .creatures
815        .iter()
816        .chain(state.room.objects.iter())
817        .find(|thing| {
818            DOCK_NOUNS.contains(&thing.noun.as_str())
819                && (thing.text.contains("dock") || thing.noun == "warden")
820        })
821        .map(|thing| format!("#{}", thing.id))
822}
823
824/// Ask the dock NPC here for the ship, twice: the first is answered with
825/// the fee, the second pays it. A ship out already is named at the first.
826/// The NPC is asked by exist id, once the room has shown it.
827async fn ask(handle: &SessionHandle, stop: &CancellationToken, hull: &str) -> Result<Asked, Halt> {
828    until(handle, stop, Duration::from_secs(4), |state| {
829        dock_npc(state).is_some()
830    })
831    .await?;
832    let state = state_of(handle).await?;
833    let Some(npc) = dock_npc(&state) else {
834        return Err(Halt::Because(
835            "I see no dock handler here to ask for your ship.".to_owned(),
836        ));
837    };
838    let ask = format!("ask {npc} about retrieve {hull}");
839    let before = state.ship.launch;
840    send(handle, stop, &ask).await?;
841    // The second goes the moment the first is answered.
842    until(handle, stop, Duration::from_secs(4), |state| {
843        state.ship.launch != before
844    })
845    .await?;
846    if state_of(handle).await?.ship.launch.piers == before.piers {
847        send(handle, stop, &ask).await?;
848        until(handle, stop, Duration::from_secs(4), |state| {
849            let launch = state.ship.launch;
850            launch.piers != before.piers || launch.refusals != before.refusals
851        })
852        .await?;
853    }
854    let state = state_of(handle).await?;
855    Ok(if state.ship.launch.piers != before.piers {
856        state
857            .ship
858            .berth
859            .map_or(Asked::Nothing, |berth| Asked::Pier(berth.pier))
860    } else if state.ship.launch.refusals != before.refusals {
861        Asked::Refused
862    } else {
863        Asked::Nothing
864    })
865}

867/// How long the ship put at a pier is waited for, and how often the pier
868/// is walked again meanwhile: the handler has it there "momentarily".
869const ARRIVES_WITHIN: Duration = Duration::from_secs(45);
870const PIER_AGAIN: Duration = Duration::from_secs(5);
871
872/// Whether the room `state` shows holds the character's ship `info`, and
873/// not another's of the same hull.
874fn own_ship_here(state: &GameState, info: &hydra_model::state::ship::reports::ShipInfo) -> bool {
875    state
876        .room
877        .objects
878        .iter()
879        .any(|thing| SHIPS.contains(&thing.noun.as_str()) && info.is_mine(&thing.text))
880}

882/// To the ship: to the port's dock handler, who is asked for it and names
883/// its pier, whether it puts the ship there or it is out already; to the
884/// bank and back for the fee when the character has not got it; and along
885/// the pier until the character's own ship is in sight -- known by its
886/// hull and its look, since another's of the same hull may lie there too
887/// -- walked again while the ship put there is still on its way.
888async fn to_the_ship(
889    handle: &SessionHandle,
890    dir: &Path,
891    roster_name: &str,
892    ocean: &'static Ocean,
893    port: &'static Port,
894    stop: &CancellationToken,
895) -> Result<(), Halt> {
896    let hull = hull_owned(handle, stop).await?;
897    let from = here(handle, dir, roster_name).await?;
898    let Some(handler) = port.handler.and_then(|uid| room_of(&from.map, uid)) else {
899        return Err(Halt::Because(format!(
900            "I do not know where {}'s dock handler is.",
901            port.name
902        )));
903    };
904    let say = |said: String| {
905        handle.say(Notice::line(NoticeKind::Info, format!("Travel: {said}")).answering());
906    };
907    say(format!("to {}'s dock handler, for the {hull}.", port.name));
908    Box::pin(walk(handle, dir, roster_name, stop, from, handler)).await?;
909    let mut asked = ask(handle, stop, &hull).await?;
910    if matches!(asked, Asked::Refused) {
911        let Some(fee) = state_of(handle).await?.ship.launch.fee else {
912            return Err(Halt::Because(format!(
913                "the dock handler wants a fee to launch your {hull}, and you have not got it."
914            )));
915        };
916        say(format!(
917            "to the bank, for the {fee} silvers the handler wants."
918        ));
919        let from = here(handle, dir, roster_name).await?;
920        let Meant::Room(bank) = route::destination(
921            &from.map,
922            &from.walker,
923            from.room,
924            "bank",
925            &from.notes.targets,
926        ) else {
927            return Err(Halt::Because(format!(
928                "the dock handler wants {fee} silvers to launch your {hull}, and I know no bank \
929                 to walk to for it."
930            )));
931        };
932        Box::pin(walk(handle, dir, roster_name, stop, from, bank)).await?;
933        send(handle, stop, &format!("withdraw {fee} silver")).await?;
934        until(handle, stop, Duration::from_secs(2), |_| false).await?;
935        let from = here(handle, dir, roster_name).await?;
936        Box::pin(walk(handle, dir, roster_name, stop, from, handler)).await?;
937        asked = ask(handle, stop, &hull).await?;
938    }
939    let pier = match asked {
940        Asked::Pier(pier) => pier,
941        Asked::Refused => {
942            return Err(Halt::Because(format!(
943                "the dock handler still will not launch your {hull}: the bank did not give me \
944                 the fee."
945            )));
946        }
947        Asked::Nothing => {
948            return Err(Halt::Because(
949                "the dock handler named no pier for your ship.".to_owned(),
950            ));
951        }
952    };
953    let info = ship_known(handle, stop, &hull).await?;
954    // A pier is several rooms: the ship is in one, once it is there.
955    let by = tokio::time::Instant::now() + ARRIVES_WITHIN;
956    loop {
957        for room in ocean.piers_titled(&pier) {
958            let from = here(handle, dir, roster_name).await?;
959            let Some(room) = room_of(&from.map, room) else {
960                continue;
961            };
962            Box::pin(walk(handle, dir, roster_name, stop, from, room)).await?;
963            if own_ship_here(&state_of(handle).await?, &info) {
964                return Ok(());
965            }
966        }
967        if tokio::time::Instant::now() >= by {
968            break;
969        }
970        say(format!("your {hull} is not at {pier} yet: looking again."));
971        until(handle, stop, PIER_AGAIN, |_| false).await?;
972    }
973    Err(Halt::Because(format!(
974        "I could not find your {hull} at {pier}."
975    )))
976}

978/// The character's `hull` as SHIP INFO describes it, asked of the game
979/// when its look has not been told: by it the ship is known among others
980/// of its hull.
981async fn ship_known(
982    handle: &SessionHandle,
983    stop: &CancellationToken,
984    hull: &str,
985) -> Result<hydra_model::state::ship::reports::ShipInfo, Halt> {
986    let known = |state: &GameState| {
987        state
988            .ship
989            .owned
990            .iter()
991            .find(|(key, _)| key.eq_ignore_ascii_case(hull))
992            .map(|(_, info)| info.clone())
993    };
994    let told = |state: &GameState| known(state).filter(|info| !info.fields.is_empty());
995    if let Some(info) = told(&state_of(handle).await?) {
996        return Ok(info);
997    }
998    send_quietly(handle, stop, &format!("ship info {hull}")).await?;
999    until(handle, stop, Duration::from_secs(3), |state| {
1000        told(state).is_some()
1001    })
1002    .await?;
1003    let state = state_of(handle).await?;
1004    Ok(told(&state).or_else(|| known(&state)).unwrap_or_default())
1005}

1007/// The hull the character owns, in the word a dock handler takes: asked of
1008/// the game when nothing has said.
1009async fn hull_owned(handle: &SessionHandle, stop: &CancellationToken) -> Result<String, Halt> {
1010    let owned = |state: &GameState| {
1011        state
1012            .ship
1013            .owned
1014            .keys()
1015            .next()
1016            .map(|hull| hull.to_lowercase())
1017    };
1018    if let Some(hull) = owned(&state_of(handle).await?) {
1019        return Ok(hull);
1020    }
1021    send_quietly(handle, stop, "ship info").await?;
1022    until(handle, stop, Duration::from_secs(3), |state| {
1023        !state.ship.owned.is_empty()
1024    })
1025    .await?;
1026    owned(&state_of(handle).await?).ok_or_else(|| {
1027        Halt::Because("SHIP INFO names no ship of yours, so I have none to sail.".to_owned())
1028    })
1029}

1031/// Walk from `from` to `goal`, the walk saying for itself how it ended.
1032async fn walk(
1033    handle: &SessionHandle,
1034    dir: &Path,
1035    roster_name: &str,
1036    stop: &CancellationToken,
1037    from: Here,
1038    goal: RoomId,
1039) -> Result<(), Halt> {
1040    if from.room == goal {
1041        return Ok(());
1042    }
1043    let (instance, name) = crate::pages::who(roster_name)?;
1044    let setting = Setting {
1045        settings: crate::pages::travel_settings(dir, roster_name),
1046        memories: from.notes.memories.clone(),
1047        last_room: Some(from.room.0),
1048        targets: from.notes.targets.clone(),
1049        into: None,
1050        instability_at: None,
1051        instabilities: file::instabilities(dir, instance).unwrap_or_default(),
1052        now: std::time::SystemTime::now()
1053            .duration_since(std::time::UNIX_EPOCH)
1054            .map_or(0, |since| since.as_secs()),
1055    };
1056    let map = Arc::clone(&from.map);
1057    let walked = drive::walk(
1058        handle,
1059        &map,
1060        goal,
1061        setting,
1062        stop,
1063        |route| {
1064            hydra_map::show_route(
1065                roster_name,
1066                Some(hydra_map::ShownRoute {
1067                    rooms: route.to_vec(),
1068                }),
1069            );
1070        },
1071        |memories| {
1072            let _ = file::save_memories(dir, instance, name, memories);
1073        },
1074    )
1075    .await;
1076    ended(handle, roster_name, &walked, true);
1077    if walked.ended == Ended::Arrived {
1078        Ok(())
1079    } else {
1080        Err(Halt::Said)
1081    }
1082}

1093/// Send `line`, unless the journey is stopped.

1100    } else {

1126/// Wait until the model says `seen`, or `most` has gone by: whether it did.
1127async fn until(
1128    handle: &SessionHandle,
1129    stop: &CancellationToken,
1130    most: Duration,
1131    seen: impl Fn(&GameState) -> bool,
1132) -> Result<bool, Halt> {
1133    let by = tokio::time::Instant::now() + most;
1134    loop {
1135        if seen(&state_of(handle).await?) {
1136            return Ok(true);
1137        }
1138        if tokio::time::Instant::now() >= by {
1139            return Ok(false);
1140        }
1141        tokio::select! {
1142            () = stop.cancelled() => return Err(Halt::Because("stopped.".to_owned())),
1143            () = tokio::time::sleep(Duration::from_millis(250)) => {}
1144        }
1145    }
1146}

1259        let rooms: Vec<hydra_travel::Room> = serde_json::from_str(

1261                 "exits":[{"to":2,"kind":"cardinal","cmd":"north","cost":1},

1267                {"id":4,"location":"the free port of Solhaven","tags":["bank"],"title":["[Bank of Vornavis, Solhaven]"]}]"#,
1268        )
1269        .expect("rooms");
1270        let here = Here {
1271            map: Arc::new(Map::from_rooms(rooms).expect("a map")),
1272            walker: hydra_travel::Walker::default(),
1273            notes: file::Notes::default(),
1274            room: RoomId(1),
1275        };
1276        let mut parser = hydra_protocol::Parser::new();
1277        let mut state = GameState::default();
1278        for line in [

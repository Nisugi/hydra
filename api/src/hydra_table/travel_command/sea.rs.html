1//! `.go2` across the sea: a walk to the ship, a voyage, and a walk on.
2//!
3//! When no way by land leads where `.go2` was asked to go, the character's
4//! own ship is the way: [`passage`] finds the port nearest by land, and the
5//! port from which the place can be walked to with the shortest voyage
6//! between the two. [`set_out`] then makes the whole journey in the walk's
7//! place, so one `.stop` stops whichever part of it is under way:
8//!
9//! 1. To the ship: to the port's dock handler, who is asked for it and
10//!    names its pier, whether the ship is put there now or is out already.
11//!    Without the handler's fee, to the bank for it and back. A pier is
12//!    several rooms long and the ship lies in one of them, so each is
13//!    walked to until the ship is seen.
14//! 2. The voyage (`hydra_travel::sail`), which boards the ship itself.
15//!    The ship is left at the far pier, not stored: it lies there a good
16//!    while, and may be wanted again.
17//! 3. Down the gangplank, and the walk on, found afresh from the far pier.
18//!
19//! The places of Sailor's Grief have open water between them, which the
20//! map does not cross: the way on, there, is a swim ([`swim`]) and then a
21//! walk, and a journey between two of them is that alone, with no ship.
22

44/// The way by sea, for a walk with none by land: `passage` when there is
45/// one and the character lets its ship be taken (`own_ship`); else `none`
46/// said, with the ship offered when it would have served.
47fn way(
48    passage: Option<Passage>,
49    (own_ship, sure): (bool, bool),
50    none: String,
51) -> Result<super::Way, super::Said> {
52    match passage {
53        // A room of the open sea, asked for from land: said back first, so
54        // a number mistyped does not put to sea.
55        Some(Passage {
56            sail: Some((from, to)),
57            what,
58            named,
59        }) if own_ship && !to.port && !sure => Err(super::Said::Lines(vec![format!(
60            "{named} is a room of the open sea, {} by ship from {}. To sail there and drop \
61             anchor: .go2 {what} --no-confirm",
62            rooms_word(to.uid, from),
63            from.name
64        )])),
65        Some(passage) if own_ship || passage.sail.is_none() => Ok(super::Way::Sea(passage)),
66        Some(Passage {
67            sail: Some((from, to)),
68            ..
69        }) => Err(super::Said::Lines(vec![format!(
70            "{none} Your own ship could take you, from {} to {}: turn on \"Use your own ship\" \n             on the Travel page.",
71            from.name, to.name
72        )])),
73        Some(_) | None => Err(super::Said::Lines(vec![none])),
74    }
75}

77/// What finds the way by sea for a walk with none by land, as the
78/// character is now: asked with where it is, what it asked for, and what to
79/// say when the sea is no way either.
80pub(super) async fn asker(
81    handle: &SessionHandle,
82    dir: &Path,
83    roster_name: &str,
84) -> impl Fn(&Here, &str, String) -> Result<super::Way, super::Said> + Send + 'static {
85    let ocean = hydra_map::ocean().as_ref().ok();
86    let state = handle.snapshot().await.map(|snapshot| snapshot.state);
87    let own_ship = crate::pages::travel_settings(dir, roster_name).on(OWN_SHIP);
88    move |here: &Here, what: &str, none: String| {
89        let (what, sure) = match what.strip_suffix(" --no-confirm") {
90            Some(what) => (what, true),
91            None => (what, false),
92        };
93        let passage = ocean
94            .zip(state.as_ref())
95            .and_then(|(ocean, state)| passage(here, ocean, state, what));
96        way(passage, (own_ship, sure), none)
97    }
98}

100/// How far `room` is from `port` by ship, in words.
101fn rooms_word(room: i64, port: &Port) -> String {
102    let ocean = hydra_map::ocean().as_ref().ok();
103    match ocean.and_then(|ocean| ocean.course(port.uid, room)) {
104        Some(course) => format!("{} rooms", course.rooms.len()),
105        None => "no known course".to_owned(),
106    }
107}

111pub(super) struct Passage {
112    /// The port the ship sails from -- the nearest by land, or the one the
113    /// ship the character stands on lies in -- and the port it sails to.

143/// A pier of `port`, as a room of the map.
144fn a_pier(map: &Map, ocean: &Ocean, port: &Port) -> Option<RoomId> {
145    let mut piers: Vec<i64> = ocean.piers_of(port).collect();
146    piers.sort_unstable();
147    piers.into_iter().find_map(|pier| room_of(map, pier))
148}

150/// The way to `what` by sea, for a character with no way there by land:
151/// `None` when no port can be walked to from here, or the place can be
152/// walked to from no other port.
153fn passage(here: &Here, ocean: &'static Ocean, state: &GameState, what: &str) -> Option<Passage> {
154    let steps = |from: RoomId, to: RoomId| {
155        route::itinerary(&here.map, &here.walker, from, to).map(|legs| legs.len())
156    };
157    let named = |goal: RoomId| {
158        let room = here.map.room(goal);
159        room.and_then(|room| room.title.first().cloned())
160            .unwrap_or_default()
161    };
162    // A swim alone, between two places of Sailor's Grief.
163    if aboard(&state.room).is_none()
164        && let Some(crossing) = swim::crossing(here, here.room, what)
165    {
166        return Some(Passage {
167            sail: None,
168            what: what.to_owned(),
169            named: named(crossing.goal),
170        });
171    }
172    // From the ship stood on, the port it lies in; else the nearest by land.
173    let from = if aboard(&state.room).is_some() {
174        match hydra_map::Navigator::default().fix(ocean, state) {
175            hydra_map::Sailing::Docked { port: Some(port) }
176            | hydra_map::Sailing::At { room: port, .. } => ocean.port_at(port),
177            _ => ocean.port_of_pier_titled(&state.ship.berth.as_ref()?.pier),
178        }?
179    } else {
180        ocean
181            .ports()
182            .iter()
183            .filter_map(|port| {
184                let pier = a_pier(&here.map, ocean, port)?;
185                Some((steps(here.room, pier)?, port))
186            })
187            .min_by_key(|(far, _)| *far)?
188            .1
189    };
190    // A room of the open sea: sailed to from the port, and the anchor dropped.
191    if let Some(bound) = ocean.bound(what).filter(|bound| !bound.port) {
192        let sea = ocean.room(bound.uid).map_or("", |room| room.sea.as_str());
193        let named = format!("{} ({sea})", bound.name);
194        return Some(Passage {
195            sail: Some((from, bound)),
196            what: what.to_owned(),
197            named,
198        });
199    }
200    let (to, goal) = ocean
201        .ports()
202        .iter()
203        .filter(|port| port.uid != from.uid)
204        .filter_map(|port| {
205            let pier = a_pier(&here.map, ocean, port)?;
206            let Meant::Room(goal) =
207                route::destination(&here.map, &here.walker, pier, what, &here.notes.targets)
208            else {
209                return None;
210            };
211            // Walked to from the pier, or swum to from it.
212            let goal = match steps(pier, goal) {
213                Some(_) => goal,
214                None => swim::crossing(here, pier, what)?.goal,
215            };
216            let voyage = ocean.course(from.uid, port.uid)?.rooms.len();
217            Some((voyage, port, goal))
218        })
219        .min_by_key(|(voyage, ..)| *voyage)
220        .map(|(_, port, goal)| (port, goal))?;
221    Some(Passage {
222        sail: Some((
223            from,
224            Bound {
225                uid: to.uid,
226                name: to.name.clone(),
227                port: true,
228            },
229        )),
230        what: what.to_owned(),
231        named: named(goal),
232    })
233}

235/// Make the journey `passage` describes, in a task of its own, in the place
236/// `claim` took: what is said of it as it sets out.
237pub(super) fn set_out(
238    handle: SessionHandle,
239    roster_name: String,
240    passage: Passage,
241    mut claim: Claim,
242) -> Result<super::Said, String> {
243    let ocean = hydra_map::ocean().as_ref().map_err(Clone::clone)?;
244    if claim.stopped() {
245        return Ok(super::Said::Lines(vec![
246            "Stopped before setting out.".to_owned(),
247        ]));
248    }
249    let said = match &passage.sail {
250        Some((_, to)) if !to.port => format!("To {}, by sea, to drop anchor there.", passage.named),
251        Some((from, to)) => format!(
252            "To {} by sea: from {} to {}, and on from there.",
253            passage.named, from.name, to.name
254        ),
255        None => format!("To {}, by a swim.", passage.named),
256    };
257    claim.set_out();
258    let (key, number, stop) = (claim.key.clone(), claim.number, claim.stop.clone());
259    crate::commands::spawn(async move {
260        let dir = crate::account::data_dir();
261        let halted = Box::pin(journey(&handle, &dir, &roster_name, ocean, &passage, &stop)).await;
262        if let Ok(mut walks) = walks().lock()
263            && let Some(walking) = walks.get_mut(&key)
264            && walking
265                .under_way
266                .as_ref()
267                .is_some_and(|(at, _)| *at == number)
268        {
269            walking.under_way = None;
270        }
271        hydra_map::show_course(&roster_name, None);
272        if let Err(Halt::Because(why)) = halted {
273            handle.say(Notice::line(NoticeKind::Error, format!("Travel: {why}")).answering());
274        }
275    });
276    Ok(super::Said::Lines(vec![said]))
277}

279async fn journey(
280    handle: &SessionHandle,
281    dir: &Path,
282    roster_name: &str,
283    ocean: &'static Ocean,
284    passage: &Passage,
285    stop: &CancellationToken,
286) -> Result<(), Halt> {
287    if let Some((from, to)) = &passage.sail {
288        Box::pin(voyage(handle, dir, roster_name, ocean, (from, to), stop)).await?;
289        // At a room of the sea the journey ends, at anchor.
290        if !to.port {
291            return Ok(());
292        }
293    }
294    Box::pin(onward(handle, dir, roster_name, stop, &passage.what)).await
295}

297/// The way on to `what` from where the character stands: a walk; or, with
298/// open water between, a swim and then a walk.
299async fn onward(
300    handle: &SessionHandle,
301    dir: &Path,
302    roster_name: &str,
303    stop: &CancellationToken,
304    what: &str,
305) -> Result<(), Halt> {
306    let none = || Halt::Because(format!("I know no way on to {what} from here."));
307    let mut from = here(handle, dir, roster_name).await?;
308    let goal = |from: &Here| {
309        let meant = route::destination(
310            &from.map,
311            &from.walker,
312            from.room,
313            what,
314            &from.notes.targets,
315        );
316        match meant {
317            Meant::Room(goal) => Some(goal),
318            _ => None,
319        }
320    };
321    let walked_to = |from: &Here, goal: RoomId| {
322        goal == from.room || route::itinerary(&from.map, &from.walker, from.room, goal).is_some()
323    };
324    if !goal(&from).is_some_and(|goal| walked_to(&from, goal)) {
325        let crossing = swim::crossing(&from, from.room, what).ok_or_else(none)?;
326        Box::pin(swim::cross(handle, dir, roster_name, stop, &crossing)).await?;
327        from = here(handle, dir, roster_name).await?;
328    }
329    let goal = goal(&from).ok_or_else(none)?;
330    Box::pin(walk(handle, dir, roster_name, stop, from, goal)).await
331}

333/// To the ship at `from`, the voyage to `to`, and down its gangplank.
334async fn voyage(
335    handle: &SessionHandle,
336    dir: &Path,
337    roster_name: &str,
338    ocean: &'static Ocean,
339    (from, to): (&'static Port, &Bound),
340    stop: &CancellationToken,
341) -> Result<(), Halt> {
342    let state = state_of(handle).await?;
343    if aboard(&state.room).is_none() {
344        Box::pin(to_the_ship(handle, dir, roster_name, ocean, from, stop)).await?;
345    }
346    let teller = handle.clone();
347    let sailed = hydra_travel::sail::sail(
348        handle,
349        stop,
350        ocean,
351        to.uid,
352        Helm::Hydra {
353            wind: crate::pages::travel_settings(dir, roster_name)
354                .on(hydra_travel::settings::CALL_WIND),
355        },
356        |rooms| {
357            hydra_map::show_course(
358                roster_name,
359                rooms.map(|rooms| hydra_map::ShownCourse {
360                    rooms: rooms.to_vec(),
361                    port: to.name.clone(),
362                }),
363            );
364        },
365        |said| {
366            teller.say(Notice::line(NoticeKind::Info, format!("Sailing: {said}")).answering());
367        },
368    )
369    .await;
370    match sailed {
371        Sailed::Arrived if !to.port => {
372            handle.say(
373                Notice::line(
374                    NoticeKind::Info,
375                    format!(
376                        "Travel: at {}, and the anchor is down. The wheel is yours.",
377                        to.name
378                    ),
379                )
380                .answering(),
381            );
382            return Ok(());
383        }
384        Sailed::Arrived => {}
385        Sailed::Stopped => return Err(Halt::Because("stopped. The wheel is yours.".to_owned())),
386        Sailed::Lost => {
387            return Err(Halt::Because(
388                "stopped, because the connection went away.".to_owned(),
389            ));
390        }
391        Sailed::Halted(why) => return Err(Halt::Because(format!("{why} The wheel is yours."))),
392    }
393    // Down the gangplank, and on by land from wherever that is.
394    send(handle, stop, "go gangplank").await?;
395    let ashore = until(handle, stop, Duration::from_secs(8), |state| {
396        state.room.title.is_some() && aboard(&state.room).is_none()
397    })
398    .await?;
399    if !ashore {
400        return Err(Halt::Because(format!(
401            "in port at {}, and I could not get ashore.",
402            to.name
403        )));
404    }
405    Ok(())
406}

408/// What a dock handler answered, asked for the ship.
409enum Asked {
410    /// The pier it is at: put there now, or out already.
411    Pier(String),
412    /// The character has not the fee with it.
413    Refused,
414    /// Nothing Hydra knows.
415    Nothing,
416}

418/// Ask the dock handler here for the ship, twice: the first is answered
419/// with the fee, the second pays it. A ship out already is named at the
420/// first.
421async fn ask(handle: &SessionHandle, stop: &CancellationToken, hull: &str) -> Result<Asked, Halt> {
422    let ask = format!("ask handler about retrieve {hull}");
423    let before = state_of(handle).await?.ship.launch;
424    send(handle, stop, &ask).await?;
425    // The second goes the moment the first is answered.
426    until(handle, stop, Duration::from_secs(4), |state| {
427        state.ship.launch != before
428    })
429    .await?;
430    if state_of(handle).await?.ship.launch.piers == before.piers {
431        send(handle, stop, &ask).await?;
432        until(handle, stop, Duration::from_secs(4), |state| {
433            let launch = state.ship.launch;
434            launch.piers != before.piers || launch.refusals != before.refusals
435        })
436        .await?;
437    }
438    let state = state_of(handle).await?;
439    Ok(if state.ship.launch.piers != before.piers {
440        state
441            .ship
442            .berth
443            .map_or(Asked::Nothing, |berth| Asked::Pier(berth.pier))
444    } else if state.ship.launch.refusals != before.refusals {
445        Asked::Refused
446    } else {
447        Asked::Nothing
448    })
449}

451/// To the ship: to the port's dock handler, who is asked for it and names
452/// its pier, whether it puts the ship there or it is out already; to the
453/// bank and back for the fee when the character has not got it; and along
454/// the pier until the ship is in sight.
455async fn to_the_ship(
456    handle: &SessionHandle,
457    dir: &Path,
458    roster_name: &str,
459    ocean: &'static Ocean,
460    port: &'static Port,
461    stop: &CancellationToken,
462) -> Result<(), Halt> {
463    let hull = hull_owned(handle, stop).await?;
464    let from = here(handle, dir, roster_name).await?;
465    let Some(handler) = port.handler.and_then(|uid| room_of(&from.map, uid)) else {
466        return Err(Halt::Because(format!(
467            "I do not know where {}'s dock handler is.",
468            port.name
469        )));
470    };
471    let say = |said: String| {
472        handle.say(Notice::line(NoticeKind::Info, format!("Travel: {said}")).answering());
473    };
474    say(format!("to {}'s dock handler, for the {hull}.", port.name));
475    Box::pin(walk(handle, dir, roster_name, stop, from, handler)).await?;
476    let mut asked = ask(handle, stop, &hull).await?;
477    if matches!(asked, Asked::Refused) {
478        let Some(fee) = state_of(handle).await?.ship.launch.fee else {
479            return Err(Halt::Because(format!(
480                "the dock handler wants a fee to launch your {hull}, and you have not got it."
481            )));
482        };
483        say(format!(
484            "to the bank, for the {fee} silvers the handler wants."
485        ));
486        let from = here(handle, dir, roster_name).await?;
487        let Meant::Room(bank) = route::destination(
488            &from.map,
489            &from.walker,
490            from.room,
491            "bank",
492            &from.notes.targets,
493        ) else {
494            return Err(Halt::Because(format!(
495                "the dock handler wants {fee} silvers to launch your {hull}, and I know no bank \
496                 to walk to for it."
497            )));
498        };
499        Box::pin(walk(handle, dir, roster_name, stop, from, bank)).await?;
500        send(handle, stop, &format!("withdraw {fee} silver")).await?;
501        until(handle, stop, Duration::from_secs(2), |_| false).await?;
502        let from = here(handle, dir, roster_name).await?;
503        Box::pin(walk(handle, dir, roster_name, stop, from, handler)).await?;
504        asked = ask(handle, stop, &hull).await?;
505    }
506    let pier = match asked {
507        Asked::Pier(pier) => pier,
508        Asked::Refused => {
509            return Err(Halt::Because(format!(
510                "the dock handler still will not launch your {hull}: the bank did not give me \
511                 the fee."
512            )));
513        }
514        Asked::Nothing => {
515            return Err(Halt::Because(
516                "the dock handler named no pier for your ship.".to_owned(),
517            ));
518        }
519    };
520    // A pier is several rooms: the ship is in one.
521    for room in ocean.piers_titled(&pier) {
522        let from = here(handle, dir, roster_name).await?;
523        let Some(room) = room_of(&from.map, room) else {
524            continue;
525        };
526        Box::pin(walk(handle, dir, roster_name, stop, from, room)).await?;
527        let state = state_of(handle).await?;
528        if state
529            .room
530            .objects
531            .iter()
532            .any(|thing| SHIPS.contains(&thing.noun.as_str()))
533        {
534            return Ok(());
535        }
536    }
537    Err(Halt::Because(format!(
538        "I could not find your ship at {pier}."
539    )))
540}

542/// The hull the character owns, in the word a dock handler takes: asked of
543/// the game when nothing has said.
544async fn hull_owned(handle: &SessionHandle, stop: &CancellationToken) -> Result<String, Halt> {
545    let owned = |state: &GameState| {
546        state
547            .ship
548            .owned
549            .keys()
550            .next()
551            .map(|hull| hull.to_lowercase())
552    };
553    if let Some(hull) = owned(&state_of(handle).await?) {
554        return Ok(hull);
555    }
556    send(handle, stop, "ship info").await?;
557    until(handle, stop, Duration::from_secs(3), |state| {
558        !state.ship.owned.is_empty()
559    })
560    .await?;
561    owned(&state_of(handle).await?).ok_or_else(|| {
562        Halt::Because("SHIP INFO names no ship of yours, so I have none to sail.".to_owned())
563    })
564}

566/// Walk from `from` to `goal`, the walk saying for itself how it ended.
567async fn walk(
568    handle: &SessionHandle,
569    dir: &Path,
570    roster_name: &str,
571    stop: &CancellationToken,
572    from: Here,
573    goal: RoomId,
574) -> Result<(), Halt> {
575    if from.room == goal {
576        return Ok(());
577    }
578    let (instance, name) = crate::pages::who(roster_name)?;
579    let setting = Setting {
580        settings: crate::pages::travel_settings(dir, roster_name),
581        memories: from.notes.memories.clone(),
582        last_room: Some(from.room.0),
583        targets: from.notes.targets.clone(),
584        into: None,
585        instability_at: None,
586        instabilities: file::instabilities(dir, instance).unwrap_or_default(),
587        now: std::time::SystemTime::now()
588            .duration_since(std::time::UNIX_EPOCH)
589            .map_or(0, |since| since.as_secs()),
590    };
591    let map = Arc::clone(&from.map);
592    let walked = drive::walk(
593        handle,
594        &map,
595        goal,
596        setting,
597        stop,
598        |route| {
599            hydra_map::show_route(
600                roster_name,
601                Some(hydra_map::ShownRoute {
602                    rooms: route.to_vec(),
603                }),
604            );
605        },
606        |memories| {
607            let _ = file::save_memories(dir, instance, name, memories);
608        },
609    )
610    .await;
611    ended(handle, roster_name, &walked, true);
612    if walked.ended == Ended::Arrived {
613        Ok(())
614    } else {
615        Err(Halt::Said)
616    }
617}
618
619/// The character's model, now.
620async fn state_of(handle: &SessionHandle) -> Result<GameState, Halt> {
621    handle
622        .snapshot()
623        .await
624        .map(|snapshot| snapshot.state)
625        .ok_or_else(|| Halt::Because("the character is not running.".to_owned()))
626}
627
628/// Send `line`, unless the journey is stopped.
629async fn send(handle: &SessionHandle, stop: &CancellationToken, line: &str) -> Result<(), Halt> {
630    if stop.is_cancelled() {
631        return Err(Halt::Because("stopped.".to_owned()));
632    }
633    if handle.write(line, Origin::Travel).await {
634        Ok(())
635    } else {
636        Err(Halt::Because(
637            "stopped, because the connection went away.".to_owned(),
638        ))
639    }
640}
641
642/// Wait until the model says `seen`, or `most` has gone by: whether it did.
643async fn until(
644    handle: &SessionHandle,
645    stop: &CancellationToken,
646    most: Duration,
647    seen: impl Fn(&GameState) -> bool,
648) -> Result<bool, Halt> {
649    let by = tokio::time::Instant::now() + most;
650    loop {
651        if seen(&state_of(handle).await?) {
652            return Ok(true);
653        }
654        if tokio::time::Instant::now() >= by {
655            return Ok(false);
656        }
657        tokio::select! {
658            () = stop.cancelled() => return Err(Halt::Because("stopped.".to_owned())),
659            () = tokio::time::sleep(Duration::from_millis(250)) => {}
660        }
661    }
662}

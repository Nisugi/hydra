93/// The instance and name a roster name (`GAME:Name`) means.
94///
95/// # Errors
96///
97/// It is not a roster name, or names a game Hydra does not know.
98pub fn who(character: &str) -> Result<(&'static str, &str), String> {
99    let (game, name) = character
100        .split_once(':')
101        .ok_or_else(|| format!("{character} is not a roster name (GAME:Name)."))?;
102    let instance =
103        hydra_net::instance(game).ok_or_else(|| format!("{game} is not a game Hydra knows."))?;
104    Ok((instance, name.trim()))
105}

107/// `character`'s pages, as its file holds them.
108///
109/// # Errors
110///
111/// As [`who`].
112pub fn pages(dir: &Path, character: &str) -> Result<Vec<Page>, String> {
113    let (instance, name) = who(character)?;
114    let file_name = settings_store::settings_path(dir, instance, name).map_or_else(
115        || format!("{instance}_{name}.settings.json"),
116        |path| {
117            path.strip_prefix(dir)
118                .unwrap_or(&path)
119                .display()
120                .to_string()
121        },
122    );
123    let page = |id: &str, title: &str, takes: &str| Page {
124        id: id.to_owned(),
125        title: title.to_owned(),
126        file: file_name.clone(),
127        takes: takes.to_owned(),
128        problem: None,
129        rows: Vec::new(),
130    };
131    let mut general = page(GENERAL, "General", "at once");
132    let mut log = page(LOG, "Player log", "the next time the character logs in");
133    let mut theme = page(THEME, "Theme", "at once");
134    let mut travel = page(TRAVEL, "Travel", "the next route asked for");
135    let mut macros = macros_page(dir, instance, name);
136    let rows = load_file(dir, instance, name).and_then(|file| {
137        Ok((
138            general_rows(
139                &section(&file, ShownIn::SECTION)?,
140                &section(&file, claimant::SECTION)?,
141                &section(&file, Shown::SECTION)?,
142                &section(&file, Sorting::SECTION)?,
143                (
144                    &section(&file, Kept::SECTION)?,
145                    &section(&file, Scrollback::SECTION)?,
146                ),
147            ),
148            log_rows(
149                &section(&file, LOG_SECTION)?,
150                character,
151                hydra_board::keeps_player_log(dir),
152            ),
153            theme_rows(
154                &section(&file, Chosen::SECTION)?,
155                &Themes::load(&dir.join("themes")).names(),
156            ),
157            travel_rows(
158                &section(&file, travel::SECTION)?,
159                &memories(dir, instance, name),
160            ),
161        ))
162    });
163    match rows {
164        Ok((general_rows, log_rows, theme_rows, travel_rows)) => {
165            general.rows = general_rows;
166            log.rows = log_rows;
167            theme.rows = theme_rows;
168            travel.rows = travel_rows;
169        }
170        Err(why) => {
171            for page in [&mut general, &mut log, &mut theme, &mut travel, &mut macros] {
172                page.problem = Some(format!("Nothing here is changed while {why}"));
173            }
174        }
175    }
176    let fights = match load_file(dir, instance, name) {
177        Ok(file) => fights::page(dir, &file, &file_name),
178        Err(why) => {
179            let mut page = fights::page(dir, &SettingsFile::new(instance, name), &file_name);
180            page.rows.clear();
181            page.problem = Some(format!("Nothing here is changed while {why}"));
182            page
183        }
184    };
185    Ok(vec![
186        general,
187        log,
188        theme,
189        travel,
190        fights,
191        macros,
192        rewards_page(dir),
193        fights::every_page(dir),
194    ])
195}

284/// A change to the *Rewards* page: `key` to `to`, written to `rewards.toml`.
285/// What was done.
286fn rewards_change(dir: &Path, key: &str, to: Option<&str>) -> Result<String, String> {
287    use hydra_rewards::schedule::Settings;
288    let mut settings = Settings::read(dir)
289        .map_err(|why| format!("Nothing was changed: {why}"))?
290        .unwrap_or_else(none_rewards);
291    let said = match key {
292        "claim" => {
293            if switch(key, to)? == Some(true) {
294                let minute = settings.minute.unwrap_or(DEFAULT_MINUTE);
295                settings.minute = Some(minute);
296                format!(
297                    "Login rewards are claimed at {:02}:{:02}.",
298                    minute / 60,
299                    minute % 60
300                )
301            } else {
302                settings.minute = None;
303                "Login rewards are not claimed.".to_owned()
304            }
305        }
306        "time" => {
307            if settings.minute.is_none() {
308                return Err("Turn on Claim the login reward first.".to_owned());
309            }
310            let hours: f64 = to
311                .and_then(|text| text.trim().parse().ok())
312                .ok_or_else(|| "The time is hours from midnight, such as 19.25.".to_owned())?;
313            if !(0.0..24.0).contains(&hours) {
314                return Err("The time is within the day.".to_owned());
315            }
316            // A quarter hour a step on the slider; whole minutes after that.
317            #[expect(
318                clippy::cast_possible_truncation,
319                clippy::cast_sign_loss,
320                reason = "within the day, checked above"
321            )]
322            let minute = ((hours * 60.0).round() as u32).min(24 * 60 - 1);
323            settings.minute = Some(minute);
324            format!(
325                "Login rewards are claimed at {:02}:{:02}.",
326                minute / 60,
327                minute % 60
328            )
329        }
330        "store" => {
331            settings.store = switch(key, to)? == Some(true);
332            if settings.store {
333                "The store's Simucoins are claimed.".to_owned()
334            } else {
335                "The store's Simucoins are not claimed.".to_owned()
336            }
337        }
338        "skip" => {
339            if settings.minute.is_none() {
340                return Err("Turn on Claim the login reward first.".to_owned());
341            }
342            settings.skip = to
343                .unwrap_or("")
344                .split(',')
345                .map(|word| word.trim().to_owned())
346                .filter(|word| !word.is_empty())
347                .collect();
348            format!(
349                "{} characters are never logged in for.",
350                settings.skip.len()
351            )
352        }
353        other => return Err(format!("The Rewards page has no setting {other}.")),
354    };
355    Settings::write(dir, Some(&settings))?;
356    Ok(said)
357}

420/// A change to the *Macros* page: `key`'s table as the page wrote it, a
421/// JSON list of rows; `None` empties it. What was done.
422fn macros_change(
423    dir: &Path,
424    (instance, name): (&str, &str),
425    key: &str,
426    to: Option<&str>,
427) -> Result<String, String> {
428    let path = hydra_board::macros_paths(dir, Some(instance), name)[1].clone();
429    // The macros file's own lock, beside the settings file's: `.macro`
430    // changes this file too.
431    hydra_session::store::changing(&path, || macros_changed(&path, key, to))
432}

434/// [`macros_change`] to the file at `path`, its lock held.
435fn macros_changed(path: &Path, key: &str, to: Option<&str>) -> Result<String, String> {
436    use hydra_ui::macros::{MacroDef, MacroFile};
437    let path = path.to_path_buf();
438    let mut file = MacroFile::load(&path).map_err(|why| format!("Nothing was changed: {why}"))?;
439    let rows: Vec<Vec<String>> = match to {
440        None => Vec::new(),
441        Some(text) => serde_json::from_str(text).map_err(|why| format!("Not a table: {why}"))?,
442    };
443    let cell =
444        |row: &[String], at: usize| row.get(at).map(|s| s.trim()).unwrap_or_default().to_owned();
445    let said = match key {
446        MACROS_TABLE => {
447            let mut macros = Vec::new();
448            for row in &rows {
449                let (made_name, chain) = (cell(row, 0), cell(row, 1));
450                if made_name.is_empty() && chain.is_empty() {
451                    continue;
452                }
453                if made_name.is_empty() || made_name.contains(char::is_whitespace) {
454                    return Err(format!("A macro's name is one word: `{made_name}` is not."));
455                }
456                hydra_session::steps::steps(&chain)
457                    .map_err(|why| format!("The macro {made_name}'s commands: {why}."))?;
458                let set = match cell(row, 3).as_str() {
459                    "" => None,
460                    digit => Some(
461                        digit
462                            .parse::<u8>()
463                            .ok()
464                            .filter(|set| *set <= 9)
465                            .ok_or_else(|| format!("A set is 0 to 9: `{digit}` is not."))?,
466                    ),
467                };
468                let mut made = file.named(&made_name).cloned().unwrap_or_else(|| MacroDef {
469                    name: made_name.clone(),
470                    ..MacroDef::default()
471                });
472                made.name = made_name;
473                made.chain = chain;
474                made.key = Some(cell(row, 2)).filter(|key| !key.is_empty());
475                made.set = set;
476                made.alias = Some(cell(row, 4)).filter(|word| !word.is_empty());
477                macros.push(made);
478            }
479            file.macros = macros;
480            format!("{} macros kept.", file.macros.len())
481        }
482        other => return Err(format!("The Macros page has no setting {other}.")),
483    };
484    let folder = path.parent().map(Path::to_path_buf).unwrap_or_default();
485    hydra_session::store::save_bytes(&folder, &path, file.to_toml().as_bytes())
486        .map_err(|why| format!("Not saved: {why}"))?;
487    Ok(said)
488}

499/// `character`'s (`GAME:Name`) section `name`, as its file says: the
500/// defaults when it says nothing or cannot be read.
501fn read<T: serde::de::DeserializeOwned + Default>(dir: &Path, character: &str, name: &str) -> T {
502    who(character)
503        .ok()
504        .and_then(|(instance, who)| load_file(dir, instance, who).ok())
505        .and_then(|file| file.section(name).ok())
506        .unwrap_or_default()
507}

509/// `character`'s settings file, with the travel settings an earlier Hydra
510/// kept in travel.json carried into its travel section: once, the first
511/// time the section is found empty, and saved. A value the page no longer
512/// has is dropped.
513fn load_file(dir: &Path, instance: &str, name: &str) -> Result<SettingsFile, String> {
514    let mut file = settings_store::load(dir, instance, name).map_err(|why| why.to_string())?;
515    let mut settings: TravelSettings = section(&file, travel::SECTION)?;
516    if !settings.0.is_empty() {
517        return Ok(file);
518    }
519    let old = hydra_travel::file::take_settings(dir, instance, name)
520        .map_err(|why| format!("travel.json: {why}"))?;
521    if old.is_empty() {
522        return Ok(file);
523    }
524    for (key, value) in &old {
525        let _ = settings.set(key, Some(value));
526    }
527    put(&mut file, travel::SECTION, &settings)?;
528    settings_store::save(dir, &file).map_err(|why| format!("Not saved: {why}"))?;
529    Ok(file)
530}

600/// indented under the setting it belongs to, and the memories the map keys
601/// on shown as settings among them.
602fn travel_rows(settings: &TravelSettings, memories: &BTreeMap<String, String>) -> Vec<Row> {
603    let mut rows = Vec::new();
604    for group in travel::GROUPS {
605        rows.push(Row {
606            key: format!("group:{}", group.title),
607            label: group.title.to_owned(),
608            help: String::new(),
609            kind: if group.folded {
610                RowKind::Fold
611            } else {
612                RowKind::Heading
613            },
614            value: Value::Unset,
615            here: false,
616            from: None,
617            needs: None,
618        });
619        if group.memories {
620            rows.extend(
621                travel::MEMORIES
622                    .iter()
623                    .map(|memory| memory_row(memory, memories)),
624            );
625        }
626        // The setting a row sits under: the one above it that does not.
627        let mut over: Option<&travel::Key> = None;
628        for (name, under) in group.keys {
629            let Some(key) = travel::key(name) else {
630                continue;
631            };
632            let mut row = travel_row(key, settings, *under);
633            if !*under {
634                over = Some(key);
635            } else if let Some(over) = over.filter(|over| !travel_set(over, settings)) {
636                row.needs = Some(match over.kind {
637                    KeyKind::Toggle => format!("Turn on {} first.", over.label),
638                    _ => format!("Set {} first.", over.label),
639                });
640            }
641            rows.push(row);
642        }
643    }
644    rows
645}

1298/// `value` out of the quotes the settings menu writes words in, a quote or
1299/// a backslash it escaped put back; anything else as it is.
1300fn unquoted(value: &str) -> String {
1301    let inside = value
1302        .trim()
1303        .strip_prefix('"')
1304        .and_then(|rest| rest.strip_suffix('"'));
1305    let Some(inside) = inside else {
1306        return value.to_owned();
1307    };
1308    let mut plain = String::with_capacity(inside.len());
1309    let mut letters = inside.chars();
1310    while let Some(letter) = letters.next() {
1311        match letter {
1312            '\\' => plain.extend(letters.next()),
1313            letter => plain.push(letter),
1314        }
1315    }
1316    plain
1317}

1319/// [`apply`], with the file's lock held.
1320fn changed(dir: &Path, (instance, name): (&str, &str), wanted: &Change) -> Result<Done, String> {
1321    let mut file = settings_store::load(dir, instance, name)
1322        .map_err(|why| format!("Nothing was changed: {why}"))?;
1323    // Words come from the settings menu in quotes: taken out of them here.
1324    let plain = wanted.to.as_deref().map(unquoted);
1325    let (key, to) = (wanted.key.as_str(), plain.as_deref());
1326    let mut symbols = None;
1327    let said = match (wanted.page.as_str(), key) {
1328        (GENERAL, SHOWN_IN) => shown_in_change(&mut file, name, to)?,
1329        (GENERAL, "symbol") => {
1330            let mut commands: claimant::Settings = section(&file, claimant::SECTION)?;
1331            commands.symbol = match to.map(str::trim) {
1332                None => None,
1333                Some(mark) => {
1334                    let mut marks = mark.chars();
1335                    match (marks.next(), marks.next()) {
1336                        (Some(one), None) if claimant::usable(one) => Some(one.to_string()),
1337                        _ => {
1338                            return Err("The command symbol is one mark, such as . or /: \
1339                                        not a letter, a digit or a space."
1340                                .to_owned());
1341                        }
1342                    }
1343                }
1344            };
1345            if commands.symbol() == commands.lich_symbol() {
1346                return Err(format!(
1347                    "{} is Lich's command symbol: Hydra's must be another.",
1348                    commands.symbol()
1349                ));
1350            }
1351            put(&mut file, claimant::SECTION, &commands)?;
1352            symbols = Some((commands.symbol(), commands.lich_symbol()));
1353            format!("The command symbol is {}.", commands.symbol())
1354        }
1355        (GENERAL, "lich_symbol") => {
1356            let (said, both) = lich_symbol_change(&mut file, to)?;
1357            symbols = Some(both);
1358            said
1359        }
1360        (GENERAL, "targetid") => tags_change(&mut file, to)?,
1361        (GENERAL, key @ (SORTER | SORTER_WIDTH | SORTER_LONG)) => {
1362            sorter_change(&mut file, key, to)?
1363        }
1364        (GENERAL, key @ (HISTORY_MIN | HISTORY_MAX | HISTORY_LINES)) => {
1365            history_change(&mut file, key, to)?
1366        }
1367        (GENERAL, key @ (STORY_LINES | STREAM_LINES)) => scrollback_change(&mut file, key, to)?,
1368        (LOG, key) => {
1369            let mut log: LogSettings = section(&file, LOG_SECTION)?;
1370            let said = log_change(&mut log, key, to)?;
1371            put(&mut file, LOG_SECTION, &log)?;
1372            said
1373        }
1374        (THEME, key) => {
1375            let mut chosen: Chosen = section(&file, Chosen::SECTION)?;
1376            let said = theme_change(&mut chosen, name, key, to)?;
1377            put(&mut file, Chosen::SECTION, &chosen)?;
1378            said
1379        }
1380        (MACROS, key) => macros_change(dir, (instance, name), key, to)?,
1381        (REWARDS, key) => rewards_change(dir, key, to)?,
1382        (fights::FIGHTS, key) => fights::change(&mut file, key, to)?,
1383        (TRAVEL, key) if travel::memory(key).is_some() => {
1384            let memory = travel::memory(key).ok_or_else(|| format!("No memory {key}."))?;
1385            memory_change(dir, (instance, name), memory, to)?
1386        }
1387        (TRAVEL, key) => {
1388            let mut settings: TravelSettings = section(&file, travel::SECTION)?;
1389            let said = settings.set(key, to)?;
1390            put(&mut file, travel::SECTION, &settings)?;
1391            said
1392        }
1393        (page, key) => return Err(format!("The {page} page has no setting {key}.")),
1394    };
1395    settings_store::save(dir, &file).map_err(|why| format!("Not saved: {why}"))?;
1396    Ok(Done { said, symbols })
1397}

1458/// The General page's history row `key` -- the shortest line kept, the
1459/// longest, or how many lines -- set to `to`, or back to its own.
1460fn history_change(file: &mut SettingsFile, key: &str, to: Option<&str>) -> Result<String, String> {
1461    let mut kept: Kept = section(file, Kept::SECTION)?;
1462    let count = |(least, most): (u16, u16), what: &str| -> Result<Option<u16>, String> {
1463        to.map(str::trim)
1464            .map(|typed| {
1465                typed
1466                    .parse::<u16>()
1467                    .ok()
1468                    .filter(|n| (least..=most).contains(n))
1469                    .ok_or_else(|| format!("`{typed}` is not {what} from {least} to {most}."))
1470            })
1471            .transpose()
1472    };
1473    let said = match key {
1474        HISTORY_MAX => {
1475            kept.max_length = count((0, Kept::LONGEST), "a count of characters")?;
1476            match kept.max_length() {
1477                None => "A line of any length is kept in the history.".to_owned(),
1478                Some(n) => {
1479                    format!("A line of more than {n} characters is not kept in the history.")
1480                }
1481            }
1482        }
1483        HISTORY_LINES => {
1484            kept.lines = count((Kept::FEWEST_LINES, Kept::MOST_LINES), "a number of lines")?;
1485            format!("The history keeps the newest {} lines.", kept.lines())
1486        }
1487        _ => {
1488            kept.min_length = count((0, 200), "a count of characters")?;
1489            match kept.min_length() {
1490                0 => "Every line typed is kept in the history.".to_owned(),
1491                n => format!("A line of fewer than {n} characters is not kept in the history."),
1492            }
1493        }
1494    };
1495    put(file, Kept::SECTION, &kept)?;
1496    Ok(said)
1497}

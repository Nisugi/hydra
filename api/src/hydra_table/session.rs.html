14/// A supervised session for `login` with `password`, not yet running: its
15/// session log open while sessions are recorded (`hydra_board::
16/// records_sessions`: on in a development build, the hub's Record switch,
17/// or `HYDRA_RECORD`), and `crit` for the combat-line parser.
18///
19/// # Errors
20///
21/// The data folder or the session log could not be made.
22pub fn supervised_with(
23    login: &Login,
24    password: String,
25    crit: Option<Arc<hydra_model::crit::CritTables>>,
26    rules: &hydra_session::stream_rules::StreamRules,
27) -> Result<SupervisedSession<LiveConnector>, String> {
28    let data = account::data_dir();
29    std::fs::create_dir_all(&data).map_err(|e| format!("{}: {e}", data.display()))?;
30
31    let mut session = SupervisedSession::new(LiveConnector::new(
32        login.clone(),
33        password,
34        data.join(PIN_FILENAME),
35    ))
36    // The character's facts and the learned menu dictionary, kept across
37    // restarts.
38    .with_store(data.clone());
39    if hydra_board::records_sessions(&data) {
40        let mut redactions = Redactions::new();
41        redactions.account_of(&login.account, &login.character);
42        let dir = hydra_net::log_dir().join(hydra_net::date_dir());
43        let sink = SessionSink::create(
44            &dir,
45            &format!("{}_{}", login.game_code, login.character),
46            &hydra_net::file_stamp(),
47            redactions,
48        )
49        .map_err(|e| format!("cannot open the session log in {}: {e}", dir.display()))?;
50        eprintln!(
51            "[hydra] recording {} on {} in {}",
52            login.character,
53            login.game_code,
54            dir.display()
55        );
56        session = session.with_sink(sink);
57    }
58    if let Some(crit) = crit {
59        session = session.with_crit_tables(crit);
60    }
61    session = session.with_stream_rules(rules.clone());
62    if hydra_board::keeps_player_log(&data) {
63        let roster_name = format!("{}:{}", login.game_code, login.character);
64        let settings = crate::pages::log_settings(&data, &roster_name);
65        session = session.with_player_log(
66            player_log(
67                &format!("{}:{}", login.game_code, login.character),
68                &settings,
69            ),
70            // Every feed a player reads: what is shown is the log
71            // window's choice alone.
72            Capture::default(),
73        );
74    }
75    Ok(session)
76}

78/// `character`'s (`GAME:Name`) player log, its writer running on this
79/// runtime, under the log folder's `player`; its closed days archived and
80/// its oldest removed first, as its `settings` say, off the runtime's
81/// workers. Must be called on the runtime.
82fn player_log(character: &str, settings: &LogSettings) -> PlayerLog {
83    let root = hydra_net::log_dir();
84    // Before the writer opens anything: a log kept before games were told
85    // apart goes to its game's folder first.
86    let adopted = writer::adopt_unscoped(&root, character);
87    let (log, sink) = PlayerLog::new();
88    // Said in the player's own log as well as here.
89    let said = match adopted {
90        Ok(writer::Adopted::LeftForPrime(old)) => Some(format!(
91            "Logs kept under this name before games were told apart are in {}, \
92             and are taken to be Prime's.",
93            old.display()
94        )),
95        Err(e) => Some(format!(
96            "The player log kept under this name alone could not be moved to its \
97             game's folder: {e}. It is tried again next time."
98        )),
99        Ok(_) => None,
100    };
101    if let Some(said) = said {
102        eprintln!("[hydra] {character}: {said}");
103        log.notice(&said);
104    }
105    // The log is written on a thread of its own, never on the threads the
106    // characters play on: a stalled disk holds the log, not the game.
107    // Counted until it ends, so Hydra waits for it before it exits.
108    let writer = writer::PlayerWriter::new(&root, character);
109    let live = hydra_net::behind::Live::new();
110    let spawned = std::thread::Builder::new()
111        .name("hydra-player-log".to_owned())
112        .spawn(move || {
113            let _live = live;
114            match tokio::runtime::Builder::new_current_thread()
115                .enable_time()
116                .build()
117            {
118                Ok(runtime) => runtime.block_on(writer.run(sink)),
119                Err(why) => eprintln!("[hydra] the player log has no thread to run on: {why}"),
120            }
121        });
122    if let Err(why) = spawned {
123        eprintln!("[hydra] {character}: the player log has no thread: {why}");
124    }
125    let character = character.to_owned();
126    let (choice, keep_days) = (
127        settings.archive.unwrap_or_default(),
128        settings.keep_days.unwrap_or(0),
129    );
130    let told = log.clone();
131    tokio::task::spawn_blocking(move || {
132        // Said in the player's own log, where its window shows it, as well
133        // as here: one bad file must not go unnoticed for good.
134        let say = |text: String| {
135            eprintln!("[hydra] {character}: {text}");
136            told.notice(&text);
137        };
138        if keep_days > 0 {
139            match retention::prune(&root, &character, keep_days, &hydra_net::date_dir()) {
140                Ok(pruned) => {
141                    for why in pruned.skipped {
142                        say(format!("An old player log was not removed: {why}"));
143                    }
144                }
145                Err(e) => say(format!("The oldest player logs were not removed: {e}")),
146            }
147        }
148        let now = hydra_net::eastern::now();
149        match archive::sweep(&root, &character, choice, now) {
150            Ok(swept) => {
151                for why in swept.failed {
152                    say(format!("Part of the player log was not archived: {why}"));
153                }
154            }
155            Err(e) => say(format!("The player log was not archived: {e}")),
156        }
157    });
158    log
159}

161/// The crit tables, for the combat-line parser: ~2,400 patterns, built once
162/// and shared; `None`, said, when they cannot be.
163#[must_use]
164pub fn crit_tables() -> Option<Arc<hydra_model::crit::CritTables>> {
165    match hydra_model::crit::CritTables::load() {
166        Ok(tables) => Some(Arc::new(tables)),
167        Err(e) => {
168            eprintln!("[hydra] crit tables DISABLED -- {e}");
169            None
170        }
171    }

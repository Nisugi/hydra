1//! `.push`: the phones paired for push to phone, a test banner sent to
2//! them, and one forgotten or allowed again. A phone is paired from
3//! HydraFE; a trigger's *Push to phone* is what sends to them in play.
4
5use std::sync::Arc;

10use crate::commands::Handler;

12/// What `.push help` says.
13const HELP: &[&str] = &[
14    "push                 the phones paired for push to phone",
15    "push test            a test banner sent to each of them",
16    "push forget <name>   that phone gets no more pushes, though it connects again",
17    "push allow <name>    a phone forgotten pairs again as it next connects",
18    "A phone pairs from HydraFE. A trigger sends to every paired phone when its Push to phone is set.",
19];
20
21/// What a `.push` line asks.
22#[derive(Clone, Debug, PartialEq, Eq)]
23enum Asked {
24    List,
25    Help,
26    Test,
27    Forget(String),
28    Allow(String),
29}
30
31fn parse(rest: &str) -> Result<Asked, String> {
32    let rest = rest.trim();
33    let (word, after) = rest
34        .split_once(char::is_whitespace)
35        .map_or((rest, ""), |(word, after)| (word, after.trim()));
36    match word.to_ascii_lowercase().as_str() {
37        "" => Ok(Asked::List),
38        "help" => Ok(Asked::Help),
39        "test" => Ok(Asked::Test),
40        "forget" if !after.is_empty() => Ok(Asked::Forget(after.to_owned())),
41        "forget" => Err("Push: forget which phone? push lists them.".to_owned()),
42        "allow" if !after.is_empty() => Ok(Asked::Allow(after.to_owned())),
43        "allow" => Err("Push: allow which phone? push lists them.".to_owned()),
44        _ => Err(format!(
45            "Push: `{word}` is not something it does; push help."
46        )),
47    }
48}

50/// The phones kept, a line each, and those forgotten.
51fn listed(kept: &hydra_board::push::Kept) -> Vec<String> {
52    let mut lines = if kept.phones.is_empty() {
53        vec!["Push: no phone is paired. Pair one from HydraFE.".to_owned()]
54    } else {
55        let mut lines = vec![format!(
56            "Push: {} phone(s), through {}.",
57            kept.phones.len(),
58            kept.relay.as_deref().unwrap_or(hydra_board::push::RELAY)
59        )];
60        lines.extend(kept.phones.iter().map(|phone| {
61            let sandbox = if phone.sandbox {
62                " (a build from Xcode)"
63            } else {
64                ""
65            };
66            format!("  {}{sandbox}", phone.name)
67        }));
68        lines
69    };
70    if !kept.forgotten.is_empty() {
71        let names: Vec<&str> = kept
72            .forgotten
73            .iter()
74            .map(|gone| gone.name.as_str())
75            .collect();
76        lines.push(format!(
77            "Forgotten, until push allow <name>: {}.",
78            names.join(", ")
79        ));
80    }
81    lines
82}

84/// What forgetting or allowing `name` came to, said.
85fn changed(asked: &Asked, data: &std::path::Path) -> String {
86    match asked {
87        Asked::Forget(name) => match hydra_board::push::forget_named(data, name) {
88            Ok(Some(name)) => {
89                format!("Push: {name} gets no more pushes; push allow {name} lets it pair again.")
90            }
91            Ok(None) => format!("Push: no phone is called {name}; push lists them."),
92            Err(why) => why,
93        },
94        Asked::Allow(name) => match hydra_board::push::allow(data, name) {
95            Ok(Some(name)) => format!("Push: {name} pairs again the next time it connects."),
96            Ok(None) => format!("Push: no phone forgotten is called {name}."),
97            Err(why) => why,
98        },
99        _ => String::new(),
100    }
101}

103pub(crate) fn family(handle: SessionHandle, roster_name: String) -> Handler {
104    Arc::new(move |line: &str, _| {
105        let line = line.trim();
106        let (word, rest) = line.split_once(char::is_whitespace).unwrap_or((line, ""));
107        if !word.eq_ignore_ascii_case("push") {
108            return None;
109        }
110        let data = crate::account::data_dir();
111        let say = |notice: Notice| handle.say(notice.answering());
112        match parse(rest) {
113            Ok(Asked::Help) => say(Notice::help_table(NoticeKind::Info, HELP)),
114            Ok(Asked::List) => say(Notice::prose(
115                NoticeKind::Info,
116                listed(&hydra_board::push::read(&data)),
117            )),
118            Ok(asked @ (Asked::Forget(_) | Asked::Allow(_))) => {
119                say(Notice::line(NoticeKind::Info, changed(&asked, &data)));
120            }
121            Ok(Asked::Test) => {
122                let kept = hydra_board::push::read(&data);
123                if kept.phones.is_empty() {
124                    say(Notice::line(
125                        NoticeKind::Error,
126                        "Push: no phone is paired. Pair one from HydraFE.",
127                    ));
128                    return Some(Claimed::Done);
129                }
130                let (game, name) = roster_name
131                    .split_once(':')
132                    .map_or(("", roster_name.as_str()), |(game, name)| (game, name));
133                let alert = hydra_board::push::Alert {
134                    title: name.to_owned(),
135                    body: "A test from Hydra: push to phone works.".to_owned(),
136                    character: name.to_owned(),
137                    game: game.to_owned(),
138                    desktop: String::new(),
139                };
140                say(Notice::line(
141                    NoticeKind::Info,
142                    format!("Push: a test sent to {} phone(s).", kept.phones.len()),
143                ));
144                let handle = handle.clone();
145                crate::commands::spawn(async move {
146                    for why in hydra_board::push::send(&data, alert).await {
147                        handle.say(Notice::line(NoticeKind::Warn, format!("Push: {why}")));
148                    }
149                });
150            }
151            Err(why) => say(Notice::line(NoticeKind::Error, why)),
152        }
153        Some(Claimed::Done)
154    })
155}

157#[cfg(test)]
158mod tests {
159    use super::*;
160
161    #[test]
162    fn each_word_is_read() {
163        assert_eq!(parse(""), Ok(Asked::List));
164        assert_eq!(parse("TEST"), Ok(Asked::Test));
165        assert_eq!(
166            parse("forget my iPhone"),
167            Ok(Asked::Forget("my iPhone".to_owned()))
168        );
169        assert!(parse("forget").is_err());
170        assert_eq!(parse("allow iPhone"), Ok(Asked::Allow("iPhone".to_owned())));
171        assert!(parse("loudly").is_err());
172    }
173
174    #[test]
175    fn no_phone_says_how_to_pair_one() {
176        let said = listed(&hydra_board::push::Kept::default());
177        assert!(said[0].contains("Pair one from HydraFE"), "{said:?}");
178    }
179
180    #[test]
181    fn a_forgotten_phone_is_listed_as_forgotten() {
182        let kept = hydra_board::push::Kept {
183            forgotten: vec![hydra_board::push::Forgotten {
184                name: "iPad".to_owned(),
185                key: String::new(),
186            }],
187            ..hydra_board::push::Kept::default()
188        };
189        let said = listed(&kept);
190        assert_eq!(
191            said.last().map(String::as_str),
192            Some("Forgotten, until push allow <name>: iPad.")
193        );
194    }

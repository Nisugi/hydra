1//! The session table behind the windows: every character this process runs,
2//! started on the host, seated in the windows, and the hub's requests
3//! answered.
4//!
5//! It starts a character -- from the command line, the hub's roster, or the
6//! hub's login -- and seats it, shown as its team or its own settings say:
7//! its play window, the web page, a terminal, or nothing; keeps the hub's
8//! roster cards current; and
9//! answers quit, remove, reconnect, forget, star, remember, list an
10//! account's characters, and shut down.
11//!
12//! A login reaching `Ready` puts its character on the roster; a password
13//! typed in the hub with its box ticked is kept in the OS keyring then, once
14//! the service has proved it, and never before.
15

48/// A character playing as Hydra restarts, to start again in the next
49/// process: by the name the roster finds it by, `GAME:Name`, and the team
50/// it was started with.
51#[derive(Clone, Debug, PartialEq, Eq)]
52pub struct Resume {
53    /// The character, as `GAME:Name`.
54    pub character: String,
55    /// The team it was started with, by name.
56    pub team: Option<String>,
57}
58
59impl Resume {
60    /// As the next process is told it: `GAME:Name`, and `=` and the team
61    /// after it when it has one. A character's name has no `=`, so the
62    /// first is where the team begins.
63    #[must_use]
64    pub fn arg(&self) -> String {
65        match &self.team {
66            Some(team) => format!("{}={team}", self.character),
67            None => self.character.clone(),
68        }
69    }
70
71    /// `arg` read back.
72    #[must_use]
73    pub fn read(arg: &str) -> Self {
74        match arg.split_once('=') {
75            Some((character, team)) => Self {
76                character: character.to_owned(),
77                team: Some(team.to_owned()),
78            },
79            None => Self {
80                character: arg.to_owned(),
81                team: None,
82            },
83        }
84    }
85}

129    fn with_host(sessions: Sessions, runtime: tokio::runtime::Handle, host: Host) -> Arc<Self> {
130        let dir = account::data_dir();
131        // The stream rules before any character starts: a window reads
132        // their file again as it changes.
133        if let Some(why) = hydra_board::stream_rules::load(&dir, &sessions.stream_rules()) {
134            eprintln!("{why}");
135        }
136        // The highlights too, so a Hydra without a window has them; and a
137        // trigger that moves its line, from an older file or an import, is
138        // made a stream rule here, whatever front end is on.
139        if let Some(why) = hydra_board::highlights::load(&dir, &sessions.highlights()) {
140            eprintln!("{why}");
141        }
142        // The lookout's trigger, once, so its stream sounds from the first.
143        if let Some(said) = hydra_board::lookout::add_trigger(&dir, &sessions.highlights()) {
144            eprintln!("[hydra] {said}");
145        }
146        if let Some(said) = hydra_board::stream_rules::fold_redirects(
147            &dir,
148            &sessions.highlights(),
149            &sessions.stream_rules(),
150        ) {
151            eprintln!("{said}");
152        }
153        Arc::new(Self {
154            host: tokio::sync::Mutex::new(host),
155            sessions,
156            runtime,
157            dir,
158            started: Mutex::default(),
159            crit: OnceLock::new(),
160            closing: AtomicBool::new(false),
161            lichs: crate::lich_command::Lichs::default(),
162            restart: Mutex::default(),
163        })
164    }

166    /// Every character running, for `.to` and `.all`: a closure over the
167    /// table, held weakly, so a session's command line does not keep the
168    /// table alive.
169    fn running(self: &Arc<Self>) -> crate::relay::Characters {
170        let table = Arc::downgrade(self);
171        Arc::new(move || {
172            let table = table.clone();
173            Box::pin(async move {
174                let Some(table) = table.upgrade() else {
175                    return Vec::new();
176                };
177                let host = table.host.lock().await;
178                host.sessions()
179                    .filter(|(_, hosted)| hosted.is_running())
180                    .map(|(_, hosted)| crate::relay::Running {
181                        game: hosted.who.game.clone(),
182                        name: hosted.who.character.clone(),
183                        handle: hosted.handle.clone(),
184                    })
185                    .collect()
186            })
187        })
188    }

202        self: &Arc<Self>,
203        login: Login,
204        password: String,
205        keep: bool,
206        team: Option<Joined>,

208        self.start_seated(login, password, keep, team, true).await

211    /// [`Table::start`], its play window kept only when `windowed`: a
212    /// character started for the web page has none until its card opens one.
213    async fn start_seated(
214        self: &Arc<Self>,
215        login: Login,
216        password: String,
217        keep: bool,
218        team: Option<Joined>,
219        windowed: bool,
220    ) -> Result<SessionId, String> {
221        let character = login.character.clone();
222        if self.closing.load(Ordering::SeqCst) {
223            return Err(format!("{character} not started: Hydra is shutting down."));
224        }
225        // A character that ended earlier leaves the table before it starts
226        // again, so the table never holds a dead one beside the live one.
227        for id in self.ended(&login).await {
228            self.take_off(id).await;
229        }
230        // Held from the check to the add: an account already in use is
231        // refused before its log is opened or its Lich started, and no other
232        // start can take the account between the two.
233        let mut host = self.host.lock().await;
234        // Checked again with the lock held: a shutdown that took every
235        // session meanwhile would otherwise be followed by this one.
236        if self.closing.load(Ordering::SeqCst) {
237            return Err(format!("{character} not started: Hydra is shutting down."));
238        }
239        host.may_add(&login.account)
240            .map_err(|e| format!("{character} not started: {e}"))?;
241        let crit = self
242            .crit
243            .get_or_init(|| hydra_mem::part(hydra_mem::Part::Combat, crate::session::crit_tables))
244            .clone();
245        let session = hydra_mem::part(hydra_mem::Part::Session, || {
246            crate::session::supervised_with(
247                &login,
248                password.clone(),
249                crit,
250                &self.sessions.stream_rules(),
251            )
252        })?;
253        let handle = session.handle();
254        crate::commands::install(
255            &handle,
256            &login,
257            self.running(),
258            &self.lichs,
259            self.sessions.stream_rules(),
260        );
261        give_typing(&handle, &login.game_code, &character);
262        let log = session.player_log();
263        // Before it runs, so the window misses none of its events.
264        let events = handle.subscribe();
265        // Its Lich, kept on, once the game has let it in; the window
266        // listens already, so it hears which text to show.
267        self.lichs
268            .once_ready(&handle, &format!("{}:{}", login.game_code, character));
269        let who = Who {
270            account: login.account.clone(),
271            character: character.clone(),
272            game: login.game_code.clone(),
273        };
274        // Run on this task's runtime, the table's.
275        let id = host
276            .add(who, session)
277            .map_err(|e| format!("{character} not started: {e}"))?;
278        drop(host);
279        let seat = Seat::start(
280            (id.0, &character, &login.game_code),
281            &handle,
282            events,
283            &self.runtime,
284            (
285                self.sessions.wake(),
286                self.sessions.highlights(),
287                self.sessions.merged(),
288            ),
289        );
290        if let Some(log) = log {
291            seat.keep_log(log);
292        }
293        // Before it is seated, so its windows open where the team keeps
294        // them.
295        if let Some(team) = &team {
296            seat.in_team(team.name.clone());
297        }
298        // Looked at before it is seated, which a terminal waiting for it
299        // sees at once. Started with no window, it is shown nowhere.
300        let placed = if windowed {
301            placed(&self.sessions, &self.dir, &login, team.as_ref())
302        } else {
303            (TeamMode::Headless, false)
304        };
305        seat_as(&self.sessions, seat, placed, || {
306            hydra_board::web_page_port(&self.dir)
307        });
308        self.started
309            .lock()
310            .unwrap_or_else(PoisonError::into_inner)
311            .insert(
312                id,
313                Started {
314                    login: login.clone(),
315                    team,
316                },
317            );
318        let table = Arc::clone(self);
319        self.runtime.spawn(async move {
320            table
321                .proven(&handle, &login, keep.then_some(password))
322                .await;
323        });
324        Ok(id)
325    }

327    /// Once `handle`'s login is `Ready`, so known right: `login` on the
328    /// roster, and `password` kept in the keyring when given.
329    async fn proven(&self, handle: &SessionHandle, login: &Login, password: Option<String>) {
330        if !until_ready(handle).await {
331            return;
332        }
333        if let Err(e) = roster::record(&self.dir, roster::Entry::of(login)) {
334            eprintln!("[{}] could not be put on the roster ({e})", login.character);
335        }
336        if let Some(password) = password {
337            match account::keep(&login.account, &password) {
338                Ok(()) => eprintln!("[login] kept in the OS keyring for {}", login.account),
339                Err(e) => eprintln!("[login] the OS keyring would not keep it ({e})"),
340            }
341        }
342        self.offer().await;
343    }

345    /// The sessions for `login`'s character that stopped on their own and
346    /// are still on the table.
347    async fn ended(&self, login: &Login) -> Vec<SessionId> {
348        self.host
349            .lock()
350            .await
351            .sessions()
352            .filter(|(_, hosted)| {
353                !hosted.is_running()
354                    && hosted.who.game.eq_ignore_ascii_case(&login.game_code)
355                    && hosted.who.character.eq_ignore_ascii_case(&login.character)
356            })
357            .map(|(id, _)| id)
358            .collect()
359    }

361    /// The hub asked for something; say what came of it.
362    pub async fn answer(self: &Arc<Self>, request: HubRequest) -> String {
363        let said = match request {
364            HubRequest::Add(name) => self.add(&name, true).await,
365            HubRequest::AddHeadless(name) => self.add(&name, false).await,
366            HubRequest::Login(login) => {
367                let typed = Login {
368                    character: login.character.trim().to_owned(),
369                    account: login.account.trim().to_owned(),
370                    game_code: login.game.trim().to_uppercase(),
371                };
372                let name = typed.character.clone();
373                match self
374                    .start(
375                        typed,
376                        login.password.expose().to_owned(),
377                        login.remember,
378                        None,
379                    )
380                    .await
381                {
382                    Ok(_) => format!("Logging {name} in."),
383                    Err(e) => e,
384                }
385            }
386            HubRequest::Remove(id) => match self.take_off(SessionId(id)).await {
387                Some(started) => format!("{} has quit.", started.login.character),
388                None => "That character is no longer on the table.".to_owned(),
389            },
390            HubRequest::Reconnect(id) => self.reconnect(SessionId(id)).await,
391            HubRequest::Forget(name) => forget(&self.dir, &name),
392            HubRequest::ForgetPassword(account) => forget_password(&account),
393            HubRequest::Favourite(name, star) => favourite(&self.dir, &name, star),
394            HubRequest::Nickname(name, nickname) => nickname_set(&self.dir, &name, &nickname),
395            HubRequest::Remember(saved) => remember(&self.dir, &saved),
396            HubRequest::Characters(account) => self.characters(account).await,
397            HubRequest::Settings(character) => {
398                return self.settings(&character);
399            }
400            HubRequest::Change(change) => {
401                let done = crate::pages::apply(&self.dir, &change);
402                if let Some(symbols) = done.symbols {
403                    self.new_symbols(&change.character, symbols).await;
404                }
405                let failed = self.settings(&change.character);
406                return if failed.is_empty() { done.said } else { failed };
407            }
408            HubRequest::SaveTeam { was, team } => {
409                teams::keep(&self.dir, was.as_deref(), &team).unwrap_or_else(|why| why)
410            }
411            HubRequest::ForgetTeam(name) => {
412                teams::forget(&self.dir, &name).unwrap_or_else(|e| e.to_string())
413            }
414            HubRequest::LaunchTeam(name) => self.launch_team(&name).await,
415            HubRequest::Shutdown => {
416                eprintln!("[hydra] shut down from the hub");
417                self.closing.store(true, Ordering::SeqCst);
418                self.sessions.close();
419                return "Hydra is shutting down: every character is quitting.".to_owned();
420            }
421            HubRequest::Restart { again } => {
422                let resume = if again {
423                    self.to_resume().await
424                } else {
425                    Vec::new()
426                };
427                eprintln!(
428                    "[hydra] restarting from the hub, {} character(s) to start again",
429                    resume.len()
430                );
431                *self.restart.lock().unwrap_or_else(PoisonError::into_inner) = Some(resume);
432                self.closing.store(true, Ordering::SeqCst);
433                self.sessions.close();
434                return "Hydra is restarting: every character is quitting.".to_owned();
435            }
436        };
437        self.offer().await;
438        said
439    }

453    /// `character` (`GAME:Name`), when it is running, marks Hydra's commands
454    /// and its Lich's with `symbols` from now: the menu says a change takes
455    /// effect at once.
456    async fn new_symbols(&self, character: &str, (symbol, lich): (char, char)) {
457        let Some((game, name)) = character.split_once(':') else {
458            return;
459        };
460        for (_, hosted) in self.host.lock().await.sessions() {
461            if hosted.who.game.eq_ignore_ascii_case(game)
462                && hosted.who.character.eq_ignore_ascii_case(name)
463            {
464                let _ = hosted.handle.set_symbol(symbol);
465                let _ = hosted.handle.set_lich_symbol(lich);
466            }
467        }
468    }

470    /// Start the roster character `name` means, with its kept password,
471    /// shown as its settings say: shown in its play window, that window
472    /// opens, the hub left hidden if the windows were closed. One already
473    /// playing is shown again instead. Without `windowed` (a start from the
474    /// phone, where nobody is at the computer) it is started shown nowhere,
475    /// and one already playing is left as it is; its card opens a window
476    /// all the same.
477    async fn add(self: &Arc<Self>, name: &str, windowed: bool) -> String {
478        let entry = match roster::find(&self.dir, name) {
479            Ok(Some(entry)) => entry,
480            Ok(None) => return format!("{name} is not on the roster."),
481            Err(e) => return e.to_string(),
482        };
483        let login = Login {
484            character: entry.character,
485            account: entry.account,
486            game_code: entry.game_code,
487        };
488        let shown = shown_as(&self.sessions, &self.dir, &login, None);
489        if let Some(seat) = self.seated(&login.game_code, &login.character) {
490            if !windowed {
491                return format!("{} is playing.", login.character);
492            }
493            let (shown, here) = placed(&self.sessions, &self.dir, &login, None);
494            self.show(&seat, shown, here);
495            return format!("Showing {}.", login.character);
496        }
497        let Some(password) = account::kept(&login.account) else {
498            return format!(
499                "No password is kept for {}; type it on its card.",
500                login.account
501            );
502        };
503        let character = login.character.clone();
504        match self
505            .start_seated(login, password, false, None, windowed)
506            .await
507        {
508            Ok(id) => {
509                if windowed && shown == TeamMode::Gui {
510                    self.sessions.open_windows(Open::Seat(id.0));
511                }
512                format!("Starting {character}.")
513            }
514            Err(e) => e,
515        }
516    }

518    /// The seat of `character` on `game`, while it plays.
519    fn seated(&self, game: &str, character: &str) -> Option<Arc<Seat>> {
520        self.sessions
521            .glance(std::time::Instant::now())
522            .seats
523            .into_iter()
524            .find(|seat| {
525                seat.lifecycle() != State::Closed
526                    && seat.game.eq_ignore_ascii_case(game)
527                    && seat.name.eq_ignore_ascii_case(character)
528            })
529    }

531    /// Show `seat` as `shown` says: shown in a terminal, one opened on it,
532    /// unless a terminal asked for it (`asked_here`) and shows it itself;
533    /// else its play window, in front, opened again if the player had
534    /// closed it, the windows opened if they were closed.
535    fn show(&self, seat: &Seat, shown: TeamMode, asked_here: bool) {
536        if shown == TeamMode::Tui {
537            if !asked_here {
538                self.sessions.open_windows(Open::Terminal(seat.id));
539            }
540        } else {
541            seat.keep_window(true);
542            self.sessions.open_windows(Open::Seat(seat.id));
543        }
544    }

546    /// Start the team `name`: each of its characters not already playing,
547    /// with its kept password, seated with the team so its windows go where
548    /// the team keeps them. A character already playing plays on, its
549    /// windows where they are. As the team's mode says: its play windows
550    /// opened, those of the members already playing shown again, the hub
551    /// left hidden if Hydra rests; none, headless; none, and the web page
552    /// served for them; or none, and a terminal opened on each, those
553    /// already playing too.
554    async fn launch_team(self: &Arc<Self>, name: &str) -> String {
555        let team = match teams::find(&self.dir, name) {
556            Ok(Some(team)) => team,
557            Ok(None) => return format!("No team is called {name}."),
558            Err(e) => return e.to_string(),
559        };
560        let (mut starting, mut playing, mut not) = (Vec::new(), Vec::new(), Vec::new());
561        for member in &team.members {
562            let entry = match roster::find(&self.dir, member) {
563                Ok(Some(entry)) => entry,
564                Ok(None) => {
565                    not.push(format!("{member} is not on the roster"));
566                    continue;
567                }
568                Err(e) => {
569                    not.push(e.to_string());
570                    continue;
571                }
572            };
573            if let Some(seat) = self.seated(&entry.game_code, &entry.character) {
574                if matches!(team.mode, TeamMode::Gui | TeamMode::Tui) {
575                    self.show(&seat, team.mode, false);
576                }
577                playing.push(entry.character);
578                continue;
579            }
580            let Some(password) = account::kept(&entry.account) else {
581                not.push(format!("no password is kept for {}", entry.account));
582                continue;
583            };
584            let login = Login {
585                character: entry.character,
586                account: entry.account,
587                game_code: entry.game_code,
588            };
589            let character = login.character.clone();
590            let joined = Joined {
591                name: team.name.clone(),
592                mode: team.mode,
593            };
594            match self.start(login, password, false, Some(joined)).await {
595                Ok(_) => starting.push(character),
596                Err(e) => not.push(e),
597            }
598        }
599        let mut said = Vec::new();
600        if !starting.is_empty() {
601            said.push(format!(
602                "Starting {} ({}): {}.",
603                team.name,
604                team.mode.said(),
605                starting.join(", ")
606            ));
607            match team.mode {
608                // Opened with Hydra resting, its play windows alone; with
609                // the windows open, each opens as it is seated.
610                TeamMode::Gui => self.sessions.open_windows(Open::PlayWindows),
611                // Served, and each terminal opened, as each was seated.
612                TeamMode::Web => said.push("The web page is served for them.".to_owned()),
613                TeamMode::Tui => said.push("A terminal opens on each.".to_owned()),
614                TeamMode::Headless => {}
615            }
616        }
617        if !playing.is_empty() {
618            said.push(format!("Playing already: {}.", playing.join(", ")));
619        }
620        if !not.is_empty() {
621            said.push(format!("Not started: {}.", not.join("; ")));
622        }
623        said.join(" ")
624    }

626    /// Start a character whose connection has ended, again.
627    async fn reconnect(self: &Arc<Self>, id: SessionId) -> String {
628        let running = self
629            .host
630            .lock()
631            .await
632            .get(id)

635            None => "That character is no longer on the table.".to_owned(),

658    /// there.
659    async fn take_off(&self, id: SessionId) -> Option<Started> {
660        let hosted = self.host.lock().await.take(id)?;
661        self.sessions.detach(id.0);
662        let character = hosted.who.character.clone();
663        let game = hosted.who.game.clone();
664        // The session quits first, through its Lich when one runs, so Lich

678    /// Ask the login service which characters `account` has on its game,
679    /// and hand the hub the list. Its password goes to the OS keyring only
680    /// when the player ticked the box, and only now the service has proved
681    /// it.
682    async fn characters(&self, account: Account) -> String {
683        let (name, game) = (account.account.trim(), account.game.trim());
684        let credentials = hydra_net::Credentials {
685            account: name,
686            password: account.password.expose(),
687            character: "",
688            game_code: game,
689        };
690        let pin = self.dir.join(hydra_net::PIN_FILENAME);
691        let listed = hydra_net::list_characters(credentials, &pin, |line| {
692            eprintln!("[login] {line}");
693        })
694        .await;
695        let characters = match listed {
696            Ok(characters) => characters,
697            Err(e) => return format!("{name}'s characters could not be listed: {e}"),
698        };
699        let kept = if account.remember {
700            match account::keep(name, account.password.expose()) {
701                Ok(()) => " Its password is kept.".to_owned(),
702                Err(e) => format!(" The OS keyring would not keep its password ({e})."),
703            }
704        } else {
705            String::new()
706        };
707        let said = match characters.len() {
708            1 => format!("{name} has one character on {game}.{kept}"),
709            n => format!("{name} has {n} characters on {game}.{kept}"),
710        };
711        self.sessions.characters(Listing {
712            account: name.to_owned(),
713            game: game.to_owned(),
714            characters,
715        });
716        said
717    }

785    /// Start again each character a restart named, with its kept password
786    /// and the team it was started with, as the team is kept now. One with
787    /// no password kept, or no longer on the roster, is said and left: the
788    /// hub offers it.
789    pub async fn resume(self: &Arc<Self>, again: Vec<Resume>) {
790        for resume in again {
791            let entry = match roster::find(&self.dir, &resume.character) {
792                Ok(Some(entry)) => entry,
793                Ok(None) => {
794                    eprintln!(
795                        "[hydra] {} is no longer on the roster: not started again",
796                        resume.character
797                    );
798                    continue;
799                }
800                Err(why) => {
801                    eprintln!("[hydra] {} not started again: {why}", resume.character);
802                    continue;
803                }
804            };
805            let Some(password) = account::kept(&entry.account) else {
806                eprintln!(
807                    "[hydra] no password is kept for {}: {} not started again",
808                    entry.account, entry.character
809                );
810                continue;
811            };
812            let joined = resume
813                .team
814                .and_then(|name| match teams::find(&self.dir, &name) {
815                    Ok(Some(team)) => Some(Joined {
816                        name: team.name,
817                        mode: team.mode,
818                    }),
819                    _ => None,
820                });
821            let login = Login {
822                character: entry.character,
823                account: entry.account,
824                game_code: entry.game_code,
825            };
826            let character = login.character.clone();
827            if let Err(why) = self.start(login, password, false, joined).await {
828                eprintln!("[hydra] {character} not started again: {why}");
829            }
830        }
831    }

833    /// Quit every character, in order.
834    pub async fn stop_everything(&self) {
835        self.closing.store(true, Ordering::SeqCst);
836        let all = {
837            let mut host = self.host.lock().await;
838            let ids: Vec<SessionId> = host.sessions().map(|(id, _)| id).collect();
839            for id in ids {
840                self.sessions.detach(id.0);
841            }
842            host.take_all()
843        };
844        for end in stop_all(all).await {
845            match end {
846                Some(end) => eprintln!("[hydra] session ended: {:?}", end.stopped_because),
847                None => eprintln!("[hydra] a session did not end cleanly"),
848            }
849        }
850        self.lichs.shutdown().await;
851        // What the recorders handed the ledger as their sessions ended.
852        crate::ledger::flush().await;
853    }

856/// How `login`'s character is shown as it starts: as `team`'s mode says
857/// when it starts with one; else as its own settings in `dir` say, when
858/// they choose; else in the terminal on `sessions` that asked for it, if
859/// one did; else as Hydra's hub is set: the windows, or a terminal.
860fn shown_as(
861    sessions: &Sessions,
862    dir: &std::path::Path,
863    login: &Login,
864    team: Option<&Joined>,
865) -> TeamMode {
866    if let Some(team) = team {
867        return team.mode;
868    }
869    let name = format!("{}:{}", login.game_code, login.character);
870    if let Some(own) = crate::pages::shown_in_own(dir, &name) {
871        return own;
872    }
873    let in_terminal = sessions.awaited_in_terminal(&login.game_code, &login.character)
874        || sessions.own().hub_in_terminal();
875    if in_terminal {
876        TeamMode::Tui
877    } else {
878        TeamMode::Gui
879    }
880}

882/// How `login`'s character is shown as it starts or is shown again, as
883/// [`shown_as`] says, and whether in the terminal that asked for it:
884/// shown in a terminal, asked for by one, and neither its own *Shown in*
885/// nor Hydra's own settings want it in a terminal of its own. Shown
886/// anywhere else, the terminal that asked is let go of it, and stays where
887/// it is.
888fn placed(
889    sessions: &Sessions,
890    dir: &std::path::Path,
891    login: &Login,
892    team: Option<&Joined>,
893) -> (TeamMode, bool) {
894    let shown = shown_as(sessions, dir, login, team);
895    let (game, character) = (&login.game_code, &login.character);
896    let asked = sessions.awaited_in_terminal(game, character);
897    let here = asked
898        && team.is_none()
899        && shown == TeamMode::Tui
900        && !crate::pages::own_terminal(dir, &format!("{game}:{character}"))
901        && !sessions.own().own_terminals();
902    if asked && !here {
903        sessions.found_in_terminal(game, character);
904    }
905    (shown, here)
906}

908/// Seat `seat`, a character starting, on `sessions`, shown as `shown` says:
909/// its play window kept for the windows only when shown in one; for the
910/// web page, the page served on the port `port` gives; in a terminal, one
911/// opened on it, unless a terminal asked for it (`asked_here`) and shows it
912/// itself.
913fn seat_as(
914    sessions: &Sessions,
915    seat: Arc<Seat>,
916    (shown, asked_here): (TeamMode, bool),
917    port: impl FnOnce() -> u16,
918) {
919    // Before it is seated, so its play window opens as it is, or not at all.
920    if shown != TeamMode::Gui {
921        seat.keep_window(false);
922    }
923    let id = seat.id;
924    sessions.attach(seat);
925    match shown {
926        TeamMode::Web => sessions.serve_web(port()),
927        TeamMode::Tui if !asked_here => sessions.open_windows(Open::Terminal(id)),
928        TeamMode::Tui | TeamMode::Gui | TeamMode::Headless => {}
929    }
930}

932/// Resolves `true` once `handle`'s session is `Ready`; `false` if it ended
933/// first.
934pub(crate) async fn until_ready(handle: &SessionHandle) -> bool {
935    let mut events = handle.subscribe();
936    if handle
937        .snapshot()
938        .await
939        .is_some_and(|snapshot| snapshot.lifecycle == State::Ready)
940    {
941        return true;
942    }
943    loop {
944        match events.recv().await {
945            Ok(Event::StateChanged(State::Ready)) => return true,
946            Ok(Event::StateChanged(State::Closed)) | Err(RecvError::Closed) => return false,
947            Ok(_) | Err(RecvError::Lagged(_)) => {}
948        }
949    }
950}

1//! `.survey`: the room survey on the command line (`hydra-survey`). One
2//! survey per character; `.survey stop` and `.stop` end it after the
3//! command under way, and `.survey resume` picks it up from its log.
4

17/// Where the run files go, under the data folder.
18const FOLDER: &str = "survey";
19
20/// Rooms between one progress line and the next.
21const TOLD_EVERY: usize = 10;
22
23/// `.survey help`.
24const HELP: &[&str] = &[
25    "survey start             start a new run from this room: every exit of every room, once each way",
26    "survey resume [<file>]   go on with the newest run of this character's, or that file",
27    "survey status            where it is: rooms, exits left, the last room entered",
28    "survey stop              stop after the command under way; resume picks it up",
29    "                         (the run is survey-<name>-<stamp>.jsonl in the data folder's survey/)",
30];
31
32/// What a running survey says of itself.
33#[derive(Clone, Debug, Default)]
34struct Status {
35    path: Option<PathBuf>,
36    rooms: usize,
37    untried: usize,
38    last: Option<(String, String)>,
39    notes: Vec<String>,
40}
41
42/// One character's survey under way.
43struct Running {
44    /// Which run this is: a run ending takes off its own entry alone, not
45    /// one started in its place while it was winding down.
46    id: u64,
47    stop: CancellationToken,
48    status: Arc<Mutex<Status>>,
49}
50
51type Surveys = Mutex<HashMap<String, Running>>;
52
53/// How many runs have started: each the next id.
54static STARTED: AtomicU64 = AtomicU64::new(0);
55
56/// Register a run for `key`: its id, or `None` while one runs already.
57fn register(key: &str, stop: &CancellationToken, status: &Arc<Mutex<Status>>) -> Option<u64> {
58    let mut running = surveys().lock().ok()?;
59    if running.contains_key(key) {
60        return None;
61    }
62    let id = STARTED.fetch_add(1, Ordering::Relaxed);
63    running.insert(
64        key.to_owned(),
65        Running {
66            id,
67            stop: stop.clone(),
68            status: Arc::clone(status),
69        },
70    );
71    Some(id)
72}
73
74/// Take the run `id` off `key`, if it is still the one registered.
75fn unregister(key: &str, id: u64) {
76    if let Ok(mut running) = surveys().lock()
77        && running.get(key).is_some_and(|running| running.id == id)
78    {
79        running.remove(key);
80    }
81}
82
83fn surveys() -> &'static Surveys {
84    static SURVEYS: OnceLock<Surveys> = OnceLock::new();
85    SURVEYS.get_or_init(Surveys::default)
86}

88/// Stop `roster_name`'s survey, if one runs: whether one did.
89pub(crate) fn stop_survey(roster_name: &str) -> bool {
90    let Ok(mut running) = surveys().lock() else {
91        return false;
92    };
93    running
94        .remove(&roster_name.to_lowercase())
95        .map(|running| running.stop.cancel())
96        .is_some()
97}

99/// The `.survey` family for the character the roster calls `roster_name`.
100pub(crate) fn family(handle: SessionHandle, roster_name: String) -> Handler {
101    Arc::new(move |line: &str, _| {
102        let line = line.trim();
103        let (word, rest) = line.split_once(char::is_whitespace).unwrap_or((line, ""));
104        if !word.eq_ignore_ascii_case("survey") {
105            return None;
106        }
107        let rest = rest.trim();
108        let (sub, arg) = rest.split_once(char::is_whitespace).unwrap_or((rest, ""));
109        let notice = match sub.to_lowercase().as_str() {
110            "" | "help" => Notice::table(
111                NoticeKind::Info,
112                HELP.iter().map(|&line| line.to_owned()).collect(),
113            ),
114            "start" => start(&handle, &roster_name, Resume::New),
115            "resume" if arg.is_empty() => start(&handle, &roster_name, Resume::Newest),
116            "resume" => start(
117                &handle,
118                &roster_name,
119                Resume::File(PathBuf::from(arg.trim())),
120            ),
121            "status" => status(&roster_name),
122            "stop" => Notice::line(
123                NoticeKind::Info,
124                if stop_survey(&roster_name) {
125                    "Survey: stopping after the command under way."
126                } else {
127                    "Survey: nothing is running."
128                },
129            ),
130            other => Notice::line(
131                NoticeKind::Error,
132                format!("Survey: I do not know survey {other}. survey help lists what it takes."),
133            ),
134        };
135        handle.say_unlogged(notice.answering());
136        Some(Claimed::Done)
137    })
138}

140/// Start a run, or resume one.
141fn start(handle: &SessionHandle, roster_name: &str, resume: Resume) -> Notice {
142    let (server, character) = match crate::pages::who(roster_name) {
143        Ok((_, name)) => (
144            roster_name.split(':').next().unwrap_or_default().to_owned(),
145            name.to_owned(),
146        ),
147        Err(why) => return Notice::line(NoticeKind::Error, format!("Survey: {why}")),
148    };
149    let key = roster_name.to_lowercase();
150    let stop = CancellationToken::new();
151    let status: Arc<Mutex<Status>> = Arc::default();
152    let Some(id) = register(&key, &stop, &status) else {
153        return Notice::line(
154            NoticeKind::Warn,
155            "Survey: one is running on this character; survey stop ends it.",
156        );
157    };
158    let setting = Setting {
159        dir: crate::account::data_dir().join(FOLDER),
160        server,
161        character,
162        resume: resume.clone(),
163    };
164    let tell = teller(handle.clone(), Arc::clone(&status));
165    let handle = handle.clone();
166    crate::commands::spawn(async move {
167        let ended = hydra_survey::survey(handle.clone(), setting, stop, tell).await;
168        unregister(&key, id);
169        let notice = match ended {
170            Ok(summary) => Notice::line(NoticeKind::Info, ended_said(&summary)),
171            Err(why) => Notice::line(NoticeKind::Error, format!("Survey: {why}")),
172        };
173        handle.say(notice.answering());
174    });
175    Notice::line(
176        NoticeKind::Info,
177        match resume {
178            Resume::New => "Survey: started from this room.".to_owned(),
179            Resume::Newest => "Survey: resuming the newest run.".to_owned(),
180            Resume::File(path) => format!("Survey: resuming {}.", path.display()),
181        },
182    )
183}

185/// How a run ended, in a line.
186fn ended_said(summary: &Summary) -> String {
187    let how = if summary.done {
188        "done: every exit tried"
189    } else if summary.untried > 0 {
190        "stopped with exits left (stuck, or told to); survey resume goes on"
191    } else {
192        "stopped"
193    };
194    format!(
195        "Survey {how}: {} rooms, {} edges, {} exits left, in {}.",
196        summary.rooms,
197        summary.edges,
198        summary.untried,
199        summary.path.display()
200    )
201}

203/// What the walk tells, kept for `.survey status` and said now and then.
204fn teller(handle: SessionHandle, status: Arc<Mutex<Status>>) -> hydra_survey::Teller {
205    Arc::new(move |progress: Progress| {
206        let said = {
207            let Ok(mut status) = status.lock() else {
208                return;
209            };
210            match progress {
211                Progress::Started { path, resumed } => {
212                    let said = format!(
213                        "Survey: {} {}.",
214                        if resumed { "resumed from" } else { "writing" },
215                        path.display()
216                    );
217                    status.path = Some(path);
218                    Some(said)
219                }
220                Progress::Room {
221                    sid,
222                    title,
223                    rooms,
224                    untried,
225                } => {
226                    status.rooms = rooms;
227                    status.untried = untried;
228                    let said = (rooms % TOLD_EVERY == 0).then(|| {
229                        format!("Survey: {rooms} rooms, {untried} exits left; at {title} ({sid}).")
230                    });
231                    status.last = Some((sid, title));
232                    said
233                }
234                Progress::Died => Some("Survey: dead; healme.".to_owned()),
235                Progress::Healed => Some("Survey: back, going on.".to_owned()),
236                Progress::Note(note) => {
237                    status.notes.push(note.clone());
238                    Some(format!("Survey: {note}."))
239                }
240            }
241        };
242        if let Some(said) = said {
243            handle.say(Notice::line(NoticeKind::Info, said).answering());
244        }
245    })
246}

248/// `.survey status`.
249fn status(roster_name: &str) -> Notice {
250    let Ok(running) = surveys().lock() else {
251        return Notice::line(NoticeKind::Error, "Survey: its record is poisoned.");
252    };
253    let Some(running) = running.get(&roster_name.to_lowercase()) else {
254        return Notice::line(NoticeKind::Info, "Survey: nothing is running.");
255    };
256    let Ok(status) = running.status.lock() else {
257        return Notice::line(NoticeKind::Error, "Survey: its status is poisoned.");
258    };
259    let mut lines = vec![format!(
260        "Survey: running; {} rooms, {} exits left.",
261        status.rooms, status.untried
262    )];
263    if let Some((sid, title)) = &status.last {
264        lines.push(format!("Last entered: {title} ({sid})."));
265    }
266    if let Some(path) = &status.path {
267        lines.push(format!("Writing {}.", path.display()));
268    }
269    for note in &status.notes {
270        lines.push(format!("Note: {note}."));
271    }
272    Notice::prose(NoticeKind::Info, lines)
273}

275#[cfg(test)]
276mod tests {
277    use super::*;
278
279    /// A run stopped and another started in its place: the stopped run,
280    /// ending after, takes off its own entry alone, and the new run stays
281    /// registered, so `survey stop` still ends it.
282    #[test]
283    fn a_run_ending_late_leaves_the_run_in_its_place() {
284        let key = "gs3:tester-late";
285        let first = register(key, &CancellationToken::new(), &Arc::default()).expect("first");
286        assert!(register(key, &CancellationToken::new(), &Arc::default()).is_none());
287        assert!(stop_survey(key));
288        let second = register(key, &CancellationToken::new(), &Arc::default()).expect("second");
289        unregister(key, first);
290        assert!(
291            surveys()
292                .lock()
293                .unwrap()
294                .get(key)
295                .is_some_and(|r| r.id == second),
296            "the second stays"
297        );
298        unregister(key, second);
299        assert!(!stop_survey(key), "gone");
300    }
301}

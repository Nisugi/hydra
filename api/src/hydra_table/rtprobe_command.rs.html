1//! `.rtprobe`: how near a roundtime's end Hydra sends the next command,
2//! and whether the game takes it. A tool for working on Hydra, left out of
3//! `.help`.
4//!
5//! Each round sends a command with a roundtime (`search`) or a cast time
6//! (`incant 601`), reads the second that ends, and then sends it again:
7//!
8//! - `wait`: through the wait a line [`Listener::ask`] makes, roundtime then
9//!   quiet spell, so the round measures how late that wait sends after the
10//!   tick the clock places for the end, and whether the game refused it.
11//! - `spam`: from a little before that tick, again and again until the game
12//!   takes it, so the round shows where the game's own gate opened against
13//!   the tick: a send taken before it means the clock places the tick late.
14//! - `burst`: from the moment the first command is answered, sent every so
15//!   many ms without waiting for answers, which come back one a send and in
16//!   order; counting them says which send each answer was. No pause, so
17//!   where the sending starts owes nothing to the tick the clock places, and
18//!   since the first command lands anywhere in its second, the rounds' sends
19//!   land at every offset from the tick, not on the same few: every send is
20//!   kept, and the gate is placed as finely as the rounds allow.
21//!
22//! The first command of a round goes out at a random point in its second:
23//! sent at once after the last roundtime, every round would land at the
24//! same point and look the same however the game counts.
25//!
26//! Every round is said in the story and written as a row of
27//! `rtprobe/<name>_<stamp>_<mode>.csv` in the log folder; the summary is
28//! said at the end.
29

42/// Rounds when none are given.
43const ROUNDS: u32 = 40;

45/// The most rounds one run takes.
46const MOST_ROUNDS: u32 = 500;

48/// How long an answer is waited for.
49const ANSWER: Duration = Duration::from_secs(3);

51/// How far before the tick `spam` starts sending.
52const SPAM_LEAD: Duration = Duration::from_millis(300);
53
54/// The pause between one `spam` send's answer and the next send.
55const SPAM_GAP: Duration = Duration::from_millis(50);

57/// The most sends `spam` makes for one round.
58const SPAM_MOST: u32 = 40;
59
60/// The spacing between `burst` sends when none is given, in ms.
61const BURST_GAP_MS: u64 = 50;
62
63/// The most sends `burst` makes for one round: a whole roundtime of them.
64const BURST_MOST: usize = 600;
65
66/// How long a send's roundtime or cast time is waited for after its
67/// first prompt.
68const STATED_WAIT: Duration = Duration::from_millis(1500);

70/// The most refusals one round of `wait` takes before it gives up.
71const WAIT_MOST: u32 = 5;

73/// `.rtprobe help`.
74const HELP: &[&str] = &[
75    "rtprobe wait <command> [rounds]   send it again through the wait a script's line makes",
76    "rtprobe spam <command> [rounds]   send it again and again from just before the tick",
77    "rtprobe burst <command> [rounds] [ms]   send it every ms (50) from just before the tick, unanswered",
78    "rtprobe stop                      stop after the round under way",
79    "                                  (search for roundtime, a cast such as incant 601 for cast time)",

83enum Mode {

90impl Said {
91    const fn word(self) -> &'static str {
92        match self {
93            Self::Wait => "wait",
94            Self::TypeAhead => "typeahead",
95            Self::Taken => "taken",
96        }
97    }

100impl Mode {
101    const fn word(self) -> &'static str {
102        match self {
103            Self::Wait => "wait",
104            Self::Spam => "spam",
105            Self::Burst(_) => "burst",
106        }
107    }
108}

110/// What one answer to a `burst` send was.
111#[derive(Clone, Copy, Debug, PartialEq, Eq)]
112enum Said {
113    /// `...wait N seconds.`: refused for roundtime.
114    Wait,
115    /// `Sorry, you may only type ahead…`: refused for typeahead.
116    TypeAhead,
117    /// `Roundtime:` or `Cast Roundtime`: taken.
118    Taken,
119}
120
121/// What `text` answers a send with, if it is an answer.
122fn said(text: &str) -> Option<Said> {
123    let text = text.trim_start();
124    if text.starts_with("...wait ") || text.starts_with("Wait ") {
125        Some(Said::Wait)
126    } else if text.starts_with("Sorry, you may only type ahead") {
127        Some(Said::TypeAhead)
128    } else if text.starts_with("Roundtime: ") || text.starts_with("Cast Roundtime ") {
129        Some(Said::Taken)
130    } else {
131        None
132    }
133}

135/// One round, its times in seconds from the run's start.
136#[derive(Clone, Debug, Default)]
137struct Round {
138    /// The server second the first command's roundtime ended.
139    end: u32,
140    /// The tick the clock placed for `end`, when the round's send went.
141    edge: f64,
142    /// `wait`: when the roundtime wait was over, before the quiet spell.
143    rt_over: Option<f64>,
144    /// When the send the game answered went.
145    sent: f64,
146    /// The last refused send before it.
147    last_refused: Option<f64>,
148    /// Sends made, the answered one counted.
149    sends: u32,
150    /// Refusals among them.
151    refused: u32,
152    /// `burst`: refusals for typeahead among them.
153    typeahead: u32,
154    /// `burst`: every send answered, when it went and what it was told.
155    tries: Vec<(f64, Said)>,
156    /// The game took the send: it stated a roundtime or a cast time.
157    taken: bool,
158    /// The seconds from the first end to the end the taken send stated.
159    stated: Option<u32>,
160}
161
162impl Round {
163    fn late(&self) -> f64 {
164        self.sent - self.edge
165    }
166}

175/// Stop `roster_name`'s probe, if one runs: whether one did.
176pub(crate) fn stop_probe(roster_name: &str) -> bool {
177    running()
178        .lock()
179        .ok()
180        .and_then(|mut running| running.remove(&roster_name.to_lowercase()))
181        .map(|stop| stop.cancel())
182        .is_some()
183}

185/// The `.rtprobe` family for the character the roster calls `roster_name`.
186pub(crate) fn family(handle: SessionHandle, roster_name: String) -> Handler {
187    Arc::new(move |line: &str, _| {
188        let line = line.trim();
189        let (word, rest) = line.split_once(char::is_whitespace).unwrap_or((line, ""));
190        if !word.eq_ignore_ascii_case("rtprobe") {
191            return None;
192        }
193        let words: Vec<&str> = rest.split_whitespace().collect();
194        let notice = match words.first().map(|w| w.to_lowercase()).as_deref() {
195            None | Some("help") => Notice::help_table(NoticeKind::Info, HELP),
196            Some("stop") => Notice::line(
197                NoticeKind::Info,
198                if stop_probe(&roster_name) {
199                    "Rtprobe: stopping after the round under way."
200                } else {
201                    "Rtprobe: nothing is running."
202                },
203            ),
204            Some("wait") => start(&handle, &roster_name, Mode::Wait, &words[1..]),
205            Some("spam") => start(&handle, &roster_name, Mode::Spam, &words[1..]),
206            Some("burst") => {
207                // The spacing is the second number at the end, after the
208                // rounds.
209                let mut args = words[1..].to_vec();
210                let gap = match args.as_slice() {
211                    [.., rounds, gap] if rounds.parse::<u32>().is_ok() => gap.parse::<u64>().ok(),
212                    _ => None,
213                };
214                if gap.is_some() {
215                    args.pop();
216                }
217                let gap = gap.unwrap_or(BURST_GAP_MS).clamp(10, 1000);
218                start(&handle, &roster_name, Mode::Burst(gap), &args)
219            }
220            Some(other) => Notice::line(
221                NoticeKind::Error,
222                format!(
223                    "Rtprobe: I do not know rtprobe {other}. rtprobe help lists what it takes."
224                ),
225            ),
226        };
227        handle.say(notice.answering());
228        Some(Claimed::Done)
229    })
230}

232/// Start a run of `mode` with `args`: the command, then the rounds.
233fn start(handle: &SessionHandle, roster_name: &str, mode: Mode, args: &[&str]) -> Notice {
234    let (rounds, command) = match args.split_last() {
235        Some((last, rest)) if !rest.is_empty() && last.parse::<u32>().is_ok() => {
236            (last.parse::<u32>().unwrap_or(ROUNDS), rest.join(" "))
237        }
238        _ => (ROUNDS, args.join(" ")),
239    };
240    if command.is_empty() {
241        return Notice::line(
242            NoticeKind::Error,
243            "Rtprobe: which command? rtprobe wait search 40, say.",
244        );
245    }
246    let rounds = rounds.clamp(1, MOST_ROUNDS);
247    let name = match crate::pages::who(roster_name) {
248        Ok((_, name)) => name.to_owned(),
249        Err(why) => return Notice::line(NoticeKind::Error, format!("Rtprobe: {why}")),
250    };
251    let key = roster_name.to_lowercase();
252    let stop = CancellationToken::new();
253    {
254        let Ok(mut running) = running().lock() else {
255            return Notice::line(NoticeKind::Error, "Rtprobe: its record is poisoned.");
256        };
257        if running.contains_key(&key) {
258            return Notice::line(
259                NoticeKind::Warn,
260                "Rtprobe: one is running on this character; rtprobe stop ends it.",
261            );
262        }
263        running.insert(key.clone(), stop.clone());
264    }
265    let path = hydra_net::log_dir().join("rtprobe").join(format!(
266        "{name}_{}_{}.csv",
267        hydra_net::file_stamp(),
268        mode.word()
269    ));
270    let started = match mode {
271        Mode::Burst(gap) => format!("Rtprobe: burst {command}, {rounds} rounds, {gap} ms apart."),
272        _ => format!("Rtprobe: {} {command}, {rounds} rounds.", mode.word()),
273    };
274    let handle = handle.clone();
275    crate::commands::spawn(async move {
276        let ended = run(&handle, mode, &command, rounds, &path, stop).await;
277        if let Ok(mut running) = running().lock() {
278            running.remove(&key);
279        }
280        let notice = match ended {
281            Ok(lines) => Notice::prose(NoticeKind::Info, lines),
282            Err(why) => Notice::line(NoticeKind::Error, format!("Rtprobe: {why}")),
283        };
284        handle.say(notice.answering());
285    });
286    Notice::line(NoticeKind::Info, started)
287}

289/// The run: its rounds, said and written; the summary's lines.
290async fn run(
291    handle: &SessionHandle,
292    mode: Mode,
293    command: &str,
294    rounds: u32,
295    path: &PathBuf,
296    stop: CancellationToken,
297) -> Result<Vec<String>, String> {
298    let mut listener = Listener::join(handle, stop.clone())
299        .await
300        .ok_or("the session has ended")?;
301    let start = Instant::now();
302    let mut done = Vec::new();
303    let mut misses = 0;
304    while done.len() < rounds as usize && !stop.is_cancelled() {
305        let round = match one_round(&mut listener, mode, command, start).await {
306            Ok(Some(round)) => round,
307            Ok(None) => {
308                misses += 1;
309                if misses >= 5 {
310                    return Err(format!(
311                        "{command} stated no roundtime or cast time five times; stopped."
312                    ));
313                }
314                continue;
315            }
316            Err(Lost::Stopped) => break,
317            Err(Lost::Connection) => return Err("the connection went.".to_owned()),
318        };
319        handle.say(Notice::line(NoticeKind::Info, round_said(done.len() + 1, &round)).answering());
320        done.push(round);
321        // A random point in the second for the next round's first command.
322        let roll = SystemTime::now()
323            .duration_since(UNIX_EPOCH)
324            .map_or(0, |t| t.subsec_millis());
325        if listener.out_of_roundtime().await.is_err()
326            || listener
327                .hold(Duration::from_millis(u64::from(200 + roll)))
328                .await
329                .is_err()
330        {
331            break;
332        }
333    }
334    let written = write_csv(path, mode, &done);
335    if matches!(mode, Mode::Burst(_)) {
336        let _ = write_sends(path, &done);
337    }
338    let mut lines = summary(mode, command, &done);
339    lines.push(match written {
340        Ok(()) => format!("Rounds written to {}.", path.display()),
341        Err(why) => format!(
342            "The rounds could not be written to {}: {why}.",
343            path.display()
344        ),
345    });
346    Ok(lines)
347}

349/// The roundtime's end and the cast time's, as the state holds them.
350fn ends(listener: &Listener) -> (Option<u32>, Option<u32>) {
351    let state = listener.state();
352    (state.roundtime_ends, state.cast_time_ends)
353}

355/// The end a command stated: the roundtime's or the cast time's that
356/// changed from `before`, waited for up to [`STATED_WAIT`] since a cast
357/// states its cast time in a second burst. `None` when neither changed.
358async fn stated_end(
359    listener: &mut Listener,
360    before: (Option<u32>, Option<u32>),
361) -> Result<Option<u32>, Lost> {
362    let until = Instant::now() + STATED_WAIT;
363    loop {
364        let (hard, cast) = ends(listener);
365        if hard != before.0 {
366            return Ok(hard);
367        }
368        if cast != before.1 {
369            return Ok(cast);
370        }
371        if Instant::now() >= until {
372            return Ok(None);
373        }
374        listener.hold(Duration::from_millis(50)).await?;
375    }
376}

378/// One round: the first command sent and the end it stated read, then the
379/// second as `mode` sends it. `None` when the first stated no end.
380async fn one_round(
381    listener: &mut Listener,
382    mode: Mode,
383    command: &str,
384    start: Instant,
385) -> Result<Option<Round>, Lost> {
386    let before = ends(listener);
387    if listener
388        .first_answer(command, Origin::Survey, ANSWER)
389        .await?
390        .is_none()
391    {
392        return Ok(None);
393    }
394    let Some(end) = stated_end(listener, before).await? else {
395        return Ok(None);
396    };
397    let at = |instant: Instant| {
398        instant.saturating_duration_since(start).as_secs_f64()
399            - start.saturating_duration_since(instant).as_secs_f64()
400    };
401    let edge = |listener: &Listener| listener.state().instant_of(end).map_or(f64::NAN, at);
402    let mut round = Round {
403        end,
404        ..Round::default()
405    };
406    match mode {
407        Mode::Wait => loop {
408            listener.out_of_roundtime().await?;
409            round.rt_over = Some(at(Instant::now()));
410            listener.quiet().await?;
411            round.sent = at(Instant::now());
412            round.edge = edge(listener);
413            round.sends += 1;
414            let before = ends(listener);
415            let Some(reply) = listener
416                .first_answer(command, Origin::Survey, ANSWER)
417                .await?
418            else {
419                break;
420            };
421            if refused(&reply) && round.refused < WAIT_MOST {
422                round.refused += 1;
423                round.last_refused = Some(round.sent);
424                continue;
425            }
426            round.stated = stated_end(listener, before)
427                .await?
428                .map(|e| e.saturating_sub(end));
429            round.taken = round.stated.is_some();
430            break;
431        },
432        Mode::Spam => {
433            if let Some(tick) = listener.state().instant_of(end) {
434                let from = tick.checked_sub(SPAM_LEAD).unwrap_or(tick);
435                listener
436                    .hold(from.saturating_duration_since(Instant::now()))
437                    .await?;
438            }
439            while round.sends < SPAM_MOST {
440                let sent = at(Instant::now());
441                round.sends += 1;
442                let before = ends(listener);
443                let Some(reply) = listener
444                    .first_answer(command, Origin::Survey, ANSWER)
445                    .await?
446                else {
447                    break;
448                };
449                if refused(&reply) {
450                    round.refused += 1;
451                    round.last_refused = Some(sent);
452                    listener.hold(SPAM_GAP).await?;
453                    continue;
454                }
455                round.sent = sent;
456                round.edge = edge(listener);
457                round.stated = stated_end(listener, before)
458                    .await?
459                    .map(|e| e.saturating_sub(end));
460                round.taken = round.stated.is_some();
461                break;
462            }
463        }
464        Mode::Burst(gap) => burst(listener, command, (end, gap), &mut round, at).await?,
465    }
466    Ok(Some(round))
467}

469/// A `burst` round: from now, `command` sent every `gap` ms until an
470/// answer says one was taken, then every answer collected. Answers come one a send, in order, so the one taken
471/// is the send its answer counts to. A round whose answers do not count
472/// up to its sends is not taken.
473async fn burst(
474    listener: &mut Listener,
475    command: &str,
476    (end, gap): (u32, u64),
477    round: &mut Round,
478    at: impl Fn(Instant) -> f64,
479) -> Result<(), Lost> {
480    let mut sent: Vec<f64> = Vec::new();
481    let mut answers: Vec<Said> = Vec::new();
482    let heard = |lines: Vec<hydra_model::state::lines::ChunkLine>, answers: &mut Vec<Said>| {
483        answers.extend(lines.iter().filter_map(|line| said(&line.text())));
484    };
485    while sent.len() < BURST_MOST && !answers.contains(&Said::Taken) {
486        sent.push(at(Instant::now()));
487        listener.send(command, Origin::Survey).await?;
488        let lines = listener
489            .lines_until(Instant::now() + Duration::from_millis(gap))
490            .await?;
491        heard(lines, &mut answers);
492    }
493    let until = Instant::now() + ANSWER;
494    while answers.len() < sent.len() && Instant::now() < until {
495        let lines = listener
496            .lines_until((Instant::now() + Duration::from_millis(100)).min(until))
497            .await?;
498        heard(lines, &mut answers);
499    }
500    round.edge = listener.state().instant_of(end).map_or(f64::NAN, &at);
501    round.sends = u32::try_from(sent.len()).unwrap_or(u32::MAX);
502    let Some(taken) = answers.iter().position(|answer| *answer == Said::Taken) else {
503        return Ok(());
504    };
505    if answers.len() != sent.len() {
506        return Ok(());
507    }
508    round.tries = sent.iter().copied().zip(answers.iter().copied()).collect();
509    let before = &answers[..taken];
510    round.refused =
511        u32::try_from(before.iter().filter(|a| **a == Said::Wait).count()).unwrap_or(u32::MAX);
512    round.typeahead =
513        u32::try_from(before.iter().filter(|a| **a == Said::TypeAhead).count()).unwrap_or(u32::MAX);
514    round.last_refused = before
515        .iter()
516        .rposition(|answer| *answer == Said::Wait)
517        .map(|index| sent[index]);
518    round.sent = sent[taken];
519    round.taken = true;
520    Ok(())
521}

523/// The game refused the line for roundtime: `...wait 2 seconds.`,
524/// `Wait 1 sec.`.
525fn refused(reply: &Reply) -> bool {
526    reply.texts().iter().any(|text| {
527        let text = text.trim_start();
528        text.starts_with("...wait ") || text.starts_with("Wait ")
529    })

532/// One round in a line.
533fn round_said(number: usize, round: &Round) -> String {
534    let mut said = format!(
535        "Rtprobe round {number}: {}, sent {:+.3} s from the tick",
536        if round.taken { "taken" } else { "not taken" },
537        round.late()
538    );
539    if let Some(over) = round.rt_over {
540        let _ = write!(
541            said,
542            " (roundtime wait over at {:+.3} s)",
543            over - round.edge
544        );
545    }
546    if round.refused > 0 {
547        let _ = write!(said, "; {} refused of {} sent", round.refused, round.sends);
548    }
549    if round.typeahead > 0 {
550        let _ = write!(said, ", {} for typeahead", round.typeahead);
551    }
552    if let Some(last) = round.last_refused {
553        let _ = write!(said, ", the last at {:+.3} s", last - round.edge);
554    }
555    said.push('.');
556    said
557}

559/// The least, middle, ninetieth and most of `values`, in a line.
560fn spread(values: &mut [f64]) -> String {
561    values.sort_by(f64::total_cmp);
562    let Some((&least, &most)) = values.first().zip(values.last()) else {
563        return "none".to_owned();
564    };
565    // The value `percent` of the way along, to the nearest.
566    let at = |percent: usize| values[((values.len() - 1) * percent + 50) / 100];
567    format!(
568        "least {least:+.3}, median {:+.3}, 90th {:+.3}, most {most:+.3} s",
569        at(50),
570        at(90)
571    )
572}

574/// The run's summary.
575fn summary(mode: Mode, command: &str, rounds: &[Round]) -> Vec<String> {
576    let taken = rounds.iter().filter(|r| r.taken).count();
577    let sends: u32 = rounds.iter().map(|r| r.sends).sum();
578    let refused: u32 = rounds.iter().map(|r| r.refused).sum();
579    let mut lines = vec![format!(
580        "Rtprobe {} {command}: {} rounds, {taken} taken; {sends} sends, {refused} refused.",
581        mode.word(),
582        rounds.len()
583    )];
584    let mut late: Vec<f64> = rounds.iter().filter(|r| r.taken).map(Round::late).collect();
585    match mode {
586        Mode::Wait => {
587            lines.push(format!("Sent after the tick: {}.", spread(&mut late)));
588            let mut over: Vec<f64> = rounds
589                .iter()
590                .filter_map(|r| r.rt_over.map(|over| over - r.edge))
591                .collect();
592            lines.push(format!(
593                "Roundtime wait over after the tick: {}.",
594                spread(&mut over)
595            ));
596            lines.push(format!(
597                "Refused per thousand sends: {:.1}.",
598                if sends == 0 {
599                    0.0
600                } else {
601                    f64::from(refused) * 1000.0 / f64::from(sends)
602                }
603            ));
604        }
605        Mode::Spam | Mode::Burst(_) => {
606            if let Mode::Burst(gap) = mode {
607                let typeahead: u32 = rounds.iter().map(|r| r.typeahead).sum();
608                lines.push(format!(
609                    "Sent {gap} ms apart; {typeahead} refused for typeahead before a taken send."
610                ));
611                lines.push(gate_said(rounds));
612            }
613            let early = late.iter().filter(|late| **late < 0.0).count();
614            lines.push(format!(
615                "Taken sends against the tick: {}.",
616                spread(&mut late)
617            ));
618            let mut last: Vec<f64> = rounds
619                .iter()
620                .filter_map(|r| r.last_refused.map(|last| last - r.edge))
621                .collect();
622            lines.push(format!(
623                "Last refused send against the tick: {}.",
624                spread(&mut last)
625            ));
626            lines.push(format!(
627                "Taken before the tick: {early} of {taken} (none means the clock never places it late)."
628            ));
629        }
630    }
631    lines
632}

634/// Where the gate opened, from every `burst` send of every round against
635/// its round's tick: the latest send refused for roundtime and the earliest
636/// taken, and how many of each fell on the wrong side of the other.
637fn gate_said(rounds: &[Round]) -> String {
638    let offsets = |wanted: Said| -> Vec<f64> {
639        rounds
640            .iter()
641            .flat_map(|r| {
642                r.tries
643                    .iter()
644                    .filter(move |(_, said)| *said == wanted)
645                    .map(move |(at, _)| at - r.edge)
646            })
647            .collect()
648    };
649    let (waits, taken) = (offsets(Said::Wait), offsets(Said::Taken));
650    let latest_wait = waits.iter().copied().fold(f64::NEG_INFINITY, f64::max);
651    let earliest_taken = taken.iter().copied().fold(f64::INFINITY, f64::min);
652    if waits.is_empty() || taken.is_empty() {
653        return "The gate: not enough sends either side to place it.".to_owned();
654    }
655    let crossed = waits.iter().filter(|at| **at > earliest_taken).count()
656        + taken.iter().filter(|at| **at < latest_wait).count();
657    format!(
658        "The gate, from {} sends: latest refused {latest_wait:+.3} s, earliest taken \
659         {earliest_taken:+.3} s against the tick; {crossed} on the wrong side of the other.",
660        waits.len() + taken.len()
661    )
662}

664/// Every `burst` send as CSV beside `path`, as `<name>.sends.csv`: its
665/// round, its offset from the tick, and its answer.
666fn write_sends(path: &std::path::Path, rounds: &[Round]) -> std::io::Result<()> {
667    let mut csv = String::from("round,offset,answer\n");
668    for (number, round) in rounds.iter().enumerate() {
669        for (at, said) in &round.tries {
670            let _ = writeln!(csv, "{},{:.3},{}", number + 1, at - round.edge, said.word());
671        }
672    }
673    std::fs::write(path.with_extension("sends.csv"), csv)
674}
675
676/// The rounds as CSV at `path`.
677fn write_csv(path: &PathBuf, mode: Mode, rounds: &[Round]) -> std::io::Result<()> {
678    if let Some(dir) = path.parent() {
679        std::fs::create_dir_all(dir)?;
680    }
681    let mut csv = String::from(
682        "mode,round,end,edge,rt_over,sent,late,last_refused,sends,refused,typeahead,taken,stated\n",
683    );
684    let opt = |value: Option<f64>| value.map(|v| format!("{v:.3}")).unwrap_or_default();
685    for (number, round) in rounds.iter().enumerate() {
686        let _ = writeln!(
687            csv,
688            "{},{},{},{:.3},{},{:.3},{:.3},{},{},{},{},{},{}",
689            mode.word(),
690            number + 1,
691            round.end,
692            round.edge,
693            opt(round.rt_over),
694            round.sent,
695            round.late(),
696            opt(round.last_refused),
697            round.sends,
698            round.refused,
699            round.typeahead,
700            round.taken,
701            round.stated.map(|rt| rt.to_string()).unwrap_or_default()
702        );
703    }
704    std::fs::write(path, csv)
705}

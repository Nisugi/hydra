1//! Hydra's own command line: every line that starts with the command symbol,
2//! `.` (`hydra_session::claimant`).
3//!
4//! **Installed when the session is built, before it connects**: the command
5//! line is there from the start, not something the player turns on.
6//!
7//! The symbol decides and the word routes: a line marked as Hydra's goes to
8//! the first family that knows its word, and a word nobody knows is answered
9//! "I do not know" by the session. It never reaches the game either way.
10//!
11//! The families so far: `help`, `version`, `history`, `theme`, `targetid`,
12//! `sorter`, `alias`, `highlight`, `to` and `all` (`relay`), `lich`, `go2` and `route2`
13//! (`travel_command`), `quick`, `timer`, `inventory`, `bestiary`, `exp`, `rewards`, `cstats`,
14//! `lookout`, and `stop`.
15

21/// Whether the room survey is offered: `.survey`, in a build with the
22/// `survey` feature. Its code is built either way.
23pub const SURVEY: bool = cfg!(feature = "survey");

25/// The line over `.help`'s groups.
26pub const HELP_FIRST: &str = "Hydra's commands start with . and none of them reaches the game. \
27                              Click one to see its own help.";

29/// `lines`, a family's answer, as a table; in help's two columns when they
30/// are its `help`, which they begin with.
31#[must_use]
32pub(crate) fn table_or_help(lines: Vec<String>, help: &[&str]) -> Notice {
33    let is_help = help
34        .first()
35        .is_some_and(|first| lines.first().map(String::as_str) == Some(*first));
36    if is_help {
37        Notice::help_table(NoticeKind::Info, &lines)
38    } else {
39        Notice::table(NoticeKind::Info, lines)
40    }
41}

183/// `.help`'s commands as this build offers them: the survey's only where it
184/// is offered.
185fn help_entries() -> impl Iterator<Item = (&'static str, &'static Entry)> {
186    HELP.iter()
187        .flat_map(|(heading, entries)| entries.iter().map(move |entry| (*heading, entry)))
188        .filter(|(_, (command, ..))| SURVEY || !command.starts_with("survey "))
189}

191/// `.help` as it is said: the line over it, then each group under its
192/// heading.
193#[must_use]
194pub fn help_notice() -> Notice {
195    use hydra_session::notice::HelpPart;
196    let mut parts = vec![HelpPart::Note(HELP_FIRST.to_owned())];
197    let mut heading = "";
198    for (group, (command, runs, says)) in help_entries() {
199        if group != heading {
200            heading = group;
201            parts.push(HelpPart::Heading(group.to_owned()));
202        }
203        parts.push(HelpPart::Entry {
204            command: (*command).to_owned(),
205            runs: runs.map(str::to_owned),
206            says: (*says).to_owned(),
207        });
208    }
209    Notice::help(NoticeKind::Info, parts)
210}

212/// A family of commands: `Some` when the line was its own, `None` when the
213/// word is not one it knows.

216/// Run a command's work as a task of its own, what it holds counted as the
217/// commands' in a build that measures (`hydra_mem`).
218pub fn spawn<F>(work: F) -> tokio::task::JoinHandle<F::Output>
219where
220    F: Future + Send + 'static,
221    F::Output: Send + 'static,
222{
223    tokio::spawn(hydra_mem::task(hydra_mem::Part::Commands, work))
224}

226/// Install the command line on `handle`, for `login`'s character, with the
227/// symbol its settings choose; `.to` and `.all` reach the characters
228/// `characters` lists, and `.lich` runs its Lich among `lichs`.
229pub fn install(
230    handle: &SessionHandle,
231    login: &crate::account::Login,
232    characters: crate::relay::Characters,
233    lichs: &crate::lich_command::Lichs,
234    rules: hydra_session::stream_rules::StreamRules,
235) {
236    let roster_name = format!("{}:{}", login.game_code, login.character);
237    let (symbol, lich) = crate::pages::symbols(&crate::account::data_dir(), &roster_name);
238    let mut families: Vec<Handler> = vec![
239        help(handle.clone()),
240        version(handle.clone()),
241        crate::history::family(handle.clone(), roster_name.clone(), rules),
242        crate::theme_command::family(handle.clone(), roster_name.clone()),
243        crate::targetid_command::family(handle.clone(), roster_name.clone()),
244        crate::sorter_command::family(handle.clone(), roster_name.clone()),
245        crate::alias_command::family(handle.clone(), roster_name.clone()),
246        crate::caster_command::family(handle.clone(), roster_name.clone()),
247        crate::highlight_command::family(handle.clone()),
248        crate::relay::family(handle.clone(), Arc::clone(&characters)),
249        crate::lich_command::family(handle.clone(), roster_name.clone(), lichs.clone()),
250        crate::travel_command::family(handle.clone(), roster_name.clone(), characters),
251        crate::quick_command::family(handle.clone()),
252        crate::hotbar_command::family(),
253        crate::note_command::family(handle.clone(), roster_name.clone()),
254        crate::macro_command::family(handle.clone(), roster_name.clone()),
255        crate::timer_command::family(handle.clone(), roster_name.clone()),
256        crate::every_command::family(handle.clone(), roster_name.clone()),
257        crate::inventory_command::family(handle.clone()),
258        crate::bestiary_command::family(handle.clone()),
259        crate::exp_command::family(handle.clone(), roster_name.clone()),
260        crate::rewards_command::family(handle.clone(), roster_name.clone()),
261        crate::sync::family(handle.clone(), crate::account::data_dir()),
262        crate::cstats_command::family(handle.clone(), roster_name.clone()),
263        crate::lookout_command::family(handle.clone(), roster_name.clone()),
264        crate::push_command::family(handle.clone(), roster_name.clone()),
265        stop(handle.clone(), roster_name.clone()),
266    ];
267    if SURVEY {
268        families.push(crate::survey_command::family(handle.clone(), roster_name));
269    }
270    let runner: Runner = Arc::new(move |line: &str, origin: Origin| {
271        families
272            .iter()
273            .find_map(|family| family(line, origin))
274            .unwrap_or(Claimed::Unknown)
275    });
276    let desk = Desk::new(Some(symbol), runner).with_words(words());
277    desk.set_lich_symbol(lich);
278    // Once, as the session is built: nothing else installs one.
279    let _ = handle.set_desk(desk);
280}

282/// Every command's first word, as `.help` lists them, with the words that
283/// stand in for one: what the window completes after the symbol.
284pub fn words() -> Vec<String> {
285    let mut words: Vec<String> = help_entries()
286        .filter_map(|(_, (command, ..))| command.split_whitespace().next())
287        .map(str::to_owned)
288        .collect();
289    words.extend(
290        ["route2", "sail", "highlight", "ver", "inv", "ecaster"]
291            .iter()
292            .map(|&word| word.to_owned()),
293    );
294    words.sort();
295    words.dedup();
296    words
297}

299/// `.help` and `.?`: the families, a line each.
300fn help(handle: SessionHandle) -> Handler {
301    Arc::new(move |line: &str, _| {
302        let line = line.trim();
303        if !(line.eq_ignore_ascii_case("help") || line == "?") {
304            return None;
305        }
306        handle.say(help_notice().answering());
307        Some(Claimed::Done)
308    })
309}

311/// Which Hydra this is: its version, and the commit it was built from when
312/// git could say, `+` after it when the build had changes not committed.
313#[must_use]
314pub fn version_said() -> String {
315    let version = crate::VERSION;
316    match env!("HYDRA_COMMIT") {
317        "" => format!("Hydra {version}"),
318        commit => format!("Hydra {version} ({commit})"),
319    }
320}
321
322/// `.version` and `.ver`: which Hydra this is.
323fn version(handle: SessionHandle) -> Handler {
324    Arc::new(move |line: &str, _| {
325        let line = line.trim();
326        if !(line.eq_ignore_ascii_case("version") || line.eq_ignore_ascii_case("ver")) {
327            return None;
328        }
329        handle.say(Notice::line(NoticeKind::Info, version_said()).answering());
330        Some(Claimed::Done)
331    })
332}

334/// `.stop`: everything on this character stops: its scripts are killed and
335/// its queue cleared. Its Lich is handed its own kill-all,
336/// `ka`, which spares the scripts Lich protects; the commands still waiting
337/// to be sent are dropped, and a walk under way stops. The one already
338/// written waits for its answer.
339fn stop(handle: SessionHandle, roster_name: String) -> Handler {
340    Arc::new(move |line: &str, _| {
341        if !line.trim().eq_ignore_ascii_case("stop") {
342            return None;
343        }
344        let walking = crate::travel_command::stop_walk(&roster_name);
345        let surveying = crate::survey_command::stop_survey(&roster_name);
346        let repeating = crate::every_command::stop_every(&roster_name);
347        let kill = format!("{}ka", handle.lich_symbol());
348        let killed = handle.to_lich(&kill);
349        let handle = handle.clone();
350        crate::commands::spawn(async move {
351            let dropped = handle.clear_queue().await;
352            let lich = if killed {
353                format!("Lich's scripts are stopped ({kill})")
354            } else if handle.lich_running() {
355                // It runs, and has not read what was typed for it: the
356                // kill-all could not be handed, and its scripts run on.
357                format!(
358                    "Lich is not reading what is typed for it, so its scripts could not be \
359                     stopped ({kill} not handed; .lich off stops Lich itself)"
360                )
361            } else {
362                "No Lich runs".to_owned()
363            };
364            let waiting = match dropped {
365                0 => "nothing was waiting to be sent".to_owned(),
366                1 => "1 command waiting to be sent is dropped".to_owned(),
367                n => format!("{n} commands waiting to be sent are dropped"),
368            };
369            let walk = if walking {
370                " The walk under way is stopped."
371            } else {
372                ""
373            };
374            let survey = if surveying {
375                " The survey is stopped."
376            } else {
377                ""
378            };
379            let every = if repeating {
380                " The lines sent again and again (every) are stopped."
381            } else {
382                ""
383            };
384            handle.say(
385                Notice::line(
386                    NoticeKind::Info,
387                    format!("Stopped. {lich}; {waiting}.{walk}{survey}{every}"),
388                )
389                .answering(),
390            );
391        });
392        Some(Claimed::Done)
393    })
394}

405    /// `.help` stays short: each command in a few words, its family's own
406    /// help saying the rest; a click runs only a help or something that
407    /// only shows.
408    #[test]
409    fn help_is_short_and_its_links_only_show() {
410        const SHOWS: &[&str] = &["version", "map", "controller", "hotbars", "loadlayout"];
411        for (_, (command, runs, says)) in help_entries() {
412            assert!(says.chars().count() <= 48, "{command}: {says}");
413            if let Some(runs) = runs {
414                assert!(
415                    runs.ends_with(" help") || SHOWS.contains(runs),
416                    "{command} runs {runs}, which may act"
417                );
418            }
419        }
420        let lines = help_notice().lines().to_vec();
421        println!("{}", lines.join("\n"));
422        assert_eq!(lines[0], HELP_FIRST);
423    }

425    /// A command typed is echoed and answered, and never written to the
426    /// game; a word nobody knows is answered too; a line without the symbol
427    /// is the game's.
428    #[tokio::test(start_paused = true)]
429    async fn the_symbol_keeps_a_line_from_the_game() {
430        let session = Session::new(ReplaySource::from_bytes(b""));
431        let handle = session.handle();
432        let login = crate::account::Login {
433            character: "Ashryn".to_owned(),
434            account: "a".to_owned(),
435            game_code: "GSX".to_owned(),
436        };
437        let alone = crate::relay::alone(&handle, &login.game_code, &login.character);
438        install(
439            &handle,
440            &login,
441            alone,
442            &crate::lich_command::Lichs::default(),
443            hydra_session::stream_rules::StreamRules::default(),
444        );
445        let mut events = handle.subscribe();
446        let deadline = Duration::from_secs(1);
447        assert_eq!(
448            handle.send(".help", Origin::Player, deadline).await,
449            Outcome::Claimed
450        );
451        assert_eq!(
452            handle.send(".hepl", Origin::Player, deadline).await,
453            Outcome::Claimed
454        );
455        assert_eq!(
456            handle.send(".ver", Origin::Player, deadline).await,
457            Outcome::Claimed
458        );
459        let heard: Vec<String> = std::iter::from_fn(|| events.try_recv().ok())
460            .map(|event| match event {
461                Event::Claimed(line) => format!("typed {line}"),
462                Event::Notice(notice) => notice.lines()[0].clone(),
463                other => format!("{other:?}"),
464            })
465            .collect();
466        assert_eq!(
467            heard,
468            [
469                "typed .help",
470                HELP_FIRST,
471                "typed .hepl",
472                "I do not know .hepl. .help lists Hydra's commands.",
473                "typed .ver",
474                &version_said(),
475            ]
476        );
477        drop(session);
478        assert_eq!(
479            handle.send("look", Origin::Player, deadline).await,
480            Outcome::Dead,
481            "a line without the symbol goes to the session as ever"
482        );
483    }

609    /// hash, `+` after it for changes not committed.
610    #[test]
611    fn the_version_names_the_build() {
612        let said = super::version_said();
613        let rest = said
614            .strip_prefix(&format!("Hydra {}", crate::VERSION))
615            .expect("the version first");
616        if let Some(commit) = rest.strip_prefix(" (").and_then(|r| r.strip_suffix(')')) {
617            let hash = commit.strip_suffix('+').unwrap_or(commit);
618            assert!(hash.len() >= 7, "{said}");
619            assert!(hash.chars().all(|c| c.is_ascii_hexdigit()), "{said}");
620        } else {
621            assert_eq!(rest, "", "{said}");
622        }
623    }

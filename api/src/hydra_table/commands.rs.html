1//! Hydra's own command line: every line that starts with the command symbol,
2//! `.` (`hydra_session::claimant`).
3//!
4//! **Installed when the session is built, before it connects**: the command
5//! line is there from the start, not something the player turns on.
6//!
7//! The symbol decides and the word routes: a line marked as Hydra's goes to
8//! the first family that knows its word, and a word nobody knows is answered
9//! "I do not know" by the session. It never reaches the game either way.
10//!
11//! The families so far: `help`, `version`, `history`, `theme`, `targetid`,
12//! `sorter`, `alias`, `highlight`, `to` and `all` (`relay`), `lich`, `go2` and `route2`
13//! (`travel_command`), `quick`, `timer`, `inventory`, `bestiary`, and `stop`.
14

56/// A family of commands: `Some` when the line was its own, `None` when the
57/// word is not one it knows.
58pub type Handler = Arc<dyn Fn(&str, Origin) -> Option<Claimed> + Send + Sync>;
59
60/// Run a command's work as a task of its own, what it holds counted as the
61/// commands' in a build that measures (`hydra_mem`).
62pub fn spawn<F>(work: F) -> tokio::task::JoinHandle<F::Output>
63where
64    F: Future + Send + 'static,
65    F::Output: Send + 'static,
66{
67    tokio::spawn(hydra_mem::task(hydra_mem::Part::Commands, work))
68}

70/// Install the command line on `handle`, for `login`'s character, with the
71/// symbol its settings choose; `.to` and `.all` reach the characters
72/// `characters` lists, and `.lich` runs its Lich among `lichs`.
73pub fn install(
74    handle: &SessionHandle,
75    login: &crate::account::Login,
76    characters: crate::relay::Characters,
77    lichs: &crate::lich_command::Lichs,
78) {
79    let roster_name = format!("{}:{}", login.game_code, login.character);
80    let (symbol, lich) = crate::pages::symbols(&crate::account::data_dir(), &roster_name);
81    let families: Vec<Handler> = vec![
82        help(handle.clone()),
83        version(handle.clone()),
84        crate::history::family(handle.clone(), roster_name.clone()),
85        crate::theme_command::family(handle.clone(), roster_name.clone()),
86        crate::targetid_command::family(handle.clone(), roster_name.clone()),
87        crate::sorter_command::family(handle.clone(), roster_name.clone()),
88        crate::alias_command::family(handle.clone(), roster_name.clone()),
89        crate::caster_command::family(handle.clone(), roster_name.clone()),
90        crate::highlight_command::family(handle.clone()),
91        crate::relay::family(handle.clone(), Arc::clone(&characters)),
92        crate::lich_command::family(handle.clone(), roster_name.clone(), lichs.clone()),
93        crate::travel_command::family(handle.clone(), roster_name.clone(), characters),
94        crate::quick_command::family(handle.clone()),
95        crate::hotbar_command::family(),
96        crate::timer_command::family(handle.clone(), roster_name.clone()),
97        crate::every_command::family(handle.clone(), roster_name.clone()),
98        crate::inventory_command::family(handle.clone()),
99        crate::bestiary_command::family(handle.clone()),
100        crate::survey_command::family(handle.clone(), roster_name.clone()),
101        stop(handle.clone(), roster_name),
102    ];
103    let runner: Runner = Arc::new(move |line: &str, origin: Origin| {
104        families
105            .iter()
106            .find_map(|family| family(line, origin))
107            .unwrap_or(Claimed::Unknown)
108    });
109    let desk = Desk::new(Some(symbol), runner).with_words(words());
110    desk.set_lich_symbol(lich);
111    // Once, as the session is built: nothing else installs one.
112    let _ = handle.set_desk(desk);
113}

115/// Every command's first word, as `.help` lists them, with the words that
116/// stand in for one: what the window completes after the symbol.
117pub fn words() -> Vec<String> {
118    let mut words: Vec<String> = HELP
119        .iter()
120        .skip(1)
121        .filter_map(|line| line.split_whitespace().next())
122        .map(str::to_owned)
123        .collect();
124    words.extend(
125        ["route2", "sail", "highlight", "ver", "inv", "ecaster"]
126            .iter()
127            .map(|&word| word.to_owned()),
128    );
129    words.sort();
130    words.dedup();
131    words
132}

134/// `.help` and `.?`: the families, a line each.
135fn help(handle: SessionHandle) -> Handler {
136    Arc::new(move |line: &str, _| {
137        let line = line.trim();
138        if !(line.eq_ignore_ascii_case("help") || line == "?") {
139            return None;
140        }
141        let lines = HELP.iter().map(|&line| line.to_owned()).collect();
142        handle.say_unlogged(Notice::table(NoticeKind::Info, lines).answering());
143        Some(Claimed::Done)
144    })
145}

147/// Which Hydra this is: its version, and the commit it was built from when
148/// git could say, `+` after it when the build had changes not committed.
149#[must_use]
150pub fn version_said() -> String {
151    let version = crate::VERSION;
152    match env!("HYDRA_COMMIT") {
153        "" => format!("Hydra {version}"),
154        commit => format!("Hydra {version} ({commit})"),
155    }
156}
157
158/// `.version` and `.ver`: which Hydra this is.
159fn version(handle: SessionHandle) -> Handler {
160    Arc::new(move |line: &str, _| {
161        let line = line.trim();
162        if !(line.eq_ignore_ascii_case("version") || line.eq_ignore_ascii_case("ver")) {
163            return None;
164        }
165        handle.say_unlogged(Notice::line(NoticeKind::Info, version_said()).answering());
166        Some(Claimed::Done)
167    })
168}

170/// `.stop`: everything on this character stops: its scripts are killed and
171/// its queue cleared. Its Lich is handed its own kill-all,
172/// `ka`, which spares the scripts Lich protects; the commands still waiting
173/// to be sent are dropped, and a walk under way stops. The one already
174/// written waits for its answer.
175fn stop(handle: SessionHandle, roster_name: String) -> Handler {
176    Arc::new(move |line: &str, _| {
177        if !line.trim().eq_ignore_ascii_case("stop") {
178            return None;
179        }
180        let walking = crate::travel_command::stop_walk(&roster_name);
181        let surveying = crate::survey_command::stop_survey(&roster_name);
182        let repeating = crate::every_command::stop_every(&roster_name);
183        let kill = format!("{}ka", handle.lich_symbol());
184        let killed = handle.to_lich(&kill);
185        let handle = handle.clone();
186        crate::commands::spawn(async move {
187            let dropped = handle.clear_queue().await;
188            let lich = if killed {
189                format!("Lich's scripts are stopped ({kill})")
190            } else if handle.lich_running() {
191                // It runs, and has not read what was typed for it: the
192                // kill-all could not be handed, and its scripts run on.
193                format!(
194                    "Lich is not reading what is typed for it, so its scripts could not be \
195                     stopped ({kill} not handed; .lich off stops Lich itself)"
196                )
197            } else {
198                "No Lich runs".to_owned()
199            };
200            let waiting = match dropped {
201                0 => "nothing was waiting to be sent".to_owned(),
202                1 => "1 command waiting to be sent is dropped".to_owned(),
203                n => format!("{n} commands waiting to be sent are dropped"),
204            };
205            let walk = if walking {
206                " The walk under way is stopped."
207            } else {
208                ""
209            };
210            let survey = if surveying {
211                " The survey is stopped."
212            } else {
213                ""
214            };
215            let every = if repeating {
216                " The line sent again and again is stopped."
217            } else {
218                ""
219            };
220            handle.say(
221                Notice::line(
222                    NoticeKind::Info,
223                    format!("Stopped. {lich}; {waiting}.{walk}{survey}{every}"),
224                )
225                .answering(),
226            );
227        });
228        Some(Claimed::Done)
229    })
230}

233mod tests {
234    use std::time::Duration;
235
236    use hydra_net::ReplaySource;
237    use hydra_session::{Event, Outcome, Session};
238
239    use super::*;
240
241    /// A command typed is echoed and answered, and never written to the
242    /// game; a word nobody knows is answered too; a line without the symbol
243    /// is the game's.
244    #[tokio::test(start_paused = true)]
245    async fn the_symbol_keeps_a_line_from_the_game() {
246        let session = Session::new(ReplaySource::from_bytes(b""));
247        let handle = session.handle();
248        let login = crate::account::Login {
249            character: "Ashryn".to_owned(),
250            account: "a".to_owned(),
251            game_code: "GSX".to_owned(),
252        };
253        let alone = crate::relay::alone(&handle, &login.game_code, &login.character);
254        install(
255            &handle,
256            &login,
257            alone,
258            &crate::lich_command::Lichs::default(),
259        );
260        let mut events = handle.subscribe();
261        let deadline = Duration::from_secs(1);
262        assert_eq!(
263            handle.send(".help", Origin::Player, deadline).await,
264            Outcome::Claimed
265        );
266        assert_eq!(
267            handle.send(".hepl", Origin::Player, deadline).await,
268            Outcome::Claimed
269        );
270        assert_eq!(
271            handle.send(".ver", Origin::Player, deadline).await,
272            Outcome::Claimed
273        );
274        let heard: Vec<String> = std::iter::from_fn(|| events.try_recv().ok())
275            .map(|event| match event {
276                Event::Claimed(line) => format!("typed {line}"),
277                Event::Notice(notice) => notice.lines()[0].clone(),
278                other => format!("{other:?}"),
279            })
280            .collect();
281        assert_eq!(
282            heard,
283            [
284                "typed .help",
285                HELP[0],
286                "typed .hepl",
287                "I do not know .hepl. .help lists Hydra's commands.",
288                "typed .ver",
289                &version_said(),
290            ]
291        );
292        drop(session);
293        assert_eq!(
294            handle.send("look", Origin::Player, deadline).await,
295            Outcome::Dead,
296            "a line without the symbol goes to the session as ever"
297        );
298    }
299}

302mod stop_tests {
303    use std::time::Duration;

305    use hydra_net::AnsweringSource;

310    /// `.stop` hands the character's Lich its kill-all and drops what waits
311    /// to be sent; the command already written still has its answer.
312    #[tokio::test(start_paused = true)]
313    async fn stop_kills_lichs_scripts_and_drops_what_waits() {
314        let (source, game) = AnsweringSource::new(b"<prompt time=\"1\">&gt;</prompt>\n");
315        game.hold_replies();
316        let session = Session::new(source);
317        let handle = session.handle();
318        let login = crate::account::Login {
319            character: "Ashryn".to_owned(),
320            account: "a".to_owned(),
321            game_code: "GSX".to_owned(),
322        };
323        let alone = crate::relay::alone(&handle, &login.game_code, &login.character);
324        install(
325            &handle,
326            &login,
327            alone,
328            &crate::lich_command::Lichs::default(),
329        );
330        let mut lich = handle.lich_door().attach().expect("a Lich");
331        let mut events = handle.subscribe();
332        tokio::spawn(session.run());
333
334        let wait = Duration::from_secs(30);
335        let sent: Vec<_> = ["look", "north", "south"]
336            .into_iter()
337            .map(|line| {
338                let handle = handle.clone();
339                tokio::spawn(async move { handle.send(line, Origin::Player, wait).await })
340            })
341            .collect();
342        tokio::time::sleep(Duration::from_millis(50)).await;
343        assert_eq!(game.lines(), ["look"], "one at a time");
344
345        handle.send(".stop", Origin::Player, wait).await;
346        tokio::time::sleep(Duration::from_millis(50)).await;
347        assert_eq!(lich.typing.next().await.as_deref(), Some(";ka"));
348        let mut outcomes = Vec::new();
349        game.release_replies();
350        for one in sent {
351            outcomes.push(one.await.unwrap());
352        }
353        assert!(
354            matches!(outcomes[0], Outcome::Confirmed(_) | Outcome::Timeout),
355            "{outcomes:?}"
356        );
357        assert_eq!(outcomes[1..], [Outcome::Stopped, Outcome::Stopped]);
358        assert_eq!(game.lines(), ["look"], "never sent");
359        let said: Vec<String> = std::iter::from_fn(|| events.try_recv().ok())
360            .filter_map(|event| match event {
361                Event::Notice(notice) => Some(notice.lines().join(" ")),
362                _ => None,
363            })
364            .collect();
365        assert!(
366            said.iter().any(|said| said
367                == "Stopped. Lich's scripts are stopped (;ka); 2 commands waiting to be sent are dropped."),
368            "{said:?}"
369        );
370    }

419#[cfg(test)]
420mod version_tests {
421    /// `.version` names the version, and the commit built from: a short
422    /// hash, `+` after it for changes not committed.
423    #[test]
424    fn the_version_names_the_build() {
425        let said = super::version_said();
426        let rest = said
427            .strip_prefix(&format!("Hydra {}", crate::VERSION))
428            .expect("the version first");
429        if let Some(commit) = rest.strip_prefix(" (").and_then(|r| r.strip_suffix(')')) {
430            let hash = commit.strip_suffix('+').unwrap_or(commit);
431            assert!(hash.len() >= 7, "{said}");
432            assert!(hash.chars().all(|c| c.is_ascii_hexdigit()), "{said}");
433        } else {
434            assert_eq!(rest, "", "{said}");
435        }
436    }

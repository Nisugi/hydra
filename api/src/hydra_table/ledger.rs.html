1//! The ledger shared by every character in this process: opened once, in
2//! the data folder, with one writer every recorder hands rows to.
3
4use std::sync::OnceLock;
5
6use hydra_ledger::{Ledger, Writer};
7
8/// The ledger and its writer, opened on first use; `None` when the file
9/// could not be opened, said once.
10fn shared() -> Option<&'static (Ledger, Writer)> {
11    static SHARED: OnceLock<Option<(Ledger, Writer)>> = OnceLock::new();
12    SHARED
13        .get_or_init(|| {
14            let path = Ledger::path_in(&crate::account::data_dir());
15            match Ledger::open(&path).and_then(|ledger| Ok((ledger.writer()?, ledger))) {
16                Ok((writer, ledger)) => Some((ledger, writer)),
17                Err(e) => {
18                    eprintln!("The ledger at {} could not be opened: {e}", path.display());
19                    None
20                }
21            }
22        })
23        .as_ref()
24}
25
26/// The ledger, and a writer on it.
27pub(crate) fn open() -> Option<(&'static Ledger, Writer)> {
28    shared().map(|(ledger, writer)| (ledger, writer.clone()))
29}
30
31/// Every row handed to the ledger so far written, waited for: Hydra
32/// stopping, the writer's thread not waited for otherwise. Nothing when the
33/// ledger was never opened.
34pub(crate) async fn flush() {
35    let Some((_, writer)) = shared() else {
36        return;
37    };
38    if let Err(why) = writer.flush().await {
39        eprintln!("[hydra] the ledger as Hydra stopped: {why}");
40    }
41}

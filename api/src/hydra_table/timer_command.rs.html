1//! `.timer`: a countdown of the player's own, by its name, typed, sent by
2//! a key's macro, or put by a Lich alias or script.
3//!
4//! `.timer <name> <time>` starts it, or starts it again: the time is
5//! seconds, or minutes and seconds (`90`, `90s`, `5m`, `1m30s`). `.timer
6//! <name>` alone starts it for its default duration, as the highlight that
7//! makes it a timer gives it. A list after either -- `active spell`,
8//! `buff`, `debuff`, `cooldown`, `custom` -- is the one it shows in; none,
9//! its highlight's, or the player's own. `.timer <name> off` clears it, and `.timer`
10//! alone lists those running. It ends
11//! by the game's clock, as a roundtime does, and is kept in the
12//! character's state, where the Timers widget shows it with the effects.
13

24/// What `.timer help` says.
25const HELP: &[&str] = &[
26    "timer <name> <time>   a countdown of your own: 90, 90s, 5m or 1m30s; again, it starts over",
27    "timer <name>          for its default duration, as its trigger gives it",
28    "timer <name> 5m buff  shown with the buffs: active spell, buff, debuff, cooldown or custom",
29    "timer <name> off      clears it",
30    "timer                 the timers running, and what is left of each",
31];

33/// What a `.timer` line asks.
34#[derive(Clone, Debug, PartialEq, Eq)]
35pub(crate) enum Asked {
36    /// The timers running.
37    List,
38    /// What it does.
39    Help,
40    /// Start `name`, `seconds` from now, in `list` when one was named.
41    Start {
42        name: String,
43        seconds: u32,
44        list: Option<List>,
45    },
46    /// Clear `name`.
47    Off(String),
48    /// Start `name` for its default duration, in `list` when one was named.
49    Default { name: String, list: Option<List> },
50}
51
52/// `90`, `90s`, `5m` or `1m30s`, in seconds; none for anything else, or
53/// nothing at all.
54pub(crate) fn seconds(time: &str) -> Option<u32> {
55    let time = time.trim().to_ascii_lowercase();
56    if let Ok(seconds) = time.parse::<u32>() {
57        return (seconds > 0).then_some(seconds);
58    }
59    let (minutes, rest) = match time.split_once('m') {
60        Some((minutes, rest)) => (minutes.parse::<u32>().ok()?, rest),
61        None => (0, time.as_str()),
62    };
63    let seconds = match rest.strip_suffix('s') {
64        Some(seconds) => seconds.parse::<u32>().ok()?,
65        None if rest.is_empty() => 0,
66        None => return None,
67    };
68    let total = minutes.checked_mul(60)?.checked_add(seconds)?;
69    (total > 0).then_some(total)
70}

72/// What `rest`, the line after `timer`, asks: the last word is the time,
73/// or `off`, and the words before it the timer's name; with neither, the
74/// whole of it is the name.
75pub(crate) fn asked(rest: &str) -> Asked {
76    let rest = rest.trim();
77    if rest.is_empty() || rest.eq_ignore_ascii_case("list") {
78        return Asked::List;
79    }
80    if rest.eq_ignore_ascii_case("help") {
81        return Asked::Help;
82    }
83    if let Some(name) = rest
84        .strip_suffix(" off")
85        .or_else(|| rest.strip_suffix(" OFF"))
86        .or_else(|| rest.strip_suffix(" Off"))
87    {
88        return Asked::Off(name.trim().to_owned());
89    }
90    // A list last: one word, or `active spell`.
91    let (rest, list) = [2, 1]
92        .into_iter()
93        .find_map(|words| {
94            let (front, back) = split_last(rest, words)?;
95            Some((front, List::named(back)?))
96        })
97        .map_or((rest, None), |(front, list)| (front, Some(list)));
98    match split_last(rest, 1) {
99        Some((name, time)) if seconds(time).is_some() => Asked::Start {
100            name: name.to_owned(),
101            seconds: seconds(time).unwrap_or_default(),
102            list,
103        },
104        _ => Asked::Default {
105            name: rest.to_owned(),
106            list,
107        },
108    }
109}

111/// `line` with its last `words` words apart from the rest, when there are
112/// more than that many.
113fn split_last(line: &str, words: usize) -> Option<(&str, &str)> {
114    let mut at = line.len();
115    for _ in 0..words {
116        at = line[..at].trim_end().rfind(char::is_whitespace)?;
117    }
118    let front = line[..at].trim();
119    (!front.is_empty()).then(|| (front, line[at..].trim()))
120}

122/// The default duration of the timer `name`, and its list, as the
123/// highlights in the data folder that are `character`'s give them.
124fn default_of(character: &str, name: &str) -> Option<(u32, List)> {
125    let file = hydra_ui::highlight::HighlightFile::load(&crate::account::data_dir()).ok()?;
126    hydra_ui::highlight::Matcher::new(&file, character)
127        .0
128        .timer_default(name)
129}

131/// The game's second now, as the state reckons it; this machine's clock
132/// before the game has said.
133fn now(state: &GameState) -> u32 {
134    state.timer_ends_in(0)
135}
136
137/// `seconds` as the list says them: `1:30`, `45s`.
138fn left(seconds: u32) -> String {
139    if seconds >= 60 {
140        format!("{}:{:02}", seconds / 60, seconds % 60)
141    } else {
142        format!("{seconds}s")
143    }
144}

146/// The timers in `state`, a line each, with what is left of them at `now`.
147pub(crate) fn listed(state: &GameState, now: u32) -> Vec<String> {
148    let width = state
149        .custom_timers
150        .iter()
151        .map(|(name, _)| name.chars().count())
152        .max()
153        .unwrap_or(0);
154    let lines: Vec<String> = state
155        .custom_timers
156        .iter()
157        .map(|(name, ends)| match ends.checked_sub(now) {
158            Some(rest) if rest > 0 => format!("{name:<width$}  {} left", left(rest)),
159            _ => format!("{name:<width$}  done"),
160        })
161        .collect();
162    if lines.is_empty() {
163        vec!["No timers running. .timer <name> <time> starts one.".to_owned()]
164    } else {
165        lines
166    }
167}

169/// The `.timer` family, for `character` (`GAME:Name`).
170pub(crate) fn family(handle: SessionHandle, character: String) -> Handler {
171    Arc::new(move |line: &str, _| {
172        let line = line.trim();
173        let (word, rest) = line.split_once(char::is_whitespace).unwrap_or((line, ""));
174        if !word.eq_ignore_ascii_case("timer") {
175            return None;
176        }
177        let asked = asked(rest);
178        let handle = handle.clone();
179        let character = character.clone();
180        crate::commands::spawn(async move {
181            let say = |kind, lines: Vec<String>| {
182                handle.say_unlogged(Notice::prose(kind, lines).answering());
183            };
184            // The usage and the listing have columns.
185            let table = |lines: Vec<String>| {
186                handle.say_unlogged(Notice::table(NoticeKind::Info, lines).answering());
187            };
188            let Some(snapshot) = handle.snapshot().await else {
189                return;
190            };
191            let state = &snapshot.state;
192            let set = |id: String, value: u32, list: List| {
193                handle.fold_in(vec![Frame::Timer {
194                    id,
195                    value,
196                    category: Some(list.name().to_owned()),
197                }])
198            };
199            let started = |name: &str, seconds: u32, list: List| {
200                let shown = match list {
201                    List::Custom => String::new(),
202                    list => format!(", with the {}s", list.name()),
203                };
204                vec![format!("Timer {name}: {}{shown}.", left(seconds))]
205            };
206            match asked {
207                Asked::List => table(listed(state, now(state))),
208                Asked::Help => table(HELP.iter().map(|&line| line.to_owned()).collect()),
209                Asked::Default { name, list } => match default_of(&character, &name) {
210                    Some((seconds, own)) => {
211                        let list = list.unwrap_or(own);
212                        if set(name.clone(), state.timer_ends_in(seconds), list) {
213                            say(NoticeKind::Info, started(&name, seconds, list));
214                        }
215                    }
216                    None => say(
217                        NoticeKind::Warn,
218                        vec![format!(
219                            "Timer {name} has no default duration: .timer {name} <time>, as 90, 5m or 1m30s; or give a trigger the timer {name} and its duration."
220                        )],
221                    ),
222                },
223                Asked::Off(name) => {
224                    let kept = state.custom_timers.get(&name).is_some();
225                    if !kept {
226                        say(
227                            NoticeKind::Info,
228                            vec![format!("No timer {name} is running.")],
229                        );
230                    } else if set(name.clone(), 0, List::Custom) {
231                        say(NoticeKind::Info, vec![format!("Timer {name}: cleared.")]);
232                    }
233                }
234                Asked::Start {
235                    name,
236                    seconds,
237                    list,
238                } => {
239                    let list = list
240                        .or_else(|| default_of(&character, &name).map(|(_, list)| list))
241                        .unwrap_or_default();
242                    if set(name.clone(), state.timer_ends_in(seconds), list) {
243                        say(NoticeKind::Info, started(&name, seconds, list));
244                    }
245                }
246            }
247        });
248        Some(Claimed::Done)
249    })
250}

260    #[test]
261    fn a_time_is_seconds_or_minutes_and_seconds() {
262        for (time, want) in [
263            ("90", Some(90)),
264            ("90s", Some(90)),
265            ("5m", Some(300)),
266            ("1m30s", Some(90)),
267            ("1M30S", Some(90)),
268            ("0", None),
269            ("soon", None),
270            ("1h", None),
271            ("m", None),
272        ] {
273            assert_eq!(seconds(time), want, "{time}");
274        }

277    /// The time is last, or a list after it, or `off`; the words before
278    /// are the name, which may be more than one.
279    #[test]
280    fn a_line_names_its_timer_then_its_time() {
281        let start = |name: &str, seconds, list| Asked::Start {
282            name: name.to_owned(),
283            seconds,
284            list,
285        };
286        let default = |name: &str, list| Asked::Default {
287            name: name.to_owned(),
288            list,
289        };
290        assert_eq!(asked(""), Asked::List);
291        assert_eq!(asked("help"), Asked::Help);
292        assert_eq!(asked("dark cata 5m"), start("dark cata", 300, None));
293        assert_eq!(
294            asked("dark cata 5m buff"),
295            start("dark cata", 300, Some(List::Buffs))
296        );
297        assert_eq!(
298            asked("cata 90 active spell"),
299            start("cata", 90, Some(List::ActiveSpells))
300        );
301        assert_eq!(asked("cata off"), Asked::Off("cata".to_owned()));
302        assert_eq!(asked("cata"), default("cata", None));
303        assert_eq!(asked("dark cata"), default("dark cata", None));
304        assert_eq!(asked("stun debuff"), default("stun", Some(List::Debuffs)));
305    }

325    /// `.timer cata 90`, typed or put by a script, sets the timer the
326    /// widgets read; `off` clears it; nothing reaches the game.
327    #[tokio::test(start_paused = true)]
328    async fn timer_sets_and_clears_the_players_timer() {
329        use hydra_net::AnsweringSource;
330        use hydra_session::Session;
331        let (source, game) = AnsweringSource::new(b"<prompt time=\"1000\">&gt;</prompt>\n");
332        let session = Session::new(source);
333        let handle = session.handle();
334        let login = crate::account::Login {
335            character: "Ashryn".to_owned(),
336            account: "a".to_owned(),
337            game_code: "GSX".to_owned(),
338        };
339        let alone = crate::relay::alone(&handle, &login.game_code, &login.character);
340        crate::commands::install(
341            &handle,
342            &login,
343            alone,
344            &crate::lich_command::Lichs::default(),
345        );
346        let running = tokio::spawn(session.run());
347        let wait = Duration::from_secs(5);
348        tokio::time::sleep(Duration::from_millis(10)).await;
349        let _ = handle.send(".timer cata 90", Origin::Lich, wait).await;
350        tokio::time::sleep(Duration::from_millis(10)).await;
351        let state = handle.snapshot().await.expect("a snapshot").state;
352        let ends = state.custom_timers.get("cata").expect("set");
353        assert!(ends > 90, "{ends}");
354        let _ = handle.send(".timer cata off", Origin::Player, wait).await;
355        tokio::time::sleep(Duration::from_millis(10)).await;
356        let state = handle.snapshot().await.expect("a snapshot").state;
357        assert_eq!(state.custom_timers.get("cata"), None);
358        assert!(game.lines().is_empty(), "{:?}", game.lines());
359        running.abort();
360    }

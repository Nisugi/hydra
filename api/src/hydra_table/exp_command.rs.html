1//! `.exp`: experience tracked, on the command line (`hydra_exp`). The
2//! words are parsed and answered in the crate; this routes the line to it
3//! and says the answer.
4
5use std::sync::Arc;
6
7use hydra_session::claimant::Claimed;
8use hydra_session::{Notice, NoticeKind, SessionHandle};
9
10use crate::commands::Handler;
11
12/// The family: `exp`, for the character the roster calls `roster_name`.
13pub(crate) fn family(handle: SessionHandle, roster_name: String) -> Handler {
14    Arc::new(move |line: &str, _| {
15        let line = line.trim();
16        let (word, rest) = line.split_once(char::is_whitespace).unwrap_or((line, ""));
17        if !word.eq_ignore_ascii_case("exp") {
18            return None;
19        }
20        let rest = rest.to_owned();
21        let handle = handle.clone();
22        let roster_name = roster_name.clone();
23        crate::commands::spawn(async move {
24            let now = hydra_net::eastern::now();
25            let said = who(&roster_name).and_then(|who| {
26                let asked = hydra_exp::parse(&rest, now)?;
27                let (ledger, writer) = crate::ledger::open()
28                    .ok_or_else(|| "the ledger could not be opened.".to_owned())?;
29                hydra_exp::answer(ledger, &writer, &who, &asked, now)
30            });
31            let notice = match said {
32                Ok(lines) => Notice::table(NoticeKind::Info, lines),
33                Err(why) => Notice::line(NoticeKind::Error, format!("Exp: {why}")),
34            };
35            handle.say_unlogged(notice.answering());
36        });
37        Some(Claimed::Done)
38    })
39}

41/// The character the roster calls `roster_name`, as the ledger names it.
42fn who(roster_name: &str) -> Result<hydra_ledger::Who, String> {
43    let (game, name) = roster_name
44        .split_once(':')
45        .ok_or_else(|| format!("{roster_name} is not a character's name."))?;
46    let instance = hydra_net::instance(game).ok_or_else(|| format!("{game} is not a game."))?;
47    Ok(hydra_ledger::Who::new(instance, name))
48}

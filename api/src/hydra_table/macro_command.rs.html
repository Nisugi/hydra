1//! `.macro`: the player's macros (`hydra_ui::macros`), from the command
2//! line, a Lich script or a key.
3//!
4//! `.macro <name>` runs one: the line is claimed here and never reaches
5//! the game, and the window, which hears every line claimed, runs the
6//! macro on its character with the character's world in view, as it
7//! presses a hotbar's button. `.macro` alone lists them, `.macro show
8//! <name>` says one, and `.macro alias <name> <word>` makes `<word>` an
9//! alias that runs it, saying so when the word is taken.
10

19/// What `.macro help` says.
20const HELP: &[&str] = &[
21    "macro <name>            run it: its commands, when its condition holds",
22    "macro                   every macro, with its commands, key and alias word",
23    "macro show <name>       one macro, whole",
24    "macro alias <name> <word>  a word that runs it from the command line, kept as an alias",
25    "Macros are kept in macros.toml in Hydra's data folder, and beside a character's layout for its own.",
26    "The commands: a comma or a line between them, s1.5 a wait, rt and ct the clock's, and the behaviour words. Settings, on the Macros page, is the editor.",
27];

29/// What a `.macro` line asks of the command line itself; a run is the
30/// window's.
31#[derive(Clone, Debug, PartialEq, Eq)]
32pub(crate) enum Asked {
33    /// What it does.
34    Help,
35    /// Every macro.
36    List,
37    /// One macro.
38    Show(String),
39    /// An alias word for one.
40    Alias { name: String, word: String },
41    /// Run one: the window's.
42    Run,
43}
44
45/// What `rest`, the line after `macro`, asks.
46pub(crate) fn parse(rest: &str) -> Asked {
47    let rest = rest.trim();
48    if rest.is_empty() {
49        return Asked::List;
50    }
51    let (word, after) = rest
52        .split_once(char::is_whitespace)
53        .map_or((rest, ""), |(word, after)| (word, after.trim()));
54    match word.to_ascii_lowercase().as_str() {
55        "help" => Asked::Help,
56        "list" => Asked::List,
57        "show" if !after.is_empty() => Asked::Show(after.to_owned()),
58        "alias" => match after.split_once(char::is_whitespace) {
59            Some((name, word)) if !word.trim().is_empty() => Asked::Alias {
60                name: name.to_owned(),
61                word: word.trim().to_owned(),
62            },
63            _ => Asked::Help,
64        },
65        _ => Asked::Run,
66    }
67}

69/// The lines `.macro` says of `file`, every macro in a line.
70fn listed(file: &hydra_ui::macros::MacroFile) -> Vec<String> {
71    if file.macros.is_empty() {
72        return vec!["No macros yet: macros.toml in the data folder holds them.".to_owned()];
73    }
74    file.macros
75        .iter()
76        .map(|made| {
77            let mut said = format!("{:<14} {}", made.name, made.chain);
78            if let Some(key) = &made.key {
79                use std::fmt::Write as _;
80                let _ = write!(said, "  [{key}");
81                if let Some(set) = made.set {
82                    let _ = write!(said, " in set {set}");
83                }
84                said.push(']');
85            }
86            if let Some(word) = &made.alias {
87                use std::fmt::Write as _;
88                let _ = write!(said, "  alias {word}");
89            }
90            said
91        })
92        .collect()
93}

95/// One macro, whole.
96fn shown(made: &hydra_ui::macros::MacroDef) -> Vec<String> {
97    let mut lines = vec![format!("{}: {}", made.name, made.chain)];
98    if let Some(key) = &made.key {
99        lines.push(match made.set {
100            Some(set) => format!("  key {key}, in set {set}"),
101            None => format!("  key {key}, in every set"),
102        });
103    }
104    if let Some(word) = &made.alias {
105        lines.push(format!("  alias {word}"));
106    }
107    if let Some(when) = &made.when {
108        let when = serde_json::to_string(when).unwrap_or_default();
109        lines.push(format!("  runs only when {when}"));
110    }
111    if let Some(after) = made.repeat_after {
112        lines.push(format!("  runs again after {after:.0} s"));
113    }
114    if made.once_per_room {
115        lines.push("  runs once a room".to_owned());
116    }
117    for state in &made.states {
118        let does = if state.unavailable {
119            "not available".to_owned()
120        } else {
121            state.chain.as_deref().map_or_else(
122                || "its own commands".to_owned(),
123                |chain| format!("runs {chain}"),
124            )
125        };
126        let when = serde_json::to_string(&state.when).unwrap_or_default();
127        lines.push(format!("  while {when}: {does}"));
128    }
129    lines
130}

132/// The family: `macro`, for the character `roster_name` names on `handle`.
133pub(crate) fn family(handle: SessionHandle, roster_name: String) -> Handler {
134    Arc::new(move |line: &str, _| {
135        let line = line.trim();
136        let (word, rest) = line.split_once(char::is_whitespace).unwrap_or((line, ""));
137        if !word.eq_ignore_ascii_case("macro") {
138            return None;
139        }
140        let asked = parse(rest);
141        if asked == Asked::Run {
142            // The window's: it hears the line claimed and runs the macro.
143            return Some(Claimed::Done);
144        }
145        let said = answer(&handle, &roster_name, &asked);
146        let notice = match said {
147            Ok(lines) => crate::commands::table_or_help(lines, HELP),
148            Err(why) => Notice::line(NoticeKind::Error, why),
149        };
150        handle.say(notice.answering());
151        Some(Claimed::Done)
152    })
153}

155/// What `asked` says, for the character `roster_name` names.
156fn answer(handle: &SessionHandle, roster_name: &str, asked: &Asked) -> Result<Vec<String>, String> {
157    let dir = crate::account::data_dir();
158    let (instance, name) = crate::pages::who(roster_name)?;
159    let (file, problems) = hydra_board::macros_for(&dir, Some(instance), name);
160    match asked {
161        Asked::Help => Ok(HELP.iter().map(|l| (*l).to_owned()).collect()),
162        Asked::List => {
163            let mut lines = listed(&file);
164            lines.extend(problems);
165            Ok(lines)
166        }
167        Asked::Show(wanted) => file
168            .named(wanted)
169            .map(shown)
170            .ok_or_else(|| format!("There is no macro {wanted}.")),
171        Asked::Alias { name, word } => {
172            let made = file
173                .named(name)
174                .ok_or_else(|| format!("There is no macro {name}."))?;
175            let symbol = handle.symbol().unwrap_or('.');
176            let target = format!("{symbol}macro {}", made.name);
177            let own = crate::alias_command::own_key(roster_name)?;
178            let was = alias::change(&dir, |file| {
179                let taken = file
180                    .of(Whose::Own(&own))
181                    .and_then(|aliases| aliases.get(word.as_str()))
182                    .cloned();
183                if let Some(taken) = taken
184                    && taken != target
185                {
186                    return Err(format!(
187                        "{word} is an alias already, for {taken}. .alias set {word} {target} overwrites it."
188                    ));
189                }
190                file.put(Whose::Own(&own), word, &target)
191            })?;
192            Ok(vec![match was {
193                Some(_) => format!("{word} runs the macro {} now.", made.name),
194                None => format!(
195                    "{word} runs the macro {}: an alias, this character's.",
196                    made.name
197                ),
198            }])
199        }
200        Asked::Run => Ok(Vec::new()),
201    }
202}

204#[cfg(test)]
205mod tests {
206    use super::*;
207
208    /// A bare `.macro` lists, a word runs, and the rest are the command
209    /// line's own.
210    #[test]
211    fn a_macro_line_is_read_by_its_first_word() {
212        assert_eq!(parse(""), Asked::List);
213        assert_eq!(parse("help"), Asked::Help);
214        assert_eq!(parse("list"), Asked::List);
215        assert_eq!(parse("show barrage"), Asked::Show("barrage".into()));
216        assert_eq!(
217            parse("alias barrage wb"),
218            Asked::Alias {
219                name: "barrage".into(),
220                word: "wb".into()
221            }
222        );
223        assert_eq!(parse("alias barrage"), Asked::Help);
224        assert_eq!(parse("barrage"), Asked::Run);
225    }
226
227    /// The listing and the showing say a macro's commands, key and word.
228    #[test]
229    fn a_macro_is_listed_and_shown() {
230        let file = hydra_ui::macros::MacroFile::parse(
231            "[[macros]]\nname = \"barrage\"\nchain = \"stance off, rt, weapon barrage\"\nkey = \"F5\"\nset = 1\nalias = \"wb\"\n",
232        )
233        .expect("reads");
234        let lines = listed(&file);
235        assert_eq!(lines.len(), 1);
236        assert!(
237            lines[0].contains("[F5 in set 1]") && lines[0].contains("alias wb"),
238            "{}",
239            lines[0]
240        );
241        let one = shown(file.named("barrage").expect("named"));
242        assert_eq!(one[0], "barrage: stance off, rt, weapon barrage");
243        assert_eq!(one[1], "  key F5, in set 1");
244        assert!(listed(&hydra_ui::macros::MacroFile::default())[0].starts_with("No macros"));
245    }

1//! A line sent and its answer collected: the game's lines until the prompt
2//! that closes them, with the roundtime they cost.
3//!
4//! A [`Listener`] joins a session once ([`SessionHandle::join`]) and keeps
5//! its own [`GameState`] from the stream, so it can tell which lines the
6//! model finished since it last looked. [`Listener::ask`] writes a line to
7//! the game at once, past the queue, waits for the session to echo it
8//! ([`Event::Sent`]) so nothing said before it is taken for its answer,
9//! then keeps every main-window line until the next prompt. A prompt that
10//! closes nothing (the game sends one for someone arriving, a spell
11//! wearing off) is not taken for the answer's own: the answer is the lines
12//! up to the first prompt after at least one line, or whatever came by the
13//! time the wait ran out. [`Reply::prompt_closed`] says which. A command
14//! can answer in more than one burst with a prompt after each (`incant`
15//! is a prepare and a cast), so the answer is held open for a short quiet
16//! spell after its prompt and takes what follows, prompts and all, and
17//! for as long as the roundtime it stated lasts (a cast's effect comes
18//! when its cast time ends). And nothing is written while the game is
19//! still talking: a line asked waits for a quiet spell first, so what a
20//! command of the player's left in the stream is not taken for the
21//! answer either. A line is never taken for the next command's answer,
22//! and one answer taken wrongly does not shift every answer after it.
23//!
24//! The roundtime is read from the state at the close: how many seconds
25//! the game said remain, as the game counts them, never the absolute
26//! epoch the frame carries.
27

41/// How long an answer is waited for once the line is written: the game
42/// answers most lines within a second, a lagging one within a few.
43pub const ANSWER_WAIT: Duration = Duration::from_secs(3);
44
45/// How long an answer is held open after its prompt for more of itself,
46/// and how long the stream must be quiet before a line is written.
47pub const SETTLE: Duration = Duration::from_millis(300);
48
49/// The longest a line waits for the stream to go quiet before it is
50/// written anyway.
51pub const QUIET_WAIT: Duration = Duration::from_secs(2);
52
53/// The longest a stated roundtime is honoured before a line is sent
54/// anyway: a stuck clock does not hold a caller for ever.
55pub const LONGEST_ROUNDTIME: Duration = Duration::from_secs(30);
56
57/// The least a roundtime wait sleeps toward its tick: a wake a hair before
58/// it, the clock still a second short, sleeps this and looks again rather
59/// than spinning.
60pub const TICK_SLACK: Duration = Duration::from_millis(5);
61
62/// One command's answer.
63#[derive(Clone, Debug, Default, PartialEq, Eq)]
64pub struct Reply {
65    /// The main-window lines the game finished after the line was sent,
66    /// in order, up to the prompt or the end of the wait.
67    pub lines: Vec<ChunkLine>,
68    /// Seconds of roundtime the game stated for it, `None` when it stated
69    /// none.
70    pub roundtime: Option<u32>,
71    /// Seconds of cast roundtime the game stated for it.
72    pub cast_time: Option<u32>,
73    /// A prompt closed the answer. `false`: the wait ran out first, and
74    /// the lines may be short of the whole answer.
75    pub prompt_closed: bool,
76    /// The character arrived in a room while the answer came: the line was
77    /// a move, or something moved the character.
78    pub arrived: bool,
79}

89/// Why a listener could not go on.
90#[derive(Clone, Copy, Debug, PartialEq, Eq)]
91pub enum Lost {
92    /// The connection dropped, or the session ended.
93    Connection,
94    /// The stop token was cancelled.
95    Stopped,
96}

98/// A session joined, its state kept from the stream, ready to ask.
99pub struct Listener {
100    handle: SessionHandle,
101    events: broadcast::Receiver<Event>,
102    state: GameState,
103    /// Lines of the open chunk already looked at, the dropped counted.
104    heard: usize,
105    /// How many lines the player has sent since the join: a walker reads
106    /// it to see the player taking the character over.
107    player_sent: u64,
108    /// When the game last said something: a line or a prompt, as this
109    /// listener folded it. The quiet spell runs from here.
110    last_said: Instant,
111    stop: CancellationToken,
112}

114impl Listener {
115    /// Join `handle`'s session: its state now, and everything from here
116    /// on. `stop` ends any wait at once. `None` once the session has ended.
117    pub async fn join(handle: &SessionHandle, stop: CancellationToken) -> Option<Self> {
118        let (snapshot, events) = handle.join().await?;
119        if matches!(snapshot.lifecycle, State::Reconnecting | State::Closed) {
120            return None;
121        }
122        let mut listener = Self {
123            handle: handle.clone(),
124            events,
125            state: snapshot.state,
126            heard: 0,
127            player_sent: 0,
128            last_said: Instant::now(),
129            stop,
130        };
131        listener.heard = listener.heard_so_far();
132        Some(listener)
133    }

138        &self.state

142    #[must_use]

147    /// Send `line` as `origin` and collect its answer: any roundtime is
148    /// waited out first, then up to `wait` for the answer's prompt.
149    ///
150    /// # Errors
151    ///
152    /// The connection or the session went, or the stop was cancelled.
153    pub async fn ask(&mut self, line: &str, origin: Origin, wait: Duration) -> Result<Reply, Lost> {
154        self.ready().await?;
155        self.answer(line, origin, wait).await
156    }
157
158    /// What [`Self::ask`] waits for before it writes: any roundtime over,
159    /// then a quiet spell.
160    ///
161    /// # Errors
162    ///
163    /// The connection or the session went, or the stop was cancelled.
164    pub async fn ready(&mut self) -> Result<(), Lost> {
165        self.out_of_roundtime().await?;
166        self.quiet().await?;
167        if self.stop.is_cancelled() {
168            return Err(Lost::Stopped);
169        }
170        Ok(())
171    }
172
173    /// Send `line` as `origin` now, waiting for nothing first, and collect
174    /// its answer as [`Self::ask`] does.
175    ///
176    /// # Errors
177    ///
178    /// The connection or the session went, or the stop was cancelled.
179    pub async fn answer(
180        &mut self,
181        line: &str,
182        origin: Origin,
183        wait: Duration,
184    ) -> Result<Reply, Lost> {
185        let arrivals = self.state.arrivals;
186        let Some(mut reply) = self.first_answer(line, origin, wait).await? else {
187            return Ok(Reply::default());
188        };
189        // The rest of the answer: what comes while the roundtime it
190        // stated lasts, then until a quiet spell.
191        let most = Instant::now() + LONGEST_ROUNDTIME;
192        while self.in_roundtime() && Instant::now() < most {
193            let (fresh, _) = self.hold_until(Instant::now() + SETTLE).await?;
194            reply.lines.extend(fresh.lines);
195            self.stated(&mut reply);
196        }
197        let mut quiet = Instant::now() + SETTLE;
198        loop {
199            let (fresh, _) = self.hold_until(quiet).await?;
200            self.stated(&mut reply);
201            if !fresh.lines.is_empty() {
202                reply.lines.extend(fresh.lines);
203                quiet = Instant::now() + SETTLE;
204            }
205            if Instant::now() >= quiet {
206                break;
207            }
208        }
209        reply.arrived = self.state.arrivals != arrivals;
210        Ok(reply)
211    }

213    /// Write `line` as `origin` now, waiting for nothing: neither the echo
214    /// nor an answer.
215    ///
216    /// # Errors
217    ///
218    /// The connection or the session went.
219    pub async fn send(&mut self, line: &str, origin: Origin) -> Result<(), Lost> {
220        if self.handle.write(line, origin).await {
221            Ok(())
222        } else {
223            Err(Lost::Connection)
224        }
225    }

227    /// The main-window lines the game finishes until `until`, in order,
228    /// the state kept current as they come.
229    ///
230    /// # Errors
231    ///
232    /// The connection or the session went, or the stop was cancelled.
233    pub async fn lines_until(&mut self, until: Instant) -> Result<Vec<ChunkLine>, Lost> {
234        let mut lines = Vec::new();
235        while Instant::now() < until {
236            let (fresh, _) = self.hold_until(until).await?;
237            lines.extend(fresh.lines);
238        }
239        Ok(lines)
240    }

242    /// Send `line` as `origin` now and take its answer to the first prompt
243    /// after a line, with the roundtime stated by then: nothing held open
244    /// after it. `None` when the line was not seen to go out within `wait`.
245    ///
246    /// # Errors
247    ///
248    /// The connection or the session went, or the stop was cancelled.
249    pub async fn first_answer(
250        &mut self,
251        line: &str,
252        origin: Origin,
253        wait: Duration,
254    ) -> Result<Option<Reply>, Lost> {
255        if !self.handle.write(line, origin).await {
256            return Err(Lost::Connection);
257        }
258        let until = Instant::now() + wait;
259        // Nothing said before the echo is the answer.
260        loop {
261            let (_, sent) = self.hold_until(until).await?;
262            if sent.as_deref() == Some(line) {
263                break;
264            }
265            if Instant::now() >= until {
266                return Ok(None);
267            }
268        }
269        self.heard = self.heard_so_far();
270        let mut reply = Reply::default();
271        loop {
272            let (fresh, _) = self.hold_until(until).await?;
273            reply.lines.extend(fresh.lines);
274            if fresh.prompt && !reply.lines.is_empty() {
275                reply.prompt_closed = true;
276                break;
277            }
278            if Instant::now() >= until {
279                break;
280            }
281        }
282        // The roundtime and cast time the answer stated, read as each
283        // burst of it comes: the most seen is the command's own.
284        self.stated(&mut reply);
285        Ok(Some(reply))
286    }

288    /// The roundtime and cast time the game states now, kept on `reply`
289    /// when more than it has.
290    fn stated(&self, reply: &mut Reply) {
291        let roundtime = self.state.roundtime_remaining().filter(|left| *left > 0);
292        let cast_time = self.state.casttime_remaining().filter(|left| *left > 0);
293        reply.roundtime = reply.roundtime.max(roundtime);
294        reply.cast_time = reply.cast_time.max(cast_time);
295    }
296
297    /// Whether the game's roundtime or cast time is still running.
298    fn in_roundtime(&self) -> bool {
299        self.state.roundtime_remaining().unwrap_or(0) > 0
300            || self.state.casttime_remaining().unwrap_or(0) > 0
301    }
302
303    /// Until nothing has come from the game for [`SETTLE`], or
304    /// [`QUIET_WAIT`] has passed: what was still coming is nobody's
305    /// answer.
306    ///
307    /// # Errors
308    ///
309    /// The connection or the session went, or the stop was cancelled.
310    pub async fn quiet(&mut self) -> Result<(), Lost> {
311        // What is already waiting is folded first, so the spell runs from
312        // the game's last word and not from one not yet read.
313        self.drain().await?;
314        let most = Instant::now() + QUIET_WAIT;
315        while Instant::now() < (self.last_said + SETTLE).min(most) {
316            self.hold_until((self.last_said + SETTLE).min(most)).await?;
317        }
318        Ok(())
319    }
320
321    /// Fold every event already waiting, waiting for none.
322    async fn drain(&mut self) -> Result<(), Lost> {
323        loop {
324            match self.events.try_recv() {
325                Ok(event) => {
326                    self.take(Ok(event)).await?;
327                }
328                Err(broadcast::error::TryRecvError::Empty) => return Ok(()),
329                Err(broadcast::error::TryRecvError::Closed) => return Err(Lost::Connection),
330                Err(broadcast::error::TryRecvError::Lagged(_)) => self.rejoin().await?,
331            }
332        }
333    }

335    /// Fold what the game says for `most`, so the state is current.
336    ///
337    /// # Errors
338    ///
339    /// The connection or the session went, or the stop was cancelled.
340    pub async fn hold(&mut self, most: Duration) -> Result<(), Lost> {
341        let until = Instant::now() + most;
342        while Instant::now() < until {
343            self.hold_until(until).await?;
344        }
345        Ok(())
346    }

348    /// Until the roundtime and cast roundtime the game stated are over, or
349    /// [`LONGEST_ROUNDTIME`].
350    ///
351    /// # Errors
352    ///
353    /// The connection or the session went, or the stop was cancelled.
354    pub async fn out_of_roundtime(&mut self) -> Result<(), Lost> {
355        let until = Instant::now() + LONGEST_ROUNDTIME;
356        loop {
357            let left = self.state.until_out_of_roundtime().unwrap_or_default();
358            if left.is_zero() || Instant::now() >= until {
359                return Ok(());
360            }
361            // To the tick, and looked at again on whatever comes before it:
362            // a new roundtime moves the end.
363            let tick = Instant::now() + left.max(TICK_SLACK);
364            self.hold_until(tick.min(until)).await?;
365        }
366    }

368    /// The next event, or `until`: the lines it finished and whether it
369    /// was a prompt, or the line it echoed when it was a send.
370    async fn hold_until(&mut self, until: Instant) -> Result<(Fresh, Option<String>), Lost> {
371        let event = tokio::select! {
372            biased;
373            () = self.stop.cancelled() => return Err(Lost::Stopped),
374            () = tokio::time::sleep_until(until.into()) => return Ok((Fresh::default(), None)),
375            event = self.events.recv() => event,
376        };
377        self.take(event).await
378    }

380    /// Take one event received: the lines it finished and whether it was
381    /// a prompt, or the line it echoed when it was a send.
382    async fn take(
383        &mut self,
384        event: Result<Event, RecvError>,
385    ) -> Result<(Fresh, Option<String>), Lost> {
386        match event {
387            Ok(Event::Frame(frame)) => {
388                let fresh = self.fold(&frame);
389                if !fresh.lines.is_empty() || fresh.prompt {
390                    self.last_said = Instant::now();
391                }
392                Ok((fresh, None))
393            }
394            Ok(Event::Sent { line, origin }) => {
395                if origin == Origin::Player {
396                    self.player_sent += 1;
397                }
398                Ok((Fresh::default(), Some(line)))
399            }
400            Ok(Event::StateChanged(State::Reconnecting | State::Closed))
401            | Err(RecvError::Closed) => Err(Lost::Connection),
402            Ok(_) => Ok((Fresh::default(), None)),
403            Err(RecvError::Lagged(_)) => {
404                self.rejoin().await?;
405                Ok((Fresh::default(), None))
406            }
407        }
408    }

410    /// Events were lost: the state taken afresh, with the stream from just
411    /// after it, rather than decided from the holes.
412    async fn rejoin(&mut self) -> Result<(), Lost> {
413        let (snapshot, events) = self.handle.join().await.ok_or(Lost::Connection)?;
414        if matches!(snapshot.lifecycle, State::Reconnecting | State::Closed) {
415            return Err(Lost::Connection);
416        }
417        self.state = snapshot.state;
418        self.events = events;
419        self.heard = self.heard_so_far();
420        // What was lost may have been the game talking.
421        self.last_said = Instant::now();
422        Ok(())
423    }
424
425    /// Take `frame` into the state; the lines it finished.
426    fn fold(&mut self, frame: &Frame) -> Fresh {
427        self.state.apply(frame);
428        match frame {
429            Frame::Prompt { .. } => {
430                self.heard = 0;
431                Fresh {
432                    lines: Vec::new(),
433                    prompt: true,
434                }
435            }
436            Frame::Text(_) => {
437                let chunk = self.state.open_chunk();
438                let lines = chunk.lines();
439                let total = chunk.dropped() + lines.len();
440                let fresh = total.saturating_sub(self.heard).min(lines.len());
441                self.heard = total;
442                Fresh {
443                    lines: lines[lines.len() - fresh..].to_vec(),
444                    prompt: false,
445                }
446            }
447            _ => Fresh::default(),
448        }
449    }
450
451    /// How many lines the open chunk has had, the dropped counted.
452    fn heard_so_far(&self) -> usize {
453        let chunk = self.state.open_chunk();
454        chunk.dropped() + chunk.lines().len()
455    }
456}

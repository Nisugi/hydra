1//! [`SupervisedSession`]: a session that outlives its connections.
2//!
3//! A [`Session`](crate::Session) is one connection. This opens connections,
4//! runs one actor over each, and decides whether to open another. Between
5//! two, in order:
6//!
7//! 1. the game state forgets what the new connection has not told it
8//!    ([`GameState::invalidate_for_reconnect`]);
9//! 2. the generation advances, so a command meant for the old connection is
10//!    not sent on the new one;
11//! 3. [`State::Reconnecting`] is published;
12//! 4. it waits one rung of the backoff ladder ([`backoff`]), answering its
13//!    handles meanwhile;
14//! 5. it asks the connector for a new connection.
15//!
16//! What stops it: a quit or a cancel; a connect error that cannot succeed by
17//! trying again ([`Retryability::Fatal`]); and [`MAX_UNATTENDED_LOSSES`]
18//! drops in a row with nobody using the session -- an idle character is not
19//! logged back in forever.
20

34/// The backoff ladder, in seconds: climbs, then holds at the last rung.
35const BACKOFF_SECONDS: &[u64] = &[1, 2, 5, 10, 30];
36
37/// A connection that lasted this long, and carried bytes, worked: the ladder
38/// starts over after it.
39pub const STABLE_CONNECTION: Duration = Duration::from_secs(30);
40
41/// Drops in a row with no command sent before the session stops.
42pub const MAX_UNATTENDED_LOSSES: u32 = 2;
43
44/// A connection that lasted this long counts as attended even with no
45/// command sent, unless the game warned it was idle.
46pub const LONG_LIVED: Duration = Duration::from_mins(5);

48/// Whether trying again can help.
49#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
50pub enum Retryability {
51    /// It might work next time: a network fault, a server hiccup.
52    #[default]
53    Transient,
54    /// It will not: a refused password, an unknown character.
55    Fatal,
56}

58/// Why a connection could not be opened.
59#[derive(Clone, Debug, PartialEq, Eq)]
60pub struct ConnectError {
61    /// Which step failed.
62    pub stage: &'static str,
63    /// What it said.
64    pub detail: String,
65    /// Whether to try again.
66    pub retryability: Retryability,
67}
68
69impl ConnectError {
70    /// A failure worth retrying.
71    #[must_use]
72    pub fn transient(stage: &'static str, detail: impl Into<String>) -> Self {
73        Self {
74            stage,
75            detail: detail.into(),
76            retryability: Retryability::Transient,
77        }
78    }
79
80    /// A failure retrying cannot fix.
81    #[must_use]
82    pub fn fatal(stage: &'static str, detail: impl Into<String>) -> Self {
83        Self {
84            stage,
85            detail: detail.into(),
86            retryability: Retryability::Fatal,
87        }
88    }
89}

114/// The delay before try `attempt` (from 0), spread by `jitter` (0 to 1) over
115/// 20% either way, so sessions dropped together do not log in together.
116#[must_use]
117pub fn backoff(attempt: u32, jitter: f64) -> Duration {
118    let index = usize::try_from(attempt)
119        .unwrap_or(usize::MAX)
120        .min(BACKOFF_SECONDS.len() - 1);
121    let base = BACKOFF_SECONDS[index].saturating_mul(1000);
122    let scale = 0.8 + jitter.clamp(0.0, 1.0) * 0.4;
123    #[expect(
124        clippy::cast_possible_truncation,
125        clippy::cast_sign_loss,
126        clippy::cast_precision_loss,
127        reason = "base is at most 30,000 and scale at most 1.2: exact in f64, \
128                  and the product is a small positive whole number of ms"
129    )]
130    Duration::from_millis((base as f64 * scale) as u64)
131}
132
133/// A jitter from the clock's sub-second digits.
134fn jitter() -> f64 {
135    use std::time::{SystemTime, UNIX_EPOCH};
136    let nanos = SystemTime::now()
137        .duration_since(UNIX_EPOCH)
138        .map_or(0, |since| since.subsec_nanos());
139    f64::from((nanos / 1_000) % 1000) / 999.0
140}

142/// Why a supervised session stopped.
143#[derive(Clone, Debug, PartialEq, Eq)]
144pub enum StoppedBecause {
145    /// Asked to: a quit or a cancel.
146    Cancelled,
147    /// A connection could not be opened, and trying again cannot help.
148    Fatal(ConnectError),
149    /// It kept dropping with nobody using it.
150    Unattended,
151}

153/// What is left when a supervised session stops.
154#[derive(Debug)]
155pub struct SupervisedEnd {
156    /// Why it stopped.
157    pub stopped_because: StoppedBecause,
158    /// The game state.
159    pub state: GameState,
160    /// Everything read and written, across every connection.
161    pub recorder: Recorder,
162}

278    /// Run until it stops.
279    pub async fn run(mut self) -> SupervisedEnd {
280        // However this ends, the player's Lich is let go with it, and what
281        // the binary installed on the handle.
282        let _closes = crate::lich::Closes(self.handle.lich.clone());
283        let _lets_go = crate::command::LetsGo(self.handle.clone());
284        let mut attempt = 0u32;
285        let mut unattended = 0u32;
286        let stopped_because = loop {
287            let connected = self.connect_answering().await;
288            let Some(connected) = connected else {
289                break StoppedBecause::Cancelled;
290            };
291            for secret in self.connector.take_secrets() {
292                if let Some(sink) = self.sink.as_mut() {
293                    sink.redact_key(&secret);
294                }
295                if let Some((log, _)) = &self.player_log {
296                    log.redact(&secret, "[KEY]");
297                }
298            }
299            let source = match connected {
300                Ok(source) => source,
301                Err(error) => {
302                    self.log(&format!("connect failed: {error}"));
303                    if error.retryability == Retryability::Fatal {
304                        break StoppedBecause::Fatal(error);
305                    }
306                    if !self
307                        .wait_before_retry(&mut attempt, &error.to_string())
308                        .await
309                    {
310                        break StoppedBecause::Cancelled;
311                    }
312                    continue;
313                }
314            };
315
316            let received_before = self.recorder.inbound_count();
317            let actor = SessionActor::supervised(
318                source,
319                std::mem::take(&mut self.state),
320                self.take_commands(),
321                self.events.clone(),
322                std::mem::take(&mut self.recorder),
323                self.sink.take(),
324                self.cancel.child_token(),
325                self.generation.get(),
326            )
327            .with_player_log(
328                self.player_log.as_ref().map(|(log, _)| log.clone()),
329                &self
330                    .player_log
331                    .as_ref()
332                    .map(|(_, capture)| capture.clone())
333                    .unwrap_or_default(),
334            )
335            .with_stream_rules(self.stream_rules.clone())
336            .with_store(self.store.clone(), self.facts_read)
337            .with_lich(self.handle.lich.clone());
338            let connected_at = tokio::time::Instant::now();
339            let end = Box::pin(actor.run()).await;
340            self.commands = end.commands;
341            self.recorder = end.recorder;
342            self.sink = end.sink;
343            self.state = end.state;
344            self.lifecycle = end.lifecycle;
345            self.facts_read = end.facts_read;
346            if !end.reason.warrants_reconnect() || self.cancel.is_cancelled() {
347                break StoppedBecause::Cancelled;
348            }
349
350            // A connection that lasted and carried bytes worked: the ladder
351            // starts over.
352            let lived = connected_at.elapsed();
353            if self.recorder.inbound_count() > received_before && lived >= STABLE_CONNECTION {
354                attempt = 0;
355            }
356            let long_lived = lived >= LONG_LIVED && !self.state.idle_warned();
357            if end.attended || long_lived {
358                unattended = 0;
359            } else {
360                unattended += 1;
361                // The game said it was idle before it dropped it: once is
362                // enough.
363                let allowed = if self.state.idle_warned() {
364                    1
365                } else {
366                    MAX_UNATTENDED_LOSSES
367                };
368                if unattended >= allowed {
369                    self.log(&format!(
370                        "no command sent across {unattended} connection(s); not reconnecting"
371                    ));
372                    break StoppedBecause::Unattended;
373                }
374            }
375
376            self.state.invalidate_for_reconnect();
377            self.generation.advance();
378            if !self
379                .wait_before_retry(&mut attempt, &format!("{:?}", end.reason))
380                .await
381            {
382                break StoppedBecause::Cancelled;
383            }
384        };
385        self.finish(stopped_because)
386    }

388    /// Between connections the receiver lives here; an actor borrows it.
389    /// Connect, answering handles meanwhile, as between tries: a login can
390    /// take a while, and a quit must not wait for it. `None` when it was told
391    /// to stop instead.
392    async fn connect_answering(&mut self) -> Option<Result<C::Source, ConnectError>> {
393        let connecting = self.connector.connect();
394        tokio::pin!(connecting);
395        let mut asking = true;
396        loop {
397            tokio::select! {
398                biased;
399                () = self.cancel.cancelled() => return None,
400                connected = &mut connecting => return Some(connected),
401                message = self.commands.recv(), if asking => match message {
402                    Some(message) => {
403                        let snapshot = || Snapshot {
404                            state: self.state.clone(),
405                            lifecycle: self.lifecycle,
406                            generation: self.generation.get(),
407                        };
408                        if answer_between(message, &snapshot, &self.events) {
409                            return None;
410                        }
411                    }
412                    // Every handle is gone; nothing more will be asked.
413                    None => asking = false,
414                },
415            }
416        }
417    }

419    fn take_commands(&mut self) -> mpsc::Receiver<Inbox> {
420        let (_, empty) = mpsc::channel(1);
421        std::mem::replace(&mut self.commands, empty)
422    }

424    /// Publish `Reconnecting`, then wait one rung, answering handles meanwhile.
425    /// `false` when it was told to stop instead.
426    async fn wait_before_retry(&mut self, attempt: &mut u32, detail: &str) -> bool {
427        self.transition(State::Reconnecting);
428        let delay = backoff(*attempt, jitter());
429        *attempt = attempt.saturating_add(1);
430        self.log(&format!(
431            "reconnecting in {}ms (attempt {attempt})",
432            delay.as_millis()
433        ));
434        let _ = self.events.send(Event::ConnectFailed {
435            attempt: *attempt,
436            delay,
437            detail: detail.to_owned(),
438        });
439        if let Some((log, _)) = &self.player_log {
440            log.notice(&format!(
441                "The connection is down ({detail}); trying again in {:.1}s.",
442                delay.as_secs_f64()
443            ));
444        }
445        let until = tokio::time::Instant::now() + delay;
446        loop {
447            tokio::select! {
448                biased;
449                () = self.cancel.cancelled() => return false,
450                () = tokio::time::sleep_until(until) => return true,
451                message = self.commands.recv() => match message {
452                    // Every handle is gone; nothing more will be asked.
453                    None => return true,
454                    Some(message) => {
455                        if answer_between(message, &|| self.snapshot(), &self.events) {
456                            return false;
457                        }
458                    }
459                },
460            }
461        }
462    }

464    fn snapshot(&self) -> Snapshot {

467            lifecycle: self.lifecycle,

469        }

472    fn finish(mut self, stopped_because: StoppedBecause) -> SupervisedEnd {
473        // The session is over, whatever ended it, and the player's Lich's
474        // copy of the stream with it: the Lich behind it is stopped.
475        crate::lich::lock(&self.handle.lich).close();
476        self.transition(State::Closed);
477        self.log(&format!("session stopped: {stopped_because:?}"));
478        if let Some((log, _)) = &self.player_log {
479            log.notice("The session has ended.");
480            // Its writer ends now, not when the last clone held somewhere
481            // goes: Hydra waits for it before it exits.
482            log.close();
483        }
484        // The recording finished on a thread of its own: a stalled disk holds
485        // it, not the threads the other characters play on. Counted until it
486        // has, so Hydra waits for it before it exits.
487        if let Some(sink) = self.sink.take() {
488            let live = hydra_net::behind::Live::new();
489            let _ = std::thread::Builder::new()
490                .name("hydra-recording".to_owned())
491                .spawn(move || {
492                    let _live = live;
493                    drop(sink);
494                });
495        }
496        SupervisedEnd {
497            stopped_because,
498            state: self.state,
499            recorder: self.recorder,
500        }
501    }
502
503    fn transition(&mut self, next: State) {
504        if self.lifecycle != next {
505            self.lifecycle = next;
506            self.log(&format!("lifecycle {next:?}"));
507            let _ = self.events.send(Event::StateChanged(next));
508        }
509    }
510
511    fn log(&mut self, line: &str) {
512        if let Some(sink) = self.sink.as_mut() {
513            let _ = sink.event(line);
514        }
515    }
516}

518/// Answer `message` while there is no connection: a command is told so, a
519/// line of Lich's is not written, what is known is given, and a quit is
520/// unsent. `true` when it was a quit, and the session is to stop.
521fn answer_between(
522    message: Inbox,
523    snapshot: &dyn Fn() -> Snapshot,
524    events: &broadcast::Sender<Event>,
525) -> bool {
526    match message {
527        Inbox::Command(envelope) => {
528            let _ = envelope.reply.send(Outcome::Disconnected);
529        }
530        // Lich's line: no connection to write it on.
531        Inbox::Now { reply, .. } => {
532            let _ = reply.send(crate::command::Written::No);
533        }
534        Inbox::Snapshot(reply) => {
535            let _ = reply.send(snapshot());
536        }
537        Inbox::Join(reply) => {
538            let _ = reply.send((snapshot(), events.subscribe()));
539        }
540        // Nothing waits between connections: each is told so.
541        Inbox::Clear(reply) => {
542            let _ = reply.send(0);
543        }
544        // Nothing to send `quit` on: the session just stops.
545        Inbox::Quit { reply, .. } => {
546            let _ = reply.send(Farewell::Unsent);
547            return true;
548        }
549        // No state to fold them into until the next connection, which the
550        // game fills afresh.
551        Inbox::Frames(_) => {}
552    }
553    false
554}

560    #[test]
561    fn the_ladder_climbs_then_holds() {
562        let seconds = |attempt| backoff(attempt, 0.5).as_secs_f64();
563        for (attempt, want) in [
564            (0, 1.0),
565            (1, 2.0),
566            (2, 5.0),
567            (3, 10.0),
568            (4, 30.0),
569            (99, 30.0),
570        ] {
571            assert!((seconds(attempt) - want).abs() < f64::EPSILON, "{attempt}");
572        }
573    }

575    #[test]
576    fn jitter_spans_twenty_percent_each_way_and_is_clamped() {
577        assert_eq!(backoff(4, 0.0), Duration::from_secs(24));
578        assert_eq!(backoff(4, 1.0), Duration::from_secs(36));
579        assert_eq!(backoff(0, -5.0), backoff(0, 0.0));
580        assert_eq!(backoff(0, 5.0), backoff(0, 1.0));
581    }

584    fn the_jitter_varies_and_stays_in_range() {

587        assert!(samples.len() > 100);
588        assert!((0..1000).all(|_| (0.0..=1.0).contains(&jitter())));

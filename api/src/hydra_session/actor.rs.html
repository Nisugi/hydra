1//! The session actor: one character's connection, parser and game state.
2//!
3//! One actor is one connection. It reads what the game sends, folds it into
4//! the game state, and publishes it; it writes what the queue lets through;
5//! and it answers its handles. [`SessionActor::run`] consumes the actor, so a
6//! reconnect is a new actor over a new connection, with the durable parts --
7//! the game state, the recorder, the log, the command channel -- carried
8//! across by the supervisor.
9//!
10//! It also keeps what the character learns: its facts
11//! ([`crate::character_store`]) and the learned menu dictionary
12//! ([`crate::menu_store`]), read when the game names the character and
13//! written off its task (`saves.rs`).
14

33/// How long a read waits before the loop looks at everything else again.
34const READ_DEADLINE: Duration = Duration::from_millis(500);
35/// How long a write may take before the connection is given up on.
36const WRITE_DEADLINE: Duration = Duration::from_secs(5);
37/// How much one read takes.
38const READ_BUF: usize = 8 * 1024;
39/// How long a `quit` typed as a command waits for the game to close.
40const QUIT_EOF_DEADLINE: Duration = Duration::from_secs(10);
41/// How long the player's Lich, handed a quit, has to stop its scripts,
42/// save, and pass the quit on to the game before it is sent here: as long
43/// as Lich gives its own shutdown before it forces itself out.
44pub const LICH_QUIT: Duration = Duration::from_mins(1);

46/// Why a connection ended.
47#[derive(Clone, Copy, Debug, PartialEq, Eq)]
48pub enum EndReason {
49    /// Asked to: a quit, or the session was cancelled.
50    Cancelled,
51    /// The game closed the connection, unasked.
52    PeerClosed,
53    /// Reading failed.
54    ReadFailed,
55    /// Writing failed or took too long.
56    WriteFailed,
57}
58
59impl EndReason {
60    /// Whether the session should open another connection.
61    #[must_use]
62    pub const fn warrants_reconnect(self) -> bool {
63        !matches!(self, Self::Cancelled)
64    }
65}

241    /// Run until the connection ends.
242    pub async fn run(mut self) -> SessionEnd {
243        self.transition(State::Syncing);
244        let mut buf = vec![0u8; READ_BUF];
245        let mut senders_gone = false;
246        let reason = loop {
247            if let Some(failed) = self.pump().await {
248                break failed;
249            }
250            self.begin_lich();
251            // A quit handed to Lich, which has since let go of what is
252            // typed -- detached, or died -- is sent here, not waited on.
253            let lich_let_go =
254                self.lich_quitting.is_some() && !crate::lich::lock(&self.lich).takes_typing();
255            if lich_let_go && let Some(handed) = self.lich_quitting.take() {
256                self.log("quit: Lich let go before passing it on; sent here");
257                if !self.quit_with(handed.timeout, handed.replies).await {
258                    break EndReason::Cancelled;
259                }
260            }
261            tokio::select! {
262                biased;
263                () = self.cancel.cancelled() => {
264                    self.finish_quit(Farewell::TimedOut);
265                    break EndReason::Cancelled;
266                }
267                () = tokio::time::sleep_until(self.save_deadline()), if self.persistence.deadline.is_some() => {
268                    self.save_character();
269                }
270                () = tokio::time::sleep_until(self.quit_deadline()), if self.quitting.is_some() => {
271                    self.log("quit: the game did not close in time");
272                    self.finish_quit(Farewell::TimedOut);
273                    break EndReason::Cancelled;
274                }
275                () = tokio::time::sleep_until(self.lich_quit_deadline()), if self.lich_quitting.is_some() => {
276                    self.log("quit: Lich did not pass it on in time; sent here");
277                    if let Some(handed) = self.lich_quitting.take()
278                        && !self.quit_with(handed.timeout, handed.replies).await
279                    {
280                        break EndReason::Cancelled;
281                    }
282                }
283                received = self.commands.recv(), if !senders_gone => match received {
284                    Some(message) => if let Some(failed) = self.handle_inbox(message).await {
285                        break failed;
286                    },
287                    None => senders_gone = true,
288                },
289                read = tokio::time::timeout(READ_DEADLINE, self.source.read(&mut buf)) => {
290                    match read {
291                        // The game closed the connection: asked to, after a
292                        // quit, or not.
293                        Ok(Ok(0)) => {
294                            self.finish_reading();
295                            break self.closed(EndReason::PeerClosed);
296                        }
297                        Ok(Err(_)) => {
298                            self.finish_reading();
299                            break self.closed(EndReason::ReadFailed);
300                        }
301                        Ok(Ok(n)) => self.ingest(&buf[..n]),
302                        Err(_elapsed) => {}
303                    }
304                }
305            }
306        };
307        self.shutdown(reason).await;
308        SessionEnd {
309            reason,
310            state: self.state,
311            lifecycle: self.lifecycle,
312            recorder: self.recorder,
313            sink: self.sink,
314            attended: self.attended,
315            commands: self.commands,
316            facts_read: self.persistence.loaded,
317        }
318    }

335    /// Bytes from the game: record them, parse them, fold each frame into
336    /// the state, and publish it. Counted as the game text's memory.
337    fn ingest(&mut self, bytes: &[u8]) {
338        hydra_mem::part(hydra_mem::Part::GameText, || self.read_in(bytes));
339    }
340
341    /// [`Self::ingest`]'s work.
342    fn read_in(&mut self, bytes: &[u8]) {
343        let _ = self.events.send(Event::Clock(std::time::Instant::now()));
344        self.recorder.inbound(bytes);
345        crate::lich::lock(&self.lich).wire(bytes);
346        self.log_wire(true, bytes);
347        for frame in self.parser.push_bytes(bytes) {
348            self.take_frame(frame);
349        }
350    }
351
352    /// The connection is over: a last line the game sent with no newline
353    /// after it is read as the rest were.
354    fn finish_reading(&mut self) {
355        for frame in self.parser.finish() {
356            self.take_frame(frame);
357        }
358    }

360    /// A frame published, unless a quiet answer waiting for its first line
361    /// holds the main window's line it is part of: a line's runs split at
362    /// every link, so the line is weighed whole when it ends, and published
363    /// then, quiet from it when it is the answer's first. A prompt lets a
364    /// line still held go as it is.
365    fn publish(&mut self, frame: Frame) {
366        if let Some(hushed) = self.quiet.as_mut()
367            && hushed.start.is_some()
368            && let Frame::Text(text) = &frame
369            && text.stream.is_empty()
370        {
371            hushed.line.push_str(&text.content);
372            let ends = text.ends_line;
373            hushed.held.push(frame);
374            if ends {
375                let first = hushed
376                    .start
377                    .as_deref()
378                    .is_some_and(|start| hushed.line.contains(start));
379                self.release_held(first);
380            }
381            return;
382        }
383        if matches!(frame, Frame::Prompt { .. }) {
384            self.release_held(false);
385        }
386        let _ = self.events.send(Event::Frame(Box::new(frame)));
387    }
388
389    /// The line a quiet answer held, published: quiet from it when `first`,
390    /// the answer's first line.
391    fn release_held(&mut self, first: bool) {
392        let Some(hushed) = self.quiet.as_mut() else {
393            return;
394        };
395        let held = std::mem::take(&mut hushed.held);
396        hushed.line.clear();
397        if first {
398            hushed.start = None;
399            let _ = self.events.send(Event::Quiet(true));
400        }
401        for frame in held {
402            let _ = self.events.send(Event::Frame(Box::new(frame)));
403        }
404    }

406    /// A prompt, for a quiet question's answer under way: the prompt after
407    /// its answer ends it; one already on its way when the question went,
408    /// before any answer, does not, unless the answer is long overdue. An
409    /// answer whose first line never came hid nothing.
410    fn quiet_prompted(&mut self) {
411        let Some(hushed) = &self.quiet else {
412            return;
413        };
414        let overdue = hushed.since.elapsed() >= QUIET_MOST;
415        if hushed.start.is_some() {
416            if overdue {
417                self.quiet = None;
418            }
419        } else if hushed.heard || overdue {
420            self.quiet = None;
421            let _ = self.events.send(Event::Quiet(false));
422        }
423    }

425    /// One frame of the game's: offered to the queue's window, logged,
426    /// folded into the state, and published.
427    fn take_frame(&mut self, frame: Frame) {
428        {
429            if self.queue.window_is_open() {
430                self.queue.offer(&frame);
431            }
432            self.feed.frame(&frame);
433            let terminator = self.state.apply(&frame);
434            // The game has named the character: there is a file to read,
435            // and a lookout to show its lines to.
436            if matches!(frame, Frame::AppInfo { .. }) {
437                self.read_stored();
438                self.watch();
439            }
440            let taught = self.state.character.take_taught();
441            self.persistence.mark(taught);
442            // A `<cmdlist>` push, or its `<cmdtimestamp>`: the game
443            // correcting the dictionary, kept the moment it arrives.
444            if let Some(pushed) = pushed_commands(&frame) {
445                self.keep_learned_commands(pushed);
446            }
447            let ready = self.lifecycle == State::Syncing
448                && matches!(frame, Frame::EndSetup | Frame::Prompt { .. });
449            let prompt = match &frame {
450                Frame::Prompt { text, .. } => Some(text.clone()),
451                _ => None,
452            };
453            let main_text = matches!(&frame, Frame::Text(text)
454                if text.stream.is_empty() && !text.content.trim().is_empty());
455            self.publish(frame);
456            if let Some(hushed) = self.quiet.as_mut()
457                && hushed.start.is_none()
458                && main_text
459            {
460                hushed.heard = true;
461            }
462            if let Some(prompt) = prompt {
463                let _ = self.events.send(Event::Prompt(prompt));
464                self.quiet_prompted();
465            }
466            if terminator {
467                self.queue.close_window();
468                let facts = self.state.combat_mut().take_facts();
469                // What the combat reader found its lines were: for the
470                // stream rules, and for a frontend to colour them by.
471                let marks: Vec<_> = facts
472                    .iter()
473                    .flat_map(|facts| facts.marks.iter().cloned())
474                    .collect();
475                // The chunk is classified, so its lines can be sorted and
476                // written (`player_log/feed.rs`).
477                let chunk = crate::player_log::feed::Chunk {
478                    fighting: crate::stream_rules::fought(&facts),
479                    hostile: self.state.creatures().targets().next().is_some(),
480                    looted: self
481                        .state
482                        .loot
483                        .events()
484                        .iter()
485                        .any(hydra_model::state::loot::LootEvent::is_search_or_skin),
486                    ours: crate::stream_rules::ours(&facts),
487                };
488                let brief = self.feed.brief(&facts);
489                self.feed.close(chunk, &marks);
490                // The facts whole, for a recorder.
491                for facts in facts {
492                    if !facts.is_empty() {
493                        let _ = self.events.send(Event::Combat(crate::event::Facts(
494                            std::sync::Arc::new(facts),
495                        )));
496                    }
497                }
498                if !marks.is_empty() {
499                    let _ = self.events.send(Event::Marked(marks));
500                }
501                for event in self.state.loot.take_events() {
502                    let _ = self.events.send(Event::Loot(Box::new(event)));
503                }
504                // Where the stream rules put the chunk's lines, for the
505                // windows: after the marks, so a line marked goes to its
506                // stream marked.
507                let sorted = self.feed.take_sorted();
508                if !sorted.is_empty() {
509                    let _ = self.events.send(Event::Sorted(sorted));
510                }
511                // The fight told briefly, after its sorts: the story tells
512                // it in its lines' place, wherever they were put.
513                if let Some(brief) = brief {
514                    let _ = self.events.send(Event::Brief(Box::new(brief)));
515                }
516                // What the lookout found new, after the stream rules' word:
517                // a line they moved is found where they put it.
518                let (lines, mark) = self.feed.take_noticed();
519                if !lines.is_empty() {
520                    let _ = self.events.send(Event::Lookout { lines, mark });
521                }
522                let _ = self.events.send(Event::ChunkDone);
523            }
524            // The burst is over at `<endSetup/>`, or at the first prompt if
525            // the game never says so.
526            if ready {
527                self.transition(State::Ready);
528            }
529        }
530    }

532    /// One message from a handle. `Some` when the connection must end.
533    async fn handle_inbox(&mut self, message: Inbox) -> Option<EndReason> {
534        match message {
535            Inbox::Command(envelope) => {
536                if envelope.reply.is_closed() {
537                    return None;
538                }
539                // Meant for a connection that has since dropped.
540                if envelope.generation != self.generation {
541                    let _ = envelope.reply.send(Outcome::Disconnected);
542                    return None;
543                }
544                if is_exit_intent(&envelope.line) {
545                    let (tx, _rx) = oneshot::channel();
546                    let sent = self.begin_quit(QUIT_EOF_DEADLINE, tx).await;
547                    let _ = envelope.reply.send(if sent {
548                        Outcome::Disconnected
549                    } else {
550                        Outcome::Dead
551                    });
552                    return (!sent).then_some(EndReason::Cancelled);
553                }
554                if self.quitting.is_some() {
555                    let _ = envelope.reply.send(Outcome::Dead);
556                } else {
557                    self.queue.admit(envelope);
558                }
559            }
560            Inbox::Now {
561                line,
562                origin,
563                quiet,
564                generation,
565                reply,
566            } => return self.write_now(line, origin, quiet, generation, reply).await,
567            Inbox::Snapshot(reply) => {
568                let _ = reply.send(self.snapshot());
569            }
570            // The frames are this actor's to publish, so a stream taken here
571            // starts exactly after the snapshot.
572            Inbox::Join(reply) => {
573                let _ = reply.send((self.snapshot(), self.events.subscribe()));
574            }
575            Inbox::Clear(reply) => {
576                let _ = reply.send(self.queue.clear_waiting());
577            }
578            Inbox::Quit { timeout, reply } => {
579                if let Some(handed) = self.lich_quitting.as_mut() {
580                    handed.replies.push(reply);
581                    return None;
582                }
583                if self.quitting.is_none() && self.hand_quit_to_lich() {
584                    self.lich_quitting = Some(LichQuitting {
585                        deadline: tokio::time::Instant::now() + LICH_QUIT,
586                        timeout,
587                        replies: vec![reply],
588                    });
589                    return None;
590                }
591                if !self.begin_quit(timeout, reply).await {
592                    return Some(EndReason::Cancelled);
593                }
594            }
595            Inbox::Frames(frames) => {
596                for frame in frames {
597                    self.state.apply(&frame);
598                    // Not the game's: a dialog these open is a script's.
599                    self.state.dialogs.scripted(&frame);
600                    let _ = self.events.send(Event::Frame(Box::new(frame)));
601                }
602            }
603        }
604        None
605    }

607    /// A Lich attached since the last turn is handed a login built from the
608    /// model as it stands, before the next chunk is read, so it misses
609    /// nothing and is told nothing twice: nothing, before the game has named
610    /// the character, when the login itself is still to come.
611    fn begin_lich(&mut self) {
612        let mut slot = crate::lich::lock(&self.lich);
613        if slot.waits() {
614            let login = self.state.login();
615            slot.begin(login.as_bytes());
616            let _ = self.events.send(Event::LichBegun {
617                login: !login.is_empty(),
618            });
619        }
620    }

622    /// Write a line Lich sent, at once. `Some` when the write failed.
623    async fn write_now(
624        &mut self,
625        line: String,
626        origin: Origin,
627        quiet: Option<Hush>,
628        generation: Generation,
629        reply: oneshot::Sender<Written>,
630    ) -> Option<EndReason> {
631        let quiet_asked = quiet.is_some();
632        // Meant for a connection that has since dropped, or after a quit.
633        if generation != self.generation || self.quitting.is_some() {
634            let _ = reply.send(Written::No);
635            return None;
636        }
637        // The player's `quit`, passed on by Lich: a quit, not a loss.
638        if is_exit_intent(&line) {
639            let (tx, _rx) = oneshot::channel();
640            let sent = self.begin_quit(QUIT_EOF_DEADLINE, tx).await;
641            let _ = reply.send(if sent { Written::Yes } else { Written::No });
642            return (!sent).then_some(EndReason::Cancelled);
643        }
644        if let Err(reason) = self.write_line(&line, origin).await {
645            let _ = reply.send(Written::No);
646            return Some(reason);
647        }
648        // Its answer is its own, never a queued command's.
649        self.queue.written_at_once();
650        if origin == Origin::Player {
651            self.attended = true;
652        }
653        let _ = self.events.send(Event::Clock(std::time::Instant::now()));
654        // The line written is left out either way; an answer that waits for
655        // its first line shows what comes before it.
656        let start = match quiet {
657            Some(Hush::From(start)) => Some(start),
658            Some(Hush::Written) | None => None,
659        };
660        let waits = start.is_some();
661        if quiet_asked {
662            self.quiet = Some(Hushed {
663                since: std::time::Instant::now(),
664                heard: false,
665                start,
666                line: String::new(),
667                held: Vec::new(),
668            });
669            let _ = self.events.send(Event::Quiet(true));
670        }
671        let _ = self.events.send(Event::Sent { line, origin });
672        if waits {
673            let _ = self.events.send(Event::Quiet(false));
674        }
675        let _ = reply.send(Written::Yes);
676        None
677    }

679    /// Write what the queue lets through. `Some` when a write failed.
680    async fn pump(&mut self) -> Option<EndReason> {
681        if self.quitting.is_some() {
682            return None;
683        }
684        while let Some(envelope) = self.queue.take_next() {
685            let Envelope {
686                line,
687                origin,
688                generation,
689                reply,
690            } = envelope;
691            if generation != self.generation {
692                let _ = reply.send(Outcome::Disconnected);
693                continue;
694            }
695            if let Err(reason) = self.write_line(&line, origin).await {
696                let _ = reply.send(Outcome::Disconnected);
697                return Some(reason);
698            }
699            if origin == Origin::Player {
700                self.attended = true;
701            }
702            let _ = self.events.send(Event::Clock(std::time::Instant::now()));
703            let _ = self.events.send(Event::Sent { line, origin });
704            self.queue.open_window(reply);
705        }
706        None
707    }
708
709    /// Write one line, as one write, within the deadline: logged as
710    /// `origin` sent it.
711    async fn write_line(&mut self, line: &str, origin: Origin) -> Result<(), EndReason> {
712        let mut message = Vec::with_capacity(line.len() + 1);
713        message.extend_from_slice(line.as_bytes());
714        message.push(b'\n');
715        // Anything the player sends answers the game's idle warning.
716        self.state.answer_idle_warning();
717        let wrote = tokio::select! {
718            biased;
719            () = self.cancel.cancelled() => return Err(EndReason::Cancelled),
720            wrote = tokio::time::timeout(WRITE_DEADLINE, self.source.write_all(&message)) => wrote,
721        };
722        match wrote {
723            Ok(Ok(())) => {
724                self.recorder.outbound(&message);
725                self.log_wire(false, &message);
726                self.feed
727                    .command(&String::from_utf8_lossy(&message), origin);
728                Ok(())
729            }
730            Ok(Err(_)) | Err(_) => Err(EndReason::WriteFailed),
731        }
732    }

734    /// Hand `quit` to the player's Lich, as if typed for it, when one is
735    /// reading what is typed: it stops its scripts, runs what each does as
736    /// it ends, saves, and then passes the quit on, which ends the session
737    /// as a typed quit does. Whether it took it.
738    fn hand_quit_to_lich(&mut self) -> bool {
739        let lich = crate::lich::lock(&self.lich);
740        let handed = lich.takes_typing() && lich.hand("quit") == Some(true);
741        drop(lich);
742        if handed {
743            self.log("quit: handed to Lich, to stop its scripts and save first");
744        }
745        handed
746    }

749    fn lich_quit_deadline(&self) -> tokio::time::Instant {
750        self.lich_quitting.as_ref().map_or_else(
751            || tokio::time::Instant::now() + Duration::from_hours(1),
752            |handed| handed.deadline,
753        )

756    /// Send `quit` for every one of `replies`, waiting `timeout` for the game
757    /// to close. `false` when it could not be sent.
758    async fn quit_with(
759        &mut self,
760        timeout: Duration,
761        mut replies: Vec<oneshot::Sender<Farewell>>,
762    ) -> bool {
763        let Some(first) = replies.pop() else {
764            let (first, _) = oneshot::channel();
765            return self.begin_quit(timeout, first).await;
766        };
767        let sent = self.begin_quit(timeout, first).await;
768        match self.quitting.as_mut() {
769            Some(pending) => pending.replies.extend(replies),
770            None => {
771                for reply in replies {
772                    let _ = reply.send(Farewell::Unsent);
773                }
774            }
775        }
776        sent
777    }

779    /// Send `quit` and wait for the game to close, until `timeout`. `false`
780    /// when it could not be sent. A quit Lich was handed and has now passed
781    /// on: its waiters wait on this one.
782    async fn begin_quit(&mut self, timeout: Duration, reply: oneshot::Sender<Farewell>) -> bool {
783        if let Some(pending) = self.quitting.as_mut() {
784            pending.replies.push(reply);
785            return true;
786        }
787        let (timeout, mut replies) = match self.lich_quitting.take() {
788            Some(handed) => (timeout.max(handed.timeout), handed.replies),
789            None => (timeout, Vec::new()),
790        };
791        replies.push(reply);
792        if self.write_line("quit", Origin::Player).await.is_err() {
793            self.log("quit: could not send, the connection is gone");
794            for reply in replies {
795                let _ = reply.send(Farewell::Unsent);
796            }
797            return false;
798        }
799        let _ = self.events.send(Event::Sent {
800            line: "quit".to_owned(),
801            origin: Origin::Player,
802        });
803        self.queue.answer_all(&Outcome::Dead);
804        self.quitting = Some(Quitting {
805            deadline: tokio::time::Instant::now() + timeout,
806            replies,
807        });
808        true
809    }
810
811    /// Tell every quit waiter how it ended. `true` when a quit was under way.
812    fn finish_quit(&mut self, farewell: Farewell) -> bool {
813        // A quit Lich still held ends with the connection too.
814        if let Some(handed) = self.lich_quitting.take() {
815            for reply in handed.replies {
816                let _ = reply.send(farewell);
817            }
818        }
819        let Some(pending) = self.quitting.as_mut() else {
820            return false;
821        };
822        for reply in std::mem::take(&mut pending.replies) {
823            let _ = reply.send(farewell);
824        }
825        true
826    }

828    fn quit_deadline(&self) -> tokio::time::Instant {
829        self.quitting.as_ref().map_or_else(
830            || tokio::time::Instant::now() + Duration::from_hours(1),
831            |pending| pending.deadline,
832        )
833    }

835    /// Close the connection and answer every waiter.
836    async fn shutdown(&mut self, reason: EndReason) {
837        self.finish_quit(Farewell::Unsent);
838        let _ = self.source.shutdown().await;
839        let outcome = if reason.warrants_reconnect() {
840            self.on_disconnect.clone()
841        } else {
842            Outcome::Dead
843        };
844        self.queue.answer_all(&outcome);
845        // What the character learned is on disk before the connection says
846        // it has ended.
847        self.save_character();
848        self.saves.finish().await;
849        self.log_saves();
850        // A loss a supervisor will reconnect is not the end of the session.
851        if !(reason.warrants_reconnect() && self.on_disconnect == Outcome::Disconnected) {
852            // The session is over, and the player's Lich's copy with it.
853            crate::lich::lock(&self.lich).close();
854            self.transition(State::Closed);
855        }
856        // Handed over, not waited on: the disk is the recording's threads' to
857        // wait on, never this task's.
858        if let Some(sink) = self.sink.as_mut() {
859            let _ = sink.hand_over();
860        }
861    }

872    /// Read the character's stored facts and the learned dictionary, once a
873    /// session, now the game has said who the character is. Read here, on
874    /// the task: once, small, and before anything could be shown from them.
875    fn read_stored(&mut self) {
876        if self.persistence.loaded {
877            return;
878        }
879        let (Some(dir), Some(name), Some(instance)) = (
880            self.persistence.dir.clone(),
881            self.state.character.name.clone(),
882            self.state.character.instance.clone(),
883        ) else {
884            return;
885        };
886        self.persistence.loaded = true;
887        let said = match crate::character_store::load(&dir, &instance, &name) {
888            Ok(snapshot) if snapshot.restore_into(&mut self.state.character) => {
889                format!("character store: read {instance}_{name}")
890            }
891            Ok(_) => "character store: refused, another schema".to_owned(),
892            Err(why) => format!("character store: {why}"),
893        };
894        self.log(&said);
895        if let Err(why) = crate::menu_store::underlay(&dir, &mut self.state.learned_commands) {
896            self.log(&format!("learned menu commands not read: {why}"));
897        }
898    }

900    /// Show the connection's lines to the lookout of the character's game,
901    /// kept in the data folder, now the game has said who the character is:
902    /// once a connection, and only for a session that keeps a data folder.
903    fn watch(&mut self) {
904        if self.feed.watched() {
905            return;
906        }
907        let (Some(dir), Some(name), Some(instance)) = (
908            self.persistence.dir.as_deref(),
909            self.state.character.name.as_deref(),
910            self.state.character.instance.as_deref(),
911        ) else {
912            return;
913        };
914        let lookout = hydra_lookout::open(dir, instance);
915        self.feed.watch(lookout, name);
916        if !self.feed.learning() {
917            self.feed
918                .learn_into(crate::chunk_kinds::open(dir, instance));
919        }
920    }

922    /// What one push taught, merged into the file, off the task. Only the
923    /// push: the rest of the session's dictionary was read from the file
924    /// when it started, and another session may have kept a newer row since.
925    fn keep_learned_commands(&mut self, pushed: hydra_model::LearnedCommands) {
926        let Some(dir) = self.persistence.dir.clone() else {
927            return;
928        };
929        self.saves.queue(
930            move || match crate::menu_store::merge_and_save(&dir, &pushed) {
931                Ok(None) => None,
932                Ok(Some(path)) => Some(format!(
933                    "learned menu commands written to {}",
934                    path.display()
935                )),
936                Err(why) => Some(format!("learned menu commands not written: {why}")),
937            },
938        );
939        self.log_saves();
940    }

942    /// When the waiting groups are written: far off when none wait.
943    fn save_deadline(&self) -> tokio::time::Instant {
944        self.persistence
945            .deadline
946            .unwrap_or_else(|| tokio::time::Instant::now() + Duration::from_hours(1))
947    }

949    /// Write the groups taught since the last write, merged over what is
950    /// stored, off the task. A failure is logged: the facts are in the model
951    /// either way.
952    fn save_character(&mut self) {
953        let Some(dir) = self.persistence.dir.clone() else {
954            // Nowhere to keep them: not held forever either.
955            self.persistence.take();
956            return;
957        };
958        if self.persistence.groups.is_empty() {
959            self.persistence.deadline = None;
960            return;
961        }
962        let (Some(name), Some(instance)) = (
963            self.state.character.name.clone(),
964            self.state.character.instance.clone(),
965        ) else {
966            // No filename yet; the marks wait for the next write.
967            self.persistence.deadline = None;
968            return;
969        };
970        let groups = self.persistence.take();
971        let snapshot = hydra_model::state::character::snapshot::CharacterSnapshot::of(
972            &name,
973            &instance,
974            &self.state.character,
975            std::collections::BTreeMap::new(),
976        );
977        self.saves.queue(move || {
978            let now = std::time::SystemTime::now();
979            Some(
980                match crate::character_store::merge_and_save(&dir, snapshot, &groups, now) {
981                    Ok(path) => format!(
982                        "character store: wrote {} group(s) to {}",
983                        groups.len(),
984                        path.display()
985                    ),
986                    Err(why) => format!("character store: not written: {why}"),
987                },
988            )
989        });
990        self.log_saves();
991    }

993    /// Log what the saves made so far said.
994    fn log_saves(&mut self) {
995        for line in self.saves.heard() {
996            self.log(&line);
997        }
998    }

1000    fn transition(&mut self, next: State) {
1001        if next == State::Ready {
1002            self.feed.ready(std::time::Instant::now());
1003        }
1004        self.lifecycle = next;
1005        self.log(&format!("lifecycle {next:?}"));
1006        let _ = self.events.send(Event::StateChanged(next));
1007    }
1008
1009    fn log(&mut self, line: &str) {
1010        if let Some(sink) = self.sink.as_mut() {
1011            let _ = sink.event(line);
1012        }
1013    }
1014
1015    fn log_wire(&mut self, inbound: bool, bytes: &[u8]) {
1016        if let Some(sink) = self.sink.as_mut() {
1017            let _ = sink.wire(inbound, bytes);
1018        }
1019    }

1022/// Whether a typed line asks the game to end the session: `quit`, `exit`,
1023/// or their shortest forms.
1024pub(crate) fn is_exit_intent(line: &str) -> bool {
1025    let word = line.trim();
1026    ["quit", "qui", "exit", "exi"]
1027        .iter()
1028        .any(|exit| word.eq_ignore_ascii_case(exit))
1029}

1031/// A session over one connection, with no reconnect: for a replay, a
1032/// recording, a test.
1033#[derive(Debug)]
1034pub struct Session<S: ByteSource> {
1035    actor: SessionActor<S>,
1036    handle: SessionHandle,
1037}
1038
1039impl<S: ByteSource> Session<S> {
1040    /// A session over `source`.
1041    #[must_use]
1042    pub fn new(source: S) -> Self {
1043        let (commands, inbox) = mpsc::channel(COMMAND_CHANNEL_BOUND);
1044        let (events, _) = broadcast::channel(EVENT_CHANNEL_BOUND);
1045        let generation = GenerationCell::default();
1046        let handle = SessionHandle::new(commands, events.clone(), generation.clone());
1047        let mut actor = SessionActor::supervised(
1048            source,
1049            GameState::default(),
1050            inbox,
1051            events,
1052            Recorder::new(),
1053            None,
1054            CancellationToken::new(),
1055            generation.get(),
1056        );
1057        // Nothing reconnects it: a loss ends it.
1058        actor.on_disconnect = Outcome::Dead;
1059        actor.lich = handle.lich.clone();
1060        Self { actor, handle }
1061    }
1062
1063    /// Record to `sink` as well.
1064    #[must_use]
1065    pub fn with_sink(mut self, sink: SessionSink) -> Self {
1066        self.actor.sink = Some(sink);
1067        self
1068    }

1//! Hydra speaking to the player: a message with a **kind** that picks its
2//! colour, prose or a **mono** block for tables, and whether it answers
3//! something the player did. Every feature that tells the player anything
4//! goes through it: a route table, a trip that ends with a sword still
5//! stored, a crossing that stops and says why.
6//!
7//! # Typed, not markup
8//!
9//! A notice is never written as game markup (`<preset id='whisper'>`,
10//! `<output class="mono"/>`, `<pushBold/>`) for the parser to read back.
11//! Hydra is one binary: emitting markup only to parse it again is a round
12//! trip with a forgery in the middle, and it would put text the game never
13//! sent into a stream whose whole value is that the game sent it.
14//!
15//! So a notice is **typed**, and travels as an [`Event`](crate::Event) beside
16//! the frames rather than among them. The terminal prints it; a web frontend
17//! draws a mono block in a fixed-width box; neither parses anything, and
18//! neither can mistake Hydra's voice for the game's.
19//!
20//! # Links
21//!
22//! A table may carry links ([`Body::Linked`]): a creature's name that opens
23//! its bestiary entry, a `[go2]` that walks to its ground, a wiki page.
24//! Each line is spans ([`Span`]), a span's link one of Hydra's own
25//! commands, sent with the command symbol before it; a game command, sent
26//! as the game's; or a web address. A frontend with no links draws the
27//! words alone ([`Notice::lines`]).
28
29/// What a notice is about, which is how a frontend colours it.
30///
31/// These are four meanings, without colour names: what colour an error is
32/// belongs to whoever draws it.
33#[derive(Clone, Copy, Debug, PartialEq, Eq)]
34pub enum NoticeKind {
35    /// Something went wrong, and the player should know.
36    Error,
37    /// Something is off, and nothing has failed yet.
38    Warn,
39    /// Something worth saying: an answer to the player, or what Hydra is
40    /// doing.
41    Info,
42    /// For whoever is chasing a fault. Whether it is shown is the
43    /// frontend's choice, since a log may want what a screen does not.
44    Debug,
45}
46
47/// What a notice says.
48#[derive(Clone, Debug, PartialEq, Eq)]
49pub enum Body {
50    /// Prose, a line at a time. May be re-wrapped by whoever draws it.
51    Lines(Vec<String>),
52    /// Lines whose **columns matter** -- a table. Drawn fixed-width and never
53    /// re-wrapped.
54    Mono(Vec<String>),
55    /// A table whose cells may be links: each line as its spans, and as
56    /// the words alone for a frontend that draws no links.
57    Linked {
58        /// The words of each line.
59        lines: Vec<String>,
60        /// Each line's spans, in order; their text is the line's.
61        spans: Vec<Vec<Span>>,
62    },
63    /// Help: commands and what each does, under headings. Drawn as two
64    /// columns, each description wrapping within its own; `lines` are the
65    /// same padded into fixed-width columns, for whoever draws none.
66    Help {
67        /// The parts padded into lines.
68        lines: Vec<String>,
69        /// The parts, in order.
70        parts: Vec<HelpPart>,
71    },
72}
73
74/// One part of a help notice.
75#[derive(Clone, Debug, PartialEq, Eq)]
76pub enum HelpPart {
77    /// A heading over the commands after it.
78    Heading(String),
79    /// A command, and what it does.
80    Entry {
81        /// The command as typed, without its symbol: `go2 help`,
82        /// `to <name> <line>`.
83        command: String,
84        /// What a click on it runs, without the symbol; none when a click
85        /// would do more than show something.
86        runs: Option<String>,
87        /// What it does.
88        says: String,
89    },
90    /// A line of prose across both columns.
91    Note(String),
92}
93
94/// What a click on a span does.
95#[derive(Clone, Debug, PartialEq, Eq)]
96pub enum NoticeLink {
97    /// One of Hydra's own commands, without its symbol: `bestiary kobold`.
98    Hydra(String),
99    /// A command for the game: `go gate`.
100    Game(String),
101    /// A web address.
102    Url(String),
103}

105/// A piece of a linked line: its words, and what a click on them does.
106#[derive(Clone, Debug, PartialEq, Eq)]
107pub struct Span {
108    /// The words.
109    pub text: String,
110    /// What a click on them does; none for plain words.
111    pub link: Option<NoticeLink>,
112}
113
114impl Span {
115    /// Words that are no link.
116    #[must_use]
117    pub fn plain(text: impl Into<String>) -> Span {
118        Span {
119            text: text.into(),
120            link: None,
121        }
122    }
123
124    /// Words a click sends as one of Hydra's own commands.
125    #[must_use]
126    pub fn hydra(text: impl Into<String>, command: impl Into<String>) -> Span {
127        Span {
128            text: text.into(),
129            link: Some(NoticeLink::Hydra(command.into())),
130        }
131    }
132
133    /// Words a click sends to the game.
134    #[must_use]
135    pub fn game(text: impl Into<String>, command: impl Into<String>) -> Span {
136        Span {
137            text: text.into(),
138            link: Some(NoticeLink::Game(command.into())),
139        }
140    }
141
142    /// Words a click opens as a web page.
143    #[must_use]
144    pub fn url(text: impl Into<String>, href: impl Into<String>) -> Span {
145        Span {
146            text: text.into(),
147            link: Some(NoticeLink::Url(href.into())),
148        }
149    }
150}

152/// One thing Hydra says to the player.
153#[derive(Clone, Debug, PartialEq, Eq)]
154pub struct Notice {
155    /// What it is about, which is how a frontend colours it (and whether it
156    /// shows a `Debug` one at all).
157    pub kind: NoticeKind,
158    /// The words: prose to wrap, or a fixed-width table.
159    pub body: Body,
160    /// It answers something the player did -- a command typed, a room
161    /// shift-clicked on the minimap -- and so is shown among the game's
162    /// lines. Otherwise it is Hydra's own: what it is doing, and what went
163    /// wrong that nobody asked about, in its own pane, a debug and info
164    /// window into Hydra.
165    pub answer: bool,
166}
167
168impl Notice {
169    /// One line of prose.
170    #[must_use]
171    pub fn line(kind: NoticeKind, text: impl Into<String>) -> Notice {
172        Notice::prose(kind, vec![text.into()])
173    }
174
175    /// Prose, `lines` of it.
176    #[must_use]
177    pub fn prose(kind: NoticeKind, lines: Vec<String>) -> Notice {
178        Notice {
179            kind,
180            body: Body::Lines(lines),
181            answer: false,
182        }
183    }
184
185    /// A table, as the lines that make it.
186    #[must_use]
187    pub fn table(kind: NoticeKind, lines: Vec<String>) -> Notice {
188        Notice {
189            kind,
190            body: Body::Mono(lines),
191            answer: false,
192        }
193    }
194
195    /// A table whose cells may be links: each line as its spans.
196    #[must_use]
197    pub fn linked(kind: NoticeKind, spans: Vec<Vec<Span>>) -> Notice {
198        let lines = spans
199            .iter()
200            .map(|line| line.iter().map(|span| span.text.as_str()).collect())
201            .collect();
202        Notice {
203            kind,
204            body: Body::Linked { lines, spans },
205            answer: false,
206        }
207    }
208
209    /// Help, as its parts: padded into lines for whoever draws no columns.
210    #[must_use]
211    pub fn help(kind: NoticeKind, parts: Vec<HelpPart>) -> Notice {
212        let widest = parts
213            .iter()
214            .filter_map(|part| match part {
215                HelpPart::Entry { command, .. } => Some(command.chars().count()),
216                _ => None,
217            })
218            .max()
219            .unwrap_or(0);
220        let lines = parts
221            .iter()
222            .map(|part| match part {
223                HelpPart::Heading(heading) => heading.clone(),
224                HelpPart::Entry { command, says, .. } => {
225                    format!("  {command:<widest$}  {says}")
226                }
227                HelpPart::Note(note) => note.clone(),
228            })
229            .collect();
230        Notice {
231            kind,
232            body: Body::Help { lines, parts },
233            answer: false,
234        }
235    }
236
237    /// Help written as a table, each line a command, two or more spaces,
238    /// and what it does; a line with no such gap is prose. Read into
239    /// [`Notice::help`]'s parts, a command that ends in `help` a link.
240    #[must_use]
241    pub fn help_table(kind: NoticeKind, lines: &[impl AsRef<str>]) -> Notice {
242        let parts = lines
243            .iter()
244            .map(|line| {
245                let line = line.as_ref().trim_end();
246                match line.split_once("  ") {
247                    Some((command, says))
248                        if !command.trim().is_empty() && !says.trim().is_empty() =>
249                    {
250                        let command = command.trim().to_owned();
251                        let runs = command.ends_with(" help").then(|| command.clone());
252                        HelpPart::Entry {
253                            command,
254                            runs,
255                            says: says.trim().to_owned(),
256                        }
257                    }
258                    _ => HelpPart::Note(line.trim().to_owned()),
259                }
260            })
261            .collect();
262        Notice::help(kind, parts)
263    }
264
265    /// This, as an answer to something the player did ([`Notice::answer`]).
266    #[must_use]
267    pub fn answering(mut self) -> Notice {
268        self.answer = true;
269        self
270    }
271
272    /// It says a hunt closed: the combat recorder's word, with the hunt's
273    /// report a click away.
274    #[must_use]
275    pub fn closes_a_hunt(&self) -> bool {
276        let Body::Linked { spans, .. } = &self.body else {
277            return false;
278        };
279        spans.iter().flatten().any(|span| {
280            span.text == "[report]"
281                && matches!(&span.link, Some(NoticeLink::Hydra(command)) if command.starts_with("cstats "))
282        })
283    }
284
285    /// The lines, whichever body holds them: what a plain terminal prints.
286    #[must_use]
287    pub fn lines(&self) -> &[String] {
288        match &self.body {
289            Body::Lines(lines)
290            | Body::Mono(lines)
291            | Body::Linked { lines, .. }
292            | Body::Help { lines, .. } => lines,
293        }
294    }
295
296    /// Whether its columns matter: a table, linked or not.
297    #[must_use]
298    pub fn is_mono(&self) -> bool {
299        matches!(
300            self.body,
301            Body::Mono(_) | Body::Linked { .. } | Body::Help { .. }
302        )
303    }
304}

306#[cfg(test)]
307mod tests {
308    use super::*;
309
310    /// A linked notice's lines are its spans' words, for whoever draws no
311    /// links; a table is mono, linked or not.
312    #[test]
313    fn a_linked_notices_lines_are_its_words() {
314        let notice = Notice::linked(
315            NoticeKind::Info,
316            vec![
317                vec![
318                    Span::plain(" 1  "),
319                    Span::hydra("kobold", "bestiary kobold"),
320                    Span::plain("  "),
321                    Span::game("[go]", "go gate"),
322                ],
323                vec![Span::url("Wiki", "https://gswiki.play.net/kobold")],
324            ],
325        );
326        assert_eq!(notice.lines(), [" 1  kobold  [go]", "Wiki"]);
327        assert!(notice.is_mono());
328        assert!(!Notice::line(NoticeKind::Info, "x").is_mono());
329        assert!(Notice::table(NoticeKind::Info, vec![]).is_mono());
330    }

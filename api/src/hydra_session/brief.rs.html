1//! A fight's chunk told briefly: who attacked, then for each target what
2//! it took and how it was left, in the place of every line the game said.
3//!
4//! The block is the attack's first line, then for each target its rolls
5//! and its flares, the damage it took on the line that ended it (`  .. 45
6//! damage!  The blow shatters ...`), and its state after (`A troll is
7//! KILLED!`). Lines are given to a target as the game names it: a line that
8//! links a creature other than the attacker turns to it, and what the game
9//! indents after it -- what each blow did -- is its until another is named.
10//! A line before the first that names a target is kept as it is, before
11//! the block.
12//!
13//! The combat reader says who attacked ([`Attacker`]) and who died: a
14//! killing blow, or a death the room confirmed; the bestiary knows a
15//! creature's death message. The rest is read off the lines as the game
16//! words them. A frontend chooses for each kind of attacker whether the
17//! story shows the game's lines, the block, or neither, and whether a block
18//! shows its rolls and its flares; the combat stream always has the game's
19//! lines.
20

29/// Who made a fight's attacks.
30#[derive(Clone, Debug, PartialEq, Eq)]
31pub enum Attacker {
32    /// The character.
33    You,
34    /// Another player, by the name the game gave: one of Hydra's own
35    /// characters, or anyone else.
36    Player(String),
37    /// A creature, or a lasting effect no cast owns.
38    Creature,
39}

41/// One creature a block tells of.
42#[derive(Clone, Debug, PartialEq, Eq)]
43pub struct Target {
44    /// Its id, as the game's links carry it.
45    pub exist: String,
46    /// Its noun.
47    pub noun: String,
48    /// Its name, as the combat reader has it.
49    pub name: String,
50}
51
52/// One line of a block.
53#[derive(Clone, Debug, PartialEq, Eq)]
54pub enum BriefLine {
55    /// One of the chunk's lines as it was, by its place among
56    /// [`Brief::lines`].
57    Line(usize),
58    /// A roll's line, shown when the rolls are.
59    Roll(usize),
60    /// A flare's line, shown when the flares are.
61    Flare(usize),
62    /// The damage a target took, before the line its turn ended on, if
63    /// any: `  .. 45 damage!  ` and that line.
64    Damage {
65        /// The points.
66        damage: u32,
67        /// The line, by its place among [`Brief::lines`].
68        then: Option<usize>,
69    },
70    /// How a target was left: `  A troll is stunned, knocked down!`.
71    Left {
72        /// The creature.
73        target: Target,
74        /// What it is, in words; `KILLED` when it died.
75        state: String,
76    },
77}

79/// A fight's chunk told briefly.
80#[derive(Clone, Debug, PartialEq, Eq)]
81pub struct Brief {
82    /// Who attacked.
83    pub attacker: Attacker,
84    /// The chunk's main lines it stands for, by their text, in order: every
85    /// one with words, those kept and those left out.
86    pub lines: Vec<String>,
87    /// The block, in the lines' place.
88    pub block: Vec<BriefLine>,
89    /// The attack's first line, by its place in [`Brief::block`].
90    pub first: usize,
91    /// How many targets beside the first the attack reached: the first
92    /// line says `and 2 others.` when there were more than one.
93    pub others: usize,
94}
95
96impl Brief {
97    /// The chunk's `lines` (each with its links), of a fight `facts` tell
98    /// of, told briefly; `None` when the reader found no attack in it, or
99    /// nothing to tell.
100    #[must_use]
101    pub fn of(lines: &[(&str, &[LogLink])], facts: &[ChunkFacts]) -> Option<Self> {
102        let attacker = attacker(facts)?;
103        let attacker_id = facts
104            .iter()
105            .flat_map(|chunk| chunk.events.iter())
106            .find_map(|event| event.attacker.as_ref().and_then(|actor| actor.id));
107        let dead = dead(facts);
108        let names = names(facts);
109        // A creature the attacks were aimed at, never the attacker, nor a
110        // thing such as the weapon that flared.
111        let creature_in = |links: &[LogLink]| {
112            links.iter().find_map(|link| match &link.does {
113                Does::Object { exist, noun, .. }
114                    if names.iter().any(|(id, _)| id == exist)
115                        && attacker_id.is_none_or(|attacker| exist.parse() != Ok(attacker)) =>
116                {
117                    Some((exist.clone(), noun.clone()))
118                }
119                _ => None,
120            })
121        };
122        // The block begins at the line that names a target, or failing
123        // one -- a creature's attack on the character -- the first that
124        // names anyone.
125        let start = lines
126            .iter()
127            .position(|(_, links)| creature_in(links).is_some())
128            .or_else(|| {
129                lines.iter().position(|(_, links)| {
130                    links
131                        .iter()
132                        .any(|link| matches!(link.does, Does::Object { .. }))
133                })
134            })
135            .unwrap_or(0);
136        let mut builder = Builder::default();
137        // The attack's first line may name its target.
138        if let Some((exist, noun)) = creature_in(lines.get(start)?.1) {
139            builder.turn_to(&exist, &noun, start, false);
140        }
141        for (at, (text, links)) in lines.iter().enumerate().skip(start + 1) {
142            builder.line(at, text, creature_in(links));
143        }
144        let mut block: Vec<BriefLine> = (0..start).map(BriefLine::Line).collect();
145        block.push(BriefLine::Line(start));
146        let first = lines[start].0;
147        for turn in &builder.turns {
148            block.extend(turn.rolls.iter().copied().map(BriefLine::Roll));
149            block.extend(turn.flares.iter().copied().map(BriefLine::Flare));
150            let mut said: Vec<usize> = if turn.aimed.is_empty() {
151                turn.last
152                    .filter(|at| lines[*at].0 != first)
153                    .into_iter()
154                    .collect()
155            } else {
156                let mut aimed = turn.aimed.clone();
157                aimed.dedup_by(|a, b| lines[*a].0 == lines[*b].0);
158                aimed
159            };
160            said.retain(|at| *at != start);
161            let mut said = said.into_iter();
162            if turn.damage > 0 {
163                block.push(BriefLine::Damage {
164                    damage: turn.damage,
165                    then: said.next(),
166                });
167            }
168            let shown: Vec<usize> = said.collect();
169            let told_down = shown.iter().any(|at| {
170                let text = lines[*at].0;
171                tells_down(text)
172            });
173            block.extend(shown.into_iter().map(BriefLine::Line));
174            let Some(exist) = &turn.exist else {
175                continue;
176            };
177            if let Some(state) = turn.left(&dead, told_down) {
178                block.push(BriefLine::Left {
179                    target: Target {
180                        exist: exist.clone(),
181                        noun: turn.noun.clone(),
182                        name: names
183                            .iter()
184                            .find(|(id, _)| id == exist)
185                            .map_or_else(|| turn.noun.clone(), |(_, name)| name.clone()),
186                    },
187                    state,
188                });
189            }
190        }
191        block.extend(builder.bounty.map(BriefLine::Line));
192        let targets = builder
193            .turns
194            .iter()
195            .filter(|turn| turn.exist.is_some())
196            .count();
197        Some(Self {
198            attacker,
199            lines: lines.iter().map(|(text, _)| (*text).to_owned()).collect(),
200            block,
201            first: start,
202            others: targets.saturating_sub(1),
203        })
204    }
205}

207/// Who made the attacks `facts` tell of: the character, when any is its
208/// own; else another player who made one; else a creature. `None` when
209/// there was no attack.
210fn attacker(facts: &[ChunkFacts]) -> Option<Attacker> {
211    let events: Vec<_> = facts.iter().flat_map(|chunk| chunk.events.iter()).collect();
212    if events.is_empty() {
213        return None;
214    }
215    if events.iter().any(|event| event.is_ours()) {
216        return Some(Attacker::You);
217    }
218    let player = events.iter().find_map(|event| {
219        event
220            .attacker
221            .as_ref()
222            .filter(|actor| actor.is_player() || event.foreign_caster)
223            .filter(|_| !event.inbound)
224    });
225    Some(player.map_or(Attacker::Creature, |actor| {
226        Attacker::Player(actor.name.clone())
227    }))
228}

230/// The creatures `facts` say died, by id.
231fn dead(facts: &[ChunkFacts]) -> Vec<String> {
232    let mut dead = Vec::new();
233    for chunk in facts {
234        for event in &chunk.events {
235            if event
236                .all_hits()
237                .any(|hit| hit.crit.as_ref().is_some_and(|crit| crit.fatal))
238                && let Some(id) = event.target.id()
239            {
240                dead.push(id.to_string());
241            }
242        }
243        for fact in &chunk.facts {
244            if let Fact::Dead { creature } = fact
245                && let Some(id) = creature.id
246            {
247                dead.push(id.to_string());
248            }
249        }
250    }
251    dead
252}
253
254/// The creatures `facts` name as targets, by id, with their names.
255fn names(facts: &[ChunkFacts]) -> Vec<(String, String)> {
256    facts
257        .iter()
258        .flat_map(|chunk| chunk.events.iter())
259        .filter_map(|event| match &event.target {
260            EventTarget::Creature(actor) => actor.id.map(|id| (id.to_string(), actor.name.clone())),
261            _ => None,
262        })
263        .collect()
264}

266/// What a target took in a chunk, line by line.
267#[derive(Debug, Default)]
268struct Turn {
269    /// The creature, by id; `None` for the turn before any is named.
270    exist: Option<String>,
271    noun: String,
272    damage: u32,
273    rolls: Vec<usize>,
274    flares: Vec<usize>,
275    /// The last line said of it.
276    last: Option<usize>,
277    /// Lines of a blow to a part of it.
278    aimed: Vec<usize>,
279    states: Vec<String>,
280}
281
282impl Turn {
283    /// How its creature was left, in words: `KILLED` when `dead` names it
284    /// or a line said so; else its states, unless a line said it was put
285    /// down (`told_down`); `None` with nothing to say.
286    fn left(&self, dead: &[String], told_down: bool) -> Option<String> {
287        let exist = self.exist.as_ref()?;
288        if dead.contains(exist) || self.states.iter().any(|state| state == "dead") {
289            return Some("KILLED".to_owned());
290        }
291        if self.states.is_empty() || told_down {
292            return None;
293        }
294        let mut states: Vec<&str> = Vec::new();
295        for state in &self.states {
296            let word = if state == "knockdown" {
297                "knocked down"
298            } else {
299                state.as_str()
300            };
301            if !states.contains(&word) {
302                states.push(word);
303            }
304        }
305        Some(states.join(", "))
306    }
307}

309/// A chunk's lines given to the targets they tell of.
310#[derive(Debug, Default)]
311struct Builder {
312    turns: Vec<Turn>,
313    /// The turn under way, by its place in `turns`.
314    at: Option<usize>,
315    /// Rolls read before a target was named, given to the next.
316    rolls: Vec<usize>,
317    bounty: Option<usize>,
318}
319
320impl Builder {
321    /// The turn under way, one begun for no one when none is.
322    fn turn(&mut self) -> &mut Turn {
323        let at = *self.at.get_or_insert_with(|| {
324            self.turns.push(Turn::default());
325            self.turns.len() - 1
326        });
327        &mut self.turns[at]
328    }
329
330    /// The creature `exist`, of `noun`, named on line `at`: its turn, one
331    /// begun when it has none; the line its last said unless it `flared`.
332    fn turn_to(&mut self, exist: &str, noun: &str, at: usize, flared: bool) {
333        let found = self
334            .turns
335            .iter()
336            .position(|turn| turn.exist.as_deref() == Some(exist));
337        let index = found.unwrap_or_else(|| {
338            self.turns.push(Turn {
339                exist: Some(exist.to_owned()),
340                noun: noun.to_owned(),
341                ..Turn::default()
342            });
343            self.turns.len() - 1
344        });
345        self.at = Some(index);
346        let rolls = std::mem::take(&mut self.rolls);
347        let turn = &mut self.turns[index];
348        turn.rolls.extend(rolls);
349        if flared {
350            turn.flares.push(at);
351        } else {
352            turn.last = Some(at);
353        }
354    }
355
356    /// Line `at`, `text`, which links the creature `named` if any.
357    fn line(&mut self, at: usize, text: &str, named: Option<(String, String)>) {
358        if text.trim().is_empty() || text.starts_with("Cast Roundtime") {
359            return;
360        }
361        if bounty_in(text) {
362            self.bounty = Some(at);
363            return;
364        }
365        let states = states_in(text);
366        if let Some(damage) = damage_in(text) {
367            self.give_rolls();
368            let turn = self.turn();
369            turn.damage = turn.damage.saturating_add(damage);
370            turn.states.extend(states);
371            return;
372        }
373        if let Some((exist, noun)) = named {
374            let died = dies(&noun, text);
375            self.turn_to(&exist, &noun, at, text.contains("**"));
376            self.turn().states.extend(states);
377            if died {
378                self.turn().states.push("dead".to_owned());
379            }
380            if aimed(text) {
381                self.turn().aimed.push(at);
382            }
383            return;
384        }
385        if text.contains("**") {
386            self.give_rolls();
387            self.turn().flares.push(at);
388        } else if rolls_in(text) {
389            self.give_rolls();
390            self.rolls.push(at);
391        } else if text.starts_with(char::is_whitespace) && !passed_over(text) {
392            // The game indents what a blow did; a line of its own that
393            // names no one -- a spell wearing off -- is no target's word.
394            self.turn().last = Some(at);
395            self.turn().states.extend(states);
396        } else {
397            self.turn().states.extend(states);
398        }
399        if aimed(text) {
400            self.turn().aimed.push(at);
401        }
402    }
403
404    /// The rolls read since a target was named, given to the one under
405    /// way, when there is one.
406    fn give_rolls(&mut self) {
407        if self.at.is_some() && !self.rolls.is_empty() {
408            let rolls = std::mem::take(&mut self.rolls);
409            self.turn().rolls.extend(rolls);
410        }
411    }
412}

414/// Whether `text` is a death message of the creature `noun` names, as the
415/// bestiary has them.
416fn dies(noun: &str, text: &str) -> bool {
417    use hydra_model::state::creature::{self, MessageKind};
418    creature::by_noun(noun).into_iter().any(|creature| {
419        hydra_model::state::creature_message::classify(creature, text, &[MessageKind::Death])
420            .is_some()
421    })
422}

424/// A pattern of Hydra's own, compiled once into `cell`; one that does not
425/// compile finds nothing.
426fn pattern(cell: &'static OnceLock<Option<Regex>>, pattern: &str) -> Option<&'static Regex> {
427    cell.get_or_init(|| hydra_model::pattern::compile(pattern).ok())
428        .as_ref()
429}
430
431/// Whether the pattern in `cell` finds `text`.
432fn finds(cell: &'static OnceLock<Option<Regex>>, written: &str, text: &str) -> bool {
433    pattern(cell, written).is_some_and(|found| found.is_match(text))
434}

436/// The damage a line states: `... and hits for 45 points of damage!`.
437fn damage_in(text: &str) -> Option<u32> {
438    static DAMAGE: OnceLock<Option<Regex>> = OnceLock::new();
439    pattern(&DAMAGE, r"(?i)(\d+) (?:points? of )?damage")?
440        .captures(text)?
441        .get(1)?
442        .as_str()
443        .parse()
444        .ok()
445}
446
447/// Whether a line is one of rolls.
448fn rolls_in(text: &str) -> bool {
449    static ROLLS: OnceLock<Option<Regex>> = OnceLock::new();
450    finds(&ROLLS, r"CS:|AS:|UAF:|d100", text)
451}
452
453/// Whether a line is a bounty's word on a kill.
454fn bounty_in(text: &str) -> bool {
455    static BOUNTY: OnceLock<Option<Regex>> = OnceLock::new();
456    finds(
457        &BOUNTY,
458        r"You succeeded in your task|kills? remaining",
459        text,
460    )
461}
462
463/// Whether a line is one that is never a target's last word.
464fn passed_over(text: &str) -> bool {
465    static PASSED: OnceLock<Option<Regex>> = OnceLock::new();
466    finds(
467        &PASSED,
468        r"Roundtime|Forcing stance down|appears to gain succour|comdemnation lifts|Feeling nervous yet|A hit|Warding failed|breaks into tiny fragments|gain succor|^\s*You |(?:arrow|bolt) sticks|lands solidly in|fragrant haze",
469        text,
470    )
471}
472
473/// Whether a line tells a target was put down, so its state need not be
474/// said again.
475fn tells_down(text: &str) -> bool {
476    static DOWN: OnceLock<Option<Regex>> = OnceLock::new();
477    finds(
478        &DOWN,
479        r"knocked|webbing|buffeted|stunned|to the (?:ground|floor)",
480        text,
481    )
482}
483
484/// Whether a line tells of a blow to a part of the body, the line a
485/// target's turn is best told by.
486fn aimed(text: &str) -> bool {
487    static PART: OnceLock<Option<Regex>> = OnceLock::new();
488    static NOT: OnceLock<Option<Regex>> = OnceLock::new();
489    finds(
490        &PART,
491        r"(?i)head|neck|eye|left hand|right hand|left leg|right leg|finger|solar plexus|elbow|abdomen|chest|skull|heart|back|brain|jugular|stomach|nose|forearm|knee|sliced open",
492        text,
493    ) && !finds(
494        &NOT,
495        r"(?i)(?:arrow|bolt) sticks|lands solidly in|cutting invisible runes",
496        text,
497    )
498}

500/// The states a line puts its target in, by name.
501fn states_in(text: &str) -> Vec<String> {
502    static STATES: OnceLock<Vec<(&'static str, Regex)>> = OnceLock::new();
503    STATES
504        .get_or_init(|| {
505            [
506                ("stunned", r"stunned|strength of holy incantation"),
507                (
508                    "frozen",
509                    r"freezes|encased in a thick block of ice|stops all movement",
510                ),
511                (
512                    "knockdown",
513                    r"falls over|(?:dragged|knocked|down|flattening itself) to the (?:\w+)|collapses on",
514                ),
515                ("sympathized", r"eyes begin to glow (?:purple|dark)"),
516                ("pinned", r"pins? (?:.*) to the"),
517                ("webbed", r"ensnared in thick strands of webbing"),
518                ("buffeted", r"buffeted by"),
519                (
520                    "dead",
521                    r"tries to crawl away on the (?:ground|floor) but|rolls over on the (?:ground|floor) and goes still|body falls to the (?:ground|floor) as it is consumed by ethereal flame|(?:collapses|crashes) to the (?:ground|floor)|grows dim as s?he falls to the (?:ground|floor)|falls to the (?:ground|floor) motionless|lets out a ragged gasp before collapsing",
522                ),
523            ]
524            .into_iter()
525            .filter_map(|(name, written)| {
526                hydra_model::pattern::compile(written)
527                    .ok()
528                    .map(|found| (name, found))
529            })
530            .collect()
531        })
532        .iter()
533        .filter(|(_, found)| found.is_match(text))
534        .map(|(name, _)| (*name).to_owned())
535        .collect()
536}

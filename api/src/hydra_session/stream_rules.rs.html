1//! Stream rules: which stream a line of the game's belongs to, as the player
2//! sorts them, beside the one the game sent it to. One list for every
3//! character, [`FILE`] in the data folder; the first rule that matches a
4//! line sorts it.
5//!
6//! A rule matches a line by its words or a regular expression, and by what
7//! Hydra knows of it: that it came in a chunk the combat reader found
8//! fighting in, that the reader marked it (a flare, a status landing on the
9//! character, a killing blow), or that the bestiary knows it as one of a
10//! creature's messages of the kinds named (its idle flavour, its arrival,
11//! its death...). A rule that names a stream looks only at the lines the
12//! game sent there. Every condition a rule gives must hold.
13//!
14//! A line a rule matches is logged under its stream after the one the game
15//! sent it to, `main/ambients`: where it came from, and where the player
16//! put it. The windows are told it too ([`crate::Event::Sorted`]): it goes
17//! to its stream's windows, and leaves the stream the game sent it to unless
18//! the rule copies it.
19//!
20//! With no file, the rules are [`Rules::default`]: the one built in, every
21//! main line of a chunk with fighting in it to `combat`, kept in main too.
22//! A file is the whole list: the built-in rule is in it to edit, or not.
23//!
24//! A rule can be made from a line the player points at ([`Rule::like`]):
25//! its words as they are, with the names in it and its numbers loosened, so
26//! it takes the lines like it.
27//!
28//! The log's past lines are sorted again as they are read back
29//! ([`Compiled::sort_past`]): a line written before a rule that takes it was
30//! made is shown in its stream, though its file says otherwise.
31

48/// One rule: what lines it matches, and the stream it puts them in.
49#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
50#[serde(deny_unknown_fields)]
51#[expect(
52    clippy::struct_excessive_bools,
53    reason = "each a condition or a choice of its own, as the file writes it"
54)]
55pub struct Rule {
56    /// What the player calls it.
57    pub name: String,
58    /// Words the line holds, whatever their case; `|` between words any of
59    /// which will do.
60    #[serde(default, skip_serializing_if = "Option::is_none")]
61    pub text: Option<String>,
62    /// A regular expression the line matches, as written: case matters
63    /// unless it says `(?i)`.
64    #[serde(default, skip_serializing_if = "Option::is_none")]
65    pub regex: Option<String>,
66    /// Only the lines the game sent to this stream: `main`, `thoughts`.
67    #[serde(default, skip_serializing_if = "Option::is_none")]
68    pub from: Option<String>,
69    /// Only the lines of a chunk the combat reader found fighting in.
70    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
71    pub fighting: bool,
72    /// Only the lines the combat reader marked: a flare, a status landing
73    /// on the character, a killing blow.
74    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
75    pub marked: bool,
76    /// Only a creature's messages of these kinds, as the bestiary has them:
77    /// `ambient`, `arrival`, `death`, `flee`, `spell_prep`...
78    #[serde(default, skip_serializing_if = "Vec::is_empty")]
79    pub creature: Vec<String>,
80    /// The stream it puts them in: one the game sends, or one of the
81    /// player's own.
82    pub to: String,
83    /// Kept in the stream the game sent them to as well.
84    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
85    pub copy: bool,
86    /// Not used.
87    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
88    pub off: bool,
89}

91impl Rule {
92    /// The built-in rule: every main line of a chunk with fighting in it to
93    /// [`COMBAT`], kept in main too.
94    #[must_use]
95    pub fn fighting() -> Self {
96        Self {
97            name: "Combat".to_owned(),
98            text: None,
99            regex: None,
100            from: Some(crate::player_log::MAIN.to_owned()),
101            fighting: true,
102            marked: false,
103            creature: Vec::new(),
104            to: COMBAT.to_owned(),
105            copy: true,
106            off: false,
107        }
108    }
109
110    /// A rule for the lines like `text` that puts them in `to`: matched by
111    /// [`like`], from the stream `from` alone when it is known which the
112    /// game sent the line to, moved. `names` are where the line names a
113    /// creature, a player or a thing, by byte: where the game linked one.
114    #[must_use]
115    pub fn like(from: Option<&str>, text: &str, names: &[Range<usize>], to: &str) -> Self {
116        let said = text.trim();
117        let mut name: String = said.chars().take(40).collect();
118        if said.chars().count() > 40 {
119            name = format!("{}...", name.trim_end());
120        }
121        Self {
122            name: format!("Like \"{name}\""),
123            text: None,
124            regex: Some(like(text, names)),
125            from: from.map(str::to_owned),
126            fighting: false,
127            marked: false,
128            creature: Vec::new(),
129            to: to.to_owned(),
130            copy: false,
131            off: false,
132        }
133    }
134
135    /// Whether it can be used.
136    ///
137    /// # Errors
138    ///
139    /// Why it cannot: as a rules file read is told why each it leaves out.
140    pub fn check(&self) -> Result<(), String> {
141        Built::of(self).map(|_| ())
142    }
143
144    /// Whether its words or its regex find `text`; true when it gives
145    /// neither. What else it asks of a line is known only as lines come.
146    ///
147    /// # Errors
148    ///
149    /// Why it cannot be used.
150    pub fn finds(&self, text: &str) -> Result<bool, String> {
151        found_by(self).map(|find| find.is_none_or(|find| find.is_match(text)))
152    }
153}

155/// A regular expression for the lines like `text`: its words as they are,
156/// each of `names` -- where it names a creature, a player or a thing, by
157/// byte -- any name, the article before one with it, and each number any
158/// number. Anchored at both ends, so it takes whole lines. A line with no
159/// words left but names is matched as it is.
160#[must_use]
161pub fn like(text: &str, names: &[Range<usize>]) -> String {
162    let body = text.trim_end();
163    let mut names: Vec<&Range<usize>> = names
164        .iter()
165        .filter(|name| {
166            name.start < name.end
167                && body.is_char_boundary(name.start)
168                && body.get(name.start..name.end).is_some()
169        })
170        .collect();
171    names.sort_by_key(|name| name.start);
172    let mut pattern = String::new();
173    let mut words = false;
174    let mut at = 0;
175    for name in names {
176        if name.start < at {
177            continue;
178        }
179        let before = without_article(&body[at..name.start]);
180        words |= before.chars().any(char::is_alphabetic);
181        pattern.push_str(&numbers_loosened(before));
182        if !pattern.ends_with(".+?") {
183            pattern.push_str(".+?");
184        }
185        at = name.end;
186    }
187    let rest = &body[at..];
188    words |= rest.chars().any(char::is_alphabetic);
189    pattern.push_str(&numbers_loosened(rest));
190    if !words {
191        pattern = regex::escape(body);
192    }
193    let end = if body.len() < text.len() {
194        r"\s*$"
195    } else {
196        "$"
197    };
198    format!("^{pattern}{end}")
199}
200
201/// `before`, the words before a name, without the article that ends them:
202/// `at a ` is `at `.
203fn without_article(before: &str) -> &str {
204    let Some(head) = before.strip_suffix(' ') else {
205        return before;
206    };
207    let from = head
208        .char_indices()
209        .rev()
210        .find(|(_, c)| c.is_whitespace())
211        .map_or(0, |(at, c)| at + c.len_utf8());
212    let word = &head[from..];
213    if ["a", "an", "the", "some"]
214        .iter()
215        .any(|article| word.eq_ignore_ascii_case(article))
216    {
217        &before[..from]
218    } else {
219        before
220    }
221}

223/// `words` as a regular expression matches them, each number -- `1,234`
224/// too -- any number.
225fn numbers_loosened(words: &str) -> String {
226    let bytes = words.as_bytes();
227    let digit = |at: usize| bytes.get(at).is_some_and(u8::is_ascii_digit);
228    let (mut out, mut from, mut at) = (String::new(), 0, 0);
229    while at < bytes.len() {
230        if !digit(at) {
231            at += 1;
232            continue;
233        }
234        out.push_str(&regex::escape(&words[from..at]));
235        while digit(at) || (bytes.get(at) == Some(&b',') && digit(at + 1)) {
236            at += 1;
237        }
238        out.push_str(r"[\d,]+");
239        from = at;
240    }
241    out.push_str(&regex::escape(&words[from..]));
242    out
243}

263impl Rules {
264    /// The rules `text`, a file's, holds.
265    ///
266    /// # Errors
267    ///
268    /// Why it is not a rules file, in TOML's words.
269    pub fn parse(text: &str) -> Result<Self, String> {
270        toml::from_str(text).map_err(|why| why.to_string())
271    }
272
273    /// The rules as a file holds them.
274    ///
275    /// # Errors
276    ///
277    /// A rule TOML cannot write.
278    pub fn to_toml(&self) -> Result<String, String> {
279        toml::to_string_pretty(self).map_err(|why| why.to_string())
280    }
281
282    /// The rules in `dir`, the data folder: [`Rules::default`] when it has
283    /// no file.
284    ///
285    /// # Errors
286    ///
287    /// The file there cannot be read, or is not a rules file.
288    pub fn load(dir: &Path) -> Result<Self, String> {
289        let path = dir.join(FILE);
290        match std::fs::read_to_string(&path) {
291            Ok(text) => Self::parse(&text).map_err(|why| format!("{}: {why}", path.display())),
292            Err(why) if why.kind() == std::io::ErrorKind::NotFound => Ok(Self::default()),
293            Err(why) => Err(format!("{}: {why}", path.display())),
294        }
295    }
296}

335/// One rule, ready to match.
336#[derive(Debug)]
337struct Built {
338    find: Option<regex::Regex>,
339    from: Option<String>,
340    fighting: bool,
341    marked: bool,
342    kinds: Vec<MessageKind>,
343    to: String,
344    copy: bool,
345}
346
347impl Built {
348    /// `rule`, ready; or why it cannot be used.
349    fn of(rule: &Rule) -> Result<Self, String> {
350        let to = rule.to.trim();
351        if to.is_empty() {
352            return Err("it puts lines in no stream".to_owned());
353        }
354        if to == crate::player_log::MAIN {
355            return Err("it puts lines in main, where the game's own lines are".to_owned());
356        }
357        let find = found_by(rule)?;
358        let kinds = rule
359            .creature
360            .iter()
361            .map(|kind| {
362                MessageKind::parse(kind.trim())
363                    .ok_or_else(|| format!("`{kind}` is no kind of creature message"))
364            })
365            .collect::<Result<Vec<_>, _>>()?;
366        let from = rule
367            .from
368            .as_deref()
369            .map(str::trim)
370            .filter(|from| !from.is_empty())
371            .map(str::to_owned);
372        if find.is_none() && from.is_none() && !rule.fighting && !rule.marked && kinds.is_empty() {
373            return Err("it would take every line: it gives nothing to match".to_owned());
374        }
375        Ok(Self {
376            find,
377            from,
378            fighting: rule.fighting,
379            marked: rule.marked,
380            kinds,
381            to: to.to_owned(),
382            copy: rule.copy,
383        })
384    }
385
386    /// Whether it matches `line`: the cheap conditions first, the bestiary
387    /// last.
388    fn matches(&self, line: &Line<'_>) -> bool {
389        self.from.as_deref().is_none_or(|from| from == line.stream)
390            && (!self.fighting || line.fighting)
391            && (!self.marked || line.marked)
392            && self
393                .find
394                .as_ref()
395                .is_none_or(|find| find.is_match(line.text))
396            && (self.kinds.is_empty() || self.creatures_say(line))
397    }
398
399    /// Whether `line` is one of a creature's messages of the kinds asked,
400    /// a creature it links by noun.
401    fn creatures_say(&self, line: &Line<'_>) -> bool {
402        line.links
403            .iter()
404            .filter_map(|link| match &link.does {
405                Does::Object { noun, .. } => Some(noun.as_str()),
406                _ => None,
407            })
408            .flat_map(creature::by_noun)
409            .any(|creature| {
410                hydra_model::state::creature_message::classify(creature, line.text, &self.kinds)
411                    .is_some()
412            })
413    }
414}

416/// What finds the lines `rule` takes by their words: its words, any of
417/// them, whatever their case, or its regex; none when it gives neither.
418fn found_by(rule: &Rule) -> Result<Option<regex::Regex>, String> {
419    Ok(match (rule.text.as_deref(), rule.regex.as_deref()) {
420        (Some(_), Some(_)) => return Err("it gives both words and a regex".to_owned()),
421        (Some(text), None) => {
422            let words: Vec<String> = text
423                .split('|')
424                .map(str::trim)
425                .filter(|word| !word.is_empty())
426                .map(regex::escape)
427                .collect();
428            if words.is_empty() {
429                return Err("its words are empty".to_owned());
430            }
431            Some(
432                regex::RegexBuilder::new(&words.join("|"))
433                    .case_insensitive(true)
434                    .build()
435                    .map_err(|why| why.to_string())?,
436            )
437        }
438        (None, Some(expression)) => Some(
439            regex::RegexBuilder::new(expression)
440                .size_limit(1 << 20)
441                .build()
442                .map_err(|why| format!("its regex is not one: {why}"))?,
443        ),
444        (None, None) => None,
445    })
446}

448/// The rules, ready to sort lines: those that cannot be used left out.
449#[derive(Debug)]
450pub struct Compiled {
451    rules: Vec<Built>,
452    /// The rules as written, every one: what an editor shows.
453    written: Rules,
454}
455
456impl Compiled {
457    /// `rules` ready, and why each left out cannot be used.
458    #[must_use]
459    pub fn new(rules: &Rules) -> (Self, Vec<String>) {
460        let mut problems = Vec::new();
461        let mut built = Vec::new();
462        for rule in rules.rules.iter().filter(|rule| !rule.off) {
463            match Built::of(rule) {
464                Ok(ready) => built.push(ready),
465                Err(why) => problems.push(format!("{}: {why}", rule.name)),
466            }
467        }
468        let written = rules.clone();
469        (
470            Self {
471                rules: built,
472                written,
473            },
474            problems,
475        )
476    }
477
478    /// The rules as written, those that cannot be used and those turned off
479    /// among them.
480    #[must_use]
481    pub fn rules(&self) -> &Rules {
482        &self.written
483    }
484
485    /// The streams the rules put lines in, each once, in the rules' order:
486    /// what a frontend offers a window for beside the game's.
487    #[must_use]
488    pub fn streams(&self) -> Vec<String> {
489        let mut streams: Vec<String> = Vec::new();
490        for rule in &self.rules {
491            if !streams.contains(&rule.to) {
492                streams.push(rule.to.clone());
493            }
494        }
495        streams
496    }
497
498    /// Where the first rule that can be tried on a line read back from the
499    /// log matches `line` puts it: one asking whether there was fighting or a
500    /// mark is passed over, as a past line cannot say. `None` when none does.
501    #[must_use]
502    pub fn sort_past(&self, line: &Line<'_>) -> Option<Sorted<'_>> {
503        self.rules
504            .iter()
505            .filter(|rule| !rule.fighting && !rule.marked)
506            .find(|rule| rule.matches(line))
507            .map(|rule| Sorted {
508                to: &rule.to,
509                copy: rule.copy,
510            })
511    }
512
513    /// Where the first rule that matches `line` puts it; `None` when none
514    /// does.
515    #[must_use]
516    pub fn sort(&self, line: &Line<'_>) -> Option<Sorted<'_>> {
517        self.rules
518            .iter()
519            .find(|rule| rule.matches(line))
520            .map(|rule| Sorted {
521                to: &rule.to,
522                copy: rule.copy,
523            })
524    }
525}

527/// The rules every session sorts its lines by, shared by whoever sets them:
528/// clones are the same, and each session sees a set at its next prompt.
529#[derive(Clone, Debug)]
530pub struct StreamRules(Arc<Mutex<Arc<Compiled>>>);
531
532impl Default for StreamRules {
533    /// The rules built in.
534    fn default() -> Self {
535        Self::new(&Rules::default()).0
536    }
537}
538
539impl StreamRules {
540    /// `rules`, shared; and why each left out cannot be used.
541    #[must_use]
542    pub fn new(rules: &Rules) -> (Self, Vec<String>) {
543        let (compiled, problems) = Compiled::new(rules);
544        (Self(Arc::new(Mutex::new(Arc::new(compiled)))), problems)
545    }
546
547    /// Sort by `rules` from now on: why each left out cannot be used.
548    pub fn set(&self, rules: &Rules) -> Vec<String> {
549        let (compiled, problems) = Compiled::new(rules);
550        *self.0.lock().unwrap_or_else(PoisonError::into_inner) = Arc::new(compiled);
551        problems
552    }
553
554    /// The rules as they are now.
555    #[must_use]
556    pub fn current(&self) -> Arc<Compiled> {
557        Arc::clone(&self.0.lock().unwrap_or_else(PoisonError::into_inner))
558    }
559}

561/// Whether a prompt's chunk had fighting in it: an attack, or a fact of
562/// one -- a status, a death, a crit's stun. A spell wearing off is no
563/// fighting, nor is a change in what is worn or up.
564#[must_use]
565pub fn fought(facts: &[ChunkFacts]) -> bool {
566    facts.iter().any(|chunk| {
567        !chunk.events.is_empty()
568            || chunk
569                .facts
570                .iter()
571                .any(|fact| !matches!(fact, Fact::SpellLoss { .. }))
572    })
573}

821    fn a_rule_made_from_a_line_takes_the_lines_like_it() {
822        let text = "A grey warg howls at the 3 moons.";
823        let rule = Rule::like(Some("main"), text, &[named(2, 11)], "ambients");
824        assert_eq!(
825            rule.regex.as_deref(),
826            Some(r"^.+? howls at the [\d,]+ moons\.$")
827        );
828        assert_eq!(rule.name, "Like \"A grey warg howls at the 3 moons.\"");
829        assert_eq!(
830            (rule.from.as_deref(), rule.to.as_str(), rule.copy),
831            (Some("main"), "ambients", false)
832        );
833        let (compiled, problems) = Compiled::new(&Rules {
834            rules: vec![rule.clone()],
835        });
836        assert!(problems.is_empty(), "{problems:?}");
837        let sorted = |stream, text| compiled.sort(&line(stream, text)).map(|sorted| sorted.to);
838        assert_eq!(sorted("main", text), Some("ambients"));
839        assert_eq!(
840            sorted("main", "An orc howls at the 1,024 moons."),
841            Some("ambients")
842        );
843        assert_eq!(sorted("main", "A warg howls at the moon."), None);
844        assert_eq!(sorted("thoughts", text), None, "the line's stream alone");
845        assert_eq!(rule.finds("Ashryn howls at the 2 moons."), Ok(true));
846        let unplaced = Rule {
847            to: String::new(),
848            ..rule.clone()
849        };
850        assert_eq!(
851            unplaced.finds(text),
852            Ok(true),
853            "its words tried before it has a stream"
854        );
855        assert_eq!(
856            like("You swing at a kobold!", &[named(15, 21)]),
857            r"^You swing at .+?!$"
858        );
859        assert_eq!(
860            like("  Two 1,000 gems.  ", &[]),
861            r"^  Two [\d,]+ gems\.\s*$"
862        );
863        assert_eq!(
864            like("Nisugi", &[named(0, 6)]),
865            "^Nisugi$",
866            "nothing but a name"
867        );
868        assert_eq!(
869            like("x", &[named(3, 9)]),
870            "^x$",
871            "a name past the end left out"
872        );
873        let long = Rule::like(None, &"word ".repeat(10), &[], "x");
874        assert_eq!(long.from, None, "from any stream");
875        assert_eq!(
876            long.name,
877            "Like \"word word word word word word word word...\""
878        );
879    }

1//! Stream rules: which stream a line of the game's belongs to, as the player
2//! sorts them, beside the one the game sent it to. One list for every
3//! character, [`FILE`] in the data folder; the first rule that matches a
4//! line sorts it.
5//!
6//! A rule matches a line by its words or a regular expression, and by what
7//! Hydra knows of it: that it came in a chunk the combat reader found
8//! fighting in, that the reader marked it (a flare, a status landing on the
9//! character, a killing blow), or that the bestiary knows it as one of a
10//! creature's messages of the kinds named (its idle flavour, its arrival,
11//! its death...). A rule that names a stream looks only at the lines the
12//! game sent there. Every condition a rule gives must hold.
13//!
14//! A line a rule matches is logged under its stream after the one the game
15//! sent it to, `main/ambients`: where it came from, and where the player
16//! put it. The windows are told it too ([`crate::Event::Sorted`]): it goes
17//! to its stream's windows, and leaves the stream the game sent it to unless
18//! the rule copies it.
19//!
20//! With no file, the rules are [`Rules::default`]: the two built in, every
21//! main line of a chunk with fighting in it to `combat`, and every main line
22//! of a chunk with a search or a skin in it to `loot`, each kept in main
23//! too. A file is the whole list: the built-in rules are in it to edit, or
24//! not. A file read before the loot rule was built in is given it once
25//! ([`Rules::parse`]).
26//!
27//! A rule can be made from a line the player points at ([`Rule::like`]):
28//! its words as they are, with the names in it and its numbers loosened, so
29//! it takes the lines like it.
30//!
31//! A command goes with its chunk to the stream of the rule that takes
32//! fights ([`Rules::command`]): a script's attack, by its first word
33//! ([`FIGHT_COMMANDS`]), when the chunk it went out in had fighting in it;
34//! a stance or a target, whoever sent it ([`SETUP_COMMANDS`]), while a
35//! hostile creature is in the room, and the chunk it went out in with it.
36//! The player's own attacks stay where they were typed. A search or a skin,
37//! whoever sent it ([`LOOT_COMMANDS`]), goes with its chunk to the rule
38//! that takes loot. A command is logged as it was sent, whatever its
39//! stream.
40//!
41//! The log's past lines are sorted again as they are read back
42//! ([`Compiled::sort_past`]): a line written before a rule that takes it was
43//! made is shown in its stream, though its file says otherwise.
44

164/// Whether `text`, a command, begins with a word of `list`, or of
165/// `built_in` when the rule names none; a spell number alone is
166/// `incant`'s.
167fn begins_with(text: &str, list: Option<&Vec<String>>, built_in: &[&str]) -> bool {
168    let Some(word) = text.split_whitespace().next() else {
169        return false;
170    };
171    let word = if word.bytes().all(|byte| byte.is_ascii_digit()) {
172        "incant".to_owned()
173    } else {
174        word.to_ascii_lowercase()
175    };
176    match list {
177        Some(list) => list
178            .iter()
179            .any(|listed| listed.trim().eq_ignore_ascii_case(&word)),
180        None => built_in.contains(&word.as_str()),
181    }
182}

305/// A regular expression for the lines like `text`: its words as they are,
306/// each of `names` -- where it names a creature, a player or a thing, by
307/// byte -- any name, the article before one with it, and each number any
308/// number. Anchored at both ends, so it takes whole lines. A line with no
309/// words left but names is matched as it is.
310#[must_use]
311pub fn like(text: &str, names: &[Range<usize>]) -> String {
312    let body = text.trim_end();
313    let mut names: Vec<&Range<usize>> = names
314        .iter()
315        .filter(|name| {
316            name.start < name.end
317                && body.is_char_boundary(name.start)
318                && body.get(name.start..name.end).is_some()
319        })
320        .collect();
321    names.sort_by_key(|name| name.start);
322    let mut pattern = String::new();
323    let mut words = false;
324    let mut at = 0;
325    for name in names {
326        if name.start < at {
327            continue;
328        }
329        let before = without_article(&body[at..name.start]);
330        words |= before.chars().any(char::is_alphabetic);
331        pattern.push_str(&numbers_loosened(before));
332        if !pattern.ends_with(".+?") {
333            pattern.push_str(".+?");
334        }
335        at = name.end;
336    }
337    let rest = &body[at..];
338    words |= rest.chars().any(char::is_alphabetic);
339    pattern.push_str(&numbers_loosened(rest));
340    if !words {
341        pattern = regex::escape(body);
342    }
343    let end = if body.len() < text.len() {
344        r"\s*$"
345    } else {
346        "$"
347    };
348    format!("^{pattern}{end}")
349}

351/// `before`, the words before a name, without the article that ends them:
352/// `at a ` is `at `.
353fn without_article(before: &str) -> &str {
354    let Some(head) = before.strip_suffix(' ') else {
355        return before;
356    };
357    let from = head
358        .char_indices()
359        .rev()
360        .find(|(_, c)| c.is_whitespace())
361        .map_or(0, |(at, c)| at + c.len_utf8());
362    let word = &head[from..];
363    if ["a", "an", "the", "some"]
364        .iter()
365        .any(|article| word.eq_ignore_ascii_case(article))
366    {
367        &before[..from]
368    } else {
369        before
370    }
371}

373/// `words` as a regular expression matches them, each number -- `1,234`
374/// too -- any number.
375fn numbers_loosened(words: &str) -> String {
376    let bytes = words.as_bytes();
377    let digit = |at: usize| bytes.get(at).is_some_and(u8::is_ascii_digit);
378    let (mut out, mut from, mut at) = (String::new(), 0, 0);
379    while at < bytes.len() {
380        if !digit(at) {
381            at += 1;
382            continue;
383        }
384        out.push_str(&regex::escape(&words[from..at]));
385        while digit(at) || (bytes.get(at) == Some(&b',') && digit(at + 1)) {
386            at += 1;
387        }
388        out.push_str(r"[\d,]+");
389        from = at;
390    }
391    out.push_str(&regex::escape(&words[from..]));
392    out
393}

395/// The rules, in the order they are tried.
396#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]

398pub struct Rules {

408impl Default for Rules {

412            loot_given: true,
413            rules: vec![Rule::fighting(), Rule::looting()],

418impl Rules {
419    /// The rules `text`, a file's, holds: with the loot rule built in after
420    /// the rules there when the file was written before it was
421    /// ([`Self::loot_given`]).
422    ///
423    /// # Errors
424    ///
425    /// Why it is not a rules file, in TOML's words.
426    pub fn parse(text: &str) -> Result<Self, String> {
427        let mut rules: Self = toml::from_str(text).map_err(|why| why.to_string())?;
428        if !rules.loot_given {
429            rules.loot_given = true;
430            if !rules.rules.iter().any(|rule| rule.looting) {
431                rules.rules.push(Rule::looting());
432            }
433        }
434        Ok(rules)
435    }
436
437    /// The rules as a file holds them.
438    ///
439    /// # Errors
440    ///
441    /// A rule TOML cannot write.
442    pub fn to_toml(&self) -> Result<String, String> {
443        toml::to_string_pretty(self).map_err(|why| why.to_string())
444    }
445
446    /// The rules in `dir`, the data folder: [`Rules::default`] when it has
447    /// no file.
448    ///
449    /// # Errors
450    ///
451    /// The file there cannot be read, or is not a rules file.
452    pub fn load(dir: &Path) -> Result<Self, String> {
453        let path = dir.join(FILE);
454        match std::fs::read_to_string(&path) {
455            Ok(text) => Self::parse(&text).map_err(|why| format!("{}: {why}", path.display())),
456            Err(why) if why.kind() == std::io::ErrorKind::NotFound => Ok(Self::default()),
457            Err(why) => Err(format!("{}: {why}", path.display())),
458        }
459    }

461    /// Whether the rules that put a fight's chunk in [`COMBAT`] keep it
462    /// in main too: `None` while none in use does. Those rules ask for
463    /// fighting and nothing of the line.
464    #[must_use]
465    pub fn combat_kept(&self) -> Option<bool> {
466        self.rules
467            .iter()
468            .find(|rule| rule.takes_fights())
469            .map(|rule| rule.copy)
470    }

472    /// Whether the rules that put a search's or a skin's chunk in their
473    /// stream keep it in main too: `None` while none in use does. Those
474    /// rules ask for looting and nothing of the line.
475    #[must_use]
476    pub fn loot_kept(&self) -> Option<bool> {
477        self.rules
478            .iter()
479            .find(|rule| rule.takes_loot())
480            .map(|rule| rule.copy)
481    }

483    /// The rules that put a search's or a skin's chunk in their stream made
484    /// to keep it in main too (`keep`), or to move it: what was done.
485    ///
486    /// # Errors
487    ///
488    /// No rule in use puts searches and skins in a stream.
489    pub fn keep_loot(&mut self, keep: bool) -> Result<String, String> {
490        let mut found = false;
491        for rule in self.rules.iter_mut().filter(|rule| rule.takes_loot()) {
492            rule.copy = keep;
493            found = true;
494        }
495        if !found {
496            return Err("No stream rule puts searches and skins in a stream.".to_owned());
497        }
498        Ok(if keep {
499            "Searches and skins are kept in the story as well as loot.".to_owned()
500        } else {
501            "Searches and skins are moved to loot, out of the story while a loot window is open."
502                .to_owned()
503        })
504    }
505
506    /// Whether `text`, a command sent in a chunk while a hostile creature
507    /// is in the room, takes that chunk to the rule that takes fights: a
508    /// stance, a target ([`SETUP_COMMANDS`]).
509    #[must_use]
510    pub fn sets_up(&self, text: &str) -> bool {
511        self.rules
512            .iter()
513            .filter(|rule| rule.takes_fights())
514            .any(|rule| begins_with(text, rule.setup.as_ref(), SETUP_COMMANDS))
515    }
516
517    /// Whether `text`, a command, is an attack a rule that takes fights
518    /// lists ([`FIGHT_COMMANDS`]): its answer is a fight's, or a word of
519    /// why there was none.
520    #[must_use]
521    pub fn attacks(&self, text: &str) -> bool {
522        self.rules
523            .iter()
524            .filter(|rule| rule.takes_fights())
525            .any(|rule| begins_with(text, rule.commands.as_ref(), FIGHT_COMMANDS))
526    }
527
528    /// Whether `text`, a command, is one a rule that takes fights or loot
529    /// lists: its answer is a fight's, a search's or a skin's, or one the
530    /// readers may not know as one.
531    #[must_use]
532    pub fn goes_with_chunks(&self, text: &str) -> bool {
533        self.rules.iter().any(|rule| {
534            if rule.takes_fights() {
535                begins_with(text, rule.commands.as_ref(), FIGHT_COMMANDS)
536                    || begins_with(text, rule.setup.as_ref(), SETUP_COMMANDS)
537            } else {
538                rule.takes_loot() && begins_with(text, rule.commands.as_ref(), LOOT_COMMANDS)
539            }
540        })
541    }
542
543    /// Where `command` goes with its chunk: to the stream of the first rule
544    /// that takes fights and lists it. A script's attack goes with a chunk
545    /// that had fighting in it ([`FIGHT_COMMANDS`]); a command that sets
546    /// up ([`Self::sets_up`]) with a chunk it took.
547    #[must_use]
548    pub fn command(&self, command: &Command<'_>) -> Option<Sorted<'_>> {
549        self.rules
550            .iter()
551            .find(|rule| {
552                if rule.takes_fights() {
553                    (command.by_script
554                        && command.fighting
555                        && begins_with(command.text, rule.commands.as_ref(), FIGHT_COMMANDS))
556                        || (command.set_up
557                            && begins_with(command.text, rule.setup.as_ref(), SETUP_COMMANDS))
558                } else {
559                    rule.takes_loot()
560                        && command.looted
561                        && begins_with(command.text, rule.commands.as_ref(), LOOT_COMMANDS)
562                }
563            })
564            .map(|rule| Sorted {
565                to: &rule.to,
566                copy: rule.copy,
567            })
568    }
569
570    /// The rules that put a fight's chunk in [`COMBAT`] made to keep it in
571    /// main too (`keep`), or to move it: what was done.
572    ///
573    /// # Errors
574    ///
575    /// No rule in use puts fights in combat.
576    pub fn keep_combat(&mut self, keep: bool) -> Result<String, String> {
577        let mut found = false;
578        for rule in self.rules.iter_mut().filter(|rule| rule.takes_fights()) {
579            rule.copy = keep;
580            found = true;
581        }
582        if !found {
583            return Err("No stream rule puts fights in combat.".to_owned());
584        }
585        Ok(if keep {
586            "Fights are kept in the story as well as combat.".to_owned()
587        } else {
588            "Fights are moved to combat, out of the story while a combat window is open.".to_owned()
589        })
590    }
591}

656/// One rule, ready to match.
657#[derive(Debug)]
658#[expect(
659    clippy::struct_excessive_bools,
660    reason = "each a condition or a choice of the rule's own"
661)]
662struct Built {
663    find: Option<regex::Regex>,
664    from: Option<String>,
665    fighting: bool,
666    looting: bool,
667    marked: bool,
668    kinds: Vec<MessageKind>,
669    to: String,
670    copy: bool,
671}
672
673impl Built {
674    /// `rule`, ready; or why it cannot be used.
675    fn of(rule: &Rule) -> Result<Self, String> {
676        let to = rule.to.trim();
677        if to.is_empty() {
678            return Err("it puts lines in no stream".to_owned());
679        }
680        if to == crate::player_log::MAIN {
681            return Err("it puts lines in main, where the game's own lines are".to_owned());
682        }
683        let find = found_by(rule)?;
684        let kinds = rule
685            .creature
686            .iter()
687            .map(|kind| {
688                MessageKind::parse(kind.trim())
689                    .ok_or_else(|| format!("`{kind}` is no kind of creature message"))
690            })
691            .collect::<Result<Vec<_>, _>>()?;
692        let from = rule
693            .from
694            .as_deref()
695            .map(str::trim)
696            .filter(|from| !from.is_empty())
697            .map(str::to_owned);
698        if find.is_none()
699            && from.is_none()
700            && !rule.fighting
701            && !rule.looting
702            && !rule.marked
703            && kinds.is_empty()
704        {
705            return Err("it would take every line: it gives nothing to match".to_owned());
706        }
707        Ok(Self {
708            find,
709            from,
710            fighting: rule.fighting,
711            looting: rule.looting,
712            marked: rule.marked,
713            kinds,
714            to: to.to_owned(),
715            copy: rule.copy,
716        })
717    }
718
719    /// Whether it matches `line`: the cheap conditions first, the bestiary
720    /// last.
721    fn matches(&self, line: &Line<'_>) -> bool {
722        self.from.as_deref().is_none_or(|from| from == line.stream)
723            && (!self.fighting || line.fighting)
724            && (!self.looting || line.looted)
725            && (!self.marked || line.marked)
726            && self
727                .find
728                .as_ref()
729                .is_none_or(|find| find.is_match(line.text))
730            && (self.kinds.is_empty() || self.creatures_say(line))
731    }
732
733    /// Whether `line` is one of a creature's messages of the kinds asked,
734    /// a creature it links by noun.
735    fn creatures_say(&self, line: &Line<'_>) -> bool {
736        line.links
737            .iter()
738            .filter_map(|link| match &link.does {
739                Does::Object { noun, .. } => Some(noun.as_str()),
740                _ => None,
741            })
742            .flat_map(creature::by_noun)
743            .any(|creature| {
744                hydra_model::state::creature_message::classify(creature, line.text, &self.kinds)
745                    .is_some()
746            })
747    }
748}
749
750/// What finds the lines `rule` takes by their words: its words, any of
751/// them, whatever their case, or its regex; none when it gives neither.
752fn found_by(rule: &Rule) -> Result<Option<regex::Regex>, String> {
753    Ok(match (rule.text.as_deref(), rule.regex.as_deref()) {
754        (Some(_), Some(_)) => return Err("it gives both words and a regex".to_owned()),
755        (Some(text), None) => {
756            let words: Vec<String> = text
757                .split('|')
758                .map(str::trim)
759                .filter(|word| !word.is_empty())
760                .map(regex::escape)
761                .collect();
762            if words.is_empty() {
763                return Err("its words are empty".to_owned());
764            }
765            Some(
766                regex::RegexBuilder::new(&words.join("|"))
767                    .case_insensitive(true)
768                    .build()
769                    .map_err(|why| why.to_string())?,
770            )
771        }
772        (None, Some(expression)) => Some(
773            regex::RegexBuilder::new(expression)
774                .size_limit(1 << 20)
775                .build()
776                .map_err(|why| format!("its regex is not one: {why}"))?,
777        ),
778        (None, None) => None,
779    })
780}

782/// The rules, ready to sort lines: those that cannot be used left out.
783#[derive(Debug)]
784pub struct Compiled {
785    rules: Vec<Built>,
786    /// The rules as written, every one: what an editor shows.
787    written: Rules,
788}
789
790impl Compiled {
791    /// `rules` ready, and why each left out cannot be used.
792    #[must_use]
793    pub fn new(rules: &Rules) -> (Self, Vec<String>) {
794        let mut problems = Vec::new();
795        let mut built = Vec::new();
796        for rule in rules.rules.iter().filter(|rule| !rule.off) {
797            match Built::of(rule) {
798                Ok(ready) => built.push(ready),
799                Err(why) => problems.push(format!("{}: {why}", rule.name)),
800            }
801        }
802        let written = rules.clone();
803        (
804            Self {
805                rules: built,
806                written,
807            },
808            problems,
809        )
810    }
811
812    /// The rules as written, those that cannot be used and those turned off
813    /// among them.
814    #[must_use]
815    pub fn rules(&self) -> &Rules {
816        &self.written
817    }
818
819    /// The streams the rules put lines in, each once, in the rules' order:
820    /// what a frontend offers a window for beside the game's.
821    #[must_use]
822    pub fn streams(&self) -> Vec<String> {
823        let mut streams: Vec<String> = Vec::new();
824        for rule in &self.rules {
825            if !streams.contains(&rule.to) {
826                streams.push(rule.to.clone());
827            }
828        }
829        streams
830    }
831
832    /// Where the first rule that can be tried on a line read back from the
833    /// log matches `line` puts it: one asking whether there was fighting,
834    /// looting or a mark is passed over, as a past line cannot say. `None`
835    /// when none does.
836    #[must_use]
837    pub fn sort_past(&self, line: &Line<'_>) -> Option<Sorted<'_>> {
838        self.rules
839            .iter()
840            .filter(|rule| !rule.fighting && !rule.looting && !rule.marked)
841            .find(|rule| rule.matches(line))
842            .map(|rule| Sorted {
843                to: &rule.to,
844                copy: rule.copy,
845            })
846    }
847
848    /// Where the first rule that matches `line` puts it; `None` when none
849    /// does.
850    #[must_use]
851    pub fn sort(&self, line: &Line<'_>) -> Option<Sorted<'_>> {
852        self.rules
853            .iter()
854            .find(|rule| rule.matches(line))
855            .map(|rule| Sorted {
856                to: &rule.to,
857                copy: rule.copy,
858            })
859    }
860}

862/// The rules every session sorts its lines by, shared by whoever sets them:
863/// clones are the same, and each session sees a set at its next prompt.
864#[derive(Clone, Debug)]
865pub struct StreamRules(Arc<Mutex<Arc<Compiled>>>);
866
867impl Default for StreamRules {
868    /// The rules built in.
869    fn default() -> Self {
870        Self::new(&Rules::default()).0
871    }
872}
873
874impl StreamRules {
875    /// `rules`, shared; and why each left out cannot be used.
876    #[must_use]
877    pub fn new(rules: &Rules) -> (Self, Vec<String>) {
878        let (compiled, problems) = Compiled::new(rules);
879        (Self(Arc::new(Mutex::new(Arc::new(compiled)))), problems)
880    }
881
882    /// Sort by `rules` from now on: why each left out cannot be used.
883    pub fn set(&self, rules: &Rules) -> Vec<String> {
884        let (compiled, problems) = Compiled::new(rules);
885        *self.0.lock().unwrap_or_else(PoisonError::into_inner) = Arc::new(compiled);
886        problems
887    }
888
889    /// The rules as they are now.
890    #[must_use]
891    pub fn current(&self) -> Arc<Compiled> {
892        Arc::clone(&self.0.lock().unwrap_or_else(PoisonError::into_inner))
893    }
894}

896/// Whether a prompt's chunk had fighting in it: an attack, or a fact of
897/// one -- a status, a death, a crit's stun. A spell wearing off is no
898/// fighting, nor is a change in what is worn or up.
899#[must_use]
900pub fn fought(facts: &[ChunkFacts]) -> bool {
901    facts.iter().any(|chunk| {
902        !chunk.events.is_empty()
903            || chunk
904                .facts
905                .iter()
906                .any(|fact| !matches!(fact, Fact::SpellLoss { .. }))
907    })
908}
909
910/// Whose attacks a prompt's chunk had in it: `Some(true)`, one of the
911/// character's own; `Some(false)`, only others' -- a creature's, another
912/// player's, a lasting effect's tick no cast owns; `None`, no attack, or
913/// none the reader could say whose.
914#[must_use]
915pub fn ours(facts: &[ChunkFacts]) -> Option<bool> {
916    let mut others = false;
917    for event in facts.iter().flat_map(|chunk| chunk.events.iter()) {
918        if event.is_ours() {
919            return Some(true);
920        }
921        others |=
922            event.inbound || event.foreign_caster || event.unowned || event.attacker.is_some();
923    }
924    others.then_some(false)
925}

1296    #[test]
1297    fn a_rule_made_from_a_line_takes_the_lines_like_it() {
1298        let text = "A grey warg howls at the 3 moons.";
1299        let rule = Rule::like(Some("main"), text, &[named(2, 11)], "ambients");
1300        assert_eq!(
1301            rule.regex.as_deref(),
1302            Some(r"^.+? howls at the [\d,]+ moons\.$")
1303        );
1304        assert_eq!(rule.name, "Like \"A grey warg howls at the 3 moons.\"");
1305        assert_eq!(
1306            (rule.from.as_deref(), rule.to.as_str(), rule.copy),
1307            (Some("main"), "ambients", false)
1308        );
1309        let (compiled, problems) = Compiled::new(&Rules {
1310            loot_given: true,
1311            rules: vec![rule.clone()],
1312        });
1313        assert!(problems.is_empty(), "{problems:?}");
1314        let sorted = |stream, text| compiled.sort(&line(stream, text)).map(|sorted| sorted.to);
1315        assert_eq!(sorted("main", text), Some("ambients"));
1316        assert_eq!(
1317            sorted("main", "An orc howls at the 1,024 moons."),
1318            Some("ambients")
1319        );
1320        assert_eq!(sorted("main", "A warg howls at the moon."), None);
1321        assert_eq!(sorted("thoughts", text), None, "the line's stream alone");
1322        assert_eq!(rule.finds("Ashryn howls at the 2 moons."), Ok(true));
1323        let unplaced = Rule {
1324            to: String::new(),
1325            ..rule.clone()
1326        };
1327        assert_eq!(
1328            unplaced.finds(text),
1329            Ok(true),
1330            "its words tried before it has a stream"
1331        );
1332        assert_eq!(
1333            like("You swing at a kobold!", &[named(15, 21)]),
1334            r"^You swing at .+?!$"
1335        );
1336        assert_eq!(
1337            like("  Two 1,000 gems.  ", &[]),
1338            r"^  Two [\d,]+ gems\.\s*$"
1339        );
1340        assert_eq!(
1341            like("Nisugi", &[named(0, 6)]),
1342            "^Nisugi$",
1343            "nothing but a name"
1344        );
1345        assert_eq!(
1346            like("x", &[named(3, 9)]),
1347            "^x$",
1348            "a name past the end left out"
1349        );
1350        let long = Rule::like(None, &"word ".repeat(10), &[], "x");
1351        assert_eq!(long.from, None, "from any stream");
1352        assert_eq!(
1353            long.name,
1354            "Like \"word word word word word word word word...\""
1355        );

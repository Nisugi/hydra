10/// One chunk's combat facts, shared: the same chunk is the same facts, so
11/// two of these are equal when they are the one chunk.
12#[derive(Clone, Debug)]
13pub struct Facts(pub std::sync::Arc<hydra_model::state::combat::ChunkFacts>);
14
15impl PartialEq for Facts {
16    fn eq(&self, other: &Self) -> bool {
17        std::sync::Arc::ptr_eq(&self.0, &other.0)
18    }
19}
20
21impl Eq for Facts {}
22
23impl std::ops::Deref for Facts {
24    type Target = hydra_model::state::combat::ChunkFacts;
25
26    fn deref(&self) -> &Self::Target {
27        &self.0
28    }
29}

34/// One thing that happened, in the order it happened.
35#[derive(Clone, Debug, PartialEq, Eq)]
36pub enum Event {
37    /// When what follows happened: before each chunk of the game's bytes,
38    /// the moment it arrived, and before each line written, the moment it
39    /// was. For a window to time itself by (`hydra_ui::perf`).
40    Clock(std::time::Instant),
41    /// A frame the game sent, after the game state has taken it.
42    Frame(Box<Frame>),
43    /// A prompt: its text. Also sent as a frame; this is the one a frontend
44    /// draws an input line by.
45    Prompt(String),
46    /// The game's lines of the chunk the last prompt closed that the combat
47    /// reader found to be a flare, a status landing on the character or a
48    /// killing blow, each with its text, for a frontend to colour.
49    Marked(Vec<hydra_model::state::combat::LineMark>),
50    /// The lines of the chunk the last prompt closed that the stream rules
51    /// sorted, each with where it goes (`crate::stream_rules`).
52    Sorted(Vec<crate::stream_rules::SortedLine>),
53    /// The lines of the chunk the last prompt closed that the lookout found
54    /// new (`hydra_lookout`): for the lookout's stream.
55    Lookout {
56        /// The lines, in order.
57        lines: Vec<crate::player_log::feed::Noticed>,
58        /// The story marks them where they came, as well.
59        mark: bool,
60    },
61    /// What the combat reader made of the chunk the last prompt closed:
62    /// its attacks and the facts beside them, for a recorder.
63    Combat(Facts),
64    /// From here (`true`) until the prompt that ends its answer
65    /// (`false`), the story is answering a question Hydra asked quietly
66    /// (`SessionHandle::write_quietly`): the main window's lines, the line
67    /// written and the prompt are kept out of the story.
68    Quiet(bool),
69    /// A line was written to the game.
70    Sent {
71        /// The line, without its newline.
72        line: String,
73        /// Who sent it.
74        origin: Origin,
75    },
76    /// The session moved to a new state.
77    StateChanged(State),
78    /// A line typed for Hydra's own command line (`claimant`): echoed where
79    /// it was typed, and never sent to the game.
80    Claimed(String),
81    /// Hydra says something to the player (`notice`).
82    Notice(crate::notice::Notice),
83    /// A frame of what the player's Lich shows, and whose line it came from
84    /// (`crate::lich`).
85    LichShown {
86        /// The frame.
87        frame: Box<Frame>,
88        /// Whose line it was.
89        mark: crate::lich::Mark,
90    },
91    /// The player's Lich is handed the game from here on: after a `login`
92    /// built from the model when it started late, which it shows again up to
93    /// that login's prompt.
94    LichBegun {
95        /// Whether it was handed a built login.
96        login: bool,
97    },
98    /// The player's Lich has gone.
99    LichEnded,
100    /// Which text a window shows while the player's Lich runs.
101    LichView(crate::lich::View),
102    /// A connection failed or dropped; the next try is after `delay`.
103    ConnectFailed {
104        /// How many tries in a row.
105        attempt: u32,
106        /// How long until the next.
107        delay: Duration,
108        /// What went wrong.
109        detail: String,
110    },
111}

1//! Behaviour words: a chain's piece Hydra carries out with intelligence,
2//! reading the model before it sends and the game after
3//! (`crate::steps::BEHAVIOURS`).
4//!
5//! # `stance <x>`
6//!
7//! 1. Already in the stance asked for: nothing sent.
8//! 2. In roundtime: it is waited for, since the game takes no command
9//!    then.
10//! 3. Otherwise it is sent at once, a cast roundtime or not: asked for
11//!    defensive in a cast roundtime, the game sets guarded and says why.
12//! 4. Sent, the bar (`pbarStance`) is the truth: done when it shows the
13//!    stance's band, or guarded for defensive asked in a cast roundtime;
14//!    said to have failed, and the chain stopped, when it does not within
15//!    a few seconds. A `...wait` is waited out and the stance sent again,
16//!    at most [`heard::SENT_AGAIN`] times.
17//!
18//! # `<assault>`
19//!
20//! In [`assault`]: `barrage #123` sent as `weapon barrage #123`. Written
21//! with `weapon`, an assault is the command as written.
22

35/// How long the bar is given to show a stance asked for.
36const TAKEN_WITHIN: Duration = Duration::from_secs(4);
37/// How often the model is looked at while waiting.
38const LOOK: Duration = Duration::from_millis(200);
39
40/// What a stance step does next, from what the model says.
41#[derive(Clone, Copy, Debug, PartialEq, Eq)]
42pub enum Move {
43    /// Already there: nothing to send.
44    Done,
45    /// A roundtime runs: wait for it.
46    WaitRoundtime,
47    /// Ask for it.
48    Send,
49}
50
51/// The stance the model shows the character in.
52#[must_use]
53pub fn current(state: &GameState) -> Option<Stance> {
54    state
55        .character
56        .stance_percent
57        .and_then(Stance::from_percent)
58        .or_else(|| {
59            state
60                .character
61                .stance
62                .as_deref()
63                .and_then(Stance::parse_bar_text)
64                .map(|(stance, _)| stance)
65        })
66}

68/// What to do next for `wanted`, from `current` and the clock: `roundtime`
69/// as the model answers whether one runs, `None` being never stated, which
70/// is not running.
71#[must_use]
72pub fn next_move(current: Option<Stance>, wanted: Stance, roundtime: Option<bool>) -> Move {
73    if current == Some(wanted) {
74        return Move::Done;
75    }
76    if roundtime == Some(true) {
77        return Move::WaitRoundtime;
78    }
79    Move::Send
80}

82/// Whether the bar showing `shown` is the game taking `wanted`: the stance
83/// itself, or guarded for defensive asked in a cast roundtime (`casttime`),
84/// which the game sets in its place.
85#[must_use]
86pub fn taken_as(shown: Option<Stance>, wanted: Stance, casttime: bool) -> bool {
87    shown == Some(wanted)
88        || (casttime && wanted == Stance::Defensive && shown == Some(Stance::Guarded))
89}

91/// `stance <rest>` carried out on `handle`; stopped with the chain by
92/// `stopped`.
93///
94/// # Errors
95///
96/// Why it could not: no such stance, the game not showing the stance
97/// asked for, or the character gone.
98pub async fn stance(
99    handle: &SessionHandle,
100    rest: &str,
101    stopped: &CancellationToken,
102) -> Result<(), String> {
103    let wanted =
104        Stance::parse(rest).ok_or_else(|| format!("stance {rest}: not one of the six stances."))?;
105    // A roundtime may begin again while one is waited out.
106    for _ in 0..6 {
107        if stopped.is_cancelled() {
108            return Ok(());
109        }
110        let state = snapshot(handle).await?;
111        let now = current(&state);
112        match next_move(now, wanted, state.in_roundtime()) {
113            Move::Done => return Ok(()),
114            Move::WaitRoundtime => handle.clock_clear(stopped, false).await,
115            Move::Send => {
116                let casttime = state.in_casttime() == Some(true);
117                return taken(handle, wanted, casttime, stopped).await;
118            }
119        }
120    }
121    Err(format!(
122        "stance {}: the game kept me from it.",
123        wanted.as_str()
124    ))
125}

127/// `stance <wanted>` sent, and the bar watched for it ([`taken_as`]), a
128/// cast roundtime running when `casttime`; a `...wait` waited out and the
129/// stance sent again.
130async fn taken(
131    handle: &SessionHandle,
132    wanted: Stance,
133    casttime: bool,
134    stopped: &CancellationToken,
135) -> Result<(), String> {
136    let line = format!("stance {}", wanted.as_str());
137    let gone = || "the character is gone.".to_owned();
138    for _ in 0..=SENT_AGAIN {
139        let mut heard = Heard::from_now(handle).await.ok_or_else(gone)?;
140        let _ = handle.send_as_typed(&line).await;
141        let by = tokio::time::Instant::now() + TAKEN_WITHIN;
142        let mut wait = None;
143        while wait.is_none() && tokio::time::Instant::now() < by {
144            if stopped.is_cancelled() {
145                return Ok(());
146            }
147            if taken_as(current(&snapshot(handle).await?), wanted, casttime) {
148                return Ok(());
149            }
150            let look = (tokio::time::Instant::now() + LOOK).min(by);
151            loop {
152                let said = match heard.next(look, stopped).await {
153                    Ok(Some(said)) => said,
154                    Ok(None) => break,
155                    Err(heard::Halt::Stopped) => return Ok(()),
156                    Err(heard::Halt::Gone) => return Err(gone()),
157                };
158                if let Some(seconds) = wait_seconds(&said) {
159                    wait = Some(seconds);
160                    break;
161                }
162            }
163        }
164        let Some(seconds) = wait else {
165            return Err(format!(
166                "stance {}: the game did not show it taken.",
167                wanted.as_str()
168            ));
169        };
170        wait_out(handle, seconds, stopped).await;
171    }
172    Err(format!(
173        "stance {}: the game kept saying wait.",
174        wanted.as_str()
175    ))
176}

178/// The model now.
179async fn snapshot(handle: &SessionHandle) -> Result<GameState, String> {
180    handle
181        .snapshot()
182        .await
183        .map(|snapshot| snapshot.state)
184        .ok_or_else(|| "the character is gone.".to_owned())
185}

187#[cfg(test)]
188mod tests {
189    use super::*;
190
191    /// Each rule, from what the model says: there already, roundtime
192    /// waited out, and the send, a cast roundtime or not.
193    #[test]
194    fn the_stance_rules_from_the_clock_and_the_bar() {
195        use Stance::{Defensive, Guarded, Offensive};
196        assert_eq!(next_move(Some(Offensive), Offensive, None), Move::Done);
197        assert_eq!(
198            next_move(Some(Offensive), Defensive, Some(true)),
199            Move::WaitRoundtime
200        );
201        assert_eq!(next_move(None, Guarded, None), Move::Send);
202        assert_eq!(
203            next_move(Some(Offensive), Defensive, Some(false)),
204            Move::Send
205        );
206        assert!(taken_as(Some(Defensive), Defensive, false));
207        assert!(
208            taken_as(Some(Guarded), Defensive, true),
209            "guarded, set by the game in a cast roundtime"
210        );
211        assert!(!taken_as(Some(Guarded), Defensive, false));
212        assert!(!taken_as(Some(Guarded), Offensive, true));
213        let mut state = GameState::default();
214        state.character.stance_percent = Some(99);
215        assert_eq!(current(&state), Some(Defensive));
216        state.character.stance_percent = None;
217        state.character.stance = Some("guarded (80%)".to_owned());
218        assert_eq!(current(&state), Some(Guarded));
219    }

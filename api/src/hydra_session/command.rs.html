12/// Who sent a command. A companion character comes with the step that
13/// brings it.
14#[derive(Clone, Copy, Debug, PartialEq, Eq)]
15pub enum Origin {
16    /// Typed by the person playing, at a window or through their Lich.
17    Player,
18    /// Put by one of the player's Lich scripts (`lich`).
19    Lich,
20    /// Hydra's own travel walking the character (`hydra-travel`).
21    Travel,
22    /// A trigger of the player's: one of their highlights that sends.
23    Trigger,
24    /// Hydra's room survey walking the character (`hydra-survey`).
25    Survey,
26}

28/// How a command's round trip ended.
29///
30/// A round trip owns what the game sends from the moment its line is
31/// written until the next prompt.
32#[derive(Clone, Debug, PartialEq, Eq)]
33pub enum Outcome {
34    /// The prompt closed the round trip; this is the first frame the game
35    /// sent for it.
36    Confirmed(Box<Frame>),
37    /// Nothing came back within the deadline, or only a prompt. It does not
38    /// mean the command did not happen.
39    Timeout,
40    /// The connection dropped first. The session reconnects; the command was
41    /// not sent, or its answer was lost with the connection.
42    Disconnected,
43    /// The session has ended.
44    Dead,
45    /// Hydra's command line took it (`claimant`): run or answered, and
46    /// never sent to the game.
47    Claimed,
48    /// Handed to the player's Lich (`lich`), which sends what it makes of
49    /// it.
50    Lich,
51    /// Still waiting to be sent when the player stopped everything, and
52    /// dropped: never sent.
53    Stopped,
54    /// Written to the game at once, as typed, its answer not waited for
55    /// ([`SessionHandle::send_typed`]).
56    Sent,
57}

59/// Whether a line Lich sent was written.
60#[derive(Clone, Copy, Debug, PartialEq, Eq)]
61pub(crate) enum Written {
62    /// It was.
63    Yes,
64    /// There was no connection for it, or the session is quitting.
65    No,

132/// A frontend's handle on a session. Cheap to clone; every clone is the
133/// same session.

135pub struct SessionHandle {

137    pub(crate) events: broadcast::Sender<Event>,
138    generation: GenerationCell,
139    /// Who runs the lines marked as Hydra's, once the binary has said.

156/// What takes a typed line, by its text: `true` when it was taken, and the

160/// What the binary installs on a session's handle, once, and lets go when
161/// the session ends: what it installs holds the handle (a command, a
162/// caster), and would otherwise keep the session, its events and its log,
163/// for as long as Hydra runs.
164pub(crate) struct Installed<T>(std::sync::Mutex<Holding<T>>);
165
166/// What an [`Installed`] holds.
167enum Holding<T> {
168    /// Nothing yet.
169    Unset,
170    /// What the binary installed.
171    Set(T),
172    /// The session has ended: nothing is installed again.
173    Gone,
174}
175
176impl<T> Default for Installed<T> {
177    fn default() -> Self {
178        Self(std::sync::Mutex::new(Holding::Unset))
179    }
180}
181
182impl<T> std::fmt::Debug for Installed<T> {
183    /// By hand: a [`Taker`] is a closure and has no `Debug`.
184    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
185        f.write_str(match *self.held() {
186            Holding::Unset => "Installed(unset)",
187            Holding::Set(_) => "Installed(set)",
188            Holding::Gone => "Installed(let go)",
189        })
190    }
191}

193impl<T: Clone> Installed<T> {
194    /// Install `value`: `false` when one is already, or the session has
195    /// ended.
196    pub(crate) fn set(&self, value: T) -> bool {
197        let mut held = self.held();
198        if !matches!(*held, Holding::Unset) {
199            return false;
200        }
201        *held = Holding::Set(value);
202        true
203    }
204
205    /// What is installed, copied, so nothing is held while it runs; `None`
206    /// before it is set and once it is let go.
207    pub(crate) fn get(&self) -> Option<T> {
208        match &*self.held() {
209            Holding::Set(value) => Some(value.clone()),
210            Holding::Unset | Holding::Gone => None,
211        }
212    }
213
214    /// Let go of what is installed: the session has ended. Dropped once the
215    /// lock is, as what it holds may hold handles of its own.
216    pub(crate) fn let_go(&self) {
217        let gone = std::mem::replace(&mut *self.held(), Holding::Gone);
218        drop(gone);
219    }
220}
221
222impl<T> Installed<T> {
223    fn held(&self) -> std::sync::MutexGuard<'_, Holding<T>> {
224        self.0
225            .lock()
226            .unwrap_or_else(std::sync::PoisonError::into_inner)
227    }
228}

372    /// Drop every command still waiting to be sent: how many. The one in
373    /// flight, already written, waits for its answer as ever. An alias still
374    /// waiting to send the rest of its commands sends none of them.
375    pub async fn clear_queue(&self) -> usize {
376        {
377            let mut waits = self
378                .alias_waits
379                .lock()
380                .unwrap_or_else(std::sync::PoisonError::into_inner);
381            waits.cancel();
382            *waits = tokio_util::sync::CancellationToken::new();
383        }
384        let (reply, answer) = oneshot::channel();
385        if !self.inbox(Inbox::Clear(reply)).await {
386            return 0;
387        }
388        answer.await.unwrap_or(0)
389    }

429    /// Send `line` and wait, up to `deadline`, for its round trip to end.
430    ///
431    /// `quit` (or `exit`) typed as a command quits the session the way
432    /// [`Self::quit`] does, so the game closing the connection afterwards is
433    /// not taken for a dropped one.
434    ///
435    /// A line marked with the command symbol is Hydra's (`claimant`): echoed,
436    /// run or answered, and never written to the game. Then one marked with
437    /// Lich's symbol, `;` unless chosen, is the player's Lich's
438    /// ([`crate::lich`]), and never the game's either; the rest goes to the
439    /// game, past Lich.
440    pub async fn send(&self, line: &str, origin: Origin, deadline: Duration) -> Outcome {
441        if self.claim(line, origin).is_some() {
442            return Outcome::Claimed;
443        }
444        if self.lichs(line) {
445            let _ = self.events.send(Event::Claimed(line.to_owned()));
446            return self.hand_to_lich(line);
447        }
448        self.to_the_game(line, origin, deadline).await
449    }

451    /// What the player typed at a window: first through the character's
452    /// aliases ([`crate::alias`]), each command an alias makes then sent as
453    /// typed, and never expanded again. A wait in an alias holds only that
454    /// alias: what follows it is sent when it is due, and what is typed
455    /// meanwhile goes at once. What an alias cannot send, or has no place
456    /// for, is said.
457    pub async fn send_typed(&self, line: &str) -> Outcome {
458        let Some(aliases) = self.aliases.get() else {
459            return self.send_as_typed(line).await;
460        };
461        let expanded = aliases.expand(line);
462        if let Some(problem) = aliases.problem() {
463            self.say_unlogged(crate::Notice::line(crate::NoticeKind::Error, problem).answering());
464        }
465        match expanded {
466            None => self.send_as_typed(line).await,
467            Some(Err(why)) => {
468                self.say_unlogged(crate::Notice::line(crate::NoticeKind::Error, why).answering());
469                Outcome::Claimed
470            }
471            Some(Ok(expanded)) => self.run_alias(expanded).await,
472        }
473    }

475    /// Send an alias's commands as typed, in order; from its first wait on,
476    /// on a task of its own, which [`Self::clear_queue`] stops.
477    async fn run_alias(&self, expanded: crate::alias::Expanded) -> Outcome {
478        use crate::steps::Step;
479        if let Some(rest) = &expanded.dropped {
480            self.say_unlogged(
481                crate::Notice::line(
482                    crate::NoticeKind::Info,
483                    format!(
484                        "The alias {} has more than one command and no \\? for `{rest}`: it was \
485                         not sent.",
486                        expanded.trigger
487                    ),
488                )
489                .answering(),
490            );
491        }
492        let mut outcome = Outcome::Claimed;
493        let mut steps = expanded.steps.into_iter();
494        while let Some(step) = steps.next() {
495            match step {
496                Step::Line(line) => outcome = self.send_as_typed(&line).await,
497                Step::Wait(wait) => {
498                    let rest: Vec<Step> = std::iter::once(Step::Wait(wait)).chain(steps).collect();
499                    let stopped = self
500                        .alias_waits
501                        .lock()
502                        .unwrap_or_else(std::sync::PoisonError::into_inner)
503                        .child_token();
504                    let handle = self.clone();
505                    tokio::spawn(async move {
506                        for step in rest {
507                            match step {
508                                Step::Wait(wait) => {
509                                    tokio::select! {
510                                        () = stopped.cancelled() => return,
511                                        () = tokio::time::sleep(wait) => {}
512                                    }
513                                }
514                                Step::Line(line) => {
515                                    if stopped.is_cancelled() {
516                                        return;
517                                    }
518                                    let _ = handle.send_as_typed(&line).await;
519                                }
520                            }
521                        }
522                    });
523                    break;
524                }
525            }
526        }
527        outcome
528    }

530    /// A line as the player typed it, past the aliases. Hydra's, by its
531    /// symbol; then Lich's, by Lich's symbol, or, while the player's Lich
532    /// runs, any line, for its aliases and hooks, and Lich sends what it
533    /// makes of it. The rest is **written to the game at once**, in the order
534    /// typed, without waiting for the last line's answer: typed lines go
535    /// straight through, and the game paces what it is sent. Waiting for each
536    /// answer, a key held on `look` was still being sent seconds after it was
537    /// let go. Only Hydra's own lines ([`Self::send`]) wait in the queue.
538    async fn send_as_typed(&self, line: &str) -> Outcome {
539        if self.claim(line, Origin::Player).is_some() {
540            return Outcome::Claimed;
541        }
542        if self.lichs(line) {
543            let _ = self.events.send(Event::Claimed(line.to_owned()));
544            return self.hand_to_lich(line);
545        }
546        if let Some(taker) = self.taker.get()
547            && taker(line)
548        {
549            return Outcome::Claimed;
550        }
551        if self.lich_running() {
552            return self.hand_to_lich(line);
553        }
554        match self.write_now(line, Origin::Player).await {
555            Written::Yes => Outcome::Sent,
556            Written::No => Outcome::Disconnected,
557        }
558    }

560    /// Write `line` to the game at once, past the queue, as `origin` sent
561    /// it: whether it was.
562    pub(crate) async fn write_now(&self, line: &str, origin: Origin) -> Written {
563        let (reply, written) = oneshot::channel();
564        let now = Inbox::Now {
565            line: line.to_owned(),
566            origin,
567            generation: self.generation(),
568            reply,
569        };
570        if !self.inbox(now).await {
571            return Written::No;
572        }
573        written.await.unwrap_or(Written::No)
574    }

593    /// Run `line` on Hydra's command line when its symbol marks it as
594    /// Hydra's, as `origin` sent it: echoed first, and answered when nothing
595    /// knows it. `None`: it is not Hydra's.
596    pub(crate) fn claim(&self, line: &str, origin: Origin) -> Option<crate::claimant::Claimed> {
597        let desk = self.desk.get().filter(|desk| desk.marks(line))?;
598        // Echoed before it runs: its answer comes after it in the story.
599        let _ = self.events.send(Event::Claimed(line.to_owned()));
600        let claimed = desk
601            .claim(line, origin)
602            .unwrap_or(crate::claimant::Claimed::Unknown);
603        if claimed == crate::claimant::Claimed::Unknown {
604            let symbol = desk.symbol();
605            let word = line.trim_start().trim_start_matches(symbol).trim();
606            self.say(
607                crate::Notice::line(
608                    crate::NoticeKind::Error,
609                    format!("I do not know {symbol}{word}. {symbol}help lists Hydra's commands."),
610                )
611                .answering(),
612            );
613        }
614        Some(claimed)
615    }

617    /// Queue `line` for the game, and wait up to `deadline` for its round
618    /// trip. The deadline covers the wait to be taken in too: a full inbox
619    /// does not hold a caller past it.
620    async fn to_the_game(&self, line: &str, origin: Origin, deadline: Duration) -> Outcome {
621        let (reply, answer) = oneshot::channel();
622        let envelope = Envelope {
623            line: line.to_owned(),
624            origin,
625            generation: self.generation.get(),
626            reply,
627        };
628        let round_trip = async {
629            if self.commands.send(Inbox::Command(envelope)).await.is_err() {
630                return Outcome::Dead;
631            }
632            answer.await.unwrap_or(Outcome::Dead)
633        };
634        // Out of time, the waiter is dropped: the queue sees it and moves
635        // on, or the line is never taken in at all.
636        tokio::time::timeout(deadline, round_trip)
637            .await
638            .unwrap_or(Outcome::Timeout)
639    }
640
641    /// Every event from now on.
642    #[must_use]
643    pub fn subscribe(&self) -> broadcast::Receiver<Event> {
644        self.events.subscribe()
645    }
646
647    /// What the session holds now; `None` once it has ended.
648    pub async fn snapshot(&self) -> Option<Snapshot> {
649        let (reply, answer) = oneshot::channel();
650        self.commands.send(Inbox::Snapshot(reply)).await.ok()?;
651        answer.await.ok()
652    }
653
654    /// Send `quit` and wait, up to `timeout`, for the game to close the
655    /// connection. The session ends either way, and does not reconnect.
656    pub async fn quit(&self, timeout: Duration) -> Farewell {
657        let (reply, answer) = oneshot::channel();
658        if self
659            .commands
660            .send(Inbox::Quit { timeout, reply })
661            .await
662            .is_err()
663        {
664            return Farewell::Unsent;
665        }
666        // A little longer than the session's own wait, so its answer wins.
667        match tokio::time::timeout(timeout + Duration::from_secs(1), answer).await {
668            Ok(Ok(farewell)) => farewell,
669            Ok(Err(_)) | Err(_) => Farewell::Unsent,
670        }
671    }

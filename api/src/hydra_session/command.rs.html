16/// What a chain has heard and not yet weighed, let go: a prompt that came
17/// while it waited on the clock, or while a behaviour ran, answers nothing
18/// it will send.
19fn heard_out(events: Option<&mut broadcast::Receiver<Event>>) {
20    if let Some(events) = events {
21        while events.try_recv().is_ok() {}

57/// How a command's round trip ended.
58///
59/// A round trip owns what the game sends from the moment its line is
60/// written until the next prompt.
61#[derive(Clone, Debug, PartialEq, Eq)]
62pub enum Outcome {
63    /// The prompt closed the round trip; this is the first frame the game
64    /// sent for it.
65    Confirmed(Box<Frame>),
66    /// Nothing came back within the deadline, or only a prompt. It does not
67    /// mean the command did not happen.
68    Timeout,
69    /// The connection dropped first. The session reconnects; the command was
70    /// not sent, or its answer was lost with the connection.
71    Disconnected,
72    /// The session has ended.
73    Dead,
74    /// Hydra's command line took it (`claimant`): run or answered, and
75    /// never sent to the game.
76    Claimed,
77    /// Handed to the player's Lich (`lich`), which sends what it makes of
78    /// it.
79    Lich,
80    /// Still waiting to be sent when the player stopped everything, and
81    /// dropped: never sent.
82    Stopped,
83    /// Written to the game at once, as typed, its answer not waited for
84    /// ([`SessionHandle::send_typed`]).
85    Sent,
86}

174/// A frontend's handle on a session. Cheap to clone; every clone is the
175/// same session.
176#[derive(Clone, Debug)]
177pub struct SessionHandle {
178    commands: mpsc::Sender<Inbox>,
179    pub(crate) events: broadcast::Sender<Event>,
180    generation: GenerationCell,
181    /// Who runs the lines marked as Hydra's, once the binary has said.
182    desk: crate::claimant::Slot,
183    /// The session's player log, when it keeps one, and what it writes:
184    /// what Hydra says is written there too.
185    log: std::sync::Arc<std::sync::OnceLock<(crate::PlayerLog, crate::player_log::Capture)>>,
186    /// The player's Lich, while one is attached (`lich`).
187    pub(crate) lich: crate::lich::Shared,
188    /// The character's aliases, once the binary has said where they are.
189    aliases: std::sync::Arc<std::sync::OnceLock<crate::alias::Aliases>>,
190    /// What stops an alias still waiting to send, and whatever else a typed
191    /// line set going ([`Self::stopping`]); replaced each time it stops them.
192    alias_waits: std::sync::Arc<std::sync::Mutex<tokio_util::sync::CancellationToken>>,
193    /// What takes a typed line the game would otherwise be sent, once the
194    /// binary has said ([`Self::set_taker`]).

197    /// `.note clip`: a window or the page puts it here as the copy is made.
198    clip: std::sync::Arc<std::sync::Mutex<Option<crate::notes::Clip>>>,

207/// What the binary installs on a session's handle, once, and lets go when
208/// the session ends: what it installs holds the handle (a command, a
209/// caster), and would otherwise keep the session, its events and its log,
210/// for as long as Hydra runs.
211pub(crate) struct Installed<T>(std::sync::Mutex<Holding<T>>);
212
213/// What an [`Installed`] holds.
214enum Holding<T> {
215    /// Nothing yet.
216    Unset,
217    /// What the binary installed.
218    Set(T),
219    /// The session has ended: nothing is installed again.
220    Gone,
221}
222
223impl<T> Default for Installed<T> {
224    fn default() -> Self {
225        Self(std::sync::Mutex::new(Holding::Unset))
226    }
227}
228
229impl<T> std::fmt::Debug for Installed<T> {
230    /// By hand: a [`Taker`] is a closure and has no `Debug`.
231    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
232        f.write_str(match *self.held() {
233            Holding::Unset => "Installed(unset)",
234            Holding::Set(_) => "Installed(set)",
235            Holding::Gone => "Installed(let go)",
236        })
237    }
238}

240impl<T: Clone> Installed<T> {
241    /// Install `value`: `false` when one is already, or the session has
242    /// ended.
243    pub(crate) fn set(&self, value: T) -> bool {
244        let mut held = self.held();
245        if !matches!(*held, Holding::Unset) {
246            return false;
247        }
248        *held = Holding::Set(value);
249        true
250    }
251
252    /// What is installed, copied, so nothing is held while it runs; `None`
253    /// before it is set and once it is let go.
254    pub(crate) fn get(&self) -> Option<T> {
255        match &*self.held() {
256            Holding::Set(value) => Some(value.clone()),
257            Holding::Unset | Holding::Gone => None,
258        }
259    }
260
261    /// Let go of what is installed: the session has ended. Dropped once the
262    /// lock is, as what it holds may hold handles of its own.
263    pub(crate) fn let_go(&self) {
264        let gone = std::mem::replace(&mut *self.held(), Holding::Gone);
265        drop(gone);
266    }
267}
268
269impl<T> Installed<T> {
270    fn held(&self) -> std::sync::MutexGuard<'_, Holding<T>> {
271        self.0
272            .lock()
273            .unwrap_or_else(std::sync::PoisonError::into_inner)
274    }
275}

454    /// Drop every command still waiting to be sent: how many. The one in
455    /// flight, already written, waits for its answer as ever. An alias still
456    /// waiting to send the rest of its commands sends none of them.
457    pub async fn clear_queue(&self) -> usize {
458        {
459            let mut waits = self
460                .alias_waits
461                .lock()
462                .unwrap_or_else(std::sync::PoisonError::into_inner);
463            waits.cancel();
464            *waits = tokio_util::sync::CancellationToken::new();
465        }
466        let (reply, answer) = oneshot::channel();
467        if !self.inbox(Inbox::Clear(reply)).await {
468            return 0;
469        }
470        answer.await.unwrap_or(0)
471    }

524    /// Send `line` and wait, up to `deadline`, for its round trip to end.
525    ///
526    /// `quit` (or `exit`) typed as a command quits the session the way
527    /// [`Self::quit`] does, so the game closing the connection afterwards is
528    /// not taken for a dropped one.
529    ///
530    /// A line marked with the command symbol is Hydra's (`claimant`): echoed,
531    /// run or answered, and never written to the game. Then one marked with
532    /// Lich's symbol, `;` unless chosen, is the player's Lich's
533    /// ([`crate::lich`]), and never the game's either; the rest goes to the
534    /// game, past Lich.
535    pub async fn send(&self, line: &str, origin: Origin, deadline: Duration) -> Outcome {
536        if self.claim(line, origin).is_some() {
537            return Outcome::Claimed;
538        }
539        if self.lichs(line) {
540            let _ = self.events.send(Event::Claimed(line.to_owned()));
541            self.log_typed(line, origin);
542            return self.hand_to_lich(line);
543        }
544        self.to_the_game(line, origin, deadline).await
545    }

547    /// What the player typed at a window: first through the character's
548    /// aliases ([`crate::alias`]), each command an alias makes then sent as
549    /// typed, and never expanded again. A wait in an alias holds only that
550    /// alias: what follows it is sent when it is due, and what is typed
551    /// meanwhile goes at once. What an alias cannot send, or has no place
552    /// for, is said.
553    pub async fn send_typed(&self, line: &str) -> Outcome {
554        let Some(aliases) = self.aliases.get() else {
555            return self.send_as_typed(line).await;
556        };
557        let expanded = aliases.expand(line);
558        if let Some(problem) = aliases.problem() {
559            self.say(crate::Notice::line(crate::NoticeKind::Error, problem).answering());
560        }
561        match expanded {
562            None => self.send_as_typed(line).await,
563            Some(Err(why)) => {
564                self.say(crate::Notice::line(crate::NoticeKind::Error, why).answering());
565                Outcome::Claimed
566            }
567            Some(Ok(expanded)) => self.run_alias(expanded).await,
568        }
569    }

571    /// Send an alias's commands as typed, in order; from its first wait on,
572    /// on a task of its own, which [`Self::clear_queue`] stops.
573    async fn run_alias(&self, expanded: crate::alias::Expanded) -> Outcome {
574        if let Some(rest) = &expanded.dropped {
575            self.say(
576                crate::Notice::line(
577                    crate::NoticeKind::Info,
578                    format!(
579                        "The alias {} has more than one command and no \\? for `{rest}`: it was \
580                         not sent.",
581                        expanded.trigger
582                    ),
583                )
584                .answering(),
585            );
586        }
587        // What an alias makes is sent as typed, and is never expanded again.
588        self.run_steps(expanded.steps, false).await
589    }

591    /// Run a chain: each command sent -- through the aliases when `typed`,
592    /// as typed otherwise -- a wait slept, `rt` and `ct` waited out on the
593    /// model's clock, a behaviour carried out. Whatever comes before the
594    /// first wait goes now and its outcome is answered; the rest runs in a
595    /// task of its own, stopped with everything else a typed line set going
596    /// ([`Self::stopping`]).
597    ///
598    /// Boxed: a typed command may be an alias, whose chain runs through
599    /// here again.
600    #[must_use]
601    pub fn run_steps(
602        &self,
603        steps: Vec<crate::steps::Step>,
604        typed: bool,
605    ) -> std::pin::Pin<Box<dyn Future<Output = Outcome> + Send + '_>> {
606        Box::pin(self.run_steps_now(steps, typed))
607    }

609    async fn run_steps_now(&self, steps: Vec<crate::steps::Step>, typed: bool) -> Outcome {
610        use crate::steps::Step;
611        // A chain that waits on the clock hears the game from its start, so
612        // an `rt` after a command can wait for that command's answer.
613        let mut events = if steps
614            .iter()
615            .any(|step| matches!(step, Step::Roundtime | Step::Casttime))
616        {
617            self.join().await.map(|(_, events)| events)
618        } else {
619            None
620        };
621        // Commands written since the chain last waited on the clock.
622        let mut sent = 0;
623        let mut outcome = Outcome::Claimed;
624        let mut steps = steps.into_iter();
625        while let Some(step) = steps.next() {
626            match step {
627                Step::Line(line) => {
628                    outcome = self.send_line(&line, typed).await;
629                    sent += 1;
630                }
631                Step::Behaviour { word, rest } => {
632                    let stopped = self.stopping();
633                    match self.behave(&word, &rest, typed, &stopped).await {
634                        Ok(went) if !stopped.is_cancelled() => outcome = went,
635                        Ok(_) | Err(None) => return outcome,
636                        Err(Some(why)) => {
637                            self.chain_stopped(&why);
638                            return outcome;
639                        }
640                    }
641                    sent = 0;
642                    heard_out(events.as_mut());
643                }
644                waits @ (Step::Wait(_) | Step::Roundtime | Step::Casttime) => {
645                    let rest: Vec<Step> = std::iter::once(waits).chain(steps).collect();
646                    let stopped = self
647                        .alias_waits
648                        .lock()
649                        .unwrap_or_else(std::sync::PoisonError::into_inner)
650                        .child_token();
651                    let handle = self.clone();
652                    tokio::spawn(async move {
653                        for step in rest {
654                            if stopped.is_cancelled() {
655                                return;
656                            }
657                            match step {
658                                Step::Wait(wait) => {
659                                    tokio::select! {
660                                        () = stopped.cancelled() => return,
661                                        () = tokio::time::sleep(wait) => {}
662                                    }
663                                }
664                                Step::Roundtime | Step::Casttime => {
665                                    if let Some(events) = events.as_mut() {
666                                        handle.answered(events, sent, &stopped).await;
667                                    }
668                                    let cast = matches!(step, Step::Casttime);
669                                    handle.clock_clear(&stopped, cast).await;
670                                    sent = 0;
671                                    heard_out(events.as_mut());
672                                }
673                                Step::Line(line) => {
674                                    let _ = handle.send_line(&line, typed).await;
675                                    sent += 1;
676                                }
677                                Step::Behaviour { word, rest } => {
678                                    if let Err(why) =
679                                        handle.behave(&word, &rest, typed, &stopped).await
680                                    {
681                                        if let Some(why) = why {
682                                            handle.chain_stopped(&why);
683                                        }
684                                        return;
685                                    }
686                                    sent = 0;
687                                    heard_out(events.as_mut());
688                                }
689                            }
690                        }
691                    });
692                    break;
693                }
694            }
695        }
696        outcome
697    }

699    /// Wait for the game to answer the `sent` commands a chain wrote since
700    /// it last waited on the clock: a prompt for each, or a roundtime or a
701    /// cast roundtime stated at one, whichever comes first; no longer than
702    /// `ANSWERED_WITHIN`, or until `stopped`. An `rt` straight after a
703    /// command would otherwise look at the clock before the game has said
704    /// the command's roundtime, and let the rest of the chain go at once.
705    async fn answered(
706        &self,
707        events: &mut broadcast::Receiver<Event>,
708        sent: usize,
709        stopped: &tokio_util::sync::CancellationToken,
710    ) {
711        let by = tokio::time::Instant::now() + ANSWERED_WITHIN;
712        let mut prompts = 0;
713        while prompts < sent {
714            let event = tokio::select! {
715                () = stopped.cancelled() => return,
716                () = tokio::time::sleep_until(by) => return,
717                event = events.recv() => event,
718            };
719            match event {
720                Ok(Event::Prompt(_)) => {
721                    prompts += 1;
722                    let Some(snapshot) = self.snapshot().await else {
723                        return;
724                    };
725                    let state = &snapshot.state;
726                    if state.in_roundtime() == Some(true) || state.in_casttime() == Some(true) {
727                        return;
728                    }
729                }
730                Ok(_) => {}
731                Err(_) => return,
732            }
733        }
734    }

746    /// A behaviour word of a chain, carried out (`crate::behaviour`); a
747    /// word with no behaviour yet is sent as the command it reads as.
748    /// `Err`: the chain stops, said why when `Some`, quietly when stopped.
749    async fn behave(
750        &self,
751        word: &str,
752        rest: &str,
753        typed: bool,
754        stopped: &tokio_util::sync::CancellationToken,
755    ) -> Result<Outcome, Option<String>> {
756        if word == "stance" {
757            return crate::behaviour::stance(self, rest, stopped)
758                .await
759                .map(|()| Outcome::Claimed)
760                .map_err(Some);
761        }
762        let line = if rest.is_empty() {
763            word.to_owned()
764        } else {
765            format!("{word} {rest}")
766        };
767        if let Some(assault) = crate::behaviour::assault::find(word) {
768            return crate::behaviour::assault::run(self, assault, &line, stopped)
769                .await
770                .map(|()| Outcome::Claimed);
771        }
772        Ok(self.send_line(&line, typed).await)
773    }

782    /// Wait until the model's roundtime -- or, `cast`, its cast roundtime
783    /// -- is over, looking every quarter second, no longer than
784    /// [`crate::steps::LONGEST_WAIT`]; or until `stopped`.
785    pub(crate) async fn clock_clear(
786        &self,
787        stopped: &tokio_util::sync::CancellationToken,
788        cast: bool,
789    ) {
790        let by = tokio::time::Instant::now() + crate::steps::LONGEST_WAIT;
791        while tokio::time::Instant::now() < by {
792            let Some(snapshot) = self.snapshot().await else {
793                return;
794            };
795            let running = if cast {
796                snapshot.state.in_casttime()
797            } else {
798                snapshot.state.in_roundtime()
799            };
800            if running != Some(true) {
801                return;
802            }
803            tokio::select! {
804                () = stopped.cancelled() => return,
805                () = tokio::time::sleep(Duration::from_millis(250)) => {}
806            }
807        }
808    }

810    /// A line as the player typed it, past the aliases. Hydra's, by its
811    /// symbol; then Lich's, by Lich's symbol, or, while the player's Lich
812    /// runs, any line, for its aliases and hooks, and Lich sends what it
813    /// makes of it. The rest is **written to the game at once**, in the order
814    /// typed, without waiting for the last line's answer: typed lines go
815    /// straight through, and the game paces what it is sent. Waiting for each
816    /// answer, a key held on `look` was still being sent seconds after it was
817    /// let go. Only Hydra's own lines ([`Self::send`]) wait in the queue.
818    ///
819    /// A line sent on another character by `.to` or `.all` comes here on
820    /// that character: typed there, Lich included, but never through its
821    /// aliases, so two characters' aliases cannot pass a line back and
822    /// forth without end.
823    pub async fn send_as_typed(&self, line: &str) -> Outcome {
824        if let Some(claimed) = self.claim(line, Origin::Player) {
825            // A word Hydra has no command for, but the aliases have: `.s
826            // sword back` runs the alias `s`, as a front end ran a script
827            // of that name by its dot.
828            if claimed == crate::claimant::Claimed::Unknown
829                && let Some(expanded) = self.alias_behind_symbol(line)
830            {
831                return self.run_alias(expanded).await;
832            }
833            return Outcome::Claimed;
834        }
835        if self.lichs(line) {
836            let _ = self.events.send(Event::Claimed(line.to_owned()));
837            self.log_typed(line, Origin::Player);
838            return self.hand_to_lich(line);
839        }
840        if let Some(taker) = self.taker.get()
841            && taker(line)
842        {
843            return Outcome::Claimed;
844        }
845        if self.lich_takes_typing() {
846            return self.hand_to_lich(line);
847        }
848        match self.write_now(line, Origin::Player).await {
849            Written::Yes => Outcome::Sent,
850            Written::No => Outcome::Disconnected,
851        }
852    }

912    /// Run `line` on Hydra's command line when its symbol marks it as
913    /// Hydra's, as `origin` sent it: echoed first, and answered when nothing
914    /// knows it. `None`: it is not Hydra's.
915    pub(crate) fn claim(&self, line: &str, origin: Origin) -> Option<crate::claimant::Claimed> {
916        let desk = self.desk.get().filter(|desk| desk.marks(line))?;
917        // Echoed before it runs: its answer comes after it in the story,
918        // and in the player log, where the player typed it.
919        let _ = self.events.send(Event::Claimed(line.to_owned()));
920        self.log_typed(line, origin);
921        let claimed = desk
922            .claim(line, origin)
923            .unwrap_or(crate::claimant::Claimed::Unknown);
924        if claimed == crate::claimant::Claimed::Unknown && self.alias_behind_symbol(line).is_none()
925        {
926            let symbol = desk.symbol();
927            let word = line.trim_start().trim_start_matches(symbol).trim();
928            self.say(
929                crate::Notice::line(
930                    crate::NoticeKind::Error,
931                    format!("I do not know {symbol}{word}. {symbol}help lists Hydra's commands."),
932                )
933                .answering(),
934            );
935        }
936        Some(claimed)
937    }

939    /// What `line`, Hydra's symbol first, expands to as an alias when its
940    /// word is not one of Hydra's commands: `.s sword back` as `s sword
941    /// back`. `None` when no alias fits, or the alias cannot be sent, which

949    /// Queue `line` for the game, and wait up to `deadline` for its round
950    /// trip. The deadline covers the wait to be taken in too: a full inbox
951    /// does not hold a caller past it.
952    async fn to_the_game(&self, line: &str, origin: Origin, deadline: Duration) -> Outcome {
953        let (reply, answer) = oneshot::channel();
954        let envelope = Envelope {
955            line: line.to_owned(),
956            origin,
957            generation: self.generation.get(),
958            reply,
959        };
960        let round_trip = async {
961            if self.commands.send(Inbox::Command(envelope)).await.is_err() {
962                return Outcome::Dead;
963            }
964            answer.await.unwrap_or(Outcome::Dead)
965        };
966        // Out of time, the waiter is dropped: the queue sees it and moves
967        // on, or the line is never taken in at all.
968        tokio::time::timeout(deadline, round_trip)
969            .await
970            .unwrap_or(Outcome::Timeout)
971    }

986    /// Send `quit` and wait, up to `timeout`, for the game to close the
987    /// connection. The session ends either way, and does not reconnect.
988    ///
989    /// With the player's Lich reading what is typed, the quit is handed to
990    /// it, to stop its scripts and save before it passes the quit on; it has
991    /// [`crate::actor::LICH_QUIT`] to, and `timeout` starts once it has.
992    pub async fn quit(&self, timeout: Duration) -> Farewell {
993        let (reply, answer) = oneshot::channel();
994        if self
995            .commands
996            .send(Inbox::Quit { timeout, reply })
997            .await
998            .is_err()
999        {
1000            return Farewell::Unsent;
1001        }
1002        // A little longer than the session's own wait, Lich's turn included,
1003        // so its answer wins.
1004        let waited = timeout + crate::actor::LICH_QUIT + Duration::from_secs(1);
1005        match tokio::time::timeout(waited, answer).await {
1006            Ok(Ok(farewell)) => farewell,
1007            Ok(Err(_)) | Err(_) => Farewell::Unsent,
1008        }
1009    }

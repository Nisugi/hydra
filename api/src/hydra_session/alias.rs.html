1//! Aliases: a short trigger typed in place of one or more commands.
2//!
3//! `aliases.json` in the data folder keeps them: everyone's, then each
4//! character's own, each a trigger and its target:
5//!
6//! ```json
7//! { "global": { "ls": "look" },
8//!   "characters": { "Prime_Ashryn": { "zap": "prep 901\\rcast \\?" } } }
9//! ```
10//!
11//! **A line expands** when it starts with a trigger, capitals aside, and
12//! the trigger is followed by a space or the end of the line; what follows
13//! it, its leading spaces dropped, is *the rest*. The character's own are
14//! tried before everyone's, and of those that fit, the longest trigger is
15//! the one meant: `go n` over `go`.
16//!
17//! **The target** is commands and waits, as a key macro's
18//! ([`crate::steps`]): `\r` between two commands, written as the two
19//! characters, and a piece `s1.5` a wait of a second and a half. `\?` is the
20//! rest:
21//!
22//! - no rest: each `\?` is dropped;
23//! - a target with a `\?`: each is the rest; a command `;e… "\?"`, run by
24//!   the player's Lich, takes it as a quoted string;
25//! - a target of one command and no `\?`: the rest goes after it, `ls in
26//!   bag` as `look in bag`;
27//! - a target of several and no `\?`: the rest has nowhere to go, and is
28//!   said to be dropped ([`Expanded::dropped`]).
29//!
30//! What an alias makes is sent as typed, and is never expanded again, so no
31//! alias can loop.
32

42/// The file's name in the data folder.
43pub const FILE: &str = "aliases.json";

45/// The rest of the line, as a target writes it.
46const REST: &str = "\\?";

48/// Two commands' break, as a target writes it.
49const BREAK: &str = "\\r";

51/// Whose an alias is.
52#[derive(Clone, Copy, Debug, PartialEq, Eq)]
53pub enum Whose<'a> {
54    /// Everyone's.
55    Everyone,
56    /// One character's, by its key ([`character`]).
57    Own(&'a str),
58}
59
60/// A character's key in the file: `<instance>_<name>`, as its other files
61/// are named.
62#[must_use]
63pub fn character(instance: &str, name: &str) -> String {
64    format!("{instance}_{name}")
65}

67/// Every alias, everyone's and each character's.
68#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
69pub struct AliasFile {
70    /// Everyone's: trigger, target.
71    #[serde(default, skip_serializing_if = "BTreeMap::is_empty")]
72    pub global: BTreeMap<String, String>,
73    /// Each character's own, by its key.
74    #[serde(default, skip_serializing_if = "BTreeMap::is_empty")]
75    pub characters: BTreeMap<String, BTreeMap<String, String>>,
76}

78/// What a line expands to.
79#[derive(Clone, Debug, PartialEq, Eq)]
80pub struct Expanded {
81    /// The trigger, as kept.
82    pub trigger: String,
83    /// Its commands and waits, in order.
84    pub steps: Vec<Step>,
85    /// The rest of the line, when the target had nowhere to put it.
86    pub dropped: Option<String>,
87}
88
89impl AliasFile {
90    /// The file in the data folder `dir`; none there, no aliases.
91    ///
92    /// # Errors
93    ///
94    /// It is there and does not read.
95    pub fn load(dir: &Path) -> Result<Self, String> {
96        match std::fs::read_to_string(dir.join(FILE)) {
97            Ok(text) => serde_json::from_str(&text).map_err(|why| format!("{FILE}: {why}")),
98            Err(why) if why.kind() == std::io::ErrorKind::NotFound => Ok(Self::default()),
99            Err(why) => Err(format!("{FILE} cannot be read: {why}")),
100        }
101    }
102
103    /// The aliases `whose` names.
104    #[must_use]
105    pub fn of(&self, whose: Whose<'_>) -> Option<&BTreeMap<String, String>> {
106        match whose {
107            Whose::Everyone => Some(&self.global),
108            Whose::Own(key) => self.characters.get(key),
109        }
110    }
111
112    fn of_mut(&mut self, whose: Whose<'_>) -> &mut BTreeMap<String, String> {
113        match whose {
114            Whose::Everyone => &mut self.global,
115            Whose::Own(key) => self.characters.entry(key.to_owned()).or_default(),
116        }
117    }
118
119    /// Keep `trigger` as `target` for `whose`, in place of one of the same
120    /// trigger, capitals aside, which keeps its own spelling. The target it
121    /// had, when it had one.
122    ///
123    /// # Errors
124    ///
125    /// No trigger, a trigger with `=` in it, or a target that cannot be
126    /// sent.
127    pub fn put(
128        &mut self,
129        whose: Whose<'_>,
130        trigger: &str,
131        target: &str,
132    ) -> Result<Option<String>, String> {
133        let trigger = trigger.trim();
134        let target = target.trim();
135        if trigger.is_empty() {
136            return Err("an alias needs a trigger".to_owned());
137        }
138        if trigger.contains('=') {
139            return Err("a trigger cannot hold `=`".to_owned());
140        }
141        steps::steps(&target.replace(BREAK, "\r")).map_err(|why| format!("its target: {why}"))?;
142        let aliases = self.of_mut(whose);
143        let kept = named(aliases, trigger).unwrap_or(trigger).to_owned();
144        Ok(aliases.insert(kept, target.to_owned()))
145    }
146
147    /// Take `trigger` away from `whose`, capitals aside: its spelling and
148    /// target, when there was one.
149    pub fn remove(&mut self, whose: Whose<'_>, trigger: &str) -> Option<(String, String)> {
150        let aliases = self.of_mut(whose);
151        let kept = named(aliases, trigger.trim())?.to_owned();
152        let target = aliases.remove(&kept)?;
153        if let Whose::Own(key) = whose
154            && self.characters.get(key).is_some_and(BTreeMap::is_empty)
155        {
156            self.characters.remove(key);
157        }
158        Some((kept, target))
159    }
160
161    /// What `line`, typed on the character keyed `character`, expands to;
162    /// `None` when no alias fits it.
163    ///
164    /// # Errors
165    ///
166    /// The alias that fits cannot be sent: a wait too long, or nothing left
167    /// to send.
168    #[must_use]
169    pub fn expand(&self, character: Option<&str>, line: &str) -> Option<Result<Expanded, String>> {
170        let own = character.and_then(|key| self.characters.get(key));
171        let (trigger, target, rest) = own
172            .and_then(|aliases| fitting(aliases, line))
173            .or_else(|| fitting(&self.global, line))?;
174        Some(
175            expanded(target, rest)
176                .map(|(steps, dropped)| Expanded {
177                    trigger: trigger.to_owned(),
178                    steps,
179                    dropped,
180                })
181                .map_err(|why| format!("The alias {trigger}: {why}")),
182        )
183    }
184}

186/// The trigger of `aliases` that is `trigger`, capitals aside.
187fn named<'a>(aliases: &'a BTreeMap<String, String>, trigger: &str) -> Option<&'a str> {
188    aliases
189        .keys()
190        .find(|have| have.eq_ignore_ascii_case(trigger))
191        .map(String::as_str)
192}
193
194/// The longest trigger of `aliases` that `line` starts with, followed by a
195/// space or the end: it, its target, and the rest of the line.
196fn fitting<'a>(
197    aliases: &'a BTreeMap<String, String>,
198    line: &'a str,
199) -> Option<(&'a str, &'a str, &'a str)> {
200    aliases
201        .iter()
202        .filter_map(|(trigger, target)| {
203            let head = line.get(..trigger.len())?;
204            let after = &line[trigger.len()..];
205            (head.eq_ignore_ascii_case(trigger)
206                && after.chars().next().is_none_or(char::is_whitespace))
207            .then(|| (trigger.as_str(), target.as_str(), after.trim_start()))
208        })
209        .max_by_key(|(trigger, ..)| trigger.len())
210}

212/// `target`'s steps with `rest` put where it says; and the rest, when it
213/// had nowhere to go.
214fn expanded(target: &str, rest: &str) -> Result<(Vec<Step>, Option<String>), String> {
215    let mut pieces: Vec<String> = target
216        .replace(BREAK, "\r")
217        .split(['\r', '\n'])
218        .map(str::to_owned)
219        .collect();
220    let mut dropped = None;
221    if rest.is_empty() {
222        for piece in &mut pieces {
223            *piece = piece.replace(REST, "");
224        }
225    } else if pieces.iter().any(|piece| piece.contains(REST)) {
226        for piece in &mut pieces {
227            *piece = if piece.trim_start().starts_with(";e") && piece.contains("\"\\?\"") {
228                piece.replace("\"\\?\"", &quoted(rest))
229            } else {
230                piece.replace(REST, rest)
231            };
232        }
233    } else {
234        let commands: Vec<usize> = pieces
235            .iter()
236            .enumerate()
237            .filter(|(_, piece)| !piece.trim().is_empty() && steps::wait(piece.trim()).is_none())
238            .map(|(at, _)| at)
239            .collect();
240        match commands.as_slice() {
241            [only] => {
242                let piece = &mut pieces[*only];
243                *piece = format!("{} {rest}", piece.trim_end());
244            }
245            _ => dropped = Some(rest.to_owned()),
246        }
247    }
248    let steps = steps::steps(&pieces.join("\r"))?;
249    Ok((steps, dropped))
250}

252/// `text` as a quoted string Lich's Ruby reads back as `text`.
253fn quoted(text: &str) -> String {
254    let mut out = String::with_capacity(text.len() + 2);
255    out.push('"');
256    let mut letters = text.chars().peekable();
257    while let Some(letter) = letters.next() {
258        match letter {
259            '\\' => out.push_str("\\\\"),
260            '"' => out.push_str("\\\""),
261            '\n' => out.push_str("\\n"),
262            '\t' => out.push_str("\\t"),
263            '#' if matches!(letters.peek(), Some('{' | '$' | '@')) => out.push_str("\\#"),
264            letter => out.push(letter),
265        }
266    }
267    out.push('"');
268    out
269}

271/// One character's aliases, as its session expands what is typed: the file
272/// read again whenever its modified time moves, so a change by hand or by
273/// `.alias` holds from the next line.
274#[derive(Debug)]
275pub struct Aliases {
276    dir: PathBuf,
277    character: String,
278    read: Mutex<Read>,
279}
280
281/// The file as last read.
282#[derive(Debug, Default)]
283struct Read {
284    /// When it was modified, as last looked; `None`, not there.
285    modified: Option<SystemTime>,
286    /// It has been looked at.
287    looked: bool,
288    file: Arc<AliasFile>,
289    /// Why it did not read, until said.
290    problem: Option<String>,
291}
292
293impl Aliases {
294    /// The aliases of the character keyed `character` ([`character`]),
295    /// from the data folder `dir`.
296    #[must_use]
297    pub fn new(dir: PathBuf, character: String) -> Self {
298        Self {
299            dir,
300            character,
301            read: Mutex::default(),
302        }
303    }
304
305    /// The aliases of the character `name` on the game `game_code`, from the
306    /// data folder `dir`; `None` for a game Hydra does not know.
307    #[must_use]
308    pub fn for_character(dir: PathBuf, game_code: &str, name: &str) -> Option<Self> {
309        let instance = hydra_net::instance(game_code)?;
310        Some(Self::new(dir, character(instance, name)))
311    }
312
313    /// What `line` expands to; `None` when no alias fits. A file that does
314    /// not read has no aliases until it does.
315    #[must_use]
316    pub fn expand(&self, line: &str) -> Option<Result<Expanded, String>> {
317        self.fresh().expand(Some(&self.character), line)
318    }
319
320    /// Why the file did not read, once.
321    pub fn problem(&self) -> Option<String> {
322        self.read
323            .lock()
324            .unwrap_or_else(PoisonError::into_inner)
325            .problem
326            .take()
327    }
328
329    /// The file, read again when it changed.
330    fn fresh(&self) -> Arc<AliasFile> {
331        let modified = std::fs::metadata(self.dir.join(FILE))
332            .and_then(|meta| meta.modified())
333            .ok();
334        let mut read = self.read.lock().unwrap_or_else(PoisonError::into_inner);
335        if !read.looked || read.modified != modified {
336            read.looked = true;
337            read.modified = modified;
338            match AliasFile::load(&self.dir) {
339                Ok(file) => {
340                    read.file = Arc::new(file);
341                    read.problem = None;
342                }
343                Err(why) => {
344                    read.file = Arc::default();
345                    read.problem = Some(format!("No aliases while {why}"));
346                }
347            }
348        }
349        Arc::clone(&read.file)
350    }
351}

353/// Change the file in the data folder `dir` by `change`, and write it,
354/// with its lock held, so two changes at once both land. What `change` said.
355///
356/// # Errors
357///
358/// The file does not read, `change` refused, or it was not written; the
359/// file is as it was. A file that does not read is never written over.
360pub fn change<T>(
361    dir: &Path,
362    change: impl FnOnce(&mut AliasFile) -> Result<T, String>,
363) -> Result<T, String> {
364    let path = dir.join(FILE);
365    crate::store::changing(&path, || {
366        let mut file = AliasFile::load(dir)?;
367        let said = change(&mut file)?;
368        crate::store::save_json(dir, &path, &file)
369            .map_err(|why| format!("{FILE} was not written: {why}"))?;
370        Ok(said)
371    })
372}

382    fn file(global: &[(&str, &str)], own: &[(&str, &str)]) -> AliasFile {
383        let mut file = AliasFile::default();
384        for (trigger, target) in global {
385            file.put(Whose::Everyone, trigger, target).unwrap();
386        }
387        for (trigger, target) in own {
388            file.put(Whose::Own(ME), trigger, target).unwrap();
389        }
390        file
391    }
392
393    /// The commands and waits `line` expands to, and what was dropped.
394    fn sent(file: &AliasFile, line: &str) -> Option<(Vec<String>, Option<String>)> {
395        let expanded = file.expand(Some(ME), line)?.unwrap();
396        let steps = expanded
397            .steps
398            .iter()
399            .map(|step| match step {
400                Step::Line(line) => line.clone(),
401                Step::Wait(wait) => format!("wait {}ms", wait.as_millis()),
402            })
403            .collect();
404        Some((steps, expanded.dropped))
405    }
406
407    fn lines(file: &AliasFile, line: &str) -> Vec<String> {
408        sent(file, line).map(|(steps, _)| steps).unwrap_or_default()
409    }

475    #[test]
476    fn the_rest_goes_where_the_target_says() {
477        let aliases = file(
478            &[
479                ("zap", "prep 901\\rcast \\?"),
480                ("ls", "look"),

486        assert_eq!(lines(&aliases, "zap kobold"), ["prep 901", "cast kobold"]);

1//! How long the player log is kept.
2//!
3//! Forever unless the player says otherwise, and no cap on size. With a number
4//! of days set, what is wholly older than that goes: a day-file once its day
5//! is, an archive once every day in it is. Never part of an archive.
6//!
7//! # Never a surprise
8//!
9//! Retention never surprises: deletion is previewed. The same [`doomed`] that
10//! [`prune`] deletes is what the *Player log* page and `.history` show the
11//! player first, so what they are told is what goes. It goes at the next login,
12//! not when the setting is changed.
13
14use std::fs;
15use std::io;
16use std::path::{Path, PathBuf};

18use hydra_net::eastern;

22/// One file that retention removes.
23#[derive(Clone, Debug, PartialEq, Eq)]
24pub struct Doomed {
25    /// Where it is.
26    pub path: PathBuf,
27    /// Its newest day, `YYYY-MM-DD`: every day in it is this old or older.
28    pub newest: String,
29    /// Its size.
30    pub bytes: u64,
31}
32
33/// What [`prune`] did.
34#[derive(Clone, Debug, Default, PartialEq, Eq)]
35pub struct Pruned {
36    /// The files removed, oldest first.
37    pub removed: Vec<Doomed>,
38    /// Archives left alone because what they hold could not be read, each
39    /// with why.
40    pub skipped: Vec<String>,
41}
42
43/// The first day kept when `keep_days` days are kept counting `today`
44/// (`YYYY-MM-DD`, the player's date); `None` to keep everything.
45///
46/// `keep_days` of 0 is *forever*, as the settings page says it.
47#[must_use]
48pub fn first_kept(today: &str, keep_days: u32) -> Option<String> {
49    if keep_days == 0 {
50        return None;
51    }
52    let today = eastern::parse(today)?;
53    Some(eastern::format(eastern::civil_from_days(
54        eastern::days_from_civil(today) - i64::from(keep_days) + 1,
55    )))
56}
57
58/// What keeping `keep_days` days, counting `today`, removes: every day-file
59/// and archive whose newest day is before the first day kept. Oldest first.
60///
61/// An archive whose manifest cannot be read is left out: what it holds is
62/// not known.
63///
64/// # Errors
65///
66/// A failure to read the directory.
67pub fn doomed(
68    root: &Path,
69    character: &str,
70    keep_days: u32,
71    today: &str,
72) -> io::Result<Vec<Doomed>> {
73    survey(root, character, keep_days, today).map(|(doomed, _)| doomed)
74}
75
76/// [`doomed`], and each archive left out because it could not be read.
77fn survey(
78    root: &Path,
79    character: &str,
80    keep_days: u32,
81    today: &str,
82) -> io::Result<(Vec<Doomed>, Vec<String>)> {
83    let mut skipped = Vec::new();
84    let Some(first) = first_kept(today, keep_days) else {
85        return Ok((Vec::new(), skipped));
86    };
87    let size = |path: &Path| fs::metadata(path).map_or(0, |m| m.len());
88    let mut doomed = Vec::new();
89    for path in writer::days(root, character)? {
90        if let Some(day) = writer::day_of(&path).filter(|day| *day < first) {
91            doomed.push(Doomed {
92                bytes: size(&path),
93                newest: day,
94                path,
95            });
96        }
97    }
98    for path in archive::archives(root, character)? {
99        let names = match archive::names(&path) {
100            Ok(names) => names,
101            Err(e) => {
102                skipped.push(format!("{}: {e}", path.display()));
103                continue;
104            }
105        };
106        let newest = names
107            .iter()
108            .filter_map(|name| writer::piece(name).map(|(day, _)| day))
109            .max();
110        if let Some(newest) = newest.filter(|day| *day < first) {
111            doomed.push(Doomed {
112                bytes: size(&path),
113                newest,
114                path,
115            });
116        }
117    }
118    doomed.sort_by(|a, b| a.newest.cmp(&b.newest).then_with(|| a.path.cmp(&b.path)));
119    Ok((doomed, skipped))
120}

122/// Remove what [`doomed`] names, and say what went, and what was left
123/// alone: an archive that cannot be read, or a file that would not go, does
124/// not stop the rest.
125///
126/// # Errors
127///
128/// A failure to list the directory.
129pub fn prune(root: &Path, character: &str, keep_days: u32, today: &str) -> io::Result<Pruned> {
130    let (doomed, mut skipped) = survey(root, character, keep_days, today)?;
131    let mut removed = Vec::new();
132    for file in doomed {
133        match fs::remove_file(&file.path) {
134            Ok(()) => removed.push(file),
135            Err(e) => skipped.push(format!("{}: {e}", file.path.display())),
136        }
137    }
138    Ok(Pruned { removed, skipped })
139}

141/// What [`doomed`] found, as one line a player reads: `nothing`, or how many
142/// files, how much, and back to when.
143#[must_use]
144pub fn preview(doomed: &[Doomed]) -> String {
145    let (Some(oldest), Some(newest)) = (doomed.first(), doomed.last()) else {
146        return "Nothing is old enough to remove.".to_owned();
147    };
148    let bytes: u64 = doomed.iter().map(|d| d.bytes).sum();
149    format!(
150        "The next login removes {} file{} ({} KB), the days up to {} (the oldest {}).",
151        doomed.len(),
152        if doomed.len() == 1 { "" } else { "s" },
153        bytes.div_ceil(1024),
154        newest.newest,
155        oldest.newest
156    )
157}

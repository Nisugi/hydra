1//! One connection's feed into the player log: what is captured, and what a
2//! line is tagged once its chunk is understood.
3//!
4//! A line is tagged where the game sent it, and after that where the stream
5//! rules put it ([`crate::stream_rules`]): the built-in rule puts each main
6//! line of a chunk with fighting in it in `combat`, so the combat feed reads
7//! apart from the rest.
8//!
9//! # A line's class is not known when the line ends
10//!
11//! The blob is the model's prompt-to-prompt chunk, and the combat tracker
12//! classifies it when the prompt closes it. So the feed **holds a chunk's
13//! lines and writes them at the prompt**, when the actor can say whether the
14//! chunk had fighting in it and which lines the reader marked. Everything is
15//! held, not only the main window's lines, so the file stays in the order
16//! things happened; each line keeps the stamp of the moment it arrived.
17//!
18//! # The tag: `source/class`
19//!
20//! A line from a combat chunk is tagged `main/combat`. **The class never
21//! replaces the source**: `main` is where the game sent it, `combat` is what
22//! the rules made of it. A line no rule matches stays `main`. A command, a
23//! readout, is never sorted.
24//!
25//! # Every feed a player reads is written
26//!
27//! [`Capture`] decides what is WRITTEN; what is SHOWN is the reader's
28//! choice alone. A player has no choice of what is written: a feed left out
29//! is a hole no reader can fill, and the feeds cost little beside the story.
30//! The readouts -- streams the game clears and rewrites whole -- are left
31//! out, the connection's recording keeping them; a capture that asks for
32//! one writes it **only when it changes**.
33

104/// What a closed chunk answered.
105///
106/// A chunk is the game's answer to one thing: the oldest command sent and
107/// not yet answered, or something the game did on its own -- a creature
108/// arriving, another player's swing, a spell wearing off. A command sent
109/// while the one before is still unanswered is queued by the game, and its
110/// answer comes in a chunk of its own after.
111#[derive(Clone, Debug, PartialEq, Eq)]
112pub enum Cause {
113    /// A command, as it went to the game.
114    Command {
115        /// What was sent.
116        text: String,
117        /// The tag it is logged under, its sender after it: `cmd/lich`.
118        feed: String,
119    },
120    /// The game's own: no command was waiting for an answer, the chunk
121    /// said nothing in the main window, or the combat reader found only
122    /// others' attacks in it.
123    Game,
124}
125
126impl Cause {
127    /// The command it was, when it was one: what was sent, and whether a
128    /// script sent it.
129    #[must_use]
130    pub fn command(&self) -> Option<(&str, bool)> {
131        match self {
132            Self::Command { text, feed } => Some((
133                text,
134                super::sender(feed) == Some(crate::Origin::Lich.sender()),
135            )),
136            Self::Game => None,
137        }
138    }

385    /// One frame the game sent: its text gathered into lines, each held
386    /// once it ends.
387    pub fn frame(&mut self, frame: &Frame) {
388        match frame {
389            Frame::Text(text) => {
390                let (line, links, room) = self.unfinished.entry(text.stream.clone()).or_default();
391                if line.is_empty() {
392                    *room = RoomPart::of(text.style.preset.as_deref());
393                }
394                let from = line.len();
395                line.push_str(&text.content);
396                if let Some(does) = text.link.as_ref().and_then(Does::of) {
397                    links::add(links, (from, line.len()), does);
398                }
399                if text.ends_line {
400                    let (whole, links) = (std::mem::take(line), std::mem::take(links));
401                    let room = std::mem::take(room);
402                    let mut at = 0;
403                    for (index, piece) in whole.split('\n').enumerate() {
404                        let piece_links = links::within(&links, (at, at + piece.len()));
405                        self.linked(
406                            &text.stream,
407                            piece,
408                            piece_links,
409                            room.filter(|_| index == 0),
410                        );
411                        at += piece.len() + 1;
412                    }
413                }
414            }
415            Frame::ClearStream { id, .. } => {
416                self.unfinished.remove(id);
417            }
418            _ => {}
419        }
420    }

435    /// One command, as it went to the game, under who sent it: held for
436    /// its chunk, and waiting for its answer. The client's own protocol
437    /// commands (`_menu`) get no answer of their own and wait for none.
438    pub fn command(&mut self, line: &str, origin: crate::Origin) {
439        let text = line.trim_end_matches(['\r', '\n']);
440        if text.is_empty() {
441            return;
442        }
443        let feed = super::command_tag(origin);
444        if !text.starts_with('_') {
445            self.awaiting.push_back(Awaiting {
446                text: text.to_owned(),
447                feed: feed.clone(),
448                sent: Instant::now(),
449            });
450            while self.awaiting.len() > AWAITING_MOST {
451                self.awaiting.pop_front();
452            }
453        }
454        self.hold(&feed, text, Vec::new(), None);
455    }

457    /// The chunk under way told briefly, as `facts` -- the combat reader's
458    /// word on it -- tell of it ([`crate::brief`]): its main lines with
459    /// words. `None` when there was no attack in it.
460    #[must_use]
461    pub fn brief(
462        &self,
463        facts: &[hydra_model::state::combat::ChunkFacts],
464    ) -> Option<crate::brief::Brief> {
465        let lines: Vec<(&str, &[LogLink])> = self
466            .held
467            .iter()
468            .filter(|line| line.feed == MAIN && !line.text.trim().is_empty())
469            .map(|line| (line.text.as_str(), line.links.as_slice()))
470            .collect();
471        crate::brief::Brief::of(&lines, facts)
472    }

474    /// What `held`, a chunk's lines, answered ([`Cause`]): the oldest
475    /// command still waiting, taken, when the chunk said anything in the
476    /// main window and its attacks, if any, are not only others'. An attack
477    /// waiting (`rules`' attacks) is answered by a chunk with no attack in
478    /// it only when the chunk speaks to the character -- `You cannot`,
479    /// `...wait` --: a creature arriving between an attack and its answer is
480    /// the game's own. A command that waited too long is let go first.
481    fn cause_of(&mut self, held: &[Held], ours: Option<bool>, rules: &Compiled) -> Cause {
482        let now = Instant::now();
483        self.awaiting
484            .retain(|waiting| now.saturating_duration_since(waiting.sent) < AWAITING_FOR);
485        let mut main = held.iter().filter(|line| line.feed == MAIN);
486        let answered = ours != Some(false)
487            && main.clone().next().is_some()
488            && (ours == Some(true)
489                || self
490                    .awaiting
491                    .front()
492                    .is_none_or(|waiting| !rules.rules().attacks(&waiting.text))
493                || main.any(|line| to_the_character(&line.text)));
494        match answered.then(|| self.awaiting.pop_front()).flatten() {
495            Some(waiting) => Cause::Command {
496                text: waiting.text,
497                feed: waiting.feed,
498            },
499            None => Cause::Game,
500        }
501    }

503    fn hold(&mut self, feed: &str, text: &str, links: Vec<LogLink>, room: Option<RoomPart>) {
504        self.held.push(Held {
505            at: hydra_net::line_time(),
506            feed: feed.to_owned(),
507            text: text.to_owned(),
508            links,
509            room,
510        });
511        if self.held.len() >= MAX_HELD {
512            self.close(Chunk::default(), &[]);
513        }
514    }

516    /// The prompt closed the chunk: write what was held, each line sorted by
517    /// the stream rules, as `chunk` says what it was; `marks` the lines the
518    /// combat reader marked, by their text.
519    ///
520    /// The command the chunk answered ([`Cause`]) goes with it
521    /// ([`crate::stream_rules::Rules::command`]), whichever chunk it was
522    /// sent in; a stance or a target answered while a hostile creature is
523    /// by takes the chunk as one with fighting in it. A command is logged
524    /// as it was sent.
525    pub fn close(&mut self, chunk: Chunk, marks: &[LineMark]) {
526        let Chunk {
527            fighting,
528            hostile,
529            looted,
530            ours,
531        } = chunk;
532        let held = std::mem::take(&mut self.held);
533        self.notice(&held);
534        let rules = self.rules.current();
535        let cause = self.cause_of(&held, ours, &rules);
536        let (fighting, looted, likely) = self.kinds_of(&held, &cause, (fighting, looted), &rules);
537        let set_up = hostile
538            && cause
539                .command()
540                .is_some_and(|(text, _)| rules.rules().sets_up(text));
541        let fought = fighting;
542        let fighting = fighting || set_up;
543        // The command it answered, sorted with it.
544        if let Cause::Command { text, feed } = &cause
545            && let Some(sorted) = rules.rules().command(&crate::stream_rules::Command {
546                text,
547                by_script: super::sender(feed) == Some(crate::Origin::Lich.sender()),
548                fighting: fought,
549                set_up,
550                looted,
551            })
552        {
553            self.sorted.push(SortedLine {
554                stream: feed.clone(),
555                text: text.clone(),
556                to: sorted.to.to_owned(),
557                copy: sorted.copy,
558                kept: false,
559            });
560        }
561        self.cause = Some(cause);
562        // Each line sorted, written or not: the windows are told. A room
563        // told in a fight's chunk -- a fear that made the character flee --
564        // is copied wherever its lines go, never moved: the story keeps
565        // where the character is.
566        let mut kept = Vec::with_capacity(held.len());
567        let mut in_room = false;
568        for line in held {
569            if line.feed == MAIN && line.room.is_some() {
570                in_room = true;
571            }
572            let room_line = in_room && line.feed == MAIN;
573            if room_line && ends_a_room(&line.text) {
574                in_room = false;
575            }
576            // A command is sorted by who sent it, a readout is no line.
577            let command = line.feed.starts_with(COMMANDS);
578            let sortable = !command && !READOUTS.contains(&line.feed.as_str());
579            let marked = || {
580                let text = line.text.trim();
581                marks.iter().any(|mark| mark.text.trim() == text)
582            };
583            // A command is sorted as its answer's, above.
584            let sorted = sortable
585                .then(|| {
586                    rules.sort(&Line {
587                        stream: &line.feed,
588                        text: &line.text,
589                        links: &line.links,
590                        fighting,
591                        looted,
592                        marked: marked(),
593                    })
594                })
595                .flatten()
596                .filter(|sorted| sorted.to != line.feed);
597            let stays = likely || (room_line && fighting);
598            if let Some(sorted) = sorted {
599                self.sorted.push(SortedLine {
600                    stream: line.feed.clone(),
601                    text: line.text.clone(),
602                    to: sorted.to.to_owned(),
603                    copy: sorted.copy || stays,
604                    kept: stays,
605                });
606            }
607            if self.log.is_some() && self.capture.wants(&line.feed) {
608                let to = sorted.map(|sorted| sorted.to.to_owned());
609                kept.push((line, to));
610            }
611        }
612        let Some(log) = self.log.clone() else {
613            return;
614        };
615        let held: Vec<&Held> = kept.iter().map(|(line, _)| line).collect();
616        let unchanged = self.unchanged_readouts(&held);
617        for (line, to) in kept {
618            if unchanged.contains(&line.feed) {
619                continue;
620            }
621            let stream = match to {
622                Some(to) => format!("{}/{to}", line.feed),
623                None => line.feed,
624            };
625            log.record(LogLine {
626                at: line.at,
627                stream,
628                text: line.text,
629                links: line.links,
630            });
631        }
632    }

634    /// What the chunk of `held`, which answered `cause`, was, once the
635    /// chunk kinds have their word ([`crate::chunk_kinds`]): whether it had
636    /// fighting in it, whether a creature was searched or skinned in it, and
637    /// whether its lines are only likely so, and so copied rather than
638    /// moved. A chunk the readers know -- a fight, a search or a skin, a
639    /// room, the answer to a command that goes with neither -- is counted
640    /// as what they know it to be; one they do not is judged by its
641    /// sentences.
642    fn kinds_of(
643        &self,
644        held: &[Held],
645        cause: &Cause,
646        (fighting, looted): (bool, bool),
647        rules: &Compiled,
648    ) -> (bool, bool, bool) {
649        let unchanged = (fighting, looted, false);
650        let Some(table) = &self.kinds else {
651            return unchanged;
652        };
653        let sentences: Vec<String> = held
654            .iter()
655            .filter(|line| line.feed == MAIN)
656            .map(|line| chunk_kinds::sentence(&line.text, &objects(&line.links)))
657            .filter(|sentence| !sentence.is_empty())
658            .collect();
659        if sentences.is_empty() {
660            return unchanged;
661        }
662        let room = held
663            .iter()
664            .any(|line| line.feed == MAIN && line.room.is_some());
665        let known = if fighting {
666            Some(Kind::Fight)
667        } else if looted {
668            Some(Kind::Loot)
669        } else if room
670            || cause
671                .command()
672                .is_some_and(|(text, _)| !rules.rules().goes_with_chunks(text))
673        {
674            Some(Kind::Other)
675        } else {
676            None
677        };
678        let mut table = chunk_kinds::lock(table);
679        if let Some(kind) = known {
680            table.learn(&sentences, kind);
681            table.save_now_and_then();
682            return unchanged;
683        }
684        match table.judge(&sentences) {
685            Verdict::Move(Kind::Fight) => (true, looted, false),
686            Verdict::Copy(Kind::Fight) => (true, looted, true),
687            Verdict::Move(Kind::Loot) => (fighting, true, false),
688            Verdict::Copy(Kind::Loot) => (fighting, true, true),
689            Verdict::Move(Kind::Other) | Verdict::Copy(Kind::Other) | Verdict::Unknown => unchanged,
690        }
691    }
692
693    /// Each of `held` the lookout finds new, kept for the windows; a
694    /// command and a readout are no line of the game's to it.
695    fn notice(&mut self, held: &[Held]) {
696        let Some(watched) = &self.lookout else {
697            return;
698        };
699        let now = hydra_lookout::now_seconds();
700        let mut lookout = hydra_lookout::lock(&watched.lookout);
701        let quiet_for = std::time::Duration::from_secs(u64::from(lookout.settings().quiet_secs));
702        let quiet = self.ready.is_none_or(|at| at.elapsed() < quiet_for);
703        for line in held {
704            if line.feed.starts_with(COMMANDS) || READOUTS.contains(&line.feed.as_str()) {
705                continue;
706            }
707            let objects = objects(&line.links);
708            let verdict = lookout.judge(
709                &watched.who,
710                &hydra_lookout::Line {
711                    stream: &line.feed,
712                    text: &line.text,
713                    objects: &objects,
714                    room: line.room == Some(RoomPart::Desc),
715                },
716                now,
717                quiet,
718            );
719            if verdict.shown() {
720                self.noticed.push(Noticed {
721                    stream: line.feed.clone(),
722                    text: line.text.clone(),
723                    stat: verdict == hydra_lookout::Verdict::Stat,
724                });
725            }
726        }
727        lookout.save_now_and_then();
728    }
729
730    /// The readouts in this chunk that say exactly what they said last time.
731    /// A readout arrives whole, with no prompt inside it, so a chunk's lines
732    /// for one readout ARE its snapshot. Remembers the new snapshot for the
733    /// ones that did change.
734    fn unchanged_readouts(&mut self, held: &[&Held]) -> BTreeSet<String> {
735        let mut snapshots: BTreeMap<&str, Vec<String>> = BTreeMap::new();
736        for line in held {
737            if READOUTS.contains(&line.feed.as_str()) {
738                snapshots
739                    .entry(&line.feed)
740                    .or_default()
741                    .push(line.text.clone());
742            }
743        }
744        let mut unchanged = BTreeSet::new();
745        for (feed, snapshot) in snapshots {
746            if self.written.get(feed) == Some(&snapshot) {
747                unchanged.insert(feed.to_owned());
748            } else {
749                self.written.insert(feed.to_owned(), snapshot);
750            }
751        }
752        unchanged
753    }
754}
755
756/// Whether `text`, a line of the game's, speaks to the character, as an
757/// answer to its command does: `You cannot see that.`, `...wait 2
758/// seconds.`, `What were you referring to?`.
759fn to_the_character(text: &str) -> bool {
760    let text = text.trim_start();
761    [
762        "You",
763        "Your",
764        "...wait",
765        "Wait ",
766        "What were you",
767        "I could not",
768    ]
769    .iter()
770    .any(|start| text.starts_with(start))
771}
772
773/// The game objects `links` name, where they sit in their line.
774fn objects(links: &[LogLink]) -> Vec<hydra_lookout::Object> {
775    links
776        .iter()
777        .filter_map(|link| match &link.does {
778            Does::Object { noun, .. } => Some(hydra_lookout::Object {
779                at: link.from..link.to,
780                noun: noun.clone(),
781            }),
782            _ => None,
783        })
784        .collect()
785}

787impl Drop for Feed {
788    /// A connection that ends mid-chunk still logs what it saw: the last lines
789    /// before a drop are the ones a reader goes looking for.
790    fn drop(&mut self) {
791        self.close(Chunk::default(), &[]);
792    }
793}

1//! One connection's feed into the player log: what is captured, and what a
2//! line is tagged once its chunk is understood.
3//!
4//! A line is tagged where the game sent it, and after that where the stream
5//! rules put it ([`crate::stream_rules`]): the built-in rule puts each main
6//! line of a chunk with fighting in it in `combat`, so the combat feed reads
7//! apart from the rest.
8//!
9//! # A line's class is not known when the line ends
10//!
11//! The blob is the model's prompt-to-prompt chunk, and the combat tracker
12//! classifies it when the prompt closes it. So the feed **holds a chunk's
13//! lines and writes them at the prompt**, when the actor can say whether the
14//! chunk had fighting in it and which lines the reader marked. Everything is
15//! held, not only the main window's lines, so the file stays in the order
16//! things happened; each line keeps the stamp of the moment it arrived.
17//!
18//! # The tag: `source/class`
19//!
20//! A line from a combat chunk is tagged `main/combat`. **The class never
21//! replaces the source**: `main` is where the game sent it, `combat` is what
22//! the rules made of it. A line no rule matches stays `main`. A command, a
23//! readout, is never sorted.
24//!
25//! # Every feed a player reads is written
26//!
27//! [`Capture`] decides what is WRITTEN; what is SHOWN is the reader's
28//! choice alone. A player has no choice of what is written: a feed left out
29//! is a hole no reader can fill, and the feeds cost little beside the story.
30//! The readouts -- streams the game clears and rewrites whole -- are left
31//! out, the connection's recording keeping them; a capture that asks for
32//! one writes it **only when it changes**.
33

51/// A chunk is written unclassified once it holds this many lines: the login
52/// burst and a long `look` in a crowded room run far past anything a combat
53/// chunk does, and a connection that never prompts must not hold its log in

57/// Which feeds are written to disk.
58///
59/// A set of what is switched OFF, so that a stream nobody has heard of is
60/// captured: a reader can filter noise and cannot fill a hole.
61#[derive(Clone, Debug, PartialEq, Eq)]
62pub struct Capture {
63    off: BTreeSet<String>,
64}
65
66impl Default for Capture {
67    /// Text feeds on; the readouts ([`READOUTS`]) off.
68    fn default() -> Self {
69        Self {
70            off: READOUTS.iter().map(|s| (*s).to_owned()).collect(),
71        }
72    }
73}

75impl Capture {
76    /// Every feed, readouts included.
77    #[must_use]
78    pub const fn everything() -> Self {
79        Self {
80            off: BTreeSet::new(),
81        }
82    }

84    /// Switch one feed on or off, by the tag it is logged under (`main`,
85    /// `cmd`, `inv`, `thoughts`...).
86    #[must_use]
87    pub fn set(mut self, feed: &str, on: bool) -> Self {
88        if on {
89            self.off.remove(feed);
90        } else {
91            self.off.insert(feed.to_owned());
92        }
93        self
94    }
95
96    /// Whether a feed is written.
97    #[must_use]
98    pub fn wants(&self, feed: &str) -> bool {
99        // A command under its sender is the commands' feed.
100        let feed = if super::sender(feed).is_some() {
101            COMMANDS
102        } else {
103            feed
104        };
105        !self.off.contains(feed)
106    }
107}

126/// The tag a stream's lines are logged under: the main window's id on the
127/// wire is the empty string, which is no use to someone grepping a file.

162/// One connection's feed: owned by the actor, which hands it each frame,
163/// each command sent, and each prompt's verdict. It sorts each chunk's lines
164/// by the stream rules whether or not it writes them, for the windows.
165#[derive(Debug)]
166pub struct Feed {
167    /// Where its lines are written; `None`, nowhere.
168    log: Option<PlayerLog>,
169    capture: Capture,
170    /// What sorts its lines.
171    rules: StreamRules,
172    /// The lines the rules sorted since last taken.
173    sorted: Vec<SortedLine>,
174    /// Each stream's line under way, as text, its links, and whether it
175    /// began as a room's description.
176    unfinished: BTreeMap<String, (String, Vec<LogLink>, bool)>,
177    held: Vec<Held>,
178    /// What each readout last wrote, so an unchanged rewrite is not written.
179    written: BTreeMap<String, Vec<String>>,
180    /// The lookout its lines are shown to, once the game has said whose.
181    lookout: Option<Watched>,
182    /// When the character became ready to play: until then, and for the
183    /// lookout's quiet minute after, its lines are learned and not shown.
184    ready: Option<std::time::Instant>,
185    /// The lines the lookout found new since last taken.
186    noticed: Vec<Noticed>,
187}

256    /// One frame the game sent: its text gathered into lines, each held
257    /// once it ends.
258    pub fn frame(&mut self, frame: &Frame) {
259        match frame {
260            Frame::Text(text) => {
261                let (line, links, room) = self.unfinished.entry(text.stream.clone()).or_default();
262                if line.is_empty() && text.style.preset.as_deref() == Some(ROOM_DESC) {
263                    *room = true;
264                }
265                let from = line.len();
266                line.push_str(&text.content);
267                if let Some(does) = text.link.as_ref().and_then(Does::of) {
268                    links::add(links, (from, line.len()), does);
269                }
270                if text.ends_line {
271                    let (whole, links) = (std::mem::take(line), std::mem::take(links));
272                    let room = std::mem::take(room);
273                    let mut at = 0;
274                    for (index, piece) in whole.split('\n').enumerate() {
275                        let piece_links = links::within(&links, (at, at + piece.len()));
276                        self.linked(&text.stream, piece, piece_links, room && index == 0);
277                        at += piece.len() + 1;
278                    }
279                }
280            }
281            Frame::ClearStream { id, .. } => {
282                self.unfinished.remove(id);
283            }
284            _ => {}
285        }
286    }

288    /// One completed line of game text, from the stream the wire named.

293    /// [`Self::line`], with its links, and whether it began as a room's

301    /// One command, as it went to the game, under who sent it.
302    pub fn command(&mut self, line: &str, origin: crate::Origin) {
303        let text = line.trim_end_matches(['\r', '\n']);
304        if !text.is_empty() {
305            self.hold(&super::command_tag(origin), text, Vec::new(), false);
306        }

309    fn hold(&mut self, feed: &str, text: &str, links: Vec<LogLink>, room: bool) {
310        self.held.push(Held {
311            at: hydra_net::line_time(),
312            feed: feed.to_owned(),
313            text: text.to_owned(),
314            links,
315            room,
316        });
317        if self.held.len() >= MAX_HELD {
318            self.close(false, &[]);
319        }
320    }

322    /// The prompt closed the chunk: write what was held, each line sorted by
323    /// the stream rules. `fighting` is whether the chunk had fighting in it
324    /// ([`crate::stream_rules::fought`]), and `marks` the lines the combat
325    /// reader marked, by their text.
326    pub fn close(&mut self, fighting: bool, marks: &[LineMark]) {
327        let held = std::mem::take(&mut self.held);
328        self.notice(&held);
329        let rules = self.rules.current();
330        // Each line sorted, written or not: the windows are told.
331        let mut kept = Vec::with_capacity(held.len());
332        for line in held {
333            // A command is sorted by who sent it, a readout is no line.
334            let sortable =
335                !line.feed.starts_with(COMMANDS) && !READOUTS.contains(&line.feed.as_str());
336            let marked = || {
337                let text = line.text.trim();
338                marks.iter().any(|mark| mark.text.trim() == text)
339            };
340            let sorted = sortable
341                .then(|| {
342                    rules.sort(&Line {
343                        stream: &line.feed,
344                        text: &line.text,
345                        links: &line.links,
346                        fighting,
347                        marked: marked(),
348                    })
349                })
350                .flatten()
351                .filter(|sorted| sorted.to != line.feed);
352            if let Some(sorted) = sorted {
353                self.sorted.push(SortedLine {
354                    stream: line.feed.clone(),
355                    text: line.text.clone(),
356                    to: sorted.to.to_owned(),
357                    copy: sorted.copy,
358                });
359            }
360            if self.log.is_some() && self.capture.wants(&line.feed) {
361                kept.push((line, sorted.map(|sorted| sorted.to.to_owned())));
362            }
363        }
364        let Some(log) = self.log.clone() else {
365            return;
366        };
367        let held: Vec<&Held> = kept.iter().map(|(line, _)| line).collect();
368        let unchanged = self.unchanged_readouts(&held);
369        for (line, to) in kept {
370            if unchanged.contains(&line.feed) {
371                continue;
372            }
373            let stream = match to {
374                Some(to) => format!("{}/{to}", line.feed),
375                None => line.feed,
376            };
377            log.record(LogLine {
378                at: line.at,
379                stream,
380                text: line.text,
381                links: line.links,
382            });
383        }
384    }

386    /// Each of `held` the lookout finds new, kept for the windows; a
387    /// command and a readout are no line of the game's to it.
388    fn notice(&mut self, held: &[Held]) {
389        let Some(watched) = &self.lookout else {
390            return;
391        };
392        let now = hydra_lookout::now_seconds();
393        let mut lookout = hydra_lookout::lock(&watched.lookout);
394        let quiet_for = std::time::Duration::from_secs(u64::from(lookout.settings().quiet_secs));
395        let quiet = self.ready.is_none_or(|at| at.elapsed() < quiet_for);
396        for line in held {
397            if line.feed.starts_with(COMMANDS) || READOUTS.contains(&line.feed.as_str()) {
398                continue;
399            }
400            let objects: Vec<hydra_lookout::Object> = line
401                .links
402                .iter()
403                .filter_map(|link| match &link.does {
404                    Does::Object { noun, .. } => Some(hydra_lookout::Object {
405                        at: link.from..link.to,
406                        noun: noun.clone(),
407                    }),
408                    _ => None,
409                })
410                .collect();
411            let verdict = lookout.judge(
412                &watched.who,
413                &hydra_lookout::Line {
414                    stream: &line.feed,
415                    text: &line.text,
416                    objects: &objects,
417                    room: line.room,
418                },
419                now,
420                quiet,
421            );
422            if verdict.shown() {
423                self.noticed.push(Noticed {
424                    stream: line.feed.clone(),
425                    text: line.text.clone(),
426                    stat: verdict == hydra_lookout::Verdict::Stat,
427                });
428            }
429        }
430        lookout.save_now_and_then();
431    }

433    /// The readouts in this chunk that say exactly what they said last time.
434    /// A readout arrives whole, with no prompt inside it, so a chunk's lines
435    /// for one readout ARE its snapshot. Remembers the new snapshot for the
436    /// ones that did change.
437    fn unchanged_readouts(&mut self, held: &[&Held]) -> BTreeSet<String> {
438        let mut snapshots: BTreeMap<&str, Vec<String>> = BTreeMap::new();
439        for line in held {
440            if READOUTS.contains(&line.feed.as_str()) {
441                snapshots
442                    .entry(&line.feed)
443                    .or_default()
444                    .push(line.text.clone());
445            }
446        }
447        let mut unchanged = BTreeSet::new();
448        for (feed, snapshot) in snapshots {
449            if self.written.get(feed) == Some(&snapshot) {
450                unchanged.insert(feed.to_owned());
451            } else {
452                self.written.insert(feed.to_owned(), snapshot);
453            }
454        }
455        unchanged
456    }

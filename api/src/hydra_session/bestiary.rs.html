1//! The creature codex, read for the player: what `.bestiary` answers,
2//! and what the Bestiary widget shows.
3//!
4//! The codex is the model's bestiary (`hydra_model::state::creature`):
5//! 627 creature templates with where each is found, how it attacks and
6//! defends, what it drops, and what players have written about it. Nothing
7//! here watches the game: the answers are the same for everyone.
8//!
9//! `<name>` is one creature's entry: its name as typed, else the one
10//! creature of that noun, else the one whose name holds the words;
11//! several, and they are listed to choose from. `here` lists what spawns
12//! around the character's room, `level`, `area`, `family`, `undead` and
13//! `search` the rest. Every list and entry is a linked table
14//! ([`crate::notice::Span`]): a name opens its entry, a level its level, a
15//! family its family, a ground's `[go2]` walks there, `[map]` tints its
16//! grounds on the Map Explorer, and the wiki a page.
17

25/// What `.bestiary help` says.
26pub const HELP: &[&str] = &[
27    "bestiary <name>          one creature's entry: its name, its noun, or words of its name",
28    "bestiary here            what spawns around this room",
29    "bestiary level <n|a-b>   every creature of that level, or in that band",
30    "bestiary area <name>     every creature of a ground whose name holds the words",
31    "bestiary family <name>   every creature of a family",
32    "bestiary undead          every undead creature",
33    "bestiary search <text>   every creature whose name holds the text",
34];
35
36/// What a `.bestiary` line asks.
37#[derive(Clone, Debug, PartialEq, Eq)]
38pub enum Asked {
39    /// `bestiary help`: how the command is used.
40    Help,
41    /// `bestiary <name>`: one creature's entry, by its name, its noun, or
42    /// words of its name.
43    Entry(String),
44    /// `bestiary here`: what spawns around this room.
45    Here,
46    /// `bestiary level <n|a-b>`: every creature of a level, or in a band
47    /// from the first level to the second.
48    Level(i32, i32),
49    /// `bestiary area <name>`: every creature of a ground whose name holds
50    /// the words.
51    Area(String),
52    /// `bestiary family <name>`: every creature of a family.
53    Family(String),
54    /// `bestiary undead`: every undead creature.
55    Undead,
56    /// `bestiary search <text>`: every creature whose name holds the text.
57    Search(String),
58}
59
60/// What `rest`, the line after `bestiary`, asks.
61///
62/// # Errors
63///
64/// A level that is no number, or a list asked for with nothing to list by.
65pub fn asked(rest: &str) -> Result<Asked, String> {
66    let rest = rest.trim();
67    let (word, after) = rest.split_once(char::is_whitespace).unwrap_or((rest, ""));
68    let after = after.trim();
69    Ok(match word.to_ascii_lowercase().as_str() {
70        "" | "help" => Asked::Help,
71        "here" => Asked::Here,
72        "undead" => Asked::Undead,
73        "level" => {
74            let (low, high) = after.split_once('-').unwrap_or((after, after));
75            let level = |text: &str| {
76                text.trim()
77                    .parse::<i32>()
78                    .map_err(|_| format!("`{after}` is not a level: 29, or a band, 25-33."))
79            };
80            Asked::Level(level(low)?, level(high)?)
81        }
82        "area" if !after.is_empty() => Asked::Area(after.to_owned()),
83        "family" if !after.is_empty() => Asked::Family(after.to_owned()),
84        "search" if !after.is_empty() => Asked::Search(after.to_owned()),
85        "area" | "family" | "search" => {
86            return Err(format!("{word} what? .bestiary {word} <words>"));
87        }
88        _ => Asked::Entry(rest.to_owned()),
89    })
90}

92/// The creatures `name` means: the ones named so, else the ones of that
93/// noun, else those whose name holds the words.
94fn named(name: &str) -> Vec<&'static Creature> {
95    let exact = codex::by_name(name);
96    if !exact.is_empty() {
97        return exact;
98    }
99    let by_noun = codex::by_noun(name);
100    if !by_noun.is_empty() {
101        return by_noun;
102    }
103    searched(name)
104}

106/// Every creature whose name holds `text`, in level order.
107fn searched(text: &str) -> Vec<&'static Creature> {
108    let wanted = text.trim().to_lowercase();
109    let mut found: Vec<&Creature> = codex::creatures()
110        .filter(|c| c.name.to_lowercase().contains(&wanted))
111        .collect();
112    found.sort_by_key(|c| (c.level.unwrap_or(i32::MAX), c.name.clone()));
113    found
114}

116/// `creatures` as a table: level, name, family, each a link.
117fn listed(heading: &str, mut creatures: Vec<&'static Creature>) -> Notice {
118    creatures.sort_by_key(|c| (c.level.unwrap_or(i32::MAX), c.name.clone()));
119    let mut lines = vec![vec![Span::plain(heading.to_owned())]];
120    if creatures.is_empty() {
121        lines.push(vec![Span::plain("  nothing.")]);
122    }
123    let width = creatures.iter().map(|c| c.name.len()).max().unwrap_or(0);
124    for creature in creatures {
125        let shown = creature
126            .level
127            .map_or_else(|| "  ?".to_owned(), |level| format!("{level:>3}"));
128        let mut line = vec![
129            match creature.level {
130                Some(level) => Span::hydra(shown, format!("bestiary level {level}")),
131                None => Span::plain(shown),
132            },
133            Span::plain("  "),
134            Span::hydra(
135                format!("{:<width$}", creature.name),
136                format!("bestiary {}", creature.name),
137            ),
138        ];
139        if let Some(family) = &creature.family {
140            line.push(Span::plain("  "));
141            line.push(Span::hydra(
142                family.clone(),
143                format!("bestiary family {family}"),
144            ));
145        }
146        if creature.undead == Some(true) {
147            line.push(Span::plain("  undead"));
148        }
149        lines.push(line);
150    }
151    Notice::linked(NoticeKind::Info, lines)
152}

154/// A stat as the entry writes it: exact, or its span.
155fn stat(stat: Stat) -> String {
156    match stat {
157        Stat::Exact(value) => value.to_string(),
158        Stat::Range(low, high) => format!("{low}-{high}"),
159    }
160}

162/// Which circle a target defense key is: `wiz_td` is the Wizard's.
163fn circle(key: &str) -> &str {
164    match key {
165        "bar_td" => "Bard",
166        "cle_td" => "Cleric",
167        "emp_td" => "Empath",
168        "pal_td" => "Paladin",
169        "ran_td" => "Ranger",
170        "sor_td" => "Sorcerer",
171        "wiz_td" => "Wizard",
172        "mje_td" => "Major Elemental",
173        "mne_td" => "Minor Elemental",
174        "mjs_td" => "Major Spirit",
175        "mns_td" => "Minor Spirit",
176        "mnm_td" => "Minor Mental",
177        other => other,
178    }
179}

181/// `text` wrapped to lines no wider than `width`, each indented two.
182fn wrapped(text: &str, width: usize) -> Vec<Vec<Span>> {
183    let mut lines = Vec::new();
184    let mut line = String::new();
185    for word in text.split_whitespace() {
186        if !line.is_empty() && line.len() + 1 + word.len() > width {
187            lines.push(vec![Span::plain(format!("  {line}"))]);
188            line.clear();
189        }
190        if !line.is_empty() {
191            line.push(' ');
192        }
193        line.push_str(word);
194    }
195    if !line.is_empty() {
196        lines.push(vec![Span::plain(format!("  {line}"))]);
197    }
198    lines
199}

201/// One creature's entry, `profession` the character's, whose tips come
202/// first.
203fn entry(creature: &Creature, profession: Option<&str>) -> Notice {
204    let mut lines: Vec<Vec<Span>> = Vec::new();
205    let mut head = vec![Span::plain(creature.name.to_uppercase())];
206    if let Some(level) = creature.level {
207        head.push(Span::plain("  level "));
208        head.push(Span::hydra(
209            level.to_string(),
210            format!("bestiary level {level}"),
211        ));
212    }
213    if !creature.areas.is_empty() {
214        head.push(Span::plain("  "));
215        head.push(Span::hydra("[map]", format!("map {}", creature.name)));
216    }
217    lines.push(head);
218    if let Some(description) = creature.description() {
219        lines.extend(wrapped(description, 70));
220    }
221    lines.push(identity(creature));
222    lines.extend(grounds(creature));
223    lines.extend(offense(creature));
224    lines.extend(defense(creature));
225    lines.extend(treasure(creature));
226    for note in &creature.special_notes {
227        lines.extend(wrapped(note, 70));
228    }
229    lines.extend(tips(creature, profession));
230    if let Some(url) = &creature.url {
231        lines.push(vec![
232            Span::plain("Wiki: "),
233            Span::url(url.clone(), url.clone()),
234        ]);
235    }
236    Notice::linked(NoticeKind::Info, lines)
237}

239/// The line that says what it is: family, type, undead, hit points.
240fn identity(creature: &Creature) -> Vec<Span> {
241    let mut line = Vec::new();
242    if let Some(family) = &creature.family {
243        line.push(Span::plain("Family: "));
244        line.push(Span::hydra(
245            family.clone(),
246            format!("bestiary family {family}"),
247        ));
248    }
249    if let Some(kind) = &creature.kind {
250        line.push(Span::plain(format!("  Type: {kind}")));
251    }
252    if creature.undead == Some(true) {
253        line.push(Span::plain("  Undead"));
254    }
255    if let Some(hp) = creature.max_hp {
256        line.push(Span::plain(format!("  HP: {}", stat(hp))));
257    }
258    if creature.boss {
259        line.push(Span::plain(format!(
260            "  Boss{}",
261            creature
262                .boss_type
263                .as_deref()
264                .map_or_else(String::new, |kind| format!(" ({kind})"))
265        )));
266    }
267    if line.is_empty() {
268        line.push(Span::plain(String::new()));
269    }
270    line
271}

273/// Where it is found: each ground, how many rooms, and a `[go2]` to its
274/// first.
275fn grounds(creature: &Creature) -> Vec<Vec<Span>> {
276    if creature.areas.is_empty() {
277        return Vec::new();
278    }
279    let mut lines = vec![vec![Span::plain("Found in:")]];
280    let mut by_name: BTreeMap<&str, Vec<&Area>> = BTreeMap::new();
281    for area in &creature.areas {
282        by_name.entry(area.name.as_str()).or_default().push(area);
283    }
284    for (name, spans) in by_name {
285        let rooms: u32 = spans
286            .iter()
287            .map(|span| span.uid_high.saturating_sub(span.uid_low) + 1)
288            .sum();
289        let first = spans.iter().map(|span| span.uid_low).min().unwrap_or(0);
290        lines.push(vec![
291            Span::plain("  "),
292            Span::hydra(name.to_owned(), format!("bestiary area {name}")),
293            Span::plain(format!("  {rooms} rooms  ")),
294            Span::hydra("[go2]", format!("go2 u{first}")),
295        ]);
296    }
297    lines
298}

300/// How it attacks: each category that has anything, a line an attack.
301fn offense(creature: &Creature) -> Vec<Vec<Span>> {
302    let mut lines = Vec::new();
303    for category in AttackCategory::ALL {
304        let attacks: Vec<_> = creature
305            .attacks
306            .iter()
307            .filter(|attack| attack.category == category)
308            .collect();
309        if attacks.is_empty() {
310            continue;
311        }
312        lines.push(vec![Span::plain(format!("{}:", category.as_str()))]);
313        for attack in attacks {
314            let mut said = format!("  {}", attack.name);
315            if let Some(strength) = attack.attack_strength {
316                let _ = write!(said, "  AS {}", stat(strength));
317            }
318            if let Some(strength) = attack.casting_strength {
319                let _ = write!(said, "  CS {}", stat(strength));
320            }
321            if let Some(note) = &attack.note {
322                let _ = write!(said, "  {note}");
323            }
324            lines.push(vec![Span::plain(said)]);
325        }
326    }
327    for ability in &creature.abilities {
328        let mut said = format!("  {}", ability.name);
329        if let Some(notes) = &ability.notes {
330            let _ = write!(said, "  {notes}");
331        }
332        if lines.is_empty() || lines.last().is_none_or(|l| l[0].text != "Abilities:") {
333            lines.push(vec![Span::plain("Abilities:")]);
334        }
335        lines.push(vec![Span::plain(said)]);
336    }
337    lines
338}

340/// How it defends: armour, DS by kind, UDF, TD by circle, immunities,
341/// its spells and abilities.
342fn defense(creature: &Creature) -> Vec<Vec<Span>> {
343    let mut parts = Vec::new();
344    if let Some(asg) = &creature.asg {
345        parts.push(format!("ASG {asg}"));
346    }
347    for (name, value) in [
348        ("Melee DS", creature.melee_ds),
349        ("Ranged DS", creature.ranged_ds),
350        ("Bolt DS", creature.bolt_ds),
351        ("UDF", creature.udf),
352    ] {
353        if let Some(value) = value {
354            parts.push(format!("{name} {}", stat(value)));
355        }
356    }
357    let mut lines = Vec::new();
358    if !parts.is_empty() {
359        lines.push(vec![Span::plain(format!("Defense: {}", parts.join("  ")))]);
360    }
361    if !creature.target_defense.is_empty() {
362        let td: Vec<String> = creature
363            .target_defense
364            .iter()
365            .map(|(key, value)| format!("{} {}", circle(key), stat(*value)))
366            .collect();
367        lines.extend(wrapped(&format!("TD: {}", td.join(", ")), 70));
368    }
369    for (name, list) in [
370        ("Immune to", &creature.immunities),
371        ("Defensive spells", &creature.defensive_spells),
372        ("Special defenses", &creature.special_defenses),
373    ] {
374        if !list.is_empty() {
375            lines.extend(wrapped(&format!("{name}: {}", list.join(", ")), 70));
376        }
377    }
378    if !creature.defensive_abilities.is_empty() {
379        let said: Vec<String> = creature
380            .defensive_abilities
381            .iter()
382            .map(|entry| match &entry.note {
383                Some(note) => format!("{} ({note})", entry.name),
384                None => entry.name.clone(),
385            })
386            .collect();
387        lines.extend(wrapped(
388            &format!("Defensive abilities: {}", said.join(", ")),
389            70,
390        ));
391    }
392    lines
393}

395/// What it drops.
396fn treasure(creature: &Creature) -> Vec<Vec<Span>> {
397    let treasure = &creature.treasure;
398    if treasure.is_empty() {
399        return Vec::new();
400    }
401    let mut parts = Vec::new();
402    for (name, has) in [
403        ("coins", treasure.coins),
404        ("gems", treasure.gems),
405        ("boxes", treasure.boxes),
406        ("magic items", treasure.magic_items),
407    ] {
408        if has == Some(true) {
409            parts.push(name.to_owned());
410        }
411    }
412    if let Some(skin) = &treasure.skin {
413        parts.push(format!("skin: {skin}"));
414    }
415    if treasure.blunt_required == Some(true) {
416        parts.push("blunt required".to_owned());
417    }
418    parts.extend(treasure.armaments.iter().cloned());
419    parts.extend(treasure.other.iter().cloned());
420    if parts.is_empty() {
421        return Vec::new();
422    }
423    wrapped(&format!("Treasure: {}", parts.join(", ")), 70)
424}

426/// The tips written about it: the character's profession's first, then
427/// the general ones, then the rest.
428fn tips(creature: &Creature, profession: Option<&str>) -> Vec<Vec<Span>> {
429    let mut lines = Vec::new();
430    let own = profession.map(str::to_lowercase);
431    let mut sections: Vec<&String> = creature.info.keys().collect();
432    sections.sort_by_key(|section| {
433        if Some(section.as_str()) == own.as_deref() {
434            0
435        } else if *section == "general" {
436            1
437        } else {
438            2
439        }
440    });
441    for section in sections {
442        for tip in creature.tips(section) {
443            lines.extend(wrapped(&format!("{section}: {tip}"), 70));
444        }
445    }
446    lines
447}

455/// What `.bestiary` answers to `asked`, for a character in room `uid` of
456/// profession `profession`.
457#[must_use]
458pub fn answered(asked: Asked, uid: Option<u32>, profession: Option<&str>) -> Notice {
459    match asked {
460        Asked::Help => Notice::help_table(NoticeKind::Info, HELP),
461        Asked::Entry(name) => match named(&name).as_slice() {
462            [] => Notice::line(
463                NoticeKind::Warn,
464                format!("The bestiary has no creature like {name}."),
465            ),
466            [one] => entry(one, profession),
467            several => listed(&format!("Creatures like {name}:"), several.to_vec()),
468        },
469        Asked::Here => match uid {
470            Some(uid) => listed(&format!("Found around room {uid}:"), codex::in_room(uid)),
471            None => Notice::line(
472                NoticeKind::Warn,
473                "The game has not said which room this is yet.",
474            ),
475        },
476        Asked::Level(low, high) => {
477            let (low, high) = (low.min(high), low.max(high));
478            let heading = if low == high {
479                format!("Level {low}:")
480            } else {
481                format!("Levels {low} to {high}:")
482            };
483            listed(
484                &heading,
485                codex::creatures()
486                    .filter(|c| c.level.is_some_and(|level| (low..=high).contains(&level)))
487                    .collect(),
488            )
489        }
490        Asked::Area(name) => {
491            let wanted = name.to_lowercase();
492            listed(
493                &format!("Found in {name}:"),
494                codex::creatures()
495                    .filter(|c| {
496                        c.areas
497                            .iter()
498                            .any(|area| area.name.to_lowercase().contains(&wanted))
499                    })
500                    .collect(),
501            )
502        }
503        Asked::Family(name) => {
504            let wanted = name.to_lowercase();
505            listed(
506                &format!("The {name} family:"),
507                codex::creatures()
508                    .filter(|c| {
509                        c.family
510                            .as_deref()
511                            .is_some_and(|family| family.to_lowercase() == wanted)
512                    })
513                    .collect(),
514            )
515        }
516        Asked::Undead => listed(
517            "The undead:",
518            codex::creatures()
519                .filter(|c| c.undead == Some(true))
520                .collect(),
521        ),
522        Asked::Search(text) => listed(&format!("Creatures like {text}:"), searched(&text)),
523    }
524}

530    /// The words after `bestiary` are read as what they ask.
531    #[test]
532    fn the_line_is_read() {

534        assert_eq!(asked("here"), Ok(Asked::Here));
535        assert_eq!(asked("level 29"), Ok(Asked::Level(29, 29)));
536        assert_eq!(asked("level 25-33"), Ok(Asked::Level(25, 33)));
537        assert!(asked("level high").is_err());
538        assert_eq!(

540            Ok(Asked::Area("the graveyard".into()))

542        assert!(asked("area").is_err());
543        assert_eq!(asked("Agresh bear"), Ok(Asked::Entry("Agresh bear".into())));

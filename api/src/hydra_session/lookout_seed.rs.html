1//! The lookout taught from the player logs: every line each character of a
2//! game read, learned as if it had come live, so the lookout starts out
3//! knowing the routine (`hydra lookout seed`).
4//!
5//! Each game's characters are read a day at a time, every character's
6//! lines of the day in the order they came, so a line a group heard at once
7//! is one occasion, as it is live. Nothing is shown. A log keeps no word of
8//! which line was a room's description, so the line after a room's title is
9//! taken to be one. The last day read is kept in `lookout.toml`, and a
10//! seeding again starts after it: no day is learned twice. Today is left
11//! for the next seeding, since it is still being played.
12
13use std::collections::BTreeSet;
14use std::path::Path;

16use crate::player_log::{LogLink, links::Does, reader, writer};

18/// What a seeding of one game read.
19#[derive(Clone, Debug, Default, PartialEq, Eq)]
20pub struct Seeded {
21    /// The game, as its lookout is named: `Prime`.
22    pub game: String,
23    /// The characters read.
24    pub characters: Vec<String>,
25    /// The days read, the first and the last.
26    pub days: Option<(String, String)>,
27    /// The lines read.
28    pub lines: usize,
29    /// How much its lookout has learned now.
30    pub counts: hydra_lookout::Counts,
31}
32
33/// Every game's lookout in `data` taught from the player logs under `root`,
34/// the logs folder, the days after those seeded before up to yesterday
35/// (`today`, `YYYY-MM-DD`); each written. `say` is told each day as it is
36/// read.
37///
38/// # Errors
39///
40/// The logs could not be listed, a day could not be read, or a lookout or
41/// its choices could not be written.
42pub fn seed(
43    root: &Path,
44    data: &Path,
45    today: &str,
46    say: &mut dyn FnMut(&str),
47) -> Result<Vec<Seeded>, String> {
48    let player = root.join(writer::SUBDIR);
49    let mut seeded = Vec::new();
50    for (folder, game) in games(&player)? {
51        let characters = folders(&player.join(&folder))?;
52        if characters.is_empty() {
53            continue;
54        }
55        let mut settings = hydra_lookout::Settings::read(data)?;
56        let after = settings.seeded.get(game).cloned().unwrap_or_default();
57        let whose: Vec<String> = characters
58            .iter()
59            .map(|name| format!("{folder}:{name}"))
60            .collect();
61        let mut days: BTreeSet<String> = BTreeSet::new();
62        for character in &whose {
63            let listed = reader::days(root, character)
64                .map_err(|why| format!("{character}'s days: {why}"))?;
65            days.extend(
66                listed
67                    .into_iter()
68                    .filter(|day| *day > after && day.as_str() < today),
69            );
70        }
71        let shared = hydra_lookout::open(data, game);
72        {
73            let mut lookout = hydra_lookout::lock(&shared);
74            for name in &characters {
75                lookout.own(name);
76            }
77        }
78        let mut lines = 0;
79        for day in &days {
80            let mut entries: Vec<(String, reader::Entry)> = Vec::new();
81            for (character, name) in whose.iter().zip(&characters) {
82                let read = reader::read_day(root, character, day)
83                    .map_err(|why| format!("{character} on {day}: {why}"))?;
84                entries.extend(read.into_iter().map(|entry| (name.clone(), entry)));
85            }
86            // In the order they came, whoever heard them: stable, so each
87            // character's own lines keep their order.
88            entries.sort_by(|a, b| a.1.at.cmp(&b.1.at));
89            lines += entries.len();
90            learn(&shared, day, &entries);
91            say(&format!("{game} {day}: {} lines", entries.len()));
92        }
93        let counts = {
94            let mut lookout = hydra_lookout::lock(&shared);
95            lookout.write()?;
96            lookout.counts()
97        };
98        if let Some(last) = days.last() {
99            settings.seeded.insert(game.to_owned(), last.clone());
100            settings.write(data)?;
101            hydra_lookout::reload(data);
102        }
103        seeded.push(Seeded {
104            game: game.to_owned(),
105            characters,
106            days: days.first().cloned().zip(days.last().cloned()),
107            lines,
108            counts,
109        });
110    }
111    Ok(seeded)
112}

114/// `entries`, one day's, each its character's, learned by `shared`.
115fn learn(shared: &hydra_lookout::Shared, day: &str, entries: &[(String, reader::Entry)]) {
116    #[expect(
117        clippy::cast_precision_loss,
118        reason = "days since 1970 fit an f64 exactly, being far under 2^52"
119    )]
120    let midnight = days_since_epoch(day).map_or(0.0, |days| days as f64 * 86_400.0);
121    let mut lookout = hydra_lookout::lock(shared);
122    // Whose last main line was a room's title: the next is its description.
123    let mut titled: Vec<(String, bool)> = Vec::new();
124    for (who, entry) in entries {
125        let stream = entry.stream.split('/').next().unwrap_or_default();
126        let after_title = {
127            let at = titled.iter().position(|(name, _)| name == who);
128            let was = at.is_some_and(|at| titled[at].1);
129            if stream == crate::player_log::MAIN {
130                let title = is_title(&entry.text);
131                match at {
132                    Some(at) => titled[at].1 = title,
133                    None => titled.push((who.clone(), title)),
134                }
135            }
136            was
137        };
138        let objects = objects(&entry.links);
139        let now = midnight + seconds_of(&entry.at);
140        lookout.judge(
141            who,
142            &hydra_lookout::Line {
143                stream,
144                text: &entry.text,
145                objects: &objects,
146                room: after_title && stream == crate::player_log::MAIN,
147            },
148            now,
149            true,
150        );
151    }
152}

154/// The game objects `links` name, as the lookout reads them.
155fn objects(links: &[LogLink]) -> Vec<hydra_lookout::Object> {
156    links
157        .iter()
158        .filter_map(|link| match &link.does {
159            Does::Object { noun, .. } => Some(hydra_lookout::Object {
160                at: link.from..link.to,
161                noun: noun.clone(),
162            }),
163            _ => None,
164        })
165        .collect()
166}

168/// Whether `text` is a room's title as the story shows one: `[Name]`, its
169/// number after it or not.
170fn is_title(text: &str) -> bool {
171    let text = text.trim();
172    let Some(inside) = text.strip_prefix('[') else {
173        return false;
174    };
175    let Some(end) = inside.find(']') else {
176        return false;
177    };
178    let after = inside[end + 1..].trim();
179    end > 0
180        && (after.is_empty()
181            || (after.starts_with('(')
182                && after.ends_with(')')
183                && after[1..after.len() - 1]
184                    .chars()
185                    .all(|c| c.is_ascii_digit())))
186}

188/// The game folders under `player` and the game each is, by its instance's
189/// name; a folder no game Hydra knows is left out.
190fn games(player: &Path) -> Result<Vec<(String, &'static str)>, String> {
191    Ok(folders(player)?
192        .into_iter()
193        .filter_map(|folder| hydra_net::instance(&folder).map(|game| (folder, game)))
194        .collect())
195}
196
197/// The folders in `dir`, by name, sorted; none when it is not there.
198fn folders(dir: &Path) -> Result<Vec<String>, String> {
199    let entries = match std::fs::read_dir(dir) {
200        Ok(entries) => entries,
201        Err(e) if e.kind() == std::io::ErrorKind::NotFound => return Ok(Vec::new()),
202        Err(e) => return Err(format!("{}: {e}", dir.display())),
203    };
204    let mut names: Vec<String> = entries
205        .filter_map(Result::ok)
206        .filter(|entry| entry.path().is_dir())
207        .filter_map(|entry| entry.file_name().to_str().map(str::to_owned))
208        .collect();
209    names.sort();
210    Ok(names)
211}

213/// Seconds into the day `at`, `HH:MM:SS.mmm`, says; 0 for words that are
214/// not a time.
215fn seconds_of(at: &str) -> f64 {
216    let mut parts = at.split(':');
217    let hours: f64 = parts.next().and_then(|h| h.parse().ok()).unwrap_or(0.0);
218    let minutes: f64 = parts.next().and_then(|m| m.parse().ok()).unwrap_or(0.0);
219    let seconds: f64 = parts.next().and_then(|s| s.parse().ok()).unwrap_or(0.0);
220    hours * 3600.0 + minutes * 60.0 + seconds
221}

223/// The days from 1970-01-01 to `day`, `YYYY-MM-DD`, on the civil calendar.
224fn days_since_epoch(day: &str) -> Option<i64> {
225    let mut parts = day.split('-');
226    let year: i64 = parts.next()?.parse().ok()?;
227    let month: i64 = parts.next()?.parse().ok()?;
228    let date: i64 = parts.next()?.parse().ok()?;
229    if !(1..=12).contains(&month) || !(1..=31).contains(&date) {
230        return None;
231    }
232    // Days from the civil date, counting years from March so a leap day
233    // ends the year.
234    let year = if month <= 2 { year - 1 } else { year };
235    let era = year.div_euclid(400);
236    let of_era = year - era * 400;
237    let of_year = (153 * (if month > 2 { month - 3 } else { month + 9 }) + 2) / 5 + date - 1;
238    let of_cycle = of_era * 365 + of_era / 4 - of_era / 100 + of_year;
239    Some(era * 146_097 + of_cycle - 719_468)
240}

242#[cfg(test)]
243mod tests {
244    use super::*;
245
246    #[test]
247    fn a_day_is_counted_from_the_epoch() {
248        assert_eq!(days_since_epoch("1970-01-01"), Some(0));
249        assert_eq!(days_since_epoch("2000-03-01"), Some(11_017));
250        assert_eq!(days_since_epoch("2026-10-24"), Some(20_750));
251        assert_eq!(days_since_epoch("2026-13-01"), None);
252        assert!((seconds_of("01:02:03.500") - 3723.5).abs() < 1e-9);
253    }
254
255    #[test]
256    fn a_title_is_known() {
257        assert!(is_title("[Wehnimer's Landing, Town Square]"));
258        assert!(is_title("[Ta'Vaalor, Victory Court] (10373)"));
259        assert!(!is_title("[General]-GSIV:Someone: \"hi\""));
260        assert!(!is_title("A gnarled arm quickly rises."));
261    }

263    /// Two characters' logs of one day, read once: a line they both heard
264    /// at the same moment is one occasion, and a second seeding reads no
265    /// day twice.
266    #[test]
267    fn the_logs_are_learned_once() {
268        let base = std::env::temp_dir().join(format!("hydra-lookout-seed-{}", std::process::id()));
269        let _ = std::fs::remove_dir_all(&base);
270        let (root, data) = (base.join("logs"), base.join("data"));
271        std::fs::create_dir_all(&data).unwrap();
272        for name in ["alpha", "beta"] {
273            let dir = root.join("player").join("gs3").join(name);
274            std::fs::create_dir_all(&dir).unwrap();
275            let text = concat!(
276                "[00:00:00.000][main] A cold wind blows.\n",
277                "[00:01:00.000][main] A cold wind blows.\n",
278                "[00:02:00.000][main] A cold wind blows.\n",
279                "[00:03:00.000][main] A cold wind blows.\n",
280                "[00:04:00.000][main] A cold wind blows.\n",
281                "[00:09:00.000][main] Both heard this once.\n",
282            );
283            std::fs::write(dir.join(format!("{name}_2026-10-01.log")), text).unwrap();
284        }
285        let mut said = Vec::new();
286        let seeded = seed(&root, &data, "2026-10-24", &mut |line| {
287            said.push(line.to_owned());
288        })
289        .unwrap();
290        assert_eq!(seeded.len(), 1);
291        assert_eq!(seeded[0].game, "Prime");
292        assert_eq!(seeded[0].lines, 12);
293        // The wind came on five occasions, each heard by both: learned.
294        // The other line came once, heard by both at the same moment: learning.
295        assert_eq!(seeded[0].counts.learned, 1);
296        assert_eq!(seeded[0].counts.learning, 1);
297        let again = seed(&root, &data, "2026-10-24", &mut |_| {}).unwrap();
298        assert_eq!(again[0].lines, 0, "no day twice");
299        let _ = std::fs::remove_dir_all(&base);
300    }

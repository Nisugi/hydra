1//! The relay's endpoints, HTTP/1.1 on localhost behind nginx.
2//!
3//! | Ask | With | Answer |
4//! |---|---|---|
5//! | `POST /register` | | `201 {"desktop": "<token>"}` |
6//! | `DELETE /register` | `Authorization: Bearer <token>` | `204` |
7//! | `POST /push` | the token, and a [`Note`] as JSON | `202`; `410` when the phone's token is no longer good |
8//! | `GET /health` | | `200` |
9//!
10//! nginx passes `/hydra-push/...` here with or without its prefix; the
11//! address asking is nginx's `X-Real-IP`.
12

27/// The most a request may carry.
28const BODY_MOST: usize = 8 * 1024;
29
30/// What the relay holds while it runs.
31pub struct Relay<A> {
32    apns: A,
33    topic: String,
34    tokens: Mutex<Tokens>,
35    desktops: Mutex<Desktops>,
36    limits: Mutex<Limits>,
37}
38
39/// How often each may ask.
40struct Limits {
41    /// Registering, by address: a few an hour.
42    register: Limit,
43    /// Pushing, by desktop.
44    desktop: Limit,
45    /// Pushed to, by phone.
46    phone: Limit,
47}
48
49/// An answer: its status and JSON body, if one.
50#[derive(Debug, PartialEq, Eq)]
51pub struct Answer {
52    /// The status.
53    pub status: StatusCode,
54    /// The body.
55    pub body: Option<serde_json::Value>,
56}
57
58fn answer(status: StatusCode, body: Option<serde_json::Value>) -> Answer {
59    Answer { status, body }
60}
61
62fn error(status: StatusCode, why: &str) -> Answer {
63    answer(status, Some(serde_json::json!({ "error": why })))
64}
65
66fn unix() -> u64 {
67    SystemTime::now()
68        .duration_since(UNIX_EPOCH)
69        .map_or(0, |since| since.as_secs())
70}
71
72fn lock<T>(held: &Mutex<T>) -> std::sync::MutexGuard<'_, T> {
73    held.lock().unwrap_or_else(PoisonError::into_inner)
74}

76/// One ask, as the endpoints read it.
77#[derive(Debug, Default)]
78pub struct Ask<'a> {
79    /// Its method.
80    pub method: Method,
81    /// Its path, nginx's prefix or not.
82    pub path: &'a str,
83    /// Its bearer token, if it gave one.
84    pub bearer: Option<&'a str>,
85    /// The address asking.
86    pub from: &'a str,
87    /// What it carried.
88    pub body: &'a [u8],
89}
90
91impl<A: Apns> Relay<A> {
92    /// A relay sending through `apns` for the app `topic`.
93    #[must_use]
94    pub fn new(apns: A, topic: &str, tokens: Tokens, desktops: Desktops) -> Self {
95        Self {
96            apns,
97            topic: topic.to_owned(),
98            tokens: Mutex::new(tokens),
99            desktops: Mutex::new(desktops),
100            limits: Mutex::new(Limits {
101                register: Limit::new(5, 1),
102                desktop: Limit::new(20, 30),
103                phone: Limit::new(10, 20),
104            }),
105        }
106    }
107
108    /// The answer to `ask`.
109    pub async fn answer(&self, ask: &Ask<'_>) -> Answer {
110        let path = ask.path.strip_prefix("/hydra-push").unwrap_or(ask.path);
111        match (&ask.method, path) {
112            (&Method::GET, "/health") => answer(StatusCode::OK, None),
113            (&Method::POST, "/register") => self.register(ask.from),
114            (&Method::DELETE, "/register") => self.forget(ask.bearer),
115            (&Method::POST, "/push") => self.push(ask.bearer, ask.body).await,
116            _ => error(StatusCode::NOT_FOUND, "no such ask"),
117        }
118    }
119
120    fn register(&self, from: &str) -> Answer {
121        if !lock(&self.limits).register.take(from, Instant::now()) {
122            return error(StatusCode::TOO_MANY_REQUESTS, "registered too often");
123        }
124        match lock(&self.desktops).register(unix()) {
125            Ok(token) => answer(
126                StatusCode::CREATED,
127                Some(serde_json::json!({ "desktop": token })),
128            ),
129            Err(why) => {
130                eprintln!("[hydra-push] register: {why}");
131                error(StatusCode::SERVICE_UNAVAILABLE, "not registered")
132            }
133        }
134    }
135
136    fn forget(&self, bearer: Option<&str>) -> Answer {
137        let Some(token) = bearer else {
138            return error(StatusCode::UNAUTHORIZED, "no desktop token");
139        };
140        match lock(&self.desktops).forget(token) {
141            Ok(true) => answer(StatusCode::NO_CONTENT, None),
142            Ok(false) => error(StatusCode::UNAUTHORIZED, "no such desktop"),
143            Err(why) => {
144                eprintln!("[hydra-push] forget: {why}");
145                error(StatusCode::SERVICE_UNAVAILABLE, "not forgotten")
146            }
147        }
148    }
149
150    async fn push(&self, bearer: Option<&str>, body: &[u8]) -> Answer {
151        let Some(desktop) = bearer.filter(|token| lock(&self.desktops).knows(token)) else {
152            return error(StatusCode::UNAUTHORIZED, "no such desktop");
153        };
154        let note: Note = match serde_json::from_slice(body) {
155            Ok(note) => note,
156            Err(why) => return error(StatusCode::BAD_REQUEST, &why.to_string()),
157        };
158        if let Some(why) = note.refused() {
159            return error(StatusCode::BAD_REQUEST, why);
160        }
161        {
162            let now = Instant::now();
163            let mut limits = lock(&self.limits);
164            if !limits.desktop.take(desktop, now) || !limits.phone.take(&note.device, now) {
165                return error(StatusCode::TOO_MANY_REQUESTS, "pushed too often");
166            }
167        }
168        for _ in 0..2 {
169            let token = lock(&self.tokens).token(Instant::now(), unix());
170            match self.apns.send(&note, &token, &self.topic).await {
171                Ok(Sent::Taken) => return answer(StatusCode::ACCEPTED, None),
172                Ok(Sent::Gone) => return error(StatusCode::GONE, "the phone's token is gone"),
173                Ok(Sent::Stale) => lock(&self.tokens).forget(),
174                Ok(Sent::Refused(status, reason)) => {
175                    eprintln!("[hydra-push] Apple refused a push: {status} {reason}");
176                    return error(StatusCode::BAD_GATEWAY, &reason);
177                }
178                Err(why) => {
179                    eprintln!("[hydra-push] Apple could not be reached: {why}");
180                    return error(StatusCode::BAD_GATEWAY, "Apple could not be reached");
181                }
182            }
183        }
184        error(StatusCode::BAD_GATEWAY, "Apple refused the provider token")
185    }
186}

188/// Serve `relay` on `at` until the process is told to stop.
189///
190/// # Errors
191///
192/// The address could not be listened on.
193pub async fn run<A: Apns>(relay: Arc<Relay<A>>, at: SocketAddr) -> Result<(), String> {
194    let listener = tokio::net::TcpListener::bind(at)
195        .await
196        .map_err(|why| format!("{at}: {why}"))?;
197    eprintln!("[hydra-push] listening on {at}");
198    loop {
199        let (stream, peer) = tokio::select! {
200            accepted = listener.accept() => match accepted {
201                Ok(accepted) => accepted,
202                Err(why) => {
203                    eprintln!("[hydra-push] accept: {why}");
204                    continue;
205                }
206            },
207            _ = tokio::signal::ctrl_c() => return Ok(()),
208        };
209        let relay = Arc::clone(&relay);
210        tokio::spawn(async move {
211            let service = hyper::service::service_fn(move |request| {
212                let relay = Arc::clone(&relay);
213                async move { Ok::<_, Infallible>(respond(&relay, request, peer).await) }
214            });
215            let io = hyper_util::rt::TokioIo::new(stream);
216            let _ = hyper::server::conn::http1::Builder::new()
217                .serve_connection(io, service)
218                .await;
219        });
220    }
221}

223async fn respond<A: Apns>(
224    relay: &Relay<A>,
225    request: Request<Incoming>,
226    peer: SocketAddr,
227) -> Response<Full<Bytes>> {
228    let (parts, body) = request.into_parts();
229    let header = |name: &str| {
230        parts
231            .headers
232            .get(name)
233            .and_then(|value| value.to_str().ok())
234    };
235    let from = header("x-real-ip").map_or_else(|| peer.ip().to_string(), str::to_owned);
236    let bearer = header("authorization")
237        .and_then(|value| {
238            value
239                .strip_prefix("Bearer ")
240                .or(value.strip_prefix("bearer "))
241        })
242        .map(str::trim);
243    let answer = match Limited::new(body, BODY_MOST).collect().await {
244        Ok(body) => {
245            let body = body.to_bytes();
246            relay
247                .answer(&Ask {
248                    method: parts.method.clone(),
249                    path: parts.uri.path(),
250                    bearer,
251                    from: &from,
252                    body: &body,
253                })
254                .await
255        }
256        Err(_) => error(StatusCode::PAYLOAD_TOO_LARGE, "too much"),
257    };
258    let body = answer
259        .body
260        .map(|body| Bytes::from(body.to_string()))
261        .unwrap_or_default();
262    let mut response = Response::new(Full::new(body));
263    *response.status_mut() = answer.status;
264    response.headers_mut().insert(
265        hyper::header::CONTENT_TYPE,
266        hyper::header::HeaderValue::from_static("application/json"),
267    );
268    response
269}

276    /// Apple, as the tests have it: each push's device, and the answer
277    /// given for it.
278    struct Double {
279        sent: Mutex<Vec<(String, String)>>,
280        answers: Mutex<Vec<Sent>>,
281    }
282
283    impl Apns for Arc<Double> {
284        async fn send(&self, note: &Note, token: &str, _: &str) -> Result<Sent, String> {
285            lock(&self.sent).push((note.device.clone(), token.to_owned()));
286            Ok(lock(&self.answers).pop().unwrap_or(Sent::Taken))
287        }
288    }
289
290    fn relay(name: &str, answers: Vec<Sent>) -> (Relay<Arc<Double>>, Arc<Double>) {
291        let dir = std::env::temp_dir().join(format!("hydra-push-{name}-{}", std::process::id()));
292        let _ = std::fs::remove_dir_all(&dir);
293        let pem = p256::ecdsa::SigningKey::from_slice(&[9u8; 32])
294            .expect("a key")
295            .to_pkcs8_pem(p256::pkcs8::LineEnding::LF)
296            .expect("written")
297            .to_string();
298        let double = Arc::new(Double {
299            sent: Mutex::new(Vec::new()),
300            answers: Mutex::new(answers),
301        });
302        let relay = Relay::new(
303            Arc::clone(&double),
304            "dev.hydrafe",
305            Tokens::new(&pem, "KEY", "TEAM").expect("read"),
306            Desktops::open(&dir).expect("opened"),
307        );
308        (relay, double)
309    }
310
311    async fn registered(relay: &Relay<Arc<Double>>) -> String {
312        let answer = relay
313            .answer(&Ask {
314                method: Method::POST,
315                path: "/hydra-push/register",
316                from: "10.0.0.1",
317                ..Ask::default()
318            })
319            .await;
320        assert_eq!(answer.status, StatusCode::CREATED);
321        answer.body.expect("a body")["desktop"]
322            .as_str()
323            .expect("a token")
324            .to_owned()
325    }
326
327    fn pushed<'a>(bearer: Option<&'a str>, body: &'a [u8]) -> Ask<'a> {
328        Ask {
329            method: Method::POST,
330            path: "/push",
331            bearer,
332            from: "10.0.0.1",
333            body,
334        }
335    }

337    const NOTE: &[u8] = br#"{"device":"ab12","sealed":"c2VhbGVk"}"#;

339    #[tokio::test]
340    async fn a_registered_desktop_pushes_and_a_stranger_does_not() {
341        let (relay, double) = relay("pushes", Vec::new());
342        let desktop = registered(&relay).await;
343        let answer = relay.answer(&pushed(Some(&desktop), NOTE)).await;
344        assert_eq!(answer.status, StatusCode::ACCEPTED);
345        assert_eq!(lock(&double.sent).len(), 1);
346        let stranger = relay.answer(&pushed(Some("not one"), NOTE)).await;
347        assert_eq!(stranger.status, StatusCode::UNAUTHORIZED);
348        let none = relay.answer(&pushed(None, NOTE)).await;
349        assert_eq!(none.status, StatusCode::UNAUTHORIZED);
350        assert_eq!(lock(&double.sent).len(), 1, "neither reached Apple");
351    }

353    #[tokio::test]
354    async fn a_gone_phone_is_said_and_a_stale_token_is_made_again() {
355        let (relay, double) = relay("gone", vec![Sent::Gone]);
356        let desktop = registered(&relay).await;
357        let answer = relay.answer(&pushed(Some(&desktop), NOTE)).await;
358        assert_eq!(answer.status, StatusCode::GONE);
359        lock(&double.answers).push(Sent::Stale);
360        let answer = relay.answer(&pushed(Some(&desktop), NOTE)).await;
361        assert_eq!(answer.status, StatusCode::ACCEPTED, "tried again");
362        assert_eq!(lock(&double.sent).len(), 3);
363    }

365    #[tokio::test]
366    async fn pushes_past_the_rate_are_refused() {
367        let (relay, _) = relay("rate", Vec::new());
368        let desktop = registered(&relay).await;
369        let mut statuses = Vec::new();
370        for _ in 0..12 {
371            statuses.push(relay.answer(&pushed(Some(&desktop), NOTE)).await.status);
372        }
373        assert!(
374            statuses[..10]
375                .iter()
376                .all(|status| *status == StatusCode::ACCEPTED)
377        );
378        assert_eq!(
379            statuses[11],
380            StatusCode::TOO_MANY_REQUESTS,
381            "a phone's burst is ten"
382        );
383    }

1//! Apple's push service: one alert to one phone, over HTTP/2.
2//!
3//! What Apple is handed says only "Hydra": the alert itself goes sealed,
4//! under `sealed`, and `mutable-content` asks the phone to let HydraFE's
5//! notification extension open it before the banner shows. A phone that
6//! cannot open it shows the words here.
7
8use std::future::Future;
9use std::time::Duration;
10
11use serde::{Deserialize, Serialize};
12
13/// Apple's production host: `TestFlight` and App Store builds.
14pub const PRODUCTION: &str = "https://api.push.apple.com";
15
16/// Apple's sandbox host: builds run from Xcode.
17pub const SANDBOX: &str = "https://api.sandbox.push.apple.com";
18
19/// The most a push may carry, as Apple counts it.
20pub const PAYLOAD_MOST: usize = 4096;
21
22/// How long Apple has to answer.
23const TIMEOUT: Duration = Duration::from_secs(15);
24
25/// One alert for one phone, as a desktop sends it.
26#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
27#[serde(deny_unknown_fields)]
28pub struct Note {
29    /// The phone's device token, in hex, as the app was given it.
30    pub device: String,
31    /// The phone runs a build from Xcode: Apple's sandbox.
32    #[serde(default)]
33    pub sandbox: bool,
34    /// The alert, sealed by the desktop: base64.
35    pub sealed: String,
36    /// Alerts with the same id replace one another on the phone.
37    #[serde(default, skip_serializing_if = "Option::is_none")]
38    pub collapse: Option<String>,
39}
40
41impl Note {
42    /// Why it cannot be sent as it is, if it cannot.
43    #[must_use]
44    pub fn refused(&self) -> Option<&'static str> {
45        let hex = |text: &str| !text.is_empty() && text.bytes().all(|b| b.is_ascii_hexdigit());
46        if !hex(&self.device) || self.device.len() > 200 {
47            return Some("the device token is not one");
48        }
49        if self
50            .collapse
51            .as_ref()
52            .is_some_and(|id| id.len() > 64 || !id.is_ascii())
53        {
54            return Some("the collapse id is longer than 64 bytes");
55        }
56        if payload(&self.sealed).len() > PAYLOAD_MOST {
57            return Some("the alert is too long for a push");
58        }
59        None
60    }
61}
62
63/// What Apple is handed for `sealed`.
64#[must_use]
65pub fn payload(sealed: &str) -> String {
66    serde_json::json!({
67        "aps": {
68            "alert": { "title": "Hydra", "body": "A trigger fired." },
69            "sound": "default",
70            "mutable-content": 1,
71        },
72        "sealed": sealed,
73    })
74    .to_string()
75}

77/// What came of a push.
78#[derive(Clone, Debug, PartialEq, Eq)]
79pub enum Sent {
80    /// Apple took it.
81    Taken,
82    /// The phone's token is no longer good: the desktop should forget it.
83    Gone,
84    /// The provider token was refused: made again and tried once more.
85    Stale,
86    /// Apple refused it, with its status and reason.
87    Refused(u16, String),
88}
89
90/// What sends a push: Apple, or a double in the tests.
91pub trait Apns: Send + Sync + 'static {
92    /// Hand `note` to Apple with `token`, the provider token, for `topic`.
93    fn send(
94        &self,
95        note: &Note,
96        token: &str,
97        topic: &str,
98    ) -> impl Future<Output = Result<Sent, String>> + Send;
99}
100
101/// Apple itself.
102#[derive(Clone, Debug)]
103pub struct Apple {
104    client: reqwest::Client,
105}
106
107impl Apple {
108    /// A client for Apple's push service.
109    ///
110    /// # Errors
111    ///
112    /// The TLS stack could not be made.
113    pub fn new() -> Result<Self, String> {
114        let client = reqwest::Client::builder()
115            .http2_prior_knowledge()
116            .timeout(TIMEOUT)
117            .build()
118            .map_err(|why| why.to_string())?;
119        Ok(Self { client })
120    }
121}
122
123/// Apple's answer for a refusal: its reason.
124#[derive(Deserialize)]
125struct Reason {
126    reason: String,
127}
128
129impl Apns for Apple {
130    async fn send(&self, note: &Note, token: &str, topic: &str) -> Result<Sent, String> {
131        let host = if note.sandbox { SANDBOX } else { PRODUCTION };
132        let mut request = self
133            .client
134            .post(format!("{host}/3/device/{}", note.device))
135            .header("authorization", format!("bearer {token}"))
136            .header("apns-topic", topic)
137            .header("apns-push-type", "alert")
138            .header("apns-priority", "10")
139            .body(payload(&note.sealed));
140        if let Some(id) = &note.collapse {
141            request = request.header("apns-collapse-id", id);
142        }
143        let response = request
144            .send()
145            .await
146            .map_err(|why| why.without_url().to_string())?;
147        let status = response.status().as_u16();
148        let reason = response
149            .bytes()
150            .await
151            .ok()
152            .and_then(|body| serde_json::from_slice::<Reason>(&body).ok())
153            .map(|said| said.reason)
154            .unwrap_or_default();
155        Ok(answered(status, reason))
156    }
157}
158
159/// Apple's `status` and `reason` as what came of the push.
160#[must_use]
161pub fn answered(status: u16, reason: String) -> Sent {
162    match (status, reason.as_str()) {
163        (200, _) => Sent::Taken,
164        (410, _) | (400, "BadDeviceToken" | "DeviceTokenNotForTopic") => Sent::Gone,
165        (403, "ExpiredProviderToken" | "InvalidProviderToken") => Sent::Stale,
166        _ => Sent::Refused(status, reason),
167    }
168}

170#[cfg(test)]
171mod tests {
172    use super::*;
173
174    fn note(device: &str, sealed: &str) -> Note {
175        Note {
176            device: device.to_owned(),
177            sandbox: false,
178            sealed: sealed.to_owned(),
179            collapse: None,
180        }
181    }
182
183    #[test]
184    fn a_note_is_refused_when_apple_would_refuse_it() {
185        assert_eq!(note("ab12", "c2VhbGVk").refused(), None);
186        assert!(note("", "x").refused().is_some());
187        assert!(note("not/hex", "x").refused().is_some());
188        assert!(note("ab", &"x".repeat(PAYLOAD_MOST)).refused().is_some());
189    }
190
191    #[test]
192    fn apples_answers_say_what_came_of_it() {
193        assert_eq!(answered(200, String::new()), Sent::Taken);
194        assert_eq!(answered(410, "Unregistered".to_owned()), Sent::Gone);
195        assert_eq!(answered(400, "BadDeviceToken".to_owned()), Sent::Gone);
196        assert_eq!(
197            answered(403, "ExpiredProviderToken".to_owned()),
198            Sent::Stale
199        );
200        assert_eq!(
201            answered(429, "TooManyRequests".to_owned()),
202            Sent::Refused(429, "TooManyRequests".to_owned())
203        );
204    }
205
206    #[test]
207    fn apple_is_handed_hydra_and_the_sealed_alert() {
208        let payload: serde_json::Value = serde_json::from_str(&payload("c2VhbGVk")).expect("json");
209        assert_eq!(payload["aps"]["mutable-content"], 1);
210        assert_eq!(payload["aps"]["alert"]["title"], "Hydra");
211        assert_eq!(payload["sealed"], "c2VhbGVk");
212    }

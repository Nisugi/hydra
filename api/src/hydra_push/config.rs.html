1//! What the relay runs with: one TOML file, `/etc/hydra-push/config.toml`
2//! unless named on the command line.
3//!
4//! ```toml
5//! listen = "127.0.0.1:8787"
6//! team_id = "ABCDE12345"
7//! key_id = "XYZ123ABCD"
8//! key_file = "/etc/hydra-push/AuthKey_XYZ123ABCD.p8"
9//! topic = "dev.hydrafe"
10//! state = "/var/lib/hydra-push"
11//! ```
12
13use std::net::SocketAddr;
14use std::path::{Path, PathBuf};
15
16use serde::Deserialize;
17
18/// Where the file is looked for when none is named.
19pub const DEFAULT: &str = "/etc/hydra-push/config.toml";
20
21/// The relay's settings.
22#[derive(Clone, Debug, PartialEq, Eq, Deserialize)]
23#[serde(deny_unknown_fields)]
24pub struct Config {
25    /// Where it listens: localhost, nginx in front.
26    #[serde(default = "listen")]
27    pub listen: SocketAddr,
28    /// The Apple team that signs HydraFE.
29    pub team_id: String,
30    /// The push key's id, as the developer portal names it.
31    pub key_id: String,
32    /// The push key, the `.p8` file the portal gave.
33    pub key_file: PathBuf,
34    /// The app's bundle id.
35    #[serde(default = "topic")]
36    pub topic: String,
37    /// Where the desktops' token hashes are kept.
38    pub state: PathBuf,
39}
40
41fn listen() -> SocketAddr {
42    SocketAddr::from(([127, 0, 0, 1], 8787))
43}
44
45fn topic() -> String {
46    "dev.hydrafe".to_owned()
47}
48
49impl Config {
50    /// The settings in `text`.
51    ///
52    /// # Errors
53    ///
54    /// They do not read.
55    pub fn parse(text: &str) -> Result<Self, String> {
56        toml::from_str(text).map_err(|why| why.to_string())
57    }
58
59    /// The settings in the file at `path`.
60    ///
61    /// # Errors
62    ///
63    /// It could not be read, or does not read.
64    pub fn read(path: &Path) -> Result<Self, String> {
65        let text =
66            std::fs::read_to_string(path).map_err(|why| format!("{}: {why}", path.display()))?;
67        Self::parse(&text).map_err(|why| format!("{}: {why}", path.display()))
68    }
69}

71#[cfg(test)]
72mod tests {
73    use super::*;
74
75    #[test]
76    fn the_settings_read_with_their_defaults() {
77        let config = Config::parse(
78            "team_id = \"T\"\nkey_id = \"K\"\nkey_file = \"/k.p8\"\nstate = \"/s\"\n",
79        );
80        assert_eq!(
81            config,
82            Ok(Config {
83                listen: listen(),
84                team_id: "T".to_owned(),
85                key_id: "K".to_owned(),
86                key_file: PathBuf::from("/k.p8"),
87                topic: "dev.hydrafe".to_owned(),
88                state: PathBuf::from("/s"),
89            })
90        );
91        assert!(Config::parse("team_id = \"T\"\nwhat = 1\n").is_err());
92    }

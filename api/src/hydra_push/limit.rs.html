1//! How often anyone may ask: a bucket for each key -- a desktop's token, a
2//! phone, an address -- that fills at a steady rate up to a burst.
3
4use std::collections::HashMap;
5use std::time::Instant;
6
7/// Buckets by key.
8#[derive(Debug)]
9pub struct Limit {
10    /// How many may come at once.
11    burst: f64,
12    /// How many more each second.
13    per_second: f64,
14    buckets: HashMap<String, (f64, Instant)>,
15}
16
17/// How many buckets are kept before the full ones are let go.
18const KEPT_MOST: usize = 10_000;
19
20impl Limit {
21    /// `burst` at once, and `per_minute` more each minute.
22    #[must_use]
23    pub fn new(burst: u32, per_minute: u32) -> Self {
24        Self {
25            burst: f64::from(burst),
26            per_second: f64::from(per_minute) / 60.0,
27            buckets: HashMap::new(),
28        }
29    }
30
31    /// Whether `key` may ask now; counted when it may.
32    pub fn take(&mut self, key: &str, now: Instant) -> bool {
33        if self.buckets.len() >= KEPT_MOST {
34            let (burst, rate) = (self.burst, self.per_second);
35            self.buckets.retain(|_, (left, at)| {
36                *left + now.saturating_duration_since(*at).as_secs_f64() * rate < burst
37            });
38        }
39        let (left, at) = self
40            .buckets
41            .entry(key.to_owned())
42            .or_insert((self.burst, now));
43        *left = (*left + now.saturating_duration_since(*at).as_secs_f64() * self.per_second)
44            .min(self.burst);
45        *at = now;
46        if *left < 1.0 {
47            return false;
48        }
49        *left -= 1.0;
50        true
51    }
52}

54#[cfg(test)]
55mod tests {
56    use super::*;
57    use std::time::Duration;
58
59    #[test]
60    fn a_burst_is_let_through_then_the_rate() {
61        let mut limit = Limit::new(3, 60);
62        let now = Instant::now();
63        assert!((0..3).all(|_| limit.take("a", now)));
64        assert!(!limit.take("a", now), "past the burst");
65        assert!(limit.take("b", now), "each key its own");
66        assert!(
67            limit.take("a", now + Duration::from_secs(1)),
68            "one a second"
69        );
70        assert!(!limit.take("a", now + Duration::from_secs(1)));
71    }

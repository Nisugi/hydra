1//! The provider token Apple asks of every push: a JWT, ES256, naming the
2//! key and the team, signed with the team's key.
3//!
4//! Apple refuses a token older than an hour, and one made again too often,
5//! so one is made and used until [`REMADE_AFTER`].
6
7use std::time::{Duration, Instant};
8
9use base64::Engine as _;
10use base64::engine::general_purpose::URL_SAFE_NO_PAD;
11use p256::ecdsa::signature::Signer as _;
12use p256::ecdsa::{Signature, SigningKey};
13use p256::pkcs8::DecodePrivateKey as _;
14
15/// How long a token is used before another is made.
16pub const REMADE_AFTER: Duration = Duration::from_mins(50);
17
18/// The team's key, and the token made with it last.
19pub struct Tokens {
20    key: SigningKey,
21    key_id: String,
22    team_id: String,
23    made: Option<(Instant, String)>,
24}
25
26impl std::fmt::Debug for Tokens {
27    /// Never the key.
28    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
29        f.debug_struct("Tokens")
30            .field("key_id", &self.key_id)
31            .field("team_id", &self.team_id)
32            .finish_non_exhaustive()
33    }
34}
35
36impl Tokens {
37    /// Tokens made with the key in `pem`, the `.p8` file's text.
38    ///
39    /// # Errors
40    ///
41    /// It is not a P-256 private key.
42    pub fn new(pem: &str, key_id: &str, team_id: &str) -> Result<Self, String> {
43        let key = SigningKey::from_pkcs8_pem(pem)
44            .map_err(|why| format!("the push key does not read: {why}"))?;
45        Ok(Self {
46            key,
47            key_id: key_id.to_owned(),
48            team_id: team_id.to_owned(),
49            made: None,
50        })
51    }
52
53    /// The token to send now, `unix` the time in seconds: the last one,
54    /// unless it is due to be made again.
55    pub fn token(&mut self, now: Instant, unix: u64) -> String {
56        if let Some((at, token)) = &self.made
57            && now.saturating_duration_since(*at) < REMADE_AFTER
58        {
59            return token.clone();
60        }
61        let token = self.made_at(unix);
62        self.made = Some((now, token.clone()));
63        token
64    }
65
66    /// A fresh one: Apple said the last was refused.
67    pub fn forget(&mut self) {
68        self.made = None;
69    }
70
71    fn made_at(&self, unix: u64) -> String {
72        let header = serde_json::json!({ "alg": "ES256", "kid": self.key_id });
73        let claims = serde_json::json!({ "iss": self.team_id, "iat": unix });
74        let signed = format!(
75            "{}.{}",
76            URL_SAFE_NO_PAD.encode(header.to_string()),
77            URL_SAFE_NO_PAD.encode(claims.to_string())
78        );
79        let signature: Signature = self.key.sign(signed.as_bytes());
80        format!("{signed}.{}", URL_SAFE_NO_PAD.encode(signature.to_bytes()))
81    }
82}

84#[cfg(test)]
85mod tests {
86    use super::*;
87    use p256::ecdsa::VerifyingKey;
88    use p256::ecdsa::signature::Verifier as _;
89    use p256::pkcs8::EncodePrivateKey as _;
90
91    fn key() -> (String, VerifyingKey) {
92        let key = SigningKey::from_slice(&[7u8; 32]).expect("a key");
93        let pem = key
94            .to_pkcs8_pem(p256::pkcs8::LineEnding::LF)
95            .expect("written")
96            .to_string();
97        (pem, *key.verifying_key())
98    }
99
100    /// The token says the key and team, Apple's way, and its signature is
101    /// the key's.
102    #[test]
103    fn a_token_names_the_key_and_team_and_is_signed() {
104        let (pem, verifying) = key();
105        let mut tokens = Tokens::new(&pem, "KEY", "TEAM").expect("read");
106        let token = tokens.token(Instant::now(), 1_700_000_000);
107        let parts: Vec<&str> = token.split('.').collect();
108        assert_eq!(parts.len(), 3);
109        let header = URL_SAFE_NO_PAD.decode(parts[0]).expect("header");
110        let claims = URL_SAFE_NO_PAD.decode(parts[1]).expect("claims");
111        assert_eq!(header, br#"{"alg":"ES256","kid":"KEY"}"#);
112        assert_eq!(claims, br#"{"iat":1700000000,"iss":"TEAM"}"#);
113        let signature = URL_SAFE_NO_PAD.decode(parts[2]).expect("signature");
114        let signature = Signature::from_slice(&signature).expect("64 bytes");
115        let signed = format!("{}.{}", parts[0], parts[1]);
116        assert!(verifying.verify(signed.as_bytes(), &signature).is_ok());
117    }
118
119    /// One token is used until it is due, then another is made.
120    #[test]
121    fn a_token_is_kept_until_it_is_due() {
122        let (pem, _) = key();
123        let mut tokens = Tokens::new(&pem, "KEY", "TEAM").expect("read");
124        let start = Instant::now();
125        let first = tokens.token(start, 100);
126        assert_eq!(tokens.token(start + Duration::from_mins(10), 700), first);
127        assert_ne!(tokens.token(start + REMADE_AFTER, 3100), first);
128        assert!(Tokens::new("not a key", "KEY", "TEAM").is_err());
129    }

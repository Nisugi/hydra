17/// First `<` in `s` that plausibly starts real markup.
18///
19/// A `<` preceded by `$` is the game's own broken escaping: ability HELP text
20/// ships `$<a href=$Q...$>...$</a$>`. Interpreting such a tag can corrupt
21/// parser state -- imagine a mangled `$<pushStream>` -- so mangled markup
22/// renders as literal text instead. It is rare in recorded traffic, but the
23/// failure mode when it does arrive is a link colour bleeding over everything
24/// after it.
25#[must_use]
26pub fn find_tag_start(s: &str) -> Option<usize> {
27    let bytes = s.as_bytes();
28    let mut from = 0;
29    while let Some(p) = s[from..].find('<').map(|p| p + from) {
30        if p > 0 && bytes[p - 1] == b'$' {
31            from = p + 1;
32            continue;
33        }
34        if !opens_markup(&bytes[p + 1..]) {
35            from = p + 1;
36            continue;
37        }
38        return Some(p);
39    }
40    None
41}

43/// Does what follows a `<` actually begin markup?
44///
45/// Without this guard any `<` would start a tag that ran to the next `>`, so
46/// the prose `"5 < 10 and 10 > 5"` would become `Text("5 ")`, `UnknownTag {
47/// name: "", raw: "< 10 and 10 >" }`, `Text(" 5")` -- and a consumer
48/// rendering text and logging unknown tags would show the player `"5   5"`.
49/// It would also produce `UnknownTag { name: "" }` for `<>` and `</>`, an
50/// element name that cannot exist.
51///
52/// An XML name starts with a letter, `_` or `:`; `/` opens a close tag and
53/// `!` a comment or declaration, both of which the caller handles. Everything
54/// else after a `<` is prose and stays prose.
55///
56/// This costs nothing recorded traffic relies on: prose `<` followed by a
57/// space is absent from it. The live cousin, `<#>` in Lich error echoes,
58/// still reaches the user correctly
59/// -- `#` is not a name-start char, so the whole thing stays text rather than
60/// becoming `UnknownTag { name: "#" }`.
61fn opens_markup(after: &[u8]) -> bool {
62    match after.first() {
63        Some(b'/' | b'!') => true,
64        Some(c) => c.is_ascii_alphabetic() || *c == b'_' || *c == b':',
65        None => false,
66    }
67}

69/// Element name of a raw tag: `<pushStream id=..>` -> `pushStream`,
70/// `</compDef>` -> `compDef`. Empty when the input is not tag-shaped.
71///
72/// The terminator set includes `/` and whitespace, which matters: the corpus
73/// carries `<style id="roomName" />` with a space before `/>`, so a scanner
74/// assuming `/` abuts `>` mis-reads the name.
75#[must_use]
76pub fn tag_name(tag: &str) -> &str {
77    let body = tag
78        .strip_prefix("</")
79        .or_else(|| tag.strip_prefix('<'))
80        .unwrap_or(tag);
81    let end = body
82        .find(|c: char| c.is_ascii_whitespace() || c == '>' || c == '/')
83        .unwrap_or(body.len());
84    &body[..end]
85}

87/// Is this a closing tag?
88///
89/// Tolerates the game's `$`-mangled form `</a$>`.
90#[must_use]
91pub fn is_close_tag(tag: &str) -> bool {
92    tag.starts_with("</")
93}

95/// Value of `name=` in a raw tag, accepting either quoting.
96///
97/// The wire mixes `id='room'` and `id="room"` freely, sometimes within one
98/// tag -- `vitals.xml` has `value="guarded" cmd='...'` on one `<dropDownBox>`.
99#[must_use]
100pub fn attribute(tag: &str, name: &str) -> Option<String> {
101    // **An empty name does not advance, and the loop never ends.**
102    //
103    // `"abc".find("")` is `Some(0)` -- for every haystack, at every offset --
104    // so `from = at + name.len()` leaves `from` where it was, index 0 is
105    // preceded by nothing (`preceded_ok`), the `=` test fails, `continue`,
106    // and the same zero-width match is found again. Not a slow path: a hang,
107    // in a `pub fn` of a `pub mod`, reachable by any caller that computes an
108    // attribute name and gets an empty string.
109    //
110    // `None` is the honest answer rather than a panic. There is no attribute
111    // with no name, so "not found" is exactly true, and a parser whose stated
112    // contract is that it never panics must not gain a panic at a leaf.
113    //
114    // (The walker below no longer searches for the name at all, so an empty
115    // name cannot hang it; the check stays, as the contract it states.)
116    if name.is_empty() {
117        return None;
118    }
119    // KNOWN LIMIT, documented rather than closed. An unquoted value
120    // (`<pushStream id=room name>`) is indistinguishable here from the
121    // attribute being absent, so the caller's `unwrap_or_default()`
122    // yields `StreamPush { id: "" }` -- a typed, confident frame that
123    // is wrong, and worse than an `UnknownTag`, because an empty id
124    // reads as a legitimate push to the main window.
125    //
126    // Closing it properly means `attribute` returning a three-way
127    // answer (absent / malformed / value) and all of its call sites
128    // deciding what to do with the middle case. That is a large change
129    // for a shape recorded traffic has never carried (**0 occurrences in
130    // over five million tags**), so the simplest thing that works stays.
131    //
132    // UPGRADE TRIGGER: the corpus replay reporting any unquoted
133    // attribute value, or a single real `StreamPush { id: "" }` that
134    // is not a bare `<pushStream/>`. Until then this crate claims
135    // only that it handles the wire it has seen.
136    pairs(tag)
137        .find(|(key, _)| *key == name)
138        // Decoded AND control-stripped: an id or title decoded to a raw ESC
139        // reaches a frontend's window title. See `decode_attribute_value`.
140        .map(|(_, value)| decode_attribute_value(value))
141}

143/// Each `key='value'` pair of a raw tag, in wire order, undecoded.
144///
145/// The one walk both [`attribute`] and [`attributes`] read by, so they
146/// agree: a search for `name=` anywhere in the tag would find one inside
147/// another attribute's quoted value (`<a text="say id=5" id='7'>` answering
148/// `5` for `id`). A key is the last token before its `=`, so a valueless
149/// attribute is dropped rather than merged into the next key; an unquoted
150/// value is skipped to the next space.
151fn pairs(tag: &str) -> impl Iterator<Item = (&str, &str)> {
152    let body = tag
153        .trim_start_matches('<')
154        .trim_end_matches('>')
155        .trim_end_matches('/');
156    // Skip `<name` / `</name`.
157    let mut rest = body.find(char::is_whitespace).map_or("", |at| &body[at..]);
158    std::iter::from_fn(move || {
159        loop {
160            let trimmed = rest.trim_start();
161            let eq = trimmed.find('=')?;
162            let key = trimmed[..eq].trim();
163            let key = key.rsplit(char::is_whitespace).next().unwrap_or(key);
164            let after_eq = trimmed[eq + 1..].trim_start();
165            let quote = after_eq.chars().next().filter(|q| *q == '\'' || *q == '"');
166            let Some((quote, end)) =
167                quote.and_then(|q| after_eq[q.len_utf8()..].find(q).map(|end| (q, end)))
168            else {
169                // A malformed pair: SKIP IT and keep going, rather than
170                // returning what has been collected so far. Returning early
171                // would silently drop every attribute after the bad one --
172                // `<crtrStatus exist='1' stunned='1' bad=unquoted bleeding='1'
173                // dead='1'/>` would yield only `exist` and `stunned`, losing
174                // `bleeding` and `dead` with no trace. For anything reading
175                // creature status that is a wrong decision, not a cosmetic
176                // one.
177                //
178                // Recorded traffic has no unquoted attribute values, so this
179                // is about what happens when the wire changes, not about
180                // traffic seen today. An unterminated quote has no next pair
181                // to find, so the walk ends.
182                let space = trimmed[eq..].find(char::is_whitespace)?;
183                rest = &trimmed[eq + space..];
184                continue;
185            };
186            let value = &after_eq[quote.len_utf8()..][..end];
187            rest = &after_eq[quote.len_utf8() + end + quote.len_utf8()..];
188            if !key.is_empty() {
189                return Some((key, value));
190            }
191        }
192    })
193}

195/// Value of `name=` parsed as a number, if it is one.
196#[must_use]
197pub fn attribute_u32(tag: &str, name: &str) -> Option<u32> {
198    attribute(tag, name)?.trim().parse().ok()
199}

201/// Every attribute in a raw tag, in wire order.
202///
203/// The shape `crtrStatus` and `roommeta` are kept in: raw, so the layer above
204/// owns the flag-name mapping. This crate does not know what `stunned` means.
205#[must_use]
206pub fn attributes(tag: &str) -> Vec<(String, String)> {
207    pairs(tag)
208        .map(|(key, value)| (key.to_owned(), decode_attribute_value(value)))
209        .collect()
210}

212/// Build a [`Link`] from an `<a>` or `<d>` tag.
213///
214/// The kind is decided by what the wire actually carries, as a [`LinkKind`].
215#[must_use]
216pub fn link_from_tag(tag: &str) -> Option<Link> {
217    let coord = attribute(tag, "coord");
218    if let Some(href) = attribute(tag, "href") {
219        return Some(Link {
220            kind: LinkKind::Url { href },
221            text: String::new(),
222            coord,
223        });
224    }
225    if let Some(id) = attribute(tag, "exist") {
226        return Some(Link {
227            kind: LinkKind::Exist {
228                id,
229                noun: attribute(tag, "noun").unwrap_or_default(),
230            },
231            text: String::new(),
232            coord,
233        });
234    }
235    let cmd = attribute(tag, "cmd")?;
236    Some(Link {
237        kind: LinkKind::Direct { cmd },
238        text: String::new(),
239        coord,
240    })
241}

247    #[test]
248    fn attributes_accept_either_quoting_and_do_not_match_suffixes() {
249        assert_eq!(
250            attribute("<nav rm='7503251'/>", "rm").as_deref(),
251            Some("7503251")
252        );
253        assert_eq!(
254            attribute("<streamWindow id=\"room\"/>", "id").as_deref(),
255            Some("room")
256        );
257        // `id` must not match inside `exist_id`; `top` must not match `stop`.
258        assert_eq!(
259            attribute("<x exist_id='7' id='real'/>", "id").as_deref(),
260            Some("real")
261        );
262        assert_eq!(attribute("<x stop='1'/>", "top"), None);
263        assert_eq!(attribute("<x/>", "missing"), None);
264        assert_eq!(attribute("<x a='1'/>", ""), None);
265    }

267    /// The two readers walk the same pairs: a name inside another
268    /// attribute's value is that value's text, not an attribute.
269    #[test]
270    fn a_name_inside_a_quoted_value_is_not_an_attribute() {
271        let tag = "<a noun='x' text=\"say id=5 to him\" id='7' cmd='look id=9'>";
272        assert_eq!(attribute(tag, "id").as_deref(), Some("7"));
273        assert_eq!(attribute(tag, "to"), None);
274        assert_eq!(
275            attributes(tag),
276            vec![
277                ("noun".to_owned(), "x".to_owned()),
278                ("text".to_owned(), "say id=5 to him".to_owned()),
279                ("id".to_owned(), "7".to_owned()),
280                ("cmd".to_owned(), "look id=9".to_owned()),
281            ]
282        );
283        // And the same answer whatever comes first, malformed pairs included.
284        for tag in [
285            "<x a=1 b='2' c=\"3\"/>",
286            "<roommeta weather='rain' bonfire inside='1'/>",
287            "<x a='unterminated/>",
288            "<x a = '1'  b= \"2\" />",
289        ] {
290            for (key, value) in attributes(tag) {
291                assert_eq!(attribute(tag, &key), Some(value), "{tag}: {key}");
292            }
293        }
294        assert_eq!(attribute("<x a = '1'/>", "a").as_deref(), Some("1"));
295    }

315    #[test]
316    fn attribute_bags_keep_wire_order_and_stop_at_malformed() {
317        let attrs = attributes("<crtrStatus exist='123' stunned='1' bleeding='0'/>");
318        assert_eq!(
319            attrs,
320            vec![
321                ("exist".to_owned(), "123".to_owned()),
322                ("stunned".to_owned(), "1".to_owned()),
323                ("bleeding".to_owned(), "0".to_owned()),
324            ]
325        );
326        // A malformed pair is SKIPPED, not a full stop. Stopping would drop
327        // every attribute after it: `bleeding` and `dead` below would vanish
328        // silently, and anything reading creature status would then act on a
329        // corpse it believed to be alive.
330        assert_eq!(
331            attributes("<x a=1 b='2'/>"),
332            vec![("b".to_owned(), "2".to_owned())]
333        );
334        assert_eq!(
335            attributes("<crtrStatus exist='1' stunned='1' bad=unquoted bleeding='1' dead='1'/>"),
336            vec![
337                ("exist".to_owned(), "1".to_owned()),
338                ("stunned".to_owned(), "1".to_owned()),
339                ("bleeding".to_owned(), "1".to_owned()),
340                ("dead".to_owned(), "1".to_owned()),
341            ]
342        );
343        // A valueless attribute is dropped, not merged into the next key.
344        // `("bonfire inside", "1")` is a name no consumer can match, and it
345        // took `inside` down with it.
346        assert_eq!(
347            attributes("<roommeta weather='rain' bonfire inside='1' sanctuary='1'/>"),
348            vec![
349                ("weather".to_owned(), "rain".to_owned()),
350                ("inside".to_owned(), "1".to_owned()),
351                ("sanctuary".to_owned(), "1".to_owned()),
352            ]
353        );
354        // An unterminated quote has no next pair to find, so the scan ends.
355        assert_eq!(attributes("<x a='unterminated/>"), vec![]);
356        assert_eq!(attributes("<popStream/>"), vec![]);
357    }

1//! A tag's inner text, the two ways a caller can want it.
2//!
3//! These free functions are apart from dispatch because they are not
4//! dispatch: none takes a `Parser`, none emits a frame, and all are about
5//! the bytes between two tags.
6
7use crate::text;
8
9/// Raw bytes between an open tag and its matching close, within one string.
10///
11/// **Still markup.** Callers that hand the result to the layer above want
12/// [`inner_display_text`] instead; this one exists for the two callers that
13/// parse the body themselves -- `prompt`, which decodes entities, and
14/// `parse_runs`, which turns markup into [`crate::runs::Runs`].
15pub(super) fn inner_text(tag: &str) -> String {
16    let Some(open_end) = tag.find('>') else {
17        return String::new();
18    };
19    let body = &tag[open_end + 1..];
20    match body.rfind("</") {
21        Some(close) => body[..close].to_owned(),
22        None => body.to_owned(),
23    }
24}

26/// The OPEN tag of a paired string: `<x a='1'>body</x>` -> `<x a='1'>`.
27///
28/// An envelope's own attributes must be read from here, never from the whole
29/// paired string. [`text::attribute`] scans every byte it is given, so asking
30/// the whole `<inventoryManager id=..><continuation root='777'/>...` for
31/// `root` would find the CHILD's and report it as the envelope's -- a
32/// fabricated value on an envelope that sent none.
33pub(super) fn open_tag(tag: &str) -> &str {
34    tag.find('>').map_or(tag, |end| &tag[..=end])
35}
36
37/// Every child element called `name` in a paired tag's body, each cut out as
38/// a standalone tag that [`text::attribute`] can read.
39///
40/// The one implementation `menu`, `objectives`, `cmdlist` and
41/// `inventoryManager` share. The NAME must match exactly -- a prefix test
42/// would read `<continuationX>` as a `<continuation>` -- and a child is cut at
43/// its own `>`, so prose after it is not folded into its attribute string.
44pub(super) fn children(tag: &str, name: &str) -> Vec<String> {
45    inner_text(tag)
46        .split('<')
47        .filter(|part| text::tag_name(part) == name)
48        .map(|part| {
49            let own = part.find('>').map_or(part, |end| &part[..end]);
50            format!("<{}>", own.trim_end_matches('/').trim_end())
51        })
52        .collect()
53}

55/// Every attribute of a row's own tag not among those `read`, in wire order.
56pub(super) fn unread(tag: &str, read: &[&str]) -> Vec<(String, String)> {
57    text::attributes(super::inner::open_tag(tag))
58        .into_iter()
59        .filter(|(name, _)| !read.contains(&name.as_str()))
60        .collect()
61}

63/// Every tag in a paired tag's body that is not one of the children `named`
64/// (open or close), each as `(name, raw)`: what an assembled envelope has no
65/// field for, to be reported rather than dropped. Tags inside a child called
66/// `holder` are left out: they are that child's own ([`inside_each`]).
67pub(super) fn other_children(
68    tag: &str,
69    named: &[&str],
70    holder: Option<&str>,
71) -> Vec<(String, String)> {
72    let mut held = false;
73    let mut others = Vec::new();
74    for (name, raw, close) in body_tags(tag) {
75        if Some(name.as_str()) == holder {
76            held = !close && !closes_itself(&raw);
77        } else if !held && !named.contains(&name.as_str()) {
78            others.push((name, raw));
79        }
80    }
81    others
82}

84/// For each child called `name` in a paired tag's body, in order, the tags
85/// between its open and its close, raw: none for one that closes itself.
86pub(super) fn inside_each(tag: &str, name: &str) -> Vec<Vec<String>> {
87    let mut each: Vec<Vec<String>> = Vec::new();
88    let mut open = false;
89    for (own, raw, close) in body_tags(tag) {
90        if own == name {
91            if !close {
92                each.push(Vec::new());
93            }
94            open = !close && !closes_itself(&raw);
95        } else if open && let Some(inside) = each.last_mut() {
96            inside.push(raw);
97        }
98    }
99    each
100}

102/// Every tag in a paired tag's body, as `(name, raw, whether it closes)`.
103fn body_tags(tag: &str) -> impl Iterator<Item = (String, String, bool)> {
104    inner_text(tag)
105        .split('<')
106        .skip(1)
107        .filter_map(|part| {
108            let own = part.find('>').map_or(part, |end| &part[..end]);
109            let raw = format!("<{own}>");
110            let name = text::tag_name(&raw).to_owned();
111            (!name.is_empty()).then(|| (name, raw, own.starts_with('/')))
112        })
113        .collect::<Vec<_>>()
114        .into_iter()
115}

117/// Whether a raw open tag closes itself: `<x a='1'/>`.
118fn closes_itself(raw: &str) -> bool {
119    raw.trim_end_matches('>').trim_end().ends_with('/')
120}

122/// A tag's inner text as the layer above should receive it: nested markup
123/// flattened away, entities decoded, control characters stripped.
124///
125/// Nothing above this crate sees an unparsed string, and five thin-tier
126/// fields would break that by handing back [`inner_text`] verbatim --
127/// `Spell.text`, `LeftHand.item`, `RightHand.item`, `WorldEvent.text` and
128/// `ActiveEffect.text`. A `<worldEvent>` reading
129/// `"A <b>great</b> storm &amp; flood!"` would reach the caller with the
130/// markup and the entity intact.
131///
132/// Unlike `Runs` it discards the style spans rather than modelling them:
133/// these are single-value fields and inventing a run vector for them would
134/// be structure no caller asked for. The upgrade trigger, recorded rather
135/// than pre-built: a consumer that needs the bold span inside a `<worldEvent>`
136/// moves that field to `Runs`, exactly as `Component` already is.
137///
138/// A `<` with no `>` after it ends the scan, so an unterminated nested tag
139/// truncates the text rather than leaking half a tag into it.
140pub(super) fn inner_display_text(tag: &str) -> String {
141    let raw = inner_text(tag);
142    let mut out = String::with_capacity(raw.len());
143    let mut rest = raw.as_str();
144    while let Some(lt) = rest.find('<') {
145        out.push_str(&rest[..lt]);
146        let Some(gt) = rest[lt..].find('>') else {
147            // An unterminated nested tag truncates the text rather than
148            // leaking half a tag into it.
149            rest = "";
150            break;
151        };
152        rest = &rest[lt + gt + 1..];
153    }
154    out.push_str(rest);
155    text::strip_control_chars(&text::decode_entities(&out))
156}

159mod tests {
160    use super::*;
161    use proptest::prelude::*;

163    /// A tag with a body, which is the shape `tagish()` in
164    /// `tests/parser_never_panics.rs` cannot produce.
165    ///
166    /// That generator emits one tag at a time -- `<x/>`, `<x>`, `</x` -- and
167    /// never a body BETWEEN two tags, so it never reaches `inner_text`'s whole
168    /// job, the `rfind("</")` arm. A property that never reaches the code
169    /// under test passes for the same reason an empty test does.
170    fn tag_with_body() -> impl Strategy<Value = String> {
171        let name = prop::sample::select(vec!["a", "d", "component", "preset", "b"]);
172        let body = prop::sample::select(vec![
173            "",
174            "plain",
175            "with <b>nested</b> markup",
176            "entity &amp; more",
177            "unterminated <b",
178            "</early>",
179            ">bare gt",
180            "a > b",
181        ]);
182        let close = prop::sample::select(vec![true, false]);
183        (name, body, close).prop_map(|(n, b, closed)| {
184            if closed {
185                format!("<{n}>{b}</{n}>")
186            } else {
187                format!("<{n}>{b}")
188            }
189        })
190    }
191
192    proptest! {
193        #![proptest_config(ProptestConfig::with_cases(1024))]
194
195        /// The result is always a **substring of the input**.
196        ///
197        /// `inner_text` slices; it never synthesises. This is what makes it
198        /// safe for `parse_runs` to re-scan the result as markup -- a function
199        /// that invented bytes could invent a tag.
200        #[test]
201        fn inner_text_only_ever_returns_a_slice_of_its_input(tag in tag_with_body()) {
202            let inner = inner_text(&tag);
203            prop_assert!(
204                inner.is_empty() || tag.contains(inner.as_str()),
205                "inner_text({tag:?}) = {inner:?}, which is not a substring"
206            );
207        }
208
209        /// The body starts strictly **after** the opening tag closes.
210        ///
211        /// Not "the opening tag appears nowhere in the result":
212        /// `<b>with <b>nested</b> markup</b>` legitimately CONTAINS `<b>`,
213        /// because the body is nested markup. `inner_text` returns markup by
214        /// contract -- [`inner_display_text`] is the flattening one.
215        ///
216        /// So the real invariant is positional, not lexical: whatever is
217        /// returned begins past the first `>`. That is what stops the opening
218        /// tag being re-scanned as part of the body.
219        #[test]
220        fn inner_text_starts_after_the_opening_tag(tag in tag_with_body()) {
221            let inner = inner_text(&tag);
222            if let (Some(open_end), false) = (tag.find('>'), inner.is_empty())
223                && let Some(at) = tag.find(inner.as_str())
224            {
225                prop_assert!(
226                    at > open_end,
227                    "body {inner:?} starts at {at}, inside the opening tag of {tag:?}"
228                );
229            }
230        }
231
232        /// Arbitrary input never panics and never grows.
233        ///
234        /// The length bound is the substring property's cheap corollary, and
235        /// it holds for inputs `tag_with_body` cannot express.
236        #[test]
237        fn inner_text_never_grows_its_input(tag in ".*") {
238            let inner = inner_text(&tag);
239            prop_assert!(inner.len() <= tag.len());
240        }
241
242        /// `inner_display_text` yields **no markup at all**.
243        ///
244        /// Nothing above this crate sees an unparsed string. A property is
245        /// the right shape for it because the claim is universal rather than
246        /// about the five fields named in [`inner_display_text`]'s doc.
247        #[test]
248        fn inner_display_text_never_leaks_a_tag(tag in tag_with_body()) {
249            let shown = inner_display_text(&tag);
250            prop_assert!(
251                !shown.contains('<'),
252                "markup reached the display text: {shown:?} from {tag:?}"
253            );
254        }
255    }

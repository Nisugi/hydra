1//! Emitting text: flushing the buffer into frames, and parsing a component
2//! body into runs.
3//!
4//! [`Parser::parse_runs_reporting`] is what makes a component body reach the
5//! layer above as structured text and links, never as a markup string.
6

9use crate::runs::{Run, Runs};

12impl Parser {
13    /// Mark the **last** text run of a finished line, in each stream it wrote
14    /// to, as ending it.
15    ///
16    /// Called once per `parse_line`: a line's runs carry `false` until the
17    /// last of each stream's is promoted. A line that writes to the main window
18    /// and then to a stream (`<pushStream>` part-way) ends in both, or its
19    /// main text would join the next main line. A line whose frames are all
20    /// non-text -- a bare `<prompt>`, an indicator -- promotes nothing, which
21    /// is correct: it contributed no display text, so there is no run for a
22    /// printer to terminate.
23    pub(super) fn mark_line_end(frames: &mut [Frame]) {
24        let mut ended: Vec<String> = Vec::new();
25        for frame in frames.iter_mut().rev() {
26            if let Frame::Text(text) = frame
27                && !ended.contains(&text.stream)
28            {
29                text.ends_line = true;
30                ended.push(text.stream.clone());
31            }
32        }
33    }
34
35    /// Emit the buffered text, if any, as a [`Frame::Text`].
36    pub(super) fn flush(&mut self, buffer: &mut String, frames: &mut Vec<Frame>) {
37        if buffer.is_empty() {
38            return;
39        }
40        let content = std::mem::take(buffer);
41        frames.push(self.text_frame(&content));
42    }

44    /// Wrap display text in the markup state currently open.
45    pub(super) fn text_frame(&mut self, content: &str) -> Frame {
46        let content = text::strip_control_chars(&text::decode_entities(content));
47        // Text accumulates on EVERY open link, not only the outermost.
48        //
49        // The innermost `<a exist=>` surfaces too, and an empty `text` would
50        // make it useless for exactly the case it exists for: `ready list`'s
51        // item would resolve to an id with no name, so a caller could not
52        // tell two katars apart.
53        //
54        // The outermost still gets the whole run, so nothing that reads
55        // `link` changes.
56        self.name_links(&content);
57        Frame::Text(TextFrame {
58            content,
59            stream: self.current_stream(),
60            style: self.style(),
61            link: self.links.first().cloned(),
62            inner_link: self.nested_object(),
63            // Set by `mark_line_end` once the line is fully parsed: a run
64            // cannot know here whether another follows it.
65            ends_line: false,
66        })
67    }

69    /// The markup open right now.
70    fn style(&self) -> Style {
71        Style {
72            bold_depth: self.bold_depth,
73            preset: self.presets.last().cloned(),
74            mono: self.mono,
75        }

78    /// The stream text currently belongs to; `""` is the main window.
79    fn current_stream(&self) -> String {
80        self.streams
81            .last()
82            .map(|s| s.id.clone())
83            .unwrap_or_default()
84    }

86    /// Parse a component body into [`Runs`], surfacing tags it carried that
87    /// this parser does not model.
88    ///
89    /// The layer above receives text and links, not a markup string.
90    ///
91    /// Markup state is saved and restored around the body, so an unbalanced
92    /// `<pushBold>` inside a component -- which the corpus does contain -- does
93    /// not leak bold onto the lines that follow it. Recomputing from a saved
94    /// snapshot removes that whole class of leak.
95    ///
96    /// # Why the body's unknown tags are collected
97    ///
98    /// `markup_state`'s `_ => {}` arm swallows anything that is not one of the
99    /// nine markup names, and inside a component body there is no frame vector
100    /// for a `Structural` or an `UnknownTag` to go into. Nor are the body's raw
101    /// bytes recoverable from the `Component` frame that encloses it:
102    /// `Component { id, body: Runs }` carries parsed runs and no raw bytes. So
103    /// without this, `<component id='room objs'>... a <newThing
104    /// id='1'/>rock.</component>` would yield runs and **no `UnknownTag` at
105    /// all**, on the room path, which is where the wire puts creatures and
106    /// loot.
107    ///
108    /// The names are collected here and emitted by the caller, after the frame
109    /// the body belongs to -- so a consumer sees the component first and then
110    /// what it could not understand, in wire order.
111    pub(super) fn parse_runs_reporting(
112        &mut self,
113        body: &str,
114        unmodelled: &mut Vec<String>,
115    ) -> Runs {
116        // **All four pieces of markup state, and all four the same way:
117        // cleared on entry, restored on exit.**
118        //
119        // Unsaved, `mono` would leave `<component id='x'><output
120        // class="mono"/>a</component>after`'s "after" -- and the rest of the
121        // session -- in the mono font. Treated differently on entry, an
122        // unbalanced `<pushBold/>` before a component would make its body bold
123        // where an unbalanced `<preset>` did not. A body is its own context in
124        // both directions: nothing leaks out of it, and nothing leaked from
125        // outside leaks in.
126        let saved_bold = std::mem::take(&mut self.bold_depth);
127        let saved_mono = std::mem::take(&mut self.mono);
128        let saved_presets = std::mem::take(&mut self.presets);
129        let saved_links = std::mem::take(&mut self.links);
130        let saved_over = std::mem::take(&mut self.links_over);
131
132        let mut runs = Vec::new();
133        let mut buffer = String::new();
134        let mut rest = body;
135        while !rest.is_empty() {
136            let Some(start) = text::find_tag_start(rest) else {
137                buffer.push_str(rest);
138                break;
139            };
140            buffer.push_str(&rest[..start]);
141            let tail = &rest[start..];
142            let Some(close) = tail.find('>') else {
143                buffer.push_str(tail);
144                break;
145            };
146            let tag = &tail[..=close];
147            rest = &tail[close + 1..];
148            self.push_run(&mut buffer, &mut runs);
149            // `markup_state`, not `markup_tag`: a component body becomes
150            // `Runs`, where the markup's effect is already carried by each
151            // run's own `style` and `link`.
152            //
153            // A tag that is NOT markup is collected rather than swallowed --
154            // see this function's header for why.
155            let name = super::text::tag_name(tag);
156            if !super::markup::is_markup(name) && !name.is_empty() {
157                // The WHOLE tag, not just the name: the raw form must reach
158                // the user, and a name alone cannot be shown.
159                unmodelled.push(tag.to_owned());
160            }
161            self.markup_state(tag);
162        }
163        self.push_run(&mut buffer, &mut runs);
164
165        self.bold_depth = saved_bold;
166        self.mono = saved_mono;
167        self.presets = saved_presets;
168        self.links = saved_links;
169        self.links_over = saved_over;
170        Runs { runs }
171    }

173    /// Emit one run of a component body, if there is text pending.
174    pub(super) fn push_run(&mut self, buffer: &mut String, runs: &mut Vec<Run>) {
175        if buffer.is_empty() {
176            return;
177        }
178        let content = text::strip_control_chars(&text::decode_entities(buffer));
179        buffer.clear();
180        self.name_links(&content);
181        runs.push(Run {
182            text: content,
183            style: self.style(),
184            link: self.links.first().cloned(),
185            inner_link: self.nested_object(),
186        });
187    }

189    /// `content` added to the text of every open link, up to
190    /// [`MAX_LINK_TEXT`] each.
191    ///
192    /// **The bound is on work, not on what is kept.** Every run inside a
193    /// link carries a copy of the outermost link, text so far included, so
194    /// unbounded, a link left open across a long line would cost the square
195    /// of its runs: a 256 KiB line of `x<pushBold/>` after one `<a>` would
196    /// make some 2 GiB of copies, under the line cap that is meant to bound
197    /// it. The run's own `content` is whole either way; a link's `text` is
198    /// the name of what it links. Of the 1,187 links in the committed
199    /// fixtures, the longest is 49 bytes.
200    fn name_links(&mut self, content: &str) {
201        for link in &mut self.links {
202            let room = MAX_LINK_TEXT.saturating_sub(link.text.len());
203            let mut fits = content.len().min(room);
204            while !content.is_char_boundary(fits) {
205                fits -= 1;
206            }
207            link.text.push_str(&content[..fits]);
208        }
209    }

211    /// The innermost open `<a exist=>`, when the outermost link is not it.
212    ///
213    /// `<d cmd=...>a <a exist=...>katar</a></d>` -- the click is the `<d>`
214    /// and the object is the `<a>`. Returning `None` when the outermost link
215    /// IS the `exist` keeps the common case from carrying a duplicate.
216    fn nested_object(&self) -> Option<crate::frame::Link> {
217        if matches!(
218            self.links.first().map(|l| &l.kind),
219            Some(crate::frame::LinkKind::Exist { .. })
220        ) {
221            return None;
222        }
223        self.links
224            .iter()
225            .rev()
226            .find(|l| matches!(l.kind, crate::frame::LinkKind::Exist { .. }))
227            .cloned()
228    }
229}

231/// The most of a link's text that is kept, in bytes: twenty times the
232/// longest the fixtures hold (`name_links`).
233pub(super) const MAX_LINK_TEXT: usize = 1024;

240    /// The text runs among `frames`.
241    fn texts(frames: &[Frame]) -> Vec<&crate::frame::TextFrame> {
242        frames
243            .iter()
244            .filter_map(|frame| match frame {
245                Frame::Text(text) => Some(text),
246                _ => None,
247            })
248            .collect()
249    }

251    /// A link left open over a long line: every run's copy of it is
252    /// bounded, where each was as long as the line so far.
253    #[test]
254    fn a_link_left_open_costs_no_more_than_its_cap_a_run() {
255        let line = format!(
256            "<a exist=\"1\" noun=\"rock\">{}",
257            "r\u{e9}<pushBold/>".repeat(4000)
258        );
259        let frames = crate::Parser::new().parse_line(&line);
260        let texts = texts(&frames);
261        assert_eq!(texts.len(), 4000, "every run is still there, whole");
262        let longest = texts
263            .iter()
264            .filter_map(|text| text.link.as_ref())
265            .map(|link| link.text.len())
266            .max();
267        assert!(
268            longest.is_some_and(|longest| longest <= MAX_LINK_TEXT && longest > MAX_LINK_TEXT - 4),
269            "kept to the cap, at a character's edge: {longest:?}"
270        );
271    }

295    fn body() -> impl Strategy<Value = String> {

317    /// The body's text with every tag removed, then decoded and stripped --
318    /// the same two transforms `push_run` applies, in the same order.
319    fn expected_text(body: &str) -> String {
320        let mut out = String::new();
321        let mut rest = body;
322        while let Some(start) = text::find_tag_start(rest) {
323            out.push_str(&rest[..start]);
324            let tail = &rest[start..];
325            let Some(close) = tail.find('>') else {
326                // An unterminated tag becomes text, which is what the
327                // function does with it too.
328                out.push_str(tail);
329                rest = "";
330                break;
331            };
332            rest = &tail[close + 1..];
333        }
334        out.push_str(rest);
335        text::strip_control_chars(&text::decode_entities(&out))
336    }

341        /// **The body's text is conserved exactly.**
342        ///
343        /// This is the property that earns the file. Concatenating every run's
344        /// text must reproduce the body's non-markup text, decoded and
345        /// control-stripped -- nothing dropped, duplicated or invented.
346        ///
347        /// It is a standing check on the invariant this crate exists to
348        /// uphold. **A property does not depend on having thought of the
349        /// case.**
350        #[test]
351        fn the_bodys_text_is_conserved(body in body()) {
352            let mut parser = Parser::new();
353            let mut unmodelled = Vec::new();
354            let runs = parser.parse_runs_reporting(&body, &mut unmodelled);
355            let joined: String = runs.runs.iter().map(|r| r.text.as_str()).collect();
356            prop_assert_eq!(
357                joined,
358                expected_text(&body),
359                "text was not conserved from {:?}",
360                body
361            );
362        }
363
364        /// **Every non-markup tag is reported, never swallowed.**
365        ///
366        /// The other half of conservation: `Component { body: Runs }` carries
367        /// no raw bytes, so a tag dropped here reaches nobody at all. The
368        /// whole tag is kept rather than its name -- a name alone cannot be
369        /// shown to a user.
370        ///
371        /// Asserting only that REPORTED tags are well-formed would pass a
372        /// parser that swallowed every unmodelled tag: over an empty vector
373        /// the claim is vacuously true. So the count is pinned to the input:
374        /// every non-markup tag in the body must appear, and only those.
375        #[test]
376        fn unmodelled_tags_are_reported_whole(body in body()) {
377            let mut parser = Parser::new();
378            let mut unmodelled = Vec::new();
379            let _ = parser.parse_runs_reporting(&body, &mut unmodelled);
380
381            // Counted from the input, independently of the parser.
382            let expected = ["<newThing id=\"1\"/>", "<futureTag attr=\"v\">", "<unknownFutureTag/>"]
383                .iter()
384                .map(|t| body.matches(t).count())
385                .sum::<usize>();
386            prop_assert_eq!(
387                unmodelled.len(),
388                expected,
389                "expected {} unmodelled tags from {:?}, got {:?}",
390                expected,
391                body,
392                unmodelled
393            );
394
395            for tag in &unmodelled {
396                prop_assert!(
397                    body.contains(tag.as_str()),
398                    "reported {tag:?}, which is not in {body:?}"
399                );
400                prop_assert!(
401                    tag.starts_with('<') && tag.ends_with('>'),
402                    "reported a fragment rather than a whole tag: {tag:?}"
403                );
404            }
405        }

407        /// A body **never leaks a whole tag into a run's text**.
408        ///
409        /// Not "no run text contains `<`": `&gt;&lt;` decodes to `><`, and
410        /// **that `<` is text the game sent**, not markup. Entities are
411        /// decoded AFTER tags are removed, so a decoded `<` is a literal the
412        /// user must see.
413        ///
414        /// The real claim is that no tag from the input survives whole into
415        /// a run -- no markup crosses into text, without forbidding a
416        /// character the wire legitimately carries.
417        #[test]
418        fn no_run_carries_a_whole_tag(body in body()) {
419            let mut parser = Parser::new();
420            let mut unmodelled = Vec::new();
421            let runs = parser.parse_runs_reporting(&body, &mut unmodelled);
422            for run in &runs.runs {
423                for tag in ["<pushBold/>", "<popBold/>", "<unknownFutureTag/>"] {
424                    prop_assert!(
425                        !run.text.contains(tag),
426                        "the tag {tag} survived into run text: {:?}",
427                        run.text

430            }
431        }

463    #[test]
464    fn a_body_neither_inherits_nor_leaks_markup_state() {
465        // Mono must not leak OUT, and bold must not leak IN, any more than
466        // presets and links do.
467        let mut parser = Parser::new();
468        let mut unmodelled = Vec::new();
469        let _ = parser.parse_runs_reporting("<output class=\"mono\"/>a", &mut unmodelled);
470        assert!(!parser.mono, "mono leaked out of the body");
471
472        parser.bold_depth = 1;
473        parser.mono = true;
474        let runs = parser.parse_runs_reporting("plain", &mut unmodelled);
475        let styles: Vec<_> = runs
476            .runs
477            .iter()
478            .map(|r| (r.style.bold_depth, r.style.mono))
479            .collect();
480        assert_eq!(styles, [(0, false)], "outer markup leaked into the body");
481        assert_eq!(
482            (parser.bold_depth, parser.mono),
483            (1, true),
484            "and is restored after"
485        );

1//! Tag dispatch: which frame each wire tag becomes.
2//!
3//! # Modelled and thin
4//!
5//! Room, prompt and vitals get real handlers here. Every other tag in
6//! [`crate::tags`] still produces a typed frame -- it parses, it reaches the
7//! caller, it renders -- but its fields are the wire's attributes rather
8//! than a modelled interpretation. There is no `todo!()` and no `panic!()`
9//! anywhere in this crate: both are denied by the workspace lints, and the
10//! parser never panics.
11//!
12//! # Nothing that arrives is dropped
13//!
14//! Every tag that comes in produces a frame. A known tag with nothing to
15//! model is not "nothing happened": discarding known-but-unhandled tags, or
16//! dropping a few names before the unknown check, would make whole tags
17//! vanish with no user-visible trace. **Being in [`crate::tags`] must never
18//! be what silences a tag** -- the same shape `tags.rs` records, where an
19//! explicit arm shields a name from the check meant to catch it.
20//!
21//! `tests/every_tag_is_observable.rs` generates every table tag in open,
22//! close and self-closing form and asserts each produces a frame. Most close
23//! forms never arrive -- only a couple of dozen element names ever appear as
24//! a close tag at all -- but a tag Simutronics starts sending tomorrow
25//! arrives without warning, so the rule has to be cheap across all of them
26//! rather than one modelled case each.
27//!
28//! [`Frame::Structural`] is how: every arm with nothing to model pushes one,
29//! carrying the raw bytes. `<a>`/`</a>` and `<d>`/`</d>` are the bulk of them
30//! and their effect genuinely IS in `Text.link` -- but only if a following
31//! text frame exists, and at end of line none does, so they emit too.
32//!
33//! # The cost, measured rather than guessed
34//!
35//! Replaying 60 stratified corpus files through `Parser::push_bytes`:
36//!
37//! ```text
38//! files=60 total_frames=5912327 structural=2098146 (35.5%)
39//!    1508330  a          130761  pushBold      2827  openDialog
40//!     212831  dialogData  73252  style         1019  resource
41//!     130761  popBold     29408  d              526  stream
42//! ```
43//!
44//! **Just over a third of all frames.** That is the honest price of the rule
45//! and it is why [`Frame::is_structural`] ships in this crate rather than
46//! being left to each consumer: a renderer, the replay differ and the model
47//! all need the same one-line predicate.
48//!
49//! The risk is that `Structural` becomes a dumping ground. Two things hold
50//! it: it carries `raw`, so nothing is unrecoverable, and
51//! `tests/every_tag_is_observable.rs` counts what lands here, so a tag
52//! *moving into* `Structural` shows up in a diff instead of silently.

66    /// Route one complete tag to its handler.
67    ///
68    /// `buffer` holds display text accumulated since the last frame; a handler
69    /// that emits a frame flushes it first so ordering is preserved.
70    pub(super) fn dispatch(&mut self, tag: &str, buffer: &mut String, frames: &mut Vec<Frame>) {
71        let name = text::tag_name(tag);
72        let closing = text::is_close_tag(tag);
73
74        // Markup tags change style state and emit no frame of their own, but
75        // they do split the text run, so the buffer flushes first.
76        if is_markup(name) {
77            self.flush(buffer, frames);
78            self.markup_tag(tag, frames);
79            return;
80        }
81
82        if closing {
83            self.close_tag(tag, buffer, frames);
84            return;
85        }
86
87        match name {
88            // --- prompt: also the resync barrier ---------------------------
89            "prompt" => {
90                self.flush(buffer, frames);
91                self.prompt(tag, frames);
92            }
93
94            // --- room -----------------------------------------------------
95            "nav" => {
96                self.flush(buffer, frames);
97                // **No `unwrap_or_default()`.** A bare `<nav/>` is a real
98                // shape -- an arrival at a room with no UID -- and turning it
99                // into `id: ""` hands a consumer an empty string it cannot
100                // tell from a real UID.
101                frames.push(Frame::RoomId {
102                    id: text::attribute(tag, "rm"),
103                });
104            }
105            "streamWindow" => {
106                self.flush(buffer, frames);
107                frames.push(Frame::StreamWindow {
108                    id: text::attribute(tag, "id").unwrap_or_default(),
109                    title: text::attribute(tag, "title"),
110                    subtitle: text::attribute(tag, "subtitle"),
111                    // Every attribute, not only the three lifted: see the
112                    // variant's docs for the census; `location` and `target`
113                    // ride ~99% of these tags.
114                    attrs: text::attributes(tag),
115                });
116            }
117            // Same fact, two controls: a streamBox and a stream window.
118            // `clearDynaStream` must not reach `thin.rs` and become a
119            // `StreamWindow` -- a clear reported as a DECLARATION.
120            "clearStream" | "clearDynaStream" => self.clear_stream(tag, buffer, frames),
121            "pushStream" => {
122                self.flush(buffer, frames);
123                let id = text::attribute(tag, "id").unwrap_or_default();
124                self.open_stream(id, false, frames);
125            }
126            // `<stream id=X>...</stream>` is the PAIRED form of the same
127            // redirect (an inline redirect), so it must establish the same
128            // routing context.
129            //
130            // Falling through to `thin.rs`'s `"streamId" | "stream" | ...`
131            // arm would make it a `Frame::StreamWindow` -- a WINDOW
132            // DECLARATION -- while its body went to `main` untagged: rows of
133            // `<stream id="Spells">`, each a spell-list entry, delivered as
134            // story prose. Anything filtering on `stream == "Spells"` would see
135            // none of them, and a consumer keeping a window registry would be
136            // told a window had just been declared.
137            //
138            // **`dynaStream` is the same shape one control over**: a
139            // text-content feed for a streamBox control, so its paired form
140            // establishes the same routing context.
141            "stream" | "dynaStream" if !tag.trim_end().ends_with("/>") => {
142                self.flush(buffer, frames);
143                let id = text::attribute(tag, "id").unwrap_or_default();
144                self.open_stream(id, true, frames);
145            }
146            // Self-closing: no body to route, and still not a window.
147            // Structural, so nothing is dropped -- `thin.rs`'s fallthrough
148            // would make it a `WindowHints` or a `StreamWindow`, the same
149            // wrong answer the paired arm above avoids, in the one form that
150            // arm's guard excludes.
151            "stream" | "dynaStream" => self.structural_only(name, tag, buffer, frames),
152            "popStream" => {
153                self.flush(buffer, frames);
154                self.pop_stream(tag, frames);
155            }
156            "compass" => {
157                self.flush(buffer, frames);
158                frames.push(Frame::Compass {
159                    directions: directions(tag),
160                });
161            }
162            // These always close on the line they open; one that does not is
163            // a MalformedTag and never reaches dispatch.
164            "component" | "compDef" | "inv" => {
165                self.flush(buffer, frames);
166                self.inline_paired(tag, frames);
167            }
168            "egMerchant" => self.eg_merchant(tag, buffer, frames),
169            // Also one line, always: even a 60-item menu arrives on one.
170            // Assembled rather than emitted per-tag, so the coordinates stay
171            // attached to the menu that answers for them.
172            // Bodies that are not prose on this line: three assembled from
173            // CHILDREN, and one captured across lines. See `assembled`.
174            "menu" | "objectives" | "inventoryManager" | "inventoryViewItem" | "cmdlist" => {
175                self.flush(buffer, frames);
176                self.assembled(name, tag, frames);
177            }
178            // `picture=` is what makes a `<resource>` a room picture. Without
179            // it there is nothing to model -- but "nothing to model" is not
180            // "nothing happened", so it is still a frame.
181            "resource" => {
182                self.flush(buffer, frames);
183                match text::attribute_u32(tag, "picture") {
184                    Some(id) => frames.push(Frame::RoomPicture { id }),
185                    None => frames.push(Frame::structural(name, tag)),
186                }
187            }
188
189            // --- vitals ---------------------------------------------------
190            "progressBar" => {
191                self.flush(buffer, frames);
192                frames.push(progress_bar(tag, self.dialog.as_deref()));
193            }
194            "dialogData" => {
195                self.flush(buffer, frames);
196                let id = text::attribute(tag, "id").unwrap_or_default();
197                // `clear='t'` on the OPEN tag is the wire's real "replace this
198                // dialog's contents" signal, and it is paired, not
199                // self-closing: measured over 40 corpus files, 26,745
200                // `clear='t'` opens against **zero** self-closing ones.
201                // Reading only `ends_with("/>")` would miss the form that
202                // actually occurs, and a stale buff list would never be
203                // cleared.
204                //
205                // A self-closing `dialogData` has no body to open either, so
206                // both it and `clear='t'` mean the same thing: clear the
207                // dialog. A paired one additionally becomes the enclosing
208                // dialog for the `<progressBar>`s inside it.
209                // A Lich script built with REXML writes `clear="true"`.
210                let clears = matches!(text::attribute(tag, "clear").as_deref(), Some("t" | "true"))
211                    || tag.ends_with("/>");
212                if clears {
213                    frames.push(Frame::ClearDialogData { id: id.clone() });
214                }
215                if !tag.ends_with("/>") {
216                    // **A paired open emits, whether or not it clears.**
217                    //
218                    // Otherwise `<dialogData id='expr'>` with no `clear=`
219                    // would produce no frame at all: the dialog opens, its
220                    // widgets follow, and nothing tells a consumer which
221                    // dialog they are in.
222                    //
223                    // `DialogOpen` is the existing frame for "a dialog is
224                    // open" -- `openDialog` uses it -- so this is the same
225                    // fact reaching the same place rather than new vocabulary.
226                    frames.push(Frame::DialogOpen {
227                        id: id.clone(),
228                        title: None,
229                        attrs: text::attributes(tag),
230                    });
231                    self.dialog = Some(id);
232                }
233            }
234            "clearDialogData" => {
235                self.flush(buffer, frames);
236                frames.push(Frame::ClearDialogData {
237                    id: text::attribute(tag, "id").unwrap_or_default(),
238                });
239            }
240
241            // --- everything else: typed, thin, never dropped ---------------
242            _ => self.thin(name, tag, buffer, frames),
243        }
244    }

246    /// A tag with no state of its own here, typed by `thin.rs`; a paired
247    /// `<openDialog>` opens its panel's markup, whose whitespace is not text
248    /// (`emit.rs`).
249    fn thin(&mut self, name: &str, tag: &str, buffer: &mut String, frames: &mut Vec<Frame>) {
250        self.flush(buffer, frames);
251        // A picture is placed in the stream it came in, which only the
252        // parser knows.
253        if matches!(name, "sagaImage" | "sagaimage") {
254            frames.push(Frame::InlineImage(inline_image(tag, self.current_stream())));
255            return;
256        }
257        frames.push(thin_frame(name, tag, self.dialog.as_deref()));
258        if name.eq_ignore_ascii_case("openDialog") {
259            self.panel = !tag.ends_with("/>");
260        }
261    }

263    /// `<popStream>`: close the stream it NAMES, not merely the innermost.
264    ///
265    /// Popping the innermost entry regardless would make
266    /// `<pushStream id='a'/><pushStream id='b'/><popStream id='a'/>` pop `b`
267    /// while announcing that `a` was popped and resumed, so every following
268    /// line would route to the wrong window. Nesting is real -- 151 lines in a
269    /// 272-file sample carry two pushes -- and since `pushStream` is not
270    /// stack-disciplined (19,142 pushes to 14,714 pops), a mismatch is the
271    /// expected case rather than the exotic one.
272    fn pop_stream(&mut self, tag: &str, frames: &mut Vec<Frame>) {
273        let id = text::attribute(tag, "id");
274        match &id {
275            // Remove the named stream wherever it sits. If it is not open at
276            // all, touch nothing: popping an unrelated stream to honour a pop
277            // that does not apply is the failure above.
278            Some(named) => {
279                if let Some(at) = self.streams.iter().rposition(|s| &s.id == named) {
280                    self.streams.remove(at);
281                }
282            }
283            // A bare `<popStream/>` means "the current one", which is 625 of
284            // the pops in that sample.
285            None => {
286                self.streams.pop();
287            }
288        }
289        frames.push(Frame::StreamPop { id });
290        self.resume_enclosing(frames);
291    }

293    /// Open a redirect, recording which form opened it.
294    ///
295    /// Shared by `<pushStream>` and the paired `<stream id=>`, which establish
296    /// the same routing context and differ only in how they close -- see
297    /// [`OpenStream`](super::OpenStream).
298    fn open_stream(&mut self, id: String, paired: bool, frames: &mut Vec<Frame>) {
299        self.streams.push(super::OpenStream {
300            id: id.clone(),
301            paired,
302        });
303        frames.push(Frame::StreamPush { id });
304    }

306    /// Announce the stream that is current again after a close, if any.
307    ///
308    /// A scalar consumer tracks one current stream, so it needs telling that an
309    /// enclosing redirect is back in force; nothing is emitted when the stack
310    /// empties, because `""` is the main window and `StreamPop` already said so.
311    fn resume_enclosing(&mut self, frames: &mut Vec<Frame>) {
312        if let Some(resumed) = self.streams.last().map(|s| s.id.clone()) {
313            frames.push(Frame::StreamResume { id: resumed });
314        }
315    }

317    /// `<prompt>`: emit the frame, then resync.
318    ///
319    /// The resync is what makes the measured `pushStream`:`popStream`
320    /// imbalance of 1.301 harmless. Prompts are trustworthy whether or not a
321    /// Lich is attached, so every prompt force-closes whatever was left open.
322    /// This bounds state corruption to one round.
323    fn prompt(&mut self, tag: &str, frames: &mut Vec<Frame>) {
324        let time = text::attribute(tag, "time").unwrap_or_default();
325        let inner = inner_text(tag);
326        // **The forced pops come BEFORE the prompt.**
327        //
328        // A consumer that snapshots on `Prompt` -- which is the documented
329        // round boundary -- would otherwise snapshot with streams still open,
330        // and see them close after the boundary they define.
331        for open in std::mem::take(&mut self.streams).into_iter().rev() {
332            frames.push(Frame::StreamPopForced { id: open.id });
333        }
334        frames.push(Frame::Prompt {
335            time,
336            // **STRIPPED, like every other text that reaches a terminal.**
337            //
338            // `emit.rs` and `inner.rs` pair `decode_entities` with
339            // `strip_control_chars` too, and a prompt can be printed straight
340            // to a terminal.
341            //
342            // The game socket is PLAIN TCP, so the bytes are not merely
343            // untrusted-because-remote, they are modifiable in flight.
344            // `<prompt time="1">&#27;]52;c;...&#7;</prompt>` decodes to a real
345            // ESC and BEL: an OSC-52 sequence writes the terminal's clipboard.
346            // Entities are decoded HERE, so the control characters do not
347            // exist on the wire for anything upstream to have caught.
348            text: text::strip_control_chars(&text::decode_entities(&inner)),
349        });
350        self.bold_depth = 0;
351        self.presets.clear();
352        self.links.clear();
353        self.links_over = 0;
354        self.mono = false;
355        self.dialog = None;
356        self.panel = false;
357    }

359    /// A `<component>`/`<compDef>`/`<inv>`, which always opens and closes on
360    /// one line (see [`menu`] for the other shape of the same idea): 447,095 opens in a 272-file corpus sample, none spanning a
361    /// line (see MULTI-LINE CAPTURES in `parser.rs`).
362    fn inline_paired(&mut self, tag: &str, frames: &mut Vec<Frame>) {
363        // From the open tag, never the whole paired string (`inner::open_tag`
364        // says why): asked of the whole, an envelope that sent no `id` would
365        // be given its first child's.
366        let id = text::attribute(open_tag(tag), "id").unwrap_or_default();
367        let mut unmodelled = Vec::new();
368        let body = self.parse_runs_reporting(&inner_text(tag), &mut unmodelled);
369        // `<inv>` is a container's contents, not a room component: same shape
370        // on the wire, different frame, so the name is what decides.
371        if text::tag_name(tag) == "inv" {
372            frames.push(Frame::ContainerItem {
373                container_id: id,
374                content: body,
375            });
376        } else {
377            frames.push(Frame::Component { id, body });
378        }
379        // **After the frame, in wire order.** A consumer sees the component it
380        // can use, then whatever the body carried that this parser does not
381        // model. Emitted here rather than inside `parse_runs_reporting`
382        // because only the caller knows which frame the body belonged to, and
383        // an unmodelled tag shows *after* the thing it was found in.
384        unmodelled_after(unmodelled, frames);
385    }

387    /// An `<egMerchant>`: its attributes from the open tag, its body as runs,
388    /// so a link in an announcement stays a link. Paired with its
389    /// announcement or rules as the body, or self-closing with none; on one
390    /// line, as every paired tag is.
391    fn eg_merchant(&mut self, tag: &str, buffer: &mut String, frames: &mut Vec<Frame>) {
392        self.flush(buffer, frames);
393        let open = open_tag(tag);
394        let mut unmodelled = Vec::new();
395        let text = self.parse_runs_reporting(&inner_text(tag), &mut unmodelled);
396        let trimmed = |name| text::attribute(open, name).map(|value| value.trim().to_owned());
397        let positive = |name| text::attribute_u32(open, name).filter(|&n| n > 0);
398        frames.push(Frame::EbonGateMerchant(EbonGateMerchant {
399            name: trimmed("name").unwrap_or_default(),
400            room: trimmed("room").unwrap_or_default(),
401            status: MerchantStatus::parse(&trimmed("status").unwrap_or_default()),
402            expires_min: positive("expires"),
403            time: positive("time"),
404            text,
405            attrs: text::attributes(open),
406        }));
407        unmodelled_after(unmodelled, frames);
408    }

410    /// A closing tag that is not markup.
411    ///
412    /// **Every close emits.** The only branch here is *which* frame: a close
413    /// whose name this parser does not know is [`Frame::UnknownTag`] and worth
414    /// a look; a close whose name it does know is [`Frame::Structural`] and is
415    /// not. Neither is silent: being in [`crate::tags`] must never be what
416    /// silences a tag (this file's header).
417    fn close_tag(&mut self, tag: &str, buffer: &mut String, frames: &mut Vec<Frame>) {
418        let name = text::tag_name(tag);
419        self.flush(buffer, frames);
420        // `</dialogData>` ends the dialog that encloses the `<progressBar>`s
421        // inside it, and `</openDialog>` the panel's markup.
422        if name == "dialogData" {
423            self.dialog = None;
424        }
425        if name.eq_ignore_ascii_case("openDialog") {
426            self.panel = false;
427        }
428        // `</stream>` ends the paired redirect its opener established, and it
429        // emits `StreamPop` to say so.
430        //
431        // Removing the entry and emitting `Frame::Structural` would keep the
432        // parser's own stack right while telling a frame-stream consumer
433        // nothing -- and the frame stream is the only thing a router can see.
434        // A run of self-contained `<stream id="Spells">` rows would leave the
435        // frame stream standing at that depth when the next prompt arrived,
436        // and the prompt barrier could not paper over it: `mem::take` would
437        // have nothing left to force-pop, because the closer had already
438        // removed the entries. The barrier's guarantee would hold for the
439        // parser and be empty for everyone downstream.
440        //
441        // It closes the innermost entry a PAIRED `<stream>` opened, so a stray
442        // close cannot unroute an enclosing `<pushStream>`. That distinction is
443        // what [`OpenStream`](super::OpenStream) records, and the measurement
444        // there is why: the paired form is exactly balanced on real traffic
445        // while `pushStream` is not, so a close with no paired entry open is not
446        // this stream's close. Such a close falls through and emits
447        // `Structural` -- "every close emits" is the rule above, and a close
448        // that popped nothing is not a pop.
449        // **`dynaStream` too.** The open arm above takes
450        // `"stream" | "dynaStream"` together and calls `open_stream(.., true)`
451        // for both, so both push a paired entry, and both must pop one: else
452        // `<dynaStream id='bugStream'>Details</dynaStream>Back in main` would
453        // leave "Back in main" tagged `bugStream`, and every line after it,
454        // for the life of the session.
455        //
456        // A tag that opens a routing context in one arm and is absent from
457        // its closer is a leak by construction, which is why the two lists
458        // are written to match.
459        if matches!(name, "stream" | "dynaStream")
460            && let Some(at) = self.streams.iter().rposition(|s| s.paired)
461        {
462            let open = self.streams.remove(at);
463            frames.push(Frame::StreamPop { id: Some(open.id) });
464            self.resume_enclosing(frames);
465            return;
466        }
467        // **The bytes the wire sent**, not a close tag made again from its
468        // name: rebuilt, `</a >` and a close that carried attributes would be
469        // reported as `</a>`, something the wire did not send.
470        if tags::is_known(name) {
471            frames.push(Frame::structural(name, tag));
472        } else {
473            frames.push(Frame::UnknownTag {
474                name: name.to_owned(),
475                raw: tag.to_owned(),
476            });
477        }
478    }

481/// `<progressBar>` -- the vitals handler.
482///
483/// `value=` is a **percentage**, not the current value; the numbers live in
484/// `text=`. See [`crate::numbers`] for the two traps this avoids.
485fn progress_bar(tag: &str, dialog: Option<&str>) -> Frame {
486    let text_attr = text::attribute(tag, "text").unwrap_or_default();
487    Frame::ProgressBar(ProgressBar {
488        id: text::attribute(tag, "id").unwrap_or_default(),
489        dialog: dialog.map(str::to_owned),
490        percent: text::attribute_u32(tag, "value").unwrap_or_default(),
491        amount: parse_amount(&text_attr),
492        time_remaining_secs: text::attribute(tag, "time")
493            .as_deref()
494            .and_then(parse_duration_secs),
495        text: text_attr,
496        attrs: text::attributes(tag),
497    })
498}

500/// Runs helper. Its only caller is `tests/golden_room.rs` -- no library code

505    #[must_use]
506    pub fn is_blank(&self) -> bool {
507        self.runs.iter().all(|r| r.text.trim().is_empty())
508    }

512    /// Emit [`Frame::ClearStream`] for `clearStream` / `clearDynaStream`.
513    ///
514    /// Split out with `structural_only` for the same reason: `dispatch` is a
515    /// match and clippy counts its lines, so a new arm pays for itself by
516    /// moving a body down.
517    fn clear_stream(&mut self, tag: &str, buffer: &mut String, frames: &mut Vec<Frame>) {
518        self.flush(buffer, frames);
519        frames.push(Frame::ClearStream {
520            id: text::attribute(tag, "id").unwrap_or_default(),
521            attrs: text::attributes(tag),
522        });
523    }

525    /// Emit a tag as [`Frame::Structural`] and nothing else.
526    ///
527    /// For a tag that is real, carries no payload this parser models, and must
528    /// not be guessed at -- a self-closing `<dynaStream/>` is the case that
529    /// needed it. Split out to keep `dispatch` under clippy's 100-line limit,
530    /// which the `dynaStream` arms took it past.
531    fn structural_only(
532        &mut self,
533        name: &str,
534        tag: &str,
535        buffer: &mut String,
536        frames: &mut Vec<Frame>,
537    ) {
538        self.flush(buffer, frames);
539        frames.push(Frame::structural(name, tag));
540    }

543/// The frame for a known tag found **inside a component body**.
544///
545/// [`Frame::Structural`] is the floor and stays the answer for almost
546/// everything: a tag whose whole content is "it happened" loses nothing by
547/// being reported that way, and nothing is lost because the raw form
548/// travels with it.
549///
550/// The exception is a tag that **carries data and has a frame for it**. Such
551/// a tag must not lose its frame because of where it appeared: a consumer
552/// cannot re-tokenize markup to recover what the parser already parsed.
553///
554/// # Why a list and not `thin_frame`
555///
556/// Routing every known tag through `thin_frame` would be too broad:
557/// `a_known_tag_inside_a_component_body_is_structural` shows it would turn
558/// `<nav rm='7503251'/>` into `WindowHints { id: "nav" }`. `<nav>` is handled
559/// in `dispatch`'s own match, not in `thin.rs`, so `thin_frame`'s fallback
560/// arm would answer with a default that is meaningless for a room change.
561///
562/// So the list names the tags whose `thin.rs` arm is correct in any position,
563/// and everything else keeps the floor. Adding to it means checking that the
564/// arm reads only the tag's own attributes -- which is the whole criterion.
565fn body_tag_frame(name: &str, raw: &str) -> Frame {
566    match name {
567        // `<crtrStatus exist= hostile= dead= .../>`: an id and a flag bag,
568        // read entirely from its own attributes. Nearly all of them (~99%)
569        // arrive inside a `<component>`.
570        "crtrStatus" => super::thin::thin_frame(name, raw, None),
571        _ => Frame::structural(name, raw),
572    }
573}

575/// A `<sagaImage>` placed in `stream`: its size at most
576/// [`IMAGE_MOST`](crate::frame::IMAGE_MOST) each way, a size of zero or one
577/// not a number none.
578fn inline_image(tag: &str, stream: String) -> crate::frame::InlineImage {
579    let size = |name| {
580        text::attribute_u32(tag, name)
581            .filter(|&n| n > 0)
582            .map(|n| n.min(crate::frame::IMAGE_MOST))
583    };
584    crate::frame::InlineImage {
585        stream,
586        src: text::attribute(tag, "src")
587            .map(|src| src.trim().to_owned())
588            .unwrap_or_default(),
589        alt: text::attribute(tag, "alt")
590            .map(|alt| alt.trim().to_owned())
591            .filter(|alt| !alt.is_empty()),
592        width: size("width"),
593        height: size("height"),
594        attrs: text::attributes(tag),
595    }
596}

598/// The tags a body carried that its frame does not model, each a frame of
599/// its own, pushed after the frame the body belonged to.
600fn unmodelled_after(unmodelled: Vec<String>, frames: &mut Vec<Frame>) {
601    for raw in unmodelled {
602        let name = text::tag_name(&raw).to_owned();
603        if tags::is_known(&name) {
604            // **Typed, not `Structural`, where the tag has a frame.**
605            // `Frame::structural` is the floor, and for a tag with no frame
606            // of its own it still is. But a tag that HAS a modelled frame
607            // must not lose it by virtue of where it appears.
608            //
609            // `<crtrStatus>` is why it matters: in recorded sessions
610            // nearly all of them (~99%) arrive inside a `<component>`,
611            // and only a few stand alone. Reported as `Structural`, its
612            // flags (`hostile`, `dead`, `stunned`, `prone`, `rooted`,
613            // `flying`, `inferior`, `immobile`) would be trapped in a raw
614            // string, and a consumer would have to re-tokenize markup to
615            // recover a fact the parser already had. Routing through
616            // `thin_frame` -- the same function the standalone path uses
617            // -- means one answer for one tag, wherever it sits.
618            frames.push(body_tag_frame(&name, &raw));
619        } else {
620            frames.push(Frame::UnknownTag { name, raw });
621        }
622    }
623}

625/// Assemble a `<menu>` and the `<mi>` items in its body.
626///
627/// The body is scanned for tags rather than run through `parse_runs`: a menu
628/// carries no prose, only items.
629///
630/// `coord`, `noun` and `menu_cat` are every attribute recorded so far (the
631/// census in [`MenuItem::menu_cat`]); any other is kept in `other`.
632fn menu(tag: &str) -> crate::frame::Menu {
633    let items = children(tag, "mi")
634        .iter()
635        .map(|raw| menu_item(raw))
636        .collect();
637    // The envelope's own attributes come from its open tag, not its body.
638    let tag = open_tag(tag);
639    crate::frame::Menu {
640        id: text::attribute(tag, "id").unwrap_or_default(),
641        path: text::attribute(tag, "path"),
642        categories: text::attribute(tag, "cat_list")
643            .map(|list| list.split_whitespace().map(str::to_owned).collect())
644            .unwrap_or_default(),
645        items,
646    }
647}

649/// A tag whose body is children, assembled into one frame.
650impl Parser {
651    /// Tags whose body is not prose on this line.
652    ///
653    /// Three carry a run of CHILDREN, always on one line, and are assembled
654    /// whole so the rows stay attached to the envelope that says what they
655    /// are. The fourth opens a capture that owns every following line until
656    /// its close -- its body is prose in `<result>` sections, 7 to 50 lines
657    /// of it (`view_item.rs`).
658    fn assembled(&mut self, name: &str, tag: &str, frames: &mut Vec<Frame>) {
659        let named: &[&str] = match name {
660            "menu" => {
661                frames.push(Frame::MenuResponse(menu(tag)));
662                &["mi"]
663            }
664            "cmdlist" => {
665                frames.push(cmdlist(tag));
666                &["cli"]
667            }
668            "inventoryManager" => {
669                let (manager, unplaced) = super::inventory::inventory_manager(tag);
670                frames.push(manager);
671                // A row the tree cannot place is reported as it came.
672                frames.extend(unplaced.iter().map(|raw| Frame::structural("i", raw)));
673                &["i", "continuation"]
674            }
675            "inventoryViewItem" => return self.open_view_item(tag, frames),
676            _ => {
677                frames.push(objectives(tag));
678                // A deleted group's `<group>`s are its rows, read above.
679                if text::attribute(open_tag(tag), "action").as_deref() == Some("delete-group") {
680                    &["objective", "group"]
681                } else {
682                    &["objective"]
683                }
684            }
685        };
686        // Every other tag in the body is reported as any tag is: a child
687        // the wire grows is seen, not dropped. What is inside an objective
688        // is kept on it instead, with whose it is.
689        let holder = (name == "objectives").then_some("objective");
690        for (child, raw) in super::inner::other_children(tag, named, holder) {
691            if tags::is_known(&child) {
692                frames.push(Frame::structural(&child, &raw));
693            } else {
694                frames.push(Frame::UnknownTag { name: child, raw });
695            }
696        }
697    }
698}

700/// Assemble an `<objectives>` and the `<objective>` rows in its body.
701/// One `<mi>` row.
702pub(super) fn menu_item(tag: &str) -> crate::frame::MenuItem {
703    crate::frame::MenuItem {
704        coord: text::attribute(tag, "coord"),
705        noun: text::attribute(tag, "noun"),
706        menu_cat: text::attribute(tag, "menu_cat"),
707        other: super::inner::unread(tag, &["coord", "noun", "menu_cat"]),
708    }
709}

711fn objectives(tag: &str) -> Frame {
712    let action = crate::frame::ObjectivesAction::parse(
713        &text::attribute(open_tag(tag), "action").unwrap_or_default(),
714    );
715    // A group deleted names its rows by `<group>`, read as the rows they are.
716    if action == crate::frame::ObjectivesAction::DeleteGroup {
717        return Frame::ObjectivesUpdate {
718            action,
719            entries: children(tag, "group")
720                .iter()
721                .map(|row| objective(row))
722                .collect(),
723        };
724    }
725    let mut inside = super::inner::inside_each(tag, "objective").into_iter();
726    let entries = children(tag, "objective")
727        .iter()
728        .map(|row| {
729            let mut objective = objective(row);
730            for tag in inside.next().unwrap_or_default() {
731                match text::tag_name(&tag) {
732                    "reward" => objective.rewards.push(crate::frame::ObjectiveReward {
733                        kind: text::attribute(&tag, "type").unwrap_or_default(),
734                        amount: text::attribute(&tag, "amount").and_then(|n| n.trim().parse().ok()),
735                        other: super::inner::unread(&tag, &["type", "amount"]),
736                    }),
737                    "action" => objective.actions.push(crate::frame::ObjectiveAction {
738                        kind: text::attribute(&tag, "type").unwrap_or_default(),
739                        command: text::attribute(&tag, "cmd").unwrap_or_default(),
740                        other: super::inner::unread(&tag, &["type", "cmd"]),
741                    }),
742                    _ => objective.children.push(tag),
743                }
744            }
745            objective
746        })
747        .collect();
748    Frame::ObjectivesUpdate { action, entries }
749}

751/// One `<objective>` row. Every attribute the wire sends is typed; see
752/// [`crate::frame::Objective`] for which are guaranteed.
753pub(super) fn objective(tag: &str) -> crate::frame::Objective {
754    crate::frame::Objective {
755        id: text::attribute(tag, "id").unwrap_or_default(),
756        kind: text::attribute(tag, "type").unwrap_or_default(),
757        state: text::attribute(tag, "state"),
758        name: text::attribute(tag, "name"),
759        description: text::attribute(tag, "description"),
760        location: text::attribute(tag, "location"),
761        cadence: text::attribute(tag, "cadence"),
762        expires: text::attribute(tag, "expires"),
763        rewards: Vec::new(),
764        actions: Vec::new(),
765        children: Vec::new(),
766        other: super::inner::unread(
767            tag,
768            &[
769                "id",
770                "type",
771                "state",
772                "name",
773                "description",
774                "location",
775                "cadence",
776                "expires",
777            ],
778        ),
779    }
780}

782/// Assemble a `<cmdlist>` and the `<cli>` dictionary rows in its body.
783///
784/// See [`crate::frame::CmdListUpdate`] for what the push is and why the
785/// client's `cmdlist1.xml` is a cache of it rather than shipped data.
786fn cmdlist(tag: &str) -> Frame {
787    let entries = children(tag, "cli")
788        .iter()
789        .map(|row| crate::frame::CmdListEntry {
790            coord: text::attribute(row, "coord").unwrap_or_default(),
791            label: text::attribute(row, "menu").unwrap_or_default(),
792            command: text::attribute(row, "command").unwrap_or_default(),
793            category: text::attribute(row, "menu_cat").unwrap_or_default(),
794            other: super::inner::unread(row, &["coord", "menu", "command", "menu_cat"]),
795        })
796        .collect();
797    Frame::CmdListUpdate(crate::frame::CmdListUpdate { entries })
798}

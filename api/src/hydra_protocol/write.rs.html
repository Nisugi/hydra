1//! Markup written back: tags as the game would send them, from what was read
2//! of them.
3//!
4//! The one parser reads the game's markup into frames; this writes the few
5//! tags a reader of the game's stream needs to be told again. Its reader is
6//! the player's own Lich started mid-session, which Hydra tells what it
7//! missed with a login built from its model, as of the moment it attaches.
8//! What a login consists of is known, so it can be built in the moment from
9//! accurate state.
10//!
11//! Here, with the parser, because the markup is this crate's: nothing above
12//! it spells a tag. What is written is read back by the parser to what it
13//! was written from (the tests below, and the model's own).
14

18/// `text` with the XML entities escaped: all five, since Lich's attribute
19/// reader decodes all five.
20#[must_use]
21pub fn escape(text: &str) -> String {
22    let mut escaped = String::with_capacity(text.len());
23    for c in text.chars() {
24        match c {
25            '&' => escaped.push_str("&amp;"),
26            '<' => escaped.push_str("&lt;"),
27            '>' => escaped.push_str("&gt;"),
28            '"' => escaped.push_str("&quot;"),
29            '\'' => escaped.push_str("&apos;"),
30            c => escaped.push(c),
31        }
32    }
33    escaped
34}

36/// [`escape`], for an attribute's value: a line break or a tab in it is
37/// a numeric entity too, as the wire sends one, since a tag never spans a
38/// line. Written raw, the tag would split, and read back as broken.
39#[must_use]
40pub fn escape_attribute(value: &str) -> String {
41    let mut escaped = String::with_capacity(value.len());
42    for c in escape(value).chars() {
43        match c {
44            '\n' => escaped.push_str("&#10;"),
45            '\r' => escaped.push_str("&#13;"),
46            '\t' => escaped.push_str("&#9;"),
47            c => escaped.push(c),
48        }
49    }
50    escaped
51}

53/// `<name a="v" .../>`.
54#[must_use]
55pub fn empty(name: &str, attrs: &[(&str, &str)]) -> String {
56    format!("<{name}{}/>", attributes(attrs))
57}
58
59/// `<name a="v" ...>body</name>`, `body` being markup already.
60#[must_use]
61pub fn element(name: &str, attrs: &[(&str, &str)], body: &str) -> String {
62    format!("<{name}{}>{body}</{name}>", attributes(attrs))
63}
64
65fn attributes(attrs: &[(&str, &str)]) -> String {
66    let mut written = String::new();
67    for (name, value) in attrs {
68        written.push(' ');
69        written.push_str(name);
70        written.push_str("=\"");
71        written.push_str(&escape_attribute(value));
72        written.push('"');
73    }
74    written
75}

77/// `runs` as markup the parser reads back to them: their bold, preset and
78/// mono regions, and their links, an object's inside a command's where the
79/// wire nests them.
80#[must_use]
81pub fn runs(runs: &Runs) -> String {
82    let mut writing = Writing::default();
83    for run in &runs.runs {
84        writing.run(run);
85    }
86    writing.finish()
87}
88
89/// What is open while runs are written.
90#[derive(Default)]
91struct Writing {
92    written: String,
93    bold: u16,
94    mono: bool,
95    preset: Option<String>,
96    link: Option<Link>,
97    inner: Option<Link>,
98}
99
100impl Writing {
101    fn run(&mut self, run: &Run) {
102        if self.preset != run.style.preset {
103            self.close_links();
104            if self.preset.take().is_some() {
105                self.written.push_str("</preset>");
106            }
107            if let Some(preset) = &run.style.preset {
108                self.written.push_str(&opening("preset", &[("id", preset)]));
109                self.preset = Some(preset.clone());
110            }
111        }
112        if !same(self.link.as_ref(), run.link.as_ref()) {
113            self.close_links();
114            if let Some(link) = &run.link {
115                self.written.push_str(&open(link));
116                self.link = Some(link.clone());
117            }
118        }
119        if !same(self.inner.as_ref(), run.inner_link.as_ref()) {
120            if self.inner.take().is_some() {
121                self.written.push_str("</a>");
122            }
123            if let Some(inner) = &run.inner_link {
124                self.written.push_str(&open(inner));
125                self.inner = Some(inner.clone());
126            }
127        }
128        while self.bold < run.style.bold_depth {
129            self.written.push_str("<pushBold/>");
130            self.bold += 1;
131        }
132        while self.bold > run.style.bold_depth {
133            self.written.push_str("<popBold/>");
134            self.bold -= 1;
135        }
136        if self.mono != run.style.mono {
137            self.written.push_str(if run.style.mono {
138                "<output class=\"mono\"/>"
139            } else {
140                "<output class=\"\"/>"
141            });
142            self.mono = run.style.mono;
143        }
144        self.written.push_str(&escape(&run.text));
145    }
146
147    fn close_links(&mut self) {
148        if self.inner.take().is_some() {
149            self.written.push_str("</a>");
150        }
151        if let Some(link) = self.link.take() {
152            self.written.push_str(close(&link));
153        }
154    }
155
156    fn finish(mut self) -> String {
157        self.close_links();
158        while self.bold > 0 {
159            self.written.push_str("<popBold/>");
160            self.bold -= 1;
161        }
162        if self.mono {
163            self.written.push_str("<output class=\"\"/>");
164        }
165        if self.preset.is_some() {
166            self.written.push_str("</preset>");
167        }
168        self.written
169    }
170}

172/// Whether two runs' links are one element: the same kind and coordinate.
173/// Not their text, which the parser gives each run as the link's text so
174/// far.
175fn same(one: Option<&Link>, other: Option<&Link>) -> bool {
176    match (one, other) {
177        (Some(one), Some(other)) => one.kind == other.kind && one.coord == other.coord,
178        (None, None) => true,
179        _ => false,
180    }
181}
182
183/// The opening tag of `link`, as the parser reads it back
184/// (`text::link_from_tag`, `parser/markup.rs`'s `open_link`).
185fn open(link: &Link) -> String {
186    let (name, mut attrs): (&str, Vec<(&str, &str)>) = match &link.kind {
187        LinkKind::Exist { id, noun } => ("a", vec![("exist", id), ("noun", noun)]),
188        LinkKind::Direct { cmd } => ("d", vec![("cmd", cmd)]),
189        LinkKind::DirectText => ("d", Vec::new()),
190        LinkKind::Url { href } => ("a", vec![("href", href)]),
191        LinkKind::NotActionable => ("a", Vec::new()),
192    };
193    if let Some(coord) = &link.coord {
194        attrs.push(("coord", coord));
195    }
196    opening(name, &attrs)
197}
198
199fn opening(name: &str, attrs: &[(&str, &str)]) -> String {
200    format!("<{name}{}>", attributes(attrs))
201}
202
203fn close(link: &Link) -> &'static str {
204    match link.kind {
205        LinkKind::Direct { .. } | LinkKind::DirectText => "</d>",
206        LinkKind::Exist { .. } | LinkKind::Url { .. } | LinkKind::NotActionable => "</a>",
207    }
208}

216    /// Every component body in `markup`, parsed.
217    fn bodies(markup: &str) -> Vec<(String, Runs)> {
218        Parser::new()
219            .push_bytes(markup.as_bytes())
220            .into_iter()
221            .filter_map(|frame| match frame {
222                Frame::Component { id, body } => Some((id, joined(body))),
223                _ => None,
224            })
225            .collect()
226    }
227
228    /// `body` with neighbouring runs of one style and link joined: a tag
229    /// that changed nothing splits a run on the wire, and means nothing to
230    /// a reader.
231    fn joined(body: Runs) -> Runs {
232        let mut runs: Vec<Run> = Vec::new();
233        for run in body.runs {
234            match runs.last_mut() {
235                Some(last)
236                    if last.style == run.style
237                        && last.link == run.link
238                        && last.inner_link == run.inner_link =>
239                {
240                    last.text.push_str(&run.text);
241                }
242                _ => runs.push(run),
243            }
244        }
245        Runs { runs }
246    }

248    /// What the parser read of a room -- objects, bold creatures, exits,
249    /// a command link with an object inside it -- is written back to markup
250    /// it reads to the same runs.
251    #[test]
252    fn a_room_is_read_back_as_it_was_read() {
253        let room = concat!(
254            include_str!("../tests/fixtures/room.xml"),
255            include_str!("../tests/fixtures/room_populated.xml"),
256            include_str!("../tests/fixtures/login_setup.xml"),
257            "<component id='room objs'>You also see <pushBold/>a <a exist=\"1\" noun=\"kobold\">kobold</a><popBold/>, ",
258            "<d cmd=\"store WEAPON clear\">a <a exist=\"2\" noun=\"katar\">bent katar</a></d> &amp; <preset id=\"speech\">a &lt;sign&gt;</preset>.</component>\n",
259        );
260        let read = bodies(room);
261        assert!(read.len() > 8, "the fixtures' components: {}", read.len());
262        for (id, body) in read {
263            let written = format!("<component id='{id}'>{}</component>\n", runs(&body));
264            let again = bodies(&written);
265            assert_eq!(again, [(id, body)], "{written}");
266        }
267    }

269    /// An attribute value with a line break in it is one tag on one line,
270    /// and reads back whole.
271    #[test]
272    fn a_line_break_in_an_attribute_stays_inside_the_tag() {
273        let tag = empty("label", &[("value", "two\nlines\there")]);
274        assert_eq!(tag, "<label value=\"two&#10;lines&#9;here\"/>");
275        assert!(!tag.contains('\n'));
276        let mut parser = crate::Parser::new();
277        let frames = parser.parse_line(&tag);
278        assert_eq!(frames.len(), 1, "{frames:?}");
279        assert!(
280            !frames
281                .iter()
282                .any(|f| matches!(f, crate::Frame::MalformedTag { .. })),
283            "{frames:?}"
284        );
285    }

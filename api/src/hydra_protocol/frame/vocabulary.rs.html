12/// One thing the game said.
13///
14/// [`Frame::UnknownTag`] and [`Frame::MalformedTag`] exist, and neither is
15/// optional, because Simutronics changes the protocol without notice: an
16/// unmodelled tag must reach the user as text and a log, never panicking and
17/// never silently dropped.
18#[derive(Clone, Debug, PartialEq, Eq)]
19#[non_exhaustive]
20pub enum Frame {
21    // --- text and framing -------------------------------------------------
22    /// A run of display text with the markup that was open around it.
23    Text(TextFrame),
24    /// `<prompt time=>`. The resync barrier; see [`crate::parser`].
25    Prompt {
26        /// `time=` as a string (epoch seconds on the wire); empty when absent.
27        time: String,
28        /// The body, e.g. `>` or `HR>`: entity-decoded, control characters
29        /// stripped.
30        text: String,
31    },
32    /// `<spell>` -- the currently prepared spell. A spell prepared comes
33    /// as an object, `<spell exist='spell'>Tangleweed</spell>`, and none as
34    /// `<spell>None</spell>`. In recorded sessions every one of the first kind
35    /// carries `exist='spell'`, and none of the second carries anything.
36    Spell {
37        /// The body as display text: nested markup removed, entity-decoded,
38        /// control characters stripped.
39        text: String,
40        /// As a hand's: from `exist=`, so the spell hand is a link, `#spell`.
41        link: Option<Link>,
42    },
43    /// `<left exist= noun=>` -- left hand contents.
44    LeftHand {
45        /// The body as display text, as `Spell::text` is.
46        item: String,
47        /// From `href=`, else `exist=`/`noun=`, else `cmd=`; `None` when the
48        /// tag carries none of them.
49        link: Option<Link>,
50    },
51    /// `<right exist= noun=>` -- right hand contents.
52    RightHand {
53        /// The body as display text, as `Spell::text` is.
54        item: String,
55        /// From `href=`, else `exist=`/`noun=`, else `cmd=`; `None` when the
56        /// tag carries none of them.
57        link: Option<Link>,
58    },
59
60    // --- timers -----------------------------------------------------------
61    /// `<roundTime value=>` -- absolute epoch second the roundtime ends.
62    RoundTime {
63        /// `value=`, epoch seconds. A tag whose value is absent or not a
64        /// number is no timer of zero: it is [`Frame::MalformedTag`]
65        /// (`parser/thin.rs`).
66        value: u32,
67    },
68    /// `<castTime value=>`.
69    CastTime {
70        /// `value=`: the epoch second the cast time ends, not a duration.
71        /// Absent or not a number, the tag is
72        /// [`Frame::MalformedTag`], as a roundtime's is.
73        value: u32,
74    },
75    /// `<timer>`, read as the aim timer. The parser does not check `id=`
76    /// (`parser/thin.rs`, the `"timer"` arm): every `<timer>` lands here, and
77    /// `aimTimer` is the only id this was written for.
78    AimTime {
79        /// `value=` as a whole number. Absent or not a number, the tag is
80        /// [`Frame::MalformedTag`], as a roundtime's is.
81        value: u32,
82    },
83
84    /// `<vellumTimer id= value=>`: a timer of the player's own, as a Lich
85    /// script sets one -- the game never sends it.
86    Timer {
87        /// `id=`: the timer's name, never empty.
88        id: String,
89        /// `value=`: the epoch second it ends, as a roundtime's; 0 clears
90        /// it. A tag with no name, or no number here, is
91        /// [`Frame::MalformedTag`].
92        value: u32,
93        /// `category=`: the list it shows in, as the player names it --
94        /// `buff`, `debuff`, `active spell`, `cooldown`, `custom`; none,
95        /// the player's own.
96        category: Option<String>,
97    },
98
99    // --- room and vitals --------------------------------------------------
100    /// `<resource picture=>`.
101    RoomPicture {
102        /// `picture=`. Only emitted when it parses as a number; a `<resource>`
103        /// without one becomes `Structural`.
104        id: u32,
105    },
106    /// `<progressBar>`, bare or inside `<dialogData>`.
107    ProgressBar(ProgressBar),
108    /// `<label id= value=>`, and **which dialog enclosed it**.
109    ///
110    /// The dialog is load-bearing for the same reason it is on
111    /// [`ProgressBar`] and
112    /// [`Self::InjuryImage`]: `<label id='yourLvl'>` means a character level
113    /// inside `expr` and a map legend inside `mapViewMain`, and the id alone
114    /// cannot tell them apart.
115    Label {
116        /// `id=`.
117        id: String,
118        /// `value=`.
119        value: String,
120        /// The enclosing `<dialogData id=>`, if any.
121        dialog: Option<String>,
122        /// Every attribute, in wire order. `<label>` carries its layout --
123        /// `top`/`left`/`width`/`height`/`align` in `character_info.xml` --
124        /// beyond `id` and `value`, and none of it is dropped.
125        attrs: Attrs,
126    },
127    /// `<compass><dir value=>` -- the obvious exits, as direction tokens.
128    Compass {
129        /// Each `<dir value=>`, entity-decoded, in wire order; a `<dir>`
130        /// without `value=` is skipped.
131        directions: Vec<String>,
132    },
133    /// `<component id=>` / `<compDef id=>` -- a named slice of the room.
134    ///
135    /// **`body` is parsed, not raw.** Storing the inner XML string verbatim
136    /// would hand markup to the layer above and make it re-parse, which is
137    /// what this crate exists to prevent. [`Runs`] carries it parsed.
138    Component {
139        /// `id=`, e.g. `room objs`; empty when absent.
140        id: String,
141        /// The tag's body, parsed into styled runs.
142        body: Runs,
143    },
144    /// `<pushStream id=>`.
145    StreamPush {
146        /// `id=` of the `<pushStream>`, or of a paired `<stream>` /
147        /// `<dynaStream>`; empty when absent.
148        id: String,
149    },
150    /// `<crtrStatus exist= ...>`; `attrs` raw so the layer above owns the
151    /// flag-name mapping.
152    CreatureStatus {
153        /// The creature's `exist=` id (there is no `id=`); empty when absent.
154        id: String,
155        /// Every attribute, in wire order, `exist` included.
156        attrs: Attrs,
157    },
158    /// `<roommeta .../>`: the room's environment codes.
159    RoomMeta(RoomMeta),
160
161    // --- stream stack -----------------------------------------------------
162    /// `<popStream/>`, bare or `<popStream id=>`.
163    StreamPop {
164        /// The stream closed: `id=` of a `<popStream>`, or the stream a
165        /// `</stream>` / `</dynaStream>` ended. `None` for a bare
166        /// `<popStream/>`, which closes whichever is current.
167        id: Option<String>,
168    },
169    /// Synthesized when a `<prompt>` force-closes a stream nobody popped.
170    StreamPopForced {
171        /// The stream that was still open; emitted innermost first, before the
172        /// `Prompt`.
173        id: String,
174    },
175    /// Synthesized by the stream stack when an inner stream ends.
176    StreamResume {
177        /// The enclosing stream, current again. Not emitted when the stack
178        /// empties: that is the main window, which `StreamPop` already implies.
179        id: String,
180    },
181    /// `<clearStream id=>`.
182    ClearStream {
183        /// `id=` of the `<clearStream>` or `<clearDynaStream>`; empty when
184        /// absent.
185        id: String,
186        /// Every attribute the tag carried, `id=` included: some carry
187        /// `ifClosed=`, where the clearing goes when the window is shut.
188        attrs: Attrs,
189    },
190    /// `<clearDialogData>`, or an empty `<dialogData>`.
191    ClearDialogData {
192        /// `id=`: the dialog to clear; empty when absent.
193        id: String,
194    },
195    /// `<closeDialog>` / `<closedialog>`.
196    CloseDialog {
197        /// `id=`: the dialog to close; empty when absent.
198        id: String,
199    },
200    /// `<exposeDialog>` / `<exposeStream>` / `<exposeContainer>`.
201    Expose {
202        /// The tag name, spelled as received: which of the three it was.
203        kind: String,
204        /// `id=`: the dialog, stream or container to expose; empty when absent.
205        id: String,
206    },
207    /// `<deleteContainer id=>`.
208    DeleteContainer {
209        /// `id=`: the container to delete; empty when absent.
210        id: String,
211    },
212    /// Placement attrs riding a `<streamWindow>`/`<openDialog>`/`<container>`.
213    WindowHints {
214        /// `id=`, or the tag name when absent. Always the tag name for
215        /// client-configuration tags such as `mode` and `settingsInfo`.
216        id: String,
217        /// Every attribute, in wire order.
218        attrs: Attrs,
219    },
220    /// `<playerID id=>`: the game's number for this character, sent once in
221    /// every login burst.
222    ///
223    /// Typed for [`Frame::EndSetup`]'s reason: a consumer keys on it (the
224    /// model's "is that link me?", for the group's leader), and matching a
225    /// `WindowHints` bag by its id string is re-reading markup. `id` is the
226    /// tag's one attribute, a number, so nothing the bag carried is lost.
227    PlayerId {
228        /// `id=`, verbatim; empty when absent.
229        id: String,
230    },
231    /// `<settingsInfo client= major= crc= instance=>`: which of the game's
232    /// instances this connection plays, by the game's own code, and the
233    /// client settings it expects. Sent once in every login burst
234    /// (`tests/fixtures/login_setup.xml`).
235    ///
236    /// Typed for [`Frame::PlayerId`]'s reason: a consumer keys on it. The
237    /// model keeps `instance`, which is how Lich knows which game it is in,
238    /// for a Lich handed the login late; and matching a `WindowHints` bag by
239    /// its id string is re-reading markup. The other attributes ride in
240    /// `attrs`, so nothing the bag carried is lost.
241    SettingsInfo {
242        /// `instance=`, verbatim; `None` when absent.
243        instance: Option<String>,
244        /// Every attribute, in wire order.
245        attrs: Attrs,
246    },
247    /// `<endSetup/>`: the login setup is over. The session keys `Ready` on the
248    /// first `<prompt>` after it. Typed rather than left in a `WindowHints`
249    /// bag, which a consumer could only recognise by string; unit because the
250    /// tag has no attributes, and every one seen in recorded sessions is bare.
251    EndSetup,
252    /// `<app char= game= title=>` -- who this connection is, and where.
253    ///
254    /// **`game` is the INSTANCE**, and a multi-session client needs it. The
255    /// wire sends `<app char="Alderin" game="Prime" title="GemStone IV:
256    /// Alderin [Prime]"/>` (as in `tests/fixtures/login_setup.xml`), and
257    /// Prime, Platinum, Shattered and Test are separate worlds: the same
258    /// character name in two of them is two different characters.
259    ///
260    /// `title` is kept rather than derived: it is what the game says to put in
261    /// a window title bar, and reassembling it from the other two would be
262    /// guessing at a format the server already sent.
263    AppInfo {
264        /// `char=`: the character's name; empty when absent.
265        character: String,
266        /// The instance: `Prime`, `Platinum`, `Shattered`, `Test`.
267        game: String,
268        /// The window title the game suggests, verbatim.
269        title: String,
270    },
271    /// `<nav rm=>` -- the room changed. **`None` for a bare `<nav/>`**: a real
272    /// shape meaning "arrived, and this room has no UID" (`DragonRealms`
273    /// sends it). An empty string here could not be told from a UID.
274    RoomId {
275        /// `rm=`, entity-decoded; `None` for a bare `<nav/>`.
276        id: Option<String>,
277    },
278    /// `<streamWindow id= title= subtitle= ...>`.
279    ///
280    /// `id`, `title` and `subtitle` are lifted because windows render them.
281    /// Everything else rides in [`attrs`](Self::StreamWindow::attrs) rather
282    /// than being dropped: a census of a 1,547-file corpus sample found
283    /// 1,176,686 `<streamWindow>` carrying **15 distinct attribute names**,
284    /// of which only three were kept. `location` (1,169,402) and `target`
285    /// (1,163,813) are on ~99% of them, and `ifClosed` (602,513),
286    /// `resident` (605,219) and `styleIfClosed` (1,353) are all live.
287    StreamWindow {
288        /// `id=`: the stream this window shows; empty when absent.
289        id: String,
290        /// `title=`; `None` when absent.
291        title: Option<String>,
292        /// `subtitle=`; `None` when absent.
293        subtitle: Option<String>,
294        /// Every attribute the tag carried, including the three above.
295        attrs: Attrs,
296    },
297
298    // --- status -----------------------------------------------------------
299    /// `<image id= name=>`, and **which dialog enclosed it**.
300    ///
301    /// The name says injuries because that is the one use worth modelling, but
302    /// the tag is not exclusive to them: MEASURED over 24 files, 1,313 of 2,155
303    /// `<image>` tags are `nomap.jpg` map tiles and ~50 are toolbar buttons.
304    /// `dialog` is what separates them, exactly as it does for
305    /// [`ProgressBar`] -- the same tag shape meaning
306    /// different things depending on the dialog that encloses it.
307    ///
308    InjuryImage {
309        /// `id=`. The body part, when `dialog` is `injuries`.
310        id: String,
311        /// `name=`. The severity (`Injury1`..`Scar3`), or the part's own name
312        /// when it is unhurt.
313        name: String,
314        /// The enclosing `<dialogData id=>`, if any.
315        dialog: Option<String>,
316        /// Every attribute, in wire order. Not only layout: the toolbar
317        /// `<image>`s in `inventory_container.xml` carry `cmd=`, `echo=` and
318        /// `tooltip=` -- a button's whole behaviour -- beyond `id` and `name`.
319        attrs: Attrs,
320    },
321    /// `<indicator id= visible=>`.
322    StatusIndicator {
323        /// `id=` verbatim, `Icon` prefix kept (e.g. `IconSTUNNED`); empty when
324        /// absent.
325        id: String,
326        /// True only when `visible='y'`; any other value, or none, is false.
327        active: bool,
328    },
329    /// A row of `ActiveSpells` / `Buffs` / `Debuffs` / `Cooldowns`.
330    ActiveEffect(ActiveEffect),
331    /// `<objectives action=>` and the `<objective>` rows it carries.
332    ///
333    /// One frame, like [`Frame::MenuResponse`] and for the same reason: as
334    /// separate frames the rows would land in [`Frame::WindowHints`] -- the
335    /// placement-attrs bag, which is not what a quest is -- apart from the
336    /// action. An update that says `delete-objective` is meaningless apart
337    /// from the rows it deletes.
338    ObjectivesUpdate {
339        /// What to do with `entries`.
340        action: ObjectivesAction,
341        /// The rows, in wire order.
342        entries: Vec<Objective>,
343    },
344    /// A context menu the game built for one object: `<menu>` and its
345    /// `<mi>` items, as **one** frame.
346    ///
347    /// As separate frames the items would carry an empty `id`, orphaned from
348    /// the menu they answer, and no consumer could tell which request they
349    /// belonged to. A menu opens and closes on one line, even one of 60
350    /// items, so the envelope is assembled here, as `<component>` is.
351    MenuResponse(Menu),
352    /// `<cmdlist>`: dictionary rows the server is teaching us.
353    ///
354    /// One frame carrying its `<cli>` children, for the same reason
355    /// [`Frame::MenuResponse`] is: the rows are the payload, and as separate
356    /// `WindowHints` bags nothing would join them.
357    CmdListUpdate(CmdListUpdate),
358    /// `<cmdtimestamp data=>`: the dictionary version the server just sent.
359    ///
360    /// Separate from [`Frame::CmdListUpdate`] because the wire sends it as a
361    /// separate tag, and because it is meaningful alone: a timestamp with no
362    /// rows says the client is already current.
363    CmdTimestamp {
364        /// `data=`, as sent; empty when absent.
365        version: String,
366    },
367

369    /// `<switchQuickBar id=>`.
370    QuickbarSwitch {
371        /// `id=`: the quickbar to show; empty when absent.
372        id: String,
373    },
374    /// `<openDialog>` / `<opendialog>`.
375    DialogOpen {
376        /// `id=`: the dialog's name; empty when absent.
377        id: String,
378        /// `title=`; always `None` when the frame came from `<dialogData>`.
379        title: Option<String>,
380        /// Every attribute, in wire order.
381        attrs: Attrs,
382    },
383    /// A resident `<openDialog>` panel.
384    DialogPanelOpen {
385        /// `id=`: the panel's name; empty when absent.
386        id: String,
387        /// `title=`; `None` when absent.
388        title: Option<String>,
389        /// Every attribute, in wire order, `resident='true'` among them.
390        attrs: Attrs,
391    },
392    /// Widgets inside a dialog, as raw attribute bags.
393    ///
394    /// Spinboxes, skins and the like are *toolkit* concepts; the wire fact is
395    /// only "the game sent an `<upDownEditBox>` with these attributes". One
396    /// variant keyed by [`DialogWidgets::kind`] keeps widget vocabulary out
397    /// of this crate: every widget tag is the same shape, so it is one
398    /// variant.
399    DialogWidgets(DialogWidgets),
400
401    // --- containers and inventory -----------------------------------------
402    /// `<container id= title= target=>`.
403    ///
404    /// Everything else rides in [`attrs`](Self::Container::attrs), as a
405    /// stream window's does: the committed fixtures' containers carry
406    /// `location=`, `save=` and `resident=` too. `location` is where a
407    /// frontend puts the container's window.
408    Container {
409        /// `id=`; empty when absent.
410        id: String,
411        /// `title=`; `None` when absent.
412        title: Option<String>,
413        /// `target=`, entity-decoded; `None` when absent.
414        target: Option<String>,
415        /// Every attribute the tag carried, including the three above.
416        attrs: Attrs,
417    },
418    /// `<clearContainer id=>`.
419    ClearContainer {
420        /// `id=`: the container to empty; empty when absent.
421        id: String,
422    },
423    /// `<inv id=>` -- one item in a container.
424    ContainerItem {
425        /// `id=` of the `<inv>`: the container the item is in; empty when
426        /// absent.
427        container_id: String,
428        /// The tag's body, parsed into styled runs.
429        content: Runs,
430    },
431    /// `<inventoryManager>`: a whole-inventory snapshot, assembled.
432    ///
433    /// One frame, like [`Frame::MenuResponse`] and
434    /// [`Frame::ObjectivesUpdate`]: a snapshot is only a snapshot whole. See
435    /// [`InventoryResponse`] for the fields.
436    InventoryManager(InventoryResponse),
437    /// `<inventoryViewItem>`: one item's detail, assembled across lines.
438    ///
439    /// The parser's only multi-line capture; `parser/view_item.rs` says

441    /// [`ItemView`] for the fields.
442    InventoryViewItem(ItemView),

444    // --- misc -------------------------------------------------------------
445    /// `<launchURL>` / `<LaunchURL>`. The scheme allowlist is the UI's job.
446    LaunchUrl {
447        /// `src=`, else `url=`, entity-decoded; empty when neither is present.
448        url: String,
449    },
450    /// `<pulse min= max= mana=>`.
451    Pulse {
452        /// True only when `mana="1"`; the wire sends `0` about as often.
453        mana: bool,
454        /// `min=` in seconds, `46` when absent -- which it always is on the
455        /// wire seen so far.
456        min: u32,
457        /// `max=` in seconds, `75` when absent, as `min` is.
458        max: u32,
459    },
460    /// `<worldEvent realm= expires=>`.
461    WorldEvent {
462        /// `realm=`, entity-decoded; empty when absent.
463        realm: String,
464        /// `expires=` as a whole number; `None` when absent or not one.
465        expires_min: Option<u32>,
466        /// The tag's body as display text, nested markup removed.
467        text: String,
468    },
469    /// `<PantheonStatus value=>`.
470    PantheonStatus {
471        /// `value=` as a whole number; `0` when absent or not a number.
472        value: u32,
473    },
474    /// The player's own typed command, echoed back in the log.
475    ///
476    /// It arrives as `<!-- CLIENT --><c>;go2 3609<!-- ENDCLIENT -->`, and it
477    /// is how a replay knows what the player did, which for a client that
478    /// replays its own logs as test fixtures is the difference between a
479    /// reproducible session and a one-sided transcript.
480    ClientCommand {
481        /// The text after `<c>` in the region, e.g. `;go2 3609`:
482        /// entity-decoded, control characters stripped.
483        command: String,
484    },
485    /// The login `<settings>` blob arrived and was consumed whole.
486    ///
487    /// A client-configuration document -- window layout, palettes, highlight
488    /// strings, macros -- not game protocol. It carries dozens of element
489    /// names of its own (`h`, `dc`, `cmdline`, `ignores`, `panels`,
490    /// `toggles`, ...), all on one line.
491    ///
492    /// The frame records that the blob arrived, because a login is worth
493    /// seeing in a replay; its contents are not modelled here.
494    ClientSettings,
495
496    // --- the escape hatches, both mandatory -------------------------------
497    /// A tag this parser does not model, carried whole.
498    ///
499    /// `raw` is the original bytes, so the tag reaches the user as text and a
500    /// diagnosing reader sees exactly what the game sent. It is the one place
501    /// raw markup crosses the vocabulary boundary.
502    ///
503    /// It is a type of its own, not `Text`, so a consumer can tell "the game
504    /// said something new" from "the game said hello".
505    UnknownTag {
506        /// The tag name, as spelled on the wire.
507        name: String,
508        /// The tag as received; a close tag is rebuilt as `</name>`.
509        raw: String,
510    },
511    /// A tag that arrived and whose effect is not carried by any
512    /// neighbouring frame's fields.
513    ///
514    /// Nothing that comes in is dropped: a known close form or self-closing
515    /// form with nothing to model emits this, so every consumer can see it.
516    ///
517    /// Distinct from [`Frame::UnknownTag`] on purpose: an `UnknownTag` means
518    /// the protocol changed and someone should look, a `Structural` means the
519    /// protocol did exactly what it always does and there is nothing to model.
520    /// Collapsing them would cry wolf on every `</a>`.
521    ///
522    /// `raw` is the original bytes, so nothing is unrecoverable. Use
523    /// [`Frame::is_structural`] to filter these out; most consumers want to.
524    Structural {
525        /// The tag name, e.g. `a` for a `</a>`.
526        name: String,
527        /// The tag as received; a close tag is rebuilt as `</name>`.
528        raw: String,
529    },
530    /// A tag that cannot be read: one that never closed before the line
531    /// ended, or a timer whose value is not a number.
532    ///
533    /// Treating the unclosed fragment as text would silently desync the
534    /// stream. Typed here so the case is visible, testable, and cannot be
535    /// mistaken for prose. A timer read as zero instead would end a roundtime
536    /// the game never ended.
537    MalformedTag {
538        /// The tag as received: the unterminated text through the end of the
539        /// line, or the whole timer tag.
540        raw: String,
541    },

lookout.toml in the data folder: the player’s choices, for every game.

occasions = 5          # occasions before a line is learned
window = 2.0           # seconds in which Hydra's characters' copies are one occasion
stat_floor = 1500      # a resolution stat at or over this is always shown
forget_days = 30       # one still learning that has not come in this many days is dropped
quiet_secs = 60        # after a login, lines are learned and not shown
mark = false           # a line shown is marked in the story too
paused = false
off = ["Zoleta"]       # characters whose lines are not read
ignore = ['^Obvious (?:exits|paths): ']   # the whole list; none, the built-in one
always = ["‹saysquietly,\"Areyouthere?\""] # sentences always shown

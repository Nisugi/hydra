12/// Who sent a command. A companion character comes with the step that
13/// brings it.
14#[derive(Clone, Copy, Debug, PartialEq, Eq)]
15pub enum Origin {
16    /// Typed by the person playing, at a window or through their Lich.
17    Player,
18    /// Put by one of the player's Lich scripts (`lich`).
19    Lich,
20    /// Hydra's own travel walking the character (`hydra-travel`).
21    Travel,
22    /// A trigger of the player's: one of their highlights that sends.
23    Trigger,
24    /// Hydra's room survey walking the character (`hydra-survey`).
25    Survey,
26}

28/// How a command's round trip ended.
29///
30/// A round trip owns what the game sends from the moment its line is
31/// written until the next prompt.
32#[derive(Clone, Debug, PartialEq, Eq)]
33pub enum Outcome {
34    /// The prompt closed the round trip; this is the first frame the game
35    /// sent for it.
36    Confirmed(Box<Frame>),
37    /// Nothing came back within the deadline, or only a prompt. It does not
38    /// mean the command did not happen.
39    Timeout,
40    /// The connection dropped first. The session reconnects; the command was
41    /// not sent, or its answer was lost with the connection.
42    Disconnected,
43    /// The session has ended.
44    Dead,
45    /// Hydra's command line took it (`claimant`): run or answered, and
46    /// never sent to the game.
47    Claimed,
48    /// Handed to the player's Lich (`lich`), which sends what it makes of
49    /// it.
50    Lich,
51    /// Still waiting to be sent when the player stopped everything, and
52    /// dropped: never sent.
53    Stopped,
54    /// Written to the game at once, as typed, its answer not waited for
55    /// ([`SessionHandle::send_typed`]).
56    Sent,
57}

59/// Whether a line Lich sent was written.
60#[derive(Clone, Copy, Debug, PartialEq, Eq)]
61pub(crate) enum Written {
62    /// It was.
63    Yes,
64    /// There was no connection for it, or the session is quitting.
65    No,

68/// How a quit ended.
69#[derive(Clone, Copy, Debug, PartialEq, Eq)]
70pub enum Farewell {
71    /// The game closed the connection after `quit`.
72    Acknowledged,
73    /// The game did not close it in time; the session closed it.
74    TimedOut,
75    /// `quit` could not be sent: the connection was already gone.

99/// What a handle asks of the session.
100#[derive(Debug)]
101pub(crate) enum Inbox {
102    /// Send a line.
103    Command(Envelope),
104    /// Write a line Lich sent, at once, past the queue: Lich keeps its own
105    /// pace, as it writes straight to its socket.
106    Now {
107        line: String,
108        origin: Origin,
109        generation: Generation,
110        reply: oneshot::Sender<Written>,
111    },
112    /// Hand back what the session holds.
113    Snapshot(oneshot::Sender<Snapshot>),
114    /// Hand back what the session holds, and every event from just after.
115    Join(oneshot::Sender<(Snapshot, broadcast::Receiver<Event>)>),
116    /// Drop every command still waiting to be sent, saying how many.
117    Clear(oneshot::Sender<usize>),
118    /// Send `quit`, and wait up to `timeout` for the game to close.
119    Quit {
120        timeout: Duration,
121        reply: oneshot::Sender<Farewell>,
122    },
123    /// Frames that did not come from the game -- a Lich script's dialog,
124    /// Hydra's own switch of the quick bar -- folded into the state and
125    /// published as the game's are.
126    Frames(Vec<hydra_protocol::Frame>),
127}

132/// A frontend's handle on a session. Cheap to clone; every clone is the
133/// same session.
134#[derive(Clone, Debug)]
135pub struct SessionHandle {
136    commands: mpsc::Sender<Inbox>,
137    pub(crate) events: broadcast::Sender<Event>,
138    generation: GenerationCell,
139    /// Who runs the lines marked as Hydra's, once the binary has said.
140    desk: crate::claimant::Slot,
141    /// The session's player log, when it keeps one: what Hydra says is
142    /// written there too.
143    log: std::sync::Arc<std::sync::OnceLock<crate::PlayerLog>>,
144    /// The player's Lich, while one is attached (`lich`).
145    pub(crate) lich: crate::lich::Shared,
146    /// The character's aliases, once the binary has said where they are.
147    aliases: std::sync::Arc<std::sync::OnceLock<crate::alias::Aliases>>,
148    /// What stops an alias still waiting to send, and whatever else a typed
149    /// line set going ([`Self::stopping`]); replaced each time it stops them.
150    alias_waits: std::sync::Arc<std::sync::Mutex<tokio_util::sync::CancellationToken>>,
151    /// What takes a typed line the game would otherwise be sent, once the
152    /// binary has said ([`Self::set_taker`]).
153    taker: std::sync::Arc<TakerSlot>,
154}

297    /// Drop every command still waiting to be sent: how many. The one in
298    /// flight, already written, waits for its answer as ever. An alias still
299    /// waiting to send the rest of its commands sends none of them.
300    pub async fn clear_queue(&self) -> usize {
301        {
302            let mut waits = self
303                .alias_waits
304                .lock()
305                .unwrap_or_else(std::sync::PoisonError::into_inner);
306            waits.cancel();
307            *waits = tokio_util::sync::CancellationToken::new();
308        }
309        let (reply, answer) = oneshot::channel();
310        if !self.inbox(Inbox::Clear(reply)).await {
311            return 0;
312        }
313        answer.await.unwrap_or(0)
314    }

354    /// Send `line` and wait, up to `deadline`, for its round trip to end.
355    ///
356    /// `quit` (or `exit`) typed as a command quits the session the way
357    /// [`Self::quit`] does, so the game closing the connection afterwards is
358    /// not taken for a dropped one.
359    ///
360    /// A line marked with the command symbol is Hydra's (`claimant`): echoed,
361    /// run or answered, and never written to the game. Then one marked with
362    /// Lich's symbol, `;` unless chosen, is the player's Lich's
363    /// ([`crate::lich`]), and never the game's either; the rest goes to the
364    /// game, past Lich.
365    pub async fn send(&self, line: &str, origin: Origin, deadline: Duration) -> Outcome {
366        if self.claim(line, origin).is_some() {
367            return Outcome::Claimed;
368        }
369        if self.lichs(line) {
370            let _ = self.events.send(Event::Claimed(line.to_owned()));
371            return self.hand_to_lich(line);
372        }
373        self.to_the_game(line, origin, deadline).await
374    }

376    /// What the player typed at a window: first through the character's
377    /// aliases ([`crate::alias`]), each command an alias makes then sent as
378    /// typed, and never expanded again. A wait in an alias holds only that
379    /// alias: what follows it is sent when it is due, and what is typed
380    /// meanwhile goes at once. What an alias cannot send, or has no place
381    /// for, is said.
382    pub async fn send_typed(&self, line: &str) -> Outcome {
383        let Some(aliases) = self.aliases.get() else {
384            return self.send_as_typed(line).await;
385        };
386        let expanded = aliases.expand(line);
387        if let Some(problem) = aliases.problem() {
388            self.say_unlogged(crate::Notice::line(crate::NoticeKind::Error, problem).answering());
389        }
390        match expanded {
391            None => self.send_as_typed(line).await,
392            Some(Err(why)) => {
393                self.say_unlogged(crate::Notice::line(crate::NoticeKind::Error, why).answering());
394                Outcome::Claimed
395            }
396            Some(Ok(expanded)) => self.run_alias(expanded).await,
397        }
398    }

400    /// Send an alias's commands as typed, in order; from its first wait on,
401    /// on a task of its own, which [`Self::clear_queue`] stops.
402    async fn run_alias(&self, expanded: crate::alias::Expanded) -> Outcome {
403        use crate::steps::Step;
404        if let Some(rest) = &expanded.dropped {
405            self.say_unlogged(
406                crate::Notice::line(
407                    crate::NoticeKind::Info,
408                    format!(
409                        "The alias {} has more than one command and no \\? for `{rest}`: it was \
410                         not sent.",
411                        expanded.trigger
412                    ),
413                )
414                .answering(),
415            );
416        }
417        let mut outcome = Outcome::Claimed;
418        let mut steps = expanded.steps.into_iter();
419        while let Some(step) = steps.next() {
420            match step {
421                Step::Line(line) => outcome = self.send_as_typed(&line).await,
422                Step::Wait(wait) => {
423                    let rest: Vec<Step> = std::iter::once(Step::Wait(wait)).chain(steps).collect();
424                    let stopped = self
425                        .alias_waits
426                        .lock()
427                        .unwrap_or_else(std::sync::PoisonError::into_inner)
428                        .child_token();
429                    let handle = self.clone();
430                    tokio::spawn(async move {
431                        for step in rest {
432                            match step {
433                                Step::Wait(wait) => {
434                                    tokio::select! {
435                                        () = stopped.cancelled() => return,
436                                        () = tokio::time::sleep(wait) => {}
437                                    }
438                                }
439                                Step::Line(line) => {
440                                    if stopped.is_cancelled() {
441                                        return;
442                                    }
443                                    let _ = handle.send_as_typed(&line).await;
444                                }
445                            }
446                        }
447                    });
448                    break;
449                }
450            }
451        }
452        outcome
453    }

455    /// A line as the player typed it, past the aliases. Hydra's, by its
456    /// symbol; then Lich's, by Lich's symbol, or, while the player's Lich
457    /// runs, any line, for its aliases and hooks, and Lich sends what it
458    /// makes of it. The rest is **written to the game at once**, in the order
459    /// typed, without waiting for the last line's answer: typed lines go
460    /// straight through, and the game paces what it is sent. Waiting for each
461    /// answer, a key held on `look` was still being sent seconds after it was
462    /// let go. Only Hydra's own lines ([`Self::send`]) wait in the queue.
463    async fn send_as_typed(&self, line: &str) -> Outcome {
464        if self.claim(line, Origin::Player).is_some() {
465            return Outcome::Claimed;
466        }
467        if self.lichs(line) {
468            let _ = self.events.send(Event::Claimed(line.to_owned()));
469            return self.hand_to_lich(line);
470        }
471        if let Some(taker) = self.taker.0.get()
472            && taker(line)
473        {
474            return Outcome::Claimed;
475        }
476        if self.lich_running() {
477            return self.hand_to_lich(line);
478        }
479        match self.write_now(line, Origin::Player).await {
480            Written::Yes => Outcome::Sent,
481            Written::No => Outcome::Disconnected,
482        }
483    }

485    /// Write `line` to the game at once, past the queue, as `origin` sent
486    /// it: whether it was.
487    pub(crate) async fn write_now(&self, line: &str, origin: Origin) -> Written {
488        let (reply, written) = oneshot::channel();
489        let now = Inbox::Now {
490            line: line.to_owned(),
491            origin,
492            generation: self.generation(),
493            reply,
494        };
495        if !self.inbox(now).await {
496            return Written::No;
497        }
498        written.await.unwrap_or(Written::No)
499    }

518    /// Run `line` on Hydra's command line when its symbol marks it as
519    /// Hydra's, as `origin` sent it: echoed first, and answered when nothing
520    /// knows it. `None`: it is not Hydra's.
521    pub(crate) fn claim(&self, line: &str, origin: Origin) -> Option<crate::claimant::Claimed> {
522        let desk = self.desk.get().filter(|desk| desk.marks(line))?;
523        // Echoed before it runs: its answer comes after it in the story.
524        let _ = self.events.send(Event::Claimed(line.to_owned()));
525        let claimed = desk
526            .claim(line, origin)
527            .unwrap_or(crate::claimant::Claimed::Unknown);
528        if claimed == crate::claimant::Claimed::Unknown {
529            let symbol = desk.symbol();
530            let word = line.trim_start().trim_start_matches(symbol).trim();
531            self.say(
532                crate::Notice::line(
533                    crate::NoticeKind::Error,
534                    format!("I do not know {symbol}{word}. {symbol}help lists Hydra's commands."),
535                )
536                .answering(),
537            );
538        }
539        Some(claimed)
540    }

542    /// Queue `line` for the game, and wait up to `deadline` for its round
543    /// trip. The deadline covers the wait to be taken in too: a full inbox
544    /// does not hold a caller past it.
545    async fn to_the_game(&self, line: &str, origin: Origin, deadline: Duration) -> Outcome {
546        let (reply, answer) = oneshot::channel();
547        let envelope = Envelope {
548            line: line.to_owned(),
549            origin,
550            generation: self.generation.get(),
551            reply,
552        };
553        let round_trip = async {
554            if self.commands.send(Inbox::Command(envelope)).await.is_err() {
555                return Outcome::Dead;
556            }
557            answer.await.unwrap_or(Outcome::Dead)
558        };
559        // Out of time, the waiter is dropped: the queue sees it and moves
560        // on, or the line is never taken in at all.
561        tokio::time::timeout(deadline, round_trip)
562            .await
563            .unwrap_or(Outcome::Timeout)
564    }

579    /// Send `quit` and wait, up to `timeout`, for the game to close the
580    /// connection. The session ends either way, and does not reconnect.
581    pub async fn quit(&self, timeout: Duration) -> Farewell {
582        let (reply, answer) = oneshot::channel();
583        if self
584            .commands
585            .send(Inbox::Quit { timeout, reply })
586            .await
587            .is_err()
588        {
589            return Farewell::Unsent;
590        }
591        // A little longer than the session's own wait, so its answer wins.
592        match tokio::time::timeout(timeout + Duration::from_secs(1), answer).await {
593            Ok(Ok(farewell)) => farewell,
594            Ok(Err(_)) | Err(_) => Farewell::Unsent,
595        }
596    }

1//! A line sent and its answer collected: the game's lines until the prompt
2//! that closes them, with the roundtime they cost.
3//!
4//! A [`Listener`] joins a session once ([`SessionHandle::join`]) and keeps
5//! its own [`GameState`] from the stream, so it can tell which lines the
6//! model finished since it last looked. [`Listener::ask`] writes a line to
7//! the game at once, past the queue, waits for the session to echo it
8//! ([`Event::Sent`]) so nothing said before it is taken for its answer,
9//! then keeps every main-window line until the next prompt. A prompt that
10//! closes nothing (the game sends one for someone arriving, a spell
11//! wearing off) is not taken for the answer's own: the answer is the lines
12//! up to the first prompt after at least one line, or whatever came by the
13//! time the wait ran out. [`Reply::prompt_closed`] says which. A command
14//! can answer in more than one burst with a prompt after each (`incant`
15//! is a prepare and a cast), so the answer is held open for a short quiet
16//! spell after its prompt and takes what follows, prompts and all, and
17//! for as long as the roundtime it stated lasts (a cast's effect comes
18//! when its cast time ends). And nothing is written while the game is
19//! still talking: a line asked waits for a quiet spell first, so what a
20//! command of the player's left in the stream is not taken for the
21//! answer either. A line is never taken for the next command's answer,
22//! and one answer taken wrongly does not shift every answer after it.
23//!
24//! The roundtime is read from the state at the close: how many seconds
25//! the game said remain, as the game counts them, never the absolute
26//! epoch the frame carries.
27

41/// How long an answer is waited for once the line is written: the game
42/// answers most lines within a second, a lagging one within a few.
43pub const ANSWER_WAIT: Duration = Duration::from_secs(3);
44
45/// How long an answer is held open after its prompt for more of itself,
46/// and how long the stream must be quiet before a line is written.
47pub const SETTLE: Duration = Duration::from_millis(300);
48
49/// The longest a line waits for the stream to go quiet before it is
50/// written anyway.
51pub const QUIET_WAIT: Duration = Duration::from_secs(2);
52
53/// The longest a stated roundtime is honoured before a line is sent
54/// anyway: a stuck clock does not hold a caller for ever.
55pub const LONGEST_ROUNDTIME: Duration = Duration::from_secs(30);

57/// One command's answer.
58#[derive(Clone, Debug, Default, PartialEq, Eq)]
59pub struct Reply {
60    /// The main-window lines the game finished after the line was sent,
61    /// in order, up to the prompt or the end of the wait.
62    pub lines: Vec<ChunkLine>,
63    /// Seconds of roundtime the game stated for it, `None` when it stated
64    /// none.
65    pub roundtime: Option<u32>,
66    /// Seconds of cast roundtime the game stated for it.
67    pub cast_time: Option<u32>,
68    /// A prompt closed the answer. `false`: the wait ran out first, and
69    /// the lines may be short of the whole answer.
70    pub prompt_closed: bool,
71    /// The character arrived in a room while the answer came: the line was
72    /// a move, or something moved the character.
73    pub arrived: bool,
74}

76impl Reply {
77    /// The lines as plain text.
78    #[must_use]
79    pub fn texts(&self) -> Vec<String> {
80        self.lines.iter().map(ChunkLine::text).collect()
81    }
82}

84/// Why a listener could not go on.
85#[derive(Clone, Copy, Debug, PartialEq, Eq)]
86pub enum Lost {
87    /// The connection dropped, or the session ended.
88    Connection,
89    /// The stop token was cancelled.
90    Stopped,
91}

93/// A session joined, its state kept from the stream, ready to ask.
94pub struct Listener {
95    handle: SessionHandle,
96    events: broadcast::Receiver<Event>,
97    state: GameState,
98    /// Lines of the open chunk already looked at, the dropped counted.
99    heard: usize,
100    /// How many lines the player has sent since the join: a walker reads
101    /// it to see the player taking the character over.
102    player_sent: u64,
103    stop: CancellationToken,
104}

106impl Listener {
107    /// Join `handle`'s session: its state now, and everything from here
108    /// on. `stop` ends any wait at once. `None` once the session has ended.
109    pub async fn join(handle: &SessionHandle, stop: CancellationToken) -> Option<Self> {
110        let (snapshot, events) = handle.join().await?;
111        if matches!(snapshot.lifecycle, State::Reconnecting | State::Closed) {
112            return None;
113        }
114        let mut listener = Self {
115            handle: handle.clone(),
116            events,
117            state: snapshot.state,
118            heard: 0,
119            player_sent: 0,
120            stop,
121        };
122        listener.heard = listener.heard_so_far();
123        Some(listener)
124    }

126    /// What the character knows, as of the last event folded.
127    #[must_use]
128    pub const fn state(&self) -> &GameState {
129        &self.state
130    }
131
132    /// How many lines the player has sent since the join.
133    #[must_use]
134    pub const fn player_sent(&self) -> u64 {
135        self.player_sent
136    }

138    /// Send `line` as `origin` and collect its answer: any roundtime is
139    /// waited out first, then up to `wait` for the answer's prompt.
140    ///
141    /// # Errors
142    ///
143    /// The connection or the session went, or the stop was cancelled.
144    pub async fn ask(&mut self, line: &str, origin: Origin, wait: Duration) -> Result<Reply, Lost> {
145        self.out_of_roundtime().await?;
146        self.quiet().await?;
147        if self.stop.is_cancelled() {
148            return Err(Lost::Stopped);
149        }
150        if !self.handle.write(line, origin).await {
151            return Err(Lost::Connection);
152        }
153        let until = Instant::now() + wait;
154        // Nothing said before the echo is the answer.
155        loop {
156            let (_, sent) = self.hold_until(until).await?;
157            if sent.as_deref() == Some(line) {
158                break;
159            }
160            if Instant::now() >= until {
161                return Ok(Reply::default());
162            }
163        }
164        self.heard = self.heard_so_far();
165        let arrivals = self.state.arrivals;
166        let mut reply = Reply::default();
167        loop {
168            let (fresh, _) = self.hold_until(until).await?;
169            reply.lines.extend(fresh.lines);
170            if fresh.prompt && !reply.lines.is_empty() {
171                reply.prompt_closed = true;
172                break;
173            }
174            if Instant::now() >= until {
175                break;
176            }
177        }
178        // The roundtime and cast time the answer stated, read as each
179        // burst of it comes: the most seen is the command's own.
180        self.stated(&mut reply);
181        // The rest of the answer: what comes while the roundtime it
182        // stated lasts, then until a quiet spell.
183        let most = Instant::now() + LONGEST_ROUNDTIME;
184        while self.in_roundtime() && Instant::now() < most {
185            let (fresh, _) = self.hold_until(Instant::now() + SETTLE).await?;
186            reply.lines.extend(fresh.lines);
187            self.stated(&mut reply);
188        }
189        let mut quiet = Instant::now() + SETTLE;
190        loop {
191            let (fresh, _) = self.hold_until(quiet).await?;
192            self.stated(&mut reply);
193            if !fresh.lines.is_empty() {
194                reply.lines.extend(fresh.lines);
195                quiet = Instant::now() + SETTLE;
196            }
197            if Instant::now() >= quiet {
198                break;
199            }
200        }
201        reply.arrived = self.state.arrivals != arrivals;
202        Ok(reply)
203    }

205    /// The roundtime and cast time the game states now, kept on `reply`
206    /// when more than it has.
207    fn stated(&self, reply: &mut Reply) {
208        let roundtime = self.state.roundtime_remaining().filter(|left| *left > 0);
209        let cast_time = self.state.casttime_remaining().filter(|left| *left > 0);
210        reply.roundtime = reply.roundtime.max(roundtime);
211        reply.cast_time = reply.cast_time.max(cast_time);
212    }
213
214    /// Whether the game's roundtime or cast time is still running.
215    fn in_roundtime(&self) -> bool {
216        self.state.roundtime_remaining().unwrap_or(0) > 0
217            || self.state.casttime_remaining().unwrap_or(0) > 0
218    }
219
220    /// Until nothing has come from the game for [`SETTLE`], or
221    /// [`QUIET_WAIT`] has passed: what was still coming is nobody's
222    /// answer.
223    ///
224    /// # Errors
225    ///
226    /// The connection or the session went, or the stop was cancelled.
227    pub async fn quiet(&mut self) -> Result<(), Lost> {
228        let most = Instant::now() + QUIET_WAIT;
229        let mut quiet = Instant::now() + SETTLE;
230        while Instant::now() < quiet.min(most) {
231            let (fresh, _) = self.hold_until(quiet.min(most)).await?;
232            if !fresh.lines.is_empty() || fresh.prompt {
233                quiet = Instant::now() + SETTLE;
234            }
235        }
236        Ok(())
237    }

239    /// Fold what the game says for `most`, so the state is current.
240    ///
241    /// # Errors
242    ///
243    /// The connection or the session went, or the stop was cancelled.
244    pub async fn hold(&mut self, most: Duration) -> Result<(), Lost> {
245        let until = Instant::now() + most;
246        while Instant::now() < until {
247            self.hold_until(until).await?;
248        }
249        Ok(())
250    }
251
252    /// Until the roundtime and cast roundtime the game stated are over, or
253    /// [`LONGEST_ROUNDTIME`].
254    ///
255    /// # Errors
256    ///
257    /// The connection or the session went, or the stop was cancelled.
258    pub async fn out_of_roundtime(&mut self) -> Result<(), Lost> {
259        let until = Instant::now() + LONGEST_ROUNDTIME;
260        loop {
261            let left = self
262                .state
263                .casttime_remaining()
264                .unwrap_or(0)
265                .max(self.state.roundtime_remaining().unwrap_or(0));
266            if left == 0 || Instant::now() >= until {
267                return Ok(());
268            }
269            let beat = Duration::from_millis(250).max(Duration::from_secs(u64::from(left)) / 2);
270            self.hold(beat.min(until - Instant::now())).await?;
271        }
272    }

274    /// The next event, or `until`: the lines it finished and whether it
275    /// was a prompt, or the line it echoed when it was a send.
276    async fn hold_until(&mut self, until: Instant) -> Result<(Fresh, Option<String>), Lost> {
277        let event = tokio::select! {
278            biased;
279            () = self.stop.cancelled() => return Err(Lost::Stopped),
280            () = tokio::time::sleep_until(until.into()) => return Ok((Fresh::default(), None)),
281            event = self.events.recv() => event,
282        };
283        match event {
284            Ok(Event::Frame(frame)) => Ok((self.fold(&frame), None)),
285            Ok(Event::Sent { line, origin }) => {
286                if origin == Origin::Player {
287                    self.player_sent += 1;
288                }
289                Ok((Fresh::default(), Some(line)))
290            }
291            Ok(Event::StateChanged(State::Reconnecting | State::Closed))
292            | Err(RecvError::Closed) => Err(Lost::Connection),
293            Ok(_) => Ok((Fresh::default(), None)),
294            Err(RecvError::Lagged(_)) => {
295                self.rejoin().await?;
296                Ok((Fresh::default(), None))
297            }
298        }
299    }
300
301    /// Events were lost: the state taken afresh, with the stream from just
302    /// after it, rather than decided from the holes.
303    async fn rejoin(&mut self) -> Result<(), Lost> {
304        let (snapshot, events) = self.handle.join().await.ok_or(Lost::Connection)?;
305        if matches!(snapshot.lifecycle, State::Reconnecting | State::Closed) {
306            return Err(Lost::Connection);
307        }
308        self.state = snapshot.state;
309        self.events = events;
310        self.heard = self.heard_so_far();
311        Ok(())
312    }
313
314    /// Take `frame` into the state; the lines it finished.
315    fn fold(&mut self, frame: &Frame) -> Fresh {
316        self.state.apply(frame);
317        match frame {
318            Frame::Prompt { .. } => {
319                self.heard = 0;
320                Fresh {
321                    lines: Vec::new(),
322                    prompt: true,
323                }
324            }
325            Frame::Text(_) => {
326                let chunk = self.state.open_chunk();
327                let lines = chunk.lines();
328                let total = chunk.dropped() + lines.len();
329                let fresh = total.saturating_sub(self.heard).min(lines.len());
330                self.heard = total;
331                Fresh {
332                    lines: lines[lines.len() - fresh..].to_vec(),
333                    prompt: false,
334                }
335            }
336            _ => Fresh::default(),
337        }
338    }
339
340    /// How many lines the open chunk has had, the dropped counted.
341    fn heard_so_far(&self) -> usize {
342        let chunk = self.state.open_chunk();
343        chunk.dropped() + chunk.lines().len()
344    }
345}

1//! One connection's feed into the player log: what is captured, and what a
2//! line is tagged once its chunk is understood.
3//!
4//! A goal of the log is a combat feed separable from the main feed: with
5//! enough definitions, a blob is classified as combat when it contains one of
6//! them.
7//!
8//! # A line's class is not known when the line ends
9//!
10//! The blob is the model's prompt-to-prompt chunk, and the combat tracker
11//! classifies it when the prompt closes it. So the feed **holds a chunk's
12//! lines and writes them at the prompt**, when the actor can say whether the
13//! chunk produced combat facts. Everything is held, not only the main
14//! window's lines, so the file stays in the order things happened; each line
15//! keeps the stamp of the moment it arrived.
16//!
17//! # The tag: `source/class`
18//!
19//! A line from a combat chunk is tagged `main/combat`. **The class never
20//! replaces the source**: `main` is where the game sent it, `combat` is what
21//! our definitions made of it. A chunk the definitions miss stays `main`.
22//!
23//! # Capture is configurable because a feed costs disk
24//!
25//! [`Capture`] decides what is WRITTEN; filtering what is SHOWN is the
26//! reader's. Every feed is available. The readouts -- streams the game clears
27//! and rewrites whole -- default to off, and when on are written **only when
28//! they change**. A character's choice of feeds is its settings file's
29//! ([`LogSettings`]), from the settings menu's *Player log* page.
30

38/// The class given to every main-window line of a chunk that produced combat
39/// facts.
40pub const COMBAT: &str = "combat";
41
42/// A chunk is written unclassified once it holds this many lines: the login
43/// burst and a long `look` in a crowded room run far past anything a combat
44/// chunk does, and a connection that never prompts must not hold its log in
45/// memory.
46pub const MAX_HELD: usize = 512;
47
48/// Which feeds are written to disk.
49///
50/// A set of what is switched OFF, so that a stream nobody has heard of is
51/// captured: a reader can filter noise and cannot fill a hole.
52#[derive(Clone, Debug, PartialEq, Eq)]
53pub struct Capture {
54    off: BTreeSet<String>,
55}
56
57impl Default for Capture {
58    /// Text feeds on; the readouts ([`READOUTS`]) off.
59    fn default() -> Self {
60        Self {
61            off: READOUTS.iter().map(|s| (*s).to_owned()).collect(),
62        }
63    }
64}
65
66impl Capture {
67    /// Every feed, readouts included.
68    #[must_use]
69    pub const fn everything() -> Self {
70        Self {
71            off: BTreeSet::new(),
72        }
73    }
74
75    /// Switch one feed on or off, by the tag it is logged under (`main`,
76    /// `cmd`, `inv`, `thoughts`...).
77    #[must_use]
78    pub fn set(mut self, feed: &str, on: bool) -> Self {
79        if on {
80            self.off.remove(feed);
81        } else {
82            self.off.insert(feed.to_owned());
83        }
84        self
85    }
86
87    /// Whether a feed is written.
88    #[must_use]
89    pub fn wants(&self, feed: &str) -> bool {
90        !self.off.contains(feed)
91    }
92}

124/// The tag a stream's lines are logged under: the main window's id on the
125/// wire is the empty string, which is no use to someone grepping a file.
126#[must_use]
127pub const fn tag(stream: &str) -> &str {
128    if stream.is_empty() { MAIN } else { stream }
129}

131/// A line waiting for its chunk to close.
132#[derive(Debug)]
133struct Held {
134    at: String,
135    feed: String,
136    text: String,
137    links: Vec<LogLink>,
138}

140/// One connection's feed: owned by the actor, which hands it each frame,
141/// each command sent, and each prompt's verdict.
142#[derive(Debug)]
143pub struct Feed {
144    log: PlayerLog,
145    capture: Capture,
146    /// Each stream's line under way, as text, and its links.
147    unfinished: BTreeMap<String, (String, Vec<LogLink>)>,
148    held: Vec<Held>,
149    /// What each readout last wrote, so an unchanged rewrite is not written.
150    written: BTreeMap<String, Vec<String>>,
151}
152
153impl Feed {
154    /// A feed writing to `log` what `capture` wants, with nothing held.
155    #[must_use]
156    pub fn new(log: PlayerLog, capture: Capture) -> Self {
157        Self {
158            log,
159            capture,
160            unfinished: BTreeMap::new(),
161            held: Vec::new(),
162            written: BTreeMap::new(),
163        }
164    }
165
166    /// One frame the game sent: its text gathered into lines, each held
167    /// once it ends.
168    pub fn frame(&mut self, frame: &Frame) {
169        match frame {
170            Frame::Text(text) => {
171                let (line, links) = self.unfinished.entry(text.stream.clone()).or_default();
172                let from = line.len();
173                line.push_str(&text.content);
174                if let Some(does) = text.link.as_ref().and_then(Does::of) {
175                    links::add(links, (from, line.len()), does);
176                }
177                if text.ends_line {
178                    let (whole, links) = (std::mem::take(line), std::mem::take(links));
179                    let mut at = 0;
180                    for piece in whole.split('\n') {
181                        let piece_links = links::within(&links, (at, at + piece.len()));
182                        self.linked(&text.stream, piece, piece_links);
183                        at += piece.len() + 1;
184                    }
185                }
186            }
187            Frame::ClearStream { id, .. } => {
188                self.unfinished.remove(id);
189            }
190            _ => {}
191        }
192    }
193
194    /// One completed line of game text, from the stream the wire named.
195    pub fn line(&mut self, stream: &str, text: &str) {
196        self.linked(stream, text, Vec::new());
197    }
198
199    /// [`Self::line`], with its links.
200    fn linked(&mut self, stream: &str, text: &str, links: Vec<LogLink>) {
201        if !text.trim().is_empty() {
202            self.hold(tag(stream), text, links);
203        }
204    }
205
206    /// One command, as it went to the game.
207    pub fn command(&mut self, line: &str) {
208        let text = line.trim_end_matches(['\r', '\n']);
209        if !text.is_empty() {
210            self.hold(COMMANDS, text, Vec::new());
211        }
212    }
213
214    fn hold(&mut self, feed: &str, text: &str, links: Vec<LogLink>) {
215        self.held.push(Held {
216            at: hydra_net::line_time(),
217            feed: feed.to_owned(),
218            text: text.to_owned(),
219            links,
220        });
221        if self.held.len() >= MAX_HELD {
222            self.close(false);
223        }
224    }
225
226    /// The prompt closed the chunk: write what was held. `combat` is whether
227    /// the chunk produced combat facts; only the main window's lines take the
228    /// class -- a thought that arrived mid-swing is someone else's words.
229    pub fn close(&mut self, combat: bool) {
230        let held: Vec<Held> = std::mem::take(&mut self.held)
231            .into_iter()
232            .filter(|line| self.capture.wants(&line.feed))
233            .collect();
234        let unchanged = self.unchanged_readouts(&held);
235        for line in held {
236            if unchanged.contains(&line.feed) {
237                continue;
238            }
239            let stream = if combat && line.feed == MAIN {
240                format!("{}/{COMBAT}", line.feed)
241            } else {
242                line.feed
243            };
244            self.log.record(LogLine {
245                at: line.at,
246                stream,
247                text: line.text,
248                links: line.links,
249            });
250        }
251    }
252
253    /// The readouts in this chunk that say exactly what they said last time.
254    /// A readout arrives whole, with no prompt inside it, so a chunk's lines
255    /// for one readout ARE its snapshot. Remembers the new snapshot for the
256    /// ones that did change.
257    fn unchanged_readouts(&mut self, held: &[Held]) -> BTreeSet<String> {
258        let mut snapshots: BTreeMap<&str, Vec<String>> = BTreeMap::new();
259        for line in held {
260            if READOUTS.contains(&line.feed.as_str()) {
261                snapshots
262                    .entry(&line.feed)
263                    .or_default()
264                    .push(line.text.clone());
265            }
266        }
267        let mut unchanged = BTreeSet::new();
268        for (feed, snapshot) in snapshots {
269            if self.written.get(feed) == Some(&snapshot) {
270                unchanged.insert(feed.to_owned());
271            } else {
272                self.written.insert(feed.to_owned(), snapshot);
273            }
274        }
275        unchanged
276    }
277}

279impl Drop for Feed {
280    /// A connection that ends mid-chunk still logs what it saw: the last lines
281    /// before a drop are the ones a reader goes looking for.
282    fn drop(&mut self) {
283        self.close(false);
284    }

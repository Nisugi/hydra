1//! The player log's archives.
2//!
3//! Once an Eastern month (or a Sunday-to-Saturday Eastern week) is over, its
4//! day-files become one archive, `<character>_2026-09.log.gz` (or
5//! `<character>_week-2026-09-27.log.gz`). Monthly by calendar month by default,
6//! weekly or off as the player chooses, and on Eastern time like the server.
7//! The day-files themselves stay on the player's clock; the writer cuts them
8//! wherever a week or a month begins in Eastern
9//! ([`file_key`](super::writer::file_key)), so every file belongs to exactly
10//! one archive.
11//!
12//! # The format: gzip members, a manifest first
13//!
14//! A gzip file may be several gzip streams one after another, and each may
15//! carry a file name. An archive is:
16//!
17//! 1. a member named `manifest`: one line per member after it,
18//!    `<file name>\t<compressed bytes>`;
19//! 2. each day-file, compressed whole, under its own name.
20//!
21//! `zcat` reads it as every day-file in turn (the manifest's lines first),
22//! so a player can still grep an archive by hand. The manifest is what lets a
23//! reader skip to one day without inflating the month before it.
24//!
25//! # Nothing is lost on the way
26//!
27//! An archive is written beside its final name, read back member by member
28//! and compared with the day-files it holds, and only then renamed into place;
29//! the day-files are removed last. A crash at any point leaves either the
30//! day-files or a whole archive (or both, which the reader handles: a
31//! day-file wins over the same name in an archive).
32

42/// How a character's closed days are kept.
43#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
44#[serde(rename_all = "lowercase")]
45pub enum Archive {
46    /// One archive per Eastern calendar month. The default.
47    #[default]
48    Monthly,
49    /// One archive per Sunday-to-Saturday Eastern week.
50    Weekly,
51    /// Every day-file stays plain text.
52    Off,
53}
54
55impl Archive {
56    /// The choices, as the settings file writes them.
57    pub const ALL: [Archive; 3] = [Archive::Monthly, Archive::Weekly, Archive::Off];
58
59    /// As the settings file writes it: `monthly`, `weekly`, `off`.
60    #[must_use]
61    pub const fn word(self) -> &'static str {
62        match self {
63            Archive::Monthly => "monthly",
64            Archive::Weekly => "weekly",
65            Archive::Off => "off",
66        }
67    }
68
69    /// The choice a word names, ignoring case.
70    #[must_use]
71    pub fn from_word(word: &str) -> Option<Self> {
72        Self::ALL
73            .into_iter()
74            .find(|choice| choice.word().eq_ignore_ascii_case(word.trim()))
75    }
76
77    /// The archive a stretch beginning on `stretch` belongs in: `2026-09`,
78    /// `week-2026-09-27`; `None` when nothing is archived.
79    #[must_use]
80    pub fn period(self, stretch: Date) -> Option<String> {
81        match self {
82            Archive::Monthly => Some(format!("{:04}-{:02}", stretch.0, stretch.1)),
83            Archive::Weekly => Some(format!(
84                "week-{}",
85                eastern::format(eastern::week_start(stretch))
86            )),
87            Archive::Off => None,
88        }
89    }
90}

95/// What one sweep did.
96#[derive(Clone, Debug, Default, PartialEq, Eq)]
97pub struct Swept {
98    /// Day-files now in an archive, and removed.
99    pub files: usize,
100    /// The archives written or added to, by name.
101    pub archives: Vec<String>,
102    /// Periods whose day-files were kept because their archive could not be
103    /// written, each with why.
104    pub failed: Vec<String>,
105}
106
107/// Archive every day-file of `character` whose period is over at Unix time
108/// `now`, as `choice` says. Nothing for [`Archive::Off`].
109///
110/// The file being written now is never touched: it is in the current
111/// stretch, so its period is not over.
112///
113/// # Errors
114///
115/// A failure to list the day-files. A period whose archive cannot be read,
116/// written or verified keeps its day-files, is named in [`Swept::failed`],
117/// and the periods after it are archived all the same.
118pub fn sweep(root: &Path, character: &str, choice: Archive, now: i64) -> io::Result<Swept> {
119    let Some(current) = choice.period(eastern::stretch_start(eastern::date(now))) else {
120        return Ok(Swept::default());
121    };
122    let mut periods: BTreeMap<String, Vec<PathBuf>> = BTreeMap::new();
123    for path in writer::days(root, character)? {
124        let Some(name) = path.file_name().and_then(|n| n.to_str()) else {
125            continue;
126        };
127        let Some((_, stretch)) = writer::piece(name) else {
128            continue;
129        };
130        if let Some(period) = choice.period(stretch).filter(|p| *p != current) {
131            periods.entry(period).or_default().push(path);
132        }
133    }
134    let mut swept = Swept::default();
135    for (period, files) in periods {
136        let archive = path(root, character, &period);
137        if let Err(e) = write(&archive, &files) {
138            swept.failed.push(format!("{period}: {e}"));
139            continue;
140        }
141        for file in &files {
142            if let Err(e) = fs::remove_file(file) {
143                swept.failed.push(format!("{}: {e}", file.display()));
144            } else {
145                swept.files += 1;
146            }
147        }
148        swept.archives.push(file_name(character, &period));
149    }
150    Ok(swept)
151}

163/// Every archive `character` has.
164///
165/// # Errors
166///
167/// A failure to read the directory; none at all is not one.
168pub fn archives(root: &Path, character: &str) -> io::Result<Vec<PathBuf>> {
169    let entries = match fs::read_dir(writer::dir(root, character)) {
170        Ok(entries) => entries,
171        Err(err) if err.kind() == io::ErrorKind::NotFound => return Ok(Vec::new()),
172        Err(err) => return Err(err),
173    };
174    let mut paths: Vec<PathBuf> = entries
175        .filter_map(Result::ok)
176        .map(|e| e.path())
177        .filter(|p| {
178            p.file_name()
179                .and_then(|n| n.to_str())
180                .is_some_and(|n| n.ends_with(".log.gz"))
181        })
182        .collect();
183    paths.sort_unstable();
184    Ok(paths)
185}

209/// What `character`'s player log takes on disk.
210///
211/// # Errors
212///
213/// A failure to read the directory or an archive's manifest.
214pub fn usage(root: &Path, character: &str) -> io::Result<Usage> {
215    let size = |path: &PathBuf| fs::metadata(path).map_or(0, |m| m.len());
216    Ok(Usage {
217        plain: writer::days(root, character)?.iter().map(size).sum(),
218        archived: archives(root, character)?.iter().map(size).sum(),
219        days: super::reader::days(root, character)?.len(),
220        unreadable: archives(root, character)?
221            .iter()
222            .filter_map(|path| {
223                names(path)
224                    .err()
225                    .map(|why| format!("{}: {why}", path.display()))
226            })
227            .collect(),
228    })
229}

231/// The names of the day-files an archive holds, from its manifest alone.
232///
233/// # Errors
234///
235/// A failure to read it, or an archive not in this format.
236pub fn names(archive: &Path) -> io::Result<Vec<String>> {
237    let mut reader = BufReader::new(fs::File::open(archive)?);
238    Ok(manifest(&mut reader)?
239        .into_iter()
240        .map(|(name, _)| name)
241        .collect())
242}

244/// The day-files in `archive` whose names `wanted` accepts, each as its name
245/// and its text, in the archive's order. Skips the others unread.
246///
247/// # Errors
248///
249/// A failure to read it, or an archive not in this format.
250pub fn read(archive: &Path, wanted: impl Fn(&str) -> bool) -> io::Result<Vec<(String, Vec<u8>)>> {
251    let mut reader = BufReader::new(fs::File::open(archive)?);
252    let members = manifest(&mut reader)?;
253    let mut found = Vec::new();
254    for (name, length) in members {
255        if wanted(&name) {
256            let start = reader.stream_position()?;
257            found.push((name, member(&mut reader)?));
258            // A member ends where the manifest says, whatever the decoder
259            // read ahead.
260            reader.seek(SeekFrom::Start(start + length))?;
261        } else {
262            reader.seek_relative(i64::try_from(length).map_err(io::Error::other)?)?;
263        }
264    }
265    Ok(found)
266}

268/// The manifest, read from the start of an archive, leaving `reader` at the
269/// first member after it.
270fn manifest(reader: &mut BufReader<fs::File>) -> io::Result<Vec<(String, u64)>> {
271    let text = String::from_utf8(member(reader)?).map_err(io::Error::other)?;
272    text.lines()
273        .map(|line| {
274            let (name, length) = line
275                .split_once('\t')
276                .ok_or_else(|| io::Error::other(format!("not a manifest line: {line}")))?;
277            let length = length.parse().map_err(io::Error::other)?;
278            Ok((name.to_owned(), length))
279        })
280        .collect()
281}

283/// One gzip member, inflated, leaving `reader` just after it.
284fn member<R: BufRead>(reader: &mut R) -> io::Result<Vec<u8>> {
285    let mut decoder = flate2::bufread::GzDecoder::new(reader);
286    let mut bytes = Vec::new();
287    decoder.read_to_end(&mut bytes)?;
288    Ok(bytes)
289}

291/// One gzip member holding `bytes` under `name`.
292fn compress(name: &str, bytes: &[u8]) -> io::Result<Vec<u8>> {
293    let mut encoder = flate2::GzBuilder::new()
294        .filename(name)
295        .write(Vec::new(), flate2::Compression::default());
296    encoder.write_all(bytes)?;
297    encoder.finish()
298}

300/// Write `files` into `archive`, keeping what it already holds, and prove
301/// the result before it takes the archive's name.
302fn write(archive: &Path, files: &[PathBuf]) -> io::Result<()> {
303    let mut members: Vec<(String, Vec<u8>)> = Vec::new();
304    let mut sources: Vec<(String, Vec<u8>)> = Vec::new();
305    for file in files {
306        let name = file
307            .file_name()
308            .and_then(|n| n.to_str())
309            .ok_or_else(|| io::Error::other("a day-file with no name"))?
310            .to_owned();
311        let bytes = fs::read(file)?;
312        members.push((name.clone(), compress(&name, &bytes)?));
313        sources.push((name, bytes));
314    }
315    sources.sort_by(|a, b| a.0.cmp(&b.0));
316    // What the archive already holds is kept as it is, compressed, unless a
317    // day-file of the same name replaces it: the day-file is the newer.
318    if archive.exists() {
319        let mut reader = BufReader::new(fs::File::open(archive)?);
320        let mut kept = Vec::new();
321        for (name, length) in manifest(&mut reader)? {
322            let mut bytes = vec![0; usize::try_from(length).map_err(io::Error::other)?];
323            reader.read_exact(&mut bytes)?;
324            if sources.iter().all(|(source, _)| *source != name) {
325                kept.push((name, bytes));
326            }
327        }
328        kept.append(&mut members);
329        members = kept;
330    }
331    members.sort_by(|a, b| a.0.cmp(&b.0));
332    let mut listed = String::new();
333    for (name, bytes) in &members {
334        use std::fmt::Write as _;
335        let _ = writeln!(listed, "{name}\t{}", bytes.len());
336    }
337
338    let partial = archive.with_extension("gz.partial");
339    {
340        let mut out = io::BufWriter::new(fs::File::create(&partial)?);
341        out.write_all(&compress(MANIFEST, listed.as_bytes())?)?;
342        for (_, bytes) in &members {
343            out.write_all(bytes)?;
344        }
345        out.into_inner()
346            .map_err(io::IntoInnerError::into_error)?
347            .sync_all()?;
348    }
349    let names: Vec<&str> = sources.iter().map(|(name, _)| name.as_str()).collect();
350    let back = read(&partial, |name| names.contains(&name))?;
351    if back.len() != sources.len()
352        || back
353            .iter()
354            .zip(&sources)
355            .any(|((a, got), (b, want))| a != b || got != want)
356    {
357        let _ = fs::remove_file(&partial);
358        return Err(io::Error::other(format!(
359            "{} did not read back as written; the day-files are kept",
360            archive.display()
361        )));
362    }
363    fs::rename(&partial, archive)

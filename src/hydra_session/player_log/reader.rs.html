1//! The player log's reader.
2//!
3//! Day listing, the tail, a time window, and literal or regex search, over the
4//! files [`super::writer`] leaves:
5//! `<root>/player/<character>/<character>_<day>.log`, one
6//! [`format_line`](super::writer::format_line) per line.
7//!
8//! # Off the actor, always
9//!
10//! Every function here is plain blocking file I/O and touches no session state.
11//! One task owns every session's socket, so a synchronous multi-file scan there
12//! would freeze EVERY connected character at once. A caller in async code runs
13//! these on a blocking task, and asks the writer to flush first
14//! ([`PlayerLog::flush`](super::PlayerLog::flush)), or a search of the live
15//! session misses its last second.
16//!
17//! # A time window is a time window
18//!
19//! [`window`] compares stamps; it never counts lines back from the end.
20//! A window approximated by a line count falls short: an 8,000-line tail
21//! reaches back 6.2 minutes of a requested 11 when a busy character logs ~820
22//! lines a minute. The stamp is `HH:MM:SS.mmm`, fixed width, so within a day
23//! text order IS time order, and the day is the file's.
24//!
25//! # Bounded
26//!
27//! At most [`MAX_HITS`] lines come back from any read, and at most [`MAX_DAYS`]
28//! files are opened by one. A read that stopped at the cap says so
29//! ([`Found::more`]).
30//!
31//! # Plain or archived, one day
32//!
33//! A day is read from wherever its files are: plain `.log` files, or members
34//! of a month's or a week's archive ([`super::archive`]). Only
35//! [`days`] and [`read_day`] know there is a difference.
36

53/// A moment in the log: the day's file and the stamp within it.
54///
55/// Ordered as the files are, day then stamp, both compared as text; that is
56/// right because both are fixed width (`YYYY-MM-DD`, `HH:MM:SS.mmm`).
57#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord)]
58pub struct Moment {
59    /// `YYYY-MM-DD`.
60    pub day: String,
61    /// `HH:MM:SS.mmm`, or any prefix of it: `06:47` is the start of that
62    /// minute, since a shorter string sorts before every longer one it begins.
63    pub at: String,
64}
65
66impl Moment {
67    /// The moment `at` on `day`.
68    #[must_use]
69    pub fn new(day: impl Into<String>, at: impl Into<String>) -> Self {
70        Self {
71            day: day.into(),
72            at: at.into(),
73        }
74    }
75}

77/// One line read back.
78#[derive(Clone, Debug, PartialEq, Eq)]
79pub struct Entry {
80    /// The day of the file it was in, `YYYY-MM-DD`.
81    pub day: String,
82    /// When it arrived, `HH:MM:SS.mmm`.
83    pub at: String,
84    /// Its tag: `main`, `thoughts`, `cmd`, `hydra`, `main/combat`.
85    pub stream: String,
86    /// What it said.
87    pub text: String,
88    /// Its links, by where they sit in `text`.
89    pub links: Vec<super::LogLink>,
90}
91
92impl Entry {
93    fn after_or_at(&self, moment: &Moment) -> bool {
94        (self.day.as_str(), self.at.as_str()) >= (moment.day.as_str(), moment.at.as_str())
95    }
96
97    fn before(&self, moment: &Moment) -> bool {
98        (self.day.as_str(), self.at.as_str()) < (moment.day.as_str(), moment.at.as_str())
99    }
100}

102/// Which tags a read keeps.
103///
104/// A tag is matched by any of its `/`-separated parts, so `main` keeps
105/// `main/combat` as well as `main`, and `combat` keeps only `main/combat`. Case
106/// does not matter: the game's own ids are mixed (`Spells`).
107///
108/// Or, made with [`Streams::classes`], by the tag's last part alone: what
109/// the log window ticks, where `main` is the story without its combat.
110#[derive(Clone, Debug, Default, PartialEq, Eq)]
111pub struct Streams {
112    names: Vec<String>,
113    by_class: bool,
114}
115
116impl Streams {
117    /// Every tag.
118    #[must_use]
119    pub fn all() -> Self {
120        Self::default()
121    }
122
123    /// Only these tags, by any part. None named is every tag.
124    #[must_use]
125    pub fn only<S: Into<String>>(names: impl IntoIterator<Item = S>) -> Self {
126        Self {
127            names: names.into_iter().map(Into::into).collect(),
128            by_class: false,
129        }
130    }
131
132    /// Only tags whose last part is one of these: `main` keeps `main` and
133    /// not `main/combat`. None named is every tag.
134    #[must_use]
135    pub fn classes<S: Into<String>>(names: impl IntoIterator<Item = S>) -> Self {
136        Self {
137            by_class: true,
138            ..Self::only(names)
139        }
140    }
141
142    /// The tags named; none for every tag.
143    #[must_use]
144    pub fn names(&self) -> &[String] {
145        &self.names
146    }
147
148    /// Whether a line tagged `tag` is kept.
149    #[must_use]
150    pub fn admits(&self, tag: &str) -> bool {
151        let named = |part: &str| self.names.iter().any(|w| w.eq_ignore_ascii_case(part));
152        self.names.is_empty()
153            || if self.by_class {
154                tag.rsplit('/').next().is_some_and(named)
155            } else {
156                tag.split('/').any(named)
157            }
158    }
159}

161/// What a search looks for in a line's text.
162#[derive(Clone, Debug)]
163pub struct Pattern(regex::Regex);
164
165impl Pattern {
166    /// The text itself, ignoring case: what a player types to find a word.
167    ///
168    /// # Errors
169    ///
170    /// Only past the regex crate's size limit, for a literal of megabytes:
171    /// an escaped literal is otherwise always a valid expression.
172    pub fn literal(text: &str) -> Result<Self, String> {
173        Self::build(&regex::escape(text), true)
174    }
175
176    /// A regular expression, as written: case matters unless it says `(?i)`.
177    ///
178    /// # Errors
179    ///
180    /// Why the expression is not one, in the regex crate's words.
181    pub fn regex(expression: &str) -> Result<Self, String> {
182        Self::build(expression, false)
183    }
184
185    fn build(expression: &str, fold: bool) -> Result<Self, String> {
186        regex::RegexBuilder::new(expression)
187            .case_insensitive(fold)
188            .build()
189            .map(Self)
190            .map_err(|e| e.to_string())
191    }
192
193    /// Whether `text` holds a match.
194    #[must_use]
195    pub fn matches(&self, text: &str) -> bool {
196        self.0.is_match(text)
197    }
198}

200/// What a bounded read found.
201#[derive(Clone, Debug, Default, PartialEq, Eq)]
202pub struct Found {
203    /// The lines, in the order the read gives them.
204    pub entries: Vec<Entry>,
205    /// Whether the read stopped at [`MAX_HITS`] or [`MAX_DAYS`] with more
206    /// left: *"the first thousand"* is not *"all of them"*, and a reader
207    /// must be able to tell.
208    pub more: bool,
209}

211/// Every day this character has a log for, `YYYY-MM-DD`, newest first.
212///
213/// An archive that cannot be read is left out, so the days elsewhere are
214/// still listed: [`archive::usage`] names it.
215///
216/// # Errors
217///
218/// A failure to read the directory. None at all is not one: a character who
219/// has never played has no days.
220pub fn days(root: &Path, character: &str) -> io::Result<Vec<String>> {
221    let mut days: BTreeSet<String> = writer::days(root, character)?
222        .iter()
223        .filter_map(|path| writer::day_of(path))
224        .collect();
225    for path in archive::archives(root, character)? {
226        let Ok(names) = archive::names(&path) else {
227            continue;
228        };
229        days.extend(
230            names
231                .iter()
232                .filter_map(|name| writer::piece(name).map(|(day, _)| day)),
233        );
234    }
235    Ok(days.into_iter().rev().collect())
236}

238/// One day's lines, in the order they were written; none for a day with no
239/// file.
240///
241/// A day may be in more than one file (the writer cuts where an Eastern week
242/// or month begins, [`writer::file_key`]) and each may be plain or archived.
243/// They are read in the order of the stretches they hold; a plain file wins
244/// over an archived one of the same name, which exists only for the moment
245/// between an archive being written and the day-files it replaces being
246/// removed.
247///
248/// A line that is not ours is skipped, and so is a last line with no newline:
249/// the writer ends every line with one, so a line without it is a write still
250/// under way (or cut off by a power loss), and half a line would read as a
251/// whole one.
252///
253/// An archive that cannot be read is passed over: what it held of the day
254/// is not given, and the rest is.
255///
256/// # Errors
257///
258/// A failure to read a day-file that exists.
259pub fn read_day(root: &Path, character: &str, day: &str) -> io::Result<Vec<Entry>> {
260    let ours = |name: &str| writer::piece(name).filter(|(of, _)| of == day);
261    let mut parts: BTreeMap<(Date, String), Vec<u8>> = BTreeMap::new();
262    for path in writer::days(root, character)? {
263        let Some(name) = path.file_name().and_then(|n| n.to_str()) else {
264            continue;
265        };
266        let Some((_, stretch)) = ours(name) else {
267            continue;
268        };
269        match fs::read(&path) {
270            Ok(bytes) => {
271                parts.insert((stretch, name.to_owned()), bytes);
272            }
273            // Archived and removed since it was listed: the archive has it.
274            Err(err) if err.kind() == io::ErrorKind::NotFound => {}
275            Err(err) => return Err(err),
276        }
277    }
278    for path in archives_for(root, character, day) {
279        // One that cannot be read gives none of its lines, and the rest of
280        // the day is still read: the disk-use summary names it
281        // ([`archive::usage`]).
282        let Ok(members) = archive::read(&path, |name| ours(name).is_some()) else {
283            continue;
284        };
285        for (name, bytes) in members {
286            if let Some((_, stretch)) = ours(&name) {
287                parts.entry((stretch, name)).or_insert(bytes);
288            }
289        }
290    }
291    Ok(parts
292        .values()
293        .flat_map(|bytes| entries(day, bytes))
294        .collect())
295}

297/// The lines of one file's bytes, as [`read_day`] describes.
298fn entries(day: &str, bytes: &[u8]) -> Vec<Entry> {
299    // Lossy: a byte the writer never wrote (a hand edit, a torn sector) costs
300    // that character, not the day.
301    let text = String::from_utf8_lossy(bytes);
302    let whole = text.rfind('\n').map_or("", |end| &text[..end]);
303    whole
304        .lines()
305        .filter_map(writer::parse_line)
306        .map(|(at, stream, text)| {
307            let (text, links) = super::links::read(text.trim_end_matches('\r'));
308            Entry {
309                day: day.to_owned(),
310                at: at.to_owned(),
311                stream: stream.to_owned(),
312                text: text.to_owned(),
313                links,
314            }
315        })
316        .collect()
317}

319/// The archives that can hold any of `day`'s files, whichever way they were
320/// archived: a file of that date holds a stretch from six days before it
321/// (the week began earlier) to the day after (a player west of Eastern, past
322/// midnight there).
323fn archives_for(root: &Path, character: &str, day: &str) -> Vec<std::path::PathBuf> {
324    let Some(date) = eastern::parse(day) else {
325        return Vec::new();
326    };
327    let days = eastern::days_from_civil(date);
328    let periods: BTreeSet<String> = (days - 6..=days + 1)
329        .map(eastern::civil_from_days)
330        .flat_map(|d| [Archive::Monthly, Archive::Weekly].map(|choice| choice.period(d)))
331        .flatten()
332        .collect();
333    periods
334        .iter()
335        .map(|period| archive::path(root, character, period))
336        .filter(|path| path.is_file())
337        .collect()
338}

340/// The last `count` lines kept by `streams`, oldest first, across as many
341/// days as it takes (at most [`MAX_DAYS`]). `count` is capped at
342/// [`MAX_HITS`].
343///
344/// # Errors
345///
346/// A failure to read the directory or a day's file.
347pub fn tail(
348    root: &Path,
349    character: &str,
350    count: usize,
351    streams: &Streams,
352) -> io::Result<Vec<Entry>> {
353    let count = count.min(MAX_HITS);
354    let mut kept: Vec<Entry> = Vec::new();
355    for day in days(root, character)?.iter().take(MAX_DAYS) {
356        if kept.len() >= count {
357            break;
358        }
359        let mut earlier: Vec<Entry> = read_day(root, character, day)?
360            .into_iter()
361            .filter(|e| streams.admits(&e.stream))
362            .collect();
363        earlier.append(&mut kept);
364        kept = earlier;
365    }
366    let cut = kept.len().saturating_sub(count);
367    kept.drain(..cut);
368    // The buffer is a whole day's: what is returned keeps only its own.
369    kept.shrink_to_fit();
370    Ok(kept)
371}

373/// The lines from `from` up to but not including `to`, kept by `streams`,
374/// oldest first: the first [`MAX_HITS`] of them, with [`Found::more`] set when
375/// there were others after.
376///
377/// Chosen by stamp, never by counting lines (the module doc).
378///
379/// # Errors
380///
381/// A failure to read the directory or a day's file.
382pub fn window(
383    root: &Path,
384    character: &str,
385    from: &Moment,
386    to: &Moment,
387    streams: &Streams,
388) -> io::Result<Found> {
389    let mut found = Found::default();
390    let mut within: Vec<String> = days(root, character)?
391        .into_iter()
392        .filter(|day| *day >= from.day && *day <= to.day)
393        .collect();
394    within.reverse();
395    found.more = within.len() > MAX_DAYS;
396    for day in within.iter().take(MAX_DAYS) {
397        for entry in read_day(root, character, day)? {
398            if !entry.after_or_at(from) || !entry.before(to) || !streams.admits(&entry.stream) {
399                continue;
400            }
401            if found.entries.len() == MAX_HITS {
402                found.more = true;
403                return Ok(found);
404            }
405            found.entries.push(entry);
406        }
407    }
408    Ok(found)
409}

411/// The lines whose text matches `pattern`, kept by `streams`, **newest
412/// first**, from the days on or after `since` (every day when `None`): the
413/// first [`MAX_HITS`] found, over at most [`MAX_DAYS`] days.
414///
415/// # Errors
416///
417/// A failure to read the directory or a day's file.
418pub fn search(
419    root: &Path,
420    character: &str,
421    pattern: &Pattern,
422    streams: &Streams,
423    since: Option<&str>,
424) -> io::Result<Found> {
425    let mut found = Found::default();
426    let within: Vec<String> = days(root, character)?
427        .into_iter()
428        .filter(|day| since.is_none_or(|since| day.as_str() >= since))
429        .collect();
430    for day in within.iter().take(MAX_DAYS) {
431        let hits = read_day(root, character, day)?
432            .into_iter()
433            .rev()
434            .filter(|e| streams.admits(&e.stream) && pattern.matches(&e.text));
435        for hit in hits {
436            if found.entries.len() == MAX_HITS {
437                found.more = true;
438                return Ok(found);
439            }
440            found.entries.push(hit);
441        }
442    }
443    found.more = within.len() > MAX_DAYS;
444    Ok(found)
445}

458/// Write every line from the days `from` to `to` (both kept, `YYYY-MM-DD`) kept
459/// by `streams` to one plain file at `out`, oldest first. The caller gives the
460/// range and the streams; this does all the I/O. At most [`MAX_DAYS`] days, the
461/// newest dropped past that.
462///
463/// Each line is `YYYY-MM-DD HH:MM:SS.mmm [tag] text`, the day in front, since
464/// an export may span many. `out` is written whole or not at all: beside its
465/// name first, then renamed.
466///
467/// # Errors
468///
469/// A failure to read the log or to write `out`.
470pub fn export(
471    root: &Path,
472    character: &str,
473    (from, to): (&str, &str),
474    streams: &Streams,
475    out: &Path,
476) -> io::Result<Exported> {
477    // Each export's own name beside `out`: see below.
478    static NEXT: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
479    let mut within: Vec<String> = days(root, character)?
480        .into_iter()
481        .filter(|day| day.as_str() >= from && day.as_str() <= to)
482        .collect();
483    within.reverse();
484    within.truncate(MAX_DAYS);
485    if let Some(parent) = out.parent() {
486        fs::create_dir_all(parent)?;
487    }
488    // Its own beside `out`: two exports to one file, asked at once, each
489    // write their own and the last renamed in wins, neither finding the
490    // other's gone from under it.
491    let partial = out.with_extension(format!(
492        "{}.{}.partial",
493        std::process::id(),
494        NEXT.fetch_add(1, std::sync::atomic::Ordering::Relaxed)
495    ));
496    let written = write_export(root, character, &within, streams, &partial);
497    let lines = match written {
498        Ok(lines) => lines,
499        Err(e) => {
500            let _ = fs::remove_file(&partial);
501            return Err(e);
502        }
503    };
504    if let Err(e) = fs::rename(&partial, out) {
505        let _ = fs::remove_file(&partial);
506        return Err(e);
507    }
508    Ok(Exported {
509        path: out.to_owned(),
510        lines,
511        days: within.len(),
512    })
513}

515/// The lines of `days` that `streams` keeps, written to `to`: how many.
516fn write_export(
517    root: &Path,
518    character: &str,
519    days: &[String],
520    streams: &Streams,
521    to: &Path,
522) -> io::Result<usize> {
523    use std::io::Write as _;
524    let mut file = io::BufWriter::new(fs::File::create(to)?);
525    let mut lines = 0;
526    for day in days {
527        for entry in read_day(root, character, day)? {
528            if streams.admits(&entry.stream) {
529                writeln!(
530                    file,
531                    "{} {} [{}] {}",
532                    entry.day, entry.at, entry.stream, entry.text
533                )?;
534                lines += 1;
535            }
536        }
537    }
538    file.into_inner()
539        .map_err(io::IntoInnerError::into_error)?
540        .sync_all()?;
541    Ok(lines)
542}

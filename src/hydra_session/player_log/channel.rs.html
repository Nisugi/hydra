1//! The record and the bounded channel.
2//!
3//! The channel is bounded so the actor, which owns the game socket, never
4//! waits on the disk; a line that does not fit is dropped and counted, so a
5//! hole in the history is always reported. A line carries no session or
6//! connection number: one log is one character's, and a reconnect is said in
7//! the log as Hydra's own line.
8
9use std::sync::atomic::{AtomicU64, Ordering};
10use std::sync::{Arc, Mutex, PoisonError};
11
12use hydra_net::Redactions;
13
14/// How many lines may wait for the writer before one is dropped.
15///
16/// **Sized against the login burst, which is the worst case measured.** A
17/// live session's login once dropped 99 events from the broadcast ring, and
18/// that burst is the densest text this client sees: room, exits, worn
19/// inventory, spell lists and the services table arriving at once.
20///
21/// 4096 is that with two orders of magnitude of headroom, which is the right
22/// shape for a buffer whose consumer does one `write` per batch. A number
23/// chosen to make the drop path unreachable in normal play, not measured to be
24/// exactly sufficient -- and the counter exists precisely because "unreachable"
25/// is a claim rather than a guarantee.
26pub const CAPACITY: usize = 4096;
27
28/// One line on its way to disk.
29///
30/// Deliberately flat: a timestamp, where it came from, what it said, and
31/// its links. No styling, no runs, no frame: the log keeps the text a person
32/// read and what it could click, not how it was drawn.
33#[derive(Clone, Debug, PartialEq, Eq)]
34pub struct LogLine {
35    /// Wall clock at the moment the line was handed over, `HH:MM:SS.mmm`.
36    ///
37    /// **One stamp, and the date is not in it.** The date is already in the
38    /// filename, as the wire log's is, and repeating it on every line is
39    /// waste.
40    ///
41    /// Same format as the wire log's, so two logs of one session line up by
42    /// eye. That is the argument for keeping both.
43    ///
44    /// **Not** the game's clock. `GameState::game_time_now` is the right thing
45    /// for a behavior to reason with; a person reading their own history wants
46    /// to know when they were at the keyboard.
47    pub at: String,
48    /// The window the line belongs to: `main`, `thoughts`, `inv`, `death`.
49    ///
50    /// Kept verbatim rather than typed. The set is the game's and it grows;
51    /// `streams.rs` already treats stream ids as open for that reason, and a
52    /// log that refused an unfamiliar stream would lose exactly the traffic
53    /// worth keeping.
54    pub stream: String,
55    /// What the line said, assembled and entity-decoded, with no markup.
56    pub text: String,
57    /// Its links, by where they sit in `text` ([`super::links`]).
58    pub links: Vec<super::LogLink>,
59}
60
61/// How many reads may be waiting on the writer's flush at once. A reader
62/// finding it full waits its turn; the flush it waits behind covers it too.
63const FLUSHES_WAITING: usize = 8;
64
65/// How long a read waits for the writer to say it has flushed.
66///
67/// The writer answers a flush between two lines; a read waiting longer than
68/// this goes ahead with what the file holds.
69const FLUSH_WAIT: std::time::Duration = std::time::Duration::from_secs(2);
70
71/// The session's end of the player log.
72///
73/// Cheap to clone; every clone shares one channel and one counter.
74#[derive(Clone, Debug)]
75pub struct PlayerLog {
76    tx: tokio::sync::mpsc::Sender<LogLine>,
77    flushes: tokio::sync::mpsc::Sender<tokio::sync::oneshot::Sender<()>>,
78    dropped: Arc<AtomicU64>,
79    redactions: Shared,
80}
81
82/// The secrets the writer strips, shared by the log and its writer: a
83/// reconnect brings a new launch key after the file is open.
84pub(crate) type Shared = Arc<Mutex<Redactions>>;
85
86/// The writer's end: where lines arrive.
87///
88/// Held by the writer task ([`super::writer`]), which is the only thing that
89/// should own one.
90#[derive(Debug)]
91pub struct LogSink {
92    rx: tokio::sync::mpsc::Receiver<LogLine>,
93    flushes: tokio::sync::mpsc::Receiver<tokio::sync::oneshot::Sender<()>>,
94    dropped: Arc<AtomicU64>,
95    redactions: Shared,
96}
97
98/// What reaches the writer: a line, or a reader asking for everything so far
99/// to be on disk: reads flush the writer's buffer first.
100#[derive(Debug)]
101pub enum Arrival {
102    /// A line to write.
103    Line(LogLine),
104    /// Write what is queued, flush, then answer here.
105    Flush(tokio::sync::oneshot::Sender<()>),
106}
107
108impl PlayerLog {
109    /// A log and its sink, joined by a bounded channel of [`CAPACITY`].
110    #[must_use]
111    pub fn new() -> (Self, LogSink) {
112        Self::with_capacity(CAPACITY)
113    }
114
115    /// [`Self::new`], with the capacity given rather than taken from the
116    /// constant. For tests that need to reach the drop path without queueing
117    /// four thousand lines.
118    ///
119    /// A capacity of zero is taken as one: `tokio`'s bounded channel needs
120    /// a slot, and a log with no room would drop every line.
121    #[must_use]
122    pub fn with_capacity(capacity: usize) -> (Self, LogSink) {
123        let (tx, rx) = tokio::sync::mpsc::channel(capacity.max(1));
124        let (flush_tx, flush_rx) = tokio::sync::mpsc::channel(FLUSHES_WAITING);
125        let dropped = Arc::new(AtomicU64::new(0));
126        let redactions = Shared::default();
127        (
128            Self {
129                tx,
130                flushes: flush_tx,
131                dropped: Arc::clone(&dropped),
132                redactions: Arc::clone(&redactions),
133            },
134            LogSink {
135                rx,
136                flushes: flush_rx,
137                dropped,
138                redactions,
139            },
140        )
141    }
142
143    /// Ask the writer to put every line recorded so far on disk, and wait
144    /// until it has: what a read does first, or a search of the live session
145    /// misses what is still buffered.
146    ///
147    /// **Never call this from the actor.** It waits, up to two seconds;
148    /// [`Self::record`] is the actor's, and never does.
149    ///
150    /// `true` when the writer answered; `false` when there is no writer or it
151    /// did not answer in time. Either way the read goes ahead: the file holds
152    /// what it holds.
153    pub async fn flush(&self) -> bool {
154        let (ack, done) = tokio::sync::oneshot::channel();
155        let asked = async {
156            self.flushes.send(ack).await.ok()?;
157            done.await.ok()
158        };
159        matches!(tokio::time::timeout(FLUSH_WAIT, asked).await, Ok(Some(())))
160    }
161
162    /// Hand one line to the writer, or count it dropped.
163    ///
164    /// **Never blocks, never awaits, never fails.** This is called from the
165    /// actor, which owns the game socket; anything that can wait here can stall
166    /// the character it is logging -- and, in a multi-session process, every
167    /// other character too.
168    ///
169    /// Returns whether the line was accepted. Callers may ignore it: the drop
170    /// is already counted, and the counter rather than the return value is what
171    /// a frontend reads.
172    // NOT `#[must_use]`, though clippy::pedantic asks for it. The drop is
173    // already counted by the time this returns, so a caller ignoring the bool
174    // loses nothing -- and the actor, which is the main caller, has nothing
175    // useful to do with it. The counter is the reporting channel.
176    #[allow(clippy::must_use_candidate)]
177    pub fn record(&self, line: LogLine) -> bool {
178        if self.tx.try_send(line).is_ok() {
179            return true;
180        }
181        // Both failure modes are a loss the reader must be able to see: a full
182        // channel means the writer is behind, and a closed one means it is
183        // gone. Neither is worth distinguishing HERE -- what a reader needs is
184        // "your history has a hole in it", and the count says that either way.
185        self.dropped.fetch_add(1, Ordering::Relaxed);
186        false
187    }
188
189    /// Strip `secret` from every line written from now, as `replacement`:
190    /// a launch key, which can come back as game text, or the account name.
191    pub fn redact(&self, secret: &str, replacement: &'static str) {
192        self.redactions
193            .lock()
194            .unwrap_or_else(PoisonError::into_inner)
195            .add(secret, replacement);
196    }
197
198    /// Record one of Hydra's own lines, `[hydra]`, now: a connection lost or
199    /// made. Its own tag, so neither a reader nor a parser mistakes Hydra's
200    /// voice for the game's.
201    pub fn notice(&self, text: &str) {
202        self.record(LogLine {
203            at: hydra_net::line_time(),
204            stream: super::NOTICES.to_owned(),
205            text: text.to_owned(),
206            links: Vec::new(),
207        });
208    }
209
210    /// How many lines this log has failed to record.
211    ///
212    /// **Non-zero means the archive has a hole**, and a frontend should say so.
213    /// The alternative -- a log that quietly skips lines under load -- produces
214    /// a file that reads as complete and is not, which is the one outcome
215    /// the count exists to prevent.
216    #[must_use]
217    pub fn dropped(&self) -> u64 {
218        self.dropped.load(Ordering::Relaxed)
219    }
220}
221
222impl LogSink {
223    /// The next line, or `None` once every [`PlayerLog`] is gone.
224    pub async fn recv(&mut self) -> Option<LogLine> {
225        self.rx.recv().await
226    }
227
228    /// The next line or flush request, or `None` once every [`PlayerLog`] is
229    /// gone. What the writer waits on.
230    pub async fn next(&mut self) -> Option<Arrival> {
231        tokio::select! {
232            line = self.rx.recv() => line.map(Arrival::Line),
233            Some(ack) = self.flushes.recv() => Some(Arrival::Flush(ack)),
234        }
235    }
236
237    /// The next line if one is already waiting, without awaiting.
238    ///
239    /// For tests that assert a line arrived, and for a writer draining what is
240    /// queued before it flushes. Returns `None` both when the channel is empty
241    /// and when it has ended -- a caller that needs to tell those apart wants
242    /// [`Self::recv`].
243    pub fn try_recv(&mut self) -> Option<LogLine> {
244        self.rx.try_recv().ok()
245    }
246
247    /// The shared drop count, so the writer can report it alongside its own
248    /// failures. The writer adds write errors to this same counter: from a
249    /// reader's side, a line lost to a full channel and one lost to a full
250    /// disk are the same hole.
251    #[must_use]
252    pub fn dropped(&self) -> u64 {
253        self.dropped.load(Ordering::Relaxed)
254    }
255
256    /// Count a line lost after it left the channel.
257    ///
258    /// For the writer, whose failures are the other half of what
259    /// [`PlayerLog::dropped`] reports.
260    pub fn note_dropped(&self) {
261        self.dropped.fetch_add(1, Ordering::Relaxed);
262    }
263
264    /// `text` with every secret the log was given stripped.
265    pub(crate) fn redacted(&self, text: &str) -> String {
266        self.redactions
267            .lock()
268            .unwrap_or_else(PoisonError::into_inner)
269            .apply(text)
270    }

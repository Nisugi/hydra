1//! Commands and waits, as a key macro and an alias write them: commands cut
2//! at each line break, empty pieces dropped, and a piece that is `s` and a
3//! number of seconds (`s1.5`) a wait before the rest.
4
5use std::time::Duration;
6
7/// The longest a wait may be, so a slip of the finger (`s600`) does not
8/// hold a character's commands for ten minutes.
9pub const LONGEST_WAIT: Duration = Duration::from_mins(1);
10
11/// One step: a command sent, or a wait before the rest.
12#[derive(Clone, Debug, PartialEq, Eq)]
13pub enum Step {
14    /// A command sent.
15    Line(String),
16    /// A wait before the rest.
17    Wait(Duration),
18}
19
20/// `text`'s commands and waits, in order: cut at each carriage return or
21/// line feed, each piece trimmed, empty ones dropped.
22///
23/// # Errors
24///
25/// Why it cannot be sent, in words for the player: a wait too long, or
26/// nothing to send.
27pub fn steps(text: &str) -> Result<Vec<Step>, String> {
28    let mut steps = Vec::new();
29    for piece in text.split(['\r', '\n']).map(str::trim) {
30        if piece.is_empty() {
31            continue;
32        }
33        match wait(piece) {
34            Some(wait) => steps.push(Step::Wait(wait?)),
35            None => steps.push(Step::Line(piece.to_owned())),
36        }
37    }
38    if !steps.iter().any(|step| matches!(step, Step::Line(_))) {
39        return Err("it sends no command".to_owned());
40    }
41    Ok(steps)
42}
43
44/// `piece` as a wait, when it is one: `s` and a number of seconds, from a
45/// moment to [`LONGEST_WAIT`].
46#[must_use]
47pub fn wait(piece: &str) -> Option<Result<Duration, String>> {
48    let seconds: f32 = piece.strip_prefix(['s', 'S'])?.parse().ok()?;
49    let wait = Duration::try_from_secs_f32(seconds)
50        .ok()
51        .filter(|wait| *wait <= LONGEST_WAIT && !wait.is_zero());
52    Some(wait.ok_or_else(|| {
53        format!(
54            "`{piece}` waits from a moment to {} seconds",
55            LONGEST_WAIT.as_secs()
56        )
57    }))
58}

60#[cfg(test)]
61mod tests {
62    use super::*;
63
64    #[test]
65    fn commands_and_waits_in_order() {
66        assert_eq!(
67            steps("stance off\r\rprep 118\r s1.5 \ncast\r"),
68            Ok(vec![
69                Step::Line("stance off".to_owned()),
70                Step::Line("prep 118".to_owned()),
71                Step::Wait(Duration::from_millis(1500)),
72                Step::Line("cast".to_owned()),
73            ])
74        );
75        assert!(steps("s1\r  \r").is_err(), "a wait alone sends nothing");
76        assert!(steps("look\rs61").is_err(), "a wait too long");
77        assert!(steps("look\rs0").is_err(), "a wait of nothing");
78        // `s` with no number is a command: the game's south.
79        assert_eq!(steps("s"), Ok(vec![Step::Line("s".to_owned())]));
80    }
81}

1//! Writing a file Hydra keeps -- a character's facts, the learned menu
2//! dictionary, a layout, a doll's calibrated picture -- so it is never left
3//! half written, and naming a character's file.
4//!
5//! Here rather than in a frontend, since the session keeps files too.
6
7use std::collections::BTreeMap;
8use std::fs;
9use std::io;
10use std::path::{Path, PathBuf};
11use std::sync::{Arc, Mutex, PoisonError};
12
13/// Each file a change has been made to, by path: the right to read it,
14/// change it and write it back ([`changing`]).
15static CHANGING: Mutex<BTreeMap<PathBuf, Arc<Mutex<()>>>> = Mutex::new(BTreeMap::new());
16
17/// Run `change` -- a read of the file at `path`, a change, and its write --
18/// with no other change to that file in this process made meanwhile: two
19/// settings changed at once must not each read the file, and the second
20/// write lose the first change. A lock per file; **not reentrant**.
21pub fn changing<T>(path: &Path, change: impl FnOnce() -> T) -> T {
22    let file = {
23        let mut files = CHANGING.lock().unwrap_or_else(PoisonError::into_inner);
24        Arc::clone(files.entry(path.to_owned()).or_default())
25    };
26    // A panic while it was held left the file whole: the rename is the only
27    // write, and it is atomic.
28    let _held = file.lock().unwrap_or_else(PoisonError::into_inner);
29    change()
30}

32/// Write `value` to `path` in `dir` as JSON, atomically ([`save_bytes`]).
33///
34/// # Errors
35///
36/// The value cannot be written as JSON, the directory cannot be created, or
37/// the file cannot be written or renamed.
38pub fn save_json<T: serde::Serialize>(dir: &Path, path: &Path, value: &T) -> io::Result<()> {
39    let text = serde_json::to_string_pretty(value)
40        .map_err(|err| io::Error::new(io::ErrorKind::InvalidData, err))?;
41    save_bytes(dir, path, text.as_bytes())
42}

44/// Write `bytes` to `path` in `dir`, atomically: to a temp file of this
45/// write's own, synced, then renamed over `path`. Two writes of one file at
46/// once never rename each other's half-written temp into place.
47///
48/// # Errors
49///
50/// The directory cannot be created, or the file cannot be written or
51/// renamed.
52pub fn save_bytes(dir: &Path, path: &Path, bytes: &[u8]) -> io::Result<()> {
53    fs::create_dir_all(dir)?;
54    let extension = path
55        .extension()
56        .map_or_else(String::new, |ext| ext.to_string_lossy().into_owned());
57    // One thread writes one file at a time, so the process and the thread
58    // name this write among any made at once.
59    let thread = {
60        use std::hash::{Hash, Hasher};
61        let mut hasher = std::collections::hash_map::DefaultHasher::new();
62        std::thread::current().id().hash(&mut hasher);
63        hasher.finish()
64    };
65    let temp = path.with_extension(format!("{extension}.{}-{thread:x}.tmp", std::process::id()));
66    // Synced before the rename: the rename alone survives a crash, not a
67    // power loss, which could leave a valid name over an empty file.
68    let mut file = fs::File::create(&temp)?;
69    io::Write::write_all(&mut file, bytes)?;
70    file.sync_all()?;
71    drop(file);
72    fs::rename(&temp, path).inspect_err(|_| {
73        let _ = fs::remove_file(&temp);
74    })?;
75    // std cannot open a directory for syncing on Windows, where NTFS
76    // journals the rename itself.
77    #[cfg(unix)]
78    fs::File::open(dir)?.sync_all()?;
79    Ok(())
80}

82/// A name from the wire as one path component: its ASCII letters and
83/// digits, **lowercased**, so `Ashryn` and `ASHRYN` are one file on every
84/// platform. Character names are alphanumeric, so filtering loses nothing.
85#[must_use]
86pub fn safe_component(name: &str) -> String {
87    name.chars()
88        .filter(char::is_ascii_alphanumeric)
89        .map(|c| c.to_ascii_lowercase())
90        .collect()
91}

93/// `<dir>/<instance>_<character><suffix>`; `None` when either half has
94/// nothing a filename can carry, rather than two such characters sharing one
95/// file.
96#[must_use]
97pub fn character_path(
98    dir: &Path,
99    instance: &str,
100    character: &str,
101    suffix: &str,
102) -> Option<PathBuf> {
103    let instance = safe_component(instance);
104    let character = safe_component(character);
105    if instance.is_empty() || character.is_empty() {
106        return None;
107    }
108    Some(dir.join(format!("{instance}_{character}{suffix}")))
109}

111#[cfg(test)]
112mod tests {
113    use std::sync::mpsc;
114    use std::time::Duration;
115
116    use super::*;
117
118    /// A change to a file waits for one being made to it; a change to
119    /// another file does not wait.
120    #[test]
121    fn a_change_waits_for_one_to_the_same_file_only() {
122        let (same, other) = (
123            Path::new("store-test-waits/same.toml"),
124            Path::new("store-test-waits/other.toml"),
125        );
126        let (tell, told) = mpsc::channel();
127        let (waiting, elsewhere) = changing(same, || {
128            let waiting = {
129                let tell = tell.clone();
130                std::thread::spawn(move || changing(same, || tell.send("same")))
131            };
132            let elsewhere = std::thread::spawn(move || changing(other, || tell.send("other")));
133            assert_eq!(told.recv_timeout(Duration::from_secs(10)), Ok("other"));
134            assert!(
135                told.recv_timeout(Duration::from_millis(200)).is_err(),
136                "the same file's change waits"
137            );
138            (waiting, elsewhere)
139        });
140        assert_eq!(told.recv_timeout(Duration::from_secs(10)), Ok("same"));
141        assert!(waiting.join().is_ok_and(|sent| sent.is_ok()));
142        assert!(elsewhere.join().is_ok_and(|sent| sent.is_ok()));
143    }

145    #[test]
146    fn a_file_is_written_whole_and_written_over() {
147        let dir = std::env::temp_dir().join(format!("hydra-store-{}", std::process::id()));
148        let path = dir.join("kept.json");
149        save_json(&dir, &path, &vec![1, 2]).unwrap();
150        save_json(&dir, &path, &vec![3]).unwrap();
151        let text = fs::read_to_string(&path).unwrap();
152        assert_eq!(serde_json::from_str::<Vec<u32>>(&text).unwrap(), [3]);
153        let left: Vec<_> = fs::read_dir(&dir).unwrap().collect();
154        assert_eq!(left.len(), 1, "no temp file left behind");
155        let _ = fs::remove_dir_all(&dir);
156    }

1//! `.go2` and `.route2`: travel on Hydra's command line
2//! (`hydra_travel::command`).
3//!
4//! `.go2 <place>` walks there (`hydra_travel::walk`), one walk a character:
5//! another `.go2` stops the first and sets out, and `.go2 stop` and `.stop`
6//! stop it. `.route2` says the route, priced for this character as it is
7//! now, and sends nothing; `.go2 targets`, `list`, `save` and `delete` list
8//! the places there are and keep the player's own; `.go2 coverage` says how
9//! much of the map Hydra walks yet. A route is drawn on the character's
10//! minimap (`hydra_map::show_route`): a walk's as it goes, a `.route2`'s
11//! until `.route2 clear` or the next. The minimap asks with these words too:
12//! `.route2 <room> --brief` for a click on a room, `.go2 <room>` for one on
13//! the goal.
14//!
15//! `to <character>`, for `.go2` and `.route2`, is the room another
16//! character running here stands in, as it is when asked: the walk goes
17//! there, and does not follow it on.
18//!
19//! `.sail <port>` makes the character's ship ready, puts to sea and sails
20//! it there (`hydra_travel::sail`), in the walk's place, and so does `.go2
21//! <port>` typed aboard a ship. `.sail route <port>` sends nothing: it
22//! keeps the course on the chart (`hydra_map::show_course`) while the
23//! player sails, as `.route2` does a walk's.
24//!
25//! A `.go2` to a place no way by land reaches goes by the character's ship
26//! ([`sea`]): a walk to it, the voyage, and a walk on.
27//!
28//! `.portal` is for a way out that is no compass direction -- a door, a
29//! gate, a stair: `go gate`, or a crossing Hydra walks itself. Alone, with
30//! one such way, it takes it; with several, it lists them numbered, and
31//! `.portal <n>` or `.portal <word>` takes one. Where the map does not know
32//! the room, the things in it that are ways through by their nouns
33//! (`hydra_map::PORTAL_NOUNS`) are the ways: `go door`.
34//!
35//! The map is the one the minimaps share (`hydra_map::shared`), loaded the
36//! first time anyone asks. Each answer reads the character afresh: its
37//! model from the session, its Travel page, and the travel file.
38

63/// `.go2` and `.route2` on the command line of the character the roster
64/// calls `roster_name` (`GAME:Name`), answering on `handle`.
65/// `.go2 to` finds the others among `characters`.
66pub(crate) fn family(
67    handle: SessionHandle,
68    roster_name: String,
69    characters: Characters,
70) -> crate::commands::Handler {
71    Arc::new(move |line: &str, _| {
72        let asked = parse(line)?;
73        let (handle, roster_name) = (handle.clone(), roster_name.clone());
74        let characters = Arc::clone(&characters);
75        crate::commands::spawn(async move {
76            let said = match asked {
77                Ok(command) => answer(&handle, &roster_name, &characters, command).await,
78                Err(why) => Err(why),
79            };
80            let notice = match said {
81                Ok(Said::Lines(lines)) => Notice::prose(NoticeKind::Info, lines),
82                Ok(Said::Table(lines)) => Notice::table(NoticeKind::Info, lines),
83                Err(why) => Notice::line(NoticeKind::Error, format!("Travel: {why}")),
84            };
85            handle.say_unlogged(notice.answering());
86        });
87        Some(Claimed::Done)
88    })
89}

106/// `command`, answered for `roster_name`.
107async fn answer(
108    handle: &SessionHandle,
109    roster_name: &str,
110    characters: &Characters,
111    command: Command,
112) -> Result<Said, String> {
113    let dir = crate::account::data_dir();
114    let (instance, name) = crate::pages::who(roster_name)?;
115    match command {
116        Command::Help => Ok(Said::Table(
117            HELP.iter().map(|&line| line.to_owned()).collect(),
118        )),
119        Command::Stop => Ok(Said::Lines(vec![if stop_walk(roster_name) {
120            "Stopped walking.".to_owned()
121        } else {
122            "Nothing is walking.".to_owned()
123        }])),
124        Command::Sail { port, route } => sail(handle, &dir, roster_name, &port, route).await,
125        Command::Coverage => {
126            let map = map().await?;
127            blocking(move || Said::Table(hydra_travel::coverage::report(&map))).await
128        }
129        Command::List => {
130            let notes = notes(&dir, instance, name)?;
131            Ok(Said::Table(listed(&notes)))
132        }
133        Command::Forget {
134            name: target,
135            global,
136        } => forgotten(&dir, (instance, name), &target, global),
137        Command::Save {
138            name: target,
139            rooms,
140            global,
141        } => {
142            let here = here(handle, &dir, roster_name).await;
143            let map = map().await?;
144            let rooms = rooms_meant(&map, &rooms, here.as_ref().ok().map(|here| here.room))?;
145            let whose = if global {
146                Whose::Everyone
147            } else {
148                Whose::Own(instance, name)
149            };
150            file::save_target(&dir, whose, &target, &rooms).map_err(|why| why.to_string())?;
151            let whose = if global {
152                "every character"
153            } else {
154                "this character"
155            };
156            Ok(Said::Lines(vec![format!(
157                "{target} means {rooms:?}, for {whose}."
158            )]))
159        }
160        Command::Places => {
161            let here = here(handle, &dir, roster_name).await?;
162            blocking(move || {
163                Said::Table(route::places(&here.map, &here.walker, &here.notes.targets))
164            })
165            .await
166        }
167        Command::ClearRoute => {
168            hydra_map::show_route(roster_name, None);
169            Ok(Said::Lines(vec![
170                "The route is off the minimap.".to_owned(),
171            ]))
172        }
173        Command::Route { what, brief } => {
174            let (what, whom) = whither(characters, roster_name, what).await?;
175            shown(handle, &dir, roster_name, what, brief, whom).await
176        }
177        Command::Portal(which) => portal(handle, &dir, roster_name, characters, which).await,
178        Command::Go(what) => {
179            // Aboard a ship, a port is sailed to: there is no walking it.
180            if let Some(port) = port_from_aboard(handle, &what).await {
181                let ocean = hydra_map::ocean().as_ref().map_err(Clone::clone)?;
182                let said = format!("Sailing for {}.", port.name);
183                let bound = hydra_map::Bound {
184                    uid: port.uid,
185                    name: port.name.clone(),
186                    port: true,
187                };
188                set_sail(handle, roster_name, ocean, bound, helm(&dir, roster_name));
189                return Ok(Said::Lines(vec![said]));
190            }
191            // A `.stop` from now stops it, while it is still finding its
192            // way: the other character first, then the map.
193            let claim = Claim::take(roster_name);
194            let (what, whom) = whither(characters, roster_name, what).await?;
195            go(handle, &dir, roster_name, what, whom, claim).await
196        }
197    }
198}

203    dir: &Path,
204    (instance, name): (&str, &str),
205    target: &str,
206    global: bool,
207) -> Result<Said, String> {
208    let whose = if global {
209        Whose::Everyone
210    } else {
211        Whose::Own(instance, name)
212    };
213    let was = file::forget_target(dir, whose, target).map_err(|why| why.to_string())?;
214    let whose = if global {
215        "everyone's"
216    } else {
217        "this character's"
218    };
219    Ok(Said::Lines(vec![if was {
220        format!("{target} is forgotten, of {whose} names.")
221    } else {
222        format!("{target} is not one of {whose} names.")
223    }]))
224}

226/// `.portal`: the ways out of the room that are no compass direction,
227/// listed, or the one `which` names taken -- the only one, when none is
228/// named and there is one.
229async fn portal(
230    handle: &SessionHandle,
231    dir: &Path,
232    roster_name: &str,
233    characters: &Characters,
234    which: Option<String>,
235) -> Result<Said, String> {
236    let ways = ways_out(handle, dir, roster_name).await?;
237    if ways.is_empty() {
238        return Err("no way out of here but the compass's.".to_owned());
239    }
240    let chosen = match which.as_deref().map(str::trim) {
241        None if ways.len() == 1 => 0,
242        None => return Ok(Said::Table(ways_listed(&ways))),
243        Some(word) => way_named(&ways, word)?,
244    };
245    let way = &ways[chosen];
246    if let Some(command) = &way.command {
247        handle.write(command, hydra_session::Origin::Player).await;
248        return Ok(Said::Lines(vec![format!("{}: {command}", way.label)]));
249    }
250    let claim = Claim::take(roster_name);
251    let (what, whom) = whither(characters, roster_name, way.to.to_string()).await?;
252    go(handle, dir, roster_name, what, whom, claim).await
253}

255/// The ways out of the character's room that are no compass direction:
256/// the map's, where it knows the room; else the things in the room that
257/// are ways through by their nouns.
258async fn ways_out(
259    handle: &SessionHandle,
260    dir: &Path,
261    roster_name: &str,
262) -> Result<Vec<hydra_map::WayOut>, String> {
263    let (instance, name) = crate::pages::who(roster_name)?;
264    let notes = notes(dir, instance, name)?;
265    let map = map().await?;
266    let snapshot = handle
267        .snapshot()
268        .await
269        .ok_or_else(|| "the character is not running.".to_owned())?;
270    if let Some(room) = route::here(&map, &snapshot.state, notes.last_room) {
271        let ways = hydra_map::ways_out_of(&map, room);
272        if !ways.is_empty() {
273            return Ok(ways);
274        }
275    }
276    let mut seen = std::collections::BTreeSet::new();
277    Ok(snapshot
278        .state
279        .room
280        .objects
281        .iter()
282        .filter(|thing| hydra_map::PORTAL_NOUNS.contains(&thing.noun.to_lowercase().as_str()))
283        .filter(|thing| seen.insert(thing.noun.to_lowercase()))
284        .map(|thing| hydra_map::WayOut {
285            label: thing.text.clone(),
286            command: Some(format!("go {}", thing.noun)),
287            to: 0,
288        })
289        .collect())
290}

292/// The ways, numbered, each with what takes it.
293fn ways_listed(ways: &[hydra_map::WayOut]) -> Vec<String> {
294    let width = ways.iter().map(|way| way.label.len()).max().unwrap_or(0);
295    ways.iter()
296        .enumerate()
297        .map(|(n, way)| {
298            let takes = way
299                .command
300                .clone()
301                .unwrap_or_else(|| format!("go2 {}", way.to));
302            format!("{:>2}  {:<width$}  {takes}", n + 1, way.label)
303        })
304        .collect()
305}

307/// Which of `ways` `word` names: its number from 1, or the one way whose
308/// label holds the word.
309fn way_named(ways: &[hydra_map::WayOut], word: &str) -> Result<usize, String> {
310    if let Ok(n) = word.parse::<usize>() {
311        return (1..=ways.len())
312            .contains(&n)
313            .then(|| n - 1)
314            .ok_or_else(|| format!("there are {} ways, not {n}.", ways.len()));
315    }
316    let wanted = word.to_lowercase();
317    let found: Vec<usize> = ways
318        .iter()
319        .enumerate()
320        .filter(|(_, way)| way.label.to_lowercase().contains(&wanted))
321        .map(|(at, _)| at)
322        .collect();
323    match found.as_slice() {
324        [one] => Ok(*one),
325        [] => Err(format!("no way out here is {word}: .portal lists them.")),
326        _ => Err(format!(
327            "{word} names more than one way: .portal lists them."
328        )),
329    }
330}

332/// One character's walks: the one under way, and a `.go2` still finding
333/// its way, each with its number and what stops it.
334#[derive(Default)]
335struct Walking {
336    under_way: Option<(u64, CancellationToken)>,
337    setting_out: Option<(u64, CancellationToken)>,
338}
339
340/// Each character's walks, by the roster's name for it in lower case.
341type Walks = Mutex<HashMap<String, Walking>>;
342
343fn walks() -> &'static Walks {
344    static WALKS: OnceLock<Walks> = OnceLock::new();
345    WALKS.get_or_init(Walks::default)
346}

348/// A walk's place among [`walks`], taken the moment `.go2` is asked --
349/// before the map is loaded and the goal found -- so a `.stop` from then on
350/// stops it. The walk under way goes on until this one sets out: a `.go2`
351/// that finds no way stops nothing. Given back when dropped, unless the
352/// walk set out.
353struct Claim {
354    key: String,
355    number: u64,
356    stop: CancellationToken,
357    set_out: bool,
358}
359
360impl Claim {
361    /// Take `roster_name`'s place, stopping any `.go2` still finding its
362    /// way before it.
363    fn take(roster_name: &str) -> Claim {
364        static NEXT: AtomicU64 = AtomicU64::new(1);
365        let number = NEXT.fetch_add(1, Ordering::Relaxed);
366        let stop = CancellationToken::new();
367        let key = roster_name.to_lowercase();
368        if let Ok(mut walks) = walks().lock()
369            && let Some((_, before)) = walks
370                .entry(key.clone())
371                .or_default()
372                .setting_out
373                .replace((number, stop.clone()))
374        {
375            before.cancel();
376        }
377        Claim {
378            key,
379            number,
380            stop,
381            set_out: false,
382        }
383    }
384
385    /// Whether a `.stop`, or another `.go2`, came since it was taken.
386    fn stopped(&self) -> bool {
387        self.stop.is_cancelled()
388    }
389
390    /// The walk sets out: in place of the one under way, which stops.
391    fn set_out(&mut self) {
392        self.set_out = true;
393        if let Ok(mut walks) = walks().lock() {
394            let walking = walks.entry(self.key.clone()).or_default();
395            if walking
396                .setting_out
397                .as_ref()
398                .is_some_and(|(at, _)| *at == self.number)
399            {
400                walking.setting_out = None;
401            }
402            if let Some((_, before)) = walking.under_way.replace((self.number, self.stop.clone())) {
403                before.cancel();
404            }
405        }
406    }
407}
408
409impl Drop for Claim {
410    fn drop(&mut self) {
411        if self.set_out {
412            return;
413        }
414        if let Ok(mut walks) = walks().lock()
415            && let Some(walking) = walks.get_mut(&self.key)
416            && walking
417                .setting_out
418                .as_ref()
419                .is_some_and(|(at, _)| *at == self.number)
420        {
421            walking.setting_out = None;
422        }
423    }
424}

426/// Stop `roster_name`'s walk, under way or still finding its way: whether
427/// there was one.
428pub(crate) fn stop_walk(roster_name: &str) -> bool {
429    let Ok(mut walks) = walks().lock() else {
430        return false;
431    };
432    let Some(walking) = walks.remove(&roster_name.to_lowercase()) else {
433        return false;
434    };
435    let stopped = [walking.under_way, walking.setting_out]
436        .into_iter()
437        .flatten()
438        .map(|(_, stop)| stop.cancel())
439        .count()
440        > 0;
441    if stopped {
442        hydra_map::show_route(roster_name, None);
443        hydra_map::show_course(roster_name, None);
444    }
445    stopped
446}

448/// `.sail <port>`: the ship made ready and sailed there. With `route`,
449/// nothing is sent: the course is kept on the chart while the player
450/// sails. `.sail help` lists the ports.
451async fn sail(
452    handle: &SessionHandle,
453    dir: &Path,
454    roster_name: &str,
455    what: &str,
456    route: bool,
457) -> Result<Said, String> {
458    let ocean = hydra_map::ocean().as_ref().map_err(Clone::clone)?;
459    if matches!(
460        what.trim().to_lowercase().as_str(),
461        "help" | "list" | "ports"
462    ) {
463        return Ok(Said::Table(ports(handle, ocean).await));
464    }
465    let bound = ocean.bound(what).ok_or_else(|| {
466        let ports: Vec<&str> = ocean
467            .ports()
468            .iter()
469            .map(|port| port.name.as_str())
470            .collect();
471        format!(
472            "I know no port called {what}, and no room of the sea numbered so. The ports are: {}.",
473            ports.join(", ")
474        )
475    })?;
476    if !route {
477        let said = format!("Sailing for {}.", bound.name);
478        set_sail(handle, roster_name, ocean, bound, helm(dir, roster_name));
479        return Ok(Said::Lines(vec![said]));
480    }
481    let course = whence(handle, ocean)
482        .await
483        .and_then(|from| ocean.course(from, bound.uid));
484    let said = match course {
485        Some(course) if course.legs.is_empty() => format!("You are at {}.", bound.name),
486        Some(course) => {
487            let legs: Vec<String> = course
488                .legs
489                .iter()
490                .map(|leg| format!("{} {}", leg.heading.word(), leg.rooms))
491                .collect();
492            format!(
493                "The course to {}: {}, then {} ({} rooms, {} turns of the wheel). It is on the chart; .stop takes it off.",
494                bound.name,
495                legs.join(", "),
496                if bound.port { "the port" } else { "the anchor" },
497                course.rooms.len(),
498                course.turns()
499            )
500        }
501        None => format!(
502            "The course to {} will be on the chart once I know where the ship is.",
503            bound.name
504        ),
505    };
506    set_sail(handle, roster_name, ocean, bound, Helm::Player);
507    Ok(Said::Lines(vec![said]))
508}

525/// Sail to `bound` with `helm` at the wheel, in the walk's place, so `.stop`
526/// and another `.go2` stop it; its course on the chart as it goes; and how
527/// it ended said.
528fn set_sail(
529    handle: &SessionHandle,
530    roster_name: &str,
531    ocean: &'static hydra_map::Ocean,
532    bound: hydra_map::Bound,
533    helm: Helm,
534) {
535    let mut claim = Claim::take(roster_name);
536    claim.set_out();
537    let (key, number, stop) = (claim.key.clone(), claim.number, claim.stop.clone());
538    let (handle, character) = (handle.clone(), roster_name.to_owned());
539    hydra_map::show_course(roster_name, None);
540    crate::commands::spawn(async move {
541        let teller = handle.clone();
542        let sailed = hydra_travel::sail::sail(
543            &handle,
544            &stop,
545            ocean,
546            bound.uid,
547            helm,
548            |rooms| {
549                hydra_map::show_course(
550                    &character,
551                    rooms.map(|rooms| hydra_map::ShownCourse {
552                        rooms: rooms.to_vec(),
553                        port: bound.name.clone(),
554                    }),
555                );
556            },
557            |said| {
558                teller.say(Notice::line(NoticeKind::Info, format!("Sailing: {said}")).answering());
559            },
560        )
561        .await;
562        if let Ok(mut walks) = walks().lock()
563            && let Some(walking) = walks.get_mut(&key)
564            && walking
565                .under_way
566                .as_ref()
567                .is_some_and(|(at, _)| *at == number)
568        {
569            walking.under_way = None;
570        }
571        // The wheel was Hydra's, and is given back; or it never was.
572        let yours = if matches!(helm, Helm::Hydra { .. }) {
573            " The wheel is yours."
574        } else {
575            ""
576        };
577        let (kind, said) = match sailed {
578            Sailed::Arrived if bound.port => {
579                (NoticeKind::Info, format!("in port at {}.", bound.name))
580            }
581            Sailed::Arrived => (
582                NoticeKind::Info,
583                format!("at {}, and the anchor is down.{yours}", bound.name),
584            ),
585            Sailed::Stopped => (NoticeKind::Info, format!("stopped.{yours}")),
586            Sailed::Lost => (
587                NoticeKind::Error,
588                "stopped, because the connection went away.".to_owned(),
589            ),
590            Sailed::Halted(why) => (NoticeKind::Error, format!("{why}{yours}")),
591        };
592        handle.say(Notice::line(kind, format!("Sailing: {said}")).answering());
593    });
594}

596/// The room of the ocean the character's ship is in or will depart into,
597/// when anything says: at sea, its room; in port or on a pier, the port's.
598async fn whence(handle: &SessionHandle, ocean: &'static hydra_map::Ocean) -> Option<i64> {
599    let snapshot = handle.snapshot().await?;
600    let state = &snapshot.state;
601    match hydra_map::Navigator::default().fix(ocean, state) {
602        hydra_map::Sailing::At { room, .. } => Some(room),
603        hydra_map::Sailing::Docked { port: Some(port) } => Some(port),
604        // On a pier, or aboard a ship put at one by a dock handler.
605        _ => state
606            .room
607            .id
608            .as_deref()
609            .and_then(|id| id.parse().ok())
610            .and_then(|land| ocean.port_of_pier(land))
611            .or_else(|| ocean.port_of_pier_titled(&state.ship.berth.as_ref()?.pier))
612            .map(|port| port.uid),
613    }
614}

616/// The ports, a line each, and how far each is from where the character
617/// or its ship is, when that is a port or a room of the ocean.
618async fn ports(handle: &SessionHandle, ocean: &'static hydra_map::Ocean) -> Vec<String> {
619    let from = whence(handle, ocean).await;
620    let mut lines = vec![
621        "sail <port>        make your ship ready and sail it there: sail kraken".to_owned(),
622        "sail route <port>  show the course there on the chart as you sail, and send nothing"
623            .to_owned(),
624    ];
625    let width = ocean
626        .ports()
627        .iter()
628        .map(|port| port.name.len())
629        .max()
630        .unwrap_or(0);
631    for port in ocean.ports() {
632        let far = from
633            .and_then(|from| ocean.course(from, port.uid))
634            .map(|course| {
635                if course.legs.is_empty() {
636                    "here".to_owned()
637                } else {
638                    format!(
639                        "{} rooms, {} turns of the wheel",
640                        course.rooms.len(),
641                        course.turns()
642                    )
643                }
644            });
645        lines.push(
646            format!("  {:width$}  {}", port.name, far.unwrap_or_default())
647                .trim_end()
648                .to_owned(),
649        );
650    }
651    lines
652}

654/// The way somewhere: by land to a room, or by the character's ship.
655enum Way {
656    Land(RoomId, Option<Into>),
657    Sea(sea::Passage),

660/// Set out for `what`: the walk runs on, and says how it ended.
661async fn go(
662    handle: &SessionHandle,
663    dir: &Path,
664    roster_name: &str,
665    what: String,
666    whom: Option<String>,
667    claim: Claim,
668) -> Result<Said, String> {
669    let here = here(handle, dir, roster_name).await?;
670    let (instance, name) = crate::pages::who(roster_name)?;
671    let _ = file::save_last_room(dir, instance, name, here.room.0);
672    // `--instability=<room>`, and the place without it.
673    let instability_at = what
674        .split_whitespace()
675        .find_map(|word| word.strip_prefix("--instability=")?.parse::<u32>().ok());
676    let what = what
677        .split_whitespace()
678        .filter(|word| !word.starts_with("--instability="))
679        .collect::<Vec<_>>()
680        .join(" ");
681    let map = Arc::clone(&here.map);
682    let (from, memories, targets) = (
683        here.room,
684        here.notes.memories.clone(),
685        here.notes.targets.clone(),
686    );
687    // The way by sea, should there be none by land.
688    let by_sea = sea::asker(handle, dir, roster_name).await;
689    let goal = blocking(move || {
690        // The Confluence's own words first: no room of the map is them.
691        if let Some(into) = instability::asked(&here.map, here.room, &what, None) {
692            return Ok(Way::Land(plane_room(&here.map), Some(into)));
693        }
694        match route::destination(
695            &here.map,
696            &here.walker,
697            here.room,
698            &what,
699            &here.notes.targets,
700        ) {
701            Meant::Room(goal) => {
702                let into = instability::asked(&here.map, here.room, &what, Some(goal));
703                // No way there: said before setting out, so the walk under
704                // way is not stopped for one that cannot go.
705                if !reachable(&here, goal) && into.is_none() {
706                    return by_sea(&here, &what, format!("I know no way to {what} from here."));
707                }
708                Ok(Way::Land(goal, into))
709            }
710            Meant::Several(rooms) => Err(Said::Table(several(&here.map, &what, &rooms))),
711            Meant::Nowhere => by_sea(
712                &here,
713                &what,
714                format!("I do not know a way to {what} from here."),
715            ),
716        }
717    })
718    .await?;
719    let (goal, into) = match goal {
720        Ok(Way::Land(goal, into)) => (goal, into),
721        // No way by land: the character's ship is the way.
722        Ok(Way::Sea(passage)) => {
723            return sea::set_out(handle.clone(), roster_name.to_owned(), passage, claim);
724        }
725        Err(said) => return Ok(said),
726    };
727    if goal == from && into.is_none() {
728        return Ok(Said::Lines(vec![already(goal, whom.as_deref())]));
729    }
730    let named = map
731        .room(goal)
732        .and_then(|room| room.title.first().cloned())
733        .unwrap_or_default();
734    let setting = Setting {
735        settings: crate::pages::travel_settings(dir, roster_name),
736        memories,
737        last_room: Some(from.0),
738        targets,
739        into,
740        instability_at,
741        instabilities: file::instabilities(dir, instance).unwrap_or_default(),
742        now: std::time::SystemTime::now()
743            .duration_since(std::time::UNIX_EPOCH)
744            .map_or(0, |since| since.as_secs()),
745    };
746    if claim.stopped() {
747        return Ok(Said::Lines(vec!["Stopped before setting out.".to_owned()]));
748    }
749    set_out(
750        handle.clone(),
751        roster_name.to_owned(),
752        map,
753        goal,
754        setting,
755        claim,
756    );
757    let said = match into {
758        Some(Into::Instability) => "Looking for an instability of your element.".to_owned(),
759        Some(Into::Plane | Into::Hot | Into::Cold) => {
760            "Into the Elemental Confluence, by an instability of your element.".to_owned()
761        }
762        Some(Into::Room(_)) | None => match &whom {
763            Some(whom) => format!("Walking to {whom}, in {named} ({}).", goal.0),
764            None => format!("Walking to {named} ({}).", goal.0),
765        },
766    };
767    Ok(Said::Lines(vec![said]))
768}

770/// Whether any way leads from where the character is to `goal`.
771fn reachable(here: &Here, goal: RoomId) -> bool {
772    goal == here.room
773        || here
774            .map
775            .routes(
776                here.room,
777                hydra_travel::Target::Room(goal),
778                route::pricing(&here.walker),
779            )
780            .path_to(goal)
781            .is_some()
782}

784/// A room of the Elemental Confluence: the goal a walk into the plane is
785/// given, though the words typed decide where it ends.
786fn plane_room(map: &Map) -> RoomId {
787    map.rooms()
788        .iter()
789        .find(|room| {
790            room.title
791                .first()
792                .is_some_and(|title| title == instability::PLANE)
793        })
794        .map_or(RoomId(0), |room| room.id)
795}

797/// The walk to `goal`, in a task of its own, in the place `claim` took.
798fn set_out(
799    handle: SessionHandle,
800    roster_name: String,
801    map: Arc<Map>,
802    goal: RoomId,
803    setting: Setting,
804    mut claim: Claim,
805) {
806    claim.set_out();
807    let (key, number, stop) = (claim.key.clone(), claim.number, claim.stop.clone());
808    let (session, character) = (handle.clone(), roster_name.clone());
809    // The walk runs in a task of its own, so one that fails outright is
810    // still noticed: its route comes off and the player is told.
811    let walking = crate::commands::spawn(async move {
812        let dir = crate::account::data_dir();
813        let who = crate::pages::who(&character).ok();
814        drive::walk(
815            &session,
816            &map,
817            goal,
818            setting,
819            &stop,
820            |route| {
821                hydra_map::show_route(
822                    &character,
823                    Some(hydra_map::ShownRoute {
824                        rooms: route.to_vec(),
825                    }),
826                );
827            },
828            |memories| {
829                // Kept at once: a memory lost strands the character.
830                if let Some((instance, name)) = who {
831                    let _ = file::save_memories(&dir, instance, name, memories);
832                }
833            },
834        )
835        .await
836    });
837    crate::commands::spawn(async move {
838        let walked = walking.await;
839        // Only its own: a walk that replaced it is not this one's to end.
840        let ours = walks().lock().is_ok_and(|mut walks| {
841            let Some(walking) = walks.get_mut(&key) else {
842                return false;
843            };
844            let ours = walking
845                .under_way
846                .as_ref()
847                .is_some_and(|(at, _)| *at == number);
848            if ours {
849                walking.under_way = None;
850            }
851            ours
852        });
853        if let Ok(walked) = walked {
854            ended(&handle, &roster_name, &walked, ours);
855        } else {
856            if ours {
857                hydra_map::show_route(&roster_name, None);
858            }
859            handle.say(
860                Notice::line(
861                    NoticeKind::Error,
862                    "Travel: the walk stopped on a fault in Hydra; nothing more is sent.",
863                )
864                .answering(),
865            );
866        }
867    });
868}

870/// What a walk that ended leaves: its route off the minimap when it is
871/// over, the last room and a typeahead learned kept, and a word to the
872/// player when it did not simply arrive.
873fn ended(handle: &SessionHandle, roster_name: &str, walked: &Walked, ours: bool) {
874    let dir = crate::account::data_dir();
875    if let (Some(room), Ok((instance, name))) = (walked.last_room, crate::pages::who(roster_name)) {
876        let _ = file::save_last_room(&dir, instance, name, room.0);
877    }
878    if let (Some(seen), Ok((instance, _))) = (&walked.instabilities, crate::pages::who(roster_name))
879    {
880        let _ = file::save_instabilities(&dir, instance, seen);
881    }
882    if let Some(typeahead) = walked.typeahead {
883        let _ = crate::pages::apply(
884            &dir,
885            &hydra_ui::settings::Change {
886                character: roster_name.to_owned(),
887                page: crate::pages::TRAVEL.to_owned(),
888                key: "typeahead".to_owned(),
889                to: Some(typeahead.to_string()),
890            },
891        );
892    }
893    if ours {
894        hydra_map::show_route(roster_name, None);
895    }
896    let why = match &walked.ended {
897        Ended::Arrived => None,
898        Ended::Stopped if !ours => None,
899        Ended::Stopped => Some("stopped.".to_owned()),
900        Ended::Failed(Why::NoRoute) => {
901            Some("there is no way there that Hydra can walk this character.".to_owned())
902        }
903        Ended::Failed(Why::OffTheMap) => {
904            Some("I do not know what room this is, so I have stopped.".to_owned())
905        }
906        Ended::Failed(Why::TooManyReplans) => {
907            Some("I keep being carried off the route, so I have stopped.".to_owned())
908        }
909        Ended::HandedToLich(line) => Some(format!(
910            "your Lich sent {line}, so I have stopped walking and left it to Lich."
911        )),
912        Ended::Lost => Some("stopped, because the connection went away.".to_owned()),
913        Ended::Halted(why) => Some(why.clone()),
914        Ended::UnknownSpell(spell) => Some(format!(
915            "the map names a spell, {spell}, that is not in the spell table."
916        )),
917    };
918    if let Some(why) = why {
919        let kind = if matches!(walked.ended, Ended::Stopped) {
920            NoticeKind::Info
921        } else {
922            NoticeKind::Error
923        };
924        handle.say(Notice::line(kind, format!("Travel: {why}")).answering());
925    }
926    for said in left_undone(walked) {
927        handle.say(Notice::line(NoticeKind::Warn, format!("Travel: {said}")).answering());
928    }
929    for (from, to) in &walked.wrong_for_the_map {
930        handle.say(Notice::line(
931            NoticeKind::Warn,
932            format!(
933                "Travel: the game says there is no way from {} to {}, which the map has.",
934                from.0, to.0
935            ),
936        ));
937    }
938}

940/// What a walk changed and did not put back, as the player is told it: a
941/// key out of its container, what is still put away, a stance or a
942/// language.
943fn left_undone(walked: &Walked) -> Vec<String> {
944    let mut said = Vec::new();
945    for (name, sent_back) in &walked.still_out {
946        said.push(if *sent_back {
947            format!("your {name} was out; I put it back once, and did not wait to see.")
948        } else {
949            format!("your {name} is out, and I did not put it back.")
950        });
951    }
952    if !walked.still_stored.is_empty() {
953        let names: Vec<&str> = walked
954            .still_stored
955            .iter()
956            .map(|stored| stored.name.as_str())
957            .collect();
958        said.push(format!("still put away: {}.", names.join(", ")));
959    }
960    for (what, before) in [
961        ("stance", &walked.stance_before),
962        ("language", &walked.speech_before),
963    ] {
964        if let Some(before) = before {
965            said.push(format!(
966                "your {what} was {before}, and I did not put it back."
967            ));
968        }
969    }
970    said
971}

973/// The way to `what`, as a table, and on the minimap.
974async fn shown(
975    handle: &SessionHandle,
976    dir: &Path,
977    roster_name: &str,
978    what: String,
979    brief: bool,
980    whom: Option<String>,
981) -> Result<Said, String> {
982    let here = here(handle, dir, roster_name).await?;
983    let (instance, name) = crate::pages::who(roster_name)?;
984    // Where it is, for a tie between rooms that read alike next time.
985    let _ = file::save_last_room(dir, instance, name, here.room.0);
986    let character = roster_name.to_owned();
987    blocking(move || {
988        // No exit of the map goes into the plane: the way is found by
989        // searching, so there is no route to show.
990        if instability::asked(&here.map, here.room, &what, None).is_some() {
991            return Ok(Said::Lines(vec![
992                "The way into the Elemental Confluence is found by searching for an \
993                 instability of your element, so there is no route to show. .go2 confluence \
994                 goes."
995                    .to_owned(),
996            ]));
997        }
998        let meant = route::destination(
999            &here.map,
1000            &here.walker,
1001            here.room,
1002            &what,
1003            &here.notes.targets,
1004        );
1005        let goal = match meant {
1006            Meant::Room(goal) => goal,
1007            Meant::Several(rooms) => return Ok(Said::Table(several(&here.map, &what, &rooms))),
1008            Meant::Nowhere => return Err(format!("I do not know a way to {what} from here.")),
1009        };
1010        let legs = route::itinerary(&here.map, &here.walker, here.room, goal)
1011            .ok_or_else(|| format!("There is no way to {what} ({}) from here.", goal.0))?;
1012        if legs.is_empty() {
1013            return Ok(Said::Lines(vec![already(goal, whom.as_deref())]));
1014        }
1015        let lines = if brief {
1016            vec![in_a_line(&here.map, goal, &legs, whom.as_deref())]
1017        } else {
1018            route::table(&here.map, here.room, &legs)
1019        };
1020        hydra_map::show_route(
1021            &character,
1022            Some(hydra_map::ShownRoute {
1023                rooms: std::iter::once(here.room.0)
1024                    .chain(legs.iter().map(|leg| leg.to.0))
1025                    .collect(),
1026            }),
1027        );
1028        Ok(if brief {
1029            Said::Lines(lines)
1030        } else {
1031            Said::Table(lines)
1032        })
1033    })
1034    .await?
1035}

1045/// The way to `goal` in a line, as a click on the minimap is answered.
1046fn in_a_line(map: &Map, goal: RoomId, legs: &[route::Leg], whom: Option<&str>) -> String {
1047    let title = map
1048        .room(goal)
1049        .and_then(|room| room.title.first().cloned())
1050        .unwrap_or_default();
1051    let seconds = legs.last().map_or(0.0, |leg| leg.so_far);
1052    let to = whom.map_or_else(String::new, |whom| format!("{whom}, in "));
1053    format!(
1054        "The way to {to}{title} ({}): {} step{}, about {seconds:.0} seconds. Click it again to \
1055         walk there.",
1056        goal.0,
1057        legs.len(),
1058        if legs.len() == 1 { "" } else { "s" },
1059    )
1060}

1071/// `what`, with `to <character>` made the number of the room that
1072/// character stands in, and the character's name: another character
1073/// running here, as `.to` names one.
1074async fn whither(
1075    characters: &Characters,
1076    roster_name: &str,
1077    what: String,
1078) -> Result<(String, Option<String>), String> {
1079    let Some(name) = to_whom(&what) else {
1080        return Ok((what, None));
1081    };
1082    let running = characters().await;
1083    let one = crate::relay::pick(name, &running).map_err(|why| why.replacen("To: ", "", 1))?;
1084    let whose = format!("{}:{}", one.game, one.name);
1085    if whose.eq_ignore_ascii_case(roster_name) {
1086        return Err(format!("{} is you.", one.name));
1087    }
1088    let map = map().await?;
1089    let (instance, name) = crate::pages::who(&whose)?;
1090    let notes = notes(&crate::account::data_dir(), instance, name)?;
1091    let snapshot = one
1092        .handle
1093        .snapshot()
1094        .await
1095        .ok_or_else(|| format!("{} is not running.", one.name))?;
1096    let room = route::here(&map, &snapshot.state, notes.last_room)
1097        .ok_or_else(|| format!("I cannot tell which room {} is in.", one.name))?;
1098    Ok((room.0.to_string(), Some(one.name.clone())))
1099}

1101/// Words that fit several rooms: which, as go2 lists them.
1102fn several(map: &Map, what: &str, rooms: &[RoomId]) -> Vec<String> {
1103    let mut lines = vec![format!(
1104        "{what} fits {} rooms; go2 one of them by its number:",
1105        rooms.len()
1106    )];
1107    lines.extend(rooms.iter().take(SEVERAL_SHOWN).map(|room| {
1108        let title = map
1109            .room(*room)
1110            .and_then(|room| room.title.first().cloned())
1111            .unwrap_or_default();
1112        format!("  {:>6}  {title}", room.0)
1113    }));
1114    if rooms.len() > SEVERAL_SHOWN {
1115        lines.push(format!("  ...and {} more.", rooms.len() - SEVERAL_SHOWN));
1116    }
1117    lines
1118}

1132/// The rooms `.go2 save` names: a number, `u<uid>`, or `current`; none
1133/// given means `current`.
1134fn rooms_meant(map: &Map, words: &[String], here: Option<RoomId>) -> Result<Vec<u32>, String> {
1135    let current = || {
1136        here.map(|room| room.0)
1137            .ok_or_else(|| "I cannot tell which room this is.".to_owned())
1138    };
1139    if words.is_empty() {
1140        return Ok(vec![current()?]);
1141    }
1142    words
1143        .iter()
1144        .map(|word| {
1145            if word == "current" {
1146                return current();
1147            }
1148            if let Some(uid) = word
1149                .strip_prefix('u')
1150                .and_then(|digits| digits.parse().ok())
1151            {
1152                return match map.ids_for_uid(Uid(uid)) {
1153                    [room] => Ok(room.0),
1154                    _ => Err(format!("{word} is not one room the map knows.")),
1155                };
1156            }
1157            word.parse()
1158                .ok()
1159                .filter(|id| map.room(RoomId(*id)).is_some())
1160                .ok_or_else(|| format!("{word} is not a room the map knows."))
1161        })
1162        .collect()
1163}

1190/// The character as it is now.
1191async fn here(handle: &SessionHandle, dir: &Path, roster_name: &str) -> Result<Here, String> {

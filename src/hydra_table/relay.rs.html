1//! `.to <character> <command>` and `.all <command>`: a line typed on one
2//! character and sent on another, or on every one.
3//!
4//! Each character has one command line, and a line typed in it goes on that
5//! character. To send a line on another character, or on all of them, the
6//! line is prefaced with a command that says which.
7//!
8//! Two words, kept apart, so a character named All is never a broadcast.
9//! The line goes on each character as if typed there: its own command line
10//! first (`.to Baelor .highlight list` runs on Baelor's), then the game, a
11//! leading command symbol made that character's own. A
12//! character is named in full, or by the start of its name when only one
13//! fits; one whose name is on two games is named with its game, `.to
14//! GSF:Baelor look`, and the name alone is refused rather than sent to
15//! either.
16//!
17//! **`.all` can leave characters out, or name only some**: `.all
18//! -Dicate,Maravel stand` sends on everyone but those two, and `.all
19//! +Nisugi,Dicate stand` on those two alone. Each name is picked as
20//! `.to` picks one; one that picks nobody refuses the whole line, so a typo
21//! never sends on the wrong characters.
22//!
23//! **Only the player's own typing relays**: a line from anything else on the
24//! command line is not taken.

34/// The characters running now: what a relay may send on. A closure, so the
35/// session table stays the binary's.
36pub type Characters =
37    Arc<dyn Fn() -> Pin<Box<dyn Future<Output = Vec<Running>> + Send>> + Send + Sync>;
38
39/// A character running now.
40#[derive(Clone)]
41pub struct Running {
42    /// The game it is on, by its code (`GS3`).
43    pub game: String,
44    /// Its name.
45    pub name: String,
46    /// Its session.
47    pub handle: SessionHandle,
48}
49
50impl Running {
51    /// How the player tells it apart among `running`: its name, with its
52    /// game when another there has the same name.
53    fn label(&self, running: &[Running]) -> String {
54        let twice = running
55            .iter()
56            .filter(|other| other.name.eq_ignore_ascii_case(&self.name))
57            .count()
58            > 1;
59        if twice {
60            format!("{}:{}", self.game, self.name)
61        } else {
62            self.name.clone()
63        }
64    }
65}

67/// How long a relayed line waits for its answer, as a typed one does.
68const DEADLINE: Duration = Duration::from_secs(10);
69
70/// A relay, as typed.
71#[derive(Clone, Debug, PartialEq, Eq)]
72pub(crate) enum Relay {
73    /// `.to <name> <line>`: `line` on the character `name` picks.
74    To {
75        /// The character, in full or its start.
76        name: String,
77        /// What to send on it.
78        line: String,
79    },
80    /// `.all [-names|+names] <line>`: `line` on every running character, or
81    /// all but some, or only some.
82    All {
83        /// Who, of those running.
84        who: Who,
85        /// What to send on each.
86        line: String,
87    },
88}

90/// The relay a line (without its symbol) asks for: `None` when it is not
91/// `to` or `all`; `Err` with how to use it when it is, but says too little.
92fn parse(line: &str) -> Option<Result<Relay, &'static str>> {
93    let line = line.trim_start();
94    let (word, rest) = line.split_once(char::is_whitespace).unwrap_or((line, ""));
95    let rest = rest.trim();
96    if word.eq_ignore_ascii_case("to") {
97        let Some((name, line)) = rest.split_once(char::is_whitespace) else {
98            return Some(Err(
99                "To: name a character and a command, as `to Baelor look`.",
100            ));
101        };
102        return Some(Ok(Relay::To {
103            name: name.to_owned(),
104            line: line.trim().to_owned(),
105        }));
106    }
107    if word.eq_ignore_ascii_case("all") {
108        let (first, after) = rest.split_once(char::is_whitespace).unwrap_or((rest, ""));
109        let names = |list: &str| -> Vec<String> {
110            list.split(',')
111                .map(str::trim)
112                .filter(|name| !name.is_empty())
113                .map(str::to_owned)
114                .collect()
115        };
116        let (who, line) = match (first.strip_prefix('-'), first.strip_prefix('+')) {
117            (Some(list), _) => (Who::Except(names(list)), after.trim()),
118            (_, Some(list)) => (Who::Only(names(list)), after.trim()),
119            _ => (Who::Everyone, rest),
120        };
121        if matches!(&who, Who::Except(list) | Who::Only(list) if list.is_empty()) {
122            return Some(Err(
123                "All: name characters after - or +, as `all -Dicate,Maravel stand`.",
124            ));
125        }
126        if line.is_empty() {
127            return Some(Err(
128                "All: say what every character is to do, as `all stand`.",
129            ));
130        }
131        return Some(Ok(Relay::All {
132            who,
133            line: line.to_owned(),
134        }));
135    }
136    None
137}

139/// `.to` and `.all` on `handle`'s command line, sending on the characters
140/// `characters` lists.
141pub fn family(handle: SessionHandle, characters: Characters) -> crate::commands::Handler {
142    Arc::new(move |line: &str, origin: Origin| {
143        let relay = parse(line)?;
144        if origin != Origin::Player {
145            return None;
146        }
147        let relay = match relay {
148            Ok(relay) => relay,
149            Err(how) => {
150                handle.say(Notice::line(NoticeKind::Error, how).answering());
151                return Some(Claimed::Done);
152            }
153        };
154        let (told, characters) = (handle.clone(), Arc::clone(&characters));
155        tokio::spawn(async move {
156            relay_on(&told, &relay, &characters().await).await;
157        });
158        Some(Claimed::Done)
159    })
160}

162/// The characters running where only `handle`'s does: the terminal client.
163#[must_use]
164pub fn alone(handle: &SessionHandle, game: &str, name: &str) -> Characters {
165    let me = Running {
166        game: game.to_owned(),
167        name: name.to_owned(),
168        handle: handle.clone(),
169    };
170    Arc::new(move || {
171        let me = me.clone();
172        Box::pin(async move { vec![me] })
173    })
174}

176/// Send `relay` on the characters it names among `running`, saying on `told`
177/// -- the character it was typed on -- whatever did not go.
178async fn relay_on(told: &SessionHandle, relay: &Relay, running: &[Running]) {
179    let (targets, line) = match relay {
180        Relay::To { name, line } => match pick(name, running) {
181            Ok(one) => (vec![one.clone()], line),
182            Err(why) => {
183                told.say(Notice::line(NoticeKind::Error, why).answering());
184                return;
185            }
186        },
187        Relay::All { who, line } => match chosen(who, running) {
188            Ok(targets) => (targets, line),
189            Err(why) => {
190                told.say(Notice::line(NoticeKind::Error, why).answering());
191                return;
192            }
193        },
194    };
195    if targets.is_empty() {
196        told.say(
197            Notice::line(
198                NoticeKind::Warn,
199                format!("All: {line} -- nobody is left to send it on."),
200            )
201            .answering(),
202        );
203        return;
204    }
205    if matches!(relay, Relay::All { .. }) {
206        let names: Vec<String> = targets.iter().map(|one| one.label(running)).collect();
207        told.say(
208            Notice::line(
209                NoticeKind::Info,
210                format!("All: {line} -- on {}.", names.join(", ")),
211            )
212            .answering(),
213        );
214    }
215    let mut sending = tokio::task::JoinSet::new();
216    let from = (told.symbol(), told.lich_symbol());
217    for target in targets {
218        let (name, target) = (target.label(running), target.handle);
219        let line = in_its_symbols(line, from, (target.symbol(), target.lich_symbol()));
220        sending.spawn(async move {
221            let outcome = target.send(&line, Origin::Player, DEADLINE).await;
222            (name, outcome)
223        });
224    }
225    while let Some(Ok((name, outcome))) = sending.join_next().await {
226        if let Some(why) = unsent(&outcome) {
227            told.say(Notice::line(NoticeKind::Warn, format!("To {name}: {why}")).answering());
228        }
229    }
230}

232/// `line`, typed with the symbols `from` (Hydra's, if one is set, and
233/// Lich's), as the character with the symbols `to` would type it: `.stop`
234/// typed where Hydra's symbol is `.` is `/stop` where it is `/`, so it
235/// stops that character rather than reaching its game as text.
236fn in_its_symbols(line: &str, from: (Option<char>, char), to: (Option<char>, char)) -> String {
237    if let (Some(ours), Some(theirs)) = (from.0, to.0)
238        && let Some(rest) = line.strip_prefix(ours)
239    {
240        return format!("{theirs}{rest}");
241    }
242    match line.strip_prefix(from.1) {
243        Some(rest) => format!("{}{rest}", to.1),
244        None => line.to_owned(),
245    }
246}

248/// Why a relayed line may not have gone, or `None` when it did.
249fn unsent(outcome: &Outcome) -> Option<&'static str> {
250    match outcome {
251        Outcome::Confirmed(_) | Outcome::Claimed | Outcome::Lich | Outcome::Sent => None,
252        Outcome::Timeout => Some("no answer yet; it may have reached the game."),
253        Outcome::Stopped => Some("stopped before it was sent."),
254        Outcome::Dead => Some("that character's session has ended."),
255        Outcome::Disconnected => {
256            Some("the connection was interrupted; it may not have reached the game.")
257        }
258    }
259}

1//! Who a relay is for: the character `.to` names, and the characters an
2//! `.all` keeps or leaves out.
3
4use super::Running;

6/// Which running characters an `.all` is for.
7#[derive(Clone, Debug, PartialEq, Eq)]
8pub(crate) enum Who {
9    /// Every one.
10    Everyone,
11    /// Every one but these, as typed.
12    Except(Vec<String>),
13    /// Only these, as typed.
14    Only(Vec<String>),
15}
16
17/// The character `typed` picks among `running`: `GAME:Name` on that game,
18/// or a name on whichever game has it. Its whole name, whatever the case,
19/// or else the one whose name starts so.
20///
21/// # Errors
22///
23/// What to tell the player: nobody fits, or more than one does -- a whole
24/// name on two games among them, rather than sending to whichever is found
25/// first.
26pub(crate) fn pick<'a>(typed: &str, running: &'a [Running]) -> Result<&'a Running, String> {
27    let (game, name) = match typed.split_once(':') {
28        Some((game, name)) => (Some(game.trim()), name.trim()),
29        None => (None, typed),
30    };
31    let on_game = || {
32        running
33            .iter()
34            .filter(move |one| game.is_none_or(|game| one.game.eq_ignore_ascii_case(game)))
35    };
36    let exact: Vec<&Running> = on_game()
37        .filter(|one| one.name.eq_ignore_ascii_case(name))
38        .collect();
39    let lower = name.to_lowercase();
40    let fits = if exact.is_empty() {
41        on_game()
42            .filter(|one| one.name.to_lowercase().starts_with(&lower))
43            .collect()
44    } else {
45        exact
46    };
47    let labels = |fits: &[&Running]| {
48        fits.iter()
49            .map(|one| one.label(running))
50            .collect::<Vec<_>>()
51    };
52    match fits.as_slice() {
53        [one] => Ok(one),
54        [] => Err(format!("To: no character named {typed} is running here.")),
55        [first, rest @ ..]
56            if rest
57                .iter()
58                .all(|one| one.name.eq_ignore_ascii_case(&first.name)) =>
59        {
60            Err(format!(
61                "To: {} is on more than one game; say which, as {}.",
62                first.name,
63                labels(&fits).join(" or ")
64            ))
65        }
66        many => Err(format!(
67            "To: {typed} could be {}; say more of the name.",
68            labels(many).join(" or ")
69        )),
70    }
71}

73/// The characters `who` names among `running`, each name picked as `.to`
74/// picks one.
75///
76/// # Errors
77///
78/// A name that picks nobody, or more than one, said as `.to` says it: the
79/// whole line is refused.
80pub(crate) fn chosen(who: &Who, running: &[Running]) -> Result<Vec<Running>, String> {
81    // By place in `running`: two characters are two entries, whatever else
82    // they share.
83    let picked = |names: &[String]| -> Result<Vec<usize>, String> {
84        names
85            .iter()
86            .map(|name| {
87                let one = pick(name, running).map_err(|why| why.replacen("To:", "All:", 1))?;
88                Ok(running
89                    .iter()
90                    .position(|each| std::ptr::eq(each, one))
91                    .unwrap_or(usize::MAX))
92            })
93            .collect()
94    };
95    let keep = |keep: &dyn Fn(usize) -> bool| -> Vec<Running> {
96        running
97            .iter()
98            .enumerate()
99            .filter(|(at, _)| keep(*at))
100            .map(|(_, one)| one.clone())
101            .collect()
102    };
103    Ok(match who {
104        Who::Everyone => running.to_vec(),
105        Who::Except(names) => {
106            let out = picked(names)?;
107            keep(&|at| !out.contains(&at))
108        }
109        Who::Only(names) => {
110            let only = picked(names)?;
111            keep(&|at| only.contains(&at))
112        }
113    })
114}

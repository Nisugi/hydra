1//! `.highlight` (or `.trigger`): the highlights on the command
2//! line, through the same writer as the editor (`hydra_ui::highlight::change`).
3//!
4//! - `list [words]`: the categories, or the highlights whose name, category
5//!   or words hold `words`;
6//! - `on <name>` and `off <name>`: a highlight, or a whole category;
7//! - `remove <name>`;
8//! - `import <file>`: a Wrayth settings file's `<strings>`, `<names>` and
9//!   `<ignores>`.
10//!
11//! The highlights live in `highlights.toml` in the data folder, where every
12//! character's window reads them again within a second of a change.
13

22/// What `.highlight help` says.
23const HELP: &[&str] = &[
24    "highlight list [words]    the categories; or the highlights whose name, category or words \
25     hold the words",
26    "highlight on|off <name>   a highlight, or a whole category",
27    "highlight remove <name>   take a highlight away",
28    "highlight import <file>   a Wrayth settings file's highlights, names and ignores; importing \
29     it again replaces what it brought",
30    "The Highlights button on the hub opens the editor.",
31    "Highlights are kept in highlights.toml in Hydra's data folder; sounds are played from its \
32     sounds folder.",
33];
34
35/// What a line of `.highlight` asks.
36#[derive(Debug, PartialEq, Eq)]
37enum Asked {
38    Help,
39    List(String),
40    Switch(String, bool),
41    Remove(String),
42    Import(PathBuf),
43}
44
45/// `line`, without its symbol, as `.highlight`: `None` when it is not one,
46/// an error when it says what it cannot.
47fn parse(line: &str) -> Option<Result<Asked, String>> {
48    let line = line.trim();
49    let (word, rest) = line.split_once(char::is_whitespace).unwrap_or((line, ""));
50    if !(word.eq_ignore_ascii_case("highlight") || word.eq_ignore_ascii_case("trigger")) {
51        return None;
52    }
53    let rest = rest.trim();
54    let (verb, rest) = rest.split_once(char::is_whitespace).unwrap_or((rest, ""));
55    let rest = rest.trim().trim_matches('"');
56    Some(match verb.to_ascii_lowercase().as_str() {
57        "" | "help" => Ok(Asked::Help),
58        "list" => Ok(Asked::List(rest.to_owned())),
59        "on" | "off" | "remove" if rest.is_empty() => {
60            Err(format!("say which, as `highlight {verb} <name>`."))
61        }
62        "on" => Ok(Asked::Switch(rest.to_owned(), true)),
63        "off" => Ok(Asked::Switch(rest.to_owned(), false)),
64        "remove" => Ok(Asked::Remove(rest.to_owned())),
65        "import" if rest.is_empty() => {
66            Err("say which file, as `highlight import <file>`.".to_owned())
67        }
68        "import" => Ok(Asked::Import(PathBuf::from(rest))),
69        other => Err(format!(
70            "there is no `highlight {other}`; `highlight help` lists what there is."
71        )),
72    })
73}

75/// `.highlight` on a character's command line, saying what it did on
76/// `handle`.
77pub(crate) fn family(handle: SessionHandle) -> crate::commands::Handler {
78    Arc::new(move |line: &str, _| {
79        let asked = parse(line)?;
80        // The usage and a listing have columns; what was done is a sentence.
81        let table = matches!(asked, Ok(Asked::Help | Asked::List(_)));
82        let said = asked.and_then(|asked| run(&crate::account::data_dir(), asked));
83        let notice = match said {
84            Ok(lines) if table => Notice::table(NoticeKind::Info, lines),
85            Ok(lines) => Notice::prose(NoticeKind::Info, lines),
86            Err(why) => Notice::line(NoticeKind::Error, format!("Highlights: {why}")),
87        };
88        handle.say_unlogged(notice.answering());
89        Some(Claimed::Done)
90    })
91}

93/// Do what was `asked`, the highlights file in the data folder `dir`. What
94/// to say, a line each.
95fn run(dir: &Path, asked: Asked) -> Result<Vec<String>, String> {
96    let one = |said: String| vec![format!("Highlights: {said}")];
97    Ok(match asked {
98        Asked::Help => HELP.iter().map(|&line| line.to_owned()).collect(),
99        Asked::List(words) => list(&HighlightFile::load(dir)?, &words),
100        Asked::Switch(name, on) => one(change(dir, |file| file.switch(&name, on))?),
101        Asked::Remove(name) => one(format!(
102            "{} is taken away.",
103            change(dir, |file| file.remove(&name))?
104        )),
105        Asked::Import(file) => vec![import(dir, &file)?],
106    })
107}

109/// The most highlights a list shows.
110const LISTED: usize = 40;
111
112/// The categories, each with how many and whether on; or, given `words`,
113/// the highlights whose name, category or words hold them.
114fn list(file: &HighlightFile, words: &str) -> Vec<String> {
115    if file.rules.is_empty() {
116        return vec!["Highlights: none yet. `highlight import <file>` brings Wrayth's.".to_owned()];
117    }
118    let words = words.trim().to_lowercase();
119    if words.is_empty() {
120        let mut said = vec![format!(
121            "Highlights: {}, in these categories:",
122            file.rules.len()
123        )];
124        for category in file.category_names() {
125            let count = file
126                .rules
127                .values()
128                .filter(|rule| rule.category == category)
129                .count();
130            let off = if file.category_on(&category) {
131                ""
132            } else {
133                " (off)"
134            };
135            said.push(format!("  {category}: {count}{off}"));
136        }
137        said.push("`highlight list <words>` lists the highlights holding them.".to_owned());
138        return said;
139    }
140    let found: Vec<String> = file
141        .rules
142        .iter()
143        .filter(|(name, rule)| {
144            [
145                Some(name.as_str()),
146                Some(rule.category.as_str()),
147                rule.text.as_deref(),
148                rule.regex.as_deref(),
149            ]
150            .into_iter()
151            .flatten()
152            .any(|said| said.to_lowercase().contains(&words))
153        })
154        .map(|(name, rule)| {
155            let looks = rule.regex.as_ref().map_or_else(
156                || format!("\"{}\"", rule.text.as_deref().unwrap_or_default()),
157                |regex| format!("/{regex}/"),
158            );
159            let off = if rule.off || !file.category_on(&rule.category) {
160                " (off)"
161            } else {
162                ""
163            };
164            format!("  {name} [{}]: {looks}{off}", rule.category)
165        })
166        .collect();
167    let mut said = vec![format!("Highlights holding \"{words}\": {}", found.len())];
168    let more = found.len().saturating_sub(LISTED);
169    said.extend(found.into_iter().take(LISTED));
170    if more > 0 {
171        said.push(format!("  and {more} more"));
172    }
173    said
174}

176/// Import the Wrayth settings file `file` into the highlights file in the
177/// data folder `dir`. What was done.
178fn import(dir: &Path, file: &Path) -> Result<String, String> {
179    let xml = std::fs::read_to_string(file)
180        .map_err(|why| format!("{} cannot be read: {why}.", file.display()))?;
181    let shown = file.file_name().map_or_else(
182        || file.display().to_string(),
183        |name| name.to_string_lossy().into_owned(),
184    );
185    let origin = format!("Wrayth: {shown}");
186    let (imported, ignores_on) = change(dir, |highlights| {
187        let imported = import_wrayth(highlights, &xml, &origin)
188            .map_err(|why| format!("{}: {why}.", file.display()))?;
189        Ok((imported, highlights.category_on("Wrayth ignores")))
190    })
191    .map_err(|why| format!("{why} Nothing was imported."))?;
192    let [strings, names, ignores] = imported.counts;
193    let mut said = format!(
194        "Highlights: from {shown}, {strings} highlight(s), {names} name(s) and {ignores} ignore(s)"
195    );
196    if imported.replaced > 0 {
197        let _ = write!(
198            said,
199            ", in place of the {} it brought before",
200            imported.replaced
201        );
202    }
203    said.push('.');
204    if ignores > 0 && !ignores_on {
205        said.push_str(" The ignores are off, as Wrayth had them.");
206    }
207    if imported.sounds > 0 {
208        let _ = write!(
209            said,
210            " {} play a sound: put the files in {}.",
211            imported.sounds,
212            dir.join("sounds").display()
213        );
214    }
215    if !imported.notes.is_empty() {
216        let more = imported.notes.len().saturating_sub(3);
217        let _ = write!(
218            said,
219            " Left out: {}{}.",
220            imported.notes[..imported.notes.len().min(3)].join("; "),
221            if more > 0 {
222                format!("; and {more} more")
223            } else {
224                String::new()
225            }
226        );
227    }
228    Ok(said)
229}

235    #[test]
236    fn the_words_parse() {
237        assert_eq!(parse("look"), None);
238        assert_eq!(parse("highlight"), Some(Ok(Asked::Help)));
239        assert_eq!(
240            parse("trigger import \"C:/Wrayth/GS4/70682.xml\""),
241            Some(Ok(Asked::Import(PathBuf::from("C:/Wrayth/GS4/70682.xml"))))
242        );
243        assert!(matches!(parse("highlight import"), Some(Err(_))));
244        assert_eq!(
245            parse("highlight list"),
246            Some(Ok(Asked::List(String::new())))
247        );
248        assert_eq!(
249            parse("highlight off Wrayth names"),
250            Some(Ok(Asked::Switch("Wrayth names".to_owned(), false)))
251        );
252        assert!(matches!(parse("highlight remove"), Some(Err(_))));
253        assert!(matches!(parse("highlight fly"), Some(Err(_))));
254    }

256    #[test]
257    fn an_import_is_written_and_a_broken_file_is_never_written_over() {
258        let dir =
259            std::env::temp_dir().join(format!("hydra-highlight-import-{}", std::process::id()));
260        let _ = std::fs::remove_dir_all(&dir);
261        std::fs::create_dir_all(&dir).unwrap();
262        let xml = dir.join("70682.xml");
263        std::fs::write(
264            &xml,
265            "<settings><strings><h color=\"#ff8000\" line=\"y\" text=\"[LNet]-\"/></strings>\
266             <ignores disable='y'><h text=\"erratically\"/></ignores></settings>",
267        )
268        .unwrap();
269        let said = import(&dir, &xml).unwrap();
270        assert!(said.contains("1 highlight(s)"), "{said}");
271        assert!(said.contains("ignores are off"), "{said}");
272        let file = HighlightFile::load(&dir).unwrap();
273        assert_eq!(file.rules.len(), 2);
274        let said = import(&dir, &xml).unwrap();
275        assert!(said.contains("in place of the 2"), "{said}");
276        let listed = run(&dir, Asked::List(String::new())).unwrap();
277        assert!(
278            listed
279                .iter()
280                .any(|line| line.contains("Wrayth ignores: 1 (off)")),
281            "{listed:?}"
282        );
283        let listed = run(&dir, Asked::List("lnet".to_owned())).unwrap();
284        assert_eq!(listed.len(), 2, "{listed:?}");
285        run(&dir, Asked::Switch("[lnet]-".to_owned(), false)).unwrap();
286        assert!(HighlightFile::load(&dir).unwrap().rules["[LNet]-"].off);
287        run(&dir, Asked::Remove("[LNet]-".to_owned())).unwrap();
288        assert!(run(&dir, Asked::Remove("[LNet]-".to_owned())).is_err());
289
290        std::fs::write(
291            dir.join(hydra_ui::highlight::FILE),
292            "[highlight.x]\nlook = 3\n",
293        )
294        .unwrap();
295        assert!(import(&dir, &xml).is_err());
296        assert_eq!(
297            std::fs::read_to_string(dir.join(hydra_ui::highlight::FILE)).unwrap(),
298            "[highlight.x]\nlook = 3\n"
299        );
300        let _ = std::fs::remove_dir_all(&dir);
301    }

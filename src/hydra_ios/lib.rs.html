1//! Hydra inside the HydraFE app: the table, every character's session and
2//! the web page, run on the phone and served to it alone, through a few C
3//! calls (`include/hydra.h`) the app makes from Swift.
4//!
5//! - `hydra_start(data_dir)`: start Hydra (once), its data under the app's
6//!   folder, and answer `{"url": ...}` once its page is served on
7//!   `127.0.0.1`, or `{"error": ...}`.
8//! - `hydra_stop()`: quit every character in order and stop.
9//! - `hydra_status()`: `{"running": ..., "url": ..., "characters": ...}`.
10//! - `hydra_asleep()`, `hydra_awake()`: the app left the screen, or came
11//!   back to it (`run`).
12//! - `hydra_string_free(s)`: a string these returned, given back.
13//!
14//! Hydra's messages go to the app's standard error (Xcode's console) and to
15//! `hydra.log` in its folder (`log`).
16//!
17//! Each string returned is Hydra's own; the app copies it and gives it back
18//! once. No call panics across the boundary: what fails is said in JSON.
19//!
20//! No Lich runs on a phone (`hydra-table` without `lich`), and no window
21//! is drawn: the page is the phone's frontend.
22
23#![expect(
24    unsafe_code,
25    reason = "the C calls the app makes, and its standard error kept"
26)]
27
28use std::ffi::{CStr, CString, c_char};

33/// `text` as a string Hydra owns until the app gives it back. JSON has no
34/// NUL in it; one would give NULL rather than a panic.
35fn to_c(text: &str) -> *mut c_char {
36    CString::new(text).map_or(std::ptr::null_mut(), CString::into_raw)
37}
38
39/// Start Hydra with what it keeps under `data_dir`, and answer where its
40/// page is.
41///
42/// # Safety
43///
44/// `data_dir` is NULL or a NUL-terminated UTF-8 C string.
45#[unsafe(no_mangle)]
46pub unsafe extern "C" fn hydra_start(data_dir: *const c_char) -> *mut c_char {
47    if data_dir.is_null() {
48        return to_c(&serde_json::json!({ "error": "no data folder was given" }).to_string());
49    }
50    // SAFETY: the caller gives a NUL-terminated C string, as the header says.
51    let dir = unsafe { CStr::from_ptr(data_dir) };
52    let answer = match dir.to_str() {
53        Ok(dir) => match run::start(std::path::Path::new(dir)) {
54            Ok(url) => serde_json::json!({ "url": url }),
55            Err(why) => serde_json::json!({ "error": why }),
56        },
57        Err(why) => serde_json::json!({ "error": format!("the data folder is not UTF-8: {why}") }),
58    };
59    to_c(&answer.to_string())
60}

62/// Quit every character in order and stop Hydra.
63#[unsafe(no_mangle)]
64pub extern "C" fn hydra_stop() {
65    run::stop();
66}
67
68/// Whether Hydra runs, where its page is, and how many characters play.
69#[unsafe(no_mangle)]
70pub extern "C" fn hydra_status() -> *mut c_char {
71    to_c(&run::status().to_string())
72}
73
74/// The app is leaving the screen.
75#[unsafe(no_mangle)]
76pub extern "C" fn hydra_asleep() {
77    run::asleep();
78}
79
80/// The app is on the screen again.
81#[unsafe(no_mangle)]
82pub extern "C" fn hydra_awake() {
83    run::awake();
84}
85
86/// Give back a string one of these calls returned.
87///
88/// # Safety
89///
90/// `text` is NULL, or a string a `hydra_` call returned and not yet given
91/// back; it is not used after.
92#[unsafe(no_mangle)]
93pub unsafe extern "C" fn hydra_string_free(text: *mut c_char) {
94    if !text.is_null() {
95        // SAFETY: it came from `CString::into_raw` in `to_c`, once.
96        drop(unsafe { CString::from_raw(text) });
97    }
98}

100#[cfg(test)]
101mod tests {
102    use super::*;
103
104    /// `text`, read and given back, as the app does.
105    fn read(text: *mut c_char) -> serde_json::Value {
106        assert!(!text.is_null());
107        // SAFETY: a string `to_c` made.
108        let said = unsafe { CStr::from_ptr(text) }.to_str().unwrap().to_owned();
109        // SAFETY: given back once.
110        unsafe { hydra_string_free(text) };
111        serde_json::from_str(&said).unwrap()
112    }
113
114    /// No data folder is said as an error, not a crash.
115    #[test]
116    fn no_data_folder_is_an_error() {
117        // SAFETY: NULL is allowed.
118        let said = read(unsafe { hydra_start(std::ptr::null()) });
119        assert!(said["error"].is_string(), "{said}");
120    }
121
122    /// Started, Hydra serves its page to this device with its pairing, and
123    /// says so; a second start answers the same; stopped, it is not running.
124    #[test]
125    fn hydra_starts_serves_its_page_here_and_stops() {
126        let dir = std::env::temp_dir().join(format!("hydra-ios-{}", std::process::id()));
127        let path = CString::new(dir.to_string_lossy().as_bytes()).unwrap();
128        // SAFETY: a NUL-terminated string.
129        let started = read(unsafe { hydra_start(path.as_ptr()) });
130        let url = started["url"].as_str().unwrap_or_default().to_owned();
131        assert!(url.starts_with("http://127.0.0.1:"), "{started}");
132        assert!(url.contains("/#pairing="), "{started}");
133        // SAFETY: as above.
134        let again = read(unsafe { hydra_start(path.as_ptr()) });
135        assert_eq!(again["url"], url.as_str(), "the same Hydra");
136        let status = read(hydra_status());
137        assert_eq!(status["running"], true, "{status}");
138        assert_eq!(status["characters"], 0);
139        hydra_asleep();
140        hydra_awake();
141        hydra_stop();
142        assert_eq!(read(hydra_status())["running"], false);
143        let _ = std::fs::remove_dir_all(dir);
144    }
145}

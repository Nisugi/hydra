1//! Hydra's messages, kept where they can be read: what Hydra writes to its
2//! standard error comes out of a pipe a line at a time, and each line goes
3//! to the standard error the app had (Xcode's console, while it runs the
4//! app) and to `hydra.log` in Hydra's folder, the one before it kept as
5//! `hydra.log.1`.
6//!
7//! Nothing here hands a line to the system's log: under a debugger the
8//! system's log copies what it is given to standard error, which is this
9//! pipe, so every line would come round again until the pipe filled and
10//! Hydra stopped at its next message.
11
12use std::sync::Once;
13
14/// How large `hydra.log` grows before it is kept as `hydra.log.1` and a
15/// new one begun.
16#[cfg(unix)]
17const LOG_SIZE: u64 = 2 * 1024 * 1024;
18
19/// The pipe is set up once.
20static KEPT: Once = Once::new();
21
22/// From now, each line written to standard error is kept: called with
23/// Hydra's folder the current directory.
24pub(crate) fn keep() {
25    KEPT.call_once(|| {
26        if let Err(why) = piped() {
27            eprintln!("[hydra] its messages are not kept: {why}");
28        }
29    });
30}
31
32/// Standard error made the writing end of a pipe, its reading end read on
33/// a thread of its own.
34#[cfg(unix)]
35fn piped() -> std::io::Result<()> {
36    use std::io::BufRead;
37    use std::os::fd::FromRawFd;
38
39    let mut ends = [0; 2];
40    // SAFETY: `ends` has room for the two descriptors `pipe` makes.
41    if unsafe { libc::pipe(ends.as_mut_ptr()) } != 0 {
42        return Err(std::io::Error::last_os_error());
43    }
44    let [reading, writing] = ends;
45    // SAFETY: standard error is open; its copy is the app's own, kept for
46    // the lines to go on to.
47    let before = unsafe { libc::dup(libc::STDERR_FILENO) };
48    if before < 0 {
49        return Err(std::io::Error::last_os_error());
50    }
51    // SAFETY: both are open descriptors; standard error becomes the
52    // writing end, which is then closed under its first name.
53    unsafe {
54        if libc::dup2(writing, libc::STDERR_FILENO) < 0 {
55            return Err(std::io::Error::last_os_error());
56        }
57        libc::close(writing);
58    }
59    // SAFETY: each is open and owned by nothing else.
60    let (read, mut app) = unsafe {
61        (
62            std::fs::File::from_raw_fd(reading),
63            std::fs::File::from_raw_fd(before),
64        )
65    };
66    std::thread::Builder::new()
67        .name("hydra-log".to_owned())
68        .spawn(move || {
69            let mut file = Kept::open();
70            for line in std::io::BufReader::new(read).lines().map_while(Result::ok) {
71                let line = format!("{} {line}\n", clock());
72                let _ = std::io::Write::write_all(&mut app, line.as_bytes());
73                file.write(&line);
74            }
75        })?;
76    Ok(())
77}

79/// Nothing to pipe where there is no Unix: the messages stay on standard
80/// error.
81#[cfg(not(unix))]
82#[expect(clippy::unnecessary_wraps, reason = "the Unix build's, which can fail")]
83fn piped() -> std::io::Result<()> {
84    Ok(())
85}

87/// `hydra.log`, as it is written.
88#[cfg(unix)]
89struct Kept {
90    file: Option<std::fs::File>,
91    written: u64,
92}
93
94#[cfg(unix)]
95impl Kept {
96    const NAME: &'static str = "hydra.log";
97    const BEFORE: &'static str = "hydra.log.1";
98
99    fn open() -> Self {
100        let file = std::fs::OpenOptions::new()
101            .create(true)
102            .append(true)
103            .open(Self::NAME)
104            .ok();
105        let written = file
106            .as_ref()
107            .and_then(|file| file.metadata().ok())
108            .map_or(0, |meta| meta.len());
109        Self { file, written }
110    }
111
112    /// `line` kept, the file begun again once it has grown to [`LOG_SIZE`].
113    fn write(&mut self, line: &str) {
114        if self.written >= LOG_SIZE {
115            self.file = None;
116            let _ = std::fs::rename(Self::NAME, Self::BEFORE);
117            *self = Self::open();
118        }
119        if let Some(file) = self.file.as_mut()
120            && std::io::Write::write_all(file, line.as_bytes()).is_ok()
121        {
122            self.written += u64::try_from(line.len()).unwrap_or(u64::MAX);
123        }
124    }
125}

127/// The time of day now, UTC, `HH:MM:SS`.
128#[cfg(unix)]
129fn clock() -> String {
130    let seconds = std::time::SystemTime::now()
131        .duration_since(std::time::UNIX_EPOCH)
132        .map_or(0, |since| since.as_secs())
133        % 86_400;
134    format!(
135        "{:02}:{:02}:{:02}",
136        seconds / 3600,
137        seconds / 60 % 60,
138        seconds % 60
139    )
140}

1//! The Survey window: the map a survey (`hydra-survey`) is making, drawn
2//! from its run file as the file grows, so the rooms appear as they are
3//! walked into.
4//!
5//! A run file is read back through the disk reader once a second and its
6//! records made into a picture: each room where its grid position puts
7//! it, a room reached by no direction set beside the room it came from
8//! in the first free slot, every edge walked drawn once in the direction
9//! it was walked, the exits still to try as stubs, a refused exit as a
10//! stub with a cross, and the room the character last entered marked.
11//! A hover on a room tells its title, its uid, what LOCATION guessed, and
12//! what splash, the cast and forage sense found there.
13

25/// Pixels per grid step at the start.
26const ZOOM: f32 = 28.0;
27const ZOOM_RANGE: (f32, f32) = (6.0, 90.0);

29/// The window's own state.
30#[derive(Debug, Default)]
31pub(crate) struct Surveyor {
32    /// Its window is open.
33    pub(crate) open: bool,
34    /// The run file shown; none, the newest.
35    chosen: Option<PathBuf>,
36    /// Where the view is and how close, kept between frames.
37    camera: Option<(Vec2, f32)>,
38    /// The character's room is kept in view as it moves.
39    follow: bool,
40}

42impl Surveyor {
43    /// Draw the window's content: the runs to pick from, and the picture
44    /// of the one chosen.
45    pub(crate) fn show(&mut self, ui: &mut egui::Ui, data: &Path) {
46        let runs = runs(&data.join("survey"));
47        egui::Panel::left(egui::Id::new("surveyor-runs"))
48            .default_size(240.0)
49            .show(ui, |ui| self.runs(ui, &runs));
50        let chosen = self.chosen.clone().or_else(|| runs.first().cloned());
51        egui::CentralPanel::default().show(ui, |ui| {
52            let Some(path) = chosen else {
53                ui.centered_and_justified(|ui| {
54                    ui.weak("No survey yet: .survey start on a character begins one.");
55                });
56                return;
57            };
58            let picture = crate::disk::Disk::hydras()
59                .kept(
60                    &format!("survey:{}", path.display()),
61                    &path,
62                    Some(&crate::repaint(ui.ctx())),
63                    move |path| {
64                        read(path)
65                            .ok()
66                            .map(|records| Picture::from_records(&records))
67                    },
68                )
69                .map(|(_, picture)| picture);
70            match picture {
71                Some(picture) => self.sheet(ui, &picture),
72                None => {
73                    ui.centered_and_justified(|ui| ui.weak("Reading the run."));
74                }
75            }
76        });
77    }

79    /// The runs there are, newest first, and what to keep in view.
80    fn runs(&mut self, ui: &mut egui::Ui, runs: &[PathBuf]) {
81        ui.heading("Survey");
82        ui.add_space(4.0);
83        ui.checkbox(&mut self.follow, "Follow the character");
84        ui.add_space(8.0);
85        ui.label("Runs");
86        egui::ScrollArea::vertical()
87            .id_salt("surveyor-run-list")
88            .show(ui, |ui| {
89                if runs.is_empty() {
90                    ui.weak("None yet.");
91                }
92                for (i, path) in runs.iter().enumerate() {
93                    let name = path
94                        .file_stem()
95                        .and_then(|name| name.to_str())
96                        .unwrap_or_default()
97                        .trim_start_matches("survey-");
98                    let chosen = match &self.chosen {
99                        Some(chosen) => chosen == path,
100                        None => i == 0,
101                    };
102                    if ui.selectable_label(chosen, name).clicked() {
103                        self.chosen = Some(path.clone());
104                        self.camera = None;
105                    }
106                }
107            });
108    }

110    /// The picture, panned by a drag and zoomed by the wheel.
111    fn sheet(&mut self, ui: &mut egui::Ui, picture: &Picture) {
112        let size = ui.available_size().max(vec2(80.0, 60.0));
113        let (rect, response) = ui.allocate_exact_size(size, Sense::click_and_drag());
114        let painter = ui.painter_at(rect);
115        painter.rect_filled(
116            rect,
117            theme::corner(ui.ctx()),
118            theme::color(ui.ctx(), T::MapBackground),
119        );
120        if picture.rooms.is_empty() {
121            minimap::waiting(&painter, rect, "No rooms yet.");
122            return;
123        }
124        let (mut centre, mut zoom) = self.camera.unwrap_or_else(|| (picture.middle(), ZOOM));
125        if self.follow
126            && let Some(here) = &picture.here
127            && let Some(room) = picture.rooms.get(here)
128        {
129            centre = room.at;
130        }
131        if response.dragged() {
132            centre -= response.drag_delta() / zoom;
133            self.follow = false;
134        }
135        if let Some(pointer) = response.hover_pos() {
136            let scroll = ui.input(|i| i.smooth_scroll_delta.y);
137            if scroll != 0.0 {
138                let before = centre + (pointer - rect.center()) / zoom;
139                zoom = (zoom * (scroll / 200.0).exp()).clamp(ZOOM_RANGE.0, ZOOM_RANGE.1);
140                centre = before - (pointer - rect.center()) / zoom;
141            }
142        }
143        self.camera = Some((centre, zoom));
144        let to_screen = |p: Vec2| rect.center() + (p - centre) * zoom;
145        let half = (zoom * 0.28).clamp(2.5, 14.0);
146
147        draw_edges(ui.ctx(), &painter, picture, &to_screen, half);
148        let hovered = draw_rooms(ui.ctx(), &painter, &response, picture, &to_screen, half);
149        if let Some(room) = hovered {
150            response.clone().on_hover_text(room.hover_text());
151        }
152        // The tally, in a corner.
153        painter.text(
154            rect.left_top() + vec2(8.0, 6.0),
155            egui::Align2::LEFT_TOP,
156            format!(
157                "{} rooms, {} edges, {} exits left{}",
158                picture.rooms.len(),
159                picture.edges.len(),
160                picture.untried,
161                if picture.done { ", done" } else { "" }
162            ),
163            egui::FontId::proportional(12.0),
164            theme::color(ui.ctx(), T::Muted),
165        );
166    }
167}

169/// A grid coordinate as drawn: small integers, exact in an f32.
170#[expect(clippy::cast_precision_loss, reason = "grid coordinates are small")]
171fn grid(n: i32) -> f32 {
172    n as f32
173}
174
175/// The edges walked, each once, the way it was walked, and the exits
176/// still to try and the refused as stubs.
177fn draw_edges(
178    ctx: &egui::Context,
179    painter: &egui::Painter,
180    picture: &Picture,
181    to_screen: &dyn Fn(Vec2) -> Pos2,
182    half: f32,
183) {
184    let line = theme::color(ctx, T::MapLine);
185    let muted = theme::color(ctx, T::MapMuted);
186    let accent = theme::color(ctx, T::Accent);
187    // The edges walked.
188    for (from, to) in &picture.edges {
189        if let (Some(a), Some(b)) = (picture.rooms.get(from), picture.rooms.get(to)) {
190            let (p, q) = (to_screen(a.at), to_screen(b.at));
191            painter.line_segment([p, q], Stroke::new(1.2, line));
192            if from != to {
193                arrow_head(painter, p, q, half, line);
194            }
195        }
196    }
197    // The exits still to try, and the refused, as stubs.
198    for room in picture.rooms.values() {
199        let centre = to_screen(room.at);
200        for (i, (cmd, refused)) in room.stubs.iter().enumerate() {
201            let dir = step(cmd).map_or_else(
202                || {
203                    let angle = grid(i32::try_from(i).unwrap_or(0)).mul_add(0.9, 0.4);
204                    vec2(angle.cos(), angle.sin())
205                },
206                |[x, y, _]| vec2(grid(x), grid(y)).normalized(),
207            );
208            let end = centre + dir * half * 2.4;
209            let colour = if *refused { muted } else { accent };
210            painter.line_segment([centre + dir * half, end], Stroke::new(1.5, colour));
211            if *refused {
212                let c = half * 0.35;
213                painter.line_segment(
214                    [end + vec2(-c, -c), end + vec2(c, c)],
215                    Stroke::new(1.2, muted),
216                );
217                painter.line_segment(
218                    [end + vec2(-c, c), end + vec2(c, -c)],
219                    Stroke::new(1.2, muted),
220                );
221            }
222        }
223    }
224}

226/// The rooms; the one under the pointer.
227fn draw_rooms<'a>(
228    ctx: &egui::Context,
229    painter: &egui::Painter,
230    response: &egui::Response,
231    picture: &'a Picture,
232    to_screen: &dyn Fn(Vec2) -> Pos2,
233    half: f32,
234) -> Option<&'a Drawn> {
235    let mut hovered: Option<&Drawn> = None;
236    for room in picture.rooms.values() {
237        let at = to_screen(room.at);
238        let square = egui::Rect::from_center_size(at, Vec2::splat(half * 2.0));
239        let fill = if picture.here.as_deref() == Some(room.sid.as_str()) {
240            theme::color(ctx, T::MapYou)
241        } else if room.guessed {
242            theme::color(ctx, T::MapConnector)
243        } else {
244            theme::color(ctx, T::MapRoom)
245        };
246        painter.rect(
247            square,
248            half * 0.3,
249            fill,
250            Stroke::new(1.0, theme::color(ctx, T::MapRoomEdge)),
251            egui::StrokeKind::Middle,
252        );
253        if room.z != 0 {
254            painter.text(
255                at,
256                egui::Align2::CENTER_CENTER,
257                if room.z > 0 { "▲" } else { "▼" },
258                egui::FontId::proportional((half * 1.2).max(7.0)),
259                theme::color(ctx, T::MapRoomEdge),
260            );
261        }
262        if response
263            .hover_pos()
264            .is_some_and(|pointer| square.expand(2.0).contains(pointer))
265        {
266            hovered = Some(room);
267        }
268    }
269    hovered
270}

272/// The run files in `dir`, newest first.
273fn runs(dir: &Path) -> Vec<PathBuf> {
274    let mut runs: Vec<PathBuf> = std::fs::read_dir(dir)
275        .ok()
276        .into_iter()
277        .flatten()
278        .filter_map(Result::ok)
279        .map(|entry| entry.path())
280        .filter(|path| {
281            path.file_name()
282                .and_then(|name| name.to_str())
283                .is_some_and(|name| name.starts_with("survey-"))
284                && path
285                    .extension()
286                    .is_some_and(|ext| ext.eq_ignore_ascii_case("jsonl"))
287        })
288        .collect();
289    runs.sort();
290    runs.reverse();
291    runs
292}

294/// A room as drawn.
295#[derive(Clone, Debug)]
296struct Drawn {
297    sid: String,
298    /// Where on the grid, as drawn: `x + z * 0.35`, `y - z * 0.35`.
299    at: Vec2,
300    /// Its floor.
301    z: i32,
302    /// Placed beside its room of origin, not by a direction.
303    guessed: bool,
304    title: String,
305    uid: Option<String>,
306    location: Option<String>,
307    /// The exits still to try (`false`) and the refused (`true`), by
308    /// command.
309    stubs: Vec<(String, bool)>,
310    /// What the verbs found, in words.
311    found: Vec<String>,
312}
313
314impl Drawn {
315    fn hover_text(&self) -> String {
316        let mut lines = vec![format!("{} ({})", self.title, self.sid)];
317        if let Some(uid) = &self.uid {
318            lines.push(format!("uid {uid}"));
319        }
320        if let Some(location) = &self.location {
321            lines.push(format!("location: {location}"));
322        }
323        lines.extend(self.found.iter().cloned());
324        let owed: Vec<&str> = self
325            .stubs
326            .iter()
327            .filter(|(_, refused)| !refused)
328            .map(|(cmd, _)| cmd.as_str())
329            .collect();
330        if !owed.is_empty() {
331            lines.push(format!("to try: {}", owed.join(", ")));
332        }
333        let refused: Vec<&str> = self
334            .stubs
335            .iter()
336            .filter(|(_, refused)| *refused)
337            .map(|(cmd, _)| cmd.as_str())
338            .collect();
339        if !refused.is_empty() {
340            lines.push(format!("refused: {}", refused.join(", ")));
341        }
342        lines.join("\n")
343    }
344}

346/// A run as drawn.
347#[derive(Clone, Debug, Default)]
348struct Picture {
349    rooms: BTreeMap<String, Drawn>,
350    /// Each edge walked that arrived, from and to.
351    edges: Vec<(String, String)>,
352    /// The room the character was last in.
353    here: Option<String>,
354    untried: usize,
355    done: bool,
356}
357
358impl Picture {
359    /// The picture of a run's records.
360    fn from_records(records: &[Record]) -> Self {
361        let graph = Graph::from_records(records);
362        let Gathered {
363            mut placed,
364            mut taken,
365            order,
366            refused,
367            mut found,
368            here,
369            edges,
370            stopped,
371        } = gathered(records);
372        let mut guessed: HashSet<String> = HashSet::new();
373        // Rooms no direction placed: beside the room they came from, in the
374        // first free slot around it, widening until one is free.
375        for sid in &order {
376            if placed.contains_key(sid) {
377                continue;
378            }
379            let Some(room) = graph.room(sid) else {
380                continue;
381            };
382            let (from, cmd) = room.via.as_ref().map_or(([0, 0, 0], ""), |via| {
383                (
384                    placed.get(&via.from).copied().unwrap_or([0, 0, 0]),
385                    via.cmd.as_str(),
386                )
387            });
388            let slot = place(from, cmd, &taken).unwrap_or_else(|| free_slot(from, &taken));
389            placed.insert(sid.clone(), slot);
390            taken.insert(slot);
391            guessed.insert(sid.clone());
392        }
393        let mut rooms = BTreeMap::new();
394        for sid in &order {
395            let Some(room) = graph.room(sid) else {
396                continue;
397            };
398            let pos = placed.get(sid).copied().unwrap_or_default();
399            let mut stubs: Vec<(String, bool)> =
400                room.untried().map(|cmd| (cmd.to_owned(), false)).collect();
401            stubs.extend(
402                refused
403                    .get(sid)
404                    .into_iter()
405                    .flatten()
406                    .map(|cmd| (cmd.clone(), true)),
407            );
408            let record = records.iter().find_map(|record| match &record.body {
409                Body::Room(room) if &room.sid == sid => Some(room),
410                _ => None,
411            });
412            rooms.insert(
413                sid.clone(),
414                Drawn {
415                    sid: sid.clone(),
416                    at: vec2(
417                        grid(pos[2]).mul_add(0.35, grid(pos[0])),
418                        grid(pos[2]).mul_add(-0.35, grid(pos[1])),
419                    ),
420                    z: pos[2],
421                    guessed: guessed.contains(sid),
422                    title: record.map(|r| r.title.clone()).unwrap_or_default(),
423                    uid: record.and_then(|r| r.uid.clone()),
424                    location: record.and_then(|r| r.location.clone()),
425                    stubs,
426                    found: found.remove(sid).unwrap_or_default(),
427                },
428            );
429        }
430        let untried = graph.untried();
431        Self {
432            rooms,
433            edges,
434            here,
435            untried,
436            done: stopped && untried == 0,
437        }
438    }
439
440    /// The middle of the rooms drawn.
441    fn middle(&self) -> Vec2 {
442        let (mut min, mut max) = (vec2(f32::MAX, f32::MAX), vec2(f32::MIN, f32::MIN));
443        for room in self.rooms.values() {
444            min = min.min(room.at);
445            max = max.max(room.at);
446        }
447        (min + max) / 2.0
448    }
449}

451/// What one pass over the records gathers.
452struct Gathered {
453    placed: HashMap<String, [i32; 3]>,
454    taken: HashSet<[i32; 3]>,
455    order: Vec<String>,
456    refused: HashMap<String, Vec<String>>,
457    found: HashMap<String, Vec<String>>,
458    here: Option<String>,
459    edges: Vec<(String, String)>,
460    stopped: bool,
461}
462
463/// One pass over the records: where each room is placed, the order they
464/// came in, the refused exits, what the verbs found, the edges, and the
465/// room the character was last in.
466fn gathered(records: &[Record]) -> Gathered {
467    let mut placed: HashMap<String, [i32; 3]> = HashMap::new();
468    let mut taken: HashSet<[i32; 3]> = HashSet::new();
469    let mut order: Vec<String> = Vec::new();
470    let mut refused: HashMap<String, Vec<String>> = HashMap::new();
471    let mut found: HashMap<String, Vec<String>> = HashMap::new();
472    let mut here = None;
473    let mut edges = Vec::new();
474    let mut stopped = false;
475    for record in records {
476        match &record.body {
477            Body::Room(room) => {
478                order.push(room.sid.clone());
479                if let Some(pos) = room.pos {
480                    placed.insert(room.sid.clone(), pos);
481                    taken.insert(pos);
482                }
483                here = Some(room.sid.clone());
484                if let Some(location) = &room.location {
485                    found
486                        .entry(room.sid.clone())
487                        .or_default()
488                        .push(format!("location: {location}"));
489                }
490            }
491            Body::Edge(edge) => {
492                stopped = false;
493                match edge.result {
494                    EdgeResult::Arrived => {
495                        if let Some(to) = &edge.to {
496                            edges.push((edge.from.clone(), to.clone()));
497                            here = Some(to.clone());
498                        }
499                    }
500                    EdgeResult::Failed => refused
501                        .entry(edge.from.clone())
502                        .or_default()
503                        .push(edge.cmd.clone()),
504                    EdgeResult::Stayed => {}
505                }
506            }
507            Body::Verb(verb) => {
508                let said = match verb.verb.as_str() {
509                    probes::SPLASH => probes::splash(&verb.lines).map(|splash| match splash {
510                        probes::Splash::Water => "water: yes",
511                        probes::Splash::None => "water: no",
512                    }),
513                    probes::FORAGE_SENSE => Some(forage_said(&verb.lines)),
514                    v if v.starts_with("incant") => {
515                        probes::magic(&verb.lines).map(|magic| match magic {
516                            probes::Magic::Works => "magic: works",
517                            probes::Magic::Blocked => "magic: blocked",
518                        })
519                    }
520                    _ => None,
521                };
522                if let Some(said) = said {
523                    found
524                        .entry(verb.sid.clone())
525                        .or_default()
526                        .push(said.to_owned());
527                }
528            }
529            Body::Event(event) => {
530                stopped = event.what == Happening::Stopped;
531                if event.what == Happening::Resumed {
532                    here = Some(event.sid.clone());
533                }
534            }
535        }
536    }
537    Gathered {
538        placed,
539        taken,
540        order,
541        refused,
542        found,
543        here,
544        edges,
545        stopped,
546    }
547}

549/// What forage sense found, in a few words.
550fn forage_said(lines: &[String]) -> &'static str {
551    let text = lines.join(" ");
552    if text.contains("should support specimens of") {
553        "forage: herbs"
554    } else if text.contains("doubt that anything interesting could be foraged") {
555        "forage: not here"
556    } else if text.contains("do not spot any forageables") {
557        "forage: none"
558    } else {
559        "forage: ?"
560    }
561}

563/// The first free slot around `from`, nearest first, on the same floor.
564fn free_slot(from: [i32; 3], taken: &HashSet<[i32; 3]>) -> [i32; 3] {
565    let mut queue = VecDeque::from([from]);
566    let mut seen = HashSet::from([from]);
567    while let Some(at) = queue.pop_front() {
568        if !taken.contains(&at) {
569            return at;
570        }
571        for (dx, dy) in [
572            (1, 0),
573            (0, 1),
574            (-1, 0),
575            (0, -1),
576            (1, 1),
577            (-1, 1),
578            (1, -1),
579            (-1, -1),
580        ] {
581            let next = [at[0] + dx, at[1] + dy, at[2]];
582            if seen.insert(next) {
583                queue.push_back(next);
584            }
585        }
586    }
587    from
588}

590/// An arrow head at `q`, pointing from `p`, drawn short of the room.
591fn arrow_head(painter: &egui::Painter, p: Pos2, q: Pos2, half: f32, colour: egui::Color32) {
592    let dir = (q - p).normalized();
593    let tip = q - dir * half * 1.1;
594    let side = vec2(-dir.y, dir.x) * half * 0.35;
595    let back = tip - dir * half * 0.7;
596    painter.line_segment([tip, back + side], Stroke::new(1.2, colour));
597    painter.line_segment([tip, back - side], Stroke::new(1.2, colour));
598}

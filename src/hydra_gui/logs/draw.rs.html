2//! What it asks to read goes on the window's list for the app to hand on.

8impl Logs {
9    /// Draw the window into `ui`.
10    pub(crate) fn show(&mut self, ui: &mut egui::Ui) {
11        self.take_replies();
12        ui.horizontal(|ui| {
13            for (tab, label) in [
14                (Tab::Recent, "Recent"),
15                (Tab::Search, "Search"),
16                (Tab::Export, "Export"),
17            ] {
18                ui.selectable_value(&mut self.tab, tab, label);
19            }
20            if self.waiting > 0 {
21                ui.spinner();
22            }
23        });
24        ui.separator();
25        match self.tab {
26            Tab::Recent => self.recent(ui),
27            Tab::Search => self.search(ui),
28            Tab::Export => self.export(ui),
29        }
30    }

32    /// A day, whole: the day picker, the filters, the lines.
33    fn recent(&mut self, ui: &mut egui::Ui) {
34        ui.horizontal(|ui| {
35            ui.label("Day");
36            let chosen = self.day.clone().unwrap_or_default();
37            let mut picked = None;
38            egui::ComboBox::from_id_salt(("log-day", &self.character))
39                .selected_text(self.day_label(&chosen))
40                .show_ui(ui, |ui| {
41                    for (day, _) in &self.days {
42                        if ui
43                            .selectable_label(*day == chosen, self.day_label(day))
44                            .clicked()
45                        {
46                            picked = Some(day.clone());
47                        }
48                    }
49                });
50            if let Some(day) = picked.filter(|day| *day != chosen) {
51                self.asked.push(Ask::Day(day.clone()));
52                self.day = Some(day);
53                self.lines.clear();
54                self.generation += 1;
55            }
56            if ui
57                .button("Refresh")
58                .on_hover_text("Read the days and this day again")
59                .clicked()
60            {
61                self.asked.push(Ask::Days);
62                if let Some(day) = self.day.clone() {
63                    self.asked.push(Ask::Day(day));
64                }
65            }
66        });
67        self.filters(ui);
68        let lines = self
69            .day_rows
70            .of(&self.lines, self.generation, &self.ticked, self.dedup);
71        self.clicked.extend(lines_area(ui, lines, false, true));
72        self.footer(ui);
73    }

75    /// The presets, *Dedup*, and a box per class the window has seen.
76    fn filters(&mut self, ui: &mut egui::Ui) {
77        ui.horizontal(|ui| {
78            for preset in Preset::ALL {
79                if ui.button(preset.label()).clicked() {
80                    self.preset(preset);
81                }
82            }
83            ui.checkbox(&mut self.dedup, "Dedup")
84                .on_hover_text("Show a run of the same line once, with its count. Nothing is left out of an export.");
85        });
86        ui.horizontal_wrapped(|ui| {
87            for (class, on) in &mut self.ticked {
88                ui.checkbox(on, class.as_str());
89            }
90        });
91        ui.separator();
92    }

94    /// Lines holding a text, newest first.
95    fn search(&mut self, ui: &mut egui::Ui) {
96        ui.horizontal(|ui| {
97            let typed = ui.add(
98                egui::TextEdit::singleline(&mut self.query)
99                    .hint_text("Text to find, any case")
100                    .desired_width(320.0),
101            );
102            ui.checkbox(&mut self.regex, "Regex")
103                .on_hover_text("A regular expression, where case matters unless it says (?i)");
104            let entered = typed.lost_focus() && ui.input(|i| i.key_pressed(egui::Key::Enter));
105            if (ui.button("Search").clicked() || entered) && !self.query.trim().is_empty() {
106                let query = self.query.trim().to_owned();
107                self.ask_search(&query, self.regex);
108            }
109        });
110        self.filters(ui);
111        if let Some(found) = &self.found {
112            let lines =
113                self.found_rows
114                    .of(&found.entries, self.generation, &self.ticked, self.dedup);
115            let more = if found.more {
116                ", and more past the first 1000"
117            } else {
118                ""
119            };
120            ui.label(format!("{} found{more}, newest first", lines.len()));
121            self.clicked.extend(lines_area(ui, lines, true, false));
122        }
123        self.footer(ui);
124    }

126    /// Days from one to another written to a file.
127    fn export(&mut self, ui: &mut egui::Ui) {
128        ui.label("Write every line of the days chosen, of the ticked kinds, to one text file.");
129        egui::Grid::new(("log-export", &self.character)).show(ui, |ui| {
130            for (label, value) in [("From", &mut self.from), ("To", &mut self.to)] {
131                ui.label(label);
132                egui::ComboBox::from_id_salt(("log-export", label, &self.character))
133                    .selected_text(value.as_str())
134                    .show_ui(ui, |ui| {
135                        for (day, _) in &self.days {
136                            ui.selectable_value(value, day.clone(), day);
137                        }
138                    });
139                ui.end_row();
140            }
141        });
142        self.filters(ui);
143        if ui.button("Create log file").clicked() && !self.from.is_empty() {
144            self.asked.push(Ask::Export(
145                self.from.clone(),
146                self.to.clone(),
147                self.export_streams(),
148            ));
149        }
150        if ui.button("Open exports folder").clicked() {
151            self.said = open_folder(&self.folder().join("exports")).err();
152        }
153        self.footer(ui);
154    }

156    /// What was said, the disk used, and the folder.
157    fn footer(&mut self, ui: &mut egui::Ui) {
158        ui.separator();
159        ui.horizontal(|ui| {
160            if ui.button("Open logs folder").clicked() {
161                self.said = open_folder(&self.folder()).err();
162            }
163            if let Some(usage) = &self.usage {
164                ui.label(format!(
165                    "{} days · {} in all, {} plain, {} archived",
166                    usage.days,
167                    size(usage.total()),
168                    size(usage.plain),
169                    size(usage.archived)
170                ));
171            }
172        });
173        // Lines the log fell behind on: the days read are not whole.
174        if self.dropped > 0 {
175            ui.colored_label(
176                crate::theme::color(ui.ctx(), crate::theme::T::Wrong),
177                format!(
178                    "{} lines this session came faster than the log could keep, and were not \
179                     recorded.",
180                    self.dropped
181                ),
182            );
183        }
184        // An archive that cannot be read: its days are not listed.
185        for why in self.usage.iter().flat_map(|usage| &usage.unreadable) {
186            ui.colored_label(
187                crate::theme::color(ui.ctx(), crate::theme::T::Wrong),
188                format!("Could not read an archive, so its days are not listed: {why}"),
189            );
190        }
191        if let Some(said) = &self.said {
192            ui.label(said.as_str());
193        }
194    }

196    /// A day as the picker shows it: with its size, or that it is archived.
197    pub(super) fn day_label(&self, day: &str) -> String {
198        match self.days.iter().find(|(d, _)| d == day) {
199            Some((_, Some(bytes))) => format!("{day} · {}", size(*bytes)),
200            Some((_, None)) => format!("{day} · archived"),
201            None if day.is_empty() => "none kept yet".to_owned(),
202            None => day.to_owned(),
203        }
204    }
205}

207/// The lines, as many as fit drawn: `time [class] text`, the day first when
208/// they span days; the newest kept in view when `follow`. Their links are
209/// drawn as the story draws them: the one clicked, if one was.
210fn lines_area(
211    ui: &mut egui::Ui,
212    lines: super::view::Rows<'_>,
213    with_day: bool,
214    follow: bool,
215) -> Option<hydra_ui::RunLink> {
216    let mut clicked = None;
217    let height = ui.text_style_height(&egui::TextStyle::Monospace) + ui.spacing().item_spacing.y;
218    egui::ScrollArea::both()
219        .auto_shrink(false)
220        .stick_to_bottom(follow)
221        .max_height(ui.available_height() - 64.0)
222        .show_rows(ui, height, lines.len(), |ui, rows| {
223            for (entry, count) in rows.filter_map(|row| lines.get(row)) {
224                ui.horizontal(|ui| {
225                    let at = entry.at.get(..8).unwrap_or(&entry.at);
226                    let when = if with_day {
227                        format!("{} {at}", entry.day)
228                    } else {
229                        at.to_owned()
230                    };
231                    ui.label(egui::RichText::new(when).monospace().weak());
232                    ui.label(
233                        egui::RichText::new(class(&entry.stream))
234                            .small()
235                            .background_color(ui.visuals().faint_bg_color),
236                    )
237                    .on_hover_text(&entry.stream);
238                    if entry.links.is_empty() {
239                        let text = if count > 1 {
240                            format!("{}  ×{count}", entry.text)
241                        } else {
242                            entry.text.clone()
243                        };
244                        ui.add(egui::Label::new(egui::RichText::new(text).monospace()).extend());
245                    } else {
246                        let mut runs = runs(entry);
247                        if count > 1 {
248                            runs.push(mono(&format!("  ×{count}"), None));
249                        }
250                        let job =
251                            crate::text::job(&runs, ui.style(), &crate::theme::palette(ui.ctx()));
252                        if let Some(crate::text::Acted::Clicked(link, _)) =
253                            crate::text::linked(ui, job, &runs)
254                        {
255                            clicked = Some(link);
256                        }
257                    }
258                });
259            }
260        });
261    clicked
262}

264/// A logged line as runs, each link its own run.
265fn runs(entry: &hydra_session::player_log::reader::Entry) -> Vec<hydra_ui::StyledRun> {
266    use hydra_session::player_log::Does;
267    let text = &entry.text;
268    let mut runs = Vec::new();
269    let mut at = 0;
270    for link in &entry.links {
271        if link.from > at {
272            runs.push(mono(&text[at..link.from], None));
273        }
274        let does = match &link.does {
275            Does::Object { exist, noun, coord } => hydra_ui::RunLink::Object {
276                exist: exist.clone(),
277                noun: noun.clone(),
278                coord: coord.clone(),
279            },
280            Does::Command { command } => hydra_ui::RunLink::Command {
281                command: command.clone(),
282            },
283            Does::Url { href } => hydra_ui::RunLink::Url { href: href.clone() },
284        };
285        runs.push(mono(&text[link.from..link.to], Some(does)));
286        at = link.to;
287    }
288    if at < text.len() {
289        runs.push(mono(&text[at..], None));
290    }
291    runs
292}

294/// `text` in the log's fixed-width face, a link when `link`.
295fn mono(text: &str, link: Option<hydra_ui::RunLink>) -> hydra_ui::StyledRun {
296    hydra_ui::StyledRun {
297        monospace: true,
298        link,
299        ..hydra_ui::StyledRun::plain(text)
300    }

1//! A widget: one thing a player reads about one character, drawn bare -- no
2//! title, no frame -- into the space it is given. Its frame comes from what
3//! holds it: a standalone window gives it one, and a custom window holds it
4//! bare beside others.
5//!
6//! Widgets are individual things: a bar for health is a widget, a bar for
7//! stamina is another, and each hand is its own. So the catalog is of small
8//! things, and composite panels are presets built from them.
9//!
10//! A closed enum, not a trait: every kind lives here, and a trait object
11//! would buy nothing. Each kind says its name, the size it would like, and
12//! draws itself; what holds it decides where.

59/// What a widget draws from: one character, as its play window has it.
60#[derive(Clone, Copy)]
61pub(crate) struct Seen<'a> {
62    /// What it knows, once its feed has seen anything.
63    pub(crate) snapshot: Option<&'a Snapshot>,
64    /// Its story and Hydra's messages.
65    pub(crate) story: &'a Story,
66    /// Whose it is, when the widget follows another character than its
67    /// window's, which it then names.
68    pub(crate) who: Option<&'a str>,
69    /// The streams a widget in this window shows, whose lines the story
70    /// then leaves out.
71    pub(crate) open: &'a [String],
72    /// A Hydra widget in this window shows Hydra's messages, which the
73    /// story then leaves out; without one the story shows them.
74    pub(crate) hydra: bool,
75    /// How long each creature's tag after its name is, when shown
76    /// (`hydra_model::targetid`); `None` until the T commands that set it
77    /// are built.
78    pub(crate) tags: Option<hydra_model::targetid::Style>,
79    /// Where it is on the map, as its minimap draws it; `None` with no map.
80    pub(crate) minimap: Option<&'a hydra_map::MinimapView>,
81    /// Its saved places and the way travel shows, for a Travel widget;
82    /// `None` on another character's window.
83    pub(crate) travel: Option<&'a travel::TravelView>,
84    /// Its skill trainer, for a Skill trainer widget; `None` on another
85    /// character's window.
86    pub(crate) trainer: Option<&'a crate::trainer::TrainerView>,
87}

89/// Another character Hydra runs, as a widget that follows it sees
90/// it: a party's vitals, say, in the Advanced place. Never its story, which no
91/// other window shows.
92#[derive(Clone, Debug)]
93pub(crate) struct Character {
94    /// Its name.
95    pub(crate) name: String,
96    /// The game it is on, by its code (`GS3`): one name may be on two.
97    pub(crate) game: String,
98    /// What it knows, once its feed has seen anything.
99    pub(crate) snapshot: Option<Arc<Snapshot>>,
100    /// Where it is on the map, as a minimap draws it.
101    pub(crate) minimap: Option<hydra_map::MinimapView>,
102}

104impl Character {
105    /// How a widget that follows it names it, among `all`: its name, or
106    /// `GAME:Name` when another of them has the same name.
107    pub(crate) fn key(&self, all: &[Character]) -> String {
108        let shared = all
109            .iter()
110            .filter(|other| other.name.eq_ignore_ascii_case(&self.name))
111            .count()
112            > 1;
113        if shared {
114            format!("{}:{}", self.game, self.name)
115        } else {
116            self.name.clone()
117        }
118    }
119
120    /// Whether `who`, a name or `GAME:Name`, is this character.
121    pub(crate) fn is(&self, who: &str) -> bool {
122        match who.split_once(':') {
123            Some((game, name)) => {
124                game.eq_ignore_ascii_case(&self.game) && name.eq_ignore_ascii_case(&self.name)
125            }
126            None => who.eq_ignore_ascii_case(&self.name),
127        }
128    }
129}

131impl Widget {
132    /// How much it has said, ever, when it is a stream of lines: a tab not
133    /// showing counts what came since it last did. `None`
134    /// for the rest, which have nothing to count.
135    pub(crate) fn count(&self, seen: &Seen<'_>) -> Option<u64> {
136        match self {
137            Widget::Story => Some(seen.story.heard),
138            Widget::Hydra => Some(seen.story.told),
139            Widget::Stream(id) => Some(seen.story.streams.get(id).map_or(0, |kept| kept.heard)),
140            _ => None,
141        }
142    }
143
144    /// Draw it into `ui`, bare, filling what it is given. `id` is its own,
145    /// so two of one kind keep apart what they remember -- a scroll, say.
146    /// A line the player asked it to send, as if typed -- a compass's
147    /// direction -- if one was.
148    #[cfg(test)]
149    pub(crate) fn draw(&self, ui: &mut egui::Ui, seen: &Seen<'_>, id: Id) -> Option<Clicked> {
150        draw::draw(self, ui, seen, id, &Chosen::default())
151    }
152
153    /// [`Self::draw`], as the player chose on its own page (each kind's own
154    /// otherwise).
155    pub(crate) fn draw_with(
156        &self,
157        ui: &mut egui::Ui,
158        seen: &Seen<'_>,
159        id: Id,
160        chosen: &Chosen,
161    ) -> Option<Clicked> {
162        draw::draw(self, ui, seen, id, chosen)
163    }
164}

166/// What a click in a widget asked for.
167#[derive(Clone, Debug, PartialEq)]
168pub(crate) enum Clicked {
169    /// A line to send, as if typed: a compass's direction.
170    Send(String),
171    /// A line the game gave, to send when clicked: a dialog's button, its
172    /// own quick link. Checked before it goes ([`Asked::FromGame`]).
173    ///
174    /// [`Asked::FromGame`]: crate::play::Asked::FromGame
175    FromGame(String),
176    /// A line to send without an echo: an object carried and let go,
177    /// `_drag #<item> <onto>` (`carry.rs`).
178    Quietly(String),
179    /// A link in a line of the game's, clicked at a place on the screen.
180    Link(hydra_ui::RunLink, egui::Pos2),
181    /// A setting of the widget's own page changed from the widget itself:
182    /// the doll's facing from the slider under it. The window applies it
183    /// as the settings page would (`play/options.rs`).
184    Set {
185        /// The page's key.
186        key: String,
187        /// Its new value, as the page writes it.
188        to: String,
189    },
190    /// A change to the character's tracked effects, from an effect row's
191    /// menu (`crate::tracked`).
192    Track(crate::tracked::Change),
193    /// A room of the map said in the story, from the minimap: its number
194    /// and title alone when `brief`.
195    Room {
196        /// The map's room number.
197        id: u32,
198        /// Only its number and title.
199        brief: bool,
200    },
201    /// A line for the character's own command line, its command symbol
202    /// put before it: the minimap's `route2` and `go2`.
203    Command(String),
204    /// What the Skill trainer widget asks of the character's trainer.
205    Trainer(crate::trainer::Ask),
206    /// A hotbar's button in slot `from` moved to slot `to`.
207    MoveButton {
208        /// The bar, by name.
209        bar: String,
210        from: u16,
211        to: u16,
212    },
213}

215impl From<crate::text::Acted> for Clicked {
216    fn from(acted: crate::text::Acted) -> Self {
217        match acted {
218            crate::text::Acted::Clicked(link, at) => Self::Link(link, at),
219            crate::text::Acted::Quietly(line) => Self::Quietly(line),
220        }
221    }
222}

261    pub(crate) profiles: Option<Vec<String>>,
262}

283    /// One name on two games is told apart by its game; a name alone
284    /// stays as it was.
285    #[test]
286    fn a_name_on_two_games_is_named_with_its_game() {
287        let all = [
288            on("GS3", "Nisugi"),
289            on("GSF", "Nisugi"),
290            on("GS3", "Baelor"),
291        ];
292        assert_eq!(all[0].key(&all), "GS3:Nisugi");
293        assert_eq!(all[1].key(&all), "GSF:Nisugi");
294        assert_eq!(all[2].key(&all), "Baelor");
295        assert!(all[1].is("GSF:Nisugi"));
296        assert!(!all[0].is("GSF:Nisugi"));
297        assert!(all[2].is("baelor"), "a bare name, as layouts kept it");
298    }

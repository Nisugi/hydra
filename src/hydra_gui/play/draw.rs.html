88impl Drawing<'_> {
89    /// What the player chose for the widget placed as `placed` on its own
90    /// page.
91    fn chosen(&self, placed: &Placed) -> crate::widget::Chosen {
92        let (placed, widget) = (placed.id, &placed.widget);
93        crate::widget::Chosen {
94            look: self
95                .looks
96                .get(&placed)
97                .cloned()
98                .zip(widget.bar_token())
99                .map(|(look, token)| look.themed(token)),
100            room: self.rooms.get(&placed).copied(),
101            lines: self.lines.get(&placed).copied(),
102            doll: self.dolls.get(&placed).cloned(),
103            // The window's own character's aim is remembered; another's
104            // is only what its own window has heard.
105            aim: (*widget == crate::widget::Widget::Aim).then(|| crate::widget::aim::AimShown {
106                look: self.aims.get(&placed).copied().unwrap_or_default(),
107                remembered: (!self.follows.contains_key(&placed))
108                    .then(|| self.remembered_aim.map(str::to_owned))
109                    .flatten(),
110            }),
111            list: self.lists.get(&placed).copied(),
112            label: self.labels.get(&placed).cloned(),
113            roster: self.rosters.get(&placed).cloned(),
114            quick: self.quicks.get(&placed).cloned(),
115            hotbar: (*widget == crate::widget::Widget::Hotbar).then(|| {
116                crate::widget::hotbar::Shown::of(self.hotbars, self.hotbar_looks.get(&placed))
117            }),
118            // The window's own character's: another's are its window's.
119            tracked: (widget.is_effects() && !self.follows.contains_key(&placed))
120                .then(|| self.tracked.clone()),
121            circles: widget.is_effects().then(|| {
122                self.circles
123                    .cloned()
124                    .unwrap_or_else(crate::widget::circles::palette)
125            }),
126            timers: self.timers.get(&placed).cloned(),
127            minimap: self.minimaps.get(&placed).copied(),
128            profiles: (*widget == crate::widget::Widget::Trainer
129                && !self.follows.contains_key(&placed))
130            .then(|| self.profiles.names()),
131        }
132    }
133}

162/// The top bar: who, how connected, the keys, and the layout's controls:
163/// its menu, the drawers, Arrange and Lock. What the hands hold and the clocks are
164/// widgets, in the layout with the rest.
165/// The story's banners, a highlight's alerts, each in a strip of the
166/// warning colour under the top bar while it lasts: newest last, gone after
167/// [`ALERT_FOR`](hydra_ui::story::ALERT_FOR). The window is drawn again when
168/// the oldest is due to go.
169pub(super) fn alerts(ui: &mut egui::Ui, session: u32, story: &hydra_ui::Story) {
170    let now = std::time::Instant::now();
171    let alerts = story.alerts_at(now);
172    let Some(oldest) = alerts.first().map(|(at, _)| *at) else {
173        return;
174    };
175    egui::Panel::top(egui::Id::new(("play-alerts", session))).show(ui, |ui| {
176        let ground = theme::color(ui.ctx(), T::Warning);
177        for (_, text) in &alerts {
178            egui::Frame::new()
179                .fill(ground)
180                .inner_margin(4.0)
181                .corner_radius(theme::corner(ui.ctx()))
182                .show(ui, |ui| {
183                    ui.set_width(ui.available_width());
184                    ui.colored_label(theme::readable_on(ground), *text);
185                });
186        }
187    });
188    let left = hydra_ui::story::ALERT_FOR.saturating_sub(now.saturating_duration_since(oldest));
189    ui.ctx().request_repaint_after(left);
190}

192pub(super) fn top(
193    ui: &mut egui::Ui,
194    view: &PlayView<'_>,
195    title: &TitleBar,
196    (grid, drawers): (&mut f32, &mut crate::layout::Drawers),
197    (arranging, locked): (&mut bool, &mut bool),
198    unsaved: Option<&str>,
199) -> Option<Top> {
200    let mut asked = None;
201    // As Hydra's own title bar, the bar's own ground moves the window when
202    // dragged and maximizes it when double-clicked: taken before the
203    // buttons, which, added after, win the press where they are.
204    if title.own {
205        title_ground(ui);
206    }
207    // Three sides across the bar: the left reads from its edge, the right
208    // from its edge back, the window's own buttons outermost; the centre
209    // takes what is left between, centred.
210    let right_width = ui.available_width() / 3.0;
211    ui.horizontal(|ui| {
212        for item in title.on(Side::Left) {
213            let pressed = bar_item(ui, view, item, (grid, drawers), (arranging, locked));
214            asked = asked.or(pressed);
215        }
216        ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
217            ui.set_min_width(right_width);
218            if title.own {
219                window_buttons(ui);
220            }
221            for item in title.on(Side::Right).iter().rev() {
222                let pressed = bar_item(ui, view, item, (grid, drawers), (arranging, locked));
223                asked = asked.or(pressed);
224            }
225            if let Some(why) = unsaved {
226                ui.colored_label(theme::color(ui.ctx(), T::Wrong), why);
227            }
228            if let Some(on) = view.numlock {
229                ui.weak(if on { "NumLock on" } else { "NumLock off" });
230            }
231            if view.set != 0 {
232                ui.weak(format!("Set {}", view.set)).on_hover_text(
233                    "The macro set in use over set 0; Alt+0 goes back to set 0 alone",
234                );
235            }
236            let center = title.on(Side::Center);
237            if !center.is_empty() {
238                ui.with_layout(
239                    egui::Layout::centered_and_justified(egui::Direction::LeftToRight),
240                    |ui| {
241                        ui.horizontal(|ui| {
242                            for item in center {
243                                let pressed =
244                                    bar_item(ui, view, item, (grid, drawers), (arranging, locked));
245                                asked = asked.or(pressed);
246                            }
247                        });
248                    },
249                );
250            }
251        });
252    });
253    asked
254}

256/// As Hydra's own title bar, the bar's ground moves the window when
257/// dragged and maximizes it when double-clicked.
258fn title_ground(ui: &mut egui::Ui) {
259    let bar = ui.available_rect_before_wrap();
260    let ground = ui.interact(
261        bar,
262        ui.id().with("title-ground"),
263        egui::Sense::click_and_drag(),
264    );
265    if ground.drag_started_by(egui::PointerButton::Primary) {
266        ui.ctx().send_viewport_cmd(egui::ViewportCommand::StartDrag);
267    }
268    if ground.double_clicked_by(egui::PointerButton::Primary) {
269        let maximized = ui.input(|input| input.viewport().maximized.unwrap_or(false));
270        ui.ctx()
271            .send_viewport_cmd(egui::ViewportCommand::Maximized(!maximized));
272    }
273}

275/// `item` as a word of the bar, when it is one: whether it was.
276fn word(ui: &mut egui::Ui, view: &PlayView<'_>, item: &Item) -> bool {
277    let state = view.snapshot.map(|snapshot| &snapshot.state);
278    match item {
279        Item::Character => {
280            ui.strong(view.name);
281        }
282        Item::Profession => {
283            let profession = state.and_then(|state| state.character.identity.profession.as_deref());
284            ui.weak(profession.unwrap_or("?"));
285        }
286        Item::Connection => {
287            let (word, token) = crate::hub::connection(view.lifecycle);
288            ui.colored_label(theme::color(ui.ctx(), token), word);
289        }
290        Item::Room => {
291            let room = state.and_then(|state| state.room.title.as_deref());
292            ui.colored_label(
293                theme::color(ui.ctx(), T::RoomName),
294                room.unwrap_or("Room unknown"),
295            );
296        }
297        Item::Number => {
298            let number = state.and_then(|state| state.room.id.as_deref());
299            ui.weak(number.unwrap_or("?"));
300        }
301        // A game value: its name, and its number as the model works it
302        // out now, whole when it is whole.
303        Item::Value(name) => {
304            let value = state.and_then(|state| state.evaluate(name));
305            let said = match value {
306                Some(n) if n.fract() == 0.0 => format!("{name} {n:.0}"),
307                Some(n) => format!("{name} {n:.1}"),
308                None => format!("{name} ?"),
309            };
310            ui.weak(said);
311        }
312        _ => return false,
313    }
314    true
315}

317/// The − and + that step the zoom of every window, reading − then + whichever
318/// way the side lays them out: what was pressed.
319fn zoom_buttons(ui: &mut egui::Ui) -> Option<Top> {
320    let plain = |ui: &mut egui::Ui, glyph: &str, said: &str, by: i8| {
321        ui.add(egui::Button::new(glyph).frame(false))
322            .on_hover_text(said)
323            .clicked()
324            .then_some(Top::Zoom(by))
325    };
326    let smaller = ("\u{2212}", "Draw everything smaller", -1);
327    let larger = ("+", "Draw everything larger", 1);
328    let (first, second) = if ui.layout().main_dir() == egui::Direction::RightToLeft {
329        (larger, smaller)
330    } else {
331        (smaller, larger)
332    };
333    plain(ui, first.0, first.1, first.2).or_else(|| plain(ui, second.0, second.1, second.2))
334}

336/// One item of the bar: a word, or a button, and what the button asked
337/// when pressed.
338fn bar_item(
339    ui: &mut egui::Ui,
340    view: &PlayView<'_>,
341    item: &Item,
342    (grid, drawers): (&mut f32, &mut crate::layout::Drawers),
343    (arranging, locked): (&mut bool, &mut bool),
344) -> Option<Top> {
345    if word(ui, view, item) {
346        return None;
347    }
348    let mut asked = None;
349    match item {
350        Item::Settings => {
351            if ui
352                .button("Settings")
353                .on_hover_text("This character's settings, its keys and its widgets' pages")
354                .clicked()
355            {
356                asked = Some(Top::Settings);
357            }
358        }
359        Item::Zoom => asked = zoom_buttons(ui),
360        Item::Log => {
361            if ui
362                .button("Log")
363                .on_hover_text("What this character read, kept a file a day")
364                .clicked()
365            {
366                asked = Some(Top::Log);
367            }
368        }
369        Item::Keys => {
370            ui.menu_button("Keys", |ui| {
371                for said in view.keys {
372                    ui.label(said);
373                }
374                if ui.button("Read the keybinds again").clicked() {
375                    asked = Some(Top::ReloadKeys);
376                    ui.close();
377                }
378                if ui.button("Change the keys...").clicked() {
379                    asked = Some(Top::Keys);
380                    ui.close();
381                }
382            });
383        }
384        Item::Layout => {
385            ui.menu_button("Layout", |ui| {
386                            ui.horizontal(|ui| {
387                                ui.label("Grid");
388                                let changed = ui
389                                    .add(egui::DragValue::new(grid).range(0.0..=64.0).suffix(" pt"))
390                                    .on_hover_text("What the windows' edges snap to; 0 for none. Shift while dragging snaps to nothing.")
391                                    .changed();
392                                if changed {
393                                    asked = Some(Top::Grid);
394                                }
395                            });
396                            if ui.button("Add a widget...").clicked() {
397                                asked = Some(Top::AddWidget);
398                                ui.close();
399                            }
400                            if ui.button("New custom window").clicked() {
401                                asked = Some(Top::NewCustom);
402                                ui.close();
403                            }
404                            if ui
405                                .button("Fit to this window")
406                                .on_hover_text("Keep the windows where they are drawn now, at this size: sized away from it, the layout scales")
407                                .clicked()
408                            {
409                                asked = Some(Top::FitHere);
410                                ui.close();
411                            }
412                            if ui.button("Lay out afresh").clicked() {
413                                asked = Some(Top::Fit);
414                                ui.close();
415                            }
416                        });
417        }
418        Item::Drawers => {
419            ui.menu_button("Drawers", |ui| {
420                if super::drawers::menu(ui, drawers) {
421                    asked = Some(Top::Drawers);
422                }
423            });
424        }
425        // On the bar, lit while it is on, so the mode is seen
426        // and left in one click: a new custom window turns it
427        // on, and with the switch only in the Layout menu a
428        // player could be left stuck in it.
429        Item::Arrange => {
430            ui.add_enabled_ui(!*locked, |ui| {
431                ui.toggle_value(arranging, "Arrange")
432                    .on_hover_text("Move, resize and drag widgets in and out of custom windows");
433            });
434        }
435        Item::Lock => {
436            ui.toggle_value(locked, "Lock")
437                .on_hover_text("Keep every window where it is: none is dragged or resized");
438        }
439        Item::Character
440        | Item::Profession
441        | Item::Connection
442        | Item::Room
443        | Item::Number
444        | Item::Value(_) => {}
445    }
446    asked
447}

449/// The window's own buttons at the bar's right end, right to left: close,
450/// maximize or restore, minimize.
451fn window_buttons(ui: &mut egui::Ui) {
452    let maximized = ui.input(|input| input.viewport().maximized.unwrap_or(false));
453    let plain = |ui: &mut egui::Ui, glyph: &str, said: &str| {
454        ui.add(egui::Button::new(glyph).frame(false))
455            .on_hover_text(said)
456            .clicked()
457    };
458    if plain(
459        ui,
460        "\u{2715}",
461        "Close this window; the character keeps playing",
462    ) {
463        ui.ctx().send_viewport_cmd(egui::ViewportCommand::Close);
464    }
465    if maximized {
466        if plain(ui, "\u{2750}", "Restore") {
467            ui.ctx()
468                .send_viewport_cmd(egui::ViewportCommand::Maximized(false));
469        }
470    } else if plain(ui, "\u{25a1}", "Maximize") {
471        ui.ctx()
472            .send_viewport_cmd(egui::ViewportCommand::Maximized(true));
473    }
474    if plain(ui, "\u{2014}", "Minimize") {
475        ui.ctx()
476            .send_viewport_cmd(egui::ViewportCommand::Minimized(true));
477    }
478}

480/// How close to the window's edge a press resizes it, points.
481const EDGE: f32 = 6.0;
482
483/// With no system frame, the window's edges resize it: the pointer near
484/// one shows the resize cursor, and a press there begins the resize.
485pub(super) fn resize_edges(ctx: &egui::Context) {
486    use egui::{CursorIcon, ResizeDirection, ViewportCommand};
487    let maximized = ctx.input(|input| input.viewport().maximized.unwrap_or(false));
488    if maximized {
489        return;
490    }
491    let Some(at) = ctx.input(|input| input.pointer.hover_pos()) else {
492        return;
493    };
494    let screen = ctx.viewport_rect();
495    let (left, right) = (at.x - screen.min.x < EDGE, screen.max.x - at.x < EDGE);
496    let (top, bottom) = (at.y - screen.min.y < EDGE, screen.max.y - at.y < EDGE);
497    let (direction, cursor) = match (left, right, top, bottom) {
498        (true, _, true, _) => (ResizeDirection::NorthWest, CursorIcon::ResizeNwSe),
499        (_, true, true, _) => (ResizeDirection::NorthEast, CursorIcon::ResizeNeSw),
500        (true, _, _, true) => (ResizeDirection::SouthWest, CursorIcon::ResizeNeSw),
501        (_, true, _, true) => (ResizeDirection::SouthEast, CursorIcon::ResizeNwSe),
502        (true, _, _, _) => (ResizeDirection::West, CursorIcon::ResizeHorizontal),
503        (_, true, _, _) => (ResizeDirection::East, CursorIcon::ResizeHorizontal),
504        (_, _, true, _) => (ResizeDirection::North, CursorIcon::ResizeVertical),
505        (_, _, _, true) => (ResizeDirection::South, CursorIcon::ResizeVertical),
506        _ => return,
507    };
508    ctx.set_cursor_icon(cursor);
509    if ctx.input(|input| input.pointer.primary_pressed()) {
510        ctx.send_viewport_cmd(ViewportCommand::BeginResize(direction));
511    }
512}

519/// What a window holds, filling it: its window is the size its layout says,
520/// never its content's. One widget is given everything; a custom window's
521/// widgets are drawn bare in their cells, once the cells are kept to the
522/// inside it has now (`Custom::fit`). Nothing spills out of its cell: a
523/// one-line widget stays one line, and the rest scroll. A tab stack shows a
524/// tab for each of its widgets, a click on one showing it, each not showing
525/// with what it has said since it last showed. Its inside, where it drew.
526pub(super) fn holder(
527    ui: &mut egui::Ui,
528    holds: &mut Holds,
529    drawing: &mut Drawing<'_>,
530) -> egui::Rect {
531    let inside = ui.available_rect_before_wrap();
532    ui.set_min_size(inside.size());
533    // The theme's frame, when it names one: the room it takes inside the
534    // window's bound, so its outline and stickers are not clipped.
535    let framed = crate::frames::worn(ui.ctx());
536    let room = framed.as_deref().map_or(0.0, crate::frames::margin);
537    let area = inside.shrink(room);
538    match holds {
539        Holds::One(placed) => {
540            if let Some(framed) = &framed {
541                let mut child = ui.new_child(UiBuilder::new().max_rect(area));
542                shown(&mut child, placed, drawing);
543                ui.advance_cursor_after_rect(inside);
544                crate::frames::paint(ui.ctx(), ui.painter(), framed, &[area]);
545            } else {
546                shown(ui, placed, drawing);
547            }
548        }
549        Holds::Custom(custom) => {
550            // Not to the pass egui lays a window out in at its narrowest as
551            // a resize begins, then discards: kept to it, a narrow cell came
552            // back from the scaling at its smallest, not its width.
553            // Resized with Arrange on, a window leaves its widgets their
554            // size, room made or taken for arranging them; with it off they
555            // scale with the window.
556            if ui.is_sizing_pass() {
557            } else if drawing.arranging {
558                custom.keep(area.size());
559            } else {
560                custom.fit(area.size());
561            }
562            if custom.cells.is_empty() {
563                ui.new_child(UiBuilder::new().max_rect(area.shrink(4.0)))
564                    .weak("Empty. With Arrange on, in the Layout menu, drop widgets here.");
565            }
566            let shape = theme::shape(ui.ctx());
567            // With a frame, the cells are set apart by its thickness and the
568            // edge fills the gap; the cells as they touch are what it is
569            // laid on.
570            let gap = framed
571                .as_deref()
572                .map_or(shape.pane_gap, hydra_ui::frame::Frame::gap);
573            let mut touching = Vec::with_capacity(custom.cells.len());
574            let strip = theme::tab_height(ui.ctx(), crate::widget::LINE);
575            for cell in &mut custom.cells {
576                let whole = cell.rect().translate(area.min.to_vec2()).intersect(area);
577                // Half the gap off each side: the gap between two.
578                let at = whole.shrink(gap / 2.0);
579                if !at.is_positive() {
580                    continue;
581                }
582                touching.push(whole);
583                let (tabs, body) = tabs_and_body(at, cell.tabs.len(), strip, cell.bar);
584                let reorder = tab_strip(ui, cell, &tabs, drawing);
585                if let Some((at, to)) = reorder
586                    && cell.move_tab(at, to)
587                {
588                    drawing.reordered = true;
589                }
590                let Some(placed) = cell.shown() else {
591                    continue;
592                };
593                let mut child =
594                    ui.new_child(UiBuilder::new().max_rect(body).id_salt(("cell", placed.id)));
595                shown(&mut child, placed, drawing);
596            }
597            ui.advance_cursor_after_rect(inside);
598            if let Some(framed) = &framed
599                && !touching.is_empty()
600            {
601                crate::frames::paint(ui.ctx(), ui.painter(), framed, &touching);
602            }
603        }
604    }
605    inside
606}

608/// `cell`'s tab strip drawn, a tab for each of its widgets at `tabs`: a
609/// click shows that tab, and a tab dragged along the strip and let go asks
610/// to move, as (from, to). With Arrange on the strip is arranging's.
611fn tab_strip(
612    ui: &mut egui::Ui,
613    cell: &mut crate::layout::Cell,
614    tabs: &[egui::Rect],
615    drawing: &mut Drawing<'_>,
616) -> Option<(usize, usize)> {
617    let bar = cell.bar;
618    let mut reorder = None;
619    for (index, (tab, rect)) in cell.tabs.iter().zip(tabs.iter().copied()).enumerate() {
620        let showing = index == cell.showing;
621        let look = drawing.tabs.get(&tab.id);
622        let quiet = look.is_some_and(|look| look.quiet);
623        // A tab showing is read; a quiet one is never counted.
624        let unread = unread(drawing.read, tab, &drawing.seen).filter(|_| !showing && !quiet);
625        if unread.is_some() {
626            drawing.unread.push(tab.id);
627        }
628        let name = crate::widget::tab::name(look, &tab.widget.name(), unread);
629        let color = match (showing, unread.is_some()) {
630            (true, _) => look.and_then(|look| look.showing),
631            (false, true) => look.and_then(|look| look.unread),
632            (false, false) => None,
633        };
634        let name = match color {
635            Some([r, g, b]) => egui::RichText::new(name).color(egui::Color32::from_rgb(r, g, b)),
636            None => egui::RichText::new(name),
637        };
638        let mut child = ui.new_child(UiBuilder::new().max_rect(rect).id_salt(("tab", tab.id)));
639        let button =
640            egui::Button::selectable(showing, name).wrap_mode(egui::TextWrapMode::Truncate);
641        let response = child.add_sized(rect.size(), button);
642        if response.clicked() {
643            cell.showing = index;
644        }
645        if drawing.arranging {
646            continue;
647        }
648        let response = response.interact(egui::Sense::drag());
649        if let Some(at) = response.interact_pointer_pos()
650            && let Some(to) = crate::layout::tab_at(tabs, at, bar.vertical())
651            && to != index
652        {
653            if response.dragged() {
654                slot_mark(ui, tabs, index, to, bar.vertical());
655            }
656            if response.drag_stopped() {
657                reorder = Some((index, to));
658            }
659        }
660    }
661    reorder
662}

664/// The place in the strip a tab dragged from `from` would take at `to`,
665/// marked: a line at the edge of that tab it would go beside, across the
666/// strip when it runs down a side (`vertical`).
667pub(super) fn slot_mark(
668    ui: &egui::Ui,
669    tabs: &[egui::Rect],
670    from: usize,
671    to: usize,
672    vertical: bool,
673) {
674    let Some(rect) = tabs.get(to) else {
675        return;
676    };
677    let stroke = egui::Stroke::new(2.0, theme::color(ui.ctx(), T::Accent));
678    if vertical {
679        let y = if to > from { rect.max.y } else { rect.min.y };
680        ui.painter().hline(rect.x_range(), y, stroke);
681    } else {
682        let x = if to > from { rect.max.x } else { rect.min.x };
683        ui.painter().vline(x, rect.y_range(), stroke);
684    }
685}

687/// `placed` drawn into `ui`, showing: so what it has said is read, up to
688/// now. The one way a widget is drawn, standalone, alone in a cell or a tab.
689/// One that follows another character draws from that character, named on
690/// it, or says that character is not running; a story never follows.
691fn shown(ui: &mut egui::Ui, placed: &Placed, drawing: &mut Drawing<'_>) {
692    let rect = ui.max_rect();
693    drawing.rects.push((placed.id, rect));
694    if ui.input(|input| input.pointer.primary_pressed()) && ui.rect_contains_pointer(rect) {
695        drawing.pressed = Some(placed.id);
696    }
697    if drawing.in_use == Some(placed.id) {
698        // The window in use, which a scrolling key acts on.
699        let stroke = egui::Stroke::new(1.0, theme::color(ui.ctx(), T::Accent).gamma_multiply(0.5));
700        ui.painter().rect_stroke(
701            rect,
702            theme::corner(ui.ctx()),
703            stroke,
704            egui::StrokeKind::Inside,
705        );
706    }
707    // The theme's room round the widget, inside its pane.
708    let padding = theme::shape(ui.ctx()).pane_padding;
709    if padding > 0.0 && rect.shrink(padding).is_positive() {
710        let mut inner = ui.new_child(UiBuilder::new().max_rect(rect.shrink(padding)));
711        drawn(&mut inner, placed, drawing);
712        ui.advance_cursor_after_rect(rect);
713    } else {
714        drawn(ui, placed, drawing);
715    }
716}

718/// `placed` drawn, as [`shown`] says: in its face, when its kind or it has
719/// one of its own.
720fn drawn(ui: &mut egui::Ui, placed: &Placed, drawing: &mut Drawing<'_>) {
721    let key = placed.widget.type_key();
722    if let Some(face) = theme::face(ui.ctx(), &key, drawing.faces.get(&placed.id)) {
723        let families = crate::fonts::loaded(ui.ctx());
724        ui.scope(|ui| {
725            theme::faced(ui.style_mut(), &face, &families);
726            drawn_in(ui, placed, drawing);
727        });
728    } else {
729        drawn_in(ui, placed, drawing);
730    }
731}
732
733/// `placed` drawn in the face `ui` has.
734fn drawn_in(ui: &mut egui::Ui, placed: &Placed, drawing: &mut Drawing<'_>) {
735    let id = widget_id(drawing.session, placed.id);
736    let follows = drawing
737        .follows
738        .get(&placed.id)
739        .filter(|_| !placed.widget.is_story());
740    let Some(who) = follows else {
741        if let Some(count) = placed.widget.count(&drawing.seen) {
742            drawing.read.insert(placed.id, count);
743        }
744        let chosen = drawing.chosen(placed);
745        match placed.widget.draw_with(ui, &drawing.seen, id, &chosen) {
746            Some(crate::widget::Clicked::Set { key, to }) => {
747                drawing.set = Some((placed.id, key, to));
748            }
749            Some(line) => drawing.sent = Some(line),
750            None => {}
751        }
752        return;
753    };
754    // A bare name two characters share -- one name on two games -- names
755    // neither: which is meant is for the player to say.
756    let mut named = drawing.others.iter().filter(|other| other.is(who));
757    let (one, another) = (named.next(), named.next());
758    if another.is_some() {
759        ui.weak(format!(
760            "More than one {who} is running: choose which in this window's Show for."
761        ));
762        return;
763    }
764    match one {
765        Some(other) => {
766            let seen = Seen {
767                snapshot: other.snapshot.as_deref(),
768                // Named by its game too when the follow does: one name
769                // may be on two games.
770                who: Some(if who.contains(':') { who } else { &other.name }),
771                minimap: other.minimap.as_ref(),
772                travel: None,
773                trainer: None,
774                ..drawing.seen
775            };
776            // Another character's widget sends nothing on this one's.
777            let chosen = drawing.chosen(placed);
778            let _ = placed.widget.draw_with(ui, &seen, id, &chosen);
779        }
780        None => {
781            ui.weak(format!("{who} is not running."));
782        }
783    }
784}

786/// A window's title: its widget's name, and whose when it follows another
787/// character; or the custom window's own.
788pub(super) fn title(holds: &Holds, follows: &BTreeMap<u32, String>) -> String {
789    match holds {
790        Holds::One(placed) => match follows.get(&placed.id) {
791            Some(who) if !placed.widget.is_story() => format!("{} ({who})", placed.widget.name()),
792            _ => placed.widget.name().into_owned(),
793        },
794        Holds::Custom(custom) => custom.title.clone(),
795    }
796}

798/// What `placed` has said since it last showed, when it counts and that is
799/// something. A widget first seen here is read up to now, so a tab counts
800/// from when it was stacked.
801pub(super) fn unread(
802    read: &mut HashMap<u32, u64>,
803    placed: &Placed,
804    seen: &Seen<'_>,
805) -> Option<u64> {
806    let count = placed.widget.count(seen)?;
807    let last = *read.entry(placed.id).or_insert(count);
808    Some(count.saturating_sub(last)).filter(|unread| *unread > 0)
809}

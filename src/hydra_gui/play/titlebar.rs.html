1//! The play window's title bar: Hydra's own in place of the system's, when
2//! the player asks, with what it shows chosen on its page.
3//!
4//! The top bar is the title bar. With the system's frame kept it sits
5//! under the system's title bar; as Hydra's own, the system frame goes and
6//! the bar is the window's title: dragged to move the window, double-
7//! clicked to maximize it, the minimize, maximize and close buttons at its
8//! right end, the window's edges dragged to resize it.
9//!
10//! The bar has three sides, left, centre and right, and each holds a row
11//! of **items** in order: the character, its profession, how it is
12//! connected, the room's title and number, each of the menus and switches,
13//! and any **game value** the model can work out by name
14//! (`GameState::evaluate`), shown as its name and its number. The page
15//! shows each side as a row of pickers, one an item, and one more to add
16//! (`crate::titlebar_page`).
17
18use hydra_ui::settings::{Page, Row, RowKind, Value};
19use serde::{Deserialize, Serialize};
20
21/// The page's id, among the widgets' pages, which the menu sends to the
22/// window.
23pub(crate) const PAGE: &str = "widget:titlebar";
24
25/// What a game value's key begins with: `value:Stats.level`.
26pub(crate) const VALUE: &str = "value:";
27
28/// A thing the bar can show.
29#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
30#[serde(rename_all = "snake_case")]
31pub(crate) enum Item {
32    /// The character's name.
33    Character,
34    /// Its profession.
35    Profession,
36    /// How it is connected.
37    Connection,
38    /// The room's title.
39    Room,
40    /// The game's number for the room.
41    Number,
42    Lock,
43    Arrange,
44    Drawers,
45    Layout,
46    Keys,
47    Log,
48    Settings,
49    /// The − and + buttons that step the zoom.
50    Zoom,
51    /// A value the model works out by name, shown as the name and its
52    /// number: `Stats.level 100`.
53    Value(String),
54}
55
56impl Item {
57    /// Every fixed item, in the order a picker lists them.
58    pub(crate) const FIXED: [Item; 13] = [
59        Item::Character,
60        Item::Profession,
61        Item::Connection,
62        Item::Room,
63        Item::Number,
64        Item::Lock,
65        Item::Arrange,
66        Item::Drawers,
67        Item::Layout,
68        Item::Keys,
69        Item::Log,
70        Item::Settings,
71        Item::Zoom,
72    ];
73
74    /// Its key on the page and in a change: a word, or `value:<name>`.
75    pub(crate) fn key(&self) -> String {
76        match self {
77            Item::Character => "character".to_owned(),
78            Item::Profession => "profession".to_owned(),
79            Item::Connection => "connection".to_owned(),
80            Item::Room => "room".to_owned(),
81            Item::Number => "number".to_owned(),
82            Item::Lock => "lock".to_owned(),
83            Item::Arrange => "arrange".to_owned(),
84            Item::Drawers => "drawers".to_owned(),
85            Item::Layout => "layout".to_owned(),
86            Item::Keys => "keys".to_owned(),
87            Item::Log => "log".to_owned(),
88            Item::Settings => "settings".to_owned(),
89            Item::Zoom => "zoom".to_owned(),
90            Item::Value(name) => format!("{VALUE}{name}"),
91        }
92    }
93
94    /// The item `key` names; none for a word that is not one.
95    pub(crate) fn of_key(key: &str) -> Option<Item> {
96        let key = key.trim();
97        if let Some(name) = key.strip_prefix(VALUE) {
98            let name = name.trim();
99            return (!name.is_empty()).then(|| Item::Value(name.to_owned()));
100        }
101        Item::FIXED.into_iter().find(|item| item.key() == key)
102    }
103
104    /// What a player calls it.
105    pub(crate) fn label(&self) -> String {
106        match self {
107            Item::Character => "Character".to_owned(),
108            Item::Profession => "Profession".to_owned(),
109            Item::Connection => "Connection".to_owned(),
110            Item::Room => "Room title".to_owned(),
111            Item::Number => "Room number".to_owned(),
112            Item::Lock => "Lock".to_owned(),
113            Item::Arrange => "Arrange".to_owned(),
114            Item::Drawers => "Drawers menu".to_owned(),
115            Item::Layout => "Layout menu".to_owned(),
116            Item::Keys => "Keys menu".to_owned(),
117            Item::Log => "Log".to_owned(),
118            Item::Settings => "Settings".to_owned(),
119            Item::Zoom => "Zoom − and +".to_owned(),
120            Item::Value(name) => name.clone(),
121        }
122    }
123}

125/// A side of the bar.
126#[derive(Clone, Copy, Debug, PartialEq, Eq)]
127pub(crate) enum Side {
128    Left,
129    Center,
130    Right,
131}

133impl Side {
134    pub(crate) const ALL: [Side; 3] = [Side::Left, Side::Center, Side::Right];
135
136    /// Its key on the page.
137    pub(crate) const fn key(self) -> &'static str {
138        match self {
139            Side::Left => "left",
140            Side::Center => "center",
141            Side::Right => "right",
142        }
143    }
144
145    /// What a player calls it.
146    pub(crate) const fn label(self) -> &'static str {
147        match self {
148            Side::Left => "Left",
149            Side::Center => "Centre",
150            Side::Right => "Right",
151        }
152    }
153
154    /// The side `key` names.
155    pub(crate) fn of_key(key: &str) -> Option<Side> {
156        Side::ALL.into_iter().find(|side| side.key() == key)
157    }

160/// What the title bar shows, kept with the layout.
161#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
162pub(crate) struct TitleBar {
163    /// Hydra's own title bar in place of the system's.
164    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
165    pub(crate) own: bool,
166    /// The left side's items, in order.
167    #[serde(default = "left_at_first")]
168    pub(crate) left: Vec<Item>,
169    /// The centre's items, in order.
170    #[serde(default, skip_serializing_if = "Vec::is_empty")]
171    pub(crate) center: Vec<Item>,
172    /// The right side's items, in order, before the window's own buttons.
173    #[serde(default = "right_at_first")]
174    pub(crate) right: Vec<Item>,
175}
176
177/// The left side as it starts: the character and its connection. The
178/// system's title bar names the room already; Hydra's own adds it.
179fn left_at_first() -> Vec<Item> {
180    vec![Item::Character, Item::Connection]
181}
182
183/// The right side as it starts: every menu and switch.
184fn right_at_first() -> Vec<Item> {
185    vec![
186        Item::Lock,
187        Item::Arrange,
188        Item::Drawers,
189        Item::Layout,
190        Item::Keys,
191        Item::Log,
192        Item::Settings,
193        Item::Zoom,
194    ]
195}
196
197impl Default for TitleBar {
198    fn default() -> Self {
199        Self {
200            own: false,
201            left: left_at_first(),
202            center: Vec::new(),
203            right: right_at_first(),
204        }
205    }
206}

208/// The page's key for the switch.
209pub(crate) const OWN: &str = "own";
210
211/// Items as a page's value or a change writes them: one key a line.
212pub(crate) fn written(items: &[Item]) -> String {
213    items.iter().map(Item::key).collect::<Vec<_>>().join("\n")
214}
215
216/// The items `written` names, one key a line; a line naming none is an
217/// error.
218pub(crate) fn read(written: &str) -> Result<Vec<Item>, String> {
219    written
220        .lines()
221        .map(str::trim)
222        .filter(|line| !line.is_empty())
223        .map(|line| {
224            Item::of_key(line).ok_or_else(|| format!("`{line}` is not a thing the bar shows."))
225        })
226        .collect()
227}
228
229impl TitleBar {
230    /// The items on `side`, in order.
231    pub(crate) fn on(&self, side: Side) -> &[Item] {
232        match side {
233            Side::Left => &self.left,
234            Side::Center => &self.center,
235            Side::Right => &self.right,
236        }
237    }
238
239    fn on_mut(&mut self, side: Side) -> &mut Vec<Item> {
240        match side {
241            Side::Left => &mut self.left,
242            Side::Center => &mut self.center,
243            Side::Right => &mut self.right,
244        }
245    }
246
247    /// Its page: the switch, then each side's items.
248    pub(crate) fn page(&self, file: String) -> Page {
249        let default = TitleBar::default();
250        let mut rows = vec![Row {
251            key: OWN.to_owned(),
252            label: "Hydra's own title bar".to_owned(),
253            help: "The top bar in place of the system's title bar: drag it to move the \
254                   window, double-click it to maximize, with the window's buttons at its \
255                   right end; the window's edges resize it. Off, the system's title bar \
256                   stays above the top bar."
257                .to_owned(),
258            kind: RowKind::Toggle,
259            value: Value::On(self.own),
260            here: self.own,
261            from: None,
262        }];
263        for side in Side::ALL {
264            rows.push(Row {
265                key: side.key().to_owned(),
266                label: side.label().to_owned(),
267                help: format!(
268                    "What the bar's {} shows, in order.",
269                    side.label().to_lowercase()
270                ),
271                kind: RowKind::Words,
272                value: Value::Text(written(self.on(side))),
273                here: self.on(side) != default.on(side),
274                from: None,
275            });
276        }
277        Page {
278            id: PAGE.to_owned(),
279            title: "Title bar".to_owned(),
280            file,
281            takes: "at once".to_owned(),
282            problem: None,
283            rows,
284        }
285    }
286
287    /// Set `key` to `to`, or back to its own. What was done.
288    pub(crate) fn set(&mut self, key: &str, to: Option<&str>) -> Result<String, String> {
289        if key == OWN {
290            self.own = match to.map(str::trim) {
291                None | Some("off") => false,
292                Some("on") => true,
293                Some(other) => return Err(format!("`{other}` is not on or off.")),
294            };
295            if self.own {
296                // Its own bar is the window's title now: the room and its
297                // number join the left, unless the player chose the left.
298                if self.left == left_at_first() {
299                    self.left.extend([Item::Room, Item::Number]);
300                }
301                return Ok("The window wears Hydra's own title bar.".to_owned());
302            }
303            return Ok("The window wears the system's title bar.".to_owned());
304        }
305        let side =
306            Side::of_key(key).ok_or_else(|| format!("The title bar has no setting {key}."))?;
307        *self.on_mut(side) = match to {
308            None => TitleBar::default().on(side).to_vec(),
309            Some(written) => read(written)?,
310        };
311        let items = self.on(side);
312        Ok(if items.is_empty() {
313            format!("The bar's {} shows nothing.", side.label().to_lowercase())
314        } else {
315            format!(
316                "The bar's {} shows {}.",
317                side.label().to_lowercase(),
318                items.iter().map(Item::label).collect::<Vec<_>>().join(", ")
319            )
320        })
321    }
322}

324#[cfg(test)]
325mod tests {
326    use super::*;

328    /// A side is set as its keys, one a line, a game value among them;
329    /// put back, it is its own again; a word that is nothing is refused.
330    #[test]
331    fn a_side_is_its_keys_in_order() {
332        let mut bar = TitleBar::default();
333        assert_eq!(bar.on(Side::Left), [Item::Character, Item::Connection]);
334        assert_eq!(bar.on(Side::Right).len(), 8);
335        bar.set("center", Some("room\nvalue:Stats.level\nnumber"))
336            .expect("set");
337        assert_eq!(
338            bar.on(Side::Center),
339            [
340                Item::Room,
341                Item::Value("Stats.level".to_owned()),
342                Item::Number
343            ]
344        );
345        assert!(bar.set("left", Some("weather")).is_err());
346        assert!(bar.set("middle", Some("room")).is_err());
347        bar.set("right", Some("")).expect("none");
348        assert!(bar.on(Side::Right).is_empty());
349        bar.set("right", None).expect("back");
350        bar.set("center", None).expect("back");
351        assert_eq!(bar, TitleBar::default());
352        let page = bar.page(String::new());
353        assert_eq!(page.id, PAGE);
354        assert_eq!(
355            page.rows[1].value,
356            Value::Text("character\nconnection".to_owned())
357        );
358    }
359
360    /// Switched on, Hydra's own bar names the room too, unless the player
361    /// chose the left already.
362    #[test]
363    fn its_own_bar_shows_the_room() {
364        let mut bar = TitleBar::default();
365        bar.set("own", Some("on")).expect("on");
366        assert!(bar.own);
367        assert_eq!(
368            bar.on(Side::Left),
369            [Item::Character, Item::Connection, Item::Room, Item::Number]
370        );
371        let mut chosen = TitleBar::default();
372        chosen.set("left", Some("profession")).expect("set");
373        chosen.set("own", Some("on")).expect("on");
374        assert_eq!(
375            chosen.on(Side::Left),
376            [Item::Profession],
377            "a choice is kept"
378        );
379    }
380
381    /// A layout kept before the title bar had a page reads with the bar
382    /// as it was; a bar written reads back whole.
383    #[test]
384    fn an_old_file_reads_as_the_default() {
385        let bar: TitleBar = serde_json::from_str("{}").expect("read");
386        assert_eq!(bar, TitleBar::default());
387        let mut bar = TitleBar::default();
388        bar.center.push(Item::Value("Society.rank".to_owned()));
389        let text = serde_json::to_string(&bar).expect("written");
390        let back: TitleBar = serde_json::from_str(&text).expect("read");
391        assert_eq!(back, bar);
392    }

1//! A widget's own page: a health bar's right-click, *Settings...*, takes
2//! the player to that bar's settings -- horizontal or vertical, its text
3//! inside or outside it, a percent, current/max, a label with numbers or
4//! any mix, its colour, its overlay. Every way in opens the one settings
5//! menu at its place (`settings_menu.rs`): the right-click's *Settings...*
6//! asks for this widget's page there.
7//!
8//! The pages are the play window's, built from its layout, and a change is
9//! made to the layout and saved with it: no behavior is behind them. A bar
10//! widget's page holds its [`Look`]; the Room widget's, which of its parts
11//! it shows ([`RoomParts`]): which streams show in the Room window.

49/// A widget's own font and size, as its page shows them: none, the theme's
50/// for its kind.
51fn face_rows(face: Option<&hydra_ui::theme::Face>) -> Vec<Row> {
52    vec![
53        Row {
54            key: FONT.to_owned(),
55            label: "Font".to_owned(),
56            help: "This widget's own font, from the fonts folder or the computer's; none, its kind's in the theme.".to_owned(),
57            kind: RowKind::Font,
58            value: face
59                .and_then(|face| face.font.clone())
60                .map_or(Value::Unset, Value::Text),
61            here: face.is_some_and(|face| face.font.is_some()),
62            from: None,
63        },
64        Row {
65            key: TEXT_SIZE.to_owned(),
66            label: "Text size".to_owned(),
67            help: "This widget's own text size, in points; 0, its kind's in the theme.".to_owned(),
68            kind: RowKind::Number {
69                min: 0.0,
70                max: 48.0,
71            },
72            value: face
73                .and_then(|face| face.size)
74                .map_or(Value::Unset, |size| Value::Text(format!("{size}"))),
75            here: face.is_some_and(|face| face.size.is_some()),
76            from: None,
77        },
78    ]
79}

81/// `face` with the page's `key` set `to`, as the page writes it, or back to
82/// the theme's (`None`).
83fn face_set(face: &mut hydra_ui::theme::Face, key: &str, to: Option<&str>) -> Result<(), String> {
84    match key {
85        FONT => {
86            face.font = to
87                .map(str::trim)
88                .filter(|name| !name.is_empty())
89                .map(str::to_owned);
90        }
91        _ => {
92            face.size = match to.map(str::trim) {
93                None => None,
94                Some(written) => {
95                    let size: f32 = written
96                        .parse()
97                        .map_err(|_| format!("`{written}` is not a size."))?;
98                    (size >= 6.0).then(|| size.min(48.0))
99                }
100            };
101        }
102    }
103    Ok(())
104}

138/// The PNGs in `folder`, by name; none when it cannot be read.
139fn pngs(folder: &std::path::Path) -> Vec<PathBuf> {
140    let mut found: Vec<PathBuf> = std::fs::read_dir(folder)
141        .map(|entries| {
142            entries
143                .filter_map(Result::ok)
144                .map(|entry| entry.path())
145                .filter(|path| {
146                    path.extension()
147                        .is_some_and(|x| x.eq_ignore_ascii_case("png"))
148                })
149                .collect()
150        })
151        .unwrap_or_default();
152    found.sort();
153    found
154}

257    /// [`Play::widget_pages`], `now` the game's clock (unix seconds) where it
258    /// is known: the time of day a page shows as the game's own.
259    pub(crate) fn widget_pages_at(&self, pictures: &Pictures, now: Option<u32>) -> Vec<Page> {
260        let Some(layout) = &self.layout else {
261            return Vec::new();
262        };
263        let file = self
264            .layouts
265            .as_ref()
266            .map_or_else(String::new, |dir| dir.display().to_string());
267        let mut pages = Vec::new();
268        for holder in &layout.holders {
269            let window = holder.holds.title();
270            for one in holder.holds.placed() {
271                let rows = match (&one.widget, one.widget.bar_look()) {
272                    (_, Some(default)) => {
273                        let pulse = one.widget == Widget::Pulse;
274                        bar_rows(
275                            layout
276                                .looks
277                                .get(&one.id)
278                                .cloned()
279                                .zip(one.widget.bar_token())
280                                .map(|(look, token)| look.themed(token))
281                                .as_ref(),
282                            &default,
283                            &pictures.overlays,
284                            pulse,
285                        )
286                    }
287                    (Widget::Injuries, None) => {
288                        doll_rows(layout.dolls.get(&one.id), &pictures.dolls, now)
289                    }
290                    (Widget::Room, None) => room_rows(layout.rooms.get(&one.id)),
291                    (Widget::Aim, None) => crate::widget::aim::rows(layout.aims.get(&one.id)),
292                    (
293                        Widget::RightHand | Widget::LeftHand | Widget::Prepared | Widget::Reserve,
294                        None,
295                    ) => crate::widget::labels::rows(&one.widget, layout.labels.get(&one.id)),
296                    (Widget::Timers, None) => {
297                        let mut rows =
298                            crate::widget::effects::timers_rows(layout.timers.get(&one.id));
299                        rows.push(crate::tracked::table(&self.tracked));
300                        rows.push(crate::widget::circles::row(layout.circles.as_ref()));
301                        rows
302                    }
303                    (Widget::Effects(_) | Widget::Missing, None) => {
304                        vec![
305                            crate::tracked::table(&self.tracked),
306                            crate::widget::circles::row(layout.circles.as_ref()),
307                        ]
308                    }
309                    (Widget::Quick, None) => crate::widget::quick::rows(layout.quicks.get(&one.id)),
310                    (Widget::Hotbar, None) => crate::widget::hotbar::rows(
311                        layout.hotbars.get(&one.id),
312                        &self.hotbars.names(),
313                    ),
314                    (Widget::Minimap, None) => {
315                        crate::widget::minimap::look::rows(layout.minimaps.get(&one.id))
316                    }
317                    (Widget::Creatures, None) => {
318                        crate::widget::roster::targets_rows(layout.rosters.get(&one.id))
319                    }
320                    (Widget::Players, None) => {
321                        crate::widget::roster::players_rows(layout.rosters.get(&one.id))
322                    }
323                    (Widget::Npcs | Widget::Objects, None) => {
324                        list_rows(layout.lists.get(&one.id).copied())
325                    }
326                    (Widget::Story | Widget::Stream(_), None) => {
327                        lines_rows(layout.lines.get(&one.id), lines_kind(&one.widget))
328                    }
329                    _ => Vec::new(),
330                };
331                let mut rows = rows;
332                // (the Injuries page's text and window under a heading of
333                // their own, over its statuses' buttons)
334                if one.widget == Widget::Injuries {
335                    rows.push(heading("Text and window"));
336                }
337                rows.extend(face_rows(layout.faces.get(&one.id)));
338                // Its tab's own, for what can be a tab of a stack.
339                if one.widget.tab_group() == crate::widget::Group::Streams {
340                    rows.extend(crate::widget::tab::rows(layout.tabs.get(&one.id)));
341                    rows.push(crate::widget::tab::bar_row(layout.tab_bar(one.id)));
342                }
343                rows.push(title_row(holder.untitled));
344                let name = one.widget.name();
345                pages.push(Page {
346                    id: page_id(one.id),
347                    title: window
348                        .map_or_else(|| name.to_string(), |window| format!("{name} ({window})")),
349                    file: file.clone(),
350                    takes: "at once".to_owned(),
351                    problem: None,
352                    rows,
353                });
354            }
355        }
356        // The window's own page, after its widgets'.
357        pages.push(layout.title.page(file));
358        pages
359    }

399    pub(crate) fn widget_change(
400        &mut self,
401        page: &str,
402        key: &str,
403        to: Option<&str>,
404    ) -> Result<String, String> {
405        if page == super::titlebar::PAGE {
406            let layout = self
407                .layout
408                .as_mut()
409                .ok_or_else(|| "The window has no layout yet.".to_owned())?;
410            let said = layout.title.set(key, to)?;
411            self.save();
412            return Ok(said);
413        }
414        let placed = page
415            .strip_prefix(PREFIX)
416            .and_then(|id| id.parse::<u32>().ok())
417            .ok_or_else(|| format!("There is no {page} page."))?;
418        // The character's, not the widget's: every effect list shares them.
419        if key == crate::tracked::TABLE {
420            let effects = match to {
421                Some(written) => crate::tracked::from_table(written)?,
422                None => Vec::new(),
423            };
424            self.track(crate::tracked::Change::All(effects));
425            return Ok("Tracked effects: changed.".to_owned());
426        }
427        let layout = self
428            .layout
429            .as_mut()
430            .ok_or_else(|| "The window has no layout yet.".to_owned())?;
431        // The character's too: the circles' colours every effect list wears.
432        if key == crate::widget::circles::TABLE {
433            layout.circles = match to {
434                Some(written) => Some(crate::widget::circles::read(written)?),
435                None => None,
436            };
437            self.save();
438            return Ok("Spell circle colours: changed.".to_owned());
439        }
440        let widget = layout
441            .placed()
442            .into_iter()
443            .find(|one| one.id == placed)
444            .map(|one| one.widget.clone())
445            .ok_or_else(|| "That widget is no longer in the window.".to_owned())?;
446        if key == TITLED {
447            let shown = match to {
448                None => true,
449                Some(to) => on_off(to)?,
450            };
451            let holder = layout
452                .holders
453                .iter_mut()
454                .find(|holder| holder.holds.placed().any(|one| one.id == placed))
455                .ok_or_else(|| "That widget is no longer in the window.".to_owned())?;
456            holder.untitled = !shown;
457            self.save();
458            return Ok(if shown {
459                "Its window shows its title.".to_owned()
460            } else {
461                "Its window's title is hidden, except while arranging.".to_owned()
462            });
463        }
464        // The cell's, not the tab's: every tab of the cell shares its strip.
465        if key == crate::widget::tab::BAR {
466            let bar = match to.map(str::trim) {
467                None => crate::layout::Bar::default(),
468                Some(to) => crate::layout::Bar::of_key(to)
469                    .ok_or_else(|| format!("`{to}` is not a side of the cell."))?,
470            };
471            if !layout.set_tab_bar(placed, bar) {
472                return Err("That widget is in no cell.".to_owned());
473            }
474            self.save();
475            return Ok(format!(
476                "Its tabs run along the {}.",
477                bar.label().to_lowercase()
478            ));
479        }
480        if key.starts_with(crate::widget::tab::PREFIX) {
481            keep(
482                &mut layout.tabs,
483                placed,
484                &crate::widget::tab::TabLook::default(),
485                |look| crate::widget::tab::set(look, key, to),
486            )?;
487            self.save();
488            return Ok(format!("{}: its tab changed.", widget.name()));
489        }
490        if key == FONT || key == TEXT_SIZE {
491            keep(
492                &mut layout.faces,
493                placed,
494                &hydra_ui::theme::Face::default(),
495                |face| face_set(face, key, to),
496            )?;
497            self.save();
498            return Ok(format!("{}: changed.", widget.name()));
499        }
500        layout_change(layout, &widget, placed, (key, to))?;
501        self.save();
502        Ok(format!("{}: changed.", widget.name()))
503    }
504}

506/// `key` set to `to` on the widget `widget` placed as `placed`, kept in
507/// `layout` by its kind.
508fn layout_change(
509    layout: &mut crate::layout::Layout,
510    widget: &Widget,
511    placed: u32,
512    (key, to): (&str, Option<&str>),
513) -> Result<(), String> {
514    match widget {
515        Widget::Room => keep(&mut layout.rooms, placed, &RoomParts::default(), |parts| {
516            room_set(parts, key, to)
517        })?,
518        Widget::Npcs | Widget::Objects => {
519            keep(&mut layout.lists, placed, &Listing::default(), |listing| {
520                list_set(listing, key, to)
521            })?;
522        }
523        Widget::Creatures | Widget::Players => keep(
524            &mut layout.rosters,
525            placed,
526            &crate::widget::roster::RosterLook::default(),
527            |look| crate::widget::roster::set(look, key, to),
528        )?,
529        Widget::Injuries => keep(&mut layout.dolls, placed, &DollLook::default(), |look| {
530            doll_set(look, key, to)
531        })?,
532        Widget::RightHand | Widget::LeftHand | Widget::Prepared | Widget::Reserve => keep(
533            &mut layout.labels,
534            placed,
535            &crate::widget::labels::LabelLook::default(),
536            |look| crate::widget::labels::set(look, key, to),
537        )?,
538        Widget::Aim => keep(
539            &mut layout.aims,
540            placed,
541            &crate::widget::aim::AimLook::default(),
542            |look| crate::widget::aim::set(look, key, to),
543        )?,
544        Widget::Timers => keep(
545            &mut layout.timers,
546            placed,
547            &crate::widget::effects::TimersLook::default(),
548            |look| crate::widget::effects::timers_set(look, key, to),
549        )?,
550        Widget::Quick => keep(
551            &mut layout.quicks,
552            placed,
553            &crate::widget::quick::QuickLook::default(),
554            |look| crate::widget::quick::set(look, key, to),
555        )?,
556        Widget::Hotbar => keep(
557            &mut layout.hotbars,
558            placed,
559            &crate::widget::hotbar::HotbarLook::default(),
560            |look| crate::widget::hotbar::set(look, key, to),
561        )?,
562        Widget::Minimap => keep(
563            &mut layout.minimaps,
564            placed,
565            &crate::widget::minimap::MinimapLook::default(),
566            |look| crate::widget::minimap::look::set(look, key, to),
567        )?,
568        Widget::Story | Widget::Stream(_) => {
569            let kind = lines_kind(widget);
570            keep(&mut layout.lines, placed, &Lines::default(), |lines| {
571                lines_set(lines, key, to, kind)
572            })?;
573        }
574        _ => {
575            let default = widget
576                .bar_look()
577                .ok_or_else(|| format!("{} has no settings of its own.", widget.name()))?;
578            keep(&mut layout.looks, placed, &default, |look| {
579                set(look, &default, key, to)
580            })?;
581        }
582    }
583    Ok(())
584}

586/// Widget `placed`'s entry in `kept` changed by `change`: kept only while it
587/// differs from its kind's own, `default`.
588fn keep<T: Clone + PartialEq>(
589    kept: &mut BTreeMap<u32, T>,
590    placed: u32,
591    default: &T,
592    change: impl FnOnce(&mut T) -> Result<(), String>,
593) -> Result<(), String> {
594    let mut now = kept
595        .get(&placed)
596        .cloned()
597        .unwrap_or_else(|| default.clone());
598    change(&mut now)?;
599    if now == *default {
600        kept.remove(&placed);
601    } else {
602        kept.insert(placed, now);
603    }
604    Ok(())
605}

635fn room_set(parts: &mut RoomParts, key: &str, to: Option<&str>) -> Result<(), String> {
636    let shown = match to {
637        Some(to) => on_off(to)?,
638        None => shows(RoomParts::default(), key),
639    };
640    *part(parts, key).ok_or_else(|| format!("The room has no part {key}."))? = shown;
641    Ok(())
642}

644/// How lines are drawn, `key` set to `to` or to the kind's own; the
645/// prompts and what was typed only on the `story`.
646/// Which widget of lines it is, for the rows only some have: the story's
647/// prompts and echo, the bounty's summary.
648#[derive(Clone, Copy, PartialEq, Eq)]
649enum LinesKind {
650    Story,
651    Bounty,
652    Stream,
653}
654
655/// The kind of `widget`, a widget of lines.
656fn lines_kind(widget: &Widget) -> LinesKind {
657    match widget {
658        Widget::Story => LinesKind::Story,
659        Widget::Stream(id) if id == hydra_model::state::bounty::STREAM => LinesKind::Bounty,
660        _ => LinesKind::Stream,
661    }
662}
663
664fn lines_set(
665    lines: &mut Lines,
666    key: &str,
667    to: Option<&str>,
668    kind: LinesKind,
669) -> Result<(), String> {
670    let story = kind == LinesKind::Story;
671    let default = Lines::default();
672    let switch = |to: Option<&str>, default: bool| to.map_or(Ok(default), on_off);
673    match (key, to) {
674        ("stamps", Some(to)) => {
675            lines.stamps = STAMPS
676                .iter()
677                .find(|(_, value, _)| *value == to)
678                .map(|(stamps, ..)| *stamps)
679                .ok_or_else(|| format!("A line's time does not go `{to}`."))?;
680        }
681        ("stamps", None) => lines.stamps = default.stamps,
682        ("hours", Some(to)) => {
683            lines.hours = HOURS
684                .iter()
685                .find(|(_, value, _)| *value == to)
686                .map(|(hours, ..)| *hours)
687                .ok_or_else(|| format!("There is no `{to}`-hour clock."))?;
688        }
689        ("hours", None) => lines.hours = default.hours,
690        ("seconds", to) => lines.seconds = switch(to, default.seconds)?,
691        ("wrap", to) => lines.wrap = switch(to, default.wrap)?,
692        ("prompts", to) if story => lines.prompts = switch(to, default.prompts)?,
693        ("echo", to) if story => lines.echo = switch(to, default.echo)?,
694        ("summary", to) if kind == LinesKind::Bounty => {
695            lines.summary = switch(to, default.summary)?;
696        }
697        (key, _) => return Err(format!("It has no setting {key}.")),
698    }
699    Ok(())
700}

702/// The Room widget's rows: each part, on or off, as `parts` says or all.
703fn room_rows(parts: Option<&RoomParts>) -> Vec<Row> {
704    let default = RoomParts::default();
705    let now = parts.copied().unwrap_or_default();
706    PARTS
707        .iter()
708        .map(|(key, label, help)| {
709            let on = shows(now, key);
710            let was = shows(default, key);
711            Row {
712                key: (*key).to_owned(),
713                label: (*label).to_owned(),
714                help: (*help).to_owned(),
715                kind: RowKind::Toggle,
716                value: Value::On(on),
717                here: on != was,
718                from: None,
719            }
720        })
721        .collect()
722}

724/// A story's or a stream's rows: how it draws its lines, as `lines` says or
725/// as it always has; the prompts and what was typed only on the `story`.
726fn lines_rows(lines: Option<&Lines>, kind: LinesKind) -> Vec<Row> {
727    let story = kind == LinesKind::Story;
728    let default = Lines::default();
729    let now = lines.copied().unwrap_or_default();
730    let row = |key: &str, label: &str, help: &str, kind: RowKind, value: Value, here: bool| Row {
731        key: key.to_owned(),
732        label: label.to_owned(),
733        help: help.to_owned(),
734        kind,
735        value,
736        here,
737        from: None,
738    };
739    let switch = |key: &str, label: &str, help: &str, now: bool, was: bool| {
740        row(
741            key,
742            label,
743            help,
744            RowKind::Toggle,
745            Value::On(now),
746            now != was,
747        )
748    };
749    let stamps = STAMPS
750        .iter()
751        .find(|(each, ..)| *each == now.stamps)
752        .map_or("", |(_, value, _)| value);
753    let hours = HOURS
754        .iter()
755        .find(|(each, ..)| *each == now.hours)
756        .map_or("", |(_, value, _)| value);
757    let mut rows = vec![
758        row(
759            "stamps",
760            "Timestamps",
761            "When each line arrived, on your clock: before it, after it, or not at all.",
762            choice(STAMPS.iter().map(|(_, value, called)| (*value, *called))),
763            Value::Text(stamps.to_owned()),
764            now.stamps != default.stamps,
765        ),
766        switch(
767            "seconds",
768            "With seconds",
769            "7:08:05 rather than 7:08.",
770            now.seconds,
771            default.seconds,
772        ),
773        row(
774            "hours",
775            "Clock",
776            "7:08 PM, or 19:08.",
777            choice(HOURS.iter().map(|(_, value, called)| (*value, *called))),
778            Value::Text(hours.to_owned()),
779            now.hours != default.hours,
780        ),
781        switch(
782            "wrap",
783            "Word wrap",
784            "A long line wraps; off, the window scrolls sideways.",
785            now.wrap,
786            default.wrap,
787        ),
788    ];
789    if story {
790        rows.push(switch(
791            "prompts",
792            "Prompts",
793            "The game's prompt, >, after what it says.",
794            now.prompts,
795            default.prompts,
796        ));
797        rows.push(switch(
798            "echo",
799            "What you type",
800            "Each line you send, after the prompt it followed.",
801            now.echo,
802            default.echo,
803        ));
804    }
805    if kind == LinesKind::Bounty {
806        rows.push(switch(
807            "summary",
808            "Summary",
809            "The task in brief, up to four lines: what kind, what and how many, where.",
810            now.summary,
811            default.summary,
812        ));
813    }
814    rows
815}

817/// A choice among `named`, each its value and what a player calls it.
818fn choice<'a>(named: impl Iterator<Item = (&'a str, &'a str)>) -> RowKind {
819    RowKind::Choice(
820        named
821            .map(|(value, called)| (value.to_owned(), called.to_owned()))
822            .collect(),
823    )
824}

826/// The overlays to choose from: none, then each image by its file's name.
827fn overlay_choice(overlays: &[PathBuf]) -> RowKind {
828    let mut images: Vec<(String, String)> = vec![(String::new(), "None".to_owned())];
829    images.extend(overlays.iter().map(|path| {
830        let mut called = path.file_name().map_or_else(
831            || path.display().to_string(),
832            |name| name.to_string_lossy().into_owned(),
833        );
834        // A pack's picture, told from the player's own of the same name.
835        if path.parent().is_some_and(|folder| {
836            folder.ends_with(std::path::Path::new("global").join("images").join("dolls"))
837        }) {
838            called.push_str(" (pack)");
839        }
840        (path.display().to_string(), called)
841    }));
842    RowKind::Choice(images)
843}

845/// A room list's one row: across with commas, or down.
846fn list_rows(listing: Option<Listing>) -> Vec<Row> {
847    let now = listing.unwrap_or_default();
848    vec![Row {
849        key: "list".to_owned(),
850        label: "List them".to_owned(),
851        help: "On a line with commas between, or one to a line.".to_owned(),
852        kind: RowKind::Choice(
853            Listing::CHOICES
854                .iter()
855                .map(|(_, value, called)| ((*value).to_owned(), (*called).to_owned()))
856                .collect(),
857        ),
858        value: Value::Text(now.value().to_owned()),
859        here: now != Listing::default(),
860        from: None,
861    }]
862}
863
864/// A room list's `key` set to `to`, or back to across.
865fn list_set(listing: &mut Listing, key: &str, to: Option<&str>) -> Result<(), String> {
866    match (key, to) {
867        ("list", None) => *listing = Listing::default(),
868        ("list", Some(to)) => {
869            *listing = Listing::of(to).ok_or_else(|| format!("A list is not laid out `{to}`."))?;
870        }
871        _ => return Err(format!("A room list has no setting {key}.")),
872    }
873    Ok(())
874}

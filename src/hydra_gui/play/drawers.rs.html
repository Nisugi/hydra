1//! A play window's drawers as drawn (the model is
2//! `crate::layout`'s `drawers.rs`): each open drawer's backdrop, the edge
3//! it is resized by, and the Drawers menu on the top bar.
4//!
5//! **Layers.** The main area's windows are egui's `Middle` order; a
6//! drawer's backdrop is `Foreground`, over them, and its windows and its
7//! edge are sublayers of the backdrop, drawn just above it wherever egui
8//! raises it (`egui::Context::set_sublayer`). One call does it, where
9//! sinking the backdrop under its windows would mean raising every other
10//! layer each frame.
11//!
12//! **Click-through** is egui's own: a backdrop that hides nothing is not
13//! interactable, so egui's hit test passes it by (`Area::interactable`), and
14//! a press reaches what is under it. The drawer's windows stay interactable,
15//! so a press on one is the window's: where there is no window there is no
16//! widget.
17
18use egui::{CursorIcon, Id, LayerId, Order, Rect, Sense, Stroke};
19
20use super::Play;
21use crate::layout::{CLEAR, Drawers, Mode, SMALLEST, THINNEST, Zone, Zones};
22use crate::theme::{self, T};

24/// How thick a drawer's edge is, to take a press: half in the drawer, half
25/// out.
26const EDGE: f32 = 6.0;
27
28/// Drawer `zone`'s backdrop layer in play window `session`: its windows
29/// are drawn just above it.
30pub(super) fn backdrop(session: u32, zone: Zone) -> LayerId {
31    LayerId::new(Order::Foreground, Id::new(("play-drawer", session, zone)))
32}
33
34/// Drawer `zone`'s edge layer.
35fn edge(session: u32, zone: Zone) -> LayerId {
36    LayerId::new(
37        Order::Foreground,
38        Id::new(("play-drawer-edge", session, zone)),
39    )
40}
41
42impl Play {
43    /// Each open drawer's backdrop in play area `area`, with the edge that
44    /// resizes it, unless the layout is locked. `true` when a resize ended,
45    /// so the layout wants saving.
46    pub(super) fn drawers(&mut self, context: &egui::Context, area: Rect) -> bool {
47        let session = self.session;
48        let Some(layout) = self.layout.as_mut() else {
49            return false;
50        };
51        let zones = Zones::of(&layout.drawers, area.size(), layout.arranged());
52        let mut ended = false;
53        for &(zone, rect) in zones.drawers() {
54            let Some(drawer) = layout.drawers.get_mut(zone) else {
55                continue;
56            };
57            let at = rect.translate(area.min.to_vec2());
58            let layer = backdrop(session, zone);
59            let opacity = drawer.shown_opacity();
60            // Seen, it takes a press on its bare backdrop. Clear, it only
61            // paints: egui's hit test stops at any widget in a layer, even
62            // the one an area made that is not interactable.
63            if !drawer.clear() {
64                egui::Area::new(layer.id)
65                    .order(Order::Foreground)
66                    .fixed_pos(at.min)
67                    .show(context, |ui| {
68                        ui.allocate_exact_size(at.size(), Sense::click_and_drag());
69                    });
70            }
71            let visuals = &context.global_style().visuals;
72            let painter = context.layer_painter(layer);
73            painter.rect_filled(at, 0.0, visuals.panel_fill.gamma_multiply(opacity));
74            // The line along its inner edge fades with it: fading only the
75            // fill would leave a hairline floating in space.
76            let line = Stroke::new(1.5, visuals.window_stroke.color.gamma_multiply(opacity));
77            let inner = Zones::edge(zone, at, 0.0);
78            painter.line_segment([inner.min, inner.max], line);
79            if layout.locked {
80                continue;
81            }
82            let strip = Zones::edge(zone, at, EDGE);
83            let edge_layer = edge(session, zone);
84            egui::Area::new(edge_layer.id)
85                .order(Order::Foreground)
86                .fixed_pos(strip.min)
87                .show(context, |ui| {
88                    let cursor = if Zones::sideways(zone) {
89                        CursorIcon::ResizeHorizontal
90                    } else {
91                        CursorIcon::ResizeVertical
92                    };
93                    let (_, response) = ui.allocate_exact_size(strip.size(), Sense::drag());
94                    let response = response.on_hover_and_drag_cursor(cursor);
95                    if response.hovered() || response.dragged() {
96                        let middle = Zones::edge(zone, at, 0.0);
97                        ui.painter().line_segment(
98                            [middle.min, middle.max],
99                            Stroke::new(2.0, theme::color(ui.ctx(), T::Accent)),
100                        );
101                    }
102                    if response.dragged() {
103                        let most = if Zones::sideways(zone) {
104                            area.width() - SMALLEST.x
105                        } else {
106                            area.height() - SMALLEST.y
107                        };
108                        let grown =
109                            drawer.size.max(THINNEST) + Zones::grown(zone, response.drag_delta());
110                        drawer.size = grown.clamp(THINNEST, most.max(THINNEST));
111                    }
112                    ended |= response.drag_stopped();
113                });
114            context.set_sublayer(layer, edge_layer);
115            context.move_to_top(edge_layer);
116        }
117        ended
118    }
119}

121/// The Drawers menu: each drawer shown or not, pushing or clipping, and a
122/// clip drawer's opacity. Whether anything changed.
123pub(super) fn menu(ui: &mut egui::Ui, drawers: &mut Drawers) -> bool {
124    let mut changed = false;
125    for zone in Zone::DRAWERS {
126        let Some(drawer) = drawers.get_mut(zone) else {
127            continue;
128        };
129        ui.horizontal(|ui| {
130            changed |= ui.checkbox(&mut drawer.open, zone.name()).changed();
131            changed |= ui
132                .radio_value(&mut drawer.mode, Mode::Push, "Push")
133                .on_hover_text("Take its room from the main area, whose windows move over")
134                .changed();
135            changed |= ui
136                .radio_value(&mut drawer.mode, Mode::Clip, "Clip")
137                .on_hover_text("Lie over the main area, which stays where it is")
138                .changed();
139        });
140        if drawer.mode == Mode::Clip {
141            ui.horizontal(|ui| {
142                ui.add_space(24.0);
143                changed |= ui
144                    .add(egui::Slider::new(&mut drawer.opacity, 0.0..=1.0).text("Opacity").fixed_decimals(2))
145                    .on_hover_text(format!(
146                        "How much of the main area it hides. Below {CLEAR:.2}, a click on the drawer where no window is reaches what is under it."
147                    ))
148                    .changed();
149            });
150        }
151    }
152    ui.weak("Drag a window into an open drawer to keep it there; drag a drawer's edge to size it.");
153    changed
154}

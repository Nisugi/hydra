1//! Carrying an object from a link. Holding the drag key and dragging a link
2//! that is an object carries it: dropped on a blank area of the story it is
3//! dropped on the ground, onto a hand widget it goes into that hand, onto a
4//! container widget or container link it goes into that container. Each
5//! move is the game's `_drag` command.
6//!
7//! egui's drag and drop holds the object from the press, once the pointer
8//! has moved; each place that takes one asks for it as the pointer lets go,
9//! and says `_drag #<item> <onto>` (`_drag #item #bag`). Let go anywhere
10//! else, nothing happens: only the story's blank area is the floor.
11//!
12//! An object is carried from a link in the story or a stream, from a hand,
13//! or from a container's contents, so items can be moved around; never
14//! from a widget showing another character's, whose ids are not this
15//! character's to send.
16
17use egui::{Id, Modifiers, Order, Sense};
18
19/// An object being carried: its id, and what the line called it.
20#[derive(Clone, Debug, PartialEq, Eq)]
21pub(crate) struct Carried {
22    /// Its exist id.
23    pub(crate) exist: String,
24    /// The link's words: `a grey rat`.
25    pub(crate) name: String,
26}
27
28/// Where the drag key is kept in the context, for every window to read.
29fn key_id() -> Id {
30    Id::new("hydra-drag-key")
31}
32
33/// The key held to carry an object: Hydra's own setting, Alt unless the
34/// player chose (`own.rs`).
35pub(crate) fn key(context: &egui::Context) -> Modifiers {
36    context
37        .data(|data| data.get_temp::<Modifiers>(key_id()))
38        .unwrap_or(Modifiers::ALT)
39}
40
41/// Whether the drag key is held, and nothing else with it
42/// (`matches_exact`), so Ctrl and Shift together, or `AltGr`, is not it.
43pub(crate) fn held(ui: &egui::Ui) -> bool {
44    let key = key(ui.ctx());
45    ui.input(|input| input.modifiers.matches_exact(key))
46}
47
48/// `object` carried from `response` once it is dragged, with the drag key
49/// held: a source's widget senses a drag only then (`sense`).
50pub(crate) fn source(response: &egui::Response, object: Carried) {
51    response.dnd_set_drag_payload(object);
52}
53
54/// How a widget that carries an object senses: a drag, while the drag key
55/// is held; otherwise nothing, and a drag moves its window as ever.
56pub(crate) fn sense(ui: &egui::Ui) -> Sense {
57    if held(ui) {
58        Sense::drag()
59    } else {
60        Sense::hover()
61    }
62}
63
64/// What is let go on `ui`'s whole space this frame, put `onto` there:
65/// `_drag #<item> <onto>`; the object `holding` there already, let go on
66/// its own place, nothing.
67pub(crate) fn target(
68    ui: &mut egui::Ui,
69    id: Id,
70    onto: &str,
71    holding: Option<&str>,
72) -> Option<String> {
73    let space = ui.interact(ui.max_rect(), id.with("drop"), Sense::hover());
74    dropped(&space, onto, holding)
75}
76
77/// What is let go on `response` this frame, put `onto` it; `itself`, the
78/// object the place is or holds, let go there, nothing.
79pub(crate) fn dropped(
80    response: &egui::Response,
81    onto: &str,
82    itself: Option<&str>,
83) -> Option<String> {
84    let carried = response.dnd_release_payload::<Carried>()?;
85    (itself != Some(carried.exist.as_str())).then(|| format!("_drag #{} {onto}", carried.exist))
86}
87
88/// While an object is carried, what it is, beside the pointer, with the
89/// grabbing hand: `Dragging: ...`.
90pub(crate) fn show(context: &egui::Context, session: u32) {
91    let Some(carried) = egui::DragAndDrop::payload::<Carried>(context) else {
92        return;
93    };
94    let Some(at) = context.pointer_latest_pos() else {
95        return;
96    };
97    context.set_cursor_icon(egui::CursorIcon::Grabbing);
98    egui::Area::new(Id::new(("carrying", session)))
99        .order(Order::Tooltip)
100        .fixed_pos(at + egui::vec2(14.0, 14.0))
101        .interactable(false)
102        .show(context, |ui| {
103            egui::Frame::popup(ui.style()).show(ui, |ui| {
104                ui.label(format!("Dragging: {}", carried.name));
105            });
106        });
107}

1//! Where a play window's layout is kept, and reading and writing it: a file
2//! per character, by game and name.
3
4use std::path::{Path, PathBuf};
5
6pub(crate) use hydra_board::kept::file;
7#[cfg(test)]
8pub(crate) use hydra_board::kept::{keep_unread, unread};
9
10use super::{Layout, VERSION};
11
12impl Layout {
13    /// `character`'s saved layout on `instance` in `dir`: by game and name,
14    /// so a character on Prime and one of the same name on Shattered each
15    /// keep their own. One saved under the name alone,
16    /// as layouts were kept before, is taken as the first. `None` when there
17    /// is none, or it cannot be read, when a fitted one does.
18    pub(crate) fn load(dir: &Path, instance: Option<&str>, character: &str) -> Option<Self> {
19        let read = |path: PathBuf| {
20            let text = std::fs::read_to_string(path).ok()?;
21            serde_json::from_str::<Self>(&text)
22                .ok()
23                .filter(|layout| layout.version == VERSION)
24        };
25        read(file(dir, instance, character))
26            .or_else(|| instance.and_then(|_| read(file(dir, None, character))))
27    }
28
29    /// Save this as `character`'s layout on `instance` in `dir`. **A file
30    /// already there that [`Layout::load`] could not read is kept**, beside
31    /// it as `.unread`: a hand edit that broke it, or a later build's
32    /// layout, would otherwise be fitted over and then written over by the
33    /// first window dragged. The keys' writer refuses such a
34    /// file; a layout is saved on every drag, so this one steps round it.
35    ///
36    /// # Errors
37    ///
38    /// The folder could not be made or the file written, or an unread file
39    /// could not be kept: one kept earlier is still there.
40    #[cfg(test)]
41    pub(crate) fn save(
42        &self,
43        dir: &Path,
44        instance: Option<&str>,
45        character: &str,
46    ) -> std::io::Result<()> {
47        let path = file(dir, instance, character);
48        keep_unread(&path, Self::reads)?;
49        hydra_session::store::save_json(dir, &path, self)
50    }
51
52    /// Kept under the character's name soon, off the window's thread, by
53    /// `later`, as [`Self::save`] keeps it.
54    pub(crate) fn save_later(
55        &self,
56        later: &mut crate::store::Later,
57        dir: &Path,
58        (instance, character): (Option<&str>, &str),
59    ) {
60        if let Ok(text) = serde_json::to_string_pretty(self) {
61            let path = file(dir, instance, character);
62            later.write_keeping_unread(dir, path, text.into_bytes(), Self::reads);
63        }
64    }
65
66    /// Whether `text` reads as a layout this build keeps.
67    fn reads(text: &str) -> bool {
68        serde_json::from_str::<Layout>(text).is_ok_and(|kept| kept.version == VERSION)
69    }
70
71    /// The layout saved as `name` in `dir`'s `named` folder, for any
72    /// character to take (`.loadlayout`); `None` when there is none, or it
73    /// cannot be read.
74    pub(crate) fn load_named(dir: &Path, name: &str) -> Option<Self> {
75        let text = std::fs::read_to_string(named_file(dir, name)?).ok()?;
76        serde_json::from_str::<Self>(&text)
77            .ok()
78            .filter(|layout| layout.version == VERSION)
79    }
80
81    /// Save this as `name` in `dir`'s `named` folder, written at once
82    /// (`.savelayout`): the file's path.
83    ///
84    /// # Errors
85    ///
86    /// The name is not one a file can have, or the file could not be
87    /// written.
88    pub(crate) fn save_named(&self, dir: &Path, name: &str) -> Result<PathBuf, String> {
89        let path = named_file(dir, name)
90            .ok_or_else(|| format!("`{name}` is not a name a layout can be saved as."))?;
91        let folder = dir.join(NAMED);
92        std::fs::create_dir_all(&folder).map_err(|why| why.to_string())?;
93        hydra_session::store::save_json(&folder, &path, self).map_err(|why| why.to_string())?;
94        Ok(path)
95    }
96}
97
98/// The names of the layouts saved in `dir`'s `named` folder, sorted.
99pub(crate) fn named_layouts(dir: &Path) -> Vec<String> {
100    let mut names: Vec<String> = std::fs::read_dir(dir.join(NAMED))
101        .map(|entries| {
102            entries
103                .filter_map(Result::ok)
104                .map(|entry| entry.path())
105                .filter(|path| path.extension().is_some_and(|x| x == "json"))
106                .filter_map(|path| {
107                    path.file_stem()
108                        .map(|stem| stem.to_string_lossy().into_owned())
109                })
110                .collect()
111        })
112        .unwrap_or_default();
113    names.sort();
114    names
115}
116
117/// The folder of `dir` where whole layouts are kept by name.
118const NAMED: &str = "named";
119
120/// Where the layout `name` is kept in `dir`; `None` when `name` is not a
121/// file's name: letters, digits, `-`, `_` and spaces, nothing else.
122fn named_file(dir: &Path, name: &str) -> Option<PathBuf> {
123    let name = name.trim();
124    let plain = !name.is_empty()
125        && name
126            .chars()
127            .all(|c| c.is_alphanumeric() || matches!(c, '-' | '_' | ' '));
128    plain.then(|| dir.join(NAMED).join(format!("{name}.json")))
129}

131#[cfg(test)]
132mod named_tests {
133    use super::*;
134
135    /// A layout saved by name reads back whole for anyone, is listed, and
136    /// a name no file can have is refused.
137    #[test]
138    fn a_layout_is_kept_by_name_for_any_character() {
139        let dir = std::env::temp_dir().join(format!("hydra-named-{}", std::process::id()));
140        let _ = std::fs::remove_dir_all(&dir);
141        let mut layout = Layout::fitted(egui::Vec2::new(900.0, 600.0));
142        layout.grid = 12.0;
143        let path = layout.save_named(&dir, "hunting").expect("saved");
144        assert!(path.ends_with("named/hunting.json") || path.ends_with("named\\hunting.json"));
145        assert_eq!(Layout::load_named(&dir, "hunting"), Some(layout));
146        assert_eq!(Layout::load_named(&dir, "resting"), None);
147        assert_eq!(named_layouts(&dir), ["hunting"]);
148        assert!(
149            Layout::fitted(egui::Vec2::new(900.0, 600.0))
150                .save_named(&dir, "../escape")
151                .is_err()
152        );
153        let _ = std::fs::remove_dir_all(&dir);
154    }

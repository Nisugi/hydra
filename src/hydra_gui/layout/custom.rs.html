1//! A custom window's inside: its widgets held bare, each in a cell -- a
2//! free rect, kept from the inside's top left -- the play window's own model
3//! one level down.
4//!
5//! When the window is resized its cells follow: across, each scales with the
6//! inside's width, so a row of bars stays as wide as the window and two
7//! clocks side by side stay halves; down, a cell on the bottom edge keeps to
8//! it, so the last list takes the new room and a line stays a line. That is
9//! the one anchor a custom window needs; no graph of anchors is solved per
10//! frame.
11

18/// How near the bottom edge a cell's bottom must be to keep to it.
19const TOUCHING: f32 = 0.5;
20
21/// The smallest a cell becomes, when its window shrinks or it is resized.
22pub(crate) const SMALLEST: Vec2 = Vec2::new(20.0, LINE);

35/// One place in a custom window: a widget, or a tab stack of several with
36/// one showing.
37#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
38pub(crate) struct Cell {

40    rect: [f32; 4],
41    /// Its widgets: one, or a tab stack's.
42    pub(crate) tabs: Vec<Placed>,
43    /// Which of them shows.
44    pub(crate) showing: usize,
45    /// Which side its tab strip runs along.
46    #[serde(default, skip_serializing_if = "Bar::is_top")]
47    pub(crate) bar: Bar,

50/// Which side of a cell its tab strip runs along.
51#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
52#[serde(rename_all = "snake_case")]
53pub(crate) enum Bar {
54    #[default]
55    Top,
56    Bottom,
57    Left,
58    Right,
59}

61/// How wide a strip down a side is, at most: a third of the cell when that
62/// is less.
63const SIDE_STRIP: f32 = 96.0;

65impl Bar {
66    pub(crate) const ALL: [Bar; 4] = [Bar::Top, Bar::Bottom, Bar::Left, Bar::Right];

68    #[expect(
69        clippy::trivially_copy_pass_by_ref,
70        reason = "serde's skip_serializing_if hands a reference"

72    fn is_top(&self) -> bool {
73        *self == Bar::Top
74    }

76    /// Down a side: the tabs run top to bottom.
77    pub(crate) const fn vertical(self) -> bool {
78        matches!(self, Bar::Left | Bar::Right)
79    }

82    pub(crate) const fn key(self) -> &'static str {
83        match self {
84            Bar::Top => "top",
85            Bar::Bottom => "bottom",
86            Bar::Left => "left",
87            Bar::Right => "right",
88        }
89    }

91    /// What a player calls it.
92    pub(crate) const fn label(self) -> &'static str {
93        match self {
94            Bar::Top => "Top",
95            Bar::Bottom => "Bottom",
96            Bar::Left => "Left",
97            Bar::Right => "Right",
98        }

108    /// An empty custom window titled `title`, whose inside is `inside`
109    /// across.
110    pub(crate) fn empty(title: &str, inside: Vec2) -> Self {
111        Self {
112            title: title.to_owned(),
113            inside: [inside.x, inside.y],
114            cells: Vec::new(),
115        }

118    /// Take widget `placed` out: its cell goes, or, when the cell is a tab
119    /// stack of others too, its tab.
120    pub(crate) fn take(&mut self, placed: u32) -> Option<Placed> {
121        let at = self
122            .cells
123            .iter()
124            .position(|cell| cell.tabs.iter().any(|tab| tab.id == placed))?;
125        let cell = &mut self.cells[at];
126        let tab = cell.tabs.iter().position(|tab| tab.id == placed)?;
127        let taken = cell.tabs.remove(tab);
128        if cell.tabs.is_empty() {
129            self.cells.remove(at);
130        } else {
131            cell.showing = cell.showing.min(cell.tabs.len() - 1);
132        }
133        Some(taken)
134    }
135
136    /// A custom window titled `title` of one cell filling its inside, `inside`
137    /// across: a tab stack of `tabs`, `showing` the one shown.
138    pub(crate) fn stack(title: &str, tabs: Vec<Placed>, showing: usize, inside: Vec2) -> Self {
139        Self {
140            title: title.to_owned(),
141            inside: [inside.x, inside.y],
142            cells: vec![Cell {
143                rect: kept(Rect::from_min_size(Pos2::ZERO, inside)),
144                tabs,
145                showing,
146                bar: Bar::default(),
147            }],
148        }
149    }
150
151    /// Take out the whole cell showing or stacking widget `placed`.
152    pub(crate) fn take_cell(&mut self, placed: u32) -> Option<Cell> {
153        let at = self
154            .cells
155            .iter()
156            .position(|cell| cell.tabs.iter().any(|tab| tab.id == placed))?;
157        Some(self.cells.remove(at))
158    }
159
160    /// The cell whose stacking middle is at `at`, from the inside's top left
161    /// (`stacks_at`): the one drawn last, on top, where cells overlap.
162    pub(crate) fn stack_at(&self, at: Pos2) -> Option<usize> {
163        self.cells
164            .iter()
165            .rposition(|cell| stacks_at(cell.rect(), at))
166    }

168    /// `tabs`, `showing` the one shown, let go at `at` inside: onto the
169    /// middle of a cell, whose tab stack they join, or else in a cell of
170    /// their own with its corner there, as big as the one shown asks and the
171    /// inside allows, and kept inside.
172    pub(crate) fn land(&mut self, tabs: Vec<Placed>, showing: usize, at: Pos2) {
173        if let Some(onto) = self.stack_at(at)
174            && stack_together(&tabs)
175            && self.cells[onto].stacks()
176        {
177            self.cells[onto].tabs.extend(tabs);
178            return;
179        }
180        let Some(shown) = tabs.get(showing).or_else(|| tabs.first()) else {
181            return;
182        };
183        let inside = Vec2::new(self.inside[0], self.inside[1]);
184        let strip = if tabs.len() > 1 { LINE } else { 0.0 };
185        let size = (shown.widget.size() + Vec2::new(0.0, strip))
186            .min(inside)
187            .max(SMALLEST);
188        let furthest = (inside - size).max(Vec2::ZERO).to_pos2();
189        self.cells.push(Cell {
190            rect: kept(Rect::from_min_size(at.clamp(Pos2::ZERO, furthest), size)),
191            tabs,
192            showing,
193            bar: Bar::default(),
194        });
195    }

197    /// `from` -- a tab, or a whole cell -- taken out and joined to the tab
198    /// stack of the cell holding widget `into`. Nothing moves when that is
199    /// where it already is.
200    pub(crate) fn stack_onto(&mut self, from: super::Taking, into: u32) {
201        let holding = |cell: &Cell, placed: u32| cell.tabs.iter().any(|tab| tab.id == placed);
202        // Only streams stack: neither side may hold anything else.
203        let (super::Taking::Tab(moving) | super::Taking::Cell(moving)) = from;
204        let stacks = self.cells.iter().all(|cell| {
205            !(holding(cell, moving) || holding(cell, into))
206                || match from {
207                    super::Taking::Tab(_) if holding(cell, moving) && !holding(cell, into) => cell
208                        .tabs
209                        .iter()
210                        .find(|tab| tab.id == moving)
211                        .is_some_and(|tab| tab.widget.stacks()),
212                    _ => cell.stacks(),
213                }
214        });
215        if !stacks {
216            return;
217        }
218        let already = match from {
219            super::Taking::Tab(placed) | super::Taking::Cell(placed) => self
220                .cells
221                .iter()
222                .any(|cell| holding(cell, placed) && holding(cell, into)),
223        };
224        if already {
225            return;
226        }
227        let tabs = match from {
228            super::Taking::Tab(placed) => self.take(placed).map(|tab| vec![tab]),
229            super::Taking::Cell(placed) => self.take_cell(placed).map(|cell| cell.tabs),
230        };
231        if let (Some(tabs), Some(cell)) =
232            (tabs, self.cells.iter_mut().find(|cell| holding(cell, into)))
233        {
234            cell.tabs.extend(tabs);
235        }
236    }

238    /// A custom window titled `title` whose inside is `inside` across, its
239    /// widgets in `rows`: a row's widgets side by side, sharing its width, as
240    /// tall as its tallest asks, and the last row down to the bottom. When
241    /// the rows ask more than the inside has, those taller than a line share
242    /// what the one-line rows leave.
243    pub(crate) fn rows(title: &str, rows: Vec<Vec<Placed>>, inside: Vec2) -> Self {
244        let asked: Vec<f32> = rows
245            .iter()
246            .map(|row| {
247                row.iter()
248                    .map(|placed| placed.widget.size().y)
249                    .fold(LINE, f32::max)
250            })
251            .collect();
252        let total: f32 = asked.iter().sum();
253        let heights: Vec<f32> = if total <= inside.y {
254            asked
255        } else {
256            let lines: f32 = asked.iter().filter(|height| **height <= LINE).sum();
257            let tall = asked.iter().filter(|height| **height > LINE).count();
258            let share = (inside.y - lines) / f32::from(u16::try_from(tall.max(1)).unwrap_or(1));
259            asked
260                .iter()
261                .map(|height| {
262                    if *height <= LINE {
263                        *height
264                    } else {
265                        share.max(LINE)
266                    }
267                })
268                .collect()
269        };
270        let count = rows.len();
271        let mut cells = Vec::new();
272        let mut y = 0.0;
273        for (at, (row, height)) in rows.into_iter().zip(heights).enumerate() {
274            let height = if at + 1 == count {
275                (inside.y - y).max(height)
276            } else {
277                height
278            };
279            let width = inside.x / f32::from(u16::try_from(row.len().max(1)).unwrap_or(1));
280            let mut x = 0.0;
281            for placed in row {
282                cells.push(Cell {
283                    rect: kept(Rect::from_min_size(pos2(x, y), Vec2::new(width, height))),
284                    tabs: vec![placed],
285                    showing: 0,
286                    bar: Bar::default(),
287                });
288                x += width;
289            }
290            y += height;
291        }
292        Self {
293            title: title.to_owned(),
294            inside: [inside.x, inside.y],
295            cells,
296        }
297    }

299    /// The inside's size its cells were last kept to.

304    /// An inside now `inside`, its cells left their size: a window resized
305    /// while arranging.
306    pub(crate) fn keep(&mut self, inside: Vec2) {
307        self.inside = [inside.x, inside.y];
308    }

310    /// Keep the cells to an inside now `inside` across, as the module says:
311    /// scaled across, the bottom row kept to the bottom. Whether anything
312    /// moved.
313    pub(crate) fn fit(&mut self, inside: Vec2) -> bool {
314        let [width, height] = self.inside;
315        if (inside.x - width).abs() < TOUCHING && (inside.y - height).abs() < TOUCHING {
316            return false;
317        }
318        let across = if width > 0.0 { inside.x / width } else { 1.0 };
319        for cell in &mut self.cells {
320            let mut at = cell.rect();
321            at.min.x *= across;
322            at.max.x = (at.max.x * across).max(at.min.x + SMALLEST.x);
323            if at.max.y >= height - TOUCHING {
324                at.max.y = inside.y.max(at.min.y + SMALLEST.y);
325            }
326            cell.rect = kept(at);
327        }
328        self.inside = [inside.x, inside.y];
329        true
330    }

333impl Cell {
334    /// Where it sits, from the inside's top left.
335    pub(crate) fn rect(&self) -> Rect {
336        rect(self.rect)
337    }
338
339    /// Whether every widget it holds may be a tab of a stack.
340    pub(crate) fn stacks(&self) -> bool {
341        stack_together(&self.tabs)
342    }
343
344    /// Tab `at` moved to `to` along the strip; the tab showing stays the
345    /// one showing. Whether anything moved.
346    pub(crate) fn move_tab(&mut self, at: usize, to: usize) -> bool {
347        if at == to || at >= self.tabs.len() || to >= self.tabs.len() {
348            return false;
349        }
350        let showing = self.tabs.get(self.showing).map(|tab| tab.id);
351        let tab = self.tabs.remove(at);
352        self.tabs.insert(to, tab);
353        if let Some(showing) = showing {
354            self.showing = self
355                .tabs
356                .iter()
357                .position(|tab| tab.id == showing)
358                .unwrap_or(0);
359        }
360        true
361    }

366    }

368    /// The widget showing: the only one, or the tab chosen.
369    pub(crate) fn shown(&self) -> Option<&Placed> {
370        self.tabs.get(self.showing).or_else(|| self.tabs.first())

374/// Whether every one of `tabs` may be a tab of a stack.

379/// Where a tab is moved along its strip.
380#[derive(Clone, Copy, Debug, PartialEq, Eq)]

387/// Which of the tab rects `tabs` (of `tabs_and_body`) the pointer at `at`
388/// is over, along the strip alone (`vertical`, down a side): a tab dragged
389/// along the strip is still over it a little off it.
390pub(crate) fn tab_at(tabs: &[Rect], at: Pos2, vertical: bool) -> Option<usize> {
391    let along = |rect: &Rect| {
392        if vertical {
393            rect.y_range().contains(at.y)
394        } else {
395            rect.x_range().contains(at.x)
396        }
397    };
398    let past = |rect: &Rect| {
399        if vertical {
400            at.y >= rect.min.y
401        } else {
402            at.x >= rect.min.x
403        }
404    };
405    tabs.iter()
406        .position(along)
407        .or_else(|| (!tabs.is_empty() && past(&tabs[0])).then_some(tabs.len() - 1))
408}

410/// Whether something let go at `at` stacks onto a cell at `rect`: over the
411/// middle half of its width. Anywhere else over it, a cell is moved or
412/// placed there instead, so a full custom window can still be rearranged.
413pub(crate) fn stacks_at(rect: Rect, at: Pos2) -> bool {
414    rect.shrink2(Vec2::new(rect.width() / 4.0, 0.0))
415        .contains(at)
416}

418/// A cell drawn at `at` holding `tabs` widgets: a rect for each tab, when
419/// it is a tab stack, in a strip along the side `bar` says -- one line
420/// tall across the top or bottom, the tabs side by side; or down the left
421/// or right, the tabs one under another, each a line tall -- and the rest,
422/// which the widget shown fills.
423pub(crate) fn tabs_and_body(at: Rect, tabs: usize, height: f32, bar: Bar) -> (Vec<Rect>, Rect) {
424    if tabs < 2 {
425        return (Vec::new(), at);
426    }
427    let count = f32::from(u16::try_from(tabs).unwrap_or(u16::MAX));
428    let (strip, body) = match bar {
429        Bar::Top => at.split_top_bottom_at_y(at.min.y + height),
430        Bar::Bottom => {
431            let (body, strip) = at.split_top_bottom_at_y(at.max.y - height);
432            (strip, body)
433        }
434        Bar::Left => at.split_left_right_at_x(at.min.x + SIDE_STRIP.min(at.width() / 3.0)),
435        Bar::Right => {
436            let (body, strip) =
437                at.split_left_right_at_x(at.max.x - SIDE_STRIP.min(at.width() / 3.0));
438            (strip, body)
439        }
440    };
441    let rects = if bar.vertical() {
442        (0..tabs)
443            .map(|n| {
444                let y = strip.min.y + height * f32::from(u16::try_from(n).unwrap_or(u16::MAX));
445                Rect::from_min_size(pos2(strip.min.x, y), Vec2::new(strip.width(), height))
446            })
447            .filter(|tab| tab.max.y <= strip.max.y + 0.5)
448            .collect()
449    } else {
450        let width = strip.width() / count;
451        (0..tabs)
452            .map(|n| {
453                let x = strip.min.x + width * f32::from(u16::try_from(n).unwrap_or(u16::MAX));
454                Rect::from_min_size(pos2(x, strip.min.y), Vec2::new(width, height))
455            })
456            .collect()
457    };
458    (rects, body)
459}
460/// Every widget placed in a layout, and its tabs turned.
461impl Layout {
462    /// Every widget showing, in the order the windows are drawn: a
463    /// standalone one, and the tab showing in each cell.
464    pub(crate) fn showing(&self) -> Vec<u32> {
465        self.holders
466            .iter()
467            .flat_map(|holder| match &holder.holds {
468                Holds::One(placed) => vec![placed.id],
469                Holds::Custom(custom) => custom
470                    .cells
471                    .iter()
472                    .filter_map(|cell| cell.shown().map(|placed| placed.id))
473                    .collect(),
474            })
475            .collect()
476    }
477
478    /// The cell whose tabs hold widget `placed`, and where in them it is.
479    fn cell_of(&mut self, placed: u32) -> Option<(&mut Cell, usize)> {
480        self.holders
481            .iter_mut()
482            .filter_map(|holder| match &mut holder.holds {
483                Holds::Custom(custom) => Some(custom),
484                Holds::One(_) => None,
485            })
486            .flat_map(|custom| custom.cells.iter_mut())
487            .find_map(|cell| {
488                let at = cell.tabs.iter().position(|tab| tab.id == placed)?;
489                Some((cell, at))
490            })
491    }
492
493    /// The tab after widget `placed` in its stack shown, or the one before
494    /// (`forward` or not), round from the last to the first: which shows
495    /// now. `None` for a widget in no custom window.
496    pub(crate) fn turn_tab(&mut self, placed: u32, forward: bool) -> Option<u32> {
497        let (cell, at) = self.cell_of(placed)?;
498        let tabs = cell.tabs.len();
499        cell.showing = if forward {
500            (at + 1) % tabs
501        } else {
502            (at + tabs - 1) % tabs
503        };
504        cell.shown().map(|shown| shown.id)
505    }
506
507    /// Show the tab that is widget `placed`; whether it is one.
508    pub(crate) fn show_tab(&mut self, placed: u32) -> bool {
509        let Some((cell, at)) = self.cell_of(placed) else {
510            return false;
511        };
512        cell.showing = at;
513        true
514    }
515
516    /// Tab `placed` moved along its strip: one left, one right, or to the
517    /// front. The tab showing stays the one showing. Whether it moved.
518    pub(crate) fn move_tab(&mut self, placed: u32, shift: Shift) -> bool {
519        let Some((cell, at)) = self.cell_of(placed) else {
520            return false;
521        };
522        let to = match shift {
523            Shift::Left => at.checked_sub(1),
524            Shift::Right => (at + 1 < cell.tabs.len()).then_some(at + 1),
525            Shift::Front => (at != 0).then_some(0),
526        };
527        to.is_some_and(|to| cell.move_tab(at, to))
528    }
529
530    /// Which side the strip of the cell holding tab `placed` runs along;
531    /// `None` for a widget in no cell.
532    pub(crate) fn tab_bar(&self, placed: u32) -> Option<Bar> {
533        self.holders
534            .iter()
535            .filter_map(|holder| match &holder.holds {
536                Holds::Custom(custom) => Some(custom),
537                Holds::One(_) => None,
538            })
539            .flat_map(|custom| custom.cells.iter())
540            .find(|cell| cell.tabs.iter().any(|tab| tab.id == placed))
541            .map(|cell| cell.bar)
542    }
543
544    /// The strip of the cell holding tab `placed` along `bar`. Whether
545    /// there was such a cell.
546    pub(crate) fn set_tab_bar(&mut self, placed: u32, bar: Bar) -> bool {
547        match self.cell_of(placed) {
548            Some((cell, _)) => {
549                cell.bar = bar;
550                true
551            }
552            None => false,
553        }
554    }
555
556    /// Every widget placed here, standalone or in a custom window, showing
557    /// or a tab behind another.
558    pub(crate) fn placed(&self) -> Vec<&Placed> {
559        self.holders
560            .iter()
561            .flat_map(|holder| holder.holds.placed())
562            .collect()
563    }
564}

569        match self {
570            Self::One(placed) => Box::new(std::iter::once(placed)),
571            Self::Custom(custom) => Box::new(custom.cells.iter().flat_map(|cell| cell.tabs.iter())),

577        match self {

579            Self::Custom(custom) => Some(custom.title.as_str()),
580        }

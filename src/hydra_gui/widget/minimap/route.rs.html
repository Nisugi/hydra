1//! The route travel shows (`.route2`), drawn over the map in the theme's
2//! route colour: a line from room to room, from where you stand to the
3//! goal, and a ring at the goal. Walking it, by hand or not, what is behind
4//! you is no longer drawn; at the goal, nothing is. Marks run along it the
5//! way it goes, when the minimap's page has them on.
6
7use egui::{FontId, Pos2, Rect, Stroke, Vec2, vec2};
8use hydra_map::{MapScene, ShownRoute};
9
10use crate::theme::{self, T};

12/// Where room `id` is drawn on any of `layers`: its square, or the way-in
13/// dot that stands for it.
14pub(super) fn drawn_at(layers: &[(&MapScene, Vec2)], id: u32) -> Option<Vec2> {
15    layers.iter().find_map(|&(scene, offset)| {
16        if let Some(room) = scene.room(id) {
17            return Some(super::cell(room.cell) + offset);
18        }
19        let door = scene.doors.iter().find(|door| door.inside == id)?;
20        Some(vec2(door.at.0, door.at.1) + offset)
21    })
22}

24/// The room `route` rings from `you`: none once you are there.
25pub(super) fn goal(route: Option<&ShownRoute>, you: u32) -> Option<u32> {
26    match route?.from(you) {
27        [_, .., goal] => Some(*goal),
28        _ => None,
29    }
30}

32/// Pixels between one mark and the next along a route, and pixels a second
33/// they run.
34const FLOW_GAP: f32 = 26.0;
35const FLOW_SPEED: f32 = 34.0;
36
37/// Marks running along the line through `along`, the way it goes: a small
38/// arrowhead every [`FLOW_GAP`] pixels, each moving toward the end. Drawn
39/// only where `rect` shows it, and the map asked to draw again, so they
40/// keep moving.
41pub(super) fn draw_flow(
42    painter: &egui::Painter,
43    rect: Rect,
44    along: &[Pos2],
45    colour: egui::Color32,
46    width: f32,
47) {
48    if along.len() < 2 {
49        return;
50    }
51    #[allow(clippy::cast_possible_truncation)]
52    let now = painter.ctx().input(|i| i.time) as f32;
53    let size = width + 3.5;
54    let stroke = Stroke::new((width * 0.6).max(1.5), colour);
55    for (tip, way) in flow_marks(along, (now * FLOW_SPEED) % FLOW_GAP) {
56        if rect.contains(tip) {
57            let back = tip - way * size;
58            let across = vec2(-way.y, way.x);
59            painter.line_segment([back + across * size, tip], stroke);
60            painter.line_segment([back - across * size, tip], stroke);
61        }
62    }
63    painter.ctx().request_repaint();
64}

66/// Where each mark is on the line through `along`, and the way the line
67/// runs there: one every [`FLOW_GAP`] pixels, the first `lead` pixels from
68/// the start, measured along the whole line and not each stretch of it.
69fn flow_marks(along: &[Pos2], lead: f32) -> Vec<(Pos2, Vec2)> {
70    let mut marks = Vec::new();
71    let mut due = lead;
72    for pair in along.windows(2) {
73        let (from, to) = (pair[0], pair[1]);
74        let length = from.distance(to);
75        if length <= f32::EPSILON {
76            continue;
77        }
78        let way = (to - from) / length;
79        while due <= length {
80            marks.push((from + way * due, way));
81            due += FLOW_GAP;
82        }
83        due -= length;
84    }
85    marks
86}

88/// `route` from `you` on, over `layers`, and how many steps are left in a
89/// corner of `rect`.
90pub(super) fn draw_route(
91    painter: &egui::Painter,
92    rect: Rect,
93    layers: &[(&MapScene, Vec2)],
94    route: &ShownRoute,
95    you: u32,
96    to_screen: &dyn Fn(Vec2) -> Pos2,
97    (zoom, flow): (f32, bool),
98) {
99    let rooms = route.from(you);
100    if rooms.len() < 2 {
101        return;
102    }
103    let colour = theme::color(painter.ctx(), T::MapRoute);
104    let width = (zoom * 0.35).clamp(2.0, 4.0);
105    for pair in rooms.windows(2) {
106        let (Some(a), Some(b)) = (drawn_at(layers, pair[0]), drawn_at(layers, pair[1])) else {
107            continue;
108        };
109        painter.line_segment([to_screen(a), to_screen(b)], Stroke::new(width, colour));
110    }
111    if flow {
112        // Only along the line: where a room of the route is not drawn the
113        // line stops, and the marks stop with it rather than cut across.
114        let mut along: Vec<Pos2> = Vec::new();
115        for room in rooms {
116            if let Some(at) = drawn_at(layers, *room) {
117                along.push(to_screen(at));
118            } else {
119                draw_flow(painter, rect, &along, colour, width);
120                along.clear();
121            }
122        }
123        draw_flow(painter, rect, &along, colour, width);
124    }
125    let goal = rooms.last().copied();
126    if let Some(at) = goal.and_then(|goal| drawn_at(layers, goal)) {
127        let square = (zoom * 0.7).clamp(3.0, 10.0);
128        painter.circle_stroke(to_screen(at), square * 0.9 + 4.0, Stroke::new(2.5, colour));
129    }
130    let steps = rooms.len() - 1;
131    let said = format!(
132        "To {}: {steps} step{}",
133        goal.unwrap_or_default(),
134        if steps == 1 { "" } else { "s" }
135    );
136    // On the map's ground, so the rooms under it do not show through.
137    let words = painter.layout_no_wrap(said, FontId::proportional(11.0), colour);
138    let at = rect.left_top() + vec2(4.0, 4.0);
139    let backing = Rect::from_min_size(at, words.size() + vec2(8.0, 4.0));
140    let ground = theme::color(painter.ctx(), T::MapBackground);
141    painter.rect_filled(backing, theme::corner(painter.ctx()), ground);
142    painter.galley(at + vec2(4.0, 2.0), words, colour);
143}

145#[cfg(test)]
146mod tests {
147    use egui::pos2;
148
149    use super::*;
150
151    /// Along a line that turns a corner, the marks keep their spacing round
152    /// it and each points the way its stretch runs.
153    #[test]
154    fn marks_keep_their_spacing_round_a_corner() {
155        let along = [pos2(0.0, 0.0), pos2(40.0, 0.0), pos2(40.0, 40.0)];
156        let marks = flow_marks(&along, 10.0);
157        let tips: Vec<Pos2> = marks.iter().map(|(tip, _)| *tip).collect();
158        // 10 and 36 along the first stretch; 62 is 22 down the second.
159        assert_eq!(tips, [pos2(10.0, 0.0), pos2(36.0, 0.0), pos2(40.0, 22.0)]);
160        assert_eq!(marks[0].1, vec2(1.0, 0.0));
161        assert_eq!(marks[2].1, vec2(0.0, 1.0));
162        // A moment later every mark is further on, and none is lost.
163        let later = flow_marks(&along, 15.0);
164        assert_eq!(later.len(), 3);
165        assert_eq!(later[0].0, pos2(15.0, 0.0));
166        assert!(flow_marks(&along[..1], 0.0).is_empty());
167    }

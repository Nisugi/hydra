10/// A direction on the compass: its word, what the rose calls it, and where
11/// it sits, in cells from the top left of a four-by-three grid.
12const DIRECTIONS: [(&str, &str, (u8, u8)); 11] = [
13    ("northwest", "NW", (0, 0)),
14    ("north", "N", (1, 0)),
15    ("northeast", "NE", (2, 0)),
16    ("west", "W", (0, 1)),
17    ("out", "OUT", (1, 1)),
18    ("east", "E", (2, 1)),
19    ("southwest", "SW", (0, 2)),
20    ("south", "S", (1, 2)),
21    ("southeast", "SE", (2, 2)),
22    ("up", "UP", (3, 0)),
23    ("down", "DN", (3, 2)),
24];

26/// The room's ways out as a compass rose, each lit when the room has it.
27/// A click on a lit one goes that way, when `goes` -- never for a compass
28/// following another character. The direction clicked, if one was.
29pub(super) fn compass(ui: &mut egui::Ui, state: Option<&GameState>, goes: bool) -> Option<String> {
30    let exits = state.and_then(|state| state.room.exits.as_ref());
31    let size = ui.available_size();
32    let cell = (size.x / 4.0).min(size.y / 3.0).max(12.0);
33    let (area, response) = ui.allocate_exact_size(vec2(cell * 4.0, cell * 3.0), Sense::hover());
34    if exits.is_none() {
35        response.widget_info(|| {
36            egui::WidgetInfo::labeled(egui::WidgetType::Label, true, "Exits unknown")
37        });
38    }
39    let mut went = None;
40    for (word, short, (column, row)) in DIRECTIONS {
41        let rect = Rect::from_min_size(
42            pos2(
43                area.min.x + f32::from(column) * cell,
44                area.min.y + f32::from(row) * cell,
45            ),
46            Vec2::splat(cell),
47        )
48        .shrink(2.0);
49        // The game names them short (`n`, `ne`): each said in full first.
50        let open =
51            exits.is_some_and(|exits| exits.iter().any(|exit| hydra_ui::exit_word(exit) == word));
52        let sense = if open && goes {
53            Sense::click()
54        } else {
55            Sense::hover()
56        };
57        let direction = ui.interact(rect, response.id.with(word), sense);
58        direction.widget_info(|| egui::WidgetInfo::labeled(egui::WidgetType::Button, open, word));
59        let painter = ui.painter();
60        let text = if open {
61            let accent = theme::color(ui.ctx(), T::Accent);
62            painter.rect_filled(rect, theme::corner(ui.ctx()), accent);
63            readable_on(accent)
64        } else {
65            painter.rect_stroke(
66                rect,
67                theme::corner(ui.ctx()),
68                Stroke::new(1.0, ui.visuals().weak_text_color()),
69                StrokeKind::Inside,
70            );
71            ui.visuals().weak_text_color()
72        };
73        painter.text(
74            rect.center(),
75            Align2::CENTER_CENTER,
76            short,
77            FontId::proportional((cell * 0.32).clamp(8.0, 14.0)),
78            text,
79        );
80        if direction.clicked() {
81            went = Some(word.to_owned());
82        }
83    }
84    went
85}

87/// One creature as the combat list shows it: read off the model's creature,
88/// so the list is drawn from what it says and nothing else.
89#[derive(Clone, Debug, PartialEq)]
90pub(super) struct Fighter {
91    /// Its name.
92    pub(super) name: String,
93    /// Its id, which its link is to.
94    pub(super) id: i64,
95    /// Its noun, when the game named one.
96    pub(super) noun: Option<String>,
97    /// Whether it is on the character's side.
98    pub(super) friend: bool,
99    /// Whether it is dead.
100    pub(super) dead: bool,
101    /// Its statuses, in words.
102    pub(super) statuses: Vec<String>,
103    /// Its health, 0-100, when the server states it.
104    pub(super) health: Option<u32>,
105}
106
107impl Fighter {
108    /// `creature` as the list shows it at `now`. A creature the game has
109    /// not said is friendly is a foe.
110    fn of(creature: &CreatureInstance, now: Option<u32>) -> Self {
111        let health = creature.hp_is_stated().then(|| {
112            // Rounded and clamped to 0-100 first, so the cast cannot truncate.
113            #[allow(
114                clippy::cast_possible_truncation,
115                clippy::cast_sign_loss,
116                reason = "rounded and clamped to 0-100 in the same expression"
117            )]
118            let percent = creature.hp_percent().clamp(0.0, 100.0).round() as u32;
119            percent
120        });
121        Self {
122            name: creature.name.clone(),
123            id: creature.id,
124            noun: creature.noun.clone(),
125            friend: creature.hostile() == Some(false),
126            dead: creature.dead(),
127            statuses: creature
128                .statuses(now)
129                .into_iter()
130                .map(|status| status.as_str().replace('_', " "))
131                .collect(),
132            health,
133        }
134    }
135}

137/// Who is fighting here: the character and its friends, then its foes;
138/// each a link to it, live when `own`, the window's own character's.
139pub(super) fn combat(
140    ui: &mut egui::Ui,
141    state: Option<&GameState>,
142    own: bool,
143) -> Option<super::Clicked> {
144    let Some(state) = state else {
145        ui.weak("Combat unknown");
146        return None;
147    };
148    let now = state.game_time_now();
149    let fighters: Vec<Fighter> = state
150        .creatures()
151        .in_room()
152        .map(|creature| Fighter::of(creature, now))
153        .collect();
154    fighting(ui, state.character.stance.as_deref(), &fighters, own)
155}

157/// The combat list: *FRIENDLY*, the character with its `stance` and its
158/// friends; *FOES*, the rest, or that there are none. Each a link to it,
159/// live when `own`.
160pub(super) fn fighting(
161    ui: &mut egui::Ui,
162    stance: Option<&str>,
163    fighters: &[Fighter],
164    own: bool,
165) -> Option<super::Clicked> {
166    ui.label(egui::RichText::new("FRIENDLY").small().weak());
167    let player = theme::color(ui.ctx(), T::Player);
168    ui.horizontal(|ui| {
169        ui.colored_label(player, "You");
170        if let Some(stance) = stance {
171            ui.weak(stance);
172        }
173    });
174    let mut clicked = None;
175    for friend in fighters.iter().filter(|fighter| fighter.friend) {
176        clicked = clicked.or(fighter(ui, friend, player, own));
177    }
178    ui.label(egui::RichText::new("FOES").small().weak());
179    let mut foes = fighters.iter().filter(|fighter| !fighter.friend).peekable();
180    if foes.peek().is_none() {
181        ui.weak("No foes.");
182    }
183    for foe in foes {
184        clicked = clicked.or(fighter(ui, foe, theme::color(ui.ctx(), T::Creature), own));
185    }
186    clicked
187}

189/// One fighter: its name, a link to it, live when `own`; whether it is
190/// dead, its statuses, and its health as a thin bar when stated.
191fn fighter(
192    ui: &mut egui::Ui,
193    fighter: &Fighter,
194    color: Color32,
195    own: bool,
196) -> Option<super::Clicked> {
197    let clicked = ui.horizontal_wrapped(|ui| {
198        let id = fighter.id.to_string();
199        let name = super::described::object_run(
200            &fighter.name,
201            (&id, fighter.noun.as_deref().unwrap_or_default()),
202            color,
203        );
204        let clicked = super::described::say(ui, &[name], own);
205        if fighter.dead {
206            ui.weak("dead");
207        }
208        if !fighter.statuses.is_empty() {
209            ui.weak(fighter.statuses.join(", "));
210        }
211        clicked
212    });
213    if let Some(percent) = fighter.health {
214        let bar = Bar::new(
215            &fighter.name,
216            Some(Amount {
217                percent,
218                current: None,
219                max: None,
220            }),
221        )
222        .fill(theme::color(ui.ctx(), T::Health))
223        .size([ui.available_width(), 6.0])
224        .says(Says {
225            label: false,
226            numbers: false,
227            percent: false,
228            words: false,
229        });
230        ui.add(bar);
231    }
232    clicked.inner
233}

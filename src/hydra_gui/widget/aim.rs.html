1//! The Aim widget's own: how it looks, and the aim remembered between
2//! runs.
3//!
4//! The game says where the character aims only when the aim changes, so a
5//! Hydra just started knows nothing of it. The last aim the game stated is
6//! kept beside the character's layout and shown until the game speaks
7//! again (`GameState::aim_said`); what it then says replaces it.
8
9use std::path::{Path, PathBuf};
10
11use hydra_ui::settings::{Row, RowKind, Value};
12use serde::{Deserialize, Serialize};
13
14/// What the player chose for one Aim widget on its own page.
15#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
16pub(crate) struct AimLook {
17    /// `Aim:` before the part, as the widget first draws it; off, the part
18    /// alone.
19    #[serde(default = "yes", skip_serializing_if = "std::ops::Not::not")]
20    pub(crate) label: bool,
21}
22
23const fn yes() -> bool {
24    true
25}
26
27impl Default for AimLook {
28    fn default() -> Self {
29        Self { label: true }
30    }
31}
32
33/// The key of the Aim widget's label row.
34const LABEL: &str = "label";
35
36/// The Aim widget's rows: whether its label shows.
37pub(crate) fn rows(look: Option<&AimLook>) -> Vec<Row> {
38    let label = look.is_none_or(|look| look.label);
39    vec![Row {
40        key: LABEL.to_owned(),
41        label: "Show the Aim label".to_owned(),
42        help: "`Aim: right eye`, or `right eye` alone.".to_owned(),
43        kind: RowKind::Toggle,
44        value: Value::On(label),
45        here: !label,
46        from: None,
47    }]
48}
49
50/// Set the Aim widget's `key` to `to`, or back to its own.
51pub(crate) fn set(look: &mut AimLook, key: &str, to: Option<&str>) -> Result<(), String> {
52    match key {
53        LABEL => {
54            look.label = match to.map(str::trim) {
55                None | Some("on") => true,
56                Some("off") => false,
57                Some(other) => return Err(format!("`{other}` is not on or off.")),
58            };
59            Ok(())
60        }
61        other => Err(format!("Aim has no setting {other}.")),
62    }
63}

65/// What the Aim widget draws: its look, and the aim to show while the
66/// game has not said one this run.
67#[derive(Clone, Debug, Default, PartialEq, Eq)]
68pub(crate) struct AimShown {
69    pub(crate) look: AimLook,
70    /// The aim remembered from the last run; `None` when none was, or the
71    /// character then aimed at nothing.
72    pub(crate) remembered: Option<String>,
73}
74
75/// The aim kept between runs, in the character's own file beside its
76/// layout.
77#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
78pub(crate) struct Remembered {
79    /// Where the character aimed when the game last said: `right eye`;
80    /// `None` for nothing in particular.
81    #[serde(default, skip_serializing_if = "Option::is_none")]
82    pub(crate) aim: Option<String>,
83}
84
85impl Remembered {
86    /// Read from `dir`, as `character` on `instance` keeps it; none with
87    /// no file, or one that does not read.
88    pub(crate) fn load(dir: &Path, instance: Option<&str>, character: &str) -> Self {
89        std::fs::read_to_string(file(dir, instance, character))
90            .ok()
91            .and_then(|text| serde_json::from_str(&text).ok())
92            .unwrap_or_default()
93    }
94
95    /// Write to `dir`, with the next batch of saves.
96    pub(crate) fn save(&self, dir: &Path, instance: Option<&str>, character: &str) {
97        let Ok(text) = serde_json::to_string_pretty(self) else {
98            return;
99        };
100        crate::store::whole(dir, &file(dir, instance, character), text.into_bytes());
101    }
102
103    /// The game has said the aim is `aim`: whether that is news, which
104    /// the caller then keeps.
105    pub(crate) fn heard(&mut self, aim: Option<&str>) -> bool {
106        if self.aim.as_deref() == aim {
107            return false;
108        }
109        self.aim = aim.map(str::to_owned);
110        true
111    }
112}
113
114/// The character's file, beside its layout.
115fn file(dir: &Path, instance: Option<&str>, character: &str) -> PathBuf {
116    crate::layout::character_file(dir, instance, character).with_extension("aim.json")
117}

119#[cfg(test)]
120mod tests {
121    use super::*;
122
123    /// The aim the game last said is written, read back, and replaced
124    /// only by news.
125    #[test]
126    fn the_aim_is_kept_between_runs() {
127        let dir = std::env::temp_dir().join(format!("hydra-aim-{}", std::process::id()));
128        let _ = std::fs::remove_dir_all(&dir);
129        std::fs::create_dir_all(&dir).expect("a folder");
130        let mut kept = Remembered::load(&dir, Some("GS3"), "Ashryn");
131        assert_eq!(kept.aim, None);
132        assert!(kept.heard(Some("right eye")));
133        assert!(!kept.heard(Some("right eye")), "no news");
134        kept.save(&dir, Some("GS3"), "Ashryn");
135        assert_eq!(
136            Remembered::load(&dir, Some("GS3"), "Ashryn").aim.as_deref(),
137            Some("right eye")
138        );
139        assert!(kept.heard(None), "aiming at nothing is news too");
140        kept.save(&dir, Some("GS3"), "Ashryn");
141        assert_eq!(Remembered::load(&dir, Some("GS3"), "Ashryn").aim, None);
142        let _ = std::fs::remove_dir_all(&dir);
143    }
144
145    #[test]
146    fn the_label_row_switches() {
147        let mut look = AimLook::default();
148        assert!(look.label);
149        set(&mut look, "label", Some("off")).expect("off");
150        assert!(!look.label);
151        assert_eq!(rows(Some(&look))[0].value, Value::On(false));
152        set(&mut look, "label", None).expect("back");
153        assert!(look.label);
154        assert!(set(&mut look, "label", Some("maybe")).is_err());
155        assert!(set(&mut look, "colour", Some("red")).is_err());
156    }

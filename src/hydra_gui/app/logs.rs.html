1//! The characters' log windows: opened from a play
2//! window's *Log* or a hub card's, each its own native window, their reads
3//! run on the runtime's blocking tasks, never on this thread -- each after
4//! the writer has put what it buffered on disk, so a read is never short.
5
6use std::sync::Arc;
7
8use hydra_session::player_log::writer;
9
10use super::{App, TITLE};
11use crate::PlayWindow as _;
12use crate::logs::Logs;
13use crate::seat::Seat;
14
15impl App {
16    /// Open `seat`'s log window, making it the first time.
17    pub(super) fn open_log(&mut self, seat: &Seat) {
18        let logs = self.logs.entry(seat.id).or_insert_with(|| {
19            let mut logs = Logs::new(&format!("{}:{}", seat.game, seat.name));
20            logs.place = crate::placement::seat_key("log", seat);
21            logs
22        });
23        logs.open = true;
24    }
25
26    /// Draw each open log window, and run what each asked to read. A window
27    /// whose character left the table goes with it.
28    pub(super) fn log_windows(&mut self, context: &egui::Context, seats: &[Arc<Seat>]) {
29        let placements = &mut self.placements;
30        self.logs.retain(|session, logs| {
31            let stays = seats.iter().any(|seat| seat.id == *session);
32            // Closed with its character, so it opens where it was left.
33            if !stays && !logs.place.is_empty() {
34                placements.closed(&logs.place);
35            }
36            stays
37        });
38        let mut clicked = Vec::new();
39        for seat in seats {
40            let Some(logs) = self.logs.get_mut(&seat.id).filter(|logs| logs.open) else {
41                continue;
42            };
43            logs.dropped = seat.log().map_or(0, hydra_session::PlayerLog::dropped);
44            let viewport = egui::ViewportId::from_hash_of(("log", seat.id));
45            let place = logs.place.clone();
46            let closed = self.placements.show(
47                context,
48                &place,
49                viewport,
50                ([1000.0, 640.0], format!("Log — {} — {TITLE}", seat.name)),
51                |ui| {
52                    egui::CentralPanel::default().show(ui, |ui| logs.show(ui));
53                },
54            );
55            if closed {
56                logs.close();
57            }
58            clicked.extend(
59                logs.take_clicks()
60                    .into_iter()
61                    .map(|link| (Arc::clone(seat), link)),
62            );
63            // Taken after every pass has drawn: what any pass asked is kept
64            // on the window until here.
65            for ask in logs.take_asks() {
66                let (inbox, character) = (logs.inbox(), logs.character.clone());
67                let (context, log) = (context.clone(), seat.log().cloned());
68                self.runtime.spawn(async move {
69                    if let Some(log) = log {
70                        log.flush().await;
71                    }
72                    let read = tokio::task::spawn_blocking(move || {
73                        hydra_mem::part(hydra_mem::Part::WindowThreads, || {
74                            crate::logs::run(&writer::root(), &character, ask)
75                        })
76                    });
77                    if let Ok(reply) = read.await {
78                        crate::logs::lock(&inbox).push(reply);
79                    }
80                    context.request_repaint_of(viewport);
81                });
82            }
83        }
84        for (seat, link) in clicked {
85            self.log_clicked(context, &seat, link);
86        }
87    }
88
89    /// A link clicked in `seat`'s log, done by its play window as a click
90    /// in its story is: a command sent, a page opened, an object's menu
91    /// asked for and shown there, in the middle of the window, opened if it
92    /// was closed.
93    fn log_clicked(&mut self, context: &egui::Context, seat: &Seat, link: hydra_ui::RunLink) {
94        let Some(character) = self.characters.get_mut(&seat.id) else {
95            return;
96        };
97        character.open = true;
98        let at = context.input_for(seat.viewport(), |input| input.content_rect().center());
99        let snapshot = seat.snapshot();
100        let story = seat.story();
101        character
102            .play
103            .log_clicked(context, (link, at), (snapshot.as_deref(), &story));
104    }
105}

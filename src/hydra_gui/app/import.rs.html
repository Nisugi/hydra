1//! The commands a play window does itself, since what they change is the
2//! window's: `.keys import`, a Wrayth key set into this character's keys or
3//! every character's; and `.doll`, its Injuries widgets turned, or a
4//! folder's doll pictures brought into Hydra's `dolls` folder; and
5//! `.controller`, the settings menu opened at its page; `.hotbars`, the
6//! hotbar editor opened, or the hotbars files read again. Answered in
7//! the story, as a command is. And `.hotbar`, a hotbar's button pressed,
8//! which the command line claims, typed or put by a script, and the window
9//! presses on its character's hotbars.

24/// What `.hotbar` says when it is not given a bar and a button.
25const HOTBAR_HELP: &str = "hotbar <bar> <button>: press a hotbar's button, by its id or its                            nickname: it sends what a click would.";
26
27/// Whether `rest`, a line's words after its command symbol, is `.hotbar`.
28fn is_hotbar_word(rest: &str) -> bool {
29    rest.split_whitespace()
30        .next()
31        .is_some_and(|word| word.eq_ignore_ascii_case("hotbar"))
32}
33
34/// What `.doll` says when it is not given a word it knows.
35const DOLL_HELP: &str = "doll face <degrees>: turn this window's Infinite doll (0 faces you, 90 \
36                         turns its front to the right, -90 to the left, 180 away); doll import \
37                         <folder>: the doll pictures in <folder> into Hydra's dolls folder, the \
38                         calibration in a <picture>.toml beside each written into it. A pack's \
39                         dolls need no import. Pick a picture on the Injuries widget's page.";

46    /// `line`, typed on `seat`'s window, done here when it is `.keys` or
47    /// `.doll` with the character's own command symbol: whether it was.
48    pub(super) fn window_command(&mut self, seat: &Seat, line: &str) -> bool {
49        let line = line.trim();
50        let Some(rest) = line.strip_prefix(self.symbol_of(seat)) else {
51            return false;
52        };
53        let (word, rest) = rest
54            .trim_start()
55            .split_once(char::is_whitespace)
56            .unwrap_or((rest.trim_start(), ""));
57        let rest = rest.trim();
58        let said = if word.eq_ignore_ascii_case("keys") {
59            self.keys_command(seat, rest)
60        } else if word.eq_ignore_ascii_case("doll") {
61            self.doll_command(seat, rest)
62        } else if word.eq_ignore_ascii_case("controller") {
63            self.controller_command(seat, rest)
64        } else if word.eq_ignore_ascii_case("hotbars") {
65            self.hotbars_command(rest)
66        } else if word.eq_ignore_ascii_case("savelayout") {
67            self.layout_command(seat, rest, true)
68        } else if word.eq_ignore_ascii_case("loadlayout") {
69            self.layout_command(seat, rest, false)
70        } else if word.eq_ignore_ascii_case("map") {
71            self.map_command(rest)
72        } else {
73            return false;
74        };
75        let kind = if said.len() > 1 {
76            NoticeKind::Warn
77        } else {
78            NoticeKind::Info
79        };
80        let mut story = seat.story();
81        story.hear(&Event::Claimed(line.to_owned()));
82        story.push(hydra_ui::Shown::Said(Notice::prose(kind, said).answering()));
83        true
84    }

86    /// Each `.hotbar` line the characters' command lines claimed, pressed
87    /// on its character's hotbars: what each sends, as a click would; why
88    /// not, said in its story.
89    pub(super) fn hotbar_pressed(&mut self) -> Vec<(Arc<Seat>, Asked)> {
90        let seats: Vec<Arc<Seat>> = self
91            .characters
92            .values()
93            .map(|character| Arc::clone(&character.seat))
94            .collect();
95        let mut asks = Vec::new();
96        for seat in seats {
97            let lines = seat.hotbar_presses();
98            if lines.is_empty() {
99                continue;
100            }
101            let symbol = self.symbol_of(&seat);
102            let Some(character) = self.characters.get_mut(&seat.id) else {
103                continue;
104            };
105            for line in lines {
106                let rest = line
107                    .trim()
108                    .strip_prefix(symbol)
109                    .map(str::trim_start)
110                    .and_then(|rest| rest.get("hotbar".len()..))
111                    .unwrap_or_default();
112                let pressed = {
113                    let story = seat.story();
114                    match (crate::hotbars::named(rest), story.world()) {
115                        (None, _) => Err(HOTBAR_HELP.to_owned()),
116                        (Some(_), None) => {
117                            Err("Not yet: the character is not known yet.".to_owned())
118                        }
119                        (Some(named), Some(world)) => character.play.hotbar_press(named, &world),
120                    }
121                };
122                // A button that presses buttons would press for ever.
123                let pressed = pressed.and_then(|command| {
124                    let own = command
125                        .trim()
126                        .strip_prefix(symbol)
127                        .is_some_and(is_hotbar_word);
128                    if own {
129                        Err("Not sent: a hotbar button never presses another.".to_owned())
130                    } else {
131                        Ok(command)
132                    }
133                });
134                match pressed {
135                    Ok(command) => asks.push((Arc::clone(&seat), Asked::Send(command))),
136                    Err(why) => {
137                        let said = Notice::line(NoticeKind::Warn, why).answering();
138                        seat.story().push(hydra_ui::Shown::Said(said));
139                    }
140                }
141            }
142        }
143        asks
144    }

146    /// `.controller`: the settings menu opened at the Controller page, the
147    /// character's own, or with `global` every character's; `.controller
148    /// reload` reads the controller files again.
149    fn controller_command(&mut self, seat: &Seat, rest: &str) -> Vec<String> {
150        if rest.eq_ignore_ascii_case("reload") {
151            self.pad.read(&self.data);
152            let mut said = vec!["The controller files are read again.".to_owned()];
153            said.extend(self.pad.said.iter().cloned());
154            return said;
155        }
156        let whose =
157            (!rest.eq_ignore_ascii_case("global")).then(|| format!("{}:{}", seat.game, seat.name));
158        let said = if whose.is_some() {
159            "The Controller page is open, on this character's own."
160        } else {
161            "The Controller page is open, on every character's."
162        };
163        self.settings
164            .open_at(whose, Some(crate::settings_menu::CONTROLLER));
165        vec![said.to_owned()]
166    }

168    /// `.keys`, its words after it `rest`: what was done.
169    fn keys_command(&mut self, seat: &Seat, rest: &str) -> Vec<String> {
170        match rest.strip_prefix("import") {
171            Some(file) if !file.trim().is_empty() => {
172                let file = file.trim();
173                let (every, file) = match file.strip_prefix("global") {
174                    Some(after) if after.starts_with(char::is_whitespace) => (true, after.trim()),
175                    _ => (false, file),
176                };
177                self.import(seat, Path::new(file.trim_matches('"')), every)
178            }
179            _ => vec![USAGE.to_owned()],
180        }
181    }

183    /// `.doll`, its words after it `rest`: what was done.
184    fn doll_command(&mut self, seat: &Seat, rest: &str) -> Vec<String> {
185        let (verb, after) = rest.split_once(char::is_whitespace).unwrap_or((rest, ""));
186        let after = after.trim();
187        match verb.to_ascii_lowercase().as_str() {
188            "face" => {
189                let Some(deg) = after
190                    .trim_end_matches('\u{b0}')
191                    .parse::<f32>()
192                    .ok()
193                    .filter(|deg| deg.is_finite())
194                else {
195                    return vec![format!(
196                        "Doll: `{after}` is not a facing in degrees. {DOLL_HELP}"
197                    )];
198                };
199                #[expect(
200                    clippy::cast_possible_truncation,
201                    reason = "wrapped to -180..180 first"
202                )]
203                let deg = ((deg + 180.0).rem_euclid(360.0) - 180.0).round() as i16;
204                let turned = self
205                    .characters
206                    .get_mut(&seat.id)
207                    .map_or(0, |character| character.play.turn_dolls(deg));
208                vec![match turned {
209                    0 => "Doll: this window has no Injuries widget to turn.".to_owned(),
210                    _ => format!("Doll: turned to {deg}\u{b0}."),
211                }]
212            }
213            "import" if !after.is_empty() => {
214                let folder = Path::new(after.trim_matches('"'));
215                match crate::doll_import::import_dolls(folder, &self.data.join("dolls")) {
216                    Ok(brought) => vec![doll_said(&brought)],
217                    Err(why) => vec![format!("Doll: {why}")],
218                }
219            }
220            _ => vec![DOLL_HELP.to_owned()],
221        }
222    }

224    /// `.savelayout <name>` (`saving`) and `.loadlayout [<name>]`: this
225    /// window's whole layout kept under a name for any character, and one
226    /// so kept taken as this window's; `.loadlayout` alone lists them.
227    fn layout_command(&mut self, seat: &Seat, name: &str, saving: bool) -> Vec<String> {
228        let name = name.trim().trim_matches('"');
229        let Some(character) = self.characters.get_mut(&seat.id) else {
230            return vec!["This character has no window.".to_owned()];
231        };
232        let play = &mut character.play;
233        if saving {
234            if name.is_empty() {
235                return vec![
236                    "savelayout <name>: save this window's layout, every window \
237                             sized and placed as it is, for any character to load."
238                        .to_owned(),
239                ];
240            }
241            return vec![play.save_layout_as(name).unwrap_or_else(|why| why)];
242        }
243        if name.is_empty() {
244            let names = play.named_layouts();
245            return vec![if names.is_empty() {
246                "No layouts are saved by name; savelayout <name> saves this one.".to_owned()
247            } else {
248                format!(
249                    "loadlayout <name> takes one of these as this window's: {}.",
250                    names.join(", ")
251                )
252            }];
253        }
254        vec![play.load_layout_named(name).unwrap_or_else(|why| why)]
255    }

257    /// `.map [<room>|<creature>]`: the Map Explorer opened; with a room,
258    /// shown there; with a creature, its grounds tinted there.
259    fn map_command(&mut self, words: &str) -> Vec<String> {
260        self.explorer.open = true;
261        self.mapped.wanted(&self.data);
262        if words.trim().is_empty() {
263            return vec!["The map is open.".to_owned()];
264        }
265        let Some((map, atlas)) = self.mapped.loaded() else {
266            return vec!["The map is still loading: ask again in a moment.".to_owned()];
267        };
268        vec![
269            self.explorer
270                .asked(map, atlas, words)
271                .unwrap_or_else(|why| why),
272        ]
273    }

275    /// `seat`'s character's Lich symbol: as its session holds it, kept up
276    /// to date by the settings menu; else as its settings file says.
277    pub(super) fn lich_symbol_of(&self, seat: &Seat) -> char {
278        if let Some(symbol) = seat.lich_symbol() {
279            return symbol;
280        }
281        seat.instance
282            .as_deref()
283            .and_then(|instance| {
284                hydra_session::settings_store::load(&self.data, instance, &seat.name).ok()
285            })
286            .and_then(|file| file.section::<claimant::Settings>(claimant::SECTION).ok())
287            .map_or(claimant::DEFAULT_LICH_SYMBOL, |settings| {
288                settings.lich_symbol()
289            })
290    }

293    /// up to date by the settings menu; else as its settings file says.
294    pub(super) fn symbol_of(&self, seat: &Seat) -> char {
295        if let Some(symbol) = seat.symbol() {
296            return symbol;

307    /// The Wrayth key sets in `file` into `seat`'s character's keys, or
308    /// every character's (`every`); what was done, a line at a time.
309    fn import(&mut self, seat: &Seat, file: &Path, every: bool) -> Vec<String> {
310        let text = match std::fs::read_to_string(file) {
311            Ok(text) => text,
312            Err(why) => return vec![format!("Keys: {} could not be read: {why}", file.display())],
313        };
314        let (imported, mut said) = wrayth::read(&text);
315        let (path, whose, whom) = if every {
316            (
317                self.keys_file.clone(),
318                Whose::Every,
319                "every character's keys".to_owned(),
320            )
321        } else {
322            (
323                self.mine_path(seat),
324                Whose::Character,
325                format!("{}'s keys", seat.name),
326            )
327        };
328        // A key already doing what it would -- in that file as it is now,
329        // its own line counted -- is left as it is.
330        let mine = (!every)
331            .then(|| self.mine.get(&path).map(|(mine, _)| mine))
332            .flatten();
333        let (mut changes, mut already, mut moved) = (Vec::new(), 0, Vec::new());
334        for key in &imported {
335            let beneath = self.keys.beneath(whose, &key.chord);
336            if key.set == 0 && self.keys.in_file(whose, mine, &key.chord) == Some(&key.does) {
337                already += 1;
338                continue;
339            }
340            if key.set == 0 && beneath.is_some_and(|beneath| *beneath != key.does) {
341                moved.push(key.chord.written());
342            }
343            changes.push(KeyChange::Bind {
344                key: key.chord.written(),
345                does: key.does.clone(),
346                was: None,
347                place: Place {
348                    set: key.set,
349                    every,
350                },
351            });
352        }
353        let name = super::keyed::name_of(&path);
354        let old = match self.keys_text(&path) {
355            Ok(old) => old,
356            Err(why) => return vec![why],
357        };
358        let (text, made, mut refused) =
359            match write::made_all(&old, (&name, whose), &changes, (&self.keys, "Keys")) {
360                Ok(done) => done,
361                Err(why) => return vec![why],
362            };
363        // The same keys made to the file with the next batch of saves.
364        let keys = self.keys.clone();
365        self.keys_written(
366            &path,
367            text,
368            Box::new(move |now: &str| {
369                write::made_all(now, (&name, whose), &changes, (&keys, "Keys"))
370                    .map(|(text, _, _)| text)
371            }),
372        );
373        let name = file
374            .file_name()
375            .map_or_else(String::new, |name| name.to_string_lossy().into_owned());
376        let mut lines = vec![format!(
377            "Keys: {made} imported from {name} into {whom}; {already} already so."
378        )];
379        if !moved.is_empty() {
380            lines.push(format!(
381                "These now do what Wrayth had them do, not Hydra's: {}. Restore one on the Keys page for Hydra's.",
382                moved.join(", ")
383            ));
384        }
385        said.append(&mut refused);
386        if !said.is_empty() {
387            lines.push("Not imported:".to_owned());
388            lines.append(&mut said);
389        }
390        lines
391    }

394/// What a doll import says it brought.
395fn doll_said(brought: &crate::doll_import::DollsImported) -> String {
396    let mut said = format!(
397        "Doll: from {}, {} picture(s) and {} overlay(s) into Hydra's dolls folder; {} \
398         calibrated from the file beside it. Pick one on the Injuries widget's page.",
399        brought.from.display(),
400        brought.pictures,
401        brought.overlays,
402        brought.calibrated
403    );
404    if !brought.kept.is_empty() {
405        let _ = write!(
406            said,
407            " Already there, so left as they are: {}.",
408            brought.kept.join(", ")
409        );
410    }
411    said

1//! The map, for the minimaps: loaded on a thread of its own the first time
2//! a play window has a minimap, every area then laid out on the spare cores
3//! (`hydra_map::Atlas`); and each character followed across it. The atlas
4//! is the windows': it is closed as they close, and goes.

18/// The Map Explorer's window, while it is open: any area, a room found,
19/// and where every character stands.
20impl super::App {
21    pub(super) fn explorer_window(&mut self, context: &egui::Context, seats: &[Arc<Seat>]) {
22        if !self.explorer.open {
23            return;
24        }
25        let standing: Vec<crate::explorer::Standing> = seats
26            .iter()
27            .filter_map(|seat| {
28                Some(crate::explorer::Standing {
29                    session: seat.id,
30                    name: seat.name.clone(),
31                    room: self.mapped.room_of(seat.id)?,
32                    saved: self.saved_places(seat),
33                })
34            })
35            .collect();
36        let loaded = self.mapped.loaded().cloned();
37        let overrides = self.mapped.overrides().clone();
38        let explorer = &mut self.explorer;
39        let mut asked = Vec::new();
40        let mut edits = Vec::new();
41        let closed = self.placements.show(
42            context,
43            EXPLORER,
44            egui::ViewportId::from_hash_of(EXPLORER),
45            ([1100.0, 720.0], format!("Map — {}", super::TITLE)),
46            |ui| {
47                egui::CentralPanel::default().show(ui, |ui| {
48                    let (asks, made) = explorer.show(ui, loaded.as_ref(), &standing, &overrides);
49                    asked.extend(asks);
50                    edits.extend(made);
51                });
52            },
53        );
54        if closed {
55            self.explorer.open = false;
56        }
57        for edit in edits {
58            let said = match edit {
59                crate::explorer::Edit::File { rooms, area } => {
60                    self.mapped.reassign(&self.data, &rooms, area.as_deref())
61                }
62                crate::explorer::Edit::Export => self.mapped.export(&self.data),
63            };
64            self.explorer.said = Some(said.unwrap_or_else(|why| why));
65        }
66        for (session, ask) in asked {
67            if let Some(seat) = seats.iter().find(|seat| seat.id == session) {
68                self.asked(context, seats, seat, ask);
69            }
70        }
71    }
72}

83        if !self.surveyor.open {
84            return;
85        }
86        let data = self.data.clone();
87        let surveyor = &mut self.surveyor;
88        let closed = self.placements.show(
89            context,
90            SURVEYOR,
91            egui::ViewportId::from_hash_of(SURVEYOR),
92            ([900.0, 640.0], format!("Survey — {}", super::TITLE)),
93            |ui| {
94                egui::CentralPanel::default().show(ui, |ui| surveyor.show(ui, &data));
95            },
96        );
97        if closed {
98            self.surveyor.open = false;
99        }
100    }

104    /// The places `seat`'s character saved (`.go2 save`), everyone's and
105    /// its own, as its travel file last read; read again once a second at
106    /// most, through the disk reader.
107    fn saved_places(&self, seat: &Seat) -> Vec<(String, Vec<u32>)> {
108        let Some(instance) = seat.instance.as_deref() else {
109            return Vec::new();
110        };
111        let (dir, instance, name) = (self.data.clone(), instance.to_owned(), seat.name.clone());
112        let key = format!("travel-targets:{instance}:{}", name.to_lowercase());
113        let file = hydra_travel::file::travel_path(&dir);
114        crate::disk::Disk::hydras()
115            .kept(&key, &file, None, move |_| {
116                hydra_travel::file::load(&dir, &instance, &name)
117                    .ok()
118                    .map(|notes| notes.targets.into_iter().collect::<Vec<_>>())
119            })
120            .map(|(_, kept)| kept)
121            .unwrap_or_default()
122    }
123}

125/// The map, once asked for, and each character's follower.
126#[derive(Default)]
127pub(super) struct Mapped {
128    /// Set by the loading thread; `None` until the map is asked for.
129    ready: Option<Arc<OnceLock<Ready>>>,
130    /// Each character followed, by session.
131    followers: HashMap<u32, Follower>,
132    /// The player's corrections to the map, laid over it as it loads.
133    overrides: Overrides,
134    /// The corrections were read from the data folder.
135    read_overrides: bool,
136}
137
138/// The atlas `ready` holds, once it does, closed: its workers stop, and it
139/// goes with the last of its owners. Every area of the map laid out would
140/// otherwise stay for the life of the process, held by its workers.
141fn close(ready: &OnceLock<Ready>) {
142    if let Some(Ok((_, atlas))) = ready.get() {
143        atlas.close();
144    }
145}
146
147/// The windows close, and their map's atlas with them.
148impl Drop for Mapped {
149    fn drop(&mut self) {
150        if let Some(ready) = &self.ready {
151            close(ready);
152        }
153    }
154}

156impl std::fmt::Debug for Mapped {
157    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
158        f.debug_struct("Mapped")
159            .field("asked", &self.ready.is_some())
160            .finish_non_exhaustive()
161    }

164impl Mapped {
165    /// Load the map on a thread of its own, its areas' layouts kept in the
166    /// data folder `data`. Once.
167    fn start(&mut self, data: &Path) {
168        if self.ready.is_some() {
169            return;
170        }
171        if !self.read_overrides {
172            self.overrides = Overrides::load(data);
173            self.read_overrides = true;
174        }
175        let ready = Arc::new(OnceLock::new());
176        let (filled, data): (_, PathBuf) = (Arc::clone(&ready), data.to_owned());
177        let overrides = self.overrides.clone();
178        let started = std::thread::Builder::new()
179            .name("map".to_owned())
180            .spawn(move || {
181                let loaded = hydra_mem::part(hydra_mem::Part::Map, || {
182                    hydra_map::shared().clone().map(|loaded| {
183                        // The corrections laid over the map before it is laid
184                        // out; the atlas's cache is keyed by each area's rooms.
185                        let map = Arc::new(overrides.applied(&loaded.map));
186                        let atlas = Atlas::start(&map, &loaded.hash, &data);
187                        (map, atlas)
188                    })
189                });
190                let _ = filled.set(loaded);
191                // The windows closed while it loaded: nothing will draw from
192                // it, and nothing else would close it.
193                if Arc::strong_count(&filled) == 1 {
194                    close(&filled);
195                }
196            });
197        if let Err(why) = started {
198            let _ = ready.set(Err(format!("the map could not be loaded: {why}")));
199        }
200        self.ready = Some(ready);
201    }
202
203    /// Load the map if it is not yet: the Explorer opened before any
204    /// minimap asked.
205    pub(super) fn wanted(&mut self, data: &Path) {
206        self.start(data);
207    }
208
209    /// The player's corrections, as applied.
210    pub(super) fn overrides(&self) -> &Overrides {
211        &self.overrides
212    }
213
214    /// `rooms` filed in `area`, or back where the map files them (`None`):
215    /// kept in the data folder, and the map loaded again with the
216    /// corrections laid over it. What was done, or why not.
217    pub(super) fn reassign(
218        &mut self,
219        data: &Path,
220        rooms: &[u32],
221        area: Option<&str>,
222    ) -> Result<String, String> {
223        for room in rooms {
224            match area {
225                Some(area) => {
226                    self.overrides.areas.insert(*room, area.to_owned());
227                }
228                None => {
229                    self.overrides.areas.remove(room);
230                }
231            }
232        }
233        self.overrides.save(data)?;
234        if let Some(ready) = self.ready.take() {
235            close(&ready);
236        }
237        self.start(data);
238        Ok(match area {
239            Some(area) => format!(
240                "{} room{} filed in {}; the map is laid out again.",
241                rooms.len(),
242                if rooms.len() == 1 { "" } else { "s" },
243                area.replace('-', " ")
244            ),
245            None => format!(
246                "{} room{} filed as the map has {}; the map is laid out again.",
247                rooms.len(),
248                if rooms.len() == 1 { "" } else { "s" },
249                if rooms.len() == 1 { "it" } else { "them" }
250            ),
251        })
252    }
253
254    /// The corrections written for the mapper. Where.
255    pub(super) fn export(&self, data: &Path) -> Result<String, String> {
256        let (map, _) = self
257            .loaded()
258            .ok_or_else(|| "The map is not loaded.".to_owned())?;
259        self.overrides
260            .export(map, data)
261            .map(|path| format!("Written to {}.", path.display()))
262    }
263
264    /// The map and its atlas, once loaded.
265    pub(super) fn loaded(&self) -> Option<&(Arc<Map>, Arc<Atlas>)> {
266        self.ready
267            .as_ref()
268            .and_then(|ready| ready.get())
269            .and_then(|ready| ready.as_ref().ok())
270    }

272    /// Where `session`'s character was last found, by the map's number.
273    pub(super) fn room_of(&self, session: u32) -> Option<u32> {
274        self.followers.get(&session).and_then(Follower::room)
275    }
276
277    /// The ways out of the room `session`'s character was last found in that
278    /// are not the compass's, as the controller's portals wheel's slices: a
279    /// crossing Hydra walks itself is its `go2` there. None before the map
280    /// is loaded.
281    pub(super) fn portals(&self, session: u32) -> Vec<crate::pad::Slice> {
282        let (Some(Some(Ok((map, _)))), Some(follower)) = (
283            self.ready.as_ref().map(|ready| ready.get()),
284            self.followers.get(&session),
285        ) else {
286            return Vec::new();
287        };
288        follower
289            .ways_out(map)
290            .into_iter()
291            .map(|way| {
292                let (command, hydra) = match way.command {
293                    Some(command) => (command, false),
294                    None => (format!("go2 {}", way.to), true),
295                };
296                crate::pad::Slice {
297                    label: way.label,
298                    command,
299                    hydra,
300                    ..crate::pad::Slice::default()
301                }
302            })
303            .collect()
304    }
305
306    /// Where each of `seen` (a session, and its last snapshot) is on the
307    /// map, the map loaded from `data` the first time `wanted`; nothing
308    /// before.
309    pub(super) fn views(
310        &mut self,
311        data: &Path,
312        wanted: bool,
313        seen: &[(u32, Option<Arc<Snapshot>>)],
314    ) -> BTreeMap<u32, MinimapView> {
315        if wanted {
316            self.start(data);
317        }
318        let Some(ready) = &self.ready else {
319            return BTreeMap::new();
320        };
321        self.followers
322            .retain(|id, _| seen.iter().any(|(session, _)| session == id));
323        seen.iter()
324            .map(|(id, snapshot)| {
325                let view = match (ready.get(), snapshot) {
326                    (None, _) => MinimapView::Waiting("Loading the map.".to_owned()),
327                    (Some(Err(why)), _) => MinimapView::Waiting(format!("No map: {why}.")),
328                    (Some(Ok(_)), None) => {
329                        MinimapView::Waiting("Where you are is not known yet.".to_owned())
330                    }
331                    (Some(Ok((map, atlas))), Some(snapshot)) => self
332                        .followers
333                        .entry(*id)
334                        .or_default()
335                        .view(map, atlas, &snapshot.state),
336                };
337                (*id, view)
338            })
339            .collect()
340    }

342    /// Room `id` of the map in words, as the story says it; `None` before
343    /// the map is loaded, or when it has no such room.
344    pub(super) fn described(&self, id: u32, brief: bool) -> Option<Vec<String>> {
345        let (map, _) = self.ready.as_ref()?.get()?.as_ref().ok()?;
346        hydra_map::described(map, id, brief)
347    }

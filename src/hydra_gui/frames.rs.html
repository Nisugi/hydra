1//! A frame drawn round a window or a card: the recipe the theme names,
2//! read from the images folder, and its pieces painted.
3//!
4//! The recipe and every picture come through the disk cache, so a recipe
5//! edited or a part replaced is drawn afresh within a second. Each piece is
6//! a textured quad: the edge in tiles from a run's middle outward, the rest
7//! one quad each, every one turned by permuting its picture's corners. A
8//! glow is a blurred copy of the part's own shape, made once per part and
9//! radius and drawn tinted behind it; one that pulses asks for a repaint a
10//! few times a second while it is shown.
11

27/// Keep the data folder for every window to find frames in, and see that
28/// the plain frame Hydra ships is in its frames folder.
29pub(crate) fn keep_frames(ctx: &egui::Context, data: &Path) {
30    let known = ctx.data(|data| data.get_temp::<PathBuf>(key()));
31    if known.as_deref() != Some(data) {
32        seed(&frames_in(data));
33        ctx.data_mut(|data_| data_.insert_temp(key(), data.to_path_buf()));
34    }
35}

37/// The data folder kept: the pool every pack of art goes into.
38pub(crate) fn pool(ctx: &egui::Context) -> Option<PathBuf> {
39    ctx.data(|data| data.get_temp::<PathBuf>(key()))
40}

55/// The plain frame Hydra ships: a flat grey edge and a grey disc on every
56/// junction, for a theme to tint. Written into `frames` where a file of it
57/// is missing, never over one the player changed.
58fn seed(frames: &Path) {
59    let parts = frames.join(hydra_ui::frame::PARTS);
60    let files: [(PathBuf, &[u8]); 3] = [
61        (
62            parts.join("edges").join("plain.png"),
63            include_bytes!("../frames/plain_edge.png"),
64        ),
65        (
66            parts.join("stickers").join("dot.png"),
67            include_bytes!("../frames/plain_dot.png"),
68        ),
69        (frames.join("plain.toml"), PLAIN_RECIPE.as_bytes()),
70    ];
71    for (path, bytes) in files {
72        if path.is_file() {
73            continue;
74        }
75        if let Some(folder) = path.parent() {
76            let _ = std::fs::create_dir_all(folder);
77        }
78        let _ = std::fs::write(&path, bytes);
79    }
80}

96/// The frame named `name`: the player's own recipe of that name, else a
97/// pack's; none while it is being read, or when it does not read.
98pub(crate) fn recipe(ctx: &egui::Context, name: &str) -> Option<Arc<Frame>> {
99    let frames = folder(ctx)?;
100    let path = listed(ctx)
101        .into_iter()
102        .find(|recipe| recipe.name == name)
103        .map(|recipe| recipe.path)?;
104    let key = path.to_string_lossy().into_owned();
105    crate::pictures::kept_watching(ctx, "frame", &key, &key, move |path| {
106        Frame::read_in(Path::new(path), &frames).ok().map(Arc::new)
107    })
108}

121/// Every recipe in the frames folder, the player's own first: read again
122/// when the folder or its packs folder changes, and when a pack lands.
123pub(crate) fn listed(ctx: &egui::Context) -> Vec<Listed> {
124    let Some(frames) = folder(ctx) else {
125        return Vec::new();
126    };
127    let watched = frames.to_string_lossy().into_owned();
128    let key = format!("{watched}#{}", packs::generation());
129    crate::pictures::kept_watching(ctx, "frames", &key, &watched, move |_| {
130        Some(
131            Frame::list(&frames)
132                .into_iter()
133                .map(|frame| Listed {
134                    name: frame.name,
135                    pack: frame.pack,
136                    path: frame.path,
137                })
138                .collect::<Vec<Listed>>(),
139        )
140    })
141    .unwrap_or_default()
142}

172/// Paint `frame` round `cells`, touching rectangles, with `painter`.
173#[allow(
174    clippy::cast_precision_loss,
175    clippy::cast_possible_truncation,
176    clippy::cast_sign_loss
177)]
178pub(crate) fn paint(ctx: &egui::Context, painter: &egui::Painter, frame: &Frame, cells: &[Rect]) {
179    let cells: Vec<hydra_ui::frame::Rect> = cells.iter().map(|rect| laid(*rect)).collect();
180    let scale = frame.scale();
181    let now = ctx.input(|input| input.time) as f32;
182    let mut pulsing = false;
183    for piece in frame.lay(&cells) {
184        let image = &frame.images[piece.image];
185        let path = image.path.to_string_lossy().into_owned();
186        let Some(texture) = crate::pictures::picture(ctx, &path) else {
187            continue;
188        };
189        let tile = if piece.part == Part::Edge && frame.run == Run::Tile {
190            Some(image.size.width as f32 * scale)
191        } else {
192            None
193        };
194        let wraps = piece.part == Part::Edge;
195        let mut breathe = |pulse: Option<f32>| {
196            pulse.map_or(1.0, |period| {
197                pulsing = true;
198                0.65 + 0.35 * (now / period * std::f32::consts::TAU).sin()
199            })
200        };
201        let px = |points: f32| (points / scale).round().clamp(0.0, 256.0) as u32;
202        // The glow behind the part.
203        if let Some(glow) = piece.look.glow {
204            let radius = px(glow.radius).max(1);
205            let pads = Pads {
206                along: if wraps { 0 } else { radius },
207                above: radius,
208                below: radius,
209            };
210            if let Some(blurred) = blurred(ctx, &path, (radius, pads), Mask::Alpha) {
211                let strength = glow.strength * breathe(glow.pulse);
212                let color = Color32::from_rgb(glow.color[0], glow.color[1], glow.color[2])
213                    .gamma_multiply(strength);
214                let spread = pads.points(scale);
215                painter.add(quads(&piece, blurred.id(), tile, spread, color));
216            }
217        }
218        let color = piece
219            .look
220            .tint
221            .map_or(Color32::WHITE, |[r, g, b]| Color32::from_rgb(r, g, b));
222        painter.add(quads(&piece, texture.id(), tile, Spread::NONE, color));
223        // The light from inside it, added over it.
224        if let Some(hot) = piece.look.emissive {
225            let radius = px(hot.radius).max(1);
226            let pads = Pads {
227                along: if wraps { 0 } else { radius },
228                above: radius,
229                below: radius + px(hot.cast),
230            };
231            let mask = Mask::Hot {
232                color: hot.color,
233                tolerance: hot.tolerance,
234                strength: hot.strength,
235            };
236            if let Some(bloom) = blurred(ctx, &path, (radius, pads), mask) {
237                let level = (255.0 * breathe(hot.pulse)).round() as u8;
238                // No alpha: what the texture holds is added to what is under.
239                let color = Color32::from_rgba_premultiplied(level, level, level, 0);
240                let spread = pads.points(scale);
241                painter.add(quads(&piece, bloom.id(), tile, spread, color));
242            }
243        }
244    }
245    // A breathing glow is drawn again a few times a second, not every
246    // frame, and only while it is shown.
247    if pulsing {
248        ctx.request_repaint_after(Duration::from_millis(50));
249    }
250}

252/// How far a blurred copy reaches past its part, in points: along the run
253/// at each end, above (outward) and below (inward).
254#[derive(Clone, Copy, Debug, PartialEq)]
255struct Spread {
256    along: f32,
257    above: f32,
258    below: f32,
259}
260
261impl Spread {
262    /// The part's own bounds.
263    const NONE: Spread = Spread {
264        along: 0.0,
265        above: 0.0,
266        below: 0.0,
267    };
268}

270/// The same reach in pixels, as a copy is padded.
271#[derive(Clone, Copy, Debug, PartialEq, Eq)]
272struct Pads {
273    along: u32,
274    above: u32,
275    below: u32,
276}
277
278impl Pads {
279    /// In points, at `scale` points per pixel.
280    #[allow(clippy::cast_precision_loss)]
281    fn points(self, scale: f32) -> Spread {
282        Spread {
283            along: self.along as f32 * scale,
284            above: self.above as f32 * scale,
285            below: self.below as f32 * scale,
286        }
287    }
288}

290/// `piece` as textured quads: one per tile of `tile` points along its run
291/// when tiled from the middle outward, else one; each reaching `spread`
292/// past the authored piece, for a glow's or a bloom's reach.
293fn quads(
294    piece: &Piece,
295    texture: TextureId,
296    tile: Option<f32>,
297    spread: Spread,
298    color: Color32,
299) -> Mesh {
300    let Spread {
301        along,
302        above,
303        below,
304    } = spread;
305    let rect = piece.rect;
306    let sideways = piece.turn.sideways();
307    let (length, thick) = if sideways {
308        (rect.h, rect.w)
309    } else {
310        (rect.w, rect.h)
311    };
312    // Authored coordinates (x along the top run, y down it) to the screen.
313    let place = |x: f32, y: f32| -> Pos2 {
314        let (sx, sy) = match piece.turn.quarters {
315            0 => (rect.x + x, rect.y + y),
316            1 => (rect.x + (thick - y), rect.y + x),
317            2 => (rect.x + (length - x), rect.y + (thick - y)),
318            _ => (rect.x + y, rect.y + (length - x)),
319        };
320        let sx = if piece.turn.flipped {
321            2.0 * (rect.x + rect.w / 2.0) - sx
322        } else {
323            sx
324        };
325        Pos2::new(sx, sy)
326    };
327    let mut mesh = Mesh::with_texture(texture);
328    let mut quad = |x0: f32, x1: f32, u0: f32, u1: f32| {
329        let at = u32::try_from(mesh.vertices.len()).unwrap_or(0);
330        for (x, y, u, v) in [
331            (x0, -above, u0, 0.0),
332            (x1, -above, u1, 0.0),
333            (x1, thick + below, u1, 1.0),
334            (x0, thick + below, u0, 1.0),
335        ] {
336            mesh.vertices.push(Vertex {
337                pos: place(x, y),
338                uv: Pos2::new(u, v),
339                color,
340            });
341        }
342        mesh.add_triangle(at, at + 1, at + 2);
343        mesh.add_triangle(at, at + 2, at + 3);
344    };
345    match tile.filter(|tile| *tile > 0.5) {
346        Some(tile) => {
347            for (x0, x1, u0, u1) in tiles(length, tile) {
348                quad(x0, x1, u0, u1);
349            }
350        }
351        None => quad(-along, length + along, 0.0, 1.0),
352    }
353    mesh
354}

356/// A run `length` long cut into tiles `tile` long from its middle outward,
357/// each as its span along the run and the part of the picture it shows.
358fn tiles(length: f32, tile: f32) -> Vec<(f32, f32, f32, f32)> {
359    let middle = length / 2.0;
360    let mut cuts = Vec::new();
361    let first = -((middle / tile) + 0.5).ceil();
362    let mut k = first;
363    loop {
364        let start = middle + (k - 0.5) * tile;
365        let end = start + tile;
366        if start >= length {
367            break;
368        }
369        if end > 0.0 {
370            let (x0, x1) = (start.max(0.0), end.min(length));
371            cuts.push((x0, x1, (x0 - start) / tile, (x1 - start) / tile));
372        }
373        k += 1.0;
374    }
375    cuts
376}

378/// What a blurred copy of a part is made from.
379#[derive(Clone, Copy, Debug, PartialEq)]
380enum Mask {
381    /// Its shape: the alpha, as a white picture to be tinted.
382    Alpha,
383    /// Its hot pixels: those within `tolerance` of `color`, in their own
384    /// colours at `strength`, premultiplied with no alpha so they add.
385    Hot {
386        color: hydra_ui::theme::Rgb,
387        tolerance: f32,
388        strength: f32,
389    },
390}
391
392impl Mask {
393    /// A name for the copy, in the cache's key.
394    fn key(self) -> String {
395        match self {
396            Mask::Alpha => "glow".to_owned(),
397            Mask::Hot {
398                color,
399                tolerance,
400                strength,
401            } => format!(
402                "hot{:02x}{:02x}{:02x}-{tolerance:.2}-{strength:.2}",
403                color[0], color[1], color[2]
404            ),
405        }
406    }
407}

409/// A copy of the picture at `path` blurred by `radius` pixels and padded by
410/// `pads`, made from `mask`: kept, and made again when the picture changes.
411/// Padded along by nothing, the copy is periodic along, an edge's.
412fn blurred(
413    ctx: &egui::Context,
414    path: &str,
415    (radius, pads): (u32, Pads),
416    mask: Mask,
417) -> Option<TextureHandle> {
418    let owned = ctx.clone();
419    let key = format!(
420        "{path}#{}-{radius}-{}-{}-{}",
421        mask.key(),
422        pads.along,
423        pads.above,
424        pads.below
425    );
426    crate::pictures::kept_watching(ctx, "blurred", &key, path, move |path| {
427        read_blurred(&owned, path, (radius, pads), mask)
428    })
429}

431/// [`blurred`]'s reading: three box blurs, near enough a Gaussian, of the
432/// mask's channels.
433#[allow(
434    clippy::cast_precision_loss,
435    clippy::cast_possible_truncation,
436    clippy::cast_sign_loss
437)]
438fn read_blurred(
439    ctx: &egui::Context,
440    path: &str,
441    (radius, pads): (u32, Pads),
442    mask: Mask,
443) -> Option<TextureHandle> {
444    let bytes = std::fs::read(Path::new(path)).ok()?;
445    let image = image::load_from_memory(&bytes).ok()?.to_rgba8();
446    let (wide, high) = (image.width() as usize, image.height() as usize);
447    let wraps = pads.along == 0;
448    let (pad_x, pad_top) = (pads.along as usize, pads.above as usize);
449    let (width, height) = (wide + 2 * pad_x, high + pad_top + pads.below as usize);
450    let mut channels: Vec<Vec<f32>> = match mask {
451        Mask::Alpha => vec![vec![0.0; width * height]],
452        Mask::Hot { .. } => vec![vec![0.0; width * height]; 3],
453    };
454    for (x, y, pixel) in image.enumerate_pixels() {
455        let at = (y as usize + pad_top) * width + x as usize + pad_x;
456        let [red, green, blue, alpha] = pixel.0.map(|v| f32::from(v) / 255.0);
457        match mask {
458            Mask::Alpha => channels[0][at] = alpha,
459            Mask::Hot {
460                color,
461                tolerance,
462                strength,
463            } => {
464                let lit = heat([red, green, blue], color, tolerance) * alpha * strength;
465                channels[0][at] = red * lit;
466                channels[1][at] = green * lit;
467                channels[2][at] = blue * lit;
468            }
469        }
470    }
471    // Three passes of a box read as a Gaussian of about the radius; the box
472    // is sized so the three together reach it.
473    let half = ((radius as f32 / 1.8).round() as usize).max(1);
474    let mut other = vec![0.0f32; width * height];
475    for channel in &mut channels {
476        for _ in 0..3 {
477            box_blur_rows(channel, &mut other, width, height, half, wraps);
478            box_blur_cols(&other, channel, width, height, half);
479        }
480    }
481    let byte = |v: f32| (v.clamp(0.0, 1.0) * 255.0).round() as u8;
482    let mut rgba = Vec::with_capacity(width * height * 4);
483    let pixels = match mask {
484        Mask::Alpha => {
485            for a in &channels[0] {
486                rgba.extend([255, 255, 255, byte(*a)]);
487            }
488            egui::ColorImage::from_rgba_unmultiplied([width, height], &rgba)
489        }
490        Mask::Hot { .. } => {
491            for ((red, green), blue) in channels[0].iter().zip(&channels[1]).zip(&channels[2]) {
492                rgba.extend([byte(*red), byte(*green), byte(*blue), 0]);
493            }
494            egui::ColorImage::from_rgba_premultiplied([width, height], &rgba)
495        }
496    };
497    let name = format!("{path}#{}-{radius}", mask.key());
498    Some(ctx.load_texture(name, pixels, egui::TextureOptions::LINEAR))
499}

501/// How hot a pixel of colour `rgb` (0 to 1 each) is: 1 at `hot`, falling
502/// to 0 at `tolerance` of the way across the colour cube from it.
503fn heat(rgb: [f32; 3], hot: hydra_ui::theme::Rgb, tolerance: f32) -> f32 {
504    let hot = hot.map(|v| f32::from(v) / 255.0);
505    let distance =
506        ((rgb[0] - hot[0]).powi(2) + (rgb[1] - hot[1]).powi(2) + (rgb[2] - hot[2]).powi(2)).sqrt()
507            / 3f32.sqrt();
508    if tolerance > 0.0 {
509        (1.0 - distance / tolerance).max(0.0)
510    } else {
511        f32::from(distance == 0.0)
512    }
513}

515/// Each row of `from` averaged over `half` either side into `to`, the row
516/// taken as periodic when it `wraps`.
517#[allow(
518    clippy::cast_precision_loss,
519    clippy::cast_possible_wrap,
520    clippy::cast_sign_loss
521)]
522fn box_blur_rows(
523    from: &[f32],
524    to: &mut [f32],
525    width: usize,
526    height: usize,
527    half: usize,
528    wraps: bool,
529) {
530    let span = (2 * half + 1) as f32;
531    for y in 0..height {
532        let row = &from[y * width..(y + 1) * width];
533        let out = &mut to[y * width..(y + 1) * width];
534        for (x, out) in out.iter_mut().enumerate() {
535            let mut sum = 0.0;
536            for d in -(half as isize)..=(half as isize) {
537                let at = x as isize + d;
538                let at = if wraps {
539                    at.rem_euclid(width as isize) as usize
540                } else if at < 0 || at >= width as isize {
541                    continue;
542                } else {
543                    at as usize
544                };
545                sum += row[at];
546            }
547            *out = sum / span;
548        }
549    }
550}
551
552/// Each column of `from` averaged over `half` either side into `to`.
553#[allow(clippy::cast_precision_loss)]
554fn box_blur_cols(from: &[f32], to: &mut [f32], width: usize, height: usize, half: usize) {
555    let span = (2 * half + 1) as f32;
556    for x in 0..width {
557        for y in 0..height {
558            let (lo, hi) = (y.saturating_sub(half), (y + half).min(height - 1));
559            let sum: f32 = (lo..=hi).map(|yy| from[yy * width + x]).sum();
560            to[y * width + x] = sum / span;
561        }
562    }
563}

719    #[test]
720    fn an_emission_keeps_the_hot_pixels_in_their_own_colours_and_adds() {
721        let dir = std::env::temp_dir().join(format!("hydra-hot-{}", std::process::id()));
722        let _ = std::fs::create_dir_all(&dir);
723        let path = dir.join("lava.png");
724        let mut image = image::RgbaImage::new(9, 1);
725        image.put_pixel(1, 0, image::Rgba([0xff, 0x6a, 0x20, 255]));
726        image.put_pixel(7, 0, image::Rgba([0x40, 0x40, 0x40, 255]));
727        image.save(&path).unwrap();
728        let at = path.to_string_lossy().into_owned();
729        let mask = Mask::Hot {
730            color: [0xff, 0x6a, 0x20],
731            tolerance: 0.2,
732            strength: 1.0,
733        };
734        let pads = Pads {
735            along: 1,
736            above: 1,
737            below: 1,
738        };
739        let (width, height) = (11usize, 3usize);
740        let (ctx, radius) = (egui::Context::default(), 1);
741        let bloom = read_blurred(&ctx, &at, (radius, pads), mask).unwrap();
742        assert_eq!(bloom.size(), [width, height]);
743        let _ = std::fs::remove_dir_all(&dir);
744
745        let orange = [0xff, 0x6a, 0x20];
746        let of = |rgb: [u8; 3]| rgb.map(|v| f32::from(v) / 255.0);
747        assert!(
748            (heat(of(orange), orange, 0.2) - 1.0).abs() < 1e-6,
749            "the colour itself is fully hot"
750        );
751        assert!(
752            heat(of([0x40, 0x40, 0x40]), orange, 0.2) < 1e-6,
753            "grey is cold"
754        );
755        let near = heat(of([0xf0, 0x70, 0x30]), orange, 0.2);
756        assert!(near > 0.5 && near < 1.0, "a nearby orange is warm: {near}");
757        assert!(
758            (heat(of(orange), orange, 0.0) - 1.0).abs() < 1e-6,
759            "no tolerance: the exact colour alone"
760        );
761        assert!(heat(of([0xfe, 0x6a, 0x20]), orange, 0.0) < 1e-6);
762    }

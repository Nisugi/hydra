1//! Keybinds: a key, and the macro it does on
2//! the character whose play window has the keyboard -- commands sent as if
3//! typed there, text put in the command input, or one of Hydra's actions
4//! (`binding.rs`).
5//!
6//! **Hydra's defaults live in the code** (`defaults.rs`); the keybinds file
7//! holds only what the player added or changed, a key written `""` unbinding
8//! a default.
9//!
10//! **A key is named by its winit code** -- `Numpad8`, `F13`, `KeyA`,
11//! `ArrowUp` -- with `ctrl+`, `shift+`, `alt+` or `cmd+` before it. The
12//! physical key, not the character it types, so a binding means the same key
13//! on every keyboard layout, and a numpad key is never its digit: egui folds
14//! `Numpad8` into `8`, which is why the numpad comes through Hydra's eframe
15//! fork instead (`eframe::Frame::numpad_keys`), with `NumLock` read from
16//! each press.
17//!
18//! The keys come three ways. The numpad's through the fork's numpad channel,
19//! which catches only the numpad keys that are bound, so an unbound one
20//! still types. The keys egui has no name for -- Pause, Scroll Lock, Print
21//! Screen, Caps Lock, the context-menu key -- through the fork's key capture
22//! (`eframe::Frame::set_key_capture`), which catches only those
23//! bound, so an unbound Caps Lock is left to egui. Every other key through
24//! egui, which names F1-F35, the arrows and the rest; a bound key's press is
25//! taken before anything else sees it. A key that types -- a letter, a digit,
26//! a mark, with no Ctrl, Alt or Cmd -- never fires a binding, or nobody could
27//! type it.
28//!
29//! **`NumLock` is nobody's.** On Windows and Linux it is the operating
30//! system's switch for the numpad; on macOS, which has none, winit calls the
31//! Clear key `NumLock`, and Clear is Hydra's switch there, the numpad typing
32//! or sending its keys (`app.rs`, `numpad_mode`).
33

52/// Ctrl, in [`Chord::held`].
53pub(crate) const CTRL: u8 = 1;
54/// Shift.
55pub(crate) const SHIFT: u8 = 2;
56/// Alt, Option on macOS.
57pub(crate) const ALT: u8 = 4;
58/// Cmd, on macOS. Nowhere else: egui-winit sets `mac_cmd` only on macOS,
59/// so a chord holding it is refused off macOS ([`Chord::refused`]) rather
60/// than written and never fired. It is not the Windows key elsewhere.
61pub(crate) const CMD: u8 = 8;

63/// A key with the modifiers held.
64#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
65pub(crate) struct Chord {
66    /// The key, as winit names its code.
67    pub(crate) key: String,
68    /// The modifiers held: [`CTRL`], [`SHIFT`], [`ALT`], [`CMD`].
69    pub(crate) held: u8,
70}
71
72impl Chord {
73    /// `key` held with `modifiers`.
74    pub(crate) fn of(key: &str, modifiers: Modifiers) -> Self {
75        let bit = |on: bool, bit: u8| if on { bit } else { 0 };
76        Self {
77            key: key.to_owned(),
78            held: bit(modifiers.ctrl, CTRL)
79                | bit(modifiers.shift, SHIFT)
80                | bit(modifiers.alt, ALT)
81                | bit(modifiers.mac_cmd, CMD),
82        }
83    }
84
85    /// A chord as written, `ctrl+shift+F1`: modifiers in any order and case,
86    /// the key last, by its winit name or the fork's numpad name (`num_8`).
87    ///
88    /// # Errors
89    ///
90    /// What is wrong with it, in words for the player.
91    pub(crate) fn parse(text: &str) -> Result<Self, String> {
92        let mut chord = Self {
93            key: String::new(),
94            held: 0,
95        };
96        let parts: Vec<&str> = text.split('+').map(str::trim).collect();
97        let Some((key, held)) = parts.split_last() else {
98            return Err(format!("`{text}` names no key."));
99        };
100        for modifier in held {
101            match modifier.to_ascii_lowercase().as_str() {
102                "ctrl" | "control" => chord.held |= CTRL,
103                "shift" => chord.held |= SHIFT,
104                "alt" | "option" => chord.held |= ALT,
105                "cmd" | "command" | "super" | "win" => chord.held |= CMD,
106                other => {
107                    return Err(format!(
108                        "`{other}` in `{text}` is not Ctrl, Shift, Alt or Cmd."
109                    ));
110                }
111            }
112        }
113        chord.key = known(key).map_err(|why| format!("`{text}`: {why}"))?;
114        Ok(chord)
115    }
116
117    /// The chord as the file and the Keys page write it: its modifiers in
118    /// one order, then its key, `Ctrl+Shift+F1`. [`Chord::parse`] reads it
119    /// back.
120    pub(crate) fn written(&self) -> String {
121        let mut written = String::new();
122        for (bit, name) in [(CTRL, "Ctrl"), (SHIFT, "Shift"), (ALT, "Alt"), (CMD, "Cmd")] {
123            if self.held & bit != 0 {
124                written.push_str(name);
125                written.push('+');
126            }
127        }
128        written.push_str(&self.key);
129        written
130    }
131
132    /// Why this chord cannot be bound, in words for the player: it types
133    /// something rather than commanding -- a letter, a digit, a mark or the
134    /// space, with no Ctrl, Alt or Cmd held -- or it is Enter alone, which
135    /// sends what is typed (a key that cannot be bound should not look
136    /// bindable); or it holds Cmd where there is none ([`CMD`]).
137    pub(crate) fn refused(&self) -> Option<String> {
138        let written = self.written();
139        if self.held & CMD != 0 && !cfg!(target_os = "macos") {
140            return Some(format!(
141                "{written}: Cmd is macOS's Command key; Hydra does not see the Windows key."
142            ));
143        }
144        if self.held & (CTRL | ALT | CMD) == 0 && types(&self.key) {
145            return Some(format!(
146                "{written} types: hold Ctrl, Alt or Cmd with it, or it could not be typed."
147            ));
148        }
149        (self.held == 0 && self.key == "Enter")
150            .then(|| "Enter sends what is typed: hold Ctrl, Shift, Alt or Cmd with it.".to_owned())
151    }
152}

154/// The keybinds: Hydra's defaults, and every character's file on top. A
155/// character's own file goes over them in [`Keybinds::of`].
156#[derive(Clone, Debug, PartialEq, Eq)]
157pub(crate) struct Keybinds {
158    /// Hydra's own (`defaults.rs`), all in set 0.
159    defaults: BTreeMap<Chord, Macro>,
160    /// Every character's: the keybinds file.
161    every: KeyFile,
162}
163
164impl Default for Keybinds {
165    /// Hydra's defaults alone: no file.
166    fn default() -> Self {
167        Self {
168            defaults: defaults::defaults(),
169            every: KeyFile::default(),
170        }
171    }
172}

230    /// What `chord` does in set 0 as the file `whose` has it now: its own
231    /// line, binding or unbinding it, or else what is beneath it. A
232    /// character's own file is `mine`.
233    pub(crate) fn in_file<'a>(
234        &'a self,
235        whose: Whose,
236        mine: Option<&'a KeyFile>,
237        chord: &Chord,
238    ) -> Option<&'a Macro> {
239        let own = match whose {
240            Whose::Character => mine.and_then(|mine| mine.sets[0].get(chord)),
241            Whose::Every => self.every.sets[0].get(chord),
242        };
243        match own {
244            Some(made) => made.as_ref(),
245            None => self.beneath(whose, chord),
246        }
247    }
248
249    /// What `chord` does in set 0 beneath a file's own line: under every
250    /// character's, Hydra's default; under a character's, every
251    /// character's, then Hydra's.
252    pub(crate) fn beneath(&self, whose: Whose, chord: &Chord) -> Option<&Macro> {
253        if whose == Whose::Character
254            && let Some(made) = self.every.sets[0].get(chord)
255        {
256            return made.as_ref();
257        }
258        self.defaults.get(chord)
259    }
260
261    /// Every key set `set` binds or unbinds -- in `mine`, a character's own
262    /// file, when the page is a character's, and in every character's --
263    /// with what it does, whose file that came from, and Hydra's default:
264    /// what the Keys page lists. Set 0's hold Hydra's defaults too.
265    pub(crate) fn rows(&self, set: u8, mine: Option<&KeyFile>) -> Vec<KeyRow> {
266        let set = usize::from(set.min(SETS - 1));
267        let layers: Vec<(Whose, &Layer)> = mine
268            .map(|mine| (Whose::Character, &mine.sets[set]))
269            .into_iter()
270            .chain(std::iter::once((Whose::Every, &self.every.sets[set])))
271            .collect();
272        let empty = BTreeMap::new();
273        let defaults = if set == 0 { &self.defaults } else { &empty };
274        let chords: BTreeSet<&Chord> = layers
275            .iter()
276            .flat_map(|(_, layer)| layer.keys())
277            .chain(defaults.keys())
278            .collect();
279        chords
280            .into_iter()
281            .map(|chord| {
282                // What the first layer after `from` binds it to, or Hydra.
283                let after = |from: usize| {
284                    layers[from..]
285                        .iter()
286                        .find_map(|(_, layer)| layer.get(chord).cloned())
287                        .unwrap_or_else(|| defaults.get(chord).cloned())
288                };
289                let (does, from, beneath) = match layers
290                    .iter()
291                    .position(|(_, layer)| layer.contains_key(chord))
292                {
293                    Some(at) => (after(at), Some(layers[at].0), after(at + 1)),
294                    None => (defaults.get(chord).cloned(), None, None),
295                };
296                KeyRow {
297                    key: chord.written(),
298                    does,
299                    default: defaults.get(chord).cloned(),
300                    from,
301                    beneath,
302                }
303            })
304            .collect()
305    }
306}

308/// The keys in effect for one character, or for none: a press looks in the
309/// chosen set, the character's own file and then every character's; then
310/// in set 0 the same way; then at Hydra's defaults. The first that binds or
311/// unbinds the key has it.
312#[derive(Clone, Copy, Debug)]
313pub(crate) struct Keys<'a> {
314    binds: &'a Keybinds,
315    mine: Option<&'a KeyFile>,
316}
317
318impl<'a> Keys<'a> {
319    /// The files' sets a press looks in, in order, before Hydra's defaults.
320    fn layers(self) -> impl Iterator<Item = &'a Layer> {
321        let chosen = self.mine.map_or(0, |mine| mine.chosen.min(SETS - 1));
322        let sets = if chosen == 0 {
323            vec![0]
324        } else {
325            vec![chosen, 0]
326        };
327        sets.into_iter().flat_map(move |set| {
328            let set = usize::from(set);
329            self.mine
330                .map(|mine| &mine.sets[set])
331                .into_iter()
332                .chain(std::iter::once(&self.binds.every.sets[set]))
333        })
334    }
335
336    /// What `chord` does, if it is bound.
337    pub(crate) fn does(self, chord: &Chord) -> Option<&'a Macro> {
338        for layer in self.layers() {
339            if let Some(made) = layer.get(chord) {
340                return made.as_ref();
341            }
342        }
343        self.binds.defaults.get(chord)
344    }
345
346    /// Each key bound, and what it does.
347    fn bound(self) -> impl Iterator<Item = (&'a Chord, &'a Macro)> {
348        let chords: BTreeSet<&'a Chord> = self
349            .layers()
350            .flat_map(BTreeMap::keys)
351            .chain(self.binds.defaults.keys())
352            .collect();
353        chords
354            .into_iter()
355            .filter_map(move |chord| self.does(chord).map(|made| (chord, made)))
356    }

358    /// The fork's names for the numpad keys that are bound, whatever the
359    /// modifiers: what it is told to catch, so an unbound numpad key still
360    /// types (`eframe::Frame::set_numpad_capture_keys`).
361    pub(crate) fn numpad_caught(self) -> HashSet<String> {
362        self.bound()
363            .filter_map(|(chord, _)| NUMPAD.iter().find(|(winit, _)| *winit == chord.key))
364            .map(|(_, fork)| (*fork).to_owned())
365            .collect()
366    }
367
368    /// The keys egui has no name for that are bound, whatever the
369    /// modifiers: what the fork is told to catch (`eframe::Frame::
370    /// set_key_capture`), so one unbound is left to egui.
371    pub(crate) fn key_capture(self) -> HashSet<winit::keyboard::KeyCode> {
372        self.bound()
373            .filter_map(|(chord, _)| CAPTURED.iter().find(|(name, _)| *name == chord.key))
374            .map(|(_, code)| *code)
375            .collect()
376    }

378    /// Take from `input` every key press this binds, and the macros they
379    /// do, in order: taken, so no widget sees a bound key. A key whose macro
380    /// `leaves` is not taken, and does nothing here; what it typed goes too.
381    /// A held key's repeats do a send again ([`repeats`]); any other macro
382    /// once a press, its repeats taken and done nothing with.
383    pub(crate) fn take(
384        self,
385        input: &mut egui::InputState,
386        leaves: impl Fn(&Macro) -> bool,
387    ) -> Vec<Macro> {
388        let (mut lines, mut typed) = (Vec::new(), false);
389        input.events.retain(|event| {
390            let egui::Event::Key {
391                key,
392                physical_key,
393                pressed: true,
394                repeat,
395                modifiers,
396            } = event
397            else {
398                return true;
399            };
400            let Some(name) = winit_name(physical_key.unwrap_or(*key)) else {
401                return true;
402            };
403            match self.does(&Chord::of(&name, *modifiers)) {
404                Some(made) if !leaves(made) => {
405                    if !*repeat || repeats(made) {
406                        lines.push(made.clone());
407                    }
408                    typed |= types(&name);
409                    false
410                }
411                _ => true,
412            }
413        });
414        if typed {
415            input
416                .events
417                .retain(|event| !matches!(event, egui::Event::Text(_)));
418        }
419        lines
420    }
421}

423/// Whether a held key's repeats do `made` again. A send does, as holding a
424/// direction keeps walking; so does an action that steps -- scrolling, the
425/// history, the next window, tab, line found or target -- as a held arrow
426/// keeps moving. A switch does not, so a held one ends where one press
427/// leaves it: a macro set, the Find bar, a drawer, Lock, Arrange, a
428/// character's window. Nor does a fill, which would only write the same
429/// words over.
430pub(crate) fn repeats(made: &Macro) -> bool {
431    use binding::Action;
432    match made {
433        Macro::Send(_) => true,
434        Macro::Fill(_) | Macro::Hotbar { .. } => false,
435        Macro::Act(action) => !matches!(
436            action,
437            Action::Set(_)
438                | Action::Find
439                | Action::DrawerTop
440                | Action::DrawerBottom
441                | Action::DrawerLeft
442                | Action::DrawerRight
443                | Action::Lock
444                | Action::Arrange
445                | Action::Character(_)
446                | Action::TargetNext
447                | Action::TargetPrevious
448                | Action::TargetClear
449        ),
450    }
451}

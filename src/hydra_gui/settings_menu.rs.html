1//! The settings menu, the one every way in opens: the main settings button,
2//! and a widget's or window's right-click menu, which opens the same menu
3//! at that setting's page.
4//!
5//! The menu around the pages: the picker of whose settings, the pages down the side, Hydra's own *Window*
6//! and *Keys* pages, a character's pages from the binary, and the widgets'
7//! own pages of the character's play window. The page drawing is
8//! [`crate::menu::Menu`]'s.
9//!
10//! A character is picked from the roster, running or not. Its pages come
11//! from the binary, which applies a change through the writer that
12//! character's settings file has; the window makes the changes to its own
13//! pages, to the keys and to a widget's page itself.
14

20/// The settings menu's native window: a window of its own, whose drags the
21/// play windows also wait out before they write their layouts.
22pub(crate) fn viewport() -> egui::ViewportId {
23    egui::ViewportId::from_hash_of("settings")
24}

26/// The *Keys* page's id.
27pub(crate) const KEYS: &str = "keys";

29/// The *Controller* page's id.
30pub(crate) const CONTROLLER: &str = crate::pad::PAGE;

32/// What the settings menu asks for.
33#[derive(Clone, Debug, PartialEq)]
34pub(crate) enum MenuAsked {
35    /// Of the binary: a character's pages, or a change to one of them.
36    Binary(HubRequest),
37    /// A change to Hydra's own *Window* page.
38    Own {
39        /// The row's key.
40        key: String,
41        /// Its value as the menu writes it; `None` for its default.
42        to: Option<String>,
43    },
44    /// A change to the keys: every character's (`None`), or a character's
45    /// own, as the roster names it.
46    Key {
47        /// The character.
48        character: Option<String>,
49        /// The change.
50        change: KeyChange,
51    },
52    /// Of the *Controller* page: every character's file (`None`), or a
53    /// character's own, as the roster names it.
54    Controller {
55        character: Option<String>,
56        asked: crate::pad::PageAsked,
57    },
58    /// A change on a widget's page, made by the character's play window.
59    Widget {
60        /// The page's id.
61        page: String,
62        /// The row's key.
63        key: String,
64        /// Its value; `None` for the kind's own.
65        to: Option<String>,
66    },
67}

69/// What the menu draws from this frame.
70#[derive(Clone, Copy, Debug)]
71pub(crate) struct MenuView<'a> {
72    /// Every character on the roster.
73    pub(crate) roster: &'a [RosterCard],
74    /// The pages the binary last gave, and whose.
75    pub(crate) pages: Option<(&'a str, &'a [Page])>,
76    /// What the binary answered the last request.
77    pub(crate) said: Option<&'a str>,
78    /// Hydra's own pages but the keys: the *Window* page.
79    pub(crate) own: &'a [Page],
80    /// The keybinds, for the *Keys* page.
81    pub(crate) keys: KeysView<'a>,
82    /// The controller file, for the *Controller* page.
83    pub(crate) controller: crate::pad::ControllerView<'a>,
84    /// The widgets' own pages, when the character shown has a play window.
85    pub(crate) widgets: &'a [Page],
86    /// The fonts a font's row offers.
87    pub(crate) fonts: &'a crate::fonts::Names,
88    /// The names the model works a value out by, for the title bar's game
89    /// values (`GameState::value_names`).
90    pub(crate) values: &'a [String],
91}

93/// A roster character's name for the binary: `GAME:Name`.
94pub(crate) fn roster_name(card: &RosterCard) -> String {
95    format!("{}:{}", card.game, card.character)
96}

98/// The settings menu's state, which outlives a frame.
99#[derive(Debug, Default)]
100pub(crate) struct SettingsMenu {
101    /// Whether its window is open.
102    pub(crate) open: bool,
103    /// Whose settings: a character as the roster names it, or `None` for
104    /// Hydra's own.
105    character: Option<String>,
106    /// The page showing, by its id.
107    page: Option<String>,
108    /// The character whose pages were last asked for, since it opened.
109    asked: Option<String>,
110    /// What the menu last said: what a change did, or why it was refused.
111    note: Option<String>,
112    /// The pages' rows as they are being typed.
113    rows: crate::menu::Menu,
114    /// The *Title bar* page's own drawing (`crate::titlebar_page`).
115    titlebar: crate::titlebar_page::TitleBarPage,
116    /// The *Keys* page.
117    keys: KeysPage,
118    /// The *Controller* page.
119    controller: crate::pad::ControllerPage,
120}

122impl SettingsMenu {
123    /// Open on `character`'s settings (`GAME:Name`), or Hydra's own, at
124    /// `page` when one is named: a page named by the start of its id is the
125    /// first of its kind. A character's pages are asked for afresh.
126    pub(crate) fn open_at(&mut self, character: Option<String>, page: Option<&str>) {
127        self.open = true;
128        self.asked = None;
129        if self.character != character {
130            self.note = None;
131        }
132        self.character = character;
133        if let Some(page) = page {
134            self.page = Some(page.to_owned());
135        }
136    }
137
138    /// Whose settings are shown, as the roster names them; `None` for
139    /// Hydra's own.
140    pub(crate) fn character(&self) -> Option<&str> {
141        self.character.as_deref()
142    }
143
144    /// The page showing, by its id.
145    #[cfg(test)]
146    pub(crate) fn page(&self) -> Option<&str> {
147        self.page.as_deref()
148    }

150    /// Whether the *Keys* page waits for a key: the window then hands it
151    /// every key the fork catches.
152    pub(crate) fn waiting_for_key(&self) -> bool {
153        self.open && self.page.as_deref() == Some(KEYS) && self.keys.waiting()
154    }
155
156    /// Whether the *Controller* page waits for a button pressed on the pad.
157    pub(crate) fn waiting_for_button(&self) -> bool {
158        self.open && self.page.as_deref() == Some(CONTROLLER) && self.controller.waiting()
159    }

161    /// The macro set the *Keys* page shows.
162    pub(crate) fn keys_set(&self) -> u8 {
163        self.keys.set()
164    }

168        self.note = Some(said);

171    /// Draw the menu over `view`: what it asks, when anything.
172    pub(crate) fn show(&mut self, ui: &mut egui::Ui, view: &MenuView<'_>) -> Option<MenuAsked> {
173        self.picker(ui, view);
174        if let Some(said) = self.note.as_deref().or(view.said) {
175            ui.weak(said);
176        }
177        ui.separator();
178        let Some(character) = self.character.clone() else {
179            return self.hydra(ui, view);
180        };
181        let mut asked = None;
182        if self.asked.as_deref() != Some(character.as_str()) {
183            self.asked = Some(character.clone());
184            asked = Some(MenuAsked::Binary(HubRequest::Settings(character.clone())));
185        }
186        let theirs = match view.pages {
187            Some((whose, pages)) if whose == character => pages,
188            _ => &[],
189        };
190        let mut titles: Vec<(&str, &str)> = theirs
191            .iter()
192            .map(|page| (page.id.as_str(), page.title.as_str()))
193            .collect();
194        titles.push((KEYS, "Keys"));
195        titles.push((CONTROLLER, "Controller"));
196        titles.extend(
197            view.widgets
198                .iter()
199                .map(|page| (page.id.as_str(), page.title.as_str())),
200        );
201        // Its own pages come first, so the menu waits for them, unless it
202        // was opened at its keys or a widget's page.
203        let elsewhere = self.page.as_deref().is_some_and(|page| {
204            page == KEYS || page == CONTROLLER || page.starts_with(crate::play::WIDGET_PAGE)
205        });
206        if theirs.is_empty() && !elsewhere {
207            ui.weak("Reading the settings...");
208            return asked;
209        }
210        ui.horizontal_top(|ui| {
211            self.contents(ui, &titles);
212            ui.separator();
213            ui.vertical(|ui| {
214                if self.page.as_deref() == Some(KEYS) {
215                    if let Some(change) = self.keys.show(ui, &view.keys, &mut self.note) {
216                        asked = Some(MenuAsked::Key {
217                            character: Some(character.clone()),
218                            change,
219                        });
220                    }
221                    return;
222                }
223                if self.page.as_deref() == Some(CONTROLLER) {
224                    if let Some(now) = self.controller.show(ui, &view.controller) {
225                        asked = Some(MenuAsked::Controller {
226                            character: Some(character.clone()),
227                            asked: now,
228                        });
229                    }
230                    return;
231                }
232                let page = theirs
233                    .iter()
234                    .chain(view.widgets)
235                    .find(|page| self.page.as_deref() == Some(page.id.as_str()));
236                let change = page.and_then(|page| {
237                    egui::ScrollArea::vertical()
238                        .id_salt("settings-page")
239                        .auto_shrink([false, false])
240                        .show(ui, |ui| {
241                            // The title bar's page is drawn its own way:
242                            // its sides as rows of pickers.
243                            if page.id == crate::play::titlebar::PAGE {
244                                self.titlebar.show(ui, page, view.values)
245                            } else {
246                                self.rows.page(ui, page, view.fonts)
247                            }
248                        })
249                        .inner
250                });
251                if let (Some(page), Some(mut change)) = (page, change) {
252                    self.note = None;
253                    change.character.clone_from(&character);
254                    asked = Some(if page.id.starts_with(crate::play::WIDGET_PAGE) {
255                        MenuAsked::Widget {
256                            page: change.page,
257                            key: change.key,
258                            to: change.to,
259                        }
260                    } else {
261                        MenuAsked::Binary(HubRequest::Change(change))
262                    });
263                }
264            });
265        });
266        asked
267    }

269    /// Whose settings: Hydra's own, or a roster character's.
270    fn picker(&mut self, ui: &mut egui::Ui, view: &MenuView<'_>) {
271        let called = |name: Option<&str>| {
272            let Some(name) = name else {
273                return "Hydra (every character)".to_owned();
274            };
275            view.roster
276                .iter()
277                .find(|card| roster_name(card) == name)
278                .map_or_else(
279                    || name.to_owned(),
280                    |card| {
281                        format!(
282                            "{} ({})",
283                            card.character,
284                            crate::launch::game_name(&card.game)
285                        )
286                    },
287                )
288        };
289        ui.horizontal(|ui| {
290            let label = ui.label("Settings for");
291            let before = self.character.clone();
292            egui::ComboBox::from_id_salt("settings-character")
293                .selected_text(called(self.character.as_deref()))
294                .show_ui(ui, |ui| {
295                    ui.selectable_value(&mut self.character, None, called(None));
296                    for card in view.roster {
297                        let name = roster_name(card);
298                        let shown = called(Some(&name));
299                        ui.selectable_value(&mut self.character, Some(name), shown);
300                    }
301                })
302                .response
303                .labelled_by(label.id);
304            if self.character != before {
305                self.note = None;
306            }
307        });
308    }

310    /// The pages' titles down the side, one showing: the one asked for, by
311    /// its id or the start of it, or the first.
312    fn contents(&mut self, ui: &mut egui::Ui, titles: &[(&str, &str)]) {
313        if self
314            .page
315            .as_deref()
316            .is_none_or(|id| titles.iter().all(|(page, _)| *page != id))
317        {
318            let started = self.page.as_deref().and_then(|start| {
319                titles
320                    .iter()
321                    .find(|(page, _)| page.starts_with(start))
322                    .map(|(page, _)| *page)
323            });
324            self.page = started
325                .or_else(|| titles.first().map(|(id, _)| *id))
326                .map(str::to_owned);
327        }
328        // Grouped: the character's own pages, then its window's, then
329        // each custom window's widgets under the window's name, folded;
330        // and scrolled, so a long layout never asks for a taller menu.
331        let groups = grouped(titles);
332        let showing = self.page.clone();
333        egui::ScrollArea::vertical()
334            .id_salt("settings-contents")
335            // As wide as its list, as tall as the menu.
336            .auto_shrink([true, false])
337            .max_width(176.0)
338            .show(ui, |ui| {
339                // Down the side, whatever the row around it lays out.
340                ui.vertical(|ui| {
341                    ui.set_width(170.0);
342                    for group in &groups {
343                        let Some(name) = &group.name else {
344                            for (id, title) in &group.pages {
345                                self.entry(ui, id, title, showing.as_deref());
346                            }
347                            continue;
348                        };
349                        let open = group
350                            .pages
351                            .iter()
352                            .any(|(id, _)| showing.as_deref() == Some(id.as_str()));
353                        egui::CollapsingHeader::new(name)
354                            .id_salt(("settings-group", name))
355                            .default_open(open)
356                            .show(ui, |ui| {
357                                for (id, title) in &group.pages {
358                                    self.entry(ui, id, title, showing.as_deref());
359                                }
360                            });
361                    }
362                });
363            });
364    }

366    /// One page's line in the contents.
367    fn entry(&mut self, ui: &mut egui::Ui, id: &str, title: &str, showing: Option<&str>) {
368        if ui.selectable_label(showing == Some(id), title).clicked() {
369            self.page = Some(id.to_owned());
370        }
371    }

373    /// Hydra's own pages: *Window*, drawn as any other page, and *Keys*,
374    /// every character's.
375    fn hydra(&mut self, ui: &mut egui::Ui, view: &MenuView<'_>) -> Option<MenuAsked> {
376        let mut titles: Vec<(&str, &str)> = view
377            .own
378            .iter()
379            .map(|page| (page.id.as_str(), page.title.as_str()))
380            .collect();
381        titles.push((KEYS, "Keys"));
382        titles.push((CONTROLLER, "Controller"));
383        let mut asked = None;
384        ui.horizontal_top(|ui| {
385            self.contents(ui, &titles);
386            ui.separator();
387            ui.vertical(|ui| {
388                if self.page.as_deref() == Some(KEYS) {
389                    asked = self
390                        .keys
391                        .show(ui, &view.keys, &mut self.note)
392                        .map(|change| MenuAsked::Key {
393                            character: None,
394                            change,
395                        });
396                    return;
397                }
398                if self.page.as_deref() == Some(CONTROLLER) {
399                    asked = self.controller.show(ui, &view.controller).map(|now| {
400                        MenuAsked::Controller {
401                            character: None,
402                            asked: now,
403                        }
404                    });
405                    return;
406                }
407                let page = view
408                    .own
409                    .iter()
410                    .find(|page| self.page.as_deref() == Some(page.id.as_str()));
411                if let Some(page) = page
412                    && let Some(change) = self.rows.page(ui, page, view.fonts)
413                {
414                    self.note = None;
415                    asked = Some(MenuAsked::Own {
416                        key: change.key,
417                        to: change.to,
418                    });
419                }
420            });
421        });
422        asked
423    }

426/// A heading of the contents and the pages under it; no heading for the
427/// character's own pages, which stand at the top.
428struct Group {
429    name: Option<String>,
430    pages: Vec<(String, String)>,
431}
432
433/// `titles` grouped for the contents: the character's own pages first,
434/// unheaded; then each window's widgets under the window's name, read off
435/// a widget page's title, `Health (Vitals)`; widgets standing alone under
436/// *Widgets*; the window's own pages, the title bar, under *Window*.
437fn grouped(titles: &[(&str, &str)]) -> Vec<Group> {
438    let mut own = Vec::new();
439    let mut window = Vec::new();
440    let mut alone = Vec::new();
441    let mut windows: Vec<Group> = Vec::new();
442    for (id, title) in titles {
443        let (id, title) = ((*id).to_owned(), (*title).to_owned());
444        if !id.starts_with(crate::play::WIDGET_PAGE) {
445            own.push((id, title));
446        } else if id == crate::play::titlebar::PAGE {
447            window.push((id, title));
448        } else if let Some((name, holder)) = title
449            .rsplit_once(" (")
450            .and_then(|(name, rest)| rest.strip_suffix(')').map(|holder| (name, holder)))
451        {
452            let holder = holder.to_owned();
453            match windows
454                .iter_mut()
455                .find(|group| group.name.as_deref() == Some(&holder))
456            {
457                Some(group) => group.pages.push((id, name.to_owned())),
458                None => windows.push(Group {
459                    name: Some(holder),
460                    pages: vec![(id, name.to_owned())],
461                }),
462            }
463        } else {
464            alone.push((id, title));
465        }
466    }
467    let mut groups = vec![Group {
468        name: None,
469        pages: own,
470    }];
471    if !window.is_empty() {
472        groups.push(Group {
473            name: Some("Window".to_owned()),
474            pages: window,
475        });
476    }
477    if !alone.is_empty() {
478        groups.push(Group {
479            name: Some("Widgets".to_owned()),
480            pages: alone,
481        });
482    }
483    groups.extend(windows);
484    groups

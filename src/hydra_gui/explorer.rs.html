1//! The Map Explorer: the whole map in a window of its own, for looking
2//! round between hunts.
3//!
4//! The minimap shows the area a character is in and follows it; the
5//! Explorer shows any area, chosen from a list or found by a room: its
6//! number, its game number (`u7120` or the digits alone), words of its
7//! title, or a tag it carries. A find lists what fits; choosing one opens
8//! its area and centres on it, ringed. Every character playing is marked
9//! on the sheet it stands on, by name. The wheel zooms about the pointer,
10//! a drag pans, and the pointer's room shows its card, as the minimap's
11//! does. A click chooses a room, and its details show down the right:
12//! its numbers, its area, where it is, what it looks like, its tags, and
13//! its exits, each with how it is crossed and where it leads; an exit into
14//! another area is coloured and names that area, and a click on where it
15//! leads chooses that room. For the character picked among those playing,
16//! *Walk here* sends its `go2`, *Show route* its `route2`, and *Say* puts
17//! the room in its story.
18//!
19//! The bestiary is the other side of the book: the room's details list
20//! what spawns there, and one chosen tints every room it is found in on
21//! the sheet, so a hunting ground shows at a glance; `.map <creature>`
22//! does the same from the command line, as the bestiary's `[map]` link
23//! sends it, and `.map <room>` finds a room.
24//!
25//! The map's one edit is where a room belongs. The room's area can be
26//! picked; an exit into another area offers *Bring here*, that one room
27//! filed in this area, and *Bring its pocket*, every room reachable from
28//! it that is filed in that other area; a room corrected offers *Reset*.
29//! The corrections are the player's (`hydra_map::Overrides`), laid over
30//! the map as it loads, and *Export for the mapper* writes them out.
31//!
32//! The marks are the minimap's: each kind of place in its colour, with a
33//! legend, and switched off as one. The picked character's saved places
34//! (`.go2 save`) are listed, each a jump to its room.
35//!
36//! The sheets are the atlas's, the same the minimaps draw, drawn with the
37//! minimap's own painters (`crate::widget::minimap`).
38

113    /// Draw the window's inside: the finder down the left, the sheet across
114    /// the rest. `loaded` is the map and its atlas once they are;
115    /// `standing` where each character is.
116    pub(crate) fn show(
117        &mut self,
118        ui: &mut egui::Ui,
119        loaded: Option<&(Arc<Map>, Arc<Atlas>)>,
120        standing: &[Standing],
121        overrides: &Overrides,
122    ) -> (Vec<Asked>, Vec<Edit>) {
123        let mut asked = Vec::new();
124        let mut edits = Vec::new();
125        let Some((map, atlas)) = loaded else {
126            ui.centered_and_justified(|ui| ui.weak("Loading the map."));
127            return (asked, edits);
128        };
129        if self.area.is_none() {
130            self.area = standing
131                .first()
132                .and_then(|who| atlas.scene_of(who.room).ok())
133                .map(|scene| scene.area.clone())
134                .or_else(|| atlas.area_names().into_iter().next());
135        }
136        egui::Panel::left(egui::Id::new("explorer-finder"))
137            .default_size(260.0)
138            .show(ui, |ui| {
139                self.finder(ui, map, atlas, standing);
140                self.marks_and_places(ui, atlas, standing);
141                edits.extend(self.corrections(ui, overrides));
142            });
143        if self.chosen.is_some() {
144            egui::Panel::right(egui::Id::new("explorer-details"))
145                .default_size(320.0)
146                .show(ui, |ui| {
147                    egui::ScrollArea::vertical()
148                        .id_salt("explorer-room")
149                        .show(ui, |ui| {
150                            let (asks, made) = self.details(ui, map, atlas, standing, overrides);
151                            asked.extend(asks);
152                            edits.extend(made);
153                        });
154                });
155        }
156        egui::CentralPanel::default().show(ui, |ui| self.sheet(ui, atlas, standing));
157        (asked, edits)
158    }

160    /// The marks' switch and legend, and the picked character's saved
161    /// places, under the finder.
162    fn marks_and_places(&mut self, ui: &mut egui::Ui, atlas: &Atlas, standing: &[Standing]) {
163        ui.add_space(6.0);
164        ui.horizontal(|ui| {
165            ui.checkbox(&mut self.marks, "Marks")
166                .on_hover_text("Each kind of place in its colour");
167            if ui
168                .small_button(if self.legend {
169                    "Legend \u{25be}"
170                } else {
171                    "Legend \u{25b8}"
172                })
173                .clicked()
174            {
175                self.legend = !self.legend;
176            }
177        });
178        if self.legend {
179            for (_, token, name) in minimap::MARK_KINDS {
180                ui.horizontal(|ui| {
181                    let (rect, _) = ui.allocate_exact_size(vec2(10.0, 10.0), Sense::hover());
182                    ui.painter()
183                        .circle_filled(rect.center(), 4.0, theme::color(ui.ctx(), token));
184                    ui.label(name);
185                });
186            }
187        }
188        let Some(who) = self.who(standing) else {
189            return;
190        };
191        if who.saved.is_empty() {
192            return;
193        }
194        ui.add_space(6.0);
195        ui.label(egui::RichText::new(format!("{}'s saved places", who.name)).strong());
196        let mut go = None;
197        for (name, rooms) in &who.saved {
198            let said = match rooms.as_slice() {
199                [one] => format!("{name}  #{one}"),
200                many => format!("{name}  {} rooms", many.len()),
201            };
202            if ui.selectable_label(false, said).clicked() {
203                go = rooms.first().copied();
204            }
205        }
206        if let Some(room) = go {
207            self.go_to(room, atlas);
208        }
209    }

211    /// The corrections so far, and the export, at the foot of the finder.

213        let mut edits = Vec::new();

223            if ui
224                .button("Export for the mapper")

227            {

229            }
230        }
231        if let Some(said) = &self.said {
232            ui.weak(said);
233        }
234        edits

237    /// Tint every room `creature` is found in, and show the first of them.
238    fn tint(&mut self, map: &Map, atlas: &Atlas, creature: &codex::Creature) {
239        self.tinted = Some(creature.id.clone());
240        self.tinted_rooms = map
241            .rooms()
242            .iter()
243            .filter(|room| {
244                room.uid
245                    .iter()
246                    .any(|uid| u32::try_from(uid.0).is_ok_and(|uid| creature.found_at(uid)))
247            })
248            .map(|room| room.id.0)
249            .collect();
250        if let Some(first) = self.tinted_rooms.iter().min().copied() {
251            self.go_to(first, atlas);
252        }
253    }

255    /// `.map <words>`: a room by its number found and shown, else the
256    /// creature named, its grounds tinted and shown. What was done.
257    pub(crate) fn asked(
258        &mut self,
259        map: &Map,
260        atlas: &Atlas,
261        words: &str,
262    ) -> Result<String, String> {
263        let words = words.trim();
264        let digits = words.strip_prefix(['u', 'U']).unwrap_or(words);
265        if digits.chars().all(|c| c.is_ascii_digit()) && !digits.is_empty() {
266            let found = hydra_map::found(map, words, 2);
267            let Some(room) = found.first() else {
268                return Err(format!("The map has no room {words}."));
269            };
270            self.go_to(room.id, atlas);
271            return Ok(format!("The map shows #{} {}.", room.id, room.title));
272        }
273        let exact = codex::by_name(words);
274        let creatures = if exact.is_empty() {
275            let nouns = codex::by_noun(words);
276            if nouns.is_empty() {
277                let wanted = words.to_lowercase();
278                codex::creatures()
279                    .filter(|c| c.name.to_lowercase().contains(&wanted))
280                    .collect()
281            } else {
282                nouns
283            }
284        } else {
285            exact
286        };
287        match creatures.as_slice() {
288            [one] => {
289                self.tint(map, atlas, one);
290                Ok(format!(
291                    "The map tints the {} rooms {} is found in.",
292                    self.tinted_rooms.len(),
293                    one.name
294                ))
295            }
296            [] => Err(format!("The bestiary has no creature like {words}.")),
297            many => Err(format!(
298                "{words} names {} creatures: .bestiary {words} lists them.",
299                many.len()
300            )),
301        }
302    }

311    /// The room chosen, in full, and what to do about it.
312    fn details(
313        &mut self,
314        ui: &mut egui::Ui,
315        map: &Map,
316        atlas: &Atlas,
317        standing: &[Standing],
318        overrides: &Overrides,
319    ) -> (Vec<Asked>, Vec<Edit>) {
320        let mut asked = Vec::new();
321        let mut edits = Vec::new();
322        let Some(id) = self.chosen else {
323            return (asked, edits);
324        };
325        let Some(room) = map.room(RoomId(id)) else {
326            ui.weak(format!("The map has no room {id}."));
327            return (asked, edits);
328        };
329        let title = room
330            .title
331            .last()
332            .map_or("", String::as_str)
333            .trim_matches(['[', ']']);
334        ui.heading(title);
335        let uids: Vec<String> = room.uid.iter().map(|uid| format!("u{}", uid.0)).collect();
336        ui.weak(format!("Room {id}  {}", uids.join(" ")));
337        let area = atlas.area_of(id).map(str::to_owned);
338        edits.extend(area_row(ui, atlas, id, area.as_deref(), overrides));
339        for (name, value) in [
340            ("Location", room.location.as_deref()),
341            ("Terrain", room.terrain.as_deref()),
342            ("Climate", room.climate.as_deref()),
343        ] {
344            if let Some(value) = value {
345                ui.label(format!("{name}: {value}"));
346            }
347        }
348        if !room.tags.is_empty() {
349            ui.label(format!("Tags: {}", room.tags.join(", ")));
350        }
351        ui.add_space(4.0);
352        asked.extend(self.spawns(ui, map, atlas, room, standing));
353        if let Some(who) = self.who(standing) {
354            if standing.len() > 1 {
355                egui::ComboBox::from_id_salt("explorer-who")
356                    .selected_text(&who.name)
357                    .show_ui(ui, |ui| {
358                        for each in standing {
359                            if ui
360                                .selectable_label(each.session == who.session, &each.name)
361                                .clicked()
362                            {
363                                self.who = Some(each.session);
364                            }
365                        }
366                    });
367            }
368            ui.horizontal(|ui| {
369                let send = |line: String| (who.session, crate::play::Asked::Command(line));
370                if ui
371                    .button("Walk here")
372                    .on_hover_text(format!("{}: .go2 {id}", who.name))
373                    .clicked()
374                {
375                    asked.push(send(format!("go2 {id}")));
376                }
377                if ui
378                    .button("Show route")
379                    .on_hover_text(format!("{}: .route2 {id}, on the minimap", who.name))
380                    .clicked()
381                {
382                    asked.push(send(format!("route2 {id}")));
383                }
384                if ui
385                    .button("Say")
386                    .on_hover_text(format!("The room, in {}'s story", who.name))
387                    .clicked()
388                {
389                    asked.push((who.session, crate::play::Asked::Room { id, brief: false }));
390                }
391            });
392        } else {
393            ui.weak("No character is playing.");
394        }
395        edits.extend(self.look(ui, map, atlas, room, area.as_deref()));
396        (asked, edits)
397    }

399    /// The area picker, the search box and what it found, and who stands
400    /// where.
401    fn finder(&mut self, ui: &mut egui::Ui, map: &Map, atlas: &Atlas, standing: &[Standing]) {
402        ui.heading("Map");
403        let areas = atlas.area_names();
404        let shown = self.area.clone().unwrap_or_default();
405        egui::ComboBox::from_id_salt("explorer-area")
406            .selected_text(shown.replace('-', " "))
407            .width(ui.available_width())
408            .show_ui(ui, |ui| {
409                for area in &areas {
410                    if ui
411                        .selectable_label(
412                            self.area.as_deref() == Some(area),
413                            area.replace('-', " "),
414                        )
415                        .clicked()
416                    {
417                        self.area = Some(area.clone());
418                        self.camera = None;
419                    }
420                }
421            });
422        ui.add_space(6.0);
423        let typed = ui.add(
424            egui::TextEdit::singleline(&mut self.search)
425                .hint_text("a room: 228, u7120, words of its title, a tag")
426                .desired_width(f32::INFINITY),
427        );
428        if typed.changed() {
429            self.found = hydra_map::found(map, &self.search, 40);
430        }
431        let mut go = None;
432        egui::ScrollArea::vertical()
433            .id_salt("explorer-found")
434            .max_height(ui.available_height() * 0.6)
435            .show(ui, |ui| {
436                for found in &self.found {
437                    let said = format!("#{} {}", found.id, found.title);
438                    if ui
439                        .selectable_label(self.chosen == Some(found.id), said)
440                        .on_hover_text(found.area.replace('-', " "))
441                        .clicked()
442                    {
443                        go = Some(found.id);
444                    }
445                }
446                if self.found.is_empty() && !self.search.trim().is_empty() {
447                    ui.weak("No room fits.");
448                }
449            });
450        if let Some(room) = go {
451            self.go_to(room, atlas);
452        }
453        if !standing.is_empty() {
454            ui.add_space(6.0);
455            ui.label(egui::RichText::new("Standing").strong());
456            for who in standing {
457                let title = atlas
458                    .scene_of(who.room)
459                    .ok()
460                    .and_then(|scene| scene.room(who.room).map(|r| r.title.clone()))
461                    .unwrap_or_else(|| format!("#{}", who.room));
462                if ui
463                    .selectable_label(false, format!("{}  {title}", who.name))
464                    .clicked()
465                {
466                    self.go_to(who.room, atlas);
467                }
468            }
469        }
470    }

472    /// What spawns in `room`, by the bestiary: each a creature, which
473    /// chosen tints its grounds on the sheet, and an *entry* for the
474    /// picked character's story.
475    fn spawns(
476        &mut self,
477        ui: &mut egui::Ui,
478        map: &Map,
479        atlas: &Atlas,
480        room: &hydra_map::Room,
481        standing: &[Standing],
482    ) -> Vec<Asked> {
483        let mut asked = Vec::new();
484        let mut creatures: Vec<&codex::Creature> = Vec::new();
485        for uid in &room.uid {
486            if let Ok(uid) = u32::try_from(uid.0) {
487                for creature in codex::in_room(uid) {
488                    if !creatures.iter().any(|c| c.id == creature.id) {
489                        creatures.push(creature);
490                    }
491                }
492            }
493        }
494        if creatures.is_empty() {
495            return asked;
496        }
497        creatures.sort_by_key(|c| (c.level.unwrap_or(i32::MAX), c.name.clone()));
498        ui.label(egui::RichText::new("Spawns here").strong());
499        let who = self.who(standing).map(|who| who.session);
500        let mut tint = None;
501        for creature in creatures {
502            ui.horizontal(|ui| {
503                let level = creature
504                    .level
505                    .map_or_else(String::new, |level| format!("{level:>3} "));
506                let chosen = self.tinted.as_deref() == Some(creature.id.as_str());
507                if ui
508                    .selectable_label(chosen, format!("{level}{}", creature.name))
509                    .on_hover_text("Tint every room it is found in")
510                    .clicked()
511                {
512                    tint = Some(if chosen { None } else { Some(creature) });
513                }
514                if let Some(who) = who
515                    && ui
516                        .small_button("entry")
517                        .on_hover_text("Its bestiary entry")
518                        .clicked()
519                {
520                    asked.push((
521                        who,
522                        crate::play::Asked::Command(format!("bestiary {}", creature.name)),
523                    ));
524                }
525            });
526        }
527        match tint {
528            Some(Some(creature)) => self.tint(map, atlas, creature),
529            Some(None) => {
530                self.tinted = None;
531                self.tinted_rooms.clear();
532            }
533            None => {}
534        }
535        ui.add_space(4.0);
536        asked
537    }

539    /// What the room looks like, its ways out, and its exits: each with
540    /// how it is crossed and where it leads, one into another area coloured
541    /// and naming that area; a click on where it leads chooses that room.
542    fn look(
543        &mut self,
544        ui: &mut egui::Ui,
545        map: &Map,
546        atlas: &Atlas,
547        room: &hydra_map::Room,
548        area: Option<&str>,
549    ) -> Vec<Edit> {
550        let mut edits = Vec::new();
551        ui.add_space(4.0);
552        if let Some(description) = room.description.last() {
553            ui.add(egui::Label::new(egui::RichText::new(description).weak()).wrap());
554        }
555        if let Some(paths) = room.paths.last() {
556            ui.weak(paths);
557        }
558        ui.add_space(4.0);
559        ui.label(egui::RichText::new(format!("Exits ({})", room.exits.len())).strong());
560        let mut go = None;
561        for exit in &room.exits {
562            let to = exit.to.0;
563            let led = map
564                .room(exit.to)
565                .and_then(|room| room.title.last())
566                .map_or_else(
567                    || format!("#{to}"),
568                    |t| t.trim_matches(['[', ']']).to_owned(),
569                );
570            let elsewhere = atlas
571                .area_of(to)
572                .filter(|other| area != Some(other))
573                .map(str::to_owned);
574            ui.horizontal_wrapped(|ui| {
575                ui.label(crossing(&exit.crossing));
576                let shown = egui::RichText::new(format!("\u{2192} {led}"));
577                let shown = match &elsewhere {
578                    Some(_) => shown.color(theme::color(ui.ctx(), T::Accent)),
579                    None => shown,
580                };
581                if ui
582                    .add(egui::Label::new(shown).sense(Sense::click()))
583                    .clicked()
584                {
585                    go = Some(to);
586                }
587                if let (Some(other), Some(here)) = (&elsewhere, area) {
588                    ui.weak(format!("in {}", other.replace('-', " ")));
589                    if ui
590                        .small_button("Bring here")
591                        .on_hover_text(format!("File that room in {}", here.replace('-', " ")))
592                        .clicked()
593                    {
594                        edits.push(Edit::File {
595                            rooms: vec![to],
596                            area: Some(here.to_owned()),
597                        });
598                    }
599                    if ui
600                        .small_button("Bring its pocket")
601                        .on_hover_text(format!(
602                            "File every room reachable from it that is in {} in {}",
603                            other.replace('-', " "),
604                            here.replace('-', " ")
605                        ))
606                        .clicked()
607                    {
608                        let area_of = |id: u32| atlas.area_of(id).map(str::to_owned);
609                        edits.push(Edit::File {
610                            rooms: hydra_map::pocket(map, to, other, &area_of),
611                            area: Some(here.to_owned()),
612                        });
613                    }
614                }
615            });
616        }
617        if let Some(to) = go {
618            self.go_to(to, atlas);
619        }
620        edits
621    }

623    /// Show room `room`: its sheet, centred on it, chosen.
624    fn go_to(&mut self, room: u32, atlas: &Atlas) {

629        self.goto = Some(room);
630    }
631
632    /// The sheet: its rooms, lines and marks, the characters on it, the
633    /// room chosen ringed; steered by the wheel and a drag.
634    fn sheet(&mut self, ui: &mut egui::Ui, atlas: &Atlas, standing: &[Standing]) {
635        let size = ui.available_size().max(vec2(80.0, 60.0));
636        let (rect, response) = ui.allocate_exact_size(size, Sense::click_and_drag());
637        let painter = ui.painter_at(rect);
638        painter.rect_filled(
639            rect,
640            theme::corner(ui.ctx()),
641            theme::color(ui.ctx(), T::MapBackground),
642        );
643        let Some(area) = self.area.clone() else {
644            minimap::waiting(&painter, rect, "No area.");
645            return;
646        };
647        let Some(scene) = atlas.sheet(&area) else {
648            minimap::waiting(&painter, rect, "Laying out the area.");
649            ui.ctx()
650                .request_repaint_after(std::time::Duration::from_millis(300));
651            return;
652        };
653        let scene = scene.as_ref();
654        let (mut centre, mut zoom) = self.camera.unwrap_or_else(|| (middle(scene), ZOOM));
655        if let Some(room) = self.goto.take()
656            && let Some(at) = scene.room(room).map(|r| minimap::cell(r.cell))
657        {
658            centre = at;
659        }
660        // The wheel zooms about the pointer; a drag pans.
661        if response.dragged() {
662            centre -= response.drag_delta() / zoom;
663        }
664        if let Some(pointer) = response.hover_pos() {
665            let scroll = ui.input(|i| i.smooth_scroll_delta.y);
666            if scroll != 0.0 {
667                let before = centre + (pointer - rect.center()) / zoom;
668                zoom = (zoom * (scroll / 200.0).exp()).clamp(ZOOM_RANGE.0, ZOOM_RANGE.1);
669                centre = before - (pointer - rect.center()) / zoom;
670            }
671        }
672        self.camera = Some((centre, zoom));
673        let to_screen = |p: Vec2| rect.center() + (p - centre) * zoom;
674        let all = |_: Option<usize>| 1.0;
675        // The tinted creature's grounds, under everything.
676        if !self.tinted_rooms.is_empty() {
677            let wash = theme::color(ui.ctx(), T::Accent).gamma_multiply(0.35);
678            let reach = (zoom * 0.7).clamp(3.0, 10.0) * 1.9;
679            for room in &scene.rooms {
680                if self.tinted_rooms.contains(&room.id) {
681                    painter.rect_filled(
682                        egui::Rect::from_center_size(
683                            to_screen(minimap::cell(room.cell)),
684                            Vec2::splat(reach),
685                        ),
686                        reach / 3.0,
687                        wash,
688                    );
689                }
690            }
691        }
692        minimap::draw_lines(&painter, scene, &to_screen, zoom, &all);
693        minimap::draw_rooms(&painter, rect, scene, &to_screen, zoom, &all);
694        if self.marks {
695            minimap::draw_marks(&painter, scene, &to_screen, zoom, &all);
696        }
697        minimap::draw_labels(&painter, scene, &to_screen, zoom, &all);
698        let square = (zoom * 0.7).clamp(3.0, 10.0);
699        if let Some(chosen) = self.chosen.and_then(|id| scene.room(id)) {
700            painter.circle_stroke(
701                to_screen(minimap::cell(chosen.cell)),
702                square * 0.9 + 4.0,
703                Stroke::new(2.0, theme::color(ui.ctx(), T::Accent)),
704            );
705        }
706        for who in standing {
707            let Some(room) = scene.room(who.room) else {
708                continue;
709            };
710            let at = to_screen(minimap::cell(room.cell));
711            let you = theme::color(ui.ctx(), T::MapYou);
712            painter.circle_stroke(at, square * 0.9 + 3.0, Stroke::new(2.5, you));
713            painter.text(
714                at + vec2(square + 6.0, 0.0),
715                egui::Align2::LEFT_CENTER,
716                &who.name,
717                egui::FontId::proportional(12.0),
718                you,
719            );
720        }
721        let layers = [(scene, Vec2::ZERO)];
722        let from_screen = |p: Pos2| centre + (p - rect.center()) / zoom;
723        let reach = (square * 0.9 + 3.0) / zoom;
724        let pointed = |at: Pos2| minimap::hit(&layers, from_screen(at), reach);
725        minimap::draw_hover(ui, &response, &pointed, &layers);
726        if response.clicked()
727            && let Some(room) = response.interact_pointer_pos().and_then(pointed)
728        {
729            self.chosen = Some(room);
730        }
731    }
732}

734/// The room's area, picked from the atlas's, with *Reset* when the player
735/// corrected it.
736fn area_row(
737    ui: &mut egui::Ui,
738    atlas: &Atlas,
739    id: u32,
740    area: Option<&str>,
741    overrides: &Overrides,
742) -> Vec<Edit> {
743    let mut edits = Vec::new();
744    ui.horizontal(|ui| {
745        ui.label("Area:");
746        egui::ComboBox::from_id_salt(("explorer-room-area", id))
747            .selected_text(area.unwrap_or("none").replace('-', " "))
748            .show_ui(ui, |ui| {
749                for name in atlas.area_names() {
750                    if ui
751                        .selectable_label(area == Some(name.as_str()), name.replace('-', " "))
752                        .clicked()
753                        && area != Some(name.as_str())
754                    {
755                        edits.push(Edit::File {
756                            rooms: vec![id],
757                            area: Some(name),
758                        });
759                    }
760                }
761            });
762        if overrides.areas.contains_key(&id)
763            && ui
764                .small_button("Reset")
765                .on_hover_text("Back where the map files it")
766                .clicked()
767        {
768            edits.push(Edit::File {
769                rooms: vec![id],
770                area: None,
771            });
772        }
773    });
774    edits
775}

777/// A crossing in words: the command, or what kind of crossing it is.
778fn crossing(crossing: &hydra_map::Crossing) -> String {
779    match crossing {
780        hydra_map::Crossing::Command(command) => command.clone(),
781        hydra_map::Crossing::Steps(steps) => format!("{} steps", steps.len()),
782        hydra_map::Crossing::Routine(routine) => {
783            let said = format!("{routine:?}");
784            format!(
785                "routine: {}",
786                said.split(['(', ' ', '{']).next().unwrap_or(&said)
787            )
788        }
789        hydra_map::Crossing::PassThrough(_) => "pass through".to_owned(),
790        hydra_map::Crossing::Unported(_) => "not walked yet".to_owned(),
791        hydra_map::Crossing::Unknown(name) => format!("unknown: {name}"),
792    }
793}

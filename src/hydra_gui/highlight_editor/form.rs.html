10/// What a banner or a notification says, as the form chooses it.
11#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
12enum Says {
13    #[default]
14    Nothing,
15    Line,
16    Words,
17}
18
19impl Says {
20    const ALL: [Self; 3] = [Self::Nothing, Self::Line, Self::Words];
21
22    fn name(self) -> &'static str {
23        match self {
24            Self::Nothing => "Nothing",
25            Self::Line => "The line",
26            Self::Words => "These words:",
27        }
28    }
29
30    /// `say` as the form shows it, and its words.
31    fn of(say: Option<&Say>) -> (Self, String) {
32        match say {
33            None | Some(Say::Line(false)) => (Self::Nothing, String::new()),
34            Some(Say::Line(true)) => (Self::Line, String::new()),
35            Some(Say::Words(words)) => (Self::Words, words.clone()),
36        }
37    }
38
39    /// As the file writes it, with `words`.
40    fn say(self, words: &str) -> Option<Say> {
41        match self {
42            Self::Nothing => None,
43            Self::Line => Some(Say::Line(true)),
44            Self::Words if words.trim().is_empty() => Some(Say::Line(true)),
45            Self::Words => Some(Say::Words(words.trim().to_owned())),
46        }
47    }
48}

56/// A flag or a status as the form edits it: its name, its seconds (none
57/// until cleared), or that it clears.
58#[derive(Clone, Debug, Default, PartialEq, Eq)]
59struct MarkDraft {
60    name: String,
61    seconds: u32,
62    clear: bool,
63}
64
65impl MarkDraft {
66    fn of(mark: Option<&Mark>) -> Self {
67        mark.map_or_else(Self::default, |mark| Self {
68            name: mark.name().to_owned(),
69            seconds: mark.seconds().unwrap_or(0),
70            clear: mark.clears(),
71        })
72    }
73
74    fn mark(&self) -> Option<Mark> {
75        let name = self.name.trim();
76        if name.is_empty() {
77            return None;
78        }
79        Some(if self.clear {
80            Mark::Full {
81                name: name.to_owned(),
82                seconds: None,
83                clear: true,
84            }
85        } else if self.seconds > 0 {
86            Mark::Full {
87                name: name.to_owned(),
88                seconds: Some(self.seconds),
89                clear: false,
90            }
91        } else {
92            Mark::Named(name.to_owned())
93        })
94    }
95
96    /// Its row: the name, then set for so long or until cleared, or cleared.
97    fn show(&mut self, ui: &mut egui::Ui, hint: &str) {
98        ui.horizontal(|ui| {
99            ui.add(
100                egui::TextEdit::singleline(&mut self.name)
101                    .hint_text(hint)
102                    .desired_width(140.0),
103            );
104            if self.name.trim().is_empty() {
105                return;
106            }
107            ui.checkbox(&mut self.clear, "clears it");
108            if !self.clear {
109                ui.label("for");
110                ui.add(
111                    egui::DragValue::new(&mut self.seconds)
112                        .range(0..=86_400)
113                        .suffix(" s"),
114                )
115                .on_hover_text("0: until cleared");
116            }
117        });
118    }
119}

251    /// The form as a highlight.
252    pub(crate) fn rule(&self) -> Rule {
253        let pattern = (!self.pattern.is_empty()).then(|| self.pattern.clone());
254        let category = self.category.trim();
255        Rule {
256            category: if category.is_empty() {
257                "General"
258            } else {
259                category
260            }
261            .to_owned(),
262            text: if self.regex { None } else { pattern.clone() },
263            regex: if self.regex { pattern } else { None },
264            case_sensitive: self.case_sensitive,
265            whole_word: self.whole_word && !self.regex,
266            color: self.color.map(hex),
267            background: self.background.map(hex),
268            line: self.line,
269            squelch: self.squelch,
270            sound: (!self.sound.trim().is_empty()).then(|| self.sound.trim().to_owned()),
271            volume: (self.volume != 100).then_some(self.volume),
272            rumble: (!self.rumble.trim().is_empty()).then(|| self.rumble.trim().to_owned()),
273            alert: self.alert.0.say(&self.alert.1),
274            notify: self.notify.0.say(&self.notify.1),
275            speak: self.speak.0.say(&self.speak.1),
276            cooldown: ((self.cooldown - COOLDOWN).abs() > f64::EPSILON).then_some(self.cooldown),
277            characters: self
278                .characters
279                .split(',')
280                .map(str::trim)
281                .filter(|name| !name.is_empty())
282                .map(str::to_owned)
283                .collect(),
284            timer: written(&self.timer),
285            seconds: (self.seconds > 0).then_some(self.seconds),
286            caught: written(&self.caught),
287            times: (self.times != 1).then_some(self.times),
288            down: written(&self.down),
289            list: (self.list != List::Custom).then(|| self.list.name().to_owned()),
290            origin: self.origin.clone(),
291            when: self.when.clone(),
292            rearm: ((self.rearm - REARM).abs() > f64::EPSILON).then_some(self.rearm),
293            substitute: (!self.substitute.is_empty()).then(|| self.substitute.clone()),
294            redirect: written(&self.redirect),
295            copy: self.copy && !self.redirect.trim().is_empty(),
296            flag: self.flag.mark(),
297            status: self.status.mark(),
298            send: self.sends(),
299            approved: self.approved.clone(),
300            off: self.off,
301        }
302    }

304    /// Draw the form into `ui`; `chosen` is the name the highlight is kept
305    /// by, `None` for a new one, and `categories` those there are. What it
306    /// asks.
307    pub(crate) fn show(
308        &mut self,
309        ui: &mut egui::Ui,
310        chosen: Option<&str>,
311        categories: &[String],
312    ) -> Vec<Asked> {
313        let mut asked = Vec::new();
314        ui.heading(chosen.map_or_else(|| "A new highlight".to_owned(), str::to_owned));
315        if let Some(origin) = &self.origin {
316            ui.weak(format!("Brought by an import: {origin}"));
317        }
318        egui::Grid::new("highlight-form")
319            .num_columns(2)
320            .spacing([12.0, 6.0])
321            .show(ui, |ui| {
322                self.what(ui, categories);
323                self.look(ui);
324                self.attention(ui);
325                self.timed(ui);
326                self.does(ui);
327            });
328        ui.add_space(6.0);
329        egui::CollapsingHeader::new("When")
330            .id_salt("highlight-when")
331            .default_open(self.when.is_some())
332            .show(ui, |ui| {
333                ui.weak("It acts only while this holds. With no words to look for, it acts each time this comes to hold.");
334                super::when::when_ui(ui, &mut self.when);
335                if self.when.is_some() && self.pattern.trim().is_empty() {
336                    ui.horizontal(|ui| {
337                        ui.label("Again only once it has not held for");
338                        ui.add(
339                            egui::DragValue::new(&mut self.rearm)
340                                .range(0.0..=3600.0)
341                                .speed(0.5)
342                                .suffix(" s"),
343                        );
344                    });
345                }
346            });
347        ui.add_space(6.0);
348        ui.horizontal(|ui| {
349            if ui.button("Save").clicked() {
350                asked.push(Asked::Save {
351                    was: chosen.map(str::to_owned),
352                    name: self.name.trim().to_owned(),
353                    rule: Box::new(self.rule()),
354                });
355            }
356            if let Some(chosen) = chosen
357                && ui.button("Remove").clicked()
358            {
359                asked.push(Asked::Remove(chosen.to_owned()));
360            }
361        });
362        ui.separator();
363        self.try_it(ui);
364        asked
365    }

606    /// A line to try the highlight on, and what it makes of it.
607    fn try_it(&mut self, ui: &mut egui::Ui) {
608        ui.strong("Try it");
609        ui.add(
610            egui::TextEdit::singleline(&mut self.test)
611                .hint_text("Paste a line from the game")
612                .desired_width(f32::INFINITY),
613        );
614        if self.test.is_empty() {
615            return;
616        }
617        let mut rule = self.rule();
618        rule.characters.clear();
619        rule.off = false;
620        if let Err(why) = Matcher::check_rule("this", &rule) {
621            ui.colored_label(ui.visuals().warn_fg_color, why);
622            return;
623        }
624        if self.pattern.trim().is_empty() {
625            ui.weak("A condition alone acts as its condition comes to hold, not on a line.");
626            return;
627        }
628        // Tried as if its condition held: the window has no character here.
629        if rule.when.take().is_some() {
630            ui.weak("It acts only while its condition holds; tried here as if it did.");
631        }
632        // Tried as if approved, so what it would send shows.
633        rule.approved.clone_from(&rule.send);
634        let mut file = HighlightFile::default();
635        file.rules.insert("this".to_owned(), rule);
636        let (matcher, _) = Matcher::new(&file, "");
637        let line = [StyledRun::plain(&self.test)];
638        let response = matcher.respond(&line);
639        let job = crate::text::job(&response.runs, ui.style(), &crate::theme::palette(ui.ctx()));
640        ui.label(job);
641        let mut said = Vec::new();
642        if response.runs == line && !response.squelched && response.fired.is_empty() {
643            said.push("It does not match this line.".to_owned());
644        }
645        if response.squelched {
646            said.push("The line is hidden.".to_owned());
647        }
648        for fired in &response.fired {
649            if let Some(alert) = &fired.alert {
650                said.push(format!("Banner: {alert}"));
651            }
652            if let Some(sound) = &fired.sound {
653                said.push(format!("Sound: {sound}"));
654            }
655            if let Some(notify) = &fired.notify {
656                said.push(format!("Notification: {notify}"));
657            }
658            if let Some(speak) = &fired.speak {
659                said.push(format!("Speech: {speak}"));
660            }
661            if let Some(rumble) = &fired.rumble {
662                said.push(format!("Rumble: {rumble}"));
663            }
664            if let Some(timing) = &fired.timer {
665                said.push(match timing.seconds {
666                    Some(seconds) => format!("Timer {}: {seconds} s", timing.name),
667                    None => format!("Timer {}: ended", timing.name),
668                });
669            }
670            for (what, mark) in [("Flag", &fired.flag), ("Status", &fired.status)] {
671                let Some(mark) = mark else {
672                    continue;
673                };
674                said.push(match (mark.clears(), mark.seconds()) {
675                    (true, _) => format!("{what} {}: cleared", mark.name()),
676                    (false, Some(seconds)) => {
677                        format!("{what} {}: set for {seconds} s", mark.name())
678                    }
679                    (false, None) => format!("{what} {}: set", mark.name()),
680                });
681            }
682            for command in &fired.send {
683                said.push(format!("Sends: {command}"));
684            }
685        }
686        if let Some((stream, copy)) = &response.redirect {
687            said.push(if *copy {
688                format!("The line goes to the {stream} stream too.")
689            } else {
690                format!("The line goes to the {stream} stream instead.")
691            });
692        }
693        for line in said {
694            ui.label(line);
695        }
696    }
697}

699/// `words`, trimmed, unless there are none.
700fn written(words: &str) -> Option<String> {
701    let words = words.trim();
702    (!words.is_empty()).then(|| words.to_owned())
703}

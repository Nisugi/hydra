1//! Where a pane lands when it is dragged or resized: its moving edges
2//! pulled to the play window's edges, to the other panes' edges, and to the
3//! grid, when one is near.
4//!
5//! Pure geometry: the gesture read per axis from the totals since it began
6//! ([`classify`]), the candidates, and the nearest legal one winning, ties
7//! to the more meaningful target (`snap_1d`, [`Target`]'s order). Two rules
8//! keep it sane: a window's centre pairs only with centre lines, never with
9//! an edge or the grid, or a move lands with both edges off the grid; and a
10//! resize that would take a pane below its smallest is skipped before the
11//! nearest is chosen, so an illegal near candidate never hides a legal one.
12//! There are no anchors and no centre lines.
13
14use egui::{Rect, Vec2};
15
16/// How close an edge must come to a target to snap to it, in points.
17pub(crate) const RADIUS: f32 = 8.0;
18
19/// What an edge can snap to, in the order ties are given: the play
20/// window's edge beats a pane flush with it, which beats a grid line
21/// running through both.
22#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
23pub(crate) enum Target {
24    /// The area's own edge.
25    Bound,
26    /// Another pane's edge.
27    Sibling,
28    /// A grid line.
29    Grid,
30}
31
32/// How a gesture moves a rect along one axis, read from the totals since
33/// the gesture began: a move shifts both edges, a resize pins one.
34#[derive(Clone, Copy, Debug, PartialEq, Eq)]
35pub(crate) enum Gesture {
36    /// This axis has not moved.
37    Idle,
38    /// Both edges together: a move.
39    Translate,
40    /// The left or top edge: a resize from that side.
41    MinEdge,
42    /// The right or bottom edge.
43    MaxEdge,
44}
45
46/// Movement below this is rounding, not a gesture.
47const MOVED: f32 = 0.6;
48
49/// Read one axis's gesture from how far its two edges have moved in all.
50pub(crate) fn classify(dmin: f32, dmax: f32) -> Gesture {
51    match (dmin.abs() > MOVED, dmax.abs() > MOVED) {
52        (false, false) => Gesture::Idle,
53        (true, false) => Gesture::MinEdge,
54        (false, true) => Gesture::MaxEdge,
55        (true, true) => Gesture::Translate,
56    }
57}

59/// One engaged snap: a line to draw while the gesture lasts.
60#[derive(Clone, Copy, Debug, PartialEq)]
61pub(crate) struct Guide {
62    /// A vertical line at x = `at`, or a horizontal one at y = `at`.
63    pub(crate) vertical: bool,
64    /// Where.
65    pub(crate) at: f32,
66    /// What it snapped to.
67    pub(crate) target: Target,
68}
69
70/// Snap `now`, which the gesture has made of `start`, against `bounds`,
71/// the other panes' rects, and a grid of `grid` points (0 for none). Axes
72/// are independent; a pane is never resized below `smallest`.
73pub(crate) fn snap(
74    start: Rect,
75    now: Rect,
76    bounds: Rect,
77    siblings: &[Rect],
78    smallest: Vec2,
79    grid: f32,
80) -> (Rect, Vec<Guide>) {
81    let mut rect = now;
82    let mut guides = Vec::new();
83    let axes = [
84        (
85            true,
86            classify(now.min.x - start.min.x, now.max.x - start.max.x),
87        ),
88        (
89            false,
90            classify(now.min.y - start.min.y, now.max.y - start.max.y),
91        ),
92    ];
93    for (vertical, gesture) in axes {
94        let along = |r: Rect| {
95            if vertical {
96                (r.min.x, r.max.x)
97            } else {
98                (r.min.y, r.max.y)
99            }
100        };
101        let (lo, hi) = along(now);
102        let (bound_lo, bound_hi) = along(bounds);
103        let mut candidates = vec![(bound_lo, Target::Bound), (bound_hi, Target::Bound)];
104        for sibling in siblings {
105            let (sibling_lo, sibling_hi) = along(*sibling);
106            candidates.push((sibling_lo, Target::Sibling));
107            candidates.push((sibling_hi, Target::Sibling));
108        }
109        let smallest = if vertical { smallest.x } else { smallest.y };
110        let Some((delta, at, target)) =
111            snap_1d(gesture, (lo, hi), smallest, &candidates, (bound_lo, grid))
112        else {
113            continue;
114        };
115        let (new_lo, new_hi) = match gesture {
116            Gesture::MinEdge => (lo + delta, hi),
117            Gesture::MaxEdge => (lo, hi + delta),
118            Gesture::Translate | Gesture::Idle => (lo + delta, hi + delta),
119        };
120        if vertical {
121            rect.min.x = new_lo;
122            rect.max.x = new_hi;
123        } else {
124            rect.min.y = new_lo;
125            rect.max.y = new_hi;
126        }
127        guides.push(Guide {
128            vertical,
129            at,
130            target,
131        });
132    }
133    (rect, guides)
134}

136/// The nearest legal snap along one axis within [`RADIUS`], as how far to
137/// move, the line it meets and what that is; `None` when nothing is near.
138/// `grid` is its origin and pitch.
139fn snap_1d(
140    gesture: Gesture,
141    (lo, hi): (f32, f32),
142    smallest: f32,
143    candidates: &[(f32, Target)],
144    (origin, pitch): (f32, f32),
145) -> Option<(f32, f32, Target)> {
146    let moving: &[f32] = match gesture {
147        Gesture::Idle => return None,
148        Gesture::Translate => &[lo, hi],
149        Gesture::MinEdge => &[lo],
150        Gesture::MaxEdge => &[hi],
151    };
152    let mut best: Option<(f32, f32, f32, Target)> = None;
153    for &position in moving {
154        let grid_line = (pitch > 0.0)
155            .then(|| origin + ((position - origin) / pitch).round() * pitch)
156            .map(|line| (line, Target::Grid));
157        for &(value, target) in candidates.iter().chain(grid_line.iter()) {
158            let distance = (value - position).abs();
159            if distance > RADIUS {
160                continue;
161            }
162            let delta = value - position;
163            let extent = match gesture {
164                Gesture::MinEdge => hi - (lo + delta),
165                Gesture::MaxEdge => (hi + delta) - lo,
166                Gesture::Translate | Gesture::Idle => hi - lo,
167            };
168            if extent < smallest - 0.01 {
169                continue;
170            }
171            let wins = best.is_none_or(|(best_distance, _, _, best_target)| {
172                distance + 0.001 < best_distance
173                    || ((distance - best_distance).abs() <= 0.001 && target < best_target)
174            });
175            if wins {
176                best = Some((distance, delta, value, target));
177            }
178        }
179    }
180    best.map(|(_, delta, at, target)| (delta, at, target))
181}

183#[cfg(test)]
184mod tests {
185    use super::*;
186    use egui::pos2;

188    fn rect(x: f32, y: f32, w: f32, h: f32) -> Rect {
189        Rect::from_min_size(pos2(x, y), Vec2::new(w, h))
190    }
191
192    const AREA: Rect = Rect {
193        min: pos2(0.0, 0.0),
194        max: pos2(600.0, 400.0),
195    };
196    const SMALLEST: Vec2 = Vec2::new(80.0, 40.0);
197
198    /// Whether two points are the same, to a hundredth.
199    fn same(a: f32, b: f32) -> bool {
200        (a - b).abs() < 0.01
201    }
202
203    #[test]
204    fn a_gesture_is_read_from_its_totals() {
205        assert_eq!(classify(0.2, -0.3), Gesture::Idle, "rounding");
206        assert_eq!(classify(12.0, 12.0), Gesture::Translate);
207        assert_eq!(classify(-7.0, 0.0), Gesture::MinEdge);
208        assert_eq!(classify(0.0, 9.0), Gesture::MaxEdge);
209    }
210
211    /// Moved to within the radius of the area's right edge, a pane butts
212    /// it; nothing near, it stays where it was dropped.
213    #[test]
214    fn a_move_meets_the_edge_it_comes_near() {
215        let start = rect(100.0, 100.0, 200.0, 100.0);
216        let (snapped, guides) = snap(
217            start,
218            start.translate(Vec2::new(295.0, 0.0)),
219            AREA,
220            &[],
221            SMALLEST,
222            0.0,
223        );
224        assert_eq!(snapped, rect(400.0, 100.0, 200.0, 100.0));
225        assert_eq!(
226            guides,
227            [Guide {
228                vertical: true,
229                at: 600.0,
230                target: Target::Bound
231            }]
232        );
233        let (free, none) = snap(
234            start,
235            start.translate(Vec2::new(40.0, 0.0)),
236            AREA,
237            &[],
238            SMALLEST,
239            0.0,
240        );
241        assert_eq!(free, start.translate(Vec2::new(40.0, 0.0)));
242        assert!(none.is_empty());
243    }

245    /// A pane moved beside another meets its edge; on the grid, a move
246    /// keeps both edges on it when the pane is a whole number of cells.
247    #[test]
248    fn a_move_meets_a_neighbour_and_the_grid() {
249        let other = rect(0.0, 0.0, 300.0, 400.0);
250        let start = rect(320.0, 50.0, 200.0, 100.0);
251        let (beside, _) = snap(
252            start,
253            start.translate(Vec2::new(-16.0, 0.0)),
254            AREA,
255            &[other],
256            SMALLEST,
257            0.0,
258        );
259        assert!(same(beside.min.x, 300.0), "{beside:?}");
260        let (gridded, guides) = snap(
261            start,
262            start.translate(Vec2::new(0.0, 13.0)),
263            AREA,
264            &[],
265            SMALLEST,
266            20.0,
267        );
268        assert!(same(gridded.min.y, 60.0), "{gridded:?}");
269        assert!(same(gridded.height(), 100.0), "a move keeps the size");
270        assert_eq!(guides[0].target, Target::Grid);
271    }

273    /// A resize moves only the edge being dragged, and never takes the
274    /// pane below its smallest: the illegal snap is skipped, not taken.
275    #[test]
276    fn a_resize_moves_one_edge_and_keeps_its_smallest() {
277        let start = rect(100.0, 100.0, 200.0, 100.0);
278        let mut dragged = start;
279        dragged.max.x = 594.0;
280        let (wider, _) = snap(start, dragged, AREA, &[], SMALLEST, 0.0);
281        assert_eq!(wider, rect(100.0, 100.0, 500.0, 100.0));
282
283        // Shrinking to 83 wide: a neighbour's near edge, 4 away, would make
284        // it 79, under the smallest, so its far edge, 6 away, is taken.
285        let start = rect(100.0, 100.0, 100.0, 100.0);
286        let mut shrunk = start;
287        shrunk.max.x = 183.0;
288        let wall = rect(179.0, 300.0, 10.0, 50.0);
289        let (kept, _) = snap(start, shrunk, AREA, &[wall], SMALLEST, 0.0);
290        assert!(same(kept.max.x, 189.0), "{kept:?}");
291    }

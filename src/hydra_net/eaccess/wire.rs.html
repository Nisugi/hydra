1//! The `EAccess` wire vocabulary: the types that cross it, and the pure
2//! functions that read and write its fields.
3//!
4//! **Everything here is pure**; the handshake that touches a socket is in
5//! [`super::handshake`].
6//!
7//! That split is what makes the sequence testable at all. A live login is a
8//! terrible place to discover an off-by-one in the `C` walk or a `split('=')`
9//! that truncates a key at its own `=` byte, and no test may run one to find
10//! out. Every function here is checked by a test; the socket half is checked
11//! only by running it live.

38/// What the caller must supply. Borrowed, not owned: nothing here needs to
39/// outlive the call, and an owned struct invites being stored.
40#[derive(Clone, Copy)]
41pub struct Credentials<'a> {
42    /// The account name, not the character name.
43    pub account: &'a str,
44    /// The account password, in the clear. Hashed against the server's key
45    /// before it reaches the socket, and never logged.
46    pub password: &'a str,
47    /// The character to launch, matched case-insensitively against the `C`
48    /// list.
49    pub character: &'a str,
50    /// The instance code. **CASE-SENSITIVE on the wire** -- see
51    /// [`authenticate`](super::authenticate)'s `M` check.
52    pub game_code: &'a str,
53}
54
55/// Deliberately not `Debug`-derived: a derived impl prints the password, and
56/// the one place a credential struct reliably leaks is a debug log written in
57/// a hurry.
58///
59/// **The account is redacted as well**, as `WebLoginRequest` and the binary's
60/// `LiveConnector` -- the other two holders of the same field -- blank it, and
61/// as the wire log registers it for redaction. It is credential-adjacent:
62/// half of what is sent at `A`, and embedded in every character code as
63/// `W_<ACCOUNT>_<SLOT>`. The character and game code stay visible; they are
64/// what a reader diagnosing a login needs.
65impl fmt::Debug for Credentials<'_> {
66    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
67        f.debug_struct("Credentials")
68            .field("account", &"<REDACTED>")
69            .field("password", &"<REDACTED>")
70            .field("character", &self.character)
71            .field("game_code", &self.game_code)
72            .finish()
73    }
74}

76/// Where the game is, and the one-shot key that opens it.
77#[derive(Clone)]
78pub struct LaunchPayload {
79    /// The game host. A plain TCP destination -- **not** the eaccess host and
80    /// **not** TLS.
81    ///
82    /// `String`, not `Option<String>`, **deliberately**. The
83    /// character-generator path (`L\t0\tSTORM`) can omit both host and port,
84    /// so that `"L\tOK\tKEY=abc\n"` is a valid success. Hydra does not support
85    /// that path: no code here sends `L\t0`. Making these `Option` today would
86    /// add a `None` arm that every caller must handle and that nothing can
87    /// produce. **If the generator is ever added, these two fields become
88    /// `Option` in the same commit.**
89    pub gamehost: String,
90    /// The game port.
91    pub gameport: u16,
92    // **`GAMECODE` is NOT carried.**
93    //
94    // Nothing would read it, and it could not be filled consistently. `L`
95    // answers with the server's FAMILY code (`GS` for a `GS3` login, VERIFIED
96    // in a capture), while the web-login fallback has no `GAMECODE` at all and
97    // could only fill it with the REQUESTED code, `GS3`. So one field would
98    // hold two different vocabularies depending on which provider ran.
99    //
100    // Not carried as `None` on the web path either, because an honest field
101    // with no reader is still a field with no reader. The need it would serve
102    // is telling GemStone from DragonRealms. When that is needed, it comes
103    // with its consumer, and `parse_launch` below records the one fact it
104    // needs: the value is a family, not an instance.
105    /// The session key. One shot, short-lived, and a credential: see this
106    /// type's `Debug`.
107    pub key: String,
108}
109
110/// Redacts the key. The payload is the natural thing to log on a successful
111/// login -- "connected to X:Y" -- and the key sits beside the host.
112impl fmt::Debug for LaunchPayload {
113    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
114        f.debug_struct("LaunchPayload")
115            .field("gamehost", &self.gamehost)
116            .field("gameport", &self.gameport)
117            .field("key", &"<REDACTED>")
118            .finish()
119    }
120}

122/// A login failure, naming the stage it happened at.
123///
124/// The stage is the whole point. A wrong password must fail cleanly and
125/// quickly, **naming the stage** -- because every failure in this sequence
126/// that does *not* name its stage points at the credential, and many of them
127/// are something else entirely.
128#[derive(Debug, Clone, PartialEq, Eq)]
129pub struct EaccessError {
130    /// Which step failed: `tls_handshake`, `a_response`, `l_response`, ...
131    pub stage: &'static str,
132    /// What went wrong. Never contains a password or a key.
133    pub detail: String,
134    /// Whether retrying is pointless: the account or the request was REFUSED,
135    /// as against the link failing on the way to asking.
136    ///
137    /// # Why this is a field and not a function of `stage`
138    ///
139    /// Because **one stage is both**. `a_response` covers the server saying
140    /// "authentication rejected" *and* `read_response` reporting "connection
141    /// closed by peer (0 bytes)" or a stage timeout -- a credential rejection
142    /// and two transport failures, under one name.
143    ///
144    /// Getting this wrong costs in both directions, and they are the same
145    /// mistake made twice:
146    ///
147    /// - a wrong password that is retried hammers the auth server, which is
148    ///   pointless and **could lock the account**;
149    /// - an EOF is a transient DROP, not a credential rejection, and treated
150    ///   as "bad credentials, stop retrying" it **strands the session**.
151    ///
152    /// So classifying on `stage` alone would strand a session on every drop
153    /// that happened to land mid-handshake. The layer that knows which it was
154    /// is the one that built the error, which is why it says so here rather
155    /// than leaving a caller to match on the message.
156    ///
157    /// **Defaults to `false`** (`err` sets it), so a failure nobody has
158    /// classified is retried. That is the direction that fails safe for the
159    /// SESSION: an unclassified error treated as fatal costs it outright.
160    ///
161    /// It is not free for the ACCOUNT, and it does not cost merely "a bounded
162    /// ladder": the supervisor retries a transient connect failure forever,
163    /// capped at one attempt per 30s, and every attempt past
164    /// `A` sends the password. So a refusal left at the default is a login
165    /// every 30 seconds until someone notices -- which is why each refusal
166    /// below that is genuinely a verdict calls [`Self::fatal`] explicitly.
167    pub fatal: bool,
168}
169
170impl EaccessError {
171    /// Mark this failure as one no retry can fix.
172    ///
173    /// Used wherever the answer is a verdict rather than a transport failure:
174    /// a rejected `A`, a game code `M` does not list, no entitlement at `F`, a
175    /// character not in `C`, an `L PROBLEM` of 1-3, and a password the hash
176    /// cannot encode.
177    #[must_use]
178    pub fn fatal(mut self) -> Self {
179        self.fatal = true;
180        self
181    }
182}
183
184impl fmt::Display for EaccessError {
185    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
186        write!(f, "[{}] {}", self.stage, self.detail)
187    }
188}

192pub(super) fn err<E: fmt::Display>(stage: &'static str, e: E) -> EaccessError {
193    EaccessError {
194        stage,
195        detail: e.to_string(),
196        // Transient unless a caller says otherwise -- see the `fatal` field.
197        fatal: false,
198    }

201/// The password hash: `((pw[i] - 32) ^ key[i]) + 32`, in `i32`.
202///
203/// # Why this refuses instead of wrapping
204///
205/// 14,336 of the 65,536 byte pairs hash outside `0..=255`. Rust `u8` wraps
206/// there in release and panics in debug. No login is known to have
207/// **successfully sent a byte outside `0..=255`**, so the server's behaviour
208/// there is completely unobserved. Masking with `& 0xFF` would emit bytes no
209/// server has been seen to accept: an untested protocol change.
210///
211/// It also closes the short-key case. Rust's `zip` would silently stop at the
212/// shorter of the two, producing a **wrong password** rather than an error.
213///
214/// # Errors
215///
216/// [`EaccessError`] with stage `hash` if the key is shorter than the password,
217/// or if any byte falls outside `0..=255`.
218///
219/// Only the first is fatal. Whether a byte falls outside the range depends on
220/// the key, and the key is the server's, new with every attempt: a password
221/// with a byte above ASCII fails some keys and passes others. Fatal, it would
222/// stop the retries and skip the web login, which takes the password as
223/// typed. Transient, the next attempt draws a new key and the fallback is
224/// tried.
225pub fn hash_password(password: &[u8], key: &[u8]) -> Result<Vec<u8>, EaccessError> {
226    if key.len() < password.len() {
227        // FATAL: the password is longer than the protocol's key can hash. That
228        // is a property of the password, not of this attempt.
229        return Err(err(
230            "hash",
231            format!(
232                "key ({} bytes) shorter than password ({} bytes) -- refusing \
233                 to truncate, which would send a wrong password rather than \
234                 fail",
235                key.len(),
236                password.len()
237            ),
238        )
239        .fatal());
240    }
241
242    let mut out = Vec::with_capacity(password.len());
243    for (i, (&p, &k)) in password.iter().zip(key.iter()).enumerate() {
244        let result = ((i32::from(p) - 32) ^ i32::from(k)) + 32;
245        if !(0..=255).contains(&result) {
246            // The password byte `p` is deliberately NOT in this message. This
247            // error propagates out of `authenticate` and is printed: it lands
248            // on stderr, in scrollback, and in any `2>` redirect.
249            //
250            // **Only the INDEX is shown.** Not the key byte and the result
251            // either: the password byte is recoverable from those two,
252            // `p = ((result - 32) ^ k) + 32`.
253            //
254            // The index is what a diagnosis actually needs ("the ninth character
255            // of your password") and reveals nothing about the byte.
256            let _ = (p, k, result);
257            return Err(err(
258                "hash",
259                format!(
260                    "password byte {i} hashes out of range, outside 0..=255. \
261                     No login is known to have sent such a byte, so the \
262                     server's behavior is UNOBSERVED. Refusing to guess. \
263                     Neither the byte nor the arithmetic is shown: the key \
264                     byte and the result together recover it."
265                ),
266            ));
267        }
268        #[expect(
269            clippy::cast_possible_truncation,
270            clippy::cast_sign_loss,
271            reason = "the range check immediately above proves 0..=255"
272        )]
273        out.push(result as u8);
274    }
275    Ok(out)
276}

278/// Replace the account name inside a character code with `<ACCOUNT>`.
279///
280/// A character code has the shape `W_<ACCOUNT>_<SLOT>` (VERIFIED in a
281/// capture), so printing one prints the account -- and the `resolve_char`
282/// progress line would do exactly that, on every login, to stderr and into the
283/// scrollback.
284///
285/// That is the same exposure [`redact`] covers for the `A` response, which
286/// would otherwise print the account and the account holder's real name in
287/// full. [`redact`] cannot cover this one, because here the account is INSIDE
288/// a field rather than a field of its own.
289///
290/// Positional, not a guess: keep the leading `W` and the trailing slot, blank
291/// what is between. The slot is taken from the END rather than from index 2, so
292/// an account name containing `_` still redacts whole. A code that does not have
293/// this shape is returned **unchanged** rather than blanked: an unrecognised
294/// code is a diagnostic, and destroying it would remove the thing a reader needs.
295#[must_use]
296pub fn redact_char_code(code: &str) -> String {
297    let parts: Vec<&str> = code.split('_').collect();
298    if parts.len() < 3 || parts[0] != "W" {
299        return code.to_owned();
300    }
301    let slot = parts[parts.len() - 1];
302    format!("W_<ACCOUNT>_{slot}")
303}

305/// Redact anything that looks like a credential before printing.
306///
307/// Two shapes: a bare 32-hex-digit field (a session key on its own), and any
308/// `KEY=` field. Everything else passes through, because the *point* of
309/// printing these lines is diagnosis -- parsing a response before seeing it
310/// fail can turn the server's bare `?` into an innocuous-looking `tier="?"`.
311#[must_use]
312pub fn redact(s: &str) -> String {
313    // The `A` success response has a FIXED shape:
314    //
315    //     A <tab> <ACCOUNT> <tab> KEY <tab> <session key> <tab> <Real Name>
316    //
317    // so the account name and the account holder's real name are POSITIONAL,
318    // and removing them is exact rather than a guess.
319    //
320    // Unredacted, both would go to the terminal, the scrollback, and into any
321    // transcript pasted elsewhere. They are personal data and nothing
322    // downstream needs them.
323    let fields: Vec<&str> = s.split('\t').collect();
324    let is_a_response = fields.first() == Some(&"A") && fields.get(2) == Some(&"KEY");
325
326    fields
327        .iter()
328        .enumerate()
329        .map(|(i, field)| {
330            if is_a_response && i == 1 {
331                "<ACCOUNT>".to_owned()
332            } else if is_a_response && i == 4 {
333                "<NAME>".to_owned()
334            } else if field.len() == 32 && field.chars().all(|c| c.is_ascii_hexdigit()) {
335                "<KEY-REDACTED>".to_owned()
336            } else if field.starts_with("KEY=") {
337                "KEY=<REDACTED>".to_owned()
338            } else {
339                (*field).to_owned()
340            }
341        })
342        .collect::<Vec<_>>()
343        .join("\t")
344}

346/// Find a character's launch code in a `C` response.
347///
348/// Format: `C \t n \t n \t n \t n \t <code> \t <Name> [\t <code> \t <Name>]...`
349/// -- four counts, then code/name pairs from field 5.
350///
351/// Split out of [`authenticate`](super::authenticate) so it can be tested
352/// without a socket: this is
353/// the one piece of parsing in the sequence with an off-by-one to get wrong,
354/// and a live login is a poor place to discover it.
355#[must_use]
356pub fn resolve_char_code<'a>(c_response: &'a str, character: &str) -> Option<&'a str> {
357    let fields: Vec<&str> = c_response.trim().split('\t').collect();
358    let mut i = 5;
359    while i + 1 < fields.len() {
360        if fields[i + 1].eq_ignore_ascii_case(character) {
361            return Some(fields[i]);
362        }
363        i += 2;
364    }
365    None
366}

368/// Every character a `C` response lists, by name, in the server's order: the
369/// names of [`resolve_char_code`]'s code and name pairs, from field 5.
370#[must_use]
371pub fn character_names(c_response: &str) -> Vec<&str> {
372    c_response.trim().split('\t').skip(6).step_by(2).collect()

383/// Whether an `L` response reports success.
384///
385/// # The guard MUST be `L<TAB>OK`, not `^L<TAB>`
386///
387/// A refusal is `L<TAB>PROBLEM<TAB><n>`, which **also** starts with `L<TAB>`. A
388/// loose guard accepts it and then parses a garbage launch payload -- so the
389/// client connects somewhere meaningless with a key that was never issued.
390///
391/// # Why this is a function rather than an inline `starts_with`
392///
393/// Inline, in `handshake::launch_character`, it would need a socket, and no
394/// test could reach it. Pulling the guard out is what makes it testable at
395/// all, which is the same `wire`/`handshake` seam this module already draws.
396#[must_use]
397pub fn is_launch_ok(l_response: &str) -> bool {
398    l_response.starts_with("L	OK")
399}

401/// Parse the `GAMEHOST` / `GAMEPORT` / `KEY` triple out of an `L\tOK` line.
402///
403/// `splitn(2, '=')`, because a `KEY` value could itself contain `=`.
404///
405/// # Errors
406///
407/// [`EaccessError`] with stage `l_response` if any of the three is absent.
408pub fn parse_launch(l_response: &str) -> Result<LaunchPayload, EaccessError> {
409    let mut gamehost = None;
410    let mut gameport = None;
411    let mut key = None;
412    for field in l_response.trim().split('\t') {
413        let mut kv = field.splitn(2, '=');
414        match (kv.next(), kv.next()) {
415            (Some("GAMEHOST"), Some(v)) => gamehost = Some(v.to_owned()),
416            (Some("GAMEPORT"), Some(v)) => gameport = v.parse::<u16>().ok(),
417            // Everything else is dropped. UPPORT, GAME, FULLGAMENAME and
418            // GAMEFILE tell a Simutronics launcher which .EXE to run. GAMECODE
419            // has no consumer yet -- see `LaunchPayload` for why it is not
420            // carried. If one arrives: it is the server's FAMILY code (`GS`),
421            // not the instance `G` selected (`GS3`).
422            (Some("KEY"), Some(v)) => key = Some(v.to_owned()),
423            _ => {}
424        }
425    }
426    Ok(LaunchPayload {
427        gamehost: gamehost.ok_or_else(|| err("l_response", "no GAMEHOST in launch payload"))?,
428        gameport: gameport.ok_or_else(|| err("l_response", "no GAMEPORT in launch payload"))?,
429        key: key.ok_or_else(|| err("l_response", "no KEY in launch payload"))?,
430    })
431}

433/// Check a response answers the command that was sent.
434///
435/// Every response in this sequence is `<letter>\t...`, echoing its command, so
436/// a mismatch means the read stream has **slipped out of step with the write
437/// stream** -- and every field read after that point is meaningless.
438///
439/// Without this check a lowercase game code looks first like an entitlement
440/// problem, then a pricing problem, then a session-state problem.
441///
442/// # Errors
443///
444/// [`EaccessError`] at `stage` if the response does not begin `<letter>\t`.
445pub fn expect_echo(response: &str, letter: char, stage: &'static str) -> Result<(), EaccessError> {
446    let want = format!("{letter}\t");
447    if response.starts_with(&want) {
448        return Ok(());
449    }
450    // **REDACTED, because the premise of this error is that the streams are
451    // out of step.** Whatever is in `response` is by definition not the
452    // response expected -- so it may be the `A` reply, which carries the
453    // session key, the account name and the account holder's real name.
454    //
455    // Printing it raw would be the exact exposure `redact` exists for, on the
456    // one path whose whole subject is "these bytes are not what we thought
457    // they were". The diagnostic value is unaffected:
458    // `redact` blanks credential-shaped fields and passes the rest through,
459    // which is why it exists rather than a blanket suppression.
460    Err(err(
461        stage,
462        format!(
463            "expected a {letter} response, got {:?} -- the read and write \
464             streams are out of step, and nothing parsed after this point is \
465             meaningful",
466            redact(response.trim())
467        ),
468    ))

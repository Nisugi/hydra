1//! What went wrong, and whether anything else should be tried.
2//!
3//! # Why this is a type and not a string
4//!
5//! The supervisor has to tell two situations apart, and getting it wrong is
6//! costly in both directions -- the same shape as
7//! [`reject`](crate::eaccess::Rejection) for the eaccess `A` response:
8//!
9//! | | Retrying |
10//! |---|---|
11//! | the credentials or the character are wrong | pointless, and each try is a strike |
12//! | the link or the server misbehaved | the only way back |
13//!
14//! An enum rather than a list of string codes: the compiler then enforces
15//! that every new failure states its verdict, which a string list cannot.
16//!
17//! # The rule that matters most
18//!
19//! **A credential rejection is never retried through a second system.** Web
20//! login would reject the same credentials eaccess rejected, so falling back
21//! would only resubmit them to a second system for no benefit; the real
22//! problem is surfaced instead.
23//!
24//! That governs the fallback in both directions: eaccess must not fall back to
25//! web login on a bad password, and web login must not retry one either.
26
27use std::fmt;
28
29/// Why a web login did not produce a launch.
30#[derive(Clone, Debug, PartialEq, Eq)]
31pub enum WebLoginFailure {
32    /// The account or password was rejected.
33    ///
34    /// A wrong password and a nonexistent account are classified identically:
35    /// both land on the same `login_error.asp` redirect and
36    /// differ only in a body heading, which is **not** used for control flow.
37    /// The practical benefit is that a stale saved account name fails fast and
38    /// fatally, rather than looping.
39    LoginRejected,
40    /// The account has no active subscription on this instance.
41    ///
42    /// Fatal: a retry cannot buy a subscription. Worth its own variant because
43    /// it is the one failure whose fix is an action the player can take, so the
44    /// message must not read as a password problem.
45    NoSubscription,
46    /// The character is not on this instance's character list.
47    ///
48    /// Fatal. Note the hazard this creates, which is why
49    /// [`Self::UnexpectedResponse`] exists: a non-200 response scrapes to no
50    /// match, so a *transient* 503 would look exactly like a missing character
51    /// and be discarded as fatal. The status is therefore checked **before**
52    /// the scrape.
53    CharacterNotFound,
54    /// The web flow has no confirmed route for this game code.
55    ///
56    /// Fatal: the table is the evidence, and there is nothing to retry into.
57    UnsupportedGameCode,
58    /// The server did something the confirmed protocol shape does not cover.
59    ///
60    /// **Transient.** This is the counterpart of
61    /// [`Rejection::Divergence`](crate::eaccess::Rejection::Divergence) and
62    /// exists for the same reason: "we do not recognise this" is not evidence
63    /// the credentials are wrong. A WAF intercept page, a 503, or a markup
64    /// change all land here.
65    ///
66    /// Carries a short static label naming *where* it diverged, never a body.
67    UnexpectedResponse(&'static str),
68    /// The transport failed before an answer arrived.
69    ///
70    /// **Transient**, and the reason this whole module exists: an unreachable
71    /// eaccess is precisely the case web login is the answer to.
72    Transport(String),
73}
74
75impl WebLoginFailure {
76    /// Whether the **credentials themselves** were refused.
77    ///
78    /// Deliberately narrower than "did this attempt fail". It is the one thing
79    /// a web-login failure can say that stops a retry ladder, because it is the
80    /// one thing that stays true on the next attempt and through another
81    /// provider. See the note below on why there is no broader predicate.
82    #[must_use]
83    pub const fn is_credential_refusal(&self) -> bool {
84        matches!(self, Self::LoginRejected)
85    }
86}
87
88// NO `is_fatal` HERE, deliberately -- and this absence is the design, not an
89// omission.
90//
91// Such a predicate would answer the WRONG QUESTION. "Fatal for web login" and "fatal for this login attempt" are
92// different, and only one variant answers both:
93//
94//   LoginRejected       the credentials were refused          -> stop everything
95//   NoSubscription      not on THIS instance                  -> eaccess may differ
96//   CharacterNotFound   the SCRAPE found nothing              -> eaccess asks `C` properly
97//   UnsupportedGameCode web login's table is narrower         -> eaccess is not
98//
99// All four end a web-login attempt; only the first is evidence against retrying
100// at all. `CharacterNotFound` is the sharpest: web login finds characters by
101// scraping HTML, which fails SILENTLY when markup changes, so treating it as
102// proof the character does not exist would turn a play.net markup change into a
103// permanently dead session.
104//
105// So the caller that matters -- `eaccess::fallback::carry_forward` -- matches
106// `LoginRejected` by name. A four-variant predicate there would have been
107// exactly wrong while looking tidier. If a caller ever genuinely needs "did
108// this web-login attempt end", add it then, with that caller in hand.
109
110impl fmt::Display for WebLoginFailure {
111    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
112        match self {
113            Self::LoginRejected => f.write_str("the account name or password was rejected"),
114            Self::NoSubscription => {
115                f.write_str("this account has no active subscription on that instance")
116            }
117            Self::CharacterNotFound => {
118                f.write_str("no character by that name is on that instance's character list")
119            }
120            Self::UnsupportedGameCode => {
121                f.write_str("web login has no confirmed route for that game code")
122            }
123            Self::UnexpectedResponse(where_) => {
124                write!(f, "unexpected response from play.net at {where_}")
125            }
126            Self::Transport(detail) => write!(f, "could not reach play.net: {detail}"),
127        }
128    }
129}

1//! The recorder: what a session did, in enough detail to run it again.
2//!
3//! A whole session is **recorded and replays deterministically** in a test,
4//! with no network.
5//!
6//! # What is recorded: BYTES, not frames
7//!
8//! Recording frames would make the replay test tautological. It would feed the
9//! parser's own output back through the parser, so a parser regression would
10//! change the recording and the replay in lockstep and the test would stay
11//! green. Recording the bytes means the replay re-parses, and a parser change
12//! that alters the frame stream is visible.
13//!
14//! # `seq`, not a timestamp
15//!
16//! A wall clock is the single easiest way to make a replay non-deterministic:
17//! two runs of the same recording produce two different files. `seq` is a
18//! monotonic counter seeded at 0, so a replay of a replay is byte-identical.
19//!
20//! # Chunk boundaries are preserved deliberately
21//!
22//! An `Inbound` entry holds exactly the bytes one `read` returned, not a
23//! re-chunked stream. Replaying the same split that happened live is what
24//! drives `Parser::push_bytes`'s partial-line path with a real boundary rather
25//! than an invented one -- the failure `push_bytes` exists to prevent.
26//!
27//! # On `bytes: Vec<u8>`
28//!
29//! Nothing **above** `hydra-protocol` sees raw wire bytes. That is not broken
30//! here: `hydra-net` is *below* `hydra-protocol`, and a transport that could
31//! not name a byte could not be a transport. Nothing above protocol ever
32//! receives one of these -- the session sees `Frame`s, and the recorder it
33//! owns is fed by the same code that feeds the parser.
34
35/// One thing that crossed the session boundary.
36#[derive(Clone, Debug, PartialEq, Eq)]
37pub enum RecordedEvent {
38    /// Bytes read from the wire, exactly as one `read` returned them.
39    Inbound {
40        /// Position in the recorder's counter, shared by both directions and
41        /// seeded at 0; not a timestamp.
42        seq: u64,
43        /// The chunk, unsplit and unjoined.
44        bytes: Vec<u8>,
45    },
46    /// Bytes written to the wire, exactly as one `write_all` sent them.
47    Outbound {
48        /// Position in the recorder's counter, shared by both directions and
49        /// seeded at 0; not a timestamp.
50        seq: u64,
51        /// The message as written in one call, its trailing newline included.
52        bytes: Vec<u8>,
53    },
54}
55
56impl RecordedEvent {
57    /// The bytes, whichever direction this event went.
58    ///
59    /// For the size accounting the byte bound needs; a caller that cares about
60    /// direction should match instead.
61    #[must_use]
62    pub fn bytes(&self) -> &[u8] {
63        match self {
64            Self::Inbound { bytes, .. } | Self::Outbound { bytes, .. } => bytes,
65        }
66    }
67}
68
69/// How many bytes of wire traffic a recorder keeps before dropping the oldest.
70///
71/// # Why this is bounded at all
72///
73/// Unbounded, it would be a leak on the production path: an append-only `Vec`
74/// that **copies every chunk**, carried across every generation, in a client
75/// designed for 3-25 simultaneous characters. A long session would hold every
76/// byte it had ever read, in memory, for the life of the process.
77///
78/// The `.bytes` sink has already written all of it to disk, so the in-memory
79/// copy buys nothing a file cannot answer -- which is what makes bounding it
80/// safe rather than a trade.
81///
82/// # Why 8 MiB, and what it costs
83///
84/// Measured: live sessions log 64-80 KB of inbound traffic per few minutes, so
85/// 8 MiB is hours of play per session and 200 MiB across a full 25-character
86/// load. That is a ceiling rather than a target: sessions that stay under it
87/// lose nothing, and replay tests use fixtures measured in kilobytes.
88///
89/// What it costs is that a session **longer than the bound cannot be replayed
90/// from its start** -- [`Recorder::events`] no longer begins at `seq` 0. That is
91/// detectable rather than silent: `seq` is independent of the `Vec`, so a
92/// consumer can see the first surviving sequence number and know what it is
93/// missing, and [`Recorder::dropped`] says how many.
94pub const MAX_RECORDED_BYTES: usize = 8 * 1024 * 1024;
95
96/// Bounded log of one session's wire traffic, oldest dropped first.
97///
98/// Held by value inside the session actor -- no process globals -- so 25
99/// sessions record 25 independent transcripts.
100#[derive(Debug, Default)]
101pub struct Recorder {
102    events: std::collections::VecDeque<RecordedEvent>,
103    next_seq: u64,
104    /// Bytes currently held, so trimming costs no rescan.
105    held_bytes: usize,
106    /// How many events were dropped to stay under [`MAX_RECORDED_BYTES`].
107    dropped: u64,
108    /// Lifetime count of outbound writes, unaffected by trimming.
109    outbound_count: u64,
110    /// Lifetime count of inbound reads, unaffected by trimming.
111    inbound_count: u64,
112}
113
114impl Recorder {
115    /// A recorder with an empty log and `seq` at 0.
116    #[must_use]
117    pub fn new() -> Self {
118        Self::default()
119    }
120
121    /// Record bytes that arrived from the wire.
122    pub fn inbound(&mut self, bytes: &[u8]) {
123        let seq = self.take_seq();
124        self.held_bytes += bytes.len();
125        self.inbound_count += 1;
126        self.events.push_back(RecordedEvent::Inbound {
127            seq,
128            bytes: bytes.to_vec(),
129        });
130        self.trim();
131    }
132
133    /// Record bytes that were sent to the wire.
134    pub fn outbound(&mut self, bytes: &[u8]) {
135        let seq = self.take_seq();
136        self.held_bytes += bytes.len();
137        self.outbound_count += 1;
138        self.events.push_back(RecordedEvent::Outbound {
139            seq,
140            bytes: bytes.to_vec(),
141        });
142        self.trim();
143    }
144
145    /// Drop the oldest events until the log fits [`MAX_RECORDED_BYTES`].
146    ///
147    /// **Never drops the last event**, however large: a recorder holding one
148    /// 10 MiB chunk keeps it, because "the most recent thing that happened" is
149    /// the one a caller is most likely to be about to read.
150    fn trim(&mut self) {
151        while self.held_bytes > MAX_RECORDED_BYTES && self.events.len() > 1 {
152            let Some(oldest) = self.events.pop_front() else {
153                break;
154            };
155            self.held_bytes = self.held_bytes.saturating_sub(oldest.bytes().len());
156            self.dropped += 1;
157        }
158    }
159
160    /// Everything still recorded, in the order it happened.
161    ///
162    /// **May not start at `seq` 0** on a session longer than
163    /// [`MAX_RECORDED_BYTES`] -- see [`Self::dropped`]. The `.bytes` sink is the
164    /// complete record; this is the recent window.
165    /// Returns a **slice**, which is why this takes `&mut self`:
166    /// `VecDeque::make_contiguous` is what lets a bounded ring keep the flat
167    /// `&[RecordedEvent]` its callers use, and a recorder is read a handful of
168    /// times per session, so one rotation costs nothing.
169    pub fn events(&mut self) -> &[RecordedEvent] {
170        self.events.make_contiguous()
171    }
172
173    /// How many events are still held. Does **not** need `&mut`, unlike
174    /// [`Self::events`], so a caller that only wants the count (a log line, a
175    /// report) is not forced to take a mutable borrow.
176    #[must_use]
177    pub fn len(&self) -> usize {
178        self.events.len()
179    }
180
181    /// Whether anything is held.
182    #[must_use]
183    pub fn is_empty(&self) -> bool {
184        self.events.is_empty()
185    }
186
187    /// How many inbound reads this recorder has seen, **including any it has
188    /// since dropped**.
189    ///
190    /// Counterpart to [`Self::outbound_count`], and for the same reason: the
191    /// supervisor needs "did this connection RECEIVE anything" to tell a login
192    /// that worked from one that opened a socket and died, and a bounded log
193    /// cannot answer that by scanning.
194    #[must_use]
195    pub const fn inbound_count(&self) -> u64 {
196        self.inbound_count
197    }
198
199    /// How many outbound writes this recorder has seen, **including any it has
200    /// since dropped**.
201    ///
202    /// A counter rather than a scan. The supervisor uses this to decide whether a
203    /// connection was "attended"; counting the log would be O(session length) on
204    /// a hot path, and wrong, because a bounded log forgets its early writes.
205    #[must_use]
206    pub const fn outbound_count(&self) -> u64 {
207        self.outbound_count
208    }
209
210    /// How many events were dropped to stay under [`MAX_RECORDED_BYTES`].
211    ///
212    /// Nonzero means [`Self::events`] is a window rather than the whole session,
213    /// which a replay built from it needs to know. Zero for every session short
214    /// enough to fit, which is every test fixture.
215    #[must_use]
216    pub const fn dropped(&self) -> u64 {
217        self.dropped
218    }
219
220    /// Bytes currently held. For a diagnostic, and for the bound's own test.
221    #[must_use]
222    pub const fn held_bytes(&self) -> usize {
223        self.held_bytes
224    }
225
226    /// The inbound chunks alone, in order, ready to build a
227    /// [`ReplaySource`](crate::replay::ReplaySource) that reproduces this
228    /// session's reads **with the same chunk boundaries**.
229    #[must_use]
230    pub fn inbound_chunks(&self) -> Vec<Vec<u8>> {
231        self.events
232            .iter()
233            .filter_map(|e| match e {
234                RecordedEvent::Inbound { bytes, .. } => Some(bytes.clone()),
235                RecordedEvent::Outbound { .. } => None,
236            })
237            .collect()
238    }
239
240    fn take_seq(&mut self) -> u64 {
241        let seq = self.next_seq;
242        self.next_seq += 1;
243        seq
244    }
245}

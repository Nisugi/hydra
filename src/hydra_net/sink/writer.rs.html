19/// Two files for one session: the raw wire, and what we did.
20///
21/// Holds its own handles rather than reaching a global: there are no process
22/// globals, and 25 sessions must write 25 independent transcripts without
23/// contending on one logger.
24#[derive(Debug)]
25pub struct SessionSink {
26    bytes: BufWriter<File>,
27    events: BufWriter<File>,
28    redactions: Redactions,
29    bytes_path: PathBuf,
30    events_path: PathBuf,
31    lines_written: u64,
32    /// `<dir>/<char>-<stamp>`, shared by every part of this session.
33    stem: PathBuf,
34    /// Which part is open. 0 is the first file.
35    part: u32,
36    /// Lines before rolling. Read once at creation, for the same reason
37    /// `stamp_bytes` is: a threshold that changed mid-session would produce
38    /// parts of inconsistent size.
39    rotate_after: u64,
40    /// The last inbound byte redacted: what the next inbound bytes follow,
41    /// so a name at their start is judged with the byte before it.
42    inbound_before: Option<u8>,
43    /// Inbound bytes held back so a secret cannot hide in a chunk boundary.
44    ///
45    /// Bounded by `longest_secret - 1`, and empty whenever nothing is
46    /// registered. See `redact_across_chunks`; `drain_pending` is what
47    /// guarantees these bytes still reach the file.
48    pending: Vec<u8>,
49    /// Whether the last inbound byte seen left a line unfinished.
50    line_open: bool,
51    /// Framed commands waiting for that line to end. See [`Self::wire`].
52    awaiting_line_end: Vec<Vec<u8>>,
53    /// Whether the bytes file carries per-line times. Read once at creation
54    /// rather than per line: an env var that changed mid-session would produce
55    /// a file that is half one format.
56    stamp_bytes: bool,
57    /// Whether the next byte written to the bytes file begins a line: the
58    /// last written was a newline, or nothing has been. A stamp goes only
59    /// there: stamped before every inbound chunk, a read boundary inside a tag
60    /// would put `HH:MM:SS.mmm: ` between `id=` and `'room'`.
61    at_line_start: bool,
62}
63
64/// Flush on drop, so a sink that is simply dropped still leaves a complete
65/// file.
66///
67/// `BufWriter` already does this for its own buffer, but the straddle tail
68/// (`redact_across_chunks`) is OURS: bytes the wire really carried, held back
69/// deliberately. Losing them would make the `.bytes` file silently short --
70/// exactly the failure mode the file exists to prevent, since it is what
71/// fixtures are cut from.
72///
73/// Errors are swallowed, as they must be in `Drop`. Callers that need to KNOW
74/// the write succeeded call [`SessionSink::flush`], which returns the error.
75impl Drop for SessionSink {
76    fn drop(&mut self) {
77        let _ = self.drain_pending();
78        let _ = self.bytes.flush();
79        let _ = self.events.flush();
80    }
81}

84    /// Create both files under `dir`, named for the character and a stamp.
85    ///
86    /// The stamp is the caller's, not read from a clock here: this crate's
87    /// [`record`](crate::record) module explains why a wall clock inside the
88    /// recording path is the easiest way to make a replay non-deterministic.
89    /// The same reasoning applies to a filename a test might assert on.
90    ///
91    /// # Errors
92    ///
93    /// Any failure to create the directory or either file.
94    pub fn create(
95        dir: &Path,
96        character: &str,
97        stamp: &str,
98        redactions: Redactions,
99    ) -> io::Result<Self> {
100        Self::create_with_rotation(dir, character, stamp, redactions, rotate_after_lines())
101    }

103    /// [`Self::create`], with the rotation threshold given rather than read
104    /// from the environment.
105    ///
106    /// Exists because a test must be able to roll a file without setting a
107    /// process-wide environment variable: `unsafe_code = "deny"` makes
108    /// `set_var` unavailable (it is `unsafe` since Rust 2024), and a test that
109    /// mutated global state would race every other test in the binary anyway.
110    ///
111    /// Taking the value is also simply better design -- the threshold is a
112    /// property of this sink, and reading it from the environment at the point
113    /// of use would be a hidden input.
114    ///
115    /// # Errors
116    ///
117    /// Any failure to create the directory or either file.
118    pub fn create_with_rotation(
119        dir: &Path,
120        character: &str,
121        stamp: &str,
122        redactions: Redactions,
123        rotate_after: u64,
124    ) -> io::Result<Self> {
125        fs::create_dir_all(dir)?;
126        // The character name is a filename component, so it must not be able
127        // to escape the directory or name a device. Characters are
128        // alphabetical in practice, with `_` between a game and a name;
129        // anything else is dropped rather than substituted, so two names
130        // cannot collide through substitution.
131        let safe: String = character
132            .chars()
133            .filter(|c| c.is_ascii_alphanumeric() || *c == '_')
134            .collect();
135        let safe = if safe.is_empty() {
136            "session".to_owned()
137        } else {
138            safe
139        };
140
141        // `<char>-<stamp>` is the stem every part of this session shares, and
142        // **every part is numbered, including the first**.
143        //
144        // An unnumbered first part -- `Tester-stamp.bytes`, then
145        // `Tester-stamp-001.bytes` -- sorts WRONG: lexicographically `-`
146        // (0x2D) precedes `.` (0x2E), so it would sort AFTER every numbered
147        // one, and reading a directory in name order would replay the session
148        // out of order. The reassembly assertion in `sink_redaction.rs` pins
149        // this.
150        //
151        // Files are made new, never opened over: a second recording with the
152        // same stem -- begun the same second -- takes the next, `_2`.
153        let (safe_stem, bytes_path, events_path, bytes, events) = {
154            let mut n = 1u32;
155            loop {
156                let stem = if n == 1 {
157                    format!("{safe}-{stamp}")
158                } else {
159                    format!("{safe}-{stamp}_{n}")
160                };
161                let bytes_path = dir.join(format!("{stem}-000.bytes"));
162                let events_path = dir.join(format!("{stem}.log"));
163                let new = |path: &Path| File::options().write(true).create_new(true).open(path);
164                match new(&bytes_path) {
165                    Ok(bytes) => {
166                        let events = new(&events_path)?;
167                        break (stem, bytes_path, events_path, bytes, events);
168                    }
169                    Err(e) if e.kind() == io::ErrorKind::AlreadyExists && n < 100 => n += 1,
170                    Err(e) => return Err(e),
171                }
172            }
173        };
174        let bytes = BufWriter::new(bytes);
175        let mut events = BufWriter::new(events);
176
177        // Say what this file is and what it is not, in the file itself. A log
178        // read six months from now must not have to guess whether it was
179        // scrubbed.
180        // The set GROWS after this line: the launch key is minted by the
181        // connect and registered afterwards (`redact_key`), because it does not
182        // exist when the file is created. So this header must describe what was
183        // registered AT CREATION and point at the in-band marker for the rest.
184        //
185        // A header claiming "nothing was registered, treat this file as raw"
186        // whenever the creation-time set is empty would be wrong for EVERY
187        // live session, since the key arrives later: the body of that same
188        // file would then show `redaction registered (session key)`, and the
189        // header would contradict its own contents in the dangerous direction.
190        writeln!(
191            events,
192            "# Hydra session log -- PRIVATE DEVELOPMENT LOG, not automatically shareable.\n\
193             # Credentials registered BEFORE the first byte: {}.\n\
194             # The session key is registered LATER, when the connect mints it --\n\
195             # look for `redaction registered (session key)` below. Everything\n\
196             # before that line predates its redaction.\n\
197             # OTHER PLAYERS' NAMES ARE NOT REDACTED. Cut fixtures through\n\
198             # hydra_protocol::scrub::Scrubber with the names named.",
199            if redactions.is_empty() {
200                "none -- no account was registered".to_owned()
201            } else {
202                redactions.kinds()
203            }
204        )?;
205
206        Ok(Self {
207            bytes,
208            events,
209            redactions,
210            bytes_path,
211            events_path,
212            lines_written: 0,
213            inbound_before: None,
214            pending: Vec::new(),
215            line_open: false,
216            awaiting_line_end: Vec::new(),
217            stamp_bytes: bytes_timestamps_enabled(),
218            at_line_start: true,
219            stem: dir.join(safe_stem),
220            part: 0,
221            rotate_after,
222        })
223    }

225    /// Append raw wire bytes, redacted, marking the direction.
226    ///
227    /// **The framing is the usual `GemStone` XML log's, not a new one.** Inbound
228    /// bytes are written verbatim; client input is wrapped in `<!-- CLIENT -->` /
229    /// `<!-- ENDCLIENT -->`. No sequence number: that format carries none, and
230    /// [`Recorder`](crate::record::Recorder) already owns ordering, so writing
231    /// one here would be a second source of truth for the same fact.
232    ///
233    /// Chunk boundaries are preserved because
234    /// [`Recorder`](crate::record::Recorder)'s own docs explain that replaying
235    /// the split that happened live is what drives `Parser::push_bytes`'s
236    /// partial-line path with a real boundary rather than an invented one.
237    ///
238    /// # Errors
239    ///
240    /// Any write failure.
241    pub fn wire(&mut self, inbound: bool, bytes: &[u8]) -> io::Result<()> {
242        if inbound {
243            // `false` means every byte is still held back waiting for more:
244            // nothing was written, so there is no line to count.
245            if !self.inbound(bytes)? {
246                return Ok(());
247            }
248        } else {
249            let record = self.client_record(bytes);
250            // **A command waits for the inbound line in progress to end.**
251            //
252            // The server's bytes before the command reached this file split
253            // wherever a TCP read ended -- or wherever the redaction hold cut
254            // them, which with any secret registered is every chunk. Writing
255            // the command at once would splice it into the middle of a line,
256            // even a tag:
257            //
258            // ```text
259            // <prompt ti<!-- CLIENT -->look<!-- ENDCLIENT -->
260            // me="1758600000">&gt;</prompt>
261            // ```
262            //
263            // That is not a parseable file, and this is the file fixtures are
264            // cut from. Interleaving at the next line end keeps every line
265            // whole. The command moves by at most the rest of one line, and
266            // `Recorder` keeps the exact order for replay.
267            //
268            // It also makes the flush below safe for redaction: the held-back
269            // tail ends in a newline, and no secret spans one.
270            if self.line_open {
271                self.awaiting_line_end.push(record);
272            } else {
273                self.drain_pending()?;
274                self.write_bytes(&record)?;
275            }
276        }
277        // **NOTHING is appended to inbound bytes**, and that fidelity is this
278        // file's whole purpose. Adding a newline to a chunk that did not end in
279        // one would split a tag straddling a read: a boundary inside
280        // `<pushStream id='room'/>` would be written with a newline in the
281        // middle of the attribute, so a replay would parse something the live
282        // session never saw.
283        //
284        // The `.bytes` file is what golden fixtures are cut from and what
285        // replay tests replay, so a byte invented here is inherited by every
286        // future fixture.
287        self.lines_written += 1;
288        if self.lines_written >= self.rotate_after {
289            self.roll()?;
290        }
291        Ok(())
292    }

303    /// Write inbound bytes, releasing any command that was waiting for the
304    /// line they finish. Returns whether anything reached the file.
305    fn inbound(&mut self, bytes: &[u8]) -> io::Result<bool> {
306        let Some(last) = bytes.last() else {
307            return Ok(false);
308        };
309        let mut wrote = false;
310        let mut rest = bytes;
311        if !self.awaiting_line_end.is_empty()
312            && let Some(at) = rest.iter().position(|&b| b == b'\n')
313        {
314            let (line_end, after) = rest.split_at(at + 1);
315            self.inbound_chunk(line_end)?;
316            // The tail now ends in a newline, so draining it cannot cut a
317            // secret in two -- and the drain releases the waiting commands.
318            self.drain_pending()?;
319            wrote = true;
320            rest = after;
321        }
322        if !rest.is_empty() {
323            wrote |= self.inbound_chunk(rest)?;
324        }
325        self.line_open = *last != b'\n';
326        Ok(wrote)
327    }

329    /// One inbound chunk through the redaction hold.
330    fn inbound_chunk(&mut self, bytes: &[u8]) -> io::Result<bool> {
331        // `None` means every byte is still held back waiting for more: nothing
332        // to write yet, and writing a stamp for it would invent a line.
333        let Some(redacted) = self.redact_across_chunks(bytes) else {
334            return Ok(false);
335        };
336        if self.stamp_bytes {
337            // Each line the chunk begins, stamped where it begins; a line
338            // the chunk continues, not.
339            for piece in redacted.split_inclusive(|byte| *byte == b'\n') {
340                if self.at_line_start {
341                    write!(self.bytes, "{}: ", line_time())?;
342                }
343                self.write_bytes(piece)?;
344            }
345        } else {
346            self.write_bytes(&redacted)?;
347        }
348        Ok(true)
349    }

351    /// `bytes` to the bytes file, noting whether they end a line.
352    fn write_bytes(&mut self, bytes: &[u8]) -> io::Result<()> {
353        self.bytes.write_all(bytes)?;
354        if let Some(last) = bytes.last() {
355            self.at_line_start = *last == b'\n';
356        }
357        Ok(())
358    }

360    /// One outbound command, redacted and framed, ready to write.
361    fn client_record(&self, bytes: &[u8]) -> Vec<u8> {
362        let redacted = self.redactions.apply_bytes(bytes);
363        let mut record = Vec::with_capacity(redacted.len() + 64);
364        if self.stamp_bytes {
365            record.extend_from_slice(format!("{}: ", line_time()).as_bytes());
366        }
367        // Client input is WRAPPED, not prefixed -- the same
368        // `<!-- CLIENT -->...<!-- ENDCLIENT -->` markers GemStone XML logs
369        // conventionally use, so a Hydra capture can be read by anything that
370        // already reads those, and vice versa. A private framing would make
371        // Hydra's logs the odd ones out.
372        record.extend_from_slice(CLIENT_OPEN);
373        record.extend_from_slice(redacted.trim_ascii_end());
374        record.extend_from_slice(CLIENT_CLOSE);
375        // **The wrapper gets its own terminator, unconditionally**, because
376        // the command's own newline was just trimmed off. Without it the
377        // next inbound line is glued to `ENDCLIENT -->` and a reader
378        // splitting on lines sees one line where there were two. (Testing
379        // whether the UNTRIMMED buffer ends in a newline would be wrong: the
380        // trimmed one is what is written.)
381        record.push(b'\n');
382        record
383    }

385    /// Redact `bytes`, carrying a tail forward so a secret cannot hide in a
386    /// chunk boundary.
387    ///
388    /// # The failure this exists for
389    ///
390    /// `apply_bytes` sees one `read` at a time, and a TCP read boundary falls
391    /// wherever the network puts it. A launch key split across two reads would
392    /// match neither half and be **written in the clear** -- the live
393    /// credential, in a file on disk.
394    ///
395    /// # How the tail works, and why it is bounded
396    ///
397    /// Holding back `longest_secret - 1` bytes guarantees that any secret is
398    /// wholly inside some chunk: a secret of length `n` cannot span more than
399    /// `n - 1` bytes of held-back tail plus the new chunk. The tail is bounded
400    /// by the longest registered secret, so it cannot grow with the session.
401    ///
402    /// # What this costs, stated plainly
403    ///
404    /// **Chunk boundaries shift** by up to `longest_secret - 1` bytes while
405    /// secrets are registered, and this function's own doc says boundaries are
406    /// preserved so a replay drives the parser's partial-line path with a real
407    /// split. That guarantee is weakened here, deliberately:
408    ///
409    /// - the boundary is still a boundary the wire *could* have produced -- no
410    ///   byte is invented, reordered or dropped, only deferred;
411    /// - [`Recorder`](crate::record::Recorder) keeps its own in-memory copy and
412    ///   is unaffected, so live replay fidelity is untouched;
413    /// - a credential in a log is a worse outcome than a shifted split in a
414    ///   fixture.
415    ///
416    /// Inbound only. Outbound commands are whole, so there is nothing to
417    /// straddle; they are placed at line ends instead (see [`Self::wire`]).
418    fn redact_across_chunks(&mut self, bytes: &[u8]) -> Option<Vec<u8>> {
419        let hold = self.redactions.longest_secret().saturating_sub(1);
420        if hold == 0 {
421            // Nothing registered: redact in place.
422            return Some(self.redact_inbound(bytes));
423        }
424
425        self.pending.extend_from_slice(bytes);
426        // Keep back the last `hold` bytes: a secret could still be completed by
427        // the next chunk.
428        let mut safe = self.pending.len().saturating_sub(hold);
429        if safe == 0 {
430            return None;
431        }
432
433        // **And do not cut through a secret that is ALREADY whole.** Holding
434        // back `hold` bytes stops an INCOMING boundary from splitting a secret,
435        // but the cut made here is a second boundary, and a naive one lands
436        // mid-secret just as easily: a key whole in `pending` sliced at byte
437        // 17 of 48 leaves neither piece matching. Walk the cut backwards past
438        // any secret that spans it.
439        safe = self.cut_clear_of_secrets(safe);
440        if safe == 0 {
441            return None;
442        }
443        let head: Vec<u8> = self.pending.drain(..safe).collect();
444        Some(self.redact_inbound(&head))
445    }

447    /// Redact the next inbound bytes, after those redacted before them.
448    fn redact_inbound(&mut self, bytes: &[u8]) -> Vec<u8> {
449        let redacted = self
450            .redactions
451            .apply_bytes_after(self.inbound_before, bytes);
452        if let Some(last) = bytes.last() {
453            self.inbound_before = Some(*last);
454        }
455        redacted
456    }

458    /// Move a proposed cut back until no registered secret spans it or ends
459    /// at it.
460    ///
461    /// A secret occupying `[start, start + len)` spans the cut when
462    /// `start < cut <= start + len`. Moving the cut to `start` puts the whole
463    /// secret in the held-back tail, where the next chunk -- or
464    /// `drain_pending` -- redacts it intact. One that ends at the cut is held
465    /// too: a name is replaced only as a whole word, which the byte after it
466    /// decides.
467    ///
468    /// Bounded: only secrets beginning within `longest_secret` bytes before the
469    /// cut can span it, so the scan is over a fixed-size window and the cut
470    /// moves back at most `longest_secret - 1` bytes.
471    fn cut_clear_of_secrets(&self, cut: usize) -> usize {
472        let window = self.redactions.longest_secret();
473        let from = cut.saturating_sub(window);
474        let mut earliest = cut;
475        for (secret, _) in self.redactions.secrets() {
476            let needle = secret.as_bytes();
477            for start in from..cut {
478                if start + needle.len() >= cut
479                    && self.pending.len() >= start + needle.len()
480                    && self.pending[start..].starts_with(needle)
481                {
482                    earliest = earliest.min(start);
483                }
484            }
485        }
486        earliest
487    }

489    /// Write out whatever the straddle tail is still holding.
490    ///
491    /// Called before a flush, a roll and a close: a tail left in memory is
492    /// bytes the wire really carried, and losing them would make the `.bytes`
493    /// file an incomplete record -- the opposite of its purpose. It is redacted
494    /// on the way out like any other chunk.
495    ///
496    /// # Errors
497    ///
498    /// Any write failure.
499    ///
500    /// Commands still waiting for a line end go out after it: a session that
501    /// ends mid-line must not lose the commands it sent.
502    fn drain_pending(&mut self) -> io::Result<()> {
503        if !self.pending.is_empty() {
504            let tail: Vec<u8> = std::mem::take(&mut self.pending);
505            let redacted = self.redact_inbound(&tail);
506            self.write_bytes(&redacted)?;
507        }
508        for record in std::mem::take(&mut self.awaiting_line_end) {
509            self.write_bytes(&record)?;
510        }
511        Ok(())
512    }

514    /// Close the current `.bytes` part and open the next.
515    ///
516    /// Only the bytes file rolls. The debug log takes a handful of lines per
517    /// session (`lifecycle Ready`, `lifecycle Closed`, errors), so rolling it
518    /// would produce empty parts and a second thing to reassemble for no gain.
519    ///
520    /// # Each part is INDEPENDENTLY VALID, and that is the point
521    ///
522    /// A part boundary falls between two whole chunks, never inside one, so
523    /// every part starts on a chunk boundary the wire really had. That is what
524    /// keeps a rolled session usable as replay input: concatenating the parts
525    /// in order reproduces the original stream exactly, and a single part
526    /// parses on its own (`Parser::push_bytes` buffers to a newline, so a part
527    /// that begins mid-tag is the one thing this must not produce).
528    ///
529    /// # Errors
530    ///
531    /// Any failure to flush the old part or create the new one. The caller
532    /// swallows it -- a session must survive a failed log --
533    /// but it is returned rather than hidden so the caller *can* report it.
534    fn roll(&mut self) -> io::Result<()> {
535        // **The pending suffix is NOT drained here.**
536        //
537        // Those bytes arrived while this part was open, so draining them into
538        // it is tempting, and it defeats the buffer's whole purpose: `pending`
539        // holds back a suffix precisely because it might be the FIRST HALF of
540        // a registered secret. With `SECRET` registered and a rotation between
541        // a chunk ending `SEC` and one starting `RET`, both fragments would be
542        // written unredacted, one per part.
543        //
544        // Carrying it across costs a little ordering: at most `hold` bytes
545        // move from the end of one part to the start of the next. That is the
546        // right trade. A part boundary that falls a few bytes early is a
547        // cosmetic imprecision in a log; a secret written in clear is not
548        // recoverable, and the whole point of this file's redaction is that
549        // it never happens.
550        //
551        // Reassembly is unaffected: concatenating the parts in order still
552        // reproduces the stream exactly, because the bytes are not lost, only
553        // deferred. The independently-valid-part claim above is the one that
554        // weakens, and only by `hold` bytes at the seam.
555        self.bytes.flush()?;
556        // `part` is committed only AFTER the file exists. Incrementing first
557        // and then using `?` on `File::create` spends the number on a part
558        // that was never opened: the next roll takes the one after it, and the
559        // sequence has a permanent hole. A reader of a capture directory
560        // cannot then tell a transient ENOSPC from a part that was written and
561        // later deleted -- and for a capture format whose whole claim is that
562        // it is the wire verbatim, "a file is missing here" must mean exactly
563        // one thing.
564        let next_part = self.part + 1;
565        let next = self.stem.with_file_name(format!(
566            "{}-{:03}.bytes",
567            self.stem.file_name().unwrap_or_default().to_string_lossy(),
568            next_part
569        ));
570        self.bytes = BufWriter::new(File::create(&next)?);
571        self.part = next_part;
572        self.bytes_path = next;
573        self.lines_written = 0;
574        Ok(())
575    }

577    /// Append one debug line: a lifecycle change, an error -- something that
578    /// is **not** on the wire.
579    ///
580    /// # Do not log wire content here
581    ///
582    /// A command, a frame or a line of game text is already in the `.bytes`
583    /// file, byte-for-byte as the server saw it. Writing it again here makes
584    /// two records of one fact that can disagree, and doubles the disk to do
585    /// it. The `.bytes` file is the whole point.
586    ///
587    /// The readable timestamped view of a session is the **user log**, a
588    /// different artifact above the parser. This is not a down-payment on it.
589    ///
590    /// # Errors
591    ///
592    /// Any write failure.
593    pub fn event(&mut self, line: &str) -> io::Result<()> {
594        // Always stamped: nothing replays this file, so a wall clock costs it
595        // nothing, and "when did that happen" is the first question anyone
596        // asks of it.
597        writeln!(
598            self.events,
599            "{} {}",
600            line_time(),
601            self.redactions.apply(line)
602        )
603    }
604
605    /// Register another launch key for redaction.
606    ///
607    /// # Why a sink outlives the secret it was opened with
608    ///
609    /// **One log spans every generation**, and a reconnect is a full re-login
610    /// (the SGE connection is strictly single-use per authentication), so
611    /// **every connection has a different launch key**. A sink
612    /// whose redaction set was fixed at creation would write generation 2's
613    /// key verbatim into a file opened during generation 1 -- the one secret
614    /// the `.bytes` file would otherwise carry in the clear, unredacted,
615    /// because it did not exist yet when the set was built.
616    ///
617    /// That is why this is narrow rather than a general `redactions_mut`: the
618    /// launch key is the only secret that is genuinely per-connection. The
619    /// account is typed once and registered at creation (the character is
620    /// not registered at all: it is display text on every line).
621    ///
622    /// Keys accumulate. An earlier generation's key stays redacted, because
623    /// the log still contains the bytes it appeared in.
624    pub fn redact_key(&mut self, key: &str) {
625        self.redactions.key(key);
626        // **Recorded in the file, not just applied.** The header is written
627        // once at creation and cannot describe a set that grows afterwards, so
628        // a reader needs an in-band marker to tell which part of a log a
629        // redaction covers. Everything above this line predates it.
630        //
631        // The KEY IS NOT LOGGED, obviously -- only the fact that one was
632        // registered. It is applied to this very line, so a bug that wrote the
633        // secret here would redact it on the way out.
634        let _ = self.event("redaction registered (session key)");
635    }
636
637    /// Flush both files.
638    ///
639    /// Called at shutdown, and worth calling on a lifecycle change: a session
640    /// that panics should still leave the log that explains why.
641    ///
642    /// # Errors
643    ///
644    /// Any flush failure.
645    pub fn flush(&mut self) -> io::Result<()> {
646        // A flush that left the straddle tail in memory would be a lie: the
647        // caller flushes to make the file complete, and a session that panics
648        // should still leave the log explaining why.
649        self.drain_pending()?;
650        self.bytes.flush()?;
651        self.events.flush()
652    }

654    /// Where the raw wire went.
655    #[must_use]
656    pub fn bytes_path(&self) -> &Path {
657        &self.bytes_path
658    }
659
660    /// Where the structured log went.
661    #[must_use]
662    pub fn events_path(&self) -> &Path {
663        &self.events_path
664    }

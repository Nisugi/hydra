1//! The ocean a ship sails: its rooms, each on a cell of one chart, the open
2//! water between them, and the ports.
3//!
4//! The game numbers ocean rooms as it numbers land rooms, and shows the
5//! number wherever it shows the room: on departing, on `LOOK OCEAN`, on
6//! `PEER`, and on every move while the ocean is watched. The rooms are kept
7//! here, in `data/ocean.tsv`, and not in the map: the ocean is a plain grid,
8//! each room one step from its neighbours, so a room's cell is a fact of
9//! the room and nothing has to lay it out.
10//!
11//! # A port is one room of the ocean and many of the land
12//!
13//! A ship lies at a pier, a land room, and is boarded from it. Departing
14//! puts it in its port's one ocean room; turning the wheel to the port
15//! there brings it back to whichever pier has space. So each pier is tied
16//! to its port ([`Ocean::port_of_pier`]), many to one.
17//!
18//! # Where the grid does not hold
19//!
20//! A few headings do not **run true**: one leads somewhere other than the
21//! next cell that way, or cannot be sailed back ([`Ocean::runs_true`]).
22//! And a few rooms are **odd**: the game's own exits there are not what
23//! its chart shows, and a ship is not sailed into one. The table marks
24//! those.
25

33/// A heading a ship can take, in the table's column order.
34#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
35pub enum Heading {
36    /// North.
37    North,
38    /// Northeast.
39    Northeast,
40    /// East.
41    East,
42    /// Southeast.
43    Southeast,
44    /// South.
45    South,
46    /// Southwest.
47    Southwest,
48    /// West.
49    West,
50    /// Northwest.
51    Northwest,
52}
53
54impl Heading {
55    /// Every heading, clockwise from north.
56    pub const ALL: [Self; 8] = [
57        Self::North,
58        Self::Northeast,
59        Self::East,
60        Self::Southeast,
61        Self::South,
62        Self::Southwest,
63        Self::West,
64        Self::Northwest,
65    ];
66
67    /// The heading the game names `word`: `north`, `Northeast`, `sw`.
68    #[must_use]
69    pub fn named(word: &str) -> Option<Self> {
70        Some(match word.trim().to_ascii_lowercase().as_str() {
71            "north" | "n" => Self::North,
72            "northeast" | "ne" => Self::Northeast,
73            "east" | "e" => Self::East,
74            "southeast" | "se" => Self::Southeast,
75            "south" | "s" => Self::South,
76            "southwest" | "sw" => Self::Southwest,
77            "west" | "w" => Self::West,
78            "northwest" | "nw" => Self::Northwest,
79            _ => return None,
80        })
81    }
82
83    /// One step this way on the chart: columns grow east, rows grow south.
84    #[must_use]
85    pub const fn step(self) -> (i32, i32) {
86        match self {
87            Self::North => (0, -1),
88            Self::Northeast => (1, -1),
89            Self::East => (1, 0),
90            Self::Southeast => (1, 1),
91            Self::South => (0, 1),
92            Self::Southwest => (-1, 1),
93            Self::West => (-1, 0),
94            Self::Northwest => (-1, -1),
95        }
96    }
97
98    /// The heading in a word, as the wheel is turned to it: `northwest`.
99    #[must_use]
100    pub const fn word(self) -> &'static str {
101        match self {
102            Self::North => "north",
103            Self::Northeast => "northeast",
104            Self::East => "east",
105            Self::Southeast => "southeast",
106            Self::South => "south",
107            Self::Southwest => "southwest",
108            Self::West => "west",
109            Self::Northwest => "northwest",
110        }
111    }
112
113    /// The heading back.
114    #[must_use]
115    pub const fn back(self) -> Self {
116        Self::ALL[(self as usize + 4) % 8]
117    }
118}

120/// One room of the ocean.
121#[derive(Clone, Debug, PartialEq, Eq)]
122pub struct OceanRoom {
123    /// The game's number for the room.
124    pub uid: i64,
125    /// The ocean the game titles the room by: `The Great Western Sea`.
126    pub ocean: String,
127    /// The sea, as the captain's map names it: `Kestrel Sea`.
128    pub sea: String,
129    /// Its cell on the chart: column, then row.
130    pub cell: (i32, i32),
131    /// The room each heading leads to, in [`Heading::ALL`]'s order.
132    pub open: [Option<i64>; 8],
133    /// The game's exits here are not what its chart shows: a room a ship
134    /// is not sailed into.
135    pub odd: bool,
136}
137
138impl OceanRoom {
139    /// The room `heading` leads to.
140    #[must_use]
141    pub const fn toward(&self, heading: Heading) -> Option<i64> {
142        self.open[heading as usize]
143    }
144}

160pub struct Port {

172/// The ocean.
173#[derive(Clone, Debug, Default)]
174pub struct Ocean {
175    rooms: Vec<OceanRoom>,
176    by_uid: HashMap<i64, usize>,
177    by_cell: HashMap<(i32, i32), usize>,
178    ports: Vec<Port>,
179    /// Each pier's land room, and the port it belongs to.
180    piers: HashMap<i64, usize>,
181    /// Each pier's title, and the port it belongs to.
182    pier_titles: HashMap<String, usize>,
183    /// Each pier's land room, and its title.
184    pier_rooms: Vec<(i64, String)>,
185}
186
187impl Ocean {
188    /// The ocean the tables hold: rooms, ports, and piers.
189    ///
190    /// # Errors
191    ///
192    /// A row is short, or a number in it is not one.
193    pub fn read(rooms: &str, ports: &str, piers: &str) -> Result<Self, String> {
194        let mut ocean = Self::default();
195        for (at, row) in rows(rooms) {
196            let room = room_of(&row).ok_or_else(|| format!("ocean room, line {at}: {row:?}"))?;
197            ocean.by_uid.insert(room.uid, ocean.rooms.len());
198            ocean.by_cell.insert(room.cell, ocean.rooms.len());
199            ocean.rooms.push(room);
200        }
201        for (at, row) in rows(ports) {
202            let port = port_of(&row).ok_or_else(|| format!("ocean port, line {at}: {row:?}"))?;
203            ocean.ports.push(port);
204        }
205        for (at, row) in rows(piers) {
206            let pier = row.first().and_then(|uid| uid.parse().ok());
207            let port = row
208                .get(1)
209                .and_then(|name| ocean.ports.iter().position(|port| port.name == *name));
210            let (Some(pier), Some(port)) = (pier, port) else {
211                return Err(format!("ocean pier, line {at}: {row:?}"));
212            };
213            ocean.piers.insert(pier, port);
214            if let Some(title) = row.get(2) {
215                ocean.pier_titles.insert((*title).to_owned(), port);
216                ocean.pier_rooms.push((pier, (*title).to_owned()));
217            }
218        }
219        Ok(ocean)
220    }
221
222    /// Every room.
223    #[must_use]
224    pub fn rooms(&self) -> &[OceanRoom] {
225        &self.rooms
226    }
227
228    /// The room the game numbers `uid`.
229    #[must_use]
230    pub fn room(&self, uid: i64) -> Option<&OceanRoom> {
231        self.by_uid.get(&uid).and_then(|&at| self.rooms.get(at))
232    }
233
234    /// The room on `cell`.
235    #[must_use]
236    pub fn at(&self, cell: (i32, i32)) -> Option<&OceanRoom> {
237        self.by_cell.get(&cell).and_then(|&at| self.rooms.get(at))
238    }
239
240    /// Whether `heading` out of `room` runs true: it leads to the room on
241    /// the next cell that way, and that room's heading back leads here.
242    #[must_use]
243    pub fn runs_true(&self, room: &OceanRoom, heading: Heading) -> bool {
244        let (dx, dy) = heading.step();
245        room.toward(heading)
246            .and_then(|uid| self.room(uid))
247            .is_some_and(|there| {
248                there.cell == (room.cell.0 + dx, room.cell.1 + dy)
249                    && there.toward(heading.back()) == Some(room.uid)
250            })
251    }
252
253    /// Every port.
254    #[must_use]
255    pub fn ports(&self) -> &[Port] {
256        &self.ports
257    }
258
259    /// The port the pier `land`, a land room's number, belongs to.
260    #[must_use]
261    pub fn port_of_pier(&self, land: i64) -> Option<&Port> {
262        self.piers.get(&land).and_then(|&at| self.ports.get(at))
263    }
264
265    /// The port the pier titled `title` belongs to: `North 'Haven Quay,
266    /// Green Pier`.
267    #[must_use]
268    pub fn port_of_pier_titled(&self, title: &str) -> Option<&Port> {
269        self.pier_titles
270            .get(title)
271            .and_then(|&at| self.ports.get(at))
272    }
273
274    /// The land rooms of the pier titled `title`, in the game's order: a
275    /// pier is several rooms long, and a ship lies in one of them.
276    #[must_use]
277    pub fn piers_titled(&self, title: &str) -> Vec<i64> {
278        let mut rooms: Vec<i64> = self
279            .pier_rooms
280            .iter()
281            .filter(|(_, known)| known == title)
282            .map(|(room, _)| *room)
283            .collect();
284        rooms.sort_unstable();
285        rooms
286    }

288    /// Where `what` means on the ocean: a port by its name, or a room by its
289    /// number (`7143382`, or `u7143382` as the map writes one). `None` for
290    /// a port not known or a number no room has.
291    #[must_use]
292    pub fn bound(&self, what: &str) -> Option<Bound> {
293        if let Some(port) = self.port_named(what) {
294            return Some(Bound {
295                uid: port.uid,
296                name: port.name.clone(),
297                port: true,
298            });
299        }
300        let number = what.trim();
301        let number = number.strip_prefix('u').unwrap_or(number);
302        let uid: i64 = number.parse().ok()?;
303        self.room(uid).map(|room| Bound {
304            uid,
305            name: room.uid.to_string(),
306            port: false,
307        })
308    }
309
310    /// The port the game describes from the water as `looks`: `diverse`.
311    #[must_use]
312    pub fn port_looking(&self, looks: &str) -> Option<&Port> {
313        self.ports
314            .iter()
315            .find(|port| port.looks.as_deref() == Some(looks))
316    }
317
318    /// The land rooms that are `port`'s piers, in no order.
319    pub fn piers_of<'a>(&'a self, port: &'a Port) -> impl Iterator<Item = i64> + 'a {
320        self.piers
321            .iter()
322            .filter(move |&(_, &at)| {
323                self.ports
324                    .get(at)
325                    .is_some_and(|known| known.uid == port.uid)
326            })
327            .map(|(&land, _)| land)
328    }
329
330    /// The port a player means by `said`: its name, the start of it, or
331    /// the start of a word of it, in any case: `kraken`, `landing`, `grief`.
332    #[must_use]
333    pub fn port_named(&self, said: &str) -> Option<&Port> {
334        let said = said.trim().to_lowercase();
335        if said.is_empty() {
336            return None;
337        }
338        let names = || {
339            self.ports
340                .iter()
341                .map(|port| (port, port.name.to_lowercase()))
342        };
343        names()
344            .find(|(_, name)| *name == said)
345            .or_else(|| names().find(|(_, name)| name.starts_with(&said)))
346            .or_else(|| {
347                names().find(|(_, name)| {
348                    name.split([' ', '\''])
349                        .any(|word| word.len() > 1 && word.starts_with(&said))
350                })
351            })
352            .map(|(port, _)| port)
353    }
354
355    /// The port docked from the room `uid`.
356    #[must_use]
357    pub fn port_at(&self, uid: i64) -> Option<&Port> {
358        self.ports.iter().find(|port| port.uid == uid)
359    }
360}

362/// The ocean built in, read the first time anyone asks.
363pub fn ocean() -> &'static Result<Ocean, String> {
364    static OCEAN: OnceLock<Result<Ocean, String>> = OnceLock::new();
365    OCEAN.get_or_init(|| Ocean::read(OCEAN_TSV, PORTS_TSV, PIERS_TSV))
366}

368/// A table's rows under its heading row, each with its line number.
369fn rows(table: &str) -> impl Iterator<Item = (usize, Vec<&str>)> {
370    table
371        .lines()
372        .enumerate()
373        .skip(1)
374        .filter(|(_, line)| !line.is_empty())
375        .map(|(at, line)| (at + 1, line.split('\t').collect()))
376}

378fn room_of(row: &[&str]) -> Option<OceanRoom> {
379    let [uid, ocean, sea, col, row, rest @ ..] = row else {
380        return None;
381    };
382    let mut open = [None; 8];
383    for (to, cell) in open.iter_mut().zip(rest) {
384        *to = number(cell).ok()?;
385    }
386    Some(OceanRoom {
387        uid: uid.parse().ok()?,
388        ocean: (*ocean).to_owned(),
389        sea: (*sea).to_owned(),
390        cell: (col.parse().ok()?, row.parse().ok()?),
391        open,
392        odd: rest.get(8) == Some(&"odd"),
393    })
394}
395
396fn port_of(row: &[&str]) -> Option<Port> {
397    let [name, uid, rest @ ..] = row else {
398        return None;
399    };
400    let looks = rest.get(1).copied().unwrap_or_default();
401    Some(Port {
402        name: (*name).to_owned(),
403        uid: uid.parse().ok()?,
404        handler: number(rest.first().copied().unwrap_or_default()).ok()?,
405        looks: (!looks.is_empty()).then(|| looks.to_owned()),
406    })
407}

409/// The number in a cell, or nothing from an empty one.
410fn number(cell: &str) -> Result<Option<i64>, std::num::ParseIntError> {
411    if cell.is_empty() {
412        Ok(None)
413    } else {
414        cell.parse().map(Some)
415    }

444    /// Nearly every heading runs true, and none of the few that do not is
445    /// taken for one.
446    #[test]
447    fn the_headings_that_do_not_run_true_are_few() {
448        let ocean = built_in();
449        let (mut all, mut untrue) = (0, 0);
450        for room in ocean.rooms() {
451            for heading in Heading::ALL {
452                if room.toward(heading).is_some() {
453                    all += 1;
454                    if !ocean.runs_true(room, heading) {
455                        untrue += 1;
456                    }
457                }
458            }
459        }
460        assert!(all > 7000, "{all}");
461        assert!(untrue < 50, "{untrue}");
462        // North out of Solhaven's port runs true; a heading with no water
463        // that way does not.
464        let solhaven = ocean.room(7_116_362).expect("a room");
465        assert!(ocean.runs_true(solhaven, North));
466        assert!(!ocean.runs_true(solhaven, South));
467        assert_eq!(ocean.rooms().iter().filter(|room| room.odd).count(), 0);
468    }

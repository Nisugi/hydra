1//! A room of the map in words, for the minimap's Shift+click and Ctrl+click.
2
3use cena_map::{Map, RoomId};
4
5/// A room a find turned up: its number, its title and its area.
6#[derive(Clone, Debug, PartialEq, Eq)]
7pub struct Found {
8    /// The room's number on the map.
9    pub id: u32,
10    /// The room's title.
11    pub title: String,
12    /// The name of the area the room is in.
13    pub area: String,
14}
15
16/// The rooms `text` names, at most `limit`: the room of that number; of
17/// that game number (`u7120`, or digits of seven or more); whose title
18/// holds the words; or that carries the tag. Nothing for nothing typed.
19#[must_use]
20pub fn found(map: &Map, text: &str, limit: usize) -> Vec<Found> {
21    let wanted = text.trim();
22    if wanted.is_empty() {
23        return Vec::new();
24    }
25    let of = |room: &cena_map::Room| Found {
26        id: room.id.0,
27        title: room
28            .title
29            .first()
30            .map(|t| t.trim_matches(['[', ']']).to_owned())
31            .unwrap_or_default(),
32        area: room.area.clone().unwrap_or_default(),
33    };
34    let digits = wanted.strip_prefix(['u', 'U']).unwrap_or(wanted);
35    if let Ok(number) = digits.parse::<i64>() {
36        let mut rooms: Vec<Found> = Vec::new();
37        if !wanted.starts_with(['u', 'U'])
38            && let Ok(id) = u32::try_from(number)
39            && let Some(room) = map.room(RoomId(id))
40        {
41            rooms.push(of(room));
42        }
43        for id in map.ids_for_uid(cena_map::Uid(number)) {
44            if let Some(room) = map.room(*id)
45                && !rooms.iter().any(|found| found.id == id.0)
46            {
47                rooms.push(of(room));
48            }
49        }
50        return rooms;
51    }
52    let lower = wanted.to_lowercase();
53    let mut rooms: Vec<Found> = map
54        .rooms()
55        .iter()
56        .filter(|room| {
57            room.tags.iter().any(|tag| tag.to_lowercase() == lower)
58                || room
59                    .title
60                    .iter()
61                    .any(|title| title.to_lowercase().contains(&lower))
62        })
63        .take(limit)
64        .map(of)
65        .collect();
66    rooms.sort_by_key(|found| found.id);
67    rooms
68}

70/// Room `id` in words: its number and uid, its title and
71/// where it is, what it looks like, and its ways out, a line each; `brief`,
72/// its number and title alone. `None` when the map has no such room.
73#[must_use]
74pub fn described(map: &Map, id: u32, brief: bool) -> Option<Vec<String>> {
75    let room = map.room(RoomId(id))?;
76    let title = room.title.last().map_or("", String::as_str);
77    if brief {
78        return Some(vec![format!("#{id} {title}")]);
79    }
80    let uid = room
81        .uid
82        .last()
83        .map_or_else(String::new, |uid| format!(" (u{})", uid.0));
84    let location = room
85        .location
86        .as_deref()
87        .map_or_else(String::new, |l| format!(" ({l})"));
88    let mut lines = vec![format!("#{id}{uid}:"), format!("{title}{location}")];
89    lines.extend(room.description.last().cloned());
90    lines.extend(room.paths.last().cloned());
91    Some(lines)
92}

94#[cfg(test)]
95mod tests {
96    use super::*;
97
98    /// Rawknuckle's in the map Hydra ships, said in full.
99    #[test]
100    fn a_room_is_said_in_full() {
101        let map = cena_map::binary::decode(cena_gs_map::GS_MAP).expect("decodes");
102        let lines = described(&map, 29877, false).expect("the tavern");
103        assert_eq!(lines[0], "#29877 (u7503251):");
104        assert!(
105            lines[1].starts_with("[Rawknuckle's, Watering Hole]"),
106            "{}",
107            lines[1]
108        );
109        assert!(
110            lines.iter().any(|l| l.starts_with("Obvious exits")),
111            "{lines:?}"
112        );
113        assert_eq!(
114            described(&map, 29877, true).expect("the tavern"),
115            ["#29877 [Rawknuckle's, Watering Hole]"]
116        );
117        assert_eq!(described(&map, u32::MAX, true), None);
118    }

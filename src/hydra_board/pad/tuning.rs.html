1//! How the controller feels: `[controller_tuning]` in the controller file.
2//! Every field is optional, and one not written is Hydra's.
3//!
4//! ```toml
5//! [controller_tuning]
6//! movement_stick = "left"    # the stick that walks: left or right
7//! opposing_stick = "scroll"  # the other, while no wheel is open: scroll or none
8//! back_slice = "down"        # where Back sits in a wheel's folder, or none
9//! deadzone = 50              # % a stick is pushed before a slice is aimed
10//! aim_dwell_ms = 150         # held on a leaf this long, it is committed
11//! nav_dwell_ms = 150         # held on a folder or Back, the wheel goes in or out
12//! fire_debounce_ms = 300     # a fire this soon after the last is not sent
13//! release_grace_ms = 40      # the walking stick stilled this long after a wheel closes
14//! fire_mode = "release"      # how a leaf fires: release, edge or retract
15//! edge_threshold = 90        # % out an edge leaf fires
16//! retract_delta = 10         # % back from its furthest a retract leaf fires
17//! trigger_open_pct = 60      # % an analog trigger is pulled to open its wheel
18//! trigger_close_pct = 40     # % it falls under to let go
19//! wheel_min_open_ms = 150    # a wheel stays open at least this long
20//! ```
21
22use super::wheel::{FireMode, Timing};
23
24/// The controller's feel.
25#[derive(Clone, Debug, PartialEq)]
26pub struct Tuning {
27    /// The right stick walks, and the left aims and scrolls.
28    pub walk_right: bool,
29    /// The stick that does not walk scrolls the window in use, or steps
30    /// through interact mode.
31    pub other_scrolls: bool,
32    /// Where Back sits in a folder: `down`, `up-left` and the rest, or
33    /// `none`.
34    pub back_slice: String,
35    /// How far a stick is pushed, in percent, before a slice is aimed.
36    pub deadzone: u8,
37    /// Held on a leaf this long, in milliseconds, it is committed.
38    pub aim_dwell_ms: u32,
39    /// Held on a folder or Back this long, in milliseconds, the wheel goes
40    /// in or out.
41    pub nav_dwell_ms: u32,
42    /// A fire this soon after the last, in milliseconds, is not sent.
43    pub fire_debounce_ms: u32,
44    /// The walking stick is stilled this long, in milliseconds, after a
45    /// wheel closes.
46    pub release_grace_ms: u32,
47    /// How a leaf fires, for a slice that does not say.
48    pub fire_mode: FireMode,
49    /// How far out, in percent, an edge leaf fires.
50    pub edge_threshold: u8,
51    /// How far back from its furthest, in percent, a retract leaf fires.
52    pub retract_delta: u8,
53    /// How far an analog trigger is pulled, in percent, to open its wheel.
54    pub trigger_open_pct: u8,
55    /// How far an analog trigger falls under, in percent, to let its wheel go.
56    pub trigger_close_pct: u8,
57    /// A wheel stays open at least this long, in milliseconds.
58    pub wheel_min_open_ms: u32,
59}

61impl Default for Tuning {
62    fn default() -> Self {
63        Self {
64            walk_right: false,
65            other_scrolls: true,
66            back_slice: "down".to_owned(),
67            deadzone: 50,
68            aim_dwell_ms: 150,
69            nav_dwell_ms: 150,
70            fire_debounce_ms: 300,
71            release_grace_ms: 40,
72            fire_mode: FireMode::Release,
73            edge_threshold: 90,
74            retract_delta: 10,
75            trigger_open_pct: 60,
76            trigger_close_pct: 40,
77            wheel_min_open_ms: 150,
78        }
79    }
80}

82impl Tuning {
83    /// What `[controller_tuning]` in `table` says, over Hydra's.
84    pub fn read(table: &toml::Table) -> Self {
85        let mut tuning = Self::default();
86        let Some(written) = table
87            .get("controller_tuning")
88            .and_then(toml::Value::as_table)
89        else {
90            return tuning;
91        };
92        let said = |key: &str| written.get(key).and_then(toml::Value::as_str);
93        let number = |key: &str| written.get(key).and_then(toml::Value::as_integer);
94        let percent = |key: &str, into: &mut u8| {
95            if let Some(read) = number(key).and_then(|read| u8::try_from(read.clamp(0, 100)).ok()) {
96                *into = read;
97            }
98        };
99        let millis = |key: &str, into: &mut u32| {
100            if let Some(read) = number(key).and_then(|read| u32::try_from(read.max(0)).ok()) {
101                *into = read;
102            }
103        };
104        if let Some(stick) = said("movement_stick") {
105            tuning.walk_right = stick.eq_ignore_ascii_case("right");
106        }
107        if let Some(other) = said("opposing_stick") {
108            tuning.other_scrolls = !other.eq_ignore_ascii_case("none");
109        }
110        if let Some(side) = said("back_slice") {
111            side.trim()
112                .to_lowercase()
113                .clone_into(&mut tuning.back_slice);
114        }
115        if let Some(mode) = said("fire_mode") {
116            tuning.fire_mode = FireMode::named(mode.trim());
117        }
118        percent("deadzone", &mut tuning.deadzone);
119        percent("edge_threshold", &mut tuning.edge_threshold);
120        percent("retract_delta", &mut tuning.retract_delta);
121        percent("trigger_open_pct", &mut tuning.trigger_open_pct);
122        percent("trigger_close_pct", &mut tuning.trigger_close_pct);
123        millis("aim_dwell_ms", &mut tuning.aim_dwell_ms);
124        millis("nav_dwell_ms", &mut tuning.nav_dwell_ms);
125        millis("fire_debounce_ms", &mut tuning.fire_debounce_ms);
126        millis("release_grace_ms", &mut tuning.release_grace_ms);
127        millis("wheel_min_open_ms", &mut tuning.wheel_min_open_ms);
128        tuning
129    }
130
131    /// The feel the wheel is aimed with.
132    #[must_use]
133    pub fn timing(&self) -> Timing {
134        Timing {
135            deadzone: (f32::from(self.deadzone) / 100.0).min(0.99),
136            aim_ms: u128::from(self.aim_dwell_ms),
137            nav_ms: u128::from(self.nav_dwell_ms),
138            fire_mode: self.fire_mode,
139            edge_threshold: f32::from(self.edge_threshold) / 100.0,
140            retract_delta: f32::from(self.retract_delta) / 100.0,
141        }
142    }
143
144    /// How far, 0 to 1, an analog trigger opens its wheel, and falls under
145    /// to let go: the second always under the first.
146    #[must_use]
147    pub fn trigger(&self) -> (f32, f32) {
148        let open = (f32::from(self.trigger_open_pct) / 100.0).clamp(0.05, 1.0);
149        let close = (f32::from(self.trigger_close_pct) / 100.0).clamp(0.0, open - 0.01);
150        (open, close)
151    }
152}

154#[cfg(test)]
155mod tests {
156    use super::*;
157
158    #[test]
159    fn the_tuning_written_is_over_hydras() {
160        let read = |text: &str| Tuning::read(&text.parse().expect("toml"));
161        assert_eq!(read(""), Tuning::default());
162        let tuning = read(
163            "[controller_tuning]\nmovement_stick = \"right\"\nopposing_stick = \"none\"\n\
164             back_slice = \"Up-Left\"\ndeadzone = 300\nfire_mode = \"edge\"\naim_dwell_ms = -5\n\
165             trigger_open_pct = 30\ntrigger_close_pct = 50\n",
166        );
167        assert!(tuning.walk_right && !tuning.other_scrolls);
168        assert_eq!(tuning.back_slice, "up-left");
169        assert_eq!(tuning.deadzone, 100, "a percent at most");
170        assert_eq!(tuning.fire_mode, FireMode::Edge);
171        assert_eq!(tuning.aim_dwell_ms, 0);
172        let (open, close) = tuning.trigger();
173        assert!(
174            (open - 0.3).abs() < 1e-6 && close < open,
175            "close under open"
176        );
177        assert!((tuning.timing().deadzone - 0.99).abs() < 1e-6);
178    }

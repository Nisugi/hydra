1//! The controller's buttons by name, and a bind's key: the modifiers held,
2//! then the button pressed, `l2+r1+dpad_down`.
3
4/// Every button that can be bound, in the order a key writes its modifiers.
5pub const BUTTONS: [&str; 17] = [
6    "south",
7    "east",
8    "north",
9    "west",
10    "dpad_up",
11    "dpad_down",
12    "dpad_left",
13    "dpad_right",
14    "l1",
15    "r1",
16    "l2",
17    "r2",
18    "l3",
19    "r3",
20    "select",
21    "start",
22    "guide",
23];
24
25/// Where `name` comes among the modifiers: by [`BUTTONS`], one not there
26/// after them all.
27fn rank(name: &str) -> usize {
28    BUTTONS
29        .iter()
30        .position(|button| *button == name)
31        .unwrap_or(BUTTONS.len())
32}
33
34/// A bind's key: the modifiers held, in [`BUTTONS`]' order and each once,
35/// and the button pressed. Written `l2+south`, or the button alone.
36#[derive(Clone, Debug, PartialEq, Eq)]
37pub struct BindKey {
38    /// The modifiers held, in [`BUTTONS`]' order.
39    pub mods: Vec<String>,
40    /// The button pressed.
41    pub button: String,
42}
43
44impl BindKey {
45    /// `button` pressed with `mods` held, the button itself not among them.
46    pub fn new(button: &str, mods: impl IntoIterator<Item = String>) -> Self {
47        let mut mods: Vec<String> = mods.into_iter().filter(|held| held != button).collect();
48        mods.sort_by(|a, b| rank(a).cmp(&rank(b)).then_with(|| a.cmp(b)));
49        mods.dedup();
50        Self {
51            mods,
52            button: button.to_owned(),
53        }
54    }
55
56    /// The key `written`: the last of its `+` pieces the button, the rest
57    /// the modifiers, in any order. None for an empty one.
58    pub fn parse(written: &str) -> Option<Self> {
59        let mut parts: Vec<&str> = written
60            .split('+')
61            .map(str::trim)
62            .filter(|part| !part.is_empty())
63            .collect();
64        let button = parts.pop()?;
65        Some(Self::new(button, parts.into_iter().map(str::to_owned)))
66    }
67
68    /// As written: the modifiers in order, then the button.
69    #[must_use]
70    pub fn canonical(&self) -> String {
71        if self.mods.is_empty() {
72            self.button.clone()
73        } else {
74            format!("{}+{}", self.mods.join("+"), self.button)
75        }
76    }
77
78    /// Whether every button it names is one of [`BUTTONS`].
79    #[must_use]
80    pub fn known(&self) -> bool {
81        std::iter::once(&self.button)
82            .chain(&self.mods)
83            .all(|name| BUTTONS.contains(&name.as_str()))
84    }
85}

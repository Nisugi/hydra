30/// The binds in force for one character: every character's, its own taken
31/// over them key by key, and the legend's entries.
32#[derive(Clone, Debug, Default, PartialEq)]
33pub struct Binds {
34    /// What each bind does, by its key as written: `l2+south`.
35    pub by_key: std::collections::BTreeMap<String, Bound>,
36    /// The legend's lines, as the file gives them.
37    pub overlay: Vec<String>,
38    /// How the controller feels.
39    pub tuning: tuning::Tuning,
40    /// The wheels, by name.
41    pub wheels: wheel::Wheels,
42    /// When and how the controller rumbles.
43    pub rumble: rumble::Rumble,
44}
45
46impl Binds {
47    /// `every` character's binds, with `own`'s over them; `own`'s legend,
48    /// when it names one, in place of `every`'s.
49    #[must_use]
50    pub fn of(every: &ControllerFile, own: Option<&ControllerFile>) -> Self {
51        let mut binds = every.binds.clone();
52        let mut overlay = every.overlay.clone().unwrap_or_default();
53        let mut tuning = every.tuning.clone().unwrap_or_default();
54        let mut wheels = every.wheels.clone();
55        let mut rumble = every.rumble.clone().unwrap_or_default();
56        if let Some(own) = own {
57            binds.extend(
58                own.binds
59                    .iter()
60                    .map(|(key, bound)| (key.clone(), bound.clone())),
61            );
62            if let Some(own) = &own.overlay {
63                overlay.clone_from(own);
64            }
65            if let Some(own) = &own.tuning {
66                tuning = own.clone();
67            }
68            wheels.over(&own.wheels);
69            if let Some(own) = &own.rumble {
70                rumble = own.clone();
71            }
72        }
73        Self {
74            by_key: binds,
75            overlay,
76            tuning,
77            wheels,
78            rumble,
79        }
80    }
81
82    /// The buttons bound to `controller_modifier`.
83    #[must_use]
84    pub fn modifiers(&self) -> BTreeSet<&str> {
85        self.by_key
86            .iter()
87            .filter(|(_, bound)| **bound == Bound::Modifier)
88            .map(|(key, _)| key.as_str())
89            .collect()
90    }
91
92    /// What pressing `button` does with `held` held: the bind of `button`
93    /// with exactly the held modifiers; none for a modifier itself, or no
94    /// such bind.
95    #[must_use]
96    pub fn pressed(&self, button: &str, held: &BTreeSet<&str>) -> Option<&Bound> {
97        let modifiers = self.modifiers();
98        if modifiers.contains(button) {
99            return None;
100        }
101        let mods = held
102            .iter()
103            .filter(|held| modifiers.contains(**held))
104            .map(|held| (*held).to_owned());
105        let key = BindKey::new(button, mods);
106        self.by_key.get(&key.canonical())
107    }
108
109    /// The modifiers held, of `held`, in the order a key writes them.
110    #[must_use]
111    pub fn held_modifiers(&self, held: &BTreeSet<&str>) -> Vec<String> {
112        let modifiers = self.modifiers();
113        BindKey::new(
114            "",
115            held.iter()
116                .filter(|held| modifiers.contains(**held))
117                .map(|held| (*held).to_owned()),
118        )
119        .mods
120    }
121
122    /// The legend's lines, each with whether its modifiers are those held:
123    /// `south - look`, `[l2] south - stand`. An entry naming no bind is
124    /// left out.
125    #[must_use]
126    pub fn legend(&self, held: &BTreeSet<&str>) -> Vec<(String, bool)> {
127        let held = self.held_modifiers(held);
128        self.overlay
129            .iter()
130            .filter_map(|entry| {
131                let key = BindKey::parse(entry)?;
132                let bound = self.by_key.get(&key.canonical())?;
133                let what = bound.said();
134                let line = if key.mods.is_empty() {
135                    format!("{} - {what}", key.button)
136                } else {
137                    format!("[{}] {} - {what}", key.mods.join("+"), key.button)
138                };
139                Some((line, key.mods == held))
140            })
141            .collect()
142    }
143}

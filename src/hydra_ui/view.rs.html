1//! What a play window shows beside the story, projected from a snapshot.
2//!
3//! Unknown is never shown as zero or empty: a gauge the game has not
4//! reported, a compass not yet seen, a clock before the first prompt are
5//! `None`.
6
7use hydra_model::{GameState, RoomItem, Vital};
8use hydra_session::State;
9
10use crate::run::{StyledRun, styled};

12/// The four gauges, in the order a window shows them: label and wire id.
13const GAUGES: [(&str, &str); 4] = [
14    ("Health", "health"),
15    ("Mana", "mana"),
16    ("Stamina", "stamina"),
17    ("Spirit", "spirit"),
18];
19
20/// One gauge.
21#[derive(Clone, Copy, Debug, PartialEq, Eq)]
22pub struct VitalView {
23    /// What the window calls it.
24    pub label: &'static str,
25    /// What the game reported; `None` until it has.
26    pub vital: Option<Vital>,
27}
28
29/// The room, as far as it has been seen.
30#[derive(Clone, Debug, Default, PartialEq, Eq)]
31pub struct RoomView {
32    /// The room's name.
33    pub title: Option<String>,
34    /// The game's room id.
35    pub id: Option<String>,
36    /// The description, as the game styled it.
37    pub description: Option<Vec<StyledRun>>,
38    /// The compass's ways out; `None` before the compass has been seen.
39    pub exits: Option<Vec<String>>,
40    /// Creatures, by what the room calls them.
41    pub creatures: Vec<String>,
42    /// Other objects.
43    pub objects: Vec<String>,
44    /// Other players, each with what it is doing when the room says.
45    pub players: Vec<String>,
46}

48/// What a play window shows beside the story.
49#[derive(Clone, Debug, PartialEq, Eq)]
50pub struct PlayView {
51    /// How the session is connected.
52    pub lifecycle: State,
53    /// The room.
54    pub room: RoomView,
55    /// Health, mana, stamina and spirit.
56    pub vitals: [VitalView; 4],
57    /// Seconds of roundtime left; `None` before the first prompt sets the
58    /// game's clock.
59    pub roundtime: Option<u32>,
60    /// Seconds of cast roundtime left, as `roundtime`.
61    pub casttime: Option<u32>,
62}
63
64impl PlayView {
65    /// The view of `state`, connected as `lifecycle`, when the game's clock
66    /// reads `server_now` (`GameState::game_time_now`; a test states it).
67    #[must_use]
68    pub fn project(state: &GameState, lifecycle: State, server_now: Option<u32>) -> Self {
69        let room = &state.room;
70        let left =
71            |ends: Option<u32>| server_now.map(|now| ends.map_or(0, |end| end.saturating_sub(now)));
72        Self {
73            lifecycle,
74            room: RoomView {
75                title: room.title.clone(),
76                id: room.id.clone(),
77                description: room.description.as_ref().map(|runs| styled(&runs.runs)),
78                exits: room.exits.clone(),
79                creatures: names(&room.creatures),
80                objects: names(&room.objects),
81                players: room
82                    .players
83                    .iter()
84                    .map(|player| match &player.status {
85                        Some(status) => format!("{} ({status})", player.text),
86                        None => player.text.clone(),
87                    })
88                    .collect(),
89            },
90            vitals: GAUGES.map(|(label, id)| VitalView {
91                label,
92                vital: state.vital(id),
93            }),
94            roundtime: left(state.roundtime_ends),
95            casttime: left(state.cast_time_ends),
96        }
97    }
98}

100/// A way out in full, as a compass is labelled and a player types it: the
101/// game's compass says `n`, `ne`, `out`. A word it does not shorten is
102/// itself.
103#[must_use]
104pub fn exit_word(exit: &str) -> &str {
105    match exit {
106        "n" => "north",
107        "ne" => "northeast",
108        "e" => "east",
109        "se" => "southeast",
110        "s" => "south",
111        "sw" => "southwest",
112        "w" => "west",
113        "nw" => "northwest",
114        "u" => "up",
115        "d" => "down",
116        other => other,
117    }
118}

120/// The room's description as styled runs, as the game styled it; `None`
121/// until one is observed.
122#[must_use]
123pub fn room_description(state: &GameState) -> Option<Vec<StyledRun>> {
124    state
125        .room
126        .description
127        .as_ref()
128        .map(|body| styled(&body.runs))
129}
130
131fn names(items: &[RoomItem]) -> Vec<String> {
132    items.iter().map(|item| item.text.clone()).collect()
133}

135#[cfg(test)]
136mod tests {
137    use hydra_protocol::Parser;
138
139    use super::*;
140
141    fn applied(wire: &str) -> GameState {
142        let mut state = GameState::default();
143        for frame in Parser::new().push_bytes(wire.as_bytes()) {
144            state.apply(&frame);
145        }
146        state
147    }
148
149    /// The compass's short names in full; a word already whole is itself.
150    #[test]
151    fn a_way_out_is_said_in_full() {
152        assert_eq!(exit_word("ne"), "northeast");
153        assert_eq!(exit_word("out"), "out");
154        assert_eq!(exit_word("north"), "north");
155    }
156
157    #[test]
158    fn nothing_seen_is_unknown_not_zero() {
159        let view = PlayView::project(&GameState::default(), State::Connecting, None);
160        assert_eq!(view.room, RoomView::default());
161        assert!(view.vitals.iter().all(|gauge| gauge.vital.is_none()));
162        assert_eq!((view.roundtime, view.casttime), (None, None));
163    }
164
165    #[test]
166    fn the_room_vitals_and_clocks_are_the_states() {
167        let state = applied(
168            "<nav rm='7503251'/>\
169             <streamWindow id='room' title='Room' subtitle=\" - [Rawknuckle's, Watering Hole]\" location='center' target='drop' ifClosed='' resident='true'/>\n\
170             <component id='room desc'>A cozy <pushBold/>bar<popBold/>.</component>\n\
171             <component id='room objs'>You also see <pushBold/><a exist=\"1\" noun=\"kobold\">a kobold</a><popBold/> and <a exist=\"2\" noun=\"stool\">a stool</a>.</component>\n\
172             <component id='room players'>Also here: <a exist=\"-3\" noun=\"Bob\">Bob</a> who is hiding.</component>\n\
173             <compass><dir value=\"n\"/><dir value=\"out\"/></compass>\n\
174             <dialogData id='minivitals'><progressBar id='health' value='87' text='health 174/200' /><progressBar id='mana' value='50' text='mana 10/20' /></dialogData>\n\
175             <roundTime value='105'/><castTime value='102'/><prompt time=\"100\">R&gt;</prompt>\n",
176        );
177        let view = PlayView::project(&state, State::Ready, Some(101));
178        assert_eq!(view.room.id.as_deref(), Some("7503251"));
179        assert_eq!(
180            view.room.title.as_deref(),
181            Some("[Rawknuckle's, Watering Hole]")
182        );
183        let description: String = view
184            .room
185            .description
186            .iter()
187            .flatten()
188            .map(|run| run.text.as_str())
189            .collect();
190        assert_eq!(description, "A cozy bar.");
191        assert_eq!(
192            view.room.exits,
193            Some(vec!["n".to_owned(), "out".to_owned()])
194        );
195        assert_eq!(view.room.creatures, ["a kobold"]);
196        assert_eq!(view.room.objects, ["a stool"]);
197        assert_eq!(view.room.players, ["Bob (hiding)"]);
198        assert_eq!(
199            view.vitals[0].vital.and_then(Vital::amount),
200            Some((174, 200))
201        );
202        assert_eq!(view.vitals[1].vital.map(|vital| vital.percent), Some(50));
203        assert_eq!(view.vitals[2].vital, None, "stamina not reported");
204        assert_eq!((view.roundtime, view.casttime), (Some(4), Some(1)));
205        let later = PlayView::project(&state, State::Ready, Some(110));
206        assert_eq!((later.roundtime, later.casttime), (Some(0), Some(0)));
207    }

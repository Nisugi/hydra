1//! An object's menu as a frontend shows it: the game's `<menu>` of bare
2//! coordinates, each labelled and given the command it sends by the model's
3//! dictionary ([`hydra_model::MenuCommands`]), grouped by category in the
4//! game's order. The model resolves once for every frontend
5//! (`crates/hydra-model/src/state/menu.rs`, *Why this lives in the model*);
6//! this is what a frontend draws of it: a link is clickable, and its menu
7//! pops up.
8
9use hydra_model::{GameState, Menu, MenuCommands, category_path};
10
11/// One entry of an object's menu.
12#[derive(Clone, Debug, PartialEq, Eq)]
13pub struct MenuEntry {
14    /// What it says: `attack`, `ask about amplify`; for a coordinate the
15    /// dictionary does not know, the coordinate, so the entry is seen.
16    pub label: String,
17    /// What choosing it sends; `None` when it cannot be sent: an unknown
18    /// coordinate, or one still wanting a second word (`%`).
19    pub command: Option<String>,
20}
21
22/// A category of an object's menu, and its entries.
23#[derive(Clone, Debug, PartialEq, Eq)]
24pub struct MenuGroup {
25    /// Its path below the top: empty for a top-level category, whose
26    /// entries a menu shows in place; `["focused multistrike"]` for one a
27    /// level down (`hydra_model::category_path`).
28    pub under: Vec<String>,
29    /// Its entries, in the game's order.
30    pub entries: Vec<MenuEntry>,
31}
32
33/// `menu` for the object clicked, its id and noun, as `state` has taught the
34/// dictionary. The dictionary's `_dialog` entries are left out: each opens a
35/// dialog of the Wrayth client's own, which no frontend here has.
36#[must_use]
37pub fn object_menu(
38    menu: &Menu,
39    clicked: (&str, &str),
40    state: Option<&GameState>,
41) -> Vec<MenuGroup> {
42    let learned = state.map(|state| &state.learned_commands);
43    MenuCommands::get()
44        .resolve_grouped(menu, clicked, None, learned)
45        .into_iter()
46        .map(|(category, items)| MenuGroup {
47            under: category.as_deref().map_or_else(Vec::new, |category| {
48                category_path(category)
49                    .into_iter()
50                    .skip(1)
51                    .map(str::to_owned)
52                    .collect()
53            }),
54            entries: items
55                .into_iter()
56                .filter(|item| {
57                    !item
58                        .command
59                        .as_deref()
60                        .is_some_and(|command| command.starts_with("_dialog"))
61                })
62                .map(|item| {
63                    let wanting = item.needs_secondary;
64                    MenuEntry {
65                        label: item.label.unwrap_or_else(|| format!("({})", item.coord)),
66                        command: item.command.filter(|_| !wanting),
67                    }
68                })
69                .collect(),
70        })
71        .filter(|group| !group.entries.is_empty())
72        .collect()
73}

75/// The command an object link's own `coord=` names, for the object `clicked`,
76/// its id and noun, as `state` has taught the dictionary: a click sends it
77/// rather than asking for a menu. `None` for a coordinate the dictionary lacks, a dialog, or one
78/// wanting a second word.
79///
80/// Resolved as a menu's entry is, the server's row first: read from the
81/// shipped table alone, a coordinate the game had changed would send its
82/// old command from a link and its new one from the menu.
83#[must_use]
84pub fn link_command(
85    coord: &str,
86    clicked: (&str, &str),
87    state: Option<&GameState>,
88) -> Option<String> {
89    let item = hydra_model::MenuItem {
90        coord: Some(coord.to_owned()),
91        ..hydra_model::MenuItem::default()
92    };
93    let learned = state.map(|state| &state.learned_commands);
94    let resolved = MenuCommands::get().resolve_item(&item, clicked, None, learned);
95    let wanting = resolved.needs_secondary;
96    resolved
97        .command
98        .filter(|command| !wanting && !command.starts_with("_dialog"))
99}

101#[cfg(test)]
102mod tests {
103    use super::*;
104
105    /// A group's entries as a test compares them: each label and command.
106    type Entries = Vec<(String, Option<String>)>;
107
108    /// A direct link and the menu agree on what a coordinate sends, the
109    /// server's row first: one the game changed, and one only it knows.
110    #[test]
111    fn a_link_and_the_menu_send_what_the_game_taught() {
112        let mut state = GameState::default();
113        let row = |coord: &str, command: &str| hydra_model::MenuCommand {
114            coord: coord.to_owned(),
115            label: command.replace('#', "@"),
116            command: command.to_owned(),
117            category: "1".to_owned(),
118        };
119        state
120            .learned_commands
121            .absorb(&[row("2524,1543", "inspect #"), row("2524,9999", "pet #")]);
122        let clicked = ("123", "kobold");
123        for (coord, sent) in [("2524,1543", "inspect #123"), ("2524,9999", "pet #123")] {
124            let menu = object_menu(&menu(&[(coord, None)]), clicked, Some(&state));
125            assert_eq!(menu[0].entries[0].command.as_deref(), Some(sent));
126            assert_eq!(
127                link_command(coord, clicked, Some(&state)).as_deref(),
128                Some(sent)
129            );
130        }
131        assert_eq!(
132            link_command("2524,1543", clicked, None).as_deref(),
133            Some("attack #123"),
134            "the shipped table, untaught"
135        );
136        assert_eq!(link_command("2524,1553", clicked, None), None, "a dialog");
137        assert_eq!(
138            link_command("2524,2184", clicked, None),
139            None,
140            "wanting a word"
141        );
142    }

144    /// A menu of the coordinates given, as the game sends one, `id` 1.
145    fn menu(items: &[(&str, Option<&str>)]) -> Menu {
146        let mut menu = Menu {
147            id: "1".to_owned(),
148            ..Menu::default()
149        };
150        for (coord, noun) in items {
151            menu.items.push(hydra_model::MenuItem::default());
152            if let Some(item) = menu.items.last_mut() {
153                item.coord = Some((*coord).to_owned());
154                item.noun = noun.map(str::to_owned);
155            }
156        }
157        menu
158    }

160    /// Each entry labelled and given its command for the object clicked,
161    /// grouped under its category's path; a dialog left out; one still
162    /// wanting its second word, and an unknown coordinate, shown unsendable.
163    #[test]
164    fn an_objects_menu_is_labelled_and_grouped() {
165        let groups = object_menu(
166            &menu(&[
167                ("2524,1543", None),
168                ("2524,1906", Some("amplify")),
169                ("2524,1553", None),
170                ("2524,2184", None),
171                ("9999,9999", None),
172            ]),
173            ("123", "kobold"),
174            None,
175        );
176        let said: Vec<(Vec<String>, Entries)> = groups
177            .into_iter()
178            .map(|group| {
179                let entries = group
180                    .entries
181                    .into_iter()
182                    .map(|entry| (entry.label, entry.command))
183                    .collect();
184                (group.under, entries)
185            })
186            .collect();
187        let entry =
188            |label: &str, command: Option<&str>| (label.to_owned(), command.map(str::to_owned));
189        assert_eq!(
190            said,
191            [
192                (vec![], vec![entry("attack", Some("attack #123"))]),
193                (
194                    vec!["questions".to_owned()],
195                    vec![entry("ask about amplify", Some("ask #123 about amplify"))]
196                ),
197                (vec![], vec![entry("transfer %", None)]),
198                (vec![], vec![entry("(9999,9999)", None)]),
199            ]
200        );
201    }

1//! Changing the highlights file: the one writer the editor and
2//! `.highlight` share. Each change reads the file afresh, changes it and
3//! writes it whole, with the file's lock held, so two changes at once both
4//! land; a file that does not read is never written over.
5
6use std::path::Path;
7
8use super::file::{FILE, HighlightFile, Rule};
9
10/// Change the highlights file in the data folder `dir` by `change`, and
11/// write it. What `change` said.
12///
13/// # Errors
14///
15/// The file does not read, `change` refused, or it was not written; the
16/// file is as it was.
17pub fn change<T>(
18    dir: &Path,
19    change: impl FnOnce(&mut HighlightFile) -> Result<T, String>,
20) -> Result<T, String> {
21    hydra_session::store::changing(&dir.join(FILE), || {
22        let mut file = HighlightFile::load(dir)?;
23        let said = change(&mut file)?;
24        file.save(dir)?;
25        Ok(said)
26    })
27}
28
29impl HighlightFile {
30    /// The name of the highlight called `name`: the one called exactly
31    /// that first, then any that is ignoring case, spaces round either
32    /// aside -- an imported name may carry them, and two imported names may
33    /// differ only in case, each its own highlight.
34    #[must_use]
35    pub fn named(&self, name: &str) -> Option<&str> {
36        let name = name.trim();
37        self.rules
38            .keys()
39            .find(|have| have.trim() == name)
40            .or_else(|| {
41                self.rules
42                    .keys()
43                    .find(|have| have.trim().eq_ignore_ascii_case(name))
44            })
45            .map(String::as_str)
46    }
47
48    /// Every category a highlight is in, or that is switched.
49    #[must_use]
50    pub fn category_names(&self) -> Vec<String> {
51        let mut names: Vec<String> = (self.rules.values().map(|rule| rule.category.clone()))
52            .chain(self.categories.keys().cloned())
53            .collect();
54        names.sort();
55        names.dedup();
56        names
57    }
58
59    /// Keep `rule` as `name`, in place of the one called `was`, when it
60    /// was one.
61    ///
62    /// # Errors
63    ///
64    /// No name, a name another highlight has, or a highlight that cannot be
65    /// built: why.
66    pub fn put(&mut self, was: Option<&str>, name: &str, rule: Rule) -> Result<(), String> {
67        let name = name.trim();
68        if name.is_empty() {
69            return Err("Give the highlight a name.".to_owned());
70        }
71        // The highlight being changed, as the file names it.
72        let was = was.and_then(|was| self.named(was)).map(str::to_owned);
73        if let Some(have) = self.named(name)
74            && was.as_deref() != Some(have)
75        {
76            return Err(format!("There is already a highlight called {have}."));
77        }
78        super::Matcher::check_rule(name, &rule)?;
79        if let Some(was) = was {
80            self.rules.remove(&was);
81        }
82        self.rules.insert(name.to_owned(), rule);
83        Ok(())
84    }
85
86    /// Switch the highlight called `named` on or off, or, when none is, the
87    /// category. What was done.
88    ///
89    /// # Errors
90    ///
91    /// No highlight or category is called that.
92    pub fn switch(&mut self, named: &str, on: bool) -> Result<String, String> {
93        let word = if on { "on" } else { "off" };
94        if let Some(name) = self.named(named).map(str::to_owned) {
95            if let Some(rule) = self.rules.get_mut(&name) {
96                rule.off = !on;
97            }
98            return Ok(format!("The highlight {name} is {word}."));
99        }
100        let category = self
101            .category_names()
102            .into_iter()
103            .find(|category| category.eq_ignore_ascii_case(named.trim()))
104            .ok_or_else(|| format!("No highlight or category is called {}.", named.trim()))?;
105        if on {
106            self.categories.remove(&category);
107        } else {
108            self.categories.insert(category.clone(), false);
109        }
110        Ok(format!("The category {category} is {word}."))
111    }
112
113    /// Take away the highlight called `named`. Its name.
114    ///
115    /// # Errors
116    ///
117    /// None is called that.
118    pub fn remove(&mut self, named: &str) -> Result<String, String> {
119        let name = self
120            .named(named)
121            .map(str::to_owned)
122            .ok_or_else(|| format!("No highlight is called {}.", named.trim()))?;
123        self.rules.remove(&name);
124        Ok(name)
125    }
126}

128#[cfg(test)]
129mod tests {
130    use super::*;
131
132    fn rule(text: &str) -> Rule {
133        Rule {
134            text: Some(text.to_owned()),
135            color: Some("#ff0000".to_owned()),
136            ..Rule::default()
137        }
138    }
139
140    #[test]
141    fn put_switch_and_remove_keep_names_and_categories_straight() {
142        let mut file = HighlightFile::default();
143        file.put(None, "Bob", rule("Bob")).unwrap();
144        assert!(file.put(None, "bob", rule("x")).is_err(), "taken, any case");
145        assert!(file.put(None, " ", rule("x")).is_err());
146        assert!(
147            file.put(
148                None,
149                "bad",
150                Rule {
151                    regex: Some("(".to_owned()),
152                    ..Rule::default()
153                }
154            )
155            .is_err()
156        );
157        file.put(Some("BOB"), "Robert", rule("Robert")).unwrap();
158        assert_eq!(file.rules.keys().collect::<Vec<_>>(), ["Robert"], "renamed");
159        file.put(Some("Robert"), "Robert", rule("Rob")).unwrap();
160        assert_eq!(
161            file.rules["Robert"].text.as_deref(),
162            Some("Rob"),
163            "kept as itself"
164        );
165
166        assert_eq!(
167            file.switch("robert", false).unwrap(),
168            "The highlight Robert is off."
169        );
170        assert!(file.rules["Robert"].off);
171        assert_eq!(
172            file.switch("general", false).unwrap(),
173            "The category General is off."
174        );
175        assert!(!file.category_on("General"));
176        file.switch("General", true).unwrap();
177        assert!(file.categories.is_empty());
178        assert!(file.switch("nothing", true).is_err());
179        assert_eq!(file.remove("ROBERT").unwrap(), "Robert");
180        assert!(file.remove("Robert").is_err());
181    }
182
183    /// A highlight kept under a name with spaces round it -- an import's --
184    /// is found by the name without them, so it can be switched and taken
185    /// away.
186    #[test]
187    fn a_name_kept_with_spaces_round_it_is_still_found() {
188        let mut file = HighlightFile::default();
189        file.rules.insert(" Bob ".to_owned(), rule("Bob"));
190        assert_eq!(file.named("bob"), Some(" Bob "));
191        assert!(file.switch("Bob", false).is_ok());
192        assert!(file.rules[" Bob "].off);
193        assert_eq!(file.remove("Bob").unwrap(), " Bob ");
194    }
195
196    /// Two highlights whose names differ only in case -- an import's -- are
197    /// two: editing one keeps the other.
198    #[test]
199    fn names_that_differ_only_in_case_are_kept_apart() {
200        let mut file = HighlightFile::default();
201        file.rules.insert("Bob".to_owned(), rule("Bob"));
202        file.rules.insert("bob".to_owned(), rule("bob"));
203        file.put(Some("bob"), "bob", rule("bobby")).unwrap();
204        assert_eq!(file.rules["Bob"].text.as_deref(), Some("Bob"), "untouched");
205        assert_eq!(file.rules["bob"].text.as_deref(), Some("bobby"));
206        assert_eq!(file.remove("bob").unwrap(), "bob");
207        assert!(file.rules.contains_key("Bob"));
208    }
209
210    /// Renaming Bob to bob, while another highlight is called bob, is
211    /// refused: the other is not written over.
212    #[test]
213    fn a_rename_onto_a_name_that_differs_only_in_case_is_refused() {
214        let mut file = HighlightFile::default();
215        file.rules.insert("Bob".to_owned(), rule("Bob"));
216        file.rules.insert("bob".to_owned(), rule("bob"));
217        assert!(file.put(Some("Bob"), "bob", rule("Bob")).is_err());
218        assert_eq!(file.rules["bob"].text.as_deref(), Some("bob"), "untouched");
219        assert!(file.rules.contains_key("Bob"));
220        // A rename of its own case, with nothing else so called, is fine.
221        let mut alone = HighlightFile::default();
222        alone.rules.insert("Bob".to_owned(), rule("Bob"));
223        alone.put(Some("Bob"), "BOB", rule("Bob")).unwrap();
224        assert!(alone.rules.contains_key("BOB") && !alone.rules.contains_key("Bob"));
225    }

227    #[test]
228    fn a_change_is_written_and_a_broken_file_never_written_over() {
229        let dir = std::env::temp_dir().join(format!("hydra-highlight-edit-{}", std::process::id()));
230        let _ = std::fs::remove_dir_all(&dir);
231        std::fs::create_dir_all(&dir).unwrap();
232        change(&dir, |file| file.put(None, "Bob", rule("Bob"))).unwrap();
233        assert!(HighlightFile::load(&dir).unwrap().rules.contains_key("Bob"));
234        assert!(change(&dir, |file| file.remove("nobody")).is_err());
235        std::fs::write(dir.join(FILE), "nonsense = [").unwrap();
236        assert!(change(&dir, |file| file.remove("Bob")).is_err());
237        assert_eq!(
238            std::fs::read_to_string(dir.join(FILE)).unwrap(),
239            "nonsense = ["
240        );
241        let _ = std::fs::remove_dir_all(&dir);
242    }

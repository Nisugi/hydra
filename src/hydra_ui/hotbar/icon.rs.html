1//! A button's icon: a cell of a sheet of icons, or an image of its own,
2//! scaled to the button; drawn grey or not, with a border of one colour or
3//! two.
4//!
5//! **As a file writes one**, any of the ways a hotbar file written for
6//! Vellum does, and Hydra's own short way for an image:
7//!
8//! ```toml
9//! icon = { sheet = "rogue", cell = 5 }                      # a sheet's cell
10//! icon = { icon = { sheet_cell = { sheet = "rogue", cell = 5 } } }
11//! icon = { icon = { image = { path = "sword.png" } } }      # an image
12//! icon = { image = "sword.png", grayscale = true }          # the same, short
13//! icon = { sheet = "rogue", cell = 5, border = "#00ff00", border_end = "#004400",
14//!          border_dir = "radial", border_width = 3 }
15//! ```
16//!
17//! A cell is counted from 1, along each row of the sheet and then down.
18//! Written back, a cell is the first way and an image the third, which
19//! Vellum reads too.
20//!
21//! **The sheets** are named in `icons.toml` in the icons folder, each its
22//! picture and its cells' size:
23//!
24//! ```toml
25//! [sheets.rogue]
26//! path = "rogue.png"    # in the icons folder, or anywhere
27//! cell = 64             # its cells' edge, in pixels; 64 unless it says
28//! ```
29

140/// An icon as a file may write it.
141#[derive(Deserialize)]
142struct IconDe {
143    #[serde(default)]
144    icon: Option<RefDe>,
145    #[serde(default)]
146    sheet: Option<String>,
147    #[serde(default)]
148    cell: Option<u32>,
149    #[serde(default)]
150    image: Option<String>,
151    #[serde(default)]
152    grayscale: bool,
153    #[serde(default)]
154    border: Option<String>,
155    #[serde(default)]
156    border_width: Option<u8>,
157    #[serde(default)]
158    border_end: Option<String>,
159    #[serde(default)]
160    border_dir: GradientDir,
161}
162
163/// What `icon = { ... }` inside an icon names.
164#[derive(Deserialize)]
165#[serde(rename_all = "snake_case")]
166enum RefDe {
167    Default,
168    None,
169    Image { path: String },
170    SheetCell { sheet: String, cell: u32 },
171}
172
173impl TryFrom<IconDe> for Icon {
174    type Error = String;
175
176    fn try_from(de: IconDe) -> Result<Self, String> {
177        let art = match (de.icon, de.sheet, de.cell, de.image) {
178            (Some(RefDe::SheetCell { sheet, cell }), ..) | (None, Some(sheet), Some(cell), _) => {
179                Art::Cell { sheet, cell }
180            }
181            (Some(RefDe::Image { path }), ..) | (None, None, None, Some(path)) => Art::Image(path),
182            (Some(RefDe::Default | RefDe::None), ..) => Art::None,
183            _ => return Err("an icon names a sheet and its cell, or an image".to_owned()),
184        };
185        if let Art::Cell { cell: 0, .. } = art {
186            return Err("a sheet's cells are counted from 1".to_owned());
187        }
188        Ok(Self {
189            art,
190            grayscale: de.grayscale,
191            border: de.border,
192            border_width: de.border_width,
193            border_end: de.border_end,
194            border_dir: de.border_dir,
195        })
196    }
197}

199/// An icon as Hydra writes it.
200#[derive(Serialize)]
201struct IconSer {
202    #[serde(skip_serializing_if = "Option::is_none")]
203    sheet: Option<String>,
204    #[serde(skip_serializing_if = "Option::is_none")]
205    cell: Option<u32>,
206    #[serde(skip_serializing_if = "Option::is_none")]
207    icon: Option<RefSer>,
208    #[serde(skip_serializing_if = "std::ops::Not::not")]
209    grayscale: bool,
210    #[serde(skip_serializing_if = "Option::is_none")]
211    border: Option<String>,
212    #[serde(skip_serializing_if = "Option::is_none")]
213    border_width: Option<u8>,
214    #[serde(skip_serializing_if = "Option::is_none")]
215    border_end: Option<String>,
216    #[serde(skip_serializing_if = "GradientDir::is_default")]
217    border_dir: GradientDir,
218}
219
220#[derive(Serialize)]
221#[serde(rename_all = "snake_case")]
222enum RefSer {
223    None,
224    Image { path: String },
225}
226
227impl From<Icon> for IconSer {
228    fn from(icon: Icon) -> Self {
229        let (sheet, cell, art) = match icon.art {
230            Art::Cell { sheet, cell } => (Some(sheet), Some(cell), None),
231            Art::Image(path) => (None, None, Some(RefSer::Image { path })),
232            Art::None => (None, None, Some(RefSer::None)),
233        };
234        Self {
235            sheet,
236            cell,
237            icon: art,
238            grayscale: icon.grayscale,
239            border: icon.border,
240            border_width: icon.border_width,
241            border_end: icon.border_end,
242            border_dir: icon.border_dir,
243        }
244    }
245}

247/// A sheet of icons: its picture, and its cells' edge.
248#[derive(Clone, Debug, PartialEq, Eq, Deserialize, Serialize)]
249pub struct Sheet {
250    /// Its picture: in the icons folder, or anywhere.
251    pub path: String,
252    /// Its cells' edge in pixels; [`CELL`] unless it says.
253    #[serde(default, skip_serializing_if = "Option::is_none")]
254    pub cell: Option<u32>,
255}
256
257impl Sheet {
258    /// Its cells' edge, in pixels.
259    #[must_use]
260    pub fn cell(&self) -> u32 {
261        self.cell.filter(|cell| *cell > 0).unwrap_or(CELL)
262    }
263}

265/// The icons folder's sheets, by name.
266#[derive(Clone, Debug, Default, PartialEq, Eq, Deserialize, Serialize)]
267pub struct Sheets {
268    /// Each sheet, by name.
269    #[serde(default)]
270    pub sheets: BTreeMap<String, Sheet>,
271}
272
273impl Sheets {
274    /// Read from `text`.
275    ///
276    /// # Errors
277    ///
278    /// Why it does not read.
279    pub fn parse(text: &str) -> Result<Self, String> {
280        toml::from_str(text).map_err(|why| why.to_string())
281    }
282
283    /// The manifest in the icons folder `dir`; none there, no sheets.
284    ///
285    /// # Errors
286    ///
287    /// It is there and does not read.
288    pub fn load(dir: &Path) -> Result<Self, String> {
289        match std::fs::read_to_string(dir.join(SHEETS)) {
290            Ok(text) => Self::parse(&text).map_err(|why| format!("{SHEETS}: {why}")),
291            Err(why) if why.kind() == std::io::ErrorKind::NotFound => Ok(Self::default()),
292            Err(why) => Err(format!("{SHEETS} cannot be read: {why}")),
293        }
294    }
295}

297/// Where `art` is drawn from, the icons folder `dir` and its `sheets`
298/// given: the picture's file, and for a sheet's cell, the cell's place in
299/// it as counted from 1 and its edge. None for no art, or a sheet not
300/// named.
301#[must_use]
302pub fn source(art: &Art, dir: &Path, sheets: &Sheets) -> Option<(PathBuf, Option<(u32, u32)>)> {
303    match art {
304        Art::None => None,
305        Art::Image(path) => Some((dir.join(path), None)),
306        Art::Cell { sheet, cell } => {
307            let sheet = sheets.sheets.get(sheet)?;
308            Some((dir.join(&sheet.path), Some((*cell, sheet.cell()))))
309        }
310    }
311}

313/// The cell `cell`, counted from 1, of a sheet `width` by `height` pixels
314/// in cells of `edge`: its left, top, right and bottom, each a fraction of
315/// the sheet. None for a cell past the sheet's last.
316#[must_use]
317pub fn cell_in(cell: u32, edge: u32, (width, height): (u32, u32)) -> Option<[f32; 4]> {
318    let (across, down) = (width / edge.max(1), height / edge.max(1));
319    if cell == 0 || across == 0 || cell > across * down {
320        return None;
321    }
322    let (column, row) = ((cell - 1) % across, (cell - 1) / across);
323    #[expect(
324        clippy::cast_precision_loss,
325        reason = "a sheet is a few thousand pixels"
326    )]
327    let part = |at: u32, of: u32| (at * edge) as f32 / of as f32;
328    Some([
329        part(column, width),
330        part(row, height),
331        part(column + 1, width),
332        part(row + 1, height),
333    ])

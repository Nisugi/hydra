1//! Creature tags: a short unique label shown after each creature's name, so
2//! creatures with the same name can be told apart. The game's own id cannot
3//! serve: the player never sees it, and it is just a string of numbers.
4//!
5//! **A tag is handed out, not worked out from the id.** What matters most is
6//! that the first character is unique in the room, so a tag need not be
7//! derived from the id; it is assigned to it. Each creature is given a
8//! [`Mark`] when the registry first sees it: a character no other creature in
9//! the room has, and random ones, kept in its instance and released when the
10//! creature is cleaned up.
11//!
12//! **What a tag shows is the player's to choose** ([`Style`]): three slots,
13//! each the unique character, a random one, a separator of the player's
14//! (`-`, `/`), or nothing; and the creature's health as a percent, three
15//! digits, before or after, or not at all. `A7-042`, `042/A`, `A7K`.
16//!
17//! Not here yet: writing the tag into the lines a player is shown (it comes
18//! with the lines published to viewers), and the commands that name a creature
19//! by its tag (`tkill A7`; with the extras).
20
21/// The characters a tag is made of: the letters first, then the digits.
22const ALPHABET: &[u8; 36] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
23
24/// The characters handed to one creature: the one no other creature in its
25/// room had when it was first seen, and a random one for each slot.
26#[derive(Clone, Copy, Debug, PartialEq, Eq)]
27pub struct Mark {
28    /// Unique in the room it was first seen in.
29    pub unique: char,
30    /// Random, one for each of the three slots.
31    pub random: [char; 3],
32}
33
34/// What one of a tag's three slots shows.
35#[derive(Clone, Copy, Debug, PartialEq, Eq)]
36pub enum Slot {
37    /// The creature's unique character.
38    Unique,
39    /// A random character of its own.
40    Random,
41    /// This character, the same for every creature: `-`, `/`.
42    Separator(char),
43    /// Nothing.
44    None,
45}
46
47impl Slot {
48    /// As the settings write it: `unique`, `random`, `none`, or the
49    /// separator itself.
50    #[must_use]
51    pub fn word(self) -> String {
52        match self {
53            Self::Unique => "unique".to_owned(),
54            Self::Random => "random".to_owned(),
55            Self::None => "none".to_owned(),
56            Self::Separator(c) => c.to_string(),
57        }
58    }
59
60    /// From the settings' `word`: a separator is one character that is not
61    /// a letter or a digit, so it is never taken for part of a tag.
62    #[must_use]
63    pub fn of(word: &str) -> Option<Self> {
64        let word = word.trim();
65        match word.to_ascii_lowercase().as_str() {
66            "unique" => Some(Self::Unique),
67            "random" => Some(Self::Random),
68            "none" | "" => Some(Self::None),
69            _ => {
70                let mut chars = word.chars();
71                match (chars.next(), chars.next()) {
72                    (Some(c), None) if !c.is_alphanumeric() && !c.is_whitespace() => {
73                        Some(Self::Separator(c))
74                    }
75                    _ => None,
76                }
77            }
78        }
79    }
80}

82/// Where a tag shows the creature's health.
83#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
84pub enum Health {
85    /// Not at all.
86    #[default]
87    Off,
88    /// Before its characters: `042/A`.
89    Front,
90    /// After them: `A7-042`.
91    Back,
92}
93
94impl Health {
95    /// As the settings write it.
96    #[must_use]
97    pub const fn word(self) -> &'static str {
98        match self {
99            Self::Off => "off",
100            Self::Front => "front",
101            Self::Back => "back",
102        }
103    }
104
105    /// From the settings' `word`.
106    #[must_use]
107    pub fn of(word: &str) -> Option<Self> {
108        match word.trim().to_ascii_lowercase().as_str() {
109            "off" | "none" => Some(Self::Off),
110            "front" => Some(Self::Front),
111            "back" => Some(Self::Back),
112            _ => None,
113        }
114    }
115}

117/// A character's settings file's `targetid` section: whether its creatures'
118/// tags are shown (`.targetid`, the General page). Only the switch so far;
119/// the look is [`Style`]'s default.
120#[derive(Clone, Debug, Default, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
121pub struct Shown {
122    /// Shown; off when unset.
123    #[serde(default, skip_serializing_if = "Option::is_none")]
124    pub enabled: Option<bool>,
125}
126
127impl Shown {
128    /// The section's name in the settings file.
129    pub const SECTION: &str = "targetid";
130
131    /// How tags look while shown; `None` while off.
132    #[must_use]
133    pub fn style(&self) -> Option<Style> {
134        self.enabled.unwrap_or(false).then(Style::default)
135    }
136}

138/// How a tag looks: its three slots and its health.
139#[derive(Clone, Copy, Debug, PartialEq, Eq)]
140pub struct Style {
141    /// The three slots, in order.
142    pub slots: [Slot; 3],
143    /// Where the health goes.
144    pub health: Health,
145}
146
147impl Default for Style {
148    /// Three characters, the first unique: `A7K`.
149    fn default() -> Self {
150        Self {
151            slots: [Slot::Unique, Slot::Random, Slot::Random],
152            health: Health::Off,
153        }
154    }
155}

157/// A mark for a new creature, `id`, in a room whose other creatures'
158/// unique characters are `taken`: the first letter free, a digit when the
159/// letters are gone, and when all are (never yet), the first again. Its
160/// random characters are drawn from the id, so the same creature seen
161/// twice from nothing gets the same ones.
162#[must_use]
163pub fn assign(id: i64, taken: impl IntoIterator<Item = char>) -> Mark {
164    let taken: Vec<char> = taken.into_iter().collect();
165    let unique = ALPHABET
166        .iter()
167        .map(|b| char::from(*b))
168        .find(|c| !taken.contains(c))
169        .unwrap_or('A');
170    let mut bits = id.unsigned_abs().wrapping_mul(0x9E37_79B9_7F4A_7C15);
171    let mut random = ['A'; 3];
172    for slot in &mut random {
173        *slot = char::from(ALPHABET[usize::try_from(bits % 36).unwrap_or(0)]);
174        bits /= 36;
175    }
176    Mark { unique, random }
177}

179/// A creature's mark and its health, a whole percent 0 to 100, as a tag
180/// wants them; `None` before the registry has handed it a mark.
181#[must_use]
182pub fn of(creature: &crate::state::creatures::CreatureInstance) -> Option<(Mark, u32)> {
183    // Clamped first, so the cast neither wraps nor truncates past the range.
184    #[expect(
185        clippy::cast_possible_truncation,
186        clippy::cast_sign_loss,
187        reason = "clamped to 0..=100 before the cast"
188    )]
189    let health = creature.hp_percent().round().clamp(0.0, 100.0) as u32;
190    Some((creature.mark()?, health))
191}

193/// The tag `style` makes of `mark`, with `health` a percent when it shows.
194#[must_use]
195pub fn tag(mark: Mark, style: Style, health: Option<u32>) -> String {
196    let characters: String = style
197        .slots
198        .iter()
199        .enumerate()
200        .filter_map(|(at, slot)| match slot {
201            Slot::Unique => Some(mark.unique),
202            Slot::Random => Some(mark.random[at]),
203            Slot::Separator(c) => Some(*c),
204            Slot::None => None,
205        })
206        .collect();
207    let health = health.map(|percent| format!("{:03}", percent.min(100)));
208    match (style.health, health) {
209        (Health::Front, Some(health)) => format!("{health}{characters}"),
210        (Health::Back, Some(health)) => format!("{characters}{health}"),
211        _ => characters,
212    }
213}

215#[cfg(test)]
216mod tests {
217    use super::{Health, Mark, Slot, Style, assign, tag};
218
219    const MARK: Mark = Mark {
220        unique: 'A',
221        random: ['K', '7', 'Q'],
222    };
223
224    /// The unique character is the first letter no other creature here
225    /// has, then the digits; the random ones come from the id, the same
226    /// each time.
227    #[test]
228    fn a_mark_is_handed_out_letters_first() {
229        assert_eq!(assign(5, []).unique, 'A');
230        assert_eq!(assign(5, ['A', 'B']).unique, 'C');
231        let letters: Vec<char> = ('A'..='Z').collect();
232        assert_eq!(assign(5, letters).unique, '0');
233        assert_eq!(assign(99, []).random, assign(99, []).random);
234    }
235
236    /// The shapes a player may choose: the slots in order, a separator as given,
237    /// nothing for none, the health three digits before or after.
238    #[test]
239    fn a_tag_is_its_slots_and_its_health() {
240        let style = |slots, health| Style { slots, health };
241        assert_eq!(tag(MARK, Style::default(), Some(42)), "A7Q");
242        assert_eq!(
243            tag(
244                MARK,
245                style(
246                    [Slot::Unique, Slot::Random, Slot::Separator('-')],
247                    Health::Back
248                ),
249                Some(42)
250            ),
251            "A7-042"
252        );
253        assert_eq!(
254            tag(
255                MARK,
256                style(
257                    [Slot::Separator('/'), Slot::Unique, Slot::None],
258                    Health::Front
259                ),
260                Some(100)
261            ),
262            "100/A"
263        );
264        assert_eq!(Slot::of("-"), Some(Slot::Separator('-')));
265        assert_eq!(Slot::of("unique"), Some(Slot::Unique));
266        assert_eq!(Slot::of("7"), None, "never a letter or digit");
267        assert_eq!(Health::of("front"), Some(Health::Front));
268    }

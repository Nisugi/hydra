1//! The room the character is in.
2//!
3//! # The room is several independent feeds
4//!
5//! `<compDef>` is the snapshot on arrival and `<component>` the live update,
6//! the same ids and the same body shape; the parser gives both as
7//! [`Frame::Component`](hydra_protocol::Frame::Component). Each id replaces
8//! only its own entry: creatures and players come and go one update at a time,
9//! and an empty body ("nobody here") is a real update. So the room keeps each
10//! component's body as it came, beside the typed lists read from it.
11//!
12//! # Creatures are the bold entries of `room objs`
13//!
14//! There is no separate creature feed: `<pushBold/>` is the only thing that
15//! tells a creature from an object in `room objs`.
16//!
17//! # Where the character is, not what it saw
18//!
19//! Only the window feed (`<nav>`, `<streamWindow id='room'>`, the room
20//! components) moves the room. The story window's styled room text is what
21//! the character saw, and abilities that look into another room send it
22//! without the window feed; reading it here would move the character
23//! wherever it looked.
24
25use std::collections::BTreeMap;

27use hydra_protocol::RoomMeta;
28use hydra_protocol::frame::LinkKind;
29use hydra_protocol::runs::{Run, Runs};

31/// One thing in the room a command can name: a creature, an object or a
32/// player, from one `<a exist= noun=>` link.
33#[derive(Clone, Debug, PartialEq, Eq)]
34pub struct RoomItem {
35    /// `exist=`, verbatim. Negative for fixtures and players, so not a number.
36    pub id: String,
37    /// `noun=`: what a command targets it by.
38    pub noun: String,
39    /// The link's text, e.g. `gaunt masked artificer`.
40    pub text: String,
41    /// The line's prose before the link, trimmed: `a`, `an`, `some`. Read
42    /// for a container's one-item lines; a room listing names many things on
43    /// one line, so there the prose between links belongs to neither and this
44    /// is `None`.
45    pub before: Option<String>,
46    /// The line's prose after the link, trimmed: `scuffed at the corners`,
47    /// `is closed.`. The game's words, not parsed into cases. As
48    /// [`Self::before`], read for container lines only.
49    pub after: Option<String>,
50    /// What the room says this player is doing: `hiding`, `sitting`, `dead`.
51    /// Read for `room players` only; `None` when the room said nothing more.
52    pub status: Option<PlayerStatus>,
53}
54
55/// What a player in the room is doing, as the room put it. The game's words,
56/// joined with a space; not a closed set.
57#[derive(Clone, Debug, PartialEq, Eq)]
58pub struct PlayerStatus(String);
59
60impl PlayerStatus {
61    /// The status as stated, e.g. `hiding`, `lying down`.
62    #[must_use]
63    pub fn as_str(&self) -> &str {
64        &self.0
65    }
66
67    /// Is this player hiding? A hiding group member is named in the roster,
68    /// unlike a hiding stranger.
69    #[must_use]
70    pub fn is_hiding(&self) -> bool {
71        self.0.split_whitespace().any(|word| word == "hiding")
72    }
73}
74
75impl std::fmt::Display for PlayerStatus {
76    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
77        f.write_str(&self.0)
78    }
79}

81/// The room the character is in, as the wire stated it.
82#[derive(Clone, Debug, Default, PartialEq, Eq)]
83pub struct Room {
84    /// `<nav rm=>`: the game's own id.
85    pub id: Option<String>,
86    /// `<streamWindow id='room' subtitle=>`, less its leading ` - `. Kept as
87    /// the wire states it, with ` - 7503251` after it when the player shows
88    /// room numbers. Arrives after `<nav>`, so `None` for a moment each move.
89    pub title: Option<String>,
90    /// `room desc`. Also in [`Self::component`].
91    pub description: Option<Runs>,
92    /// `<compass>`: direction tokens.
93    ///
94    /// Three states: `None` is not told yet; `Some(vec![])` is told there are
95    /// no cardinal exits, which is real twice over -- a room left only by a
96    /// door or portal, and a room with no way out at all.
97    pub exits: Option<Vec<String>>,
98    /// The bold entries of `room objs`.
99    pub creatures: Vec<RoomItem>,
100    /// The other entries of `room objs`: items and fixtures.
101    pub objects: Vec<RoomItem>,
102    /// The entries of `room players`.
103    pub players: Vec<RoomItem>,
104    /// `<roommeta>`: the room's environment, as the game's own codes.
105    pub meta: Option<RoomMeta>,
106    /// Each component's body as it came, by id. Private, so that one never
107    /// sent reads as absent through [`Self::component`], not as empty.
108    components: BTreeMap<String, Runs>,
109}
110
111impl Room {
112    /// One component's body, if the game has sent it. `None` is never told;
113    /// an empty body is told, and empty.
114    #[must_use]
115    pub fn component(&self, id: &str) -> Option<&Runs> {
116        self.components.get(id)
117    }
118
119    /// Every component the game has sent for this room, in id order.
120    pub fn components(&self) -> impl Iterator<Item = (&str, &Runs)> {
121        self.components.iter().map(|(id, body)| (id.as_str(), body))
122    }
123
124    /// Whether the game has said who is in the room. An empty
125    /// [`Self::players`] is both "nobody" and "not told"; this tells them apart.
126    #[must_use]
127    pub fn saw_players(&self) -> bool {
128        self.components.contains_key("room players")
129    }
130
131    /// Forget who and what is here, keeping the place: for a reconnect.
132    ///
133    /// The id, name, description, exits and environment describe somewhere
134    /// that cannot have changed while the character was out of the world.
135    /// The roster is the opposite: creatures wander and players come and go
136    /// whether or not we watch, and each entry is an id a command could
137    /// target. The raw bodies of the roster components go too, so
138    /// [`Self::saw_players`] answers "not told" rather than "nobody here".
139    pub(super) fn forget_contents(&mut self) {
140        self.creatures.clear();
141        self.objects.clear();
142        self.players.clear();
143        self.components.retain(|key, _| {
144            !matches!(
145                key.as_str(),
146                "room objs" | "room players" | "room creatures"
147            )
148        });
149    }
150
151    /// `<nav>`: an arrival, unless it names the room already here.
152    ///
153    /// A new room forgets everything about the last one: its creatures above
154    /// all, which a command must never be aimed at from the next room.
155    ///
156    /// The login burst re-declares the room the character stands in, with no
157    /// compass after it, so the same id keeps what is known. Only a stated id
158    /// can be the same: two id-less arrivals are two rooms as far as anything
159    /// here can tell.
160    ///
161    /// `true` when it was an arrival.
162    pub(super) fn arrive(&mut self, id: Option<&str>) -> bool {
163        let same_room = id.is_some() && id == self.id.as_deref();
164        if !same_room {
165            *self = Self {
166                id: id.map(str::to_owned),
167                ..Self::default()
168            };
169        }
170        !same_room
171    }
172
173    /// `<streamWindow>`: the `room` window's subtitle names the room. Only
174    /// that window's: `main` carries the same text as its own caption, and
175    /// other windows' subtitles are about themselves.
176    pub(super) fn name(&mut self, window: &str, subtitle: Option<&str>) {
177        if window == "room"
178            && let Some(subtitle) = subtitle
179        {
180            let name = subtitle.strip_prefix(" - ").unwrap_or(subtitle);
181            self.title = Some(name.to_owned());
182        }
183    }
184
185    /// One component body: it replaces its own entry and touches no other.
186    pub(super) fn apply_component(&mut self, id: &str, body: &Runs) {
187        match id {
188            "room desc" => self.description = Some(body.clone()),
189            "room objs" => {
190                self.creatures = items(body, true, false);
191                self.objects = items(body, false, false);
192            }
193            "room players" => self.players = items(body, false, true),
194            _ => {}
195        }
196        self.components.insert(id.to_owned(), body.clone());
197    }
198}

200/// The linked things in a body, bold or not, each once.
201///
202/// A link the game styles in parts (a preset or a bold change inside the
203/// `<a>`) arrives as several runs that each carry it; adjacent runs naming
204/// the same object are one occurrence, its text the last run's, a creature
205/// when bold appears anywhere in it. Only `<a exist=>` links are things: a
206/// `<d>` is a command, which keeps `room exits`' directions out of the list.
207///
208/// `with_status` reads a player's status from the prose after the link. Only
209/// `room players` asks: for objects it would give a disk a player's `hiding`.
210fn items(body: &Runs, bold: bool, with_status: bool) -> Vec<RoomItem> {
211    let runs = &body.runs;
212    let mut occurrences: Vec<(usize, usize)> = Vec::new();
213    for (index, run) in runs.iter().enumerate() {
214        let Some(id) = exist_id(run) else {
215            continue;
216        };
217        match occurrences.last_mut() {
218            Some((_, last)) if *last + 1 == index && exist_id(&runs[*last]) == Some(id) => {
219                *last = index;
220            }
221            _ => occurrences.push((index, index)),
222        }
223    }
224    occurrences
225        .into_iter()
226        .filter(|&(first, last)| {
227            runs[first..=last]
228                .iter()
229                .any(|run| run.style.bold_depth > 0)
230                == bold
231        })
232        .filter_map(|(_, last)| {
233            let link = runs[last].link.as_ref()?;
234            let LinkKind::Exist { id, noun } = &link.kind else {
235                return None;
236            };
237            let status = if with_status {
238                runs.get(last + 1)
239                    .filter(|next| next.link.is_none())
240                    .and_then(|next| parse_status(&next.text))
241            } else {
242                None
243            };
244            Some(RoomItem {
245                id: id.clone(),
246                noun: noun.clone(),
247                text: link.text.clone(),
248                before: None,
249                after: None,
250                status,
251            })
252        })
253        .collect()
254}

256/// The object a run's link names, when it names one.
257fn exist_id(run: &Run) -> Option<&str> {
258    match &run.link.as_ref()?.kind {
259        LinkKind::Exist { id, .. } => Some(id.as_str()),
260        _ => None,
261    }
262}

264/// A status clause in the prose after a player's link.
265///
266/// Two forms: ` who is hiding,` / ` who appears dead.`, and ` (hiding)`, which
267/// may come twice and is joined with a space.
268/// An ordinary separator (`, `, ` and `, `.`) is no status.
269fn parse_status(text: &str) -> Option<PlayerStatus> {
270    if let Some(rest) = text
271        .strip_prefix(" who is ")
272        .or_else(|| text.strip_prefix(" who appears "))
273    {
274        let status: String = rest
275            .chars()
276            .take_while(|c| c.is_alphanumeric() || c.is_whitespace())
277            .collect();
278        let status = status.trim();
279        if !status.is_empty() {
280            return Some(PlayerStatus(status.to_owned()));
281        }
282    }
283
284    let mut parts: Vec<&str> = Vec::new();
285    let mut rest = text;
286    while let Some(open) = rest.find('(') {
287        let Some(close) = rest[open..].find(')') else {
288            break;
289        };
290        let inner = rest[open + 1..open + close].trim();
291        if !inner.is_empty() && inner.chars().all(|c| c.is_alphanumeric() || c == ' ') {
292            parts.push(inner);
293        }
294        rest = &rest[open + close + 1..];
295    }
296    if parts.is_empty() {
297        None
298    } else {
299        Some(PlayerStatus(parts.join(" ")))
300    }

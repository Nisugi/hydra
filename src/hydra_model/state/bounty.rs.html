1//! Bounty tasks: what the Adventurer's Guild has asked for.
2//!
3//! The bounty report, and the questions a parsed task answers. A stateless
4//! classifier over one command's output, the shape `skills.rs` and `psm.rs`
5//! already take.
6//!
7//! # Two stages, and the wire distinguishes them
8//!
9//! A bounty has an **assignment** -- "go see the gem dealer" -- and then a
10//! **task** with requirements once you ask about it. The matcher table covers
11//! both, which is why there is a `GemAssignment` beside a `Gem`, and why a
12//! caller checking "do I have work to do" wants [`TaskKind::is_actionable`]
13//! rather than merely "is it not `None`".
14//!
15//! # ORDER IS LOAD-BEARING
16//!
17//! The matchers are walked in declaration order and the first match wins.
18//! That is not incidental: `Bandit` is declared before `Cull` and its pattern
19//! is `Cull`'s with `bandit` substituted for the creature capture, so a
20//! bandit task matches **both** and only the order decides. Same for
21//! `DangerousSpawned` before `Dangerous`.
22//!
23//! The matcher table preserves that order and `the_order_is_load_bearing` in the
24//! tests asserts the two pairs specifically, so a future sort or a `HashMap`
25//! cannot quietly reverse them.
26//!
27//! # The town is not always the captured town
28//!
29//! The town is overridden in five cases, four of them because the guard
30//! phrasing names no town at all:
31//!
32//! | Phrase | Town |
33//! |---|---|
34//! | `the sentry just outside town.` | Kraken's Fall |
35//! | `the tavernkeeper at Rawknuckle's Common House.` | Cold River |
36//! | `the elderly guard in the East Guardtower.` | Mist Harbor |
37//! | mentions Captain, Reiya, Ataum or Galeb | Contempt |
38//! | `gem dealer in has received` | Contempt |
39//!
40//! **The last is a workaround for a live typo** in the game's messaging,
41//! to be removed only if the game ever fixes it. Commented where it lives,
42//! because a reader who finds it and "fixes" it breaks Contempt gem bounties.
43

52/// What kind of bounty this is.
53///
54/// The `*Assignment` variants are the guild's referral; the others are the
55/// task itself, with requirements.
56#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
57pub enum TaskKind {
58    /// No task assigned.
59    None,
60    /// Go see someone about a bandit problem.
61    BanditAssignment,
62    /// Go see someone about a creature problem.
63    CreatureAssignment,
64    /// Go see the gem dealer.
65    GemAssignment,
66    /// Go track down a lost heirloom.
67    HeirloomAssignment,
68    /// Go see the herbalist.
69    HerbAssignment,
70    /// A local resident needs help.
71    RescueAssignment,
72    /// Go see the furrier.
73    SkinAssignment,
74    /// The task is done; report to the guild.
75    Taskmaster,
76    /// The heirloom is found; return it.
77    HeirloomFound,
78    /// The task is done; report to a specific guard.
79    Guard,
80    /// A dangerous creature has been provoked and must be killed.
81    DangerousSpawned,
82    /// The child has been found and must be escorted back.
83    RescueSpawned,
84    /// Kill N bandits.
85    Bandit,
86    /// Hunt a dangerous creature, which must first be provoked.
87    Dangerous,
88    /// Escort a client between two places.
89    Escort,
90    /// Collect N gems.
91    Gem,
92    /// Recover a lost heirloom from a creature's territory.
93    Heirloom,
94    /// Collect N herb samples.
95    Herb,
96    /// Clear creatures until the child appears.
97    Rescue,
98    /// Collect N skins of a given quality.
99    Skin,
100    /// Kill N of a creature type.
101    Cull,
102    /// The task was failed.
103    Failed,
104}
105
106impl TaskKind {
107    /// Is there work to do right now?
108    ///
109    /// **False for an assignment**, which is a referral rather than a task:
110    /// the character must go and ASK about BOUNTIES before there is anything
111    /// to hunt. Also false for the terminal states.
112    #[must_use]
113    pub const fn is_actionable(self) -> bool {
114        matches!(
115            self,
116            Self::Bandit
117                | Self::Cull
118                | Self::Dangerous
119                | Self::DangerousSpawned
120                | Self::Escort
121                | Self::Gem
122                | Self::Heirloom
123                | Self::HeirloomFound
124                | Self::Herb
125                | Self::Rescue
126                | Self::RescueSpawned
127                | Self::Skin
128        )
129    }
130
131    /// Is the task finished, one way or the other?
132    #[must_use]
133    pub const fn is_done(self) -> bool {
134        matches!(self, Self::Taskmaster | Self::Guard | Self::Failed)
135    }
136
137    /// Is this a referral rather than a task?
138    #[must_use]
139    pub const fn is_assignment(self) -> bool {
140        matches!(
141            self,
142            Self::BanditAssignment
143                | Self::CreatureAssignment
144                | Self::GemAssignment
145                | Self::HeirloomAssignment
146                | Self::HerbAssignment
147                | Self::RescueAssignment
148                | Self::SkinAssignment
149        )
150    }
151}

153/// A parsed bounty.
154#[derive(Debug, Clone, PartialEq, Eq)]
155pub struct Task {
156    /// Which kind.
157    pub kind: TaskKind,
158    /// The town the task belongs to, where one is known.
159    pub town: Option<String>,
160    /// Every named capture the pattern produced, by name.
161    ///
162    /// **A map rather than named fields**, and this is the one place in the
163    /// character model where typed fields lose: the capture set differs per
164    /// task kind (a skin task has `quality`, a gem task has `gem`, an escort
165    /// has `start` and `destination`), so named fields would be twenty
166    /// mostly-`None` options.
167    pub requirements: BTreeMap<String, String>,
168}
169
170impl Task {
171    /// One requirement, if the pattern captured it.
172    #[must_use]
173    pub fn requirement(&self, name: &str) -> Option<&str> {
174        self.requirements.get(name).map(String::as_str)
175    }
176
177    /// How many of the thing are needed, where the task counts.
178    ///
179    /// Returns `None` rather than `0` for a task with no count, because "no count" and "zero
180    /// left" are different answers.
181    #[must_use]
182    pub fn number(&self) -> Option<u32> {
183        self.requirement("number")?.parse().ok()
184    }
185
186    /// The creature to kill, normalised.
187    #[must_use]
188    pub fn creature(&self) -> Option<&str> {
189        self.requirement("creature")
190    }
191
192    /// Where the task takes place.
193    #[must_use]
194    pub fn area(&self) -> Option<&str> {
195        self.requirement("area")
196    }
197}

199/// The task patterns, **in match order**.
200///
201/// See the module docs: the order decides between `Bandit`/`Cull` and
202/// `DangerousSpawned`/`Dangerous`, whose patterns overlap.
203type Matcher = (TaskKind, &'static str);

205/// Shared sub-patterns, expanded into the matchers below.
206///
207/// Named for what they match.
208mod frag {
209    /// The guild's preamble, optional.
210    pub const HMM: &str = r"(?:Hmm, I've got a task here from .*?(?P<town>[A-Z].*?)\..*?)?";
211    /// Where a task takes place.
212    pub const LOCATION: &str = r"(?:on|in|near) (?:the\s+)?(?P<area>[^.]+?)(?:\s+(?:near|between|under) (?P<town2>[^.]+))?";
213    /// The many ways the game names a guard to report to.
214    pub const GUARD: &str = r"(?:one of the guardsmen just inside the (?P<town3>Ta'Illistim) City Gate|one of the guardsmen just inside the Sapphire Gate|one of the guardsmen just inside the gate|one of the (?P<town4>.*) (?:gate|tunnel) guards|one of the (?P<town5>Icemule Trace) gate guards or the halfing Belle at the Pinefar Trading Post|Quin Telaren of (?P<town6>Wehnimer's Landing)|the dwarven militia sergeant near the (?P<town7>Kharam-Dzu) town gates|the sentry just outside town|the sentry just outside (?P<town8>Kraken's Fall)|the purser of (?P<town9>River's Rest)|the tavernkeeper at Rawknuckle's Common House|the captain of the (?P<town10>Contempt)|the elderly guard in the East Guardtower)";
215    /// The herb task's body.
216    pub const CONCOCTION: &str = r"is working on a concoction that requires (?:an?|some|several) (?P<herb>[^.]+?) found [oi]n (?:the\s+)?(?P<area>[^.]+?)(?:\s+(?:near|under|between) [^.]+)?\.  These samples must be in pristine condition\.  You have been tasked to retrieve (?P<number>\d+) (?:more\s+)?samples?\.";
217    /// The taskmaster's quoted preamble.
218    pub const MAYBE: &str = r"(?:The taskmaster told you:  \x22)";
219}

221/// Build the ordered matcher table.
222///
223/// Rust's `regex` crate has no alternation of named groups with the same
224/// name -- so the `GUARD` union numbers its town captures
225/// `town3`..`town10` and [`town_from`] reads whichever fired.
226fn matchers() -> &'static Vec<(TaskKind, Regex)> {
227    static MATCHERS: OnceLock<Vec<(TaskKind, Regex)>> = OnceLock::new();
228    MATCHERS.get_or_init(|| {
229        let specs: Vec<Matcher> = vec![
230            (TaskKind::None, r"^You are not currently assigned a task"),
231            (
232                TaskKind::BanditAssignment,
233                concat_frag(
234                    frag::HMM,
235                    r"It appears they have a bandit problem they'd like you to solve",
236                ),
237            ),
238        ];
239        // The table is built in two halves only because a `concat!` of
240        // runtime strings needs a helper; the ORDER across both is the
241        // declaration order and must not change.
242        let mut out: Vec<(TaskKind, Regex)> = Vec::new();
243        for (kind, pattern) in specs {
244            if let Ok(re) = Regex::new(pattern) {
245                out.push((kind, re));
246            }
247        }
248        for (kind, pattern) in remaining_specs() {
249            if let Ok(re) = Regex::new(&pattern) {
250                out.push((kind, re));
251            }
252        }
253        out
254    })
255}

257/// Join a fragment to a literal.
258fn concat_frag(prefix: &str, rest: &str) -> &'static str {
259    // Leaked deliberately and once: `matchers()` is a `OnceLock`, so this runs
260    // at most once per pattern for the life of the process, and the
261    // alternative is threading owned `String`s through a `&'static` table for
262    // no gain.
263    Box::leak(format!("{prefix}{rest}").into_boxed_str())
264}

266/// The matchers that need runtime string assembly.
267///
268/// Declaration order continues from [`matchers`]; see the module docs on why
269/// it matters.
270fn remaining_specs() -> Vec<(TaskKind, String)> {
271    // Split into three only to respect the 100-line function cap; the ORDER
272    // across all three is the declaration order and must not change. See the
273    // module docs.
274    let mut out = assignment_specs();
275    out.extend(terminal_specs());
276    out.extend(task_specs());
277    out
278}

433/// A capture group's requirement name: the group's name without the suffix
434/// that tells its copies apart, `2` or `2b` -- `town2b` is `town`.
435fn group_key(name: &str) -> &str {
436    let letters = name.trim_end_matches(|c: char| c.is_ascii_lowercase());
437    let base = if letters.ends_with(|c: char| c.is_ascii_digit()) {
438        letters
439    } else {
440        name
441    };
442    base.trim_end_matches(|c: char| c.is_ascii_digit())
443}

445/// Normalise a creature name.
446///
447/// Two families the game qualifies with an adjective that is not part of the
448/// creature's identity for hunting purposes.
449fn normalize_creature(raw: &str) -> String {
450    let words: Vec<&str> = raw.split_whitespace().collect();
451    match words.as_slice() {
452        [_, "being"] => "being".to_owned(),
453        [_, "magna", "vereri"] => "magna vereri".to_owned(),
454        _ => raw.to_owned(),
455    }
456}

458/// Decide the town.
459///
460/// Four overrides exist because the guard phrasing names no town; the fifth is
461/// a workaround for a typo in the game's own messaging.
462fn town_from(description: &str, captured: Option<&str>) -> Option<String> {
463    if description.ends_with("the sentry just outside town.") {
464        return Some("Kraken's Fall".to_owned());
465    }
466    if description.ends_with("the tavernkeeper at Rawknuckle's Common House.") {
467        return Some("Cold River".to_owned());
468    }
469    if description.ends_with("the elderly guard in the East Guardtower.") {
470        return Some("Mist Harbor".to_owned());
471    }
472    // **The typo workaround**, for an actual typo in the game's messaging.
473    // Remove it only if the game ever fixes the typo: removing it now breaks
474    // Contempt gem bounties.
475    if description.contains("gem dealer in has received") {
476        return Some("Contempt".to_owned());
477    }
478    for name in ["Captain", "Reiya", "Ataum", "Galeb"] {
479        if description
480            .split(|c: char| !c.is_alphanumeric() && c != '\'')
481            .any(|word| word == name)
482        {
483            return Some("Contempt".to_owned());
484        }
485    }
486    captured.map(str::to_owned)
487}

489/// Classify one bounty description.
490///
491/// Returns `None` for text that is not a bounty at all -- `state.rs:306-309`'s
492/// rule that a player can say anything.
493#[must_use]
494pub fn classify(description: &str) -> Option<Task> {
495    let description = description.trim();
496    if description.is_empty() {
497        return None;
498    }
499    for (kind, regex) in matchers() {
500        let Some(captures) = regex.captures(description) else {
501            continue;
502        };
503
504        let mut requirements = BTreeMap::new();
505        let mut captured_town = None;
506        for name in regex.capture_names().flatten() {
507            let Some(value) = captures.name(name) else {
508                continue;
509            };
510            let value = value.as_str();
511            // The duplicated-name suffixes exist only because Rust's `regex`
512            // forbids two groups of one name in an alternation; strip them so
513            // a caller sees one name per requirement.
514            let key = group_key(name);
515            if key == "town" {
516                captured_town = Some(value);
517                continue;
518            }
519            let value = match key {
520                "creature" => normalize_creature(value),
521                // The action is downcased, so a caller comparing against
522                // "loot" does not have to know the wire shouts it.
523                "action" => value.to_ascii_lowercase(),
524                _ => value.to_owned(),
525            };
526            requirements.entry(key.to_owned()).or_insert(value);
527        }
528
529        let town = town_from(description, captured_town);
530        if let Some(town) = town.clone() {
531            requirements.insert("town".to_owned(), town);
532        }
533        return Some(Task {
534            kind: *kind,
535            town,
536            requirements,
537        });
538    }
539    None
540}

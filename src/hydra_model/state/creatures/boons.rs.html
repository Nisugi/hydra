1//! A creature's boons: what its name's adjectives say it can do.
2//!
3//! 34 traits, each with the adjectives that name it. The feed shows a boon
4//! only as an adjective before the creature's name, and `assess` lists them
5//! (`The <creature> appears to be flickering and robust.`), so the table is
6//! how a trait is known at all. [`assessed`] reads that line.
7//!
8//! The adjectives the bestiary strips before a lookup (`instance.rs`'s
9//! `BOON_ADJECTIVES`) are a different list, for finding the template, where
10//! this one is for knowing the trait. They mostly agree and are kept apart
11//! because they answer different questions.

51/// The trait an adjective names, ignoring case.
52#[must_use]
53pub fn trait_of(adjective: &str) -> Option<&'static str> {
54    let adjective = adjective.trim();
55    TRAITS.iter().find_map(|(name, adjectives)| {
56        adjectives
57            .iter()
58            .any(|a| a.eq_ignore_ascii_case(adjective))
59            .then_some(*name)
60    })
61}

63/// The traits an `assess` line names, in order and once each: `The
64/// <creature> appears to be flickering, robust and keen.` `None`: not
65/// such a line. An adjective the table does not know is skipped.
66#[must_use]
67pub fn assessed(line: &str) -> Option<Vec<&'static str>> {
68    let (_, rest) = line.split_once("appears to be ")?;
69    let phrase = rest.split('.').next().unwrap_or(rest).to_ascii_lowercase();
70    let mut traits = Vec::new();
71    for part in phrase.split(',').flat_map(|p| p.split(" and ")) {
72        if let Some(name) = trait_of(part)
73            && !traits.contains(&name)
74        {
75            traits.push(name);
76        }
77    }
78    Some(traits)
79}

81#[cfg(test)]
82mod tests {
83    use super::*;
84
85    #[test]
86    fn an_assess_line_is_read_as_traits() {
87        assert_eq!(
88            assessed("The kobold appears to be flickering, robust and sickly green."),
89            Some(vec!["blink", "boosted_hp", "poisonous"])
90        );
91        assert_eq!(assessed("The kobold appears to be calm."), Some(vec![]));
92        assert_eq!(assessed("You see nothing unusual."), None);
93        assert_eq!(TRAITS.len(), 34);
94    }

1//! The Order of Voln: 26 symbols, powered by favor.
2//!
3//! # Favor cost is level-scaled, and the table is the game's
4//!
5//! Every symbol costs a fraction of one number: the favor cost of Symbol of
6//! Return at the member's character level. [`BASE_FAVOR_COST_BY_LEVEL`] is that
7//! number for levels 3..=100, and each symbol carries the fraction. The table
8//! is the game wiki's: <https://gswiki.play.net/Favor#Symbol_Use_Favor_Cost>.
9//!
10//! **All 98 values were checked against that page.** The wiki also states
11//! that levels 3 to 42 and level 100 are measured in game and 43-99 are
12//! interpolated, *"accurate to within 1 or 2 favor"* -- so [`favor_cost`] is
13//! an estimate in that band, and a caller deciding whether a symbol is
14//! affordable should leave itself that margin.
15//!
16//! # Symbol of Retribution has two costs
17//!
18//! The wiki's factor table lists `Retribution (attack version)` at 0.04 and
19//! `Retribution (self-cast version)` at 0.30. A table that kept only the
20//! attack cost would report a self-cast Retribution as costing an eighth of
21//! what it does: at level 100, 87 favor against a true 653.
22//!
23//! So both are in the table, as an alternate cost the way Symbol of Blessing
24//! carries its own, and `retribution_has_both_costs` in the tests asserts it.
25//!
26//! # Free is not the same as unknown
27//!
28//! Five symbols cost nothing and one costs nothing *because it does nothing*.
29//! Recognition, Kai's Strike, Kai's Smite and Seeking are [`Cost::Free`] with a
30//! modifier of 0.00. **Symbol of Thought is different**: it has no modifier
31//! at all, because the symbol was retired -- its own summary says it *"is no
32//! longer required and will be replaced in the future"*. Both are `Free`
33//! here, since a retired symbol that cannot be used costs nothing to not use,
34//! and the distinction a caller actually wants is carried by the summary text
35//! rather than by a third cost state.
36

40/// Favor cost of one Symbol of Return activation, indexed by character level.
41///
42/// Index 0, 1 and 2 are `None`: the Order does not admit characters below
43/// level 3, so there is no cost for those levels rather than a cost of zero.
44///
45/// VERIFIED against the game wiki's Favor page, section `Symbol Use Favor
46/// Cost`: 98 values, zero disagreements.
47pub const BASE_FAVOR_COST_BY_LEVEL: [Option<u16>; 101] = {
48    let mut table = [None; 101];
49    let costs: [u16; 98] = [
50        13, 22, 32, 43, 56, 70, 85, 100, 117, 134, 151, 169, 188, 207, 226, 246, 266, 286, 307,
51        328, 349, 370, 391, 412, 434, 456, 478, 500, 522, 544, 577, 590, 613, 636, 659, 682, 705,
52        728, 751, 774, 797, 820, 843, 866, 889, 912, 935, 958, 981, 1004, 1027, 1050, 1073, 1097,
53        1121, 1145, 1169, 1193, 1217, 1241, 1265, 1289, 1313, 1337, 1361, 1385, 1409, 1433, 1457,
54        1481, 1505, 1529, 1553, 1577, 1601, 1625, 1649, 1674, 1699, 1724, 1749, 1774, 1799, 1824,
55        1849, 1874, 1899, 1924, 1949, 1974, 1999, 2024, 2049, 2074, 2099, 2124, 2149, 2174,
56    ];
57    let mut level = 3;
58    while level <= 100 {
59        table[level] = Some(costs[level - 3]);
60        level += 1;
61    }
62    table
63};

71/// What one use of a symbol costs, at this character level.
72///
73/// The product is rounded up, so a cost is never rounded down in the
74/// member's favour.
75///
76/// Returns `None` when the symbol is free, when the level is outside 3..=100,
77/// or when the level has no measured cost.
78///
79/// ```
80/// use hydra_model::state::societies::voln;
81/// // Transcendence is 0.60 of the base; at level 40 that is 728 * 0.60.
82/// let transcendence = voln::symbol("transcendence").expect("rank 12 symbol");
83/// assert_eq!(voln::favor_cost(&transcendence.cost, 40), Some(437));
84/// ```
85#[must_use]
86pub fn favor_cost(cost: &Cost, level: usize) -> Option<u32> {
87    let Cost::Favor { modifier, .. } = cost else {
88        return None;
89    };
90    favor_cost_with(*modifier, level)
91}
92
93/// The alternate cost, for the two symbols that have one.
94///
95/// Returns `None` when the symbol has no alternate use, which is every symbol
96/// but Blessing and Retribution.
97#[must_use]
98pub fn alternate_favor_cost(cost: &Cost, level: usize) -> Option<u32> {
99    let Cost::Favor {
100        alternate: Some(alt),
101        ..
102    } = cost
103    else {
104        return None;
105    };
106    favor_cost_with(alt.modifier, level)
107}
108
109/// Resolve one modifier against the level table.
110fn favor_cost_with(modifier: f64, level: usize) -> Option<u32> {
111    if modifier <= 0.0 {
112        return None;
113    }
114    let base = f64::from(*BASE_FAVOR_COST_BY_LEVEL.get(level)?.as_ref()?);
115    // Rounded up. The product of a value under 2200
116    // and a modifier at most 1.0 cannot leave u32.
117    #[allow(clippy::cast_possible_truncation, clippy::cast_sign_loss)]
118    Some((base * modifier).ceil() as u32)
119}

121/// Look up one symbol by short or long name.
122///
123/// ```
124/// use hydra_model::state::societies::voln;
125/// assert_eq!(voln::symbol("holiness").map(|s| s.rank), Some(9));
126/// assert_eq!(voln::symbol("Symbol of Holiness").map(|s| s.rank), Some(9));
127/// assert!(voln::symbol("nothing of the kind").is_none());
128/// ```
129#[must_use]
130pub fn symbol(name: &str) -> Option<&'static Ability> {
131    Society::OrderOfVoln.ability(name)
132}

139    kind: AbilityKind,

146        short_name,
147        long_name,
148        kind,
149        cost,
150        spell_number,
151        usage: None,
152    }
153}

155/// A symbol whose cost is a plain fraction of the base.
156const fn favor(modifier: f64) -> Cost {
157    Cost::Favor {
158        modifier,
159        alternate: None,
160    }
161}

1//! A Survivalist's Kit, as `analyze` and `look in` describe it, read from
2//! the chunk: whether a container is a kit, what it holds, and its
3//! extractor.
4//!
5//! A kit is not an ordinary container. It holds herbs as doses it counts,
6//! listed in one line per form -- *The kit contains DOSEs acantha leaf (12),
7//! basal moss (3).* and *contains TINCTUREs ...* -- each herb a link with its
8//! own id, which is what `get` takes. `analyze` says whether a container is a
9//! kit at all, its capacity tier out of five, whether it has the Liquid
10//! Extractor, and what the extractor is working on.
11
12use std::collections::BTreeMap;
13
14use super::containers::ItemRef;
15use super::lines::{Chunk, ChunkLine};

17/// One herb a kit holds, with its count.
18#[derive(Clone, Debug, PartialEq, Eq)]
19pub struct KitHerb {
20    /// The herb, with the id `get` takes.
21    pub item: ItemRef,
22    /// Doses of it.
23    pub count: u32,
24    /// Listed as a TINCTURE: drunk, not eaten.
25    pub liquid: bool,
26}
27
28/// What `analyze` said of a container.
29#[derive(Clone, Debug, Default, PartialEq, Eq)]
30pub struct KitAnalysis {
31    /// It is a Survivalist's Kit.
32    pub is_kit: bool,
33    /// `Capacity: N/5`.
34    pub tier: Option<u32>,
35    /// It has the Liquid Extractor unlock.
36    pub extractor: bool,
37    /// What the extractor is working on, when it is.
38    pub distilling: Option<String>,
39}
40
41/// Kits by container id: what each last analyzed as and last listed.
42#[derive(Clone, Debug, Default, PartialEq, Eq)]
43pub struct Kits {
44    analyses: BTreeMap<String, KitAnalysis>,
45    listings: BTreeMap<String, Vec<KitHerb>>,
46}
47
48impl Kits {
49    /// What `analyze` last said of this container.
50    #[must_use]
51    pub fn analysis(&self, id: &str) -> Option<&KitAnalysis> {
52        self.analyses.get(id)
53    }
54
55    /// The kit's herbs as last listed; `None` when never listed.
56    #[must_use]
57    pub fn listing(&self, id: &str) -> Option<&[KitHerb]> {
58        self.listings.get(id).map(Vec::as_slice)
59    }
60
61    /// A new connection: every item id changes at a login, so what was
62    /// known by the old ids is forgotten.
63    pub(crate) fn clear(&mut self) {
64        self.analyses.clear();
65        self.listings.clear();
66    }
67
68    /// Read one chunk.
69    pub(crate) fn read_chunk(&mut self, chunk: &Chunk) {
70        let mut analyzing: Option<String> = None;
71        // A listing is restated whole: both forms arrive in the same answer,
72        // so a kit's list is replaced, not merged, the first time it is seen
73        // in a chunk.
74        let mut listed: Vec<String> = Vec::new();
75        for line in chunk.lines() {
76            let text = line.text();
77            let text = text.trim();
78            if text.starts_with("You analyze") {
79                analyzing = first_id(line);
80                if let Some(id) = &analyzing {
81                    self.analyses.insert(id.clone(), KitAnalysis::default());
82                }
83                continue;
84            }
85            let forms = forms(text);
86            if !forms.is_empty() {
87                let Some(kit) = first_id(line) else { continue };
88                if !listed.contains(&kit) {
89                    listed.push(kit.clone());
90                    self.listings.insert(kit.clone(), Vec::new());
91                }
92                let herbs = herbs_on(line, text, &kit, &forms);
93                self.listings.entry(kit).or_default().extend(herbs);
94                continue;
95            }
96            let Some(analysis) = analyzing.as_ref().and_then(|id| self.analyses.get_mut(id)) else {
97                continue;
98            };
99            if text.contains("is a Survivalist's Kit, which is a specialized container") {
100                analysis.is_kit = true;
101            }
102            if let Some(rest) = text.split("Capacity: ").nth(1)
103                && let Some((tier, _)) = rest.split_once("/5")
104            {
105                analysis.tier = tier.trim().parse().ok();
106            }
107            if text.contains("has the Liquid Extractor unlock") {
108                analysis.extractor = true;
109            }
110            if let Some(rest) = text.split("The extractor is currently targeting ").nth(1) {
111                analysis.distilling = rest
112                    .split(", with around")
113                    .next()
114                    .map(|what| what.trim().to_owned());
115            }
116        }
117    }
118}

120/// Each form a listing line states, in order: whether it is liquid, and
121/// where its herbs begin, after `contains DOSEs` or `contains TINCTUREs`.
122/// One line may state both.
123fn forms(text: &str) -> Vec<(bool, usize)> {
124    let mut forms: Vec<(bool, usize)> = [("contains DOSEs", false), ("contains TINCTUREs", true)]
125        .into_iter()
126        .flat_map(|(word, liquid)| {
127            text.match_indices(word)
128                .map(move |(at, _)| (liquid, at + word.len()))
129        })
130        .collect();
131    forms.sort_by_key(|&(_, at)| at);
132    forms
133}
134
135/// The herbs a listing line names. Each form opens with the kit named
136/// again (`The kit contains ...`), so the herbs after each naming of the
137/// kit are that form's, each paired with the next `(N)` in its text.
138fn herbs_on(line: &ChunkLine, text: &str, kit: &str, forms: &[(bool, usize)]) -> Vec<KitHerb> {
139    let mut groups: Vec<Vec<ItemRef>> = Vec::new();
140    for item in line.objects().filter_map(item_ref) {
141        if item.id == kit {
142            groups.push(Vec::new());
143        } else if let Some(group) = groups.last_mut() {
144            group.push(item);
145        }
146    }
147    // The last namings of the kit are the forms', in order.
148    let skip = groups.len().saturating_sub(forms.len());
149    let mut herbs = Vec::new();
150    for (index, (group, &(liquid, at))) in groups.into_iter().skip(skip).zip(forms).enumerate() {
151        let end = forms.get(index + 1).map_or(text.len(), |&(_, next)| next);
152        let mut counts = text[at..end].split('(').skip(1).filter_map(|chunk| {
153            chunk
154                .split_once(')')
155                .and_then(|(n, _)| n.trim().parse::<u32>().ok())
156        });
157        herbs.extend(group.into_iter().filter_map(|item| {
158            counts.next().map(|count| KitHerb {
159                item,
160                count,
161                liquid,
162            })
163        }));
164    }
165    herbs
166}

168fn first_id(line: &ChunkLine) -> Option<String> {
169    line.objects().find_map(item_ref).map(|item| item.id)
170}

172fn item_ref(link: &hydra_protocol::frame::Link) -> Option<ItemRef> {
173    match &link.kind {
174        hydra_protocol::frame::LinkKind::Exist { id, noun } => Some(ItemRef {
175            id: id.clone(),
176            noun: noun.clone(),
177            text: link.text.clone(),
178        }),
179        _ => None,
180    }

1//! What a PSM costs to use, and whether it can be used now.
2//!
3//! # The table
4//!
5//! `data/psm_costs.tsv`, generated by `tools/extract_psm_costs.rb`: 187 rows,
6//! 11 armor, 80 combat maneuver, 33 feat, 33 shield, 24 weapon and 6 warcry,
7//! each with the line it was cut from. Every cost is one gauge and a number,
8//! stamina for all but Excoriate (10 mana), and two are two numbers: Burst of
9//! Swiftness and Surge of Strength cost 60 stamina while their own cooldown
10//! is up and 30 otherwise.
11//!
12//! **Those two are priced at the moment of the question**, from the
13//! Cooldowns dialog as it stands then, never from a value fixed earlier. A
14//! cost passes only when the gauge holds strictly more than it: exactly 30
15//! stamina does not afford a 30-stamina cost.
16//!
17//! Two feats share the mnemonic `wps` (`weighting` and `padding`). Both rows
18//! are kept; they cost the same, which the tool checks, so a lookup by
19//! mnemonic does not depend on which it finds.
20//!
21//! Read on each lookup rather than parsed once into a `static`: a hunt asks
22//! once a step, and 187 short lines is not worth a process global.
23//!
24//! # The rule
25//!
26//! [`GameState::psm_availability`] answers four tests separately, so a caller
27//! can tell *why* as well as whether:
28//!
29//! | test | passes when | read from |
30//! |---|---|---|
31//! | known | ranks `>=` 1 | [`PsmSet::get`](super::PsmSet::get); unknown until the category's list is read |
32//! | affordable | cost **`<`** current points, strictly | the gauge's current points, when the bar states them |
33//! | not cooling | not in `Cooldowns` by name | [`Effects::active_named`](crate::Effects::active_named) with the long name |
34//! | not overexerted | `Overexerted` not in `Debuffs` | the same, for `Overexerted` |
35//!
36//! And the waivers: a weapon or shield `area_of_effect` technique is
37//! affordable whatever it costs while `Glorious Momentum` is up, and a weapon
38//! technique ignores its cooldown while `Glorious Momentum` is up for an
39//! `area_of_effect` one, or `Ardor of the Scourge` for an `assault`.
40//!
41//! # What is not answered
42//!
43//! - **Warcries' availability.** [`warcry_cost`] gives the cost, but whether
44//!   a warcry can be used also turns on being cutthroated or silenced, and
45//!   the known warcries
46//!   ([`Standing::warcries`](super::super::standing::Standing::warcries)) are a
47//!   set with no "never stated" of their own. A caller gating warcries applies
48//!   its own thresholds.
49//! - **Forced roundtime**, which raises the cost by `25 + 10 x count` percent.
50//! - **Ignoring the cooldown** for Burst and Surge, which a caller would ask
51//!   for explicitly.
52

58/// Which gauge a PSM spends.
59#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
60pub enum Gauge {
61    /// Stamina: every row but one.
62    Stamina,
63    /// Mana: Excoriate.
64    Mana,
65}
66
67impl Gauge {
68    /// The `<progressBar>` id that states it.
69    #[must_use]
70    pub const fn bar(self) -> &'static str {
71        match self {
72            Self::Stamina => "stamina",
73            Self::Mana => "mana",
74        }
75    }
76
77    fn parse(text: &str) -> Option<Self> {
78        match text {
79            "stamina" => Some(Self::Stamina),
80            "mana" => Some(Self::Mana),
81            _ => None,
82        }
83    }
84}

86/// One row of the cost table.
87#[derive(Clone, Copy, Debug, PartialEq, Eq)]
88pub struct PsmCost {
89    /// The long name, e.g. `coup_de_grace`: what the Cooldowns dialog
90    /// names it, once folded ([`Effects::active_named`](crate::Effects::active_named)).
91    pub long_name: &'static str,
92    /// The technique's type, e.g. `attack`, `area_of_effect`, `assault`,
93    /// `passive`.
94    pub kind: &'static str,
95    /// The gauge it spends.
96    pub gauge: Gauge,
97    /// What it costs.
98    pub amount: u16,
99    /// What it costs while its own cooldown is up, when that differs: Burst
100    /// of Swiftness and Surge of Strength, 60.
101    pub while_cooling: Option<u16>,
102    /// The table's `source` column: where the row was cut from.
103    pub source: &'static str,
104}
105
106/// Every row: the table's category word (`cman`, ..., `warcry`), the
107/// mnemonic, and the cost.
108fn rows() -> impl Iterator<Item = (&'static str, &'static str, PsmCost)> {
109    PSM_COSTS_TSV
110        .lines()
111        .filter(|line| !line.starts_with('#') && !line.starts_with("category\t"))
112        .filter_map(read_row)
113}
114
115fn read_row(line: &'static str) -> Option<(&'static str, &'static str, PsmCost)> {
116    let mut field = line.split('\t');
117    let category = field.next()?;
118    let mnemonic = field.next()?;
119    let long_name = field.next()?;
120    let kind = field.next()?;
121    let gauge = Gauge::parse(field.next()?)?;
122    let amount = field.next()?.parse().ok()?;
123    let while_cooling = match field.next()? {
124        "" => None,
125        hot => Some(hot.parse().ok()?),
126    };
127    let source = field.next()?;
128    Some((
129        category,
130        mnemonic,
131        PsmCost {
132            long_name,
133            kind,
134            gauge,
135            amount,
136            while_cooling,
137            source,
138        },
139    ))
140}

142/// What one PSM costs, by category and mnemonic (`cman`, `coupdegrace`).
143///
144/// `None`: the table does not list it. The PSM list itself is an open set
145/// (`psm.rs`), so a mnemonic the game prints can be missing here.
146#[must_use]
147pub fn psm_cost(category: PsmCategory, mnemonic: &str) -> Option<PsmCost> {
148    row(category.as_str(), mnemonic)
149}
150
151/// What one warcry costs.
152///
153/// `Option` only because the table is data: all six are in it, which
154/// `tests/psm_cost.rs` asserts.
155#[must_use]
156pub fn warcry_cost(cry: Warcry) -> Option<PsmCost> {
157    row("warcry", cry.short_name())
158}
159
160/// Every row, in the table's order.
161pub fn every_cost() -> impl Iterator<Item = (&'static str, &'static str, PsmCost)> {
162    rows()
163}
164
165fn row(category: &str, mnemonic: &str) -> Option<PsmCost> {
166    rows()
167        .find(|(c, m, _)| *c == category && *m == mnemonic)
168        .map(|(_, _, cost)| cost)
169}

171/// The four tests for one PSM, each answered on its own
172/// ([`GameState::psm_availability`]).
173///
174/// Every `None` is **not known**, never "no": the list unread, the bar
175/// stating no points, the dialog never stated, or the clock not yet set.
176#[derive(Clone, Copy, Debug, PartialEq, Eq)]
177pub struct PsmAvailability {
178    /// The table row, or `None` when the table does not list it.
179    pub cost: Option<PsmCost>,
180    /// Trained, ranks `>= 1`. `None`: this category's list has not been read.
181    pub known: Option<bool>,
182    /// The gauge's current points exceed the cost. `None`: no row, or the bar
183    /// has not stated its points.
184    pub affordable: Option<bool>,
185    /// Listed and live in the Cooldowns dialog.
186    pub cooling: Option<bool>,
187    /// A weapon waiver lets it be used while cooling.
188    pub cooldown_waived: bool,
189    /// `Overexerted` is live in the Debuffs dialog.
190    pub overexerted: Option<bool>,
191}
192
193impl PsmAvailability {
194    /// Whether it can be used: known, affordable, not cooling unless waived,
195    /// and not overexerted.
196    ///
197    /// `Some(false)` as soon as any test is known to fail, whatever the
198    /// others; `Some(true)` only when all four are known to pass.
199    #[must_use]
200    pub fn available(&self) -> Option<bool> {
201        let cooling = if self.cooldown_waived {
202            Some(false)
203        } else {
204            self.cooling
205        };
206        let passes = [
207            self.known,
208            self.affordable,
209            cooling.map(|c| !c),
210            self.overexerted.map(|o| !o),
211        ];
212        if passes.contains(&Some(false)) {
213            Some(false)
214        } else if passes.contains(&None) {
215            None
216        } else {
217            Some(true)
218        }
219    }
220}
221
222impl GameState {
223    /// Whether one PSM can be used, test by test.
224    ///
225    /// The effects are read at [`GameState::game_time_now`]; before the first
226    /// prompt teaches the clock, `cooling` and `overexerted` are `None`.
227    #[must_use]
228    pub fn psm_availability(&self, category: PsmCategory, mnemonic: &str) -> PsmAvailability {
229        let cost = psm_cost(category, mnemonic);
230        let psms = &self.character.psms;
231        let known = match psms.get(category, mnemonic) {
232            Some(ranks) => Some(ranks.ranks >= 1),
233            None => psms.has_table(category).then_some(false),
234        };
235        let now = self.game_time_now();
236        let named = |dialog: &str, name: &str| {
237            now.and_then(|now| self.effects.active_named(dialog, name, now))
238        };
239        let cooling = cost.and_then(|c| named("Cooldowns", c.long_name));
240        let overexerted = named("Debuffs", "Overexerted");
241        let momentum = named("Buffs", "Glorious Momentum") == Some(true);
242        let ardor = named("Buffs", "Ardor of the Scourge") == Some(true);
243        let area = cost.is_some_and(|c| c.kind == "area_of_effect");
244        let assault = cost.is_some_and(|c| c.kind == "assault");
245        let waived_cost =
246            area && momentum && matches!(category, PsmCategory::Weapon | PsmCategory::Shield);
247        let cooldown_waived =
248            category == PsmCategory::Weapon && ((area && momentum) || (assault && ardor));
249        let affordable = if waived_cost {
250            Some(true)
251        } else {
252            cost.and_then(|c| self.affords(c, cooling))
253        };
254        PsmAvailability {
255            cost,
256            known,
257            affordable,
258            cooling,
259            cooldown_waived,
260            overexerted,
261        }
262    }
263
264    /// Whether the gauge's current points exceed the cost, strictly.
265    ///
266    /// A cost that doubles while cooling, asked with the cooldown unknown,
267    /// answers only when both costs give the same answer.
268    fn affords(&self, cost: PsmCost, cooling: Option<bool>) -> Option<bool> {
269        let current = self.vital(cost.gauge.bar())?.current?;
270        let covers = |amount: u16| i32::from(amount) < current;
271        match (cost.while_cooling, cooling) {
272            (None, _) | (Some(_), Some(false)) => Some(covers(cost.amount)),
273            (Some(hot), Some(true)) => Some(covers(hot)),
274            (Some(hot), None) => {
275                let cold = covers(cost.amount);
276                (cold == covers(hot)).then_some(cold)
277            }
278        }
279    }
280}

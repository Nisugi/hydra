1//! A bard's song: how long it lasts, what it costs, what it is worth.
2//!
3//! **There is no classifier here at all.** Every function is a formula over
4//! facts the model already holds — level, Elemental Lore - Air, Mental Lore -
5//! Telepathy, and ranks in the Bard circle.
6//!
7//! That makes it the module most worth getting numerically exact, because
8//! nothing in the wire will ever correct a wrong constant. A duration that is
9//! ten seconds out just quietly renews the song too late.
10//!
11//! # The inputs, and the one that needed parsing
12//!
13//! Three of the four are already typed. The fourth is not:
14//! [`Experience::level`](super::Experience::level) is the wire's label
15//! **verbatim** — `"Level 100"`, not `100` — and its doc records why: turning
16//! it into a number is a guess about a format the game may change, and every
17//! consumer that wants to display it wants the string.
18//!
19//! Spellsong is the first consumer that wants the number. So it parses, here,
20//! where a failure is visible as `None` rather than defaulting to level 0 —
21//! which would silently give every bard the shortest possible song.
22//!
23//! # Time left and renewal cost
24//!
25//! These read the effects list and the spell table, which a character alone
26//! does not hold, so they are [`GameState::spellsong_timeleft`] and
27//! [`GameState::spellsong_renew_cost`] (`state/spell_time.rs`). Time left
28//! there reads the game's own time left on an active song, rather than a
29//! timestamp kept on the side that would need re-aligning to it.
30//!
31//! [`GameState::spellsong_timeleft`]: crate::state::GameState::spellsong_timeleft
32//! [`GameState::spellsong_renew_cost`]: crate::state::GameState::spellsong_renew_cost
33

36/// Bard song arithmetic over a character's current facts.
37///
38/// Every method returns `Option` where an input may be unknown: a
39/// character whose skills have never been read has **no** song duration, and
40/// reporting the base 120 seconds would be inventing one.
41#[derive(Clone, Copy, Debug)]
42pub struct Spellsong<'a> {
43    /// The character's level.
44    level: u16,
45    /// Their skills, for the two lores this arithmetic reads.
46    skills: &'a SkillSet,
47    /// Ranks in the Bard circle.
48    bard: u16,
49}
50
51impl<'a> Spellsong<'a> {
52    /// Read the inputs from a character.
53    ///
54    /// `None` when the level has never been reported: every formula here
55    /// depends on it, so there is nothing honest to return.
56    #[must_use]
57    pub fn of(character: &'a super::Character) -> Option<Self> {
58        Some(Self {
59            level: level_of(character.experience.level.as_deref()?)?,
60            skills: &character.skills,
61            // The Bard spell circle's ranks, which arrive as a circle row of
62            // the `spell` table.
63            bard: character.skills.circle("Bard").unwrap_or(0),
64        })
65    }
66
67    /// The base song duration in seconds, before renewal.
68    ///
69    /// ```text
70    /// base(level) + LOG bonus + (INF bonus * 3) + (Telepathy ranks * 2)
71    /// ```
72    ///
73    /// The level is required to build a [`Spellsong`] at all, so there is no
74    /// answer without it, rather than a default of 120 seconds.
75    #[must_use]
76    pub fn duration(&self, logic_bonus: i16, influence_bonus: i16) -> u32 {
77        let telepathy = self
78            .skills
79            .ranks(SkillKind::MentalLoreTelepathy)
80            .unwrap_or(0);
81        let base = i32::try_from(base_duration(self.level)).unwrap_or(i32::MAX);
82        let total = base
83            + i32::from(logic_bonus)
84            + i32::from(influence_bonus) * 3
85            + i32::from(telepathy) * 2;
86        u32::try_from(total.max(0)).unwrap_or(0)
87    }
88
89    /// Ranks in Elemental Lore - Air, 0 when the table has not been read.
90    fn air(&self) -> u16 {
91        self.skills.ranks(SkillKind::ElementalLoreAir).unwrap_or(0)
92    }
93
94    /// The bonus Air ranks confer.
95    ///
96    /// Prefers the **game's own figure** from the skill table, falling back to
97    /// [`to_bonus`] only when the table has not stated one. Always computing
98    /// it would throw away the real number in favour of a reconstruction —
99    /// and the two disagree whenever an enhancive is on, because the table's
100    /// bonus includes it and the curve cannot.
101    fn air_bonus(&self) -> u16 {
102        self.skills
103            .get(SkillKind::ElementalLoreAir)
104            .and_then(|s| s.bonus)
105            .unwrap_or_else(|| to_bonus(self.air()))
106    }
107
108    /// Sonic Armor's durability.
109    #[must_use]
110    pub fn sonic_armor_durability(&self) -> u32 {
111        210 + u32::from(self.level / 2) + u32::from(self.air_bonus())
112    }
113
114    /// Sonic Blade's durability. Also called sonic weapon durability.
115    #[must_use]
116    pub fn sonic_blade_durability(&self) -> u32 {
117        160 + u32::from(self.level / 2) + u32::from(self.air_bonus())
118    }
119
120    /// Sonic Shield's durability.
121    #[must_use]
122    pub fn sonic_shield_durability(&self) -> u32 {
123        125 + u32::from(self.level / 2) + u32::from(self.air_bonus())
124    }
125
126    /// The haste bonus from Tonis, which is **negative**.
127    ///
128    /// Starts at −1 and improves by one at 30 and 75 ranks of Air.
129    #[must_use]
130    pub fn tonis_haste_bonus(&self) -> i16 {
131        let air = self.air();
132        -1 - i16::from(air >= 30) - i16::from(air >= 75)
133    }
134
135    /// Tonis's dodge bonus.
136    ///
137    /// 20, plus one for each of twenty Air thresholds passed.
138    #[must_use]
139    pub fn tonis_dodge_bonus(&self) -> u16 {
140        const THRESHOLDS: [u16; 20] = [
141            1, 2, 3, 5, 8, 10, 14, 17, 21, 26, 31, 36, 42, 49, 55, 63, 70, 78, 87, 96,
142        ];
143        let air = self.air();
144        20 + u16::try_from(THRESHOLDS.iter().filter(|&&t| air >= t).count()).unwrap_or(0)
145    }
146
147    /// How far Depression pushes down.
148    #[must_use]
149    pub fn depression_pushdown(&self) -> u16 {
150        20 + self
151            .skills
152            .ranks(SkillKind::MentalLoreTelepathy)
153            .unwrap_or(0)
154    }
155
156    /// Depression's slow, which is **negative**.
157    #[must_use]
158    pub fn depression_slow(&self) -> i16 {
159        const THRESHOLDS: [u16; 5] = [10, 25, 45, 70, 100];
160        let telepathy = self
161            .skills
162            .ranks(SkillKind::MentalLoreTelepathy)
163            .unwrap_or(0);
164        -2 - i16::try_from(THRESHOLDS.iter().filter(|&&t| telepathy >= t).count()).unwrap_or(0)
165    }
166
167    /// How many targets Holding Song can hold: `1 + (Bard ranks - 1) / 7`.
168    ///
169    /// **Zero ranks is one target.** The subtraction saturates, so the zero
170    /// case is explicit rather than relying on how a negative quotient
171    /// rounds (toward zero gives 1; flooring would give 0).
172    #[must_use]
173    pub fn holding_targets(&self) -> u16 {
174        1 + self.bard.saturating_sub(1) / 7
175    }
176
177    /// Mirrors' dodge bonus.
178    ///
179    /// **Saturating below 19 Bard ranks.** Unclamped,
180    /// `20 + (Bard ranks - 19) / 2` falls below 20 for a bard with fewer than
181    /// 19 ranks — a dodge bonus of 15 at rank 9. The spell
182    /// cannot be cast without the ranks to know it, so the expression is
183    /// unreachable in play; clamping makes that explicit rather than
184    /// propagating a number that means nothing.
185    #[must_use]
186    pub fn mirrors_dodge_bonus(&self) -> u16 {
187        20 + self.bard.saturating_sub(19) / 2
188    }
189
190    /// Mirrors' cost as `(to cast, to renew)`.
191    #[must_use]
192    pub fn mirrors_cost(&self) -> (u16, u16) {
193        let over = self.bard.saturating_sub(19);
194        (19 + over / 5, 8 + over / 10)
195    }
196
197    /// The shared sonic bonus, `Bard ranks / 2`.
198    #[must_use]
199    pub fn sonic_bonus(&self) -> u16 {
200        self.bard / 2
201    }
202
203    /// Sonic Armor's bonus.
204    #[must_use]
205    pub fn sonic_armor_bonus(&self) -> u16 {
206        self.sonic_bonus() + 15
207    }
208
209    /// Sonic Blade's bonus.
210    #[must_use]
211    pub fn sonic_blade_bonus(&self) -> u16 {
212        self.sonic_bonus() + 10
213    }
214
215    /// Sonic Shield's bonus.
216    #[must_use]
217    pub fn sonic_shield_bonus(&self) -> u16 {
218        self.sonic_bonus() + 10
219    }
220
221    /// Valor's bonus.
222    ///
223    /// Capped by whichever is lower, Bard ranks or level.
224    #[must_use]
225    pub fn valor_bonus(&self) -> u16 {
226        10 + self.bard.min(self.level).saturating_sub(10) / 2
227    }
228
229    /// Valor's cost as `(to cast, to renew)`.
230    #[must_use]
231    pub fn valor_cost(&self) -> (u16, u16) {
232        let bonus = self.valor_bonus();
233        (10 + bonus / 2, 3 + bonus / 5)
234    }
235
236    /// Luck's cost as `(to cast, to renew)`.
237    ///
238    /// **The renew term is UNVERIFIED.** It is `6 + over/8`, the `/ 2`
239    /// applying to the inner quotient only, and not `(6 + over/4) / 2` as the
240    /// parallel with every other `*_cost` would suggest. Changing it would be
241    /// a guess at the game's real number until someone measures it in play.
242    #[must_use]
243    pub fn luck_cost(&self) -> (u16, u16) {
244        let over = self.bard.saturating_sub(6);
245        (6 + over / 4, 6 + over / 4 / 2)
246    }
247}

267/// The base duration for a level, in seconds.
268///
269/// A four-band piecewise curve. **Above level 100 the last band continues**:
270/// the bands are cumulative and the last one continues cleanly, so it is
271/// extended rather than cut off or reset to 120 — a level 101 bard is not a
272/// level 0 bard.
273#[must_use]
274pub fn base_duration(level: u16) -> u32 {
275    let level = u32::from(level);
276    match level {
277        0..=25 => 120 + level * 4,
278        26..=50 => 220 + (level - 25) * 3,
279        51..=75 => 295 + (level - 50) * 2,
280        _ => 345 + (level - 75),
281    }
282}
283
284/// The bonus a count of ranks confers.
285///
286/// The game's cost curve: the first ten ranks are worth five points each and
287/// every further band is worth one less, flattening to one point per rank past
288/// forty. **Not bard-specific** — it lives here because spellsong is its first
289/// caller, and it moves when a second one appears.
290///
291/// Prefer a skill's own `bonus` when the table has stated one: this
292/// reconstruction cannot see an enhancive, and the table can.
293#[must_use]
294pub fn to_bonus(ranks: u16) -> u16 {
295    let ranks = u32::from(ranks);
296    let bonus = match ranks {
297        0..=10 => ranks * 5,
298        11..=20 => 50 + (ranks - 10) * 4,
299        21..=30 => 90 + (ranks - 20) * 3,
300        31..=40 => 120 + (ranks - 30) * 2,
301        _ => 140 + (ranks - 40),
302    };
303    u16::try_from(bonus).unwrap_or(u16::MAX)
304}

306/// The number in `"Level 100"`.
307///
308/// See the module doc: the level is stored verbatim by deliberate decision,
309/// and this is the first consumer that needs it as a number. `None` rather
310/// than `0` on an unexpected format, so a format change shows up as "unknown"
311/// instead of as a level 0 bard with the shortest song in the game.
312fn level_of(label: &str) -> Option<u16> {
313    label
314        .trim()
315        .rsplit_once(' ')
316        .map_or(label.trim(), |(_, n)| n)
317        .parse()
318        .ok()
319}

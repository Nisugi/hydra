1//! The closed vocabularies of a character sheet.
2//!
3//! Closed-vocabulary values get typed named fields. These are the sets that
4//! are genuinely closed -- the game has a fixed list and the wire spells them
5//! exactly one way.
6//!
7//! # Why enums
8//!
9//! **Every set below is the game's own list, spelled as the wire spells it.**
10//! An enum makes a typo a compile error and an unknown value a visible `None`
11//! rather than a row that silently never matches.
12//!
13//! What is NOT here, and deliberately:
14//!
15//! * **Race** -- one or two words. No fixed list is worth trusting. Genuinely
16//!   open.
17//! * **Profession** -- one word, hyphens allowed. Also unenumerated.
18//! * **Citizenship town** -- free text.
19//!
20//! Those three stay `String`. Inventing an enum for them would turn a new
21//! Simutronics race into a parse failure, which is worse than a string.
22//!
23//! # The pattern
24//!
25//! Each enum carries the `crit/types.rs` trio -- `ALL`, `as_str`, `parse` -- so
26//! round-tripping is testable and the list lives in one place. `parse` is an
27//! inherent fn returning `Option<Self>`, matching `DamageType::parse`
28//! (`crit/types.rs`) rather than `FromStr`, because a wire value that is not
29//! in the set is data we have not seen, not an error to propagate.
30

33/// How badly death has worn on the character.
34///
35/// From the experience report's `Death's Sting:` field: `None`, `Light`,
36/// `Moderate`, `Sharp`, `Harsh`, `Piercing` or `Crushing`.
37///
38/// Ordered as the game orders it, least to worst, and `Ord` follows that order
39/// so a behavior can ask whether the sting is worse than `Light` without a
40/// lookup table.
41#[derive(
42    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, serde::Serialize, serde::Deserialize,
43)]
44pub enum DeathsSting {
45    /// Printed `None`: no sting at all, the least severe value.
46    None,
47    /// Printed `Light`.
48    Light,
49    /// Printed `Moderate`.
50    Moderate,
51    /// Printed `Sharp`.
52    Sharp,
53    /// Printed `Harsh`.
54    Harsh,
55    /// Printed `Piercing`.
56    Piercing,
57    /// Printed `Crushing`: the most severe value.
58    Crushing,
59}
60
61impl DeathsSting {
62    /// Every variant, in the game's own severity order.
63    pub const ALL: [Self; 7] = [
64        Self::None,
65        Self::Light,
66        Self::Moderate,
67        Self::Sharp,
68        Self::Harsh,
69        Self::Piercing,
70        Self::Crushing,
71    ];
72
73    /// The wire spelling.
74    #[must_use]
75    pub const fn as_str(self) -> &'static str {
76        match self {
77            Self::None => "None",
78            Self::Light => "Light",
79            Self::Moderate => "Moderate",
80            Self::Sharp => "Sharp",
81            Self::Harsh => "Harsh",
82            Self::Piercing => "Piercing",
83            Self::Crushing => "Crushing",
84        }
85    }
86
87    /// Parse the wire spelling.
88    #[must_use]
89    pub fn parse(text: &str) -> Option<Self> {
90        Self::ALL.into_iter().find(|v| v.as_str() == text)
91    }
92}

100/// The three societies a character may belong to.
101///
102/// Printed `Order of Voln`, `Council of Light` or `Guardians of Sunfist`.
103///
104/// **"Not a member" is absence, not a variant.** A string sentinel such as
105/// `'None'` is something a reader must know to special-case;
106/// `Option<Society>` says the same thing in the type. See [`Society::max_rank`]
107/// for the one piece of arithmetic the
108/// societies do not share.
109#[derive(
110    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, serde::Serialize, serde::Deserialize,
111)]
112pub enum Society {
113    /// Printed `Order of Voln`; a Master holds rank 26.
114    OrderOfVoln,
115    /// Printed `Council of Light`; a Master holds rank 20.
116    CouncilOfLight,
117    /// Printed `Guardians of Sunfist`; a Master holds rank 20.
118    GuardiansOfSunfist,
119}
120
121impl Society {
122    /// Every society.
123    pub const ALL: [Self; 3] = [
124        Self::OrderOfVoln,
125        Self::CouncilOfLight,
126        Self::GuardiansOfSunfist,
127    ];
128
129    /// The wire spelling.
130    #[must_use]
131    pub const fn as_str(self) -> &'static str {
132        match self {
133            Self::OrderOfVoln => "Order of Voln",
134            Self::CouncilOfLight => "Council of Light",
135            Self::GuardiansOfSunfist => "Guardians of Sunfist",
136        }
137    }
138
139    /// Parse the wire spelling.
140    #[must_use]
141    pub fn parse(text: &str) -> Option<Self> {
142        Self::ALL.into_iter().find(|v| v.as_str() == text)
143    }
144
145    /// The rank a Master holds, which differs by society.
146    ///
147    /// **The wire does not send it.** The rank is absent for a Master -- the
148    /// line reads "You are a Master in the Order of Voln." with no number --
149    /// so the rank has to be supplied from knowledge of the game: Voln 26,
150    /// the other two 20.
151    ///
152    /// That asymmetry is real (Voln has 26 steps, the others 20) and is the
153    /// reason this is a method rather than one constant.
154    #[must_use]
155    pub const fn max_rank(self) -> u8 {
156        match self {
157            Self::OrderOfVoln => 26,
158            Self::CouncilOfLight | Self::GuardiansOfSunfist => 20,
159        }
160    }
161}

165        f.write_str(self.as_str())
166    }

328/// The five PSM categories.
329///
330/// `cman`, `feat`, `armor`, `shield` and `weapon`. Ascension is **not** one,
331/// although its `<x> list all` command sits beside theirs -- see
332/// [`AscensionTable`](super::psm::AscensionTable) for the three ways
333/// ascension's output differs.
334///
335/// Each PSM has its own `<category> list all` command and its own rank table.
336///
337/// **The header phrase and the key prefix differ**, and not uniformly:
338/// "Combat Maneuvers" is stored under `cman`. Here they are two methods on
339/// one value, so the pair cannot drift apart.
340#[derive(
341    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, serde::Serialize, serde::Deserialize,
342)]
343pub enum PsmCategory {

345    Armor,
346    /// Key prefix `cman`; `list all` heading `Combat Maneuvers`.
347    CombatManeuver,
348    /// Key prefix `feat`; `list all` heading `Feats`.
349    Feat,
350    /// Key prefix `shield`; `list all` heading `Shield Specializations`.
351    Shield,
352    /// Key prefix `weapon`; `list all` heading `Weapon Techniques`.
353    Weapon,
354}
355
356impl PsmCategory {
357    /// Every category.

359        Self::Armor,
360        Self::CombatManeuver,
361        Self::Feat,
362        Self::Shield,
363        Self::Weapon,
364    ];
365
366    /// The key prefix, e.g. `cman`.

374            Self::Weapon => "weapon",

411/// The account's subscription tier.
412///
413/// The wire says `F2P`, `Standard`, `Premium` or `Platinum`.
414///
415/// # This is four wire values, not three
416///
417/// All four are kept. Platinum is a different game instance with different
418/// mechanics, so folding it into Premium would lose a fact a behavior may
419/// need, with no way back.
420#[derive(
421    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, serde::Serialize, serde::Deserialize,
422)]
423pub enum AccountType {
424    /// Wire `F2P`.
425    Free,
426    /// Wire `Standard`.
427    Normal,
428    /// Wire `Premium`.
429    Premium,
430    /// Wire `Platinum`. Kept distinct from Premium.
431    Platinum,
432}
433
434impl AccountType {
435    /// Every tier.
436    pub const ALL: [Self; 4] = [Self::Free, Self::Normal, Self::Premium, Self::Platinum];
437
438    /// The canonical name.
439    #[must_use]
440    pub const fn as_str(self) -> &'static str {
441        match self {
442            Self::Free => "Free",
443            Self::Normal => "Normal",
444            Self::Premium => "Premium",
445            Self::Platinum => "Platinum",
446        }
447    }
448
449    /// Parse the **wire** spelling, which is not the canonical name.
450    ///
451    /// The wire says `F2P` and `Standard` where this type says `Free` and
452    /// `Normal`; both spellings are accepted so a caller need not know which
453    /// side it is on.
454    #[must_use]
455    pub fn parse(text: &str) -> Option<Self> {
456        match text {
457            "F2P" | "Free" => Some(Self::Free),
458            "Standard" | "Normal" => Some(Self::Normal),
459            "Premium" => Some(Self::Premium),
460            "Platinum" => Some(Self::Platinum),
461            _ => None,
462        }
463    }
464}

466impl fmt::Display for AccountType {
467    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
468        f.write_str(self.as_str())
469    }
470}

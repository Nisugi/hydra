1//! Damage lines: `... and hit for 20 points of damage!` and its many shapes.
2//!
3//! Reads the `damage` family: four lists in one order --
4//! basic, spell, environmental, fallback -- with the broad fallbacks last so
5//! the specific patterns win their better target captures first.
6//!
7//! # The gate is a substring
8//!
9//! Every damage pattern contains `damage` except the maneuver `N hits!`
10//! form, so a substring check skips the regex scan on the ~95% of lines that
11//! cannot match. It is a correctness statement about the patterns, not a
12//! performance guess, and the test that every damage pattern contains one of
13//! the two substrings makes it one.
14
15use super::defs::defs;
16use crate::state::lines::ChunkLine;
17
18/// One damage line, classified.
19#[derive(Debug, Clone, PartialEq, Eq)]
20pub struct DamageLine {
21    /// Points of damage.
22    pub amount: u32,
23    /// Which damage list matched: `basic`, `spell`, `environmental`,
24    /// `fallback`.
25    pub list: String,
26    /// The target the line names, as captured, where the shape names one.
27    pub target_text: Option<String>,
28}
29
30/// The substrings every damage pattern contains one of.
31pub const GATE: [&str; 2] = ["damage", " hits!"];
32
33impl DamageLine {
34    /// Classify one line as a damage line.
35    #[must_use]
36    pub fn classify(line: &ChunkLine) -> Option<Self> {
37        let text = line.text();
38        if !GATE.iter().any(|g| text.contains(g)) {
39            return None;
40        }
41        let (def, caps) = defs().first_match("damage", &text)?;
42        Some(Self {
43            amount: caps.name("damage")?.as_str().parse().ok()?,
44            list: def.name.clone(),
45            target_text: caps.name("target").map(|m| m.as_str().to_owned()),
46        })
47    }
48}

1//! Roll lines and outcome lines.
2//!
3//! Both attach to whatever the cursor points at --
4//! an active flare owns its own SMR, the swing owns its AS/DS -- and run
5//! AFTER target switching, because an outcome names its target.
6

11use crate::state::lines::ChunkLine;

16impl Cursor<'_> {
17    /// Route a roll line to its sink.
18    ///
19    /// Roll routing differs by roll class (fixture-verified): SMR/SSR/
20    /// maneuver rolls PRECEDE their per-target line, so one arriving on a
21    /// settled event belongs to the NEXT target and is held; AS/DS-class
22    /// rolls always FOLLOW their attack line, so a multi-strike keeps every
23    /// roll on the attack-born event.
24    fn handle_resolution(&mut self, ctx: &LineCtx, res: Resolution) {
25        // A PHYSICAL roll arriving on a flare-released spell that has ALREADY
26        // resolved is the interrupted swing's: resume the swing first. A
27        // released BOLT's own AS/DS is the first roll after its cast line.
28        if let (Some(cur), Some(io)) = (self.current, self.interrupted_own) {
29            let e = self.ev(cur);
30            if e.released
31                && matches!(res.kind, ResolutionKind::AsDs | ResolutionKind::UafUdf)
32                && (!e.resolutions.is_empty() || !e.hits.is_empty() || !e.outcomes.is_empty())
33            {
34                self.save_event(cur);
35                self.interrupted_own = None;
36                self.resume(io);
37                self.seeking_damage = true;
38                self.flare_ctx = None;
39            }
40        }
41        // A damaging flare claims a roll only while it has no damage yet; one
42        // that already dealt its damage is complete (an acid proc must not
43        // steal the next swing's AS/DS).
44        let settled_flare = self.flare_ctx.filter(|fc| !self.flare(*fc).hits.is_empty());
45        if settled_flare.is_some() {
46            self.flare_ctx = None;
47        }
48        let mut sink = self.flare_ctx.map(Sink::Flare);
49        // Crit RIDER roll: a knockdown crit rolls its own SMR after the damage
50        // and narrates the fall on the very next line. It belongs to the hit
51        // that caused it.
52        if sink.is_none()
53            && matches!(res.kind, ResolutionKind::Smr | ResolutionKind::ManeuverRoll)
54            && self.lines.get(ctx.index + 1).is_some_and(crit_rider_line)
55        {
56            sink = settled_flare.map(Sink::Flare).or_else(|| {
57                self.current
58                    .filter(|&c| !self.ev(c).hits.is_empty())
59                    .map(Sink::Event)
60            });
61        }
62        if sink.is_none()
63            && let Some(cur) = self.current
64        {
65            let e = self.ev(cur);
66            if res.kind.precedes_its_line() {
67                // On an ATTACK-BORN event only damage settles it (cripple:
68                // init -> resisted -> SMR). On a switch-born event an
69                // outcome or a STATUS settles too (eviscerate rolls once
70                // per onlooker, each BEFORE the creature it applies to).
71                let settled = !e.hits.is_empty()
72                    || (!e.attack_born && (!e.outcomes.is_empty() || e.had_status));
73                if !settled {
74                    sink = Some(Sink::Event(cur));
75                }
76            } else if e.attack_born || (e.hits.is_empty() && e.outcomes.is_empty()) {
77                sink = Some(Sink::Event(cur));
78            }
79        }
80        match sink {
81            Some(Sink::Flare(fr)) => self.flare_mut(fr).resolutions.push(res),
82            Some(Sink::Event(id)) => self.ev_mut(id).resolutions.push(res),
83            None => self.pending_resolutions.push(res),
84        }
85    }

87    /// Route an outcome line.
88    fn handle_outcome(&mut self, ctx: &mut LineCtx, out: &Outcome) {
89        // A pre-emptive evade ("bounds to safety as you move to attack it")
90        // is OUR swing resolving. Scoped to a live nock: only there do we
91        // know an outbound swing is outstanding.
92        let preempts_open_inbound = self.nocked.is_some()
93            && self.cur().is_some_and(|e| e.inbound)
94            && self.flare_ctx.is_none()
95            && ctx.line_target.as_ref().is_some_and(|a| a.id.is_some())
96            && !out.names_attacker;
97        if ctx.line_attack.is_some() {
98            // An outcome ON an initiation line belongs to the event that line
99            // opens below.
100            ctx.same_line_outcome = Some(out.kind);
101            return;
102        }
103        if let Some(fc) = self.flare_ctx {
104            self.flare_mut(fc).outcomes.push(out.kind);
105            return;
106        }
107        if let Some(cur) = self.current.filter(|_| !preempts_open_inbound) {
108            self.ev_mut(cur).outcomes.push(out.kind);
109            return;
110        }
111        let Some(lt) = ctx.line_target.clone().filter(|a| a.id.is_some()) else {
112            // No event, no named target: an initiation we have no def for.
113            self.orphan_outcomes.push(out.kind);
114            return;
115        };
116        // Reached with an open event only via preempts_open_inbound: the
117        // creature's swing that interleaved between our nock and the evade.
118        // Save it before opening ours.
119        if let Some(cur) = self.current
120            && self.savable(cur)
121        {
122            self.save_event(cur);
123        }
124        let hidden = self.pending_ambush.as_ref().is_some_and(|a| a.hidden);
125        let kind = self.pending_ambush.as_ref().map(|a| a.kind);
126        if out.names_attacker {
127            // The line names who attacked US: an INBOUND unknown with the
128            // creature as attacker (a wholly-negated ambush is this shape).
129            let mut ev = self.blank(if hidden { "ambush" } else { "unknown" }, EventTarget::None);
130            ev.inbound = true;
131            ev.attacker = Some(lt);
132            ev.outcomes.push(out.kind);
133            ev.ambush = hidden;
134            ev.attack_kind = kind;
135            ev.resolutions.append(&mut self.pending_resolutions);
136            self.pending_ambush = None;
137            let id = self.open(ev);
138            self.current = Some(id);
139            self.seeking_damage = true;
140        } else {
141            // Inside a volley bracket the round IS a volley arrow; a nocked
142            // bow with no fire line printed means the outcome IS the fire.
143            let round_name = self
144                .active_sequence
145                .filter(|s| s.rounds_are_attacks())
146                .map(SequenceName::as_str);
147            let nock_fire = self.nocked.is_some() && !hidden && round_name.is_none();
148            let name = if hidden {
149                "ambush"
150            } else {
151                round_name.unwrap_or(if nock_fire { "fire" } else { "unknown" })
152            };
153            let mut ev = self.blank(name, EventTarget::Creature(lt.clone()));
154            ev.weapon = if nock_fire { self.nocked.clone() } else { None };
155            ev.outcomes.push(out.kind);
156            ev.ambush = hidden;
157            ev.attack_kind = kind;
158            ev.resolutions.append(&mut self.pending_resolutions);
159            let id = self.open(ev);
160            if round_name.is_some() {
161                // lineage as the attack branch would stamp it
162                let root = *self.spawn_root.get_or_insert(id);
163                self.ev_mut(id).root = Some(root);
164            }
165            self.pending_ambush = None;
166            if nock_fire {
167                self.nocked = None;
168            }
169            self.current_target = Some(lt);
170            self.current = Some(id);
171            self.seeking_damage = true;
172        }
173    }
174
175    /// The roll-or-outcome branch.
176    pub fn handle_rolls(&mut self, ctx: &mut LineCtx, line: &ChunkLine) {
177        if let Some(res) = Resolution::classify(line) {
178            self.handle_resolution(ctx, res);
179        } else if let Some(out) = Outcome::classify(line) {
180            self.handle_outcome(ctx, &out);
181        }
182    }

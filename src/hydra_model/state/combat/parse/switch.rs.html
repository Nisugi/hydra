1//! Target switching, for multi-target attacks.
2//!
3//! An inbound event is aimed at us and has no creature target by
4//! construction: it must never adopt one, or a creature's own emote later in
5//! the chunk hands it the damage it dealt US (seen in a recorded session: an
6//! ogre's "laughs hysterically"). A foreign-target event is
7//! bound to a non-creature for the same reason. A flare announce that names
8//! ITS OWN target is not a switch either: the damage that follows is the
9//! flare's, creature-attributed.
10
11use crate::state::combat::event::EventTarget;
12
13use super::LineCtx;
14use super::cursor::Cursor;
15
16impl Cursor<'_> {
17    pub fn handle_target_switch(&mut self, ctx: &LineCtx) {
18        let Some(lt) = ctx.line_target.clone() else {
19            return;
20        };
21        if ctx.line_redirect.is_some() || !self.seeking_damage {
22            return;
23        }
24        let Some(cur_id) = self.current else {
25            return;
26        };
27        let cur = self.ev(cur_id);
28        let flare_owned_target = ctx.line_flare
29            || (ctx.line_attack.is_none()
30                && cur
31                    .flares
32                    .iter()
33                    .any(|f| f.target.as_ref().and_then(|a| a.id) == lt.id));
34        if flare_owned_target || cur.inbound || cur.is_foreign_target() || cur.foreign_caster {
35            return;
36        }
37
38        match &self.current_target {
39            Some(ct) if ct.id != lt.id => {
40                // A real switch: save the previous event if it has data, and
41                // open an inherited one -- same attack name, same lineage,
42                // another AoE target sits at the SAME spawn-tree node.
43                if self.savable(cur_id) {
44                    self.save_event(cur_id);
45                }
46                let src = self.ev(cur_id);
47                let mut ev = self.blank(&src.name, EventTarget::Creature(lt.clone()));
48                ev.weapon.clone_from(&src.weapon);
49                ev.parent_flare.clone_from(&src.parent_flare);
50                ev.root = src.root;
51                ev.parent = src.parent;
52                ev.parent_confidence = src.parent_confidence;
53                // A line can be BOTH a switch and a new attack: mark the
54                // birth line so the attack branch can tell an artifact.
55                ev.born_line = Some(ctx.index);
56                // A held roll belongs to the target this line names: volley's
57                // per-arrow SMR precedes the arrow line.
58                ev.resolutions.append(&mut self.pending_resolutions);
59                let id = self.open(ev);
60                self.current = Some(id);
61                self.flare_ctx = None;
62                self.current_target = Some(lt);
63            }
64            None => {
65                // First target for the event. A creature can never be its own
66                // victim: on a targetless 3p initiation the only link is the
67                // ATTACKER (a triton assassin's ambush, 2024-11-21).
68                let attacker_id = cur.attacker.as_ref().and_then(|a| a.id);
69                if !(attacker_id.is_some() && attacker_id == lt.id) {
70                    self.ev_mut(cur_id).target = EventTarget::Creature(lt.clone());
71                    self.current_target = Some(lt);
72                }
73            }
74            Some(_) => {} // same target
75        }
76    }
77}

1//! The always-on fact handlers: statuses, spell losses, UCS.
2//!
3//! These run on every line, in or out of combat,
4//! before the event branches -- and each yields a [`Fact`] rather than a
5//! mutation, because the creature registry is the consumer above this one.

19    /// Which of the open event's flares a status rides on, 1-based
20    /// (`status_flare_seq`).
21    fn status_flare_seq(&self, cid: i64, line: &ChunkLine) -> Option<usize> {
22        let cur_id = self.current?;
23        let cur = self.ev(cur_id);
24        // A line that is itself a flare AND a status rides the flare it
25        // announces -- the NEXT position, since the flare branch has not
26        // appended it yet. Not when the flare is about to resume an
27        // interrupted swing: the positions belong to another event.
28        if let Some(lf) = FlareLine::classify(line)
29            && !(cur.inbound && self.interrupted_own.is_some())
30        {
31            let lf_id = lf.target.as_ref().and_then(|a| a.id);
32            if lf_id == Some(cid) || (lf_id.is_none() && cur.target.id() == Some(cid)) {
33                return Some(cur.flares.len() + 1);
34            }
35        }
36        let mut f = None;
37        if let Some(fc) = self.flare_ctx {
38            let ft = self.flare(fc).target.as_ref();
39            if let Some(ft) = ft {
40                if ft.id == Some(cid) {
41                    f = Some(fc);
42                }
43            } else if cur.target.id() == Some(cid) {
44                f = Some(fc);
45            }
46        }
47        // Only flares that printed AFTER the swing line: a status right
48        // after the swing's crit is the swing's, even when a pre-flare named
49        // the creature first.
50        let idx = match f {
51            Some(FlareRef::OnEvent(id, i)) if id == cur_id => Some(i),
52            Some(_) => None,
53            None => cur
54                .flares
55                .iter()
56                .rposition(|x| !x.pre && x.target.as_ref().and_then(|a| a.id) == Some(cid)),
57        };
58        idx.map(|i| i + 1)
59    }

61    /// The event a status rides on -- ONLY when that event touched the
62    /// subject.
63    fn status_event_for(&self, cid: i64, seq: Option<usize>) -> Option<usize> {
64        let cur_id = self.current?;
65        if seq.is_some() {
66            return Some(cur_id);
67        }
68        let cur = self.ev(cur_id);
69        let touched = cur.target.id() == Some(cid)
70            || cur.attacker.as_ref().and_then(|a| a.id) == Some(cid)
71            || cur
72                .flares
73                .iter()
74                .any(|f| f.target.as_ref().and_then(|a| a.id) == Some(cid));
75        touched.then_some(cur_id)
76    }

78    /// A status line, if this is one.
79    pub fn handle_status(&mut self, ctx: &mut LineCtx, line: &ChunkLine) {
80        let Some(st) = StatusLine::classify(line) else {
81            return;
82        };
83        // A player's line -- linked to a player -- is no creature's.
84        if st.target.as_ref().is_some_and(Actor::is_player) {
85            return;
86        }
87        let target_named = st.target_text.as_deref().filter(|t| !target::is_self(t));
88        if let Some(lt) = ctx.line_target.clone().filter(|a| a.id.is_some()) {
89            // ID-based: the most reliable
90            let cid = lt.id.unwrap_or_default();
91            ctx.line_status_id = Some(cid);
92            let seq = self.status_flare_seq(cid, line);
93            let event = self.status_event_for(cid, seq);
94            self.facts.push(Fact::Status {
95                subject: Subject::Creature(lt),
96                status: st.status,
97                action: st.action,
98                event,
99                flare_seq: seq,
100                line: ctx.index,
101            });
102        } else if let Some(name) = target_named {
103            // Name-based fallback: bound to nothing.
104            //
105            // A second-person "you" would be dropped by the registry's
106            // name lookup if it took this branch. A self capture falls
107            // through to the 2p branch instead, so the fact survives.
108            self.facts.push(Fact::Status {
109                subject: Subject::Creature(Actor::unlinked(name)),
110                status: st.status,
111                action: st.action,
112                event: None,
113                flare_seq: None,
114                line: ctx.index,
115            });
116        } else if !starts_your_or_you(&ctx.text) {
117            // Pronoun status lines ("It is knocked to the ground!") describe
118            // the creature we are fighting; during an INBOUND event an
119            // active flare's own target still binds (shield-spike knockdown).
120            let subject = self
121                .current_target
122                .clone()
123                .filter(|a| a.id.is_some())
124                .or_else(|| self.flare_ctx_target().cloned());
125            if let Some(subject) = subject {
126                let cid = subject.id.unwrap_or_default();
127                ctx.line_status_id = subject.id;
128                let seq = self.status_flare_seq(cid, line);
129                let event = self.status_event_for(cid, seq);
130                self.facts.push(Fact::Status {
131                    subject: Subject::Creature(subject),
132                    status: st.status,
133                    action: st.action,
134                    event,
135                    flare_seq: seq,
136                    line: ctx.index,
137                });
138            }
139        } else {
140            // 2p: the status is OURS. Never a creature application, but it
141            // IS a fact -- inbound attacks stun US.
142            self.facts.push(Fact::Status {
143                subject: Subject::Us,
144                status: st.status,
145                action: st.action,
146                event: None,
147                flare_seq: None,
148                line: ctx.index,
149            });
150        }
151    }

153    /// A spell wear-off line, with its cause when the chunk shows one.
154    pub fn handle_spell_loss(&mut self, line: &ChunkLine) {
155        let Some(loss) = SpellLoss::classify(line) else {
156            return;
157        };
158        let id = loss.target.id;
159        let cause = if self.chunk_dispels.iter().any(|d| d.is_none() || *d == id) {
160            Some(LossCause::Dispel)
161        } else if id.is_some_and(|i| self.chunk_deaths.contains(&i)) {
162            // This chunk's kills only; the registry's dead flag is not
163            // consulted here.
164            Some(LossCause::Death)
165        } else {
166            None
167        };
168        self.facts.push(Fact::SpellLoss {
169            subject: loss.target,
170            spell: loss.spell,
171            spell_name: loss.spell_name,
172            cause,
173        });
174    }

176    /// A UCS line.
177    pub fn handle_ucs(&mut self, line: &ChunkLine) {
178        let Some(u) = UcsLine::classify(line) else {
179            return;
180        };
181        let (creature, kind) = match u {
182            UcsLine::Position { tier, target } => (target, UcsKind::Position(tier)),
183            UcsLine::PositionInbound { tier, attacker } => {
184                (attacker, UcsKind::PositionInbound(tier))
185            }
186            // A tierup names no creature: it is against the current target.
187            UcsLine::Tierup { attack } => (self.current_target.clone(), UcsKind::Tierup(attack)),
188            UcsLine::SmiteApplied { target } | UcsLine::SmiteHeld { target } => {
189                (target, UcsKind::SmiteOn)
190            }
191            UcsLine::SmiteRemoved { target } => (target, UcsKind::SmiteOff),
192        };
193        if let Some(creature) = creature.filter(|a| a.id.is_some()) {
194            self.facts.push(Fact::Ucs { creature, kind });
195        }
196    }

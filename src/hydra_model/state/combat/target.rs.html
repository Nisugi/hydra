1//! Who a line is about: resolving captures to links, by position.
2//!
3//! Target, attacker, creature target, a link's name, and whether a capture
4//! is us -- without re-scanning. A [`ChunkLine`] is plain text plus the runs
5//! it was reassembled from; a regex capture is a byte span of that text; the
6//! runs the span overlaps are where the links are.
7//!
8//! # The rules
9//!
10//! | Question | Rule | Here |
11//! |---|---|---|
12//! | target from a `(?<target>)` capture | the **first** link in the capture text; a negative id (a player) is no target | [`link_in`] with [`Pick::First`], `id >= 0` |
13//! | attacker from an `(?<attacker>)` capture | the **last** link -- a flavour prefix can carry the attacker's own pronoun first (*"Froth bubbling on `<his>` lips, `<a berserker>` swings"*, where the first link would make the attacker `"his"`) | [`Pick::Last`], any id |
14//! | target from the whole line | only a **bolded** link; unbolded links are equipment and bystanders | [`bolded_link`], `id > 0` |
15//! | a creature's name | strip a possessive `'s` the game puts INSIDE the link, squeeze doubled spaces | [`link_name`] |
16//!
17//! The "possessive inside the link" case is no special case: the capture
18//! ends at the apostrophe and the closing tag falls outside it, but a byte
19//! span that ends inside a run still overlaps that run, so it is the same
20//! lookup.

28/// Someone a line names: a creature or a player.
29///
30/// `id` is the wire's `exist`, **negative for players**, and `None` when the
31/// line named them in prose with no link at all (a foreign caster named
32/// mid-sentence, or a pre-stripped log). `noun` is likewise the link's.
33#[derive(Debug, Clone, PartialEq, Eq)]
34pub struct Actor {
35    /// The `exist` id. Negative is a player.
36    pub id: Option<i64>,
37    /// The `noun`, when linked.
38    pub noun: Option<String>,
39    /// The display name, cleaned by [`link_name`].
40    pub name: String,
41}
42
43impl Actor {
44    /// Named in prose only: no link, no id.
45    #[must_use]
46    pub fn unlinked(name: &str) -> Self {
47        Self {
48            id: None,
49            noun: None,
50            name: name.trim().to_owned(),
51        }
52    }
53
54    /// Is this a player rather than a creature?
55    ///
56    /// Players have NEGATIVE exist ids.
57    #[must_use]
58    pub fn is_player(&self) -> bool {
59        self.id.is_some_and(|id| id < 0)
60    }
61}

63/// Which link to take when a span overlaps several.
64#[derive(Debug, Clone, Copy, PartialEq, Eq)]
65pub enum Pick {
66    /// The first: a target capture.
67    First,
68    /// The last: an attacker capture, whose prefix may carry a pronoun link.
69    Last,
70}
71
72/// A creature's display name from its link text.
73///
74/// The game sometimes puts the possessive INSIDE the link (`<a>grim gigas
75/// skald's</a> mastery of music`), which would register a second creature
76/// named `"grim gigas skald's"`. Same id, same creature: strip it. A
77/// hidden adjective leaves a doubled space (`halfling  cannibal`), squeezed
78/// for the same reason.
79#[must_use]
80pub fn link_name(text: &str) -> String {
81    let stripped = text.strip_suffix("'s").unwrap_or(text);
82    let mut out = String::with_capacity(stripped.len());
83    let mut last_space = false;
84    for c in stripped.chars() {
85        if c == ' ' {
86            if !last_space {
87                out.push(c);
88            }
89            last_space = true;
90        } else {
91            out.push(c);
92            last_space = false;
93        }
94    }
95    out.trim().to_owned()
96}

98/// Each run's byte span in the plain text, with its link if any.
99fn spans(line: &ChunkLine) -> Vec<(Range<usize>, &hydra_protocol::runs::Run)> {
100    let mut at = 0;
101    line.runs
102        .runs
103        .iter()
104        .map(|run| {
105            let start = at;
106            at += run.text.len();
107            (start..at, run)
108        })
109        .collect()
110}
111
112fn actor_of(run: &hydra_protocol::runs::Run) -> Option<Actor> {
113    let link = run.link.as_ref()?;
114    let LinkKind::Exist { id, noun } = &link.kind else {
115        return None;
116    };
117    Some(Actor {
118        id: id.parse::<i64>().ok(),
119        noun: Some(noun.clone()),
120        name: link_name(&link.text),
121    })
122}

124/// The `exist` link a capture span overlaps, first or last.
125///
126/// `span` is a byte range of `line.text()`, as `regex::Match::range` gives it.
127/// Returns `None` when no run in the span carries an `exist` link.
128#[must_use]
129pub fn link_in(line: &ChunkLine, span: Range<usize>, pick: Pick) -> Option<Actor> {
130    let mut found = spans(line)
131        .into_iter()
132        .filter(|(r, _)| r.start < span.end && r.end > span.start)
133        .filter_map(|(_, run)| actor_of(run));
134    match pick {
135        Pick::First => found.next(),
136        Pick::Last => found.next_back(),
137    }
138}

140/// The first **bolded** `exist` link on the line, with a positive id.
141///
142/// ONLY bolded creatures are targets. Non-bolded links are equipment,
143/// objects, or other non-combatants. The bold is the wire's own hostility
144/// marker
145/// (`frame/payload.rs`, `bold_depth`).
146#[must_use]
147pub fn bolded_link(line: &ChunkLine) -> Option<Actor> {
148    spans(line)
149        .into_iter()
150        .filter(|(_, run)| run.style.bold_depth > 0)
151        .filter_map(|(_, run)| actor_of(run))
152        .find(|a| a.id.is_some_and(|id| id > 0))
153}
154
155/// The first `exist` link anywhere on the line, any sign.
156///
157/// What `spell_loss` uses: a wear-off applies to players in view as well as
158/// creatures, so a negative id is a valid answer.
159#[must_use]
160pub fn any_link(line: &ChunkLine) -> Option<Actor> {
161    spans(line).into_iter().find_map(|(_, run)| actor_of(run))
162}

164/// Does a target capture refer to us?
165///
166/// `you` or `your` as a whole word, case-insensitive, anchored at the start:
167/// so *"a youngling"* is not us, and *"you with its tusk"* is.
168#[must_use]
169pub fn is_self(target_text: &str) -> bool {
170    let t = target_text.trim();
171    let lower = t.to_ascii_lowercase();
172    for word in ["your", "you"] {
173        if let Some(rest) = lower.strip_prefix(word) {
174            // `\b`: end of text, or a non-word character follows.
175            if rest
176                .chars()
177                .next()
178                .is_none_or(|c| !c.is_alphanumeric() && c != '_')
179            {
180                return true;
181            }
182        }
183    }
184    false
185}

1//! Unarmed Combat System lines: position tier, tierup, smite.
2//!
3//! Reads the `ucs` family. Five of its six patterns would read the
4//! creature's `exist` out of the tag (`<a exist="([0-9]+)"`), so they are
5//! written by hand (`defs::HAND_PORTED`): the sentence without the tag, and
6//! the creature from the line's first link.
7//!
8//! # Position is a tier, stored as a number
9//!
10//! Positioning tier words map to an ordinal, so a recorder can persist
11//! positioning as a number and queries can compare outbound and inbound
12//! tiers directly. [`PositionTier`] carries that ordinal.
13
14use super::defs::defs;
15use super::target::{self, Actor};
16use crate::state::lines::ChunkLine;

18/// How well positioned: `decent` < `good` < `excellent`.
19#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
20pub enum PositionTier {
21    /// Tier 1.
22    Decent,
23    /// Tier 2.
24    Good,
25    /// Tier 3.
26    Excellent,
27}
28
29impl PositionTier {
30    /// Every tier, ascending.
31    pub const ALL: [Self; 3] = [Self::Decent, Self::Good, Self::Excellent];
32
33    /// The ordinal recorded: 1, 2, 3.
34    #[must_use]
35    pub const fn ordinal(self) -> u8 {
36        match self {
37            Self::Decent => 1,
38            Self::Good => 2,
39            Self::Excellent => 3,
40        }
41    }
42
43    /// The word the game uses.
44    #[must_use]
45    pub const fn as_str(self) -> &'static str {
46        match self {
47            Self::Decent => "decent",
48            Self::Good => "good",
49            Self::Excellent => "excellent",
50        }
51    }
52
53    /// Parse the game's word, case-insensitively.
54    #[must_use]
55    pub fn parse(text: &str) -> Option<Self> {
56        Self::ALL
57            .into_iter()
58            .find(|t| t.as_str().eq_ignore_ascii_case(text.trim()))
59    }
60}

62/// The followup a tierup opens.
63#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
64#[allow(missing_docs)]
65pub enum UcsAttack {
66    Jab,
67    Grapple,
68    Punch,
69    Kick,
70}
71
72impl UcsAttack {
73    /// Every attack.
74    pub const ALL: [Self; 4] = [Self::Jab, Self::Grapple, Self::Punch, Self::Kick];
75
76    /// The game's word.
77    #[must_use]
78    pub const fn as_str(self) -> &'static str {
79        match self {
80            Self::Jab => "jab",
81            Self::Grapple => "grapple",
82            Self::Punch => "punch",
83            Self::Kick => "kick",
84        }
85    }
86
87    /// Parse the game's word, case-insensitively.
88    #[must_use]
89    pub fn parse(text: &str) -> Option<Self> {
90        Self::ALL
91            .into_iter()
92            .find(|a| a.as_str().eq_ignore_ascii_case(text.trim()))
93    }
94}

96/// One UCS line, classified.
97#[derive(Debug, Clone, PartialEq, Eq)]
98pub enum UcsLine {
99    /// `You have good positioning against a kobold.`
100    Position {
101        /// Our tier.
102        tier: PositionTier,
103        /// Against whom.
104        target: Option<Actor>,
105    },
106    /// `The triton brawler has decent positioning against you.` -- the
107    /// creature's tier against US, printed inside its own attack block.
108    PositionInbound {
109        /// Its tier.
110        tier: PositionTier,
111        /// Who.
112        attacker: Option<Actor>,
113    },
114    /// `Strike leaves foe vulnerable to a followup jab attack!`
115    Tierup {
116        /// Which followup.
117        attack: UcsAttack,
118    },
119    /// `A crimson mist suddenly surrounds ...` -- smite applied.
120    SmiteApplied {
121        /// The creature.
122        target: Option<Actor>,
123    },
124    /// `... held in the corporeal plane` -- smite holding.
125    SmiteHeld {
126        /// The creature.
127        target: Option<Actor>,
128    },
129    /// `... returns to an ethereal state` -- smite gone.
130    SmiteRemoved {
131        /// The creature.
132        target: Option<Actor>,
133    },
134}
135
136/// The substrings every UCS pattern contains one of.
137pub const GATE: [&str; 3] = [
138    "positioning against",
139    "vulnerable to a followup",
140    "crimson mist",
141];
142
143/// The line's first creature link, positive id.
144fn creature(line: &ChunkLine) -> Option<Actor> {
145    target::any_link(line).filter(|a| a.id.is_some_and(|id| id > 0))
146}
147
148impl UcsLine {
149    /// Classify one line as a UCS event.
150    #[must_use]
151    pub fn classify(line: &ChunkLine) -> Option<Self> {
152        let text = line.text();
153        if !GATE.iter().any(|g| text.contains(g)) {
154            return None;
155        }
156        let (def, caps) = defs().first_match("ucs", &text)?;
157        match def.name.as_str() {
158            "position" => Some(Self::Position {
159                tier: PositionTier::parse(caps.name("tier")?.as_str())?,
160                target: creature(line),
161            }),
162            "position_inbound" => Some(Self::PositionInbound {
163                tier: PositionTier::parse(caps.name("tier")?.as_str())?,
164                attacker: creature(line),
165            }),
166            // The pattern has an unnamed group: `(jab|grapple|punch|kick)`.
167            "tierup" => Some(Self::Tierup {
168                attack: UcsAttack::parse(caps.get(1)?.as_str())?,
169            }),
170            "smite_applied" => Some(Self::SmiteApplied {
171                target: creature(line),
172            }),
173            "smite_held" => Some(Self::SmiteHeld {
174                target: creature(line),
175            }),
176            "smite_removed" => Some(Self::SmiteRemoved {
177                target: creature(line),
178            }),
179            _ => None,
180        }
181    }
182}

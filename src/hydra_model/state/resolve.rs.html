1//! What does a name mean? Turning `"kron"` into an item the game will accept.
2//!
3//! # Resolution only
4//!
5//! Using an item splits in two:
6//!
7//! | | What it does |
8//! |---|---|
9//! | **Resolution** | "which item does this name mean?" |
10//! | **Manipulation** | get it, wear it, swap hands, retry |
11//!
12//! Manipulation sends commands and watches for the reply, which needs a
13//! connection and roundtime; it belongs above this crate, which has no
14//! socket. Resolution needs nothing but state, and is what everything above
15//! it will ask first. It is all this module does.
16//!
17//! # The ordering is the real content
18//!
19//! [`GameState::find_items`] sorts by two keys, in the order the game itself
20//! resolves a bare noun in:
21//!
22//! 1. **How specific the match is** -- the whole name beats part of it beats
23//!    a loose hit ([`Specificity`]).
24//! 2. **Where the item is** -- hands, then the ready list, then worn, then
25//!    inside a container ([`Location`]).
26//!
27//! Both matter, and the second is why this cannot live below `GameState`: it
28//! reads the hands, the ready list and the inventory tree at once.
29//!
30//! # No regex
31//!
32//! Matching is over words, so a caller's string is never compiled into a
33//! pattern: a profile naming a bag `"pack (old)"` cannot raise or mismatch
34//! on its metacharacters.
35

39/// How well a name matched an item, best first.
40///
41/// A type rather than `0 | 1 | 2`.
42#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
43pub enum Specificity {
44    /// The name is the item's whole name.
45    Exact,
46    /// The name is a run of whole words inside the item's name.
47    Phrase,
48    /// The name matched loosely -- a single word, or words in order but apart.
49    Loose,
50}
51
52/// Where an item is, in the order the game resolves a bare noun.
53///
54/// Hands, the ready list, worn, in a container, and last items known only
55/// from the inventory tree.
56#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
57pub enum Location {
58    /// In a hand. Already held, so nothing needs fetching.
59    Hand,
60    /// Named by the ready list.
61    Ready,
62    /// Worn. Needs `remove`, not `get`.
63    Worn,
64    /// Inside a container that has been looked in.
65    Container,
66    /// Known only from the inventory snapshot -- possibly in a container
67    /// nobody has opened.
68    Inventory,
69}

71/// One candidate for a name.
72#[derive(Clone, Debug, PartialEq, Eq)]
73pub struct Match {
74    /// The item's exist id. **What a command targets.**
75    pub id: String,
76    /// Its display name.
77    pub name: String,
78    /// How well the name matched.
79    pub specificity: Specificity,
80    /// Where it is.
81    pub location: Location,
82}
83
84/// Whether `name` matches `item`, and how well.
85///
86/// Whether it matches and how well are one question, so one function answers
87/// both.
88///
89/// **Word-based, not regex.** The three cases are tested directly, so no
90/// caller's string is ever compiled.
91#[must_use]
92pub fn match_of(item_name: &str, name: &str) -> Option<Specificity> {
93    let wanted = name.trim().to_lowercase();
94    if wanted.is_empty() {
95        return None;
96    }
97    let actual = item_name.trim().to_lowercase();
98    if actual == wanted {
99        return Some(Specificity::Exact);
100    }
101    // A run of whole words: `"long bow"` in `"a scorched glowbark long bow"`,
102    // but not `"ow"` in `"bow"`.
103    if contains_words(&actual, &wanted) {
104        return Some(Specificity::Phrase);
105    }
106    // The loose form: the first word, then anything, then the rest. Words
107    // in order but not adjacent.
108    words_in_order(&actual, &wanted).then_some(Specificity::Loose)
109}
110
111/// Whether `haystack` contains `needle` on whole-word boundaries.
112fn contains_words(haystack: &str, needle: &str) -> bool {
113    haystack
114        .match_indices(needle)
115        .any(|(at, _)| starts_word(haystack, at) && ends_word(haystack, at + needle.len()))
116}
117
118/// Whether position `at` begins a word.
119fn starts_word(text: &str, at: usize) -> bool {
120    at == 0 || text[..at].ends_with(char::is_whitespace)
121}
122
123/// Whether position `at` ends a word.
124fn ends_word(text: &str, at: usize) -> bool {
125    at == text.len() || text[at..].starts_with(char::is_whitespace)
126}
127
128/// Whether every word of `needle` appears in `haystack`, in order, on word
129/// boundaries.
130fn words_in_order(haystack: &str, needle: &str) -> bool {
131    let mut rest = haystack;
132    for word in needle.split_whitespace() {
133        let Some(at) = rest
134            .match_indices(word)
135            .find(|&(at, _)| starts_word(rest, at) && ends_word(rest, at + word.len()))
136            .map(|(at, _)| at)
137        else {
138            return false;
139        };
140        rest = &rest[at + word.len()..];
141    }
142    true
143}

145impl GameState {
146    /// Every item this name could mean, best first.
147    ///
148    /// **Sends nothing** -- it reads the
149    /// hands, the ready list and the inventory this state already holds, so
150    /// the answer is only as good as what the game has said. A caller that
151    /// needs certainty refreshes inventory first, which is a behavior's job.
152    ///
153    /// Ordered by [`Specificity`] then [`Location`]: the order the game
154    /// itself resolves a bare noun in. Duplicate ids collapse; items with the
155    /// **same name** collapse too, because they are interchangeable -- and a
156    /// caller picking between two identical names has no basis to choose.
157    #[must_use]
158    pub fn find_items(&self, name: &str) -> Vec<Match> {
159        let mut found: Vec<Match> = self
160            .candidates()
161            .filter_map(|(id, item_name, location)| {
162                match_of(&item_name, name).map(|specificity| Match {
163                    id,
164                    name: item_name,
165                    specificity,
166                    location,
167                })
168            })
169            .collect();
170        // `sort_by_key` is stable, so equal (specificity, location) pairs keep
171        // the order `candidates` yielded, which is what makes a replay
172        // deterministic.
173        found.sort_by_key(|m| (m.specificity, m.location));
174        let mut seen_ids = Vec::new();
175        let mut seen_names = Vec::new();
176        found.retain(|m| {
177            let fresh = !seen_ids.contains(&m.id) && !seen_names.contains(&m.name);
178            if fresh {
179                seen_ids.push(m.id.clone());
180                seen_names.push(m.name.clone());
181            }
182            fresh
183        });
184        found
185    }
186
187    /// The single best item this name means, or `None`.
188    ///
189    /// A caller decides what an unresolvable name means, because at this
190    /// layer it is a question, not an error.
191    #[must_use]
192    pub fn find_item(&self, name: &str) -> Option<Match> {
193        self.find_items(name).into_iter().next()
194    }
195
196    /// Which hand holds this id, if either.
197    ///
198    /// **By id, not name** -- which is what
199    /// [`Hand`](super::hands::Hand) carrying the wire's `exist=` bought.
200    #[must_use]
201    pub fn hand_holding(&self, id: &str) -> Option<Held> {
202        if self.right_hand.holds(id) {
203            return Some(Held::Right);
204        }
205        self.left_hand.holds(id).then_some(Held::Left)
206    }
207
208    /// Whether the object with exist id `id` (a leading `#` allowed) is known
209    /// to be the character's own: in a hand, worn, named by a stow or ready
210    /// slot, in the inventory tree, or listed inside a container window whose
211    /// object is one of those.
212    ///
213    /// **Known, not merely possible.** `false` for anything the model has
214    /// not seen as carried -- an object in the room, a player, a bin, or an
215    /// item of the character's the game has not listed yet -- so a caller
216    /// that must never put something where it is lost refuses on `false`. A container window is
217    /// not taken as the character's on its own: only one whose object is
218    /// carried, one level down.
219    #[must_use]
220    pub fn carries(&self, id: &str) -> bool {
221        let id = id.strip_prefix('#').unwrap_or(id);
222        if id.is_empty() {
223            return false;
224        }
225        self.carries_directly(id)
226            || self.inventory.containers().any(|(window, container)| {
227                let object = container.target.as_deref().unwrap_or(window);
228                self.carries_directly(object) && container.items.iter().any(|item| item.id == id)
229            })
230    }
231
232    /// [`Self::carries`] without looking inside a container window.
233    fn carries_directly(&self, id: &str) -> bool {
234        let listed = |items: Option<&[super::containers::ItemRef]>| {
235            items.is_some_and(|items| items.iter().any(|item| item.id == id))
236        };
237        self.hand_holding(id).is_some()
238            || listed(self.worn.items())
239            || StowSlot::ALL
240                .into_iter()
241                .any(|slot| self.containers.stow(slot).is_some_and(|item| item.id == id))
242            || ReadySlot::ALL.into_iter().any(|slot| {
243                self.containers
244                    .ready(slot)
245                    .is_some_and(|item| item.id == id)
246            })
247            || self.inventory_snapshot.get(id).is_some()
248    }
249
250    /// Every item the game has named, with where it is.
251    ///
252    /// One pass, including the inventory-only fallback. Yielded in
253    /// [`Location`] order so the sort above is stable against it.
254    fn candidates(&self) -> impl Iterator<Item = (String, String, Location)> + '_ {
255        let hands = [&self.right_hand, &self.left_hand]
256            .into_iter()
257            .filter_map(|hand| {
258                Some((
259                    hand.id()?.to_owned(),
260                    hand.name()?.to_owned(),
261                    Location::Hand,
262                ))
263            });
264        let ready = ReadySlot::ALL.into_iter().filter_map(|slot| {
265            let item = self.containers.ready(slot)?;
266            Some((item.id.clone(), item.text.clone(), Location::Ready))
267        });
268        let carried = self.inventory_snapshot.all().map(|item| {
269            // `worn,player` is worn; anything else the snapshot lists is
270            // inside something. The distinction decides `remove` against
271            // `get`.
272            let where_it_is = if item.relation == "worn" {
273                Location::Worn
274            } else {
275                Location::Inventory
276            };
277            (item.id.clone(), item.name.clone(), where_it_is)
278        });
279        let in_containers = self.inventory.containers().flat_map(|(_, container)| {
280            container
281                .items
282                .iter()
283                .map(|item| (item.id.clone(), item.text.clone(), Location::Container))
284        });
285        hands.chain(ready).chain(carried).chain(in_containers)
286    }
287}

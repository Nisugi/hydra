1//! The thin tier: a typed frame for every wire tag not modelled in depth.
2//!
3//! Room, prompt and vitals have real handlers in `dispatch.rs`. Everything
4//! else lands here, and it still **parses and renders**: a typed variant
5//! carrying the wire's own attributes, never a `todo!()` and never a drop.
6
7use super::inner::inner_display_text;
8use crate::frame::{ActiveEffect, Attrs, DialogWidgets, Frame};
9use crate::tags;
10use crate::text;

12/// Shortest interval between pulses, in seconds.
13///
14/// The wire sends `<pulse mana="1"/>` bare -- the only form in a 272-file
15/// corpus sample, 1,419 of 1,419 occurrences -- so the interval is implied
16/// rather than stated and a default is the only way to have one. `0` would
17/// claim the pulse is instantaneous. Measured over 4,068 gaps between two
18/// pulses in 250 Lich logs: 45 seconds at the fifth percentile, 60 at the
19/// median, 75 at the ninety-fifth, spread evenly between, each gap its own.
20const PULSE_MIN_SECS: u32 = 45;
21
22/// Longest interval between pulses, in seconds. See [`PULSE_MIN_SECS`].
23const PULSE_MAX_SECS: u32 = 75;

25/// The thin tier: a typed frame carrying the wire's own attributes.
26///
27/// A name in [`tags::is_known`] gets its
28/// variant; a name that is not gets [`Frame::UnknownTag`] with the raw bytes,
29/// so a protocol change announces itself instead of being swallowed.
30pub(super) fn thin_frame(name: &str, tag: &str, dialog: Option<&str>) -> Frame {
31    // Built on demand, not up front. Most arms never read the bag -- the
32    // hands, the timers, `indicator` -- and building it eagerly would walk
33    // every attribute of every one of them into owned strings only to drop
34    // them: in a release build, 1M `parse_line` calls over `<indicator>`,
35    // `<roundTime>`, `<left>` and `<spell>` take ~800 ms eager against
36    // ~480 ms lazy. The paired ones are worse than wasted: `attributes` over
37    // `<left exist=..>sword</left>` walks the body too.
38    let attrs = || -> Attrs { text::attributes(tag) };
39    let id = || text::attribute(tag, "id").unwrap_or_default();
40    match name {
41        "spell" => Frame::Spell {
42            text: inner_display_text(tag),
43            link: text::link_from_tag(tag),
44        },
45        "left" => Frame::LeftHand {
46            item: inner_display_text(tag),
47            link: text::link_from_tag(tag),
48        },
49        "right" => Frame::RightHand {
50            item: inner_display_text(tag),
51            link: text::link_from_tag(tag),
52        },
53        // A timer whose value is not a number -- absent, negative, too large,
54        // or words -- is a tag that cannot be read, not a timer of zero: a
55        // zero ends a roundtime the game never ended. The frame keeps its
56        // bytes, and the model leaves the timer as it was.
57        "roundTime" | "castTime" | "timer" => {
58            let Some(value) = text::attribute_u32(tag, "value") else {
59                return Frame::MalformedTag {
60                    raw: tag.to_owned(),
61                };
62            };
63            match name {
64                "roundTime" => Frame::RoundTime { value },
65                "castTime" => Frame::CastTime { value },
66                _ => Frame::AimTime { value },
67            }
68        }
69        // KNOWN LIMIT, measured and deliberately not closed: a `<label>` with
70        // a BODY (`<label id='x'>text</label>`) reports an empty `value` and
71        // its text arrives as a separate `Text` frame.
72        //
73        // The body fallback below only fires for a body on the open tag, and
74        // `<label>` is not in `is_paired`, so the paired form never reaches
75        // here whole. Adding it there would close the gap -- and it is not
76        // worth the line, because the form does not exist on this wire:
77        //
78        //   $ grep -h -o '<label'   <272 stratified files> | wc -l  -> 1486284
79        //   $ grep -h -o '</label>' <272 stratified files> | wc -l  ->       0
80        //
81        // 1,486,284 opens, zero closes. Every one is self-closing with
82        // `value=`. UPGRADE TRIGGER: a non-zero `</label>` count in the corpus
83        // replay means adding `"label"` to `is_paired` in `parser.rs`.
84        "label" => Frame::Label {
85            id: id(),
86            value: text::attribute(tag, "value").unwrap_or_else(|| inner_display_text(tag)),
87            dialog: dialog.map(str::to_owned),
88            attrs: attrs(),
89        },
90        "crtrStatus" => Frame::CreatureStatus {
91            id: text::attribute(tag, "exist").unwrap_or_default(),
92            attrs: attrs(),
93        },
94        "roommeta" => Frame::RoomMeta(room_meta(tag)),
95        "closeDialog" | "closedialog" => Frame::CloseDialog { id: id() },
96        "exposeDialog" | "exposeStream" | "exposeContainer" => Frame::Expose {
97            kind: name.to_owned(),
98            id: id(),
99        },
100        "deleteContainer" => Frame::DeleteContainer { id: id() },
101        "app" => Frame::AppInfo {
102            character: text::attribute(tag, "char").unwrap_or_default(),
103            // `game` is the instance, and two characters of the same name in
104            // Prime and Platinum are two different characters -- so a
105            // multi-session client cannot identify a session without it.
106            game: text::attribute(tag, "game").unwrap_or_default(),
107            title: text::attribute(tag, "title").unwrap_or_default(),
108        },
109        "image" => Frame::InjuryImage {
110            id: id(),
111            name: text::attribute(tag, "name").unwrap_or_default(),
112            dialog: dialog.map(str::to_owned),
113            attrs: attrs(),
114        },
115        // The `Icon` prefix is kept, deliberately. The wire id IS
116        // `IconSTUNNED` in all 22,749 occurrences across a 272-file sample,
117        // and stripping is a rename: it makes the frame disagree with the
118        // bytes. Parsing is here and naming above; the layer that wants
119        // `STUNNED` can strip five characters.
120        //
121        // `visible=` absent is treated as inactive, and that is safe rather
122        // than merely convenient: the attribute is present in all 22,749 of
123        // those occurrences, so "absent" is not a case the wire produces.
124        "indicator" => Frame::StatusIndicator {
125            id: id(),
126            active: text::attribute(tag, "visible").as_deref() == Some("y"),
127        },
128        // `<objectives>` has NO arm here, deliberately. It is a paired tag,
129        // so every form of it -- self-closing, empty, or carrying rows --
130        // reaches `dispatch::objectives` and never this function. An arm
131        // here would be unreachable code that nothing tests.
132        //
133        // A stray `<objective>` outside any envelope IS reachable: typed as
134        // itself rather than falling through to the placement-attrs bag.
135        "objective" => stray_objective(tag),
136        // A `<menu>` is assembled whole in `dispatch.rs`, every form of it,
137        // and never reaches here: it has no arm, as `<objectives>` has none.
138        // This is the stray-item path -- an `<mi>` outside any menu, which
139        // the wire is not known to send. Typed as a one-item menu with no id
140        // rather than dropped, so a wire change goes visible instead of
141        // silent.
142        "mi" => Frame::MenuResponse(crate::frame::Menu {
143            id: id(),
144            path: text::attribute(tag, "path"),
145            categories: Vec::new(),
146            items: vec![crate::parser::dispatch::menu_item(tag)],
147        }),
148        "switchQuickBar" => Frame::QuickbarSwitch { id: id() },
149        "vellumTimer" => script_timer(tag),
150        "openDialog" | "opendialog" => {
151            let title = text::attribute(tag, "title");
152            if text::attribute(tag, "resident").as_deref() == Some("true") {
153                Frame::DialogPanelOpen {
154                    id: id(),
155                    title,
156                    attrs: attrs(),
157                }
158            } else {
159                Frame::DialogOpen {
160                    id: id(),
161                    title,
162                    attrs: attrs(),
163                }
164            }
165        }
166        "cmdButton" | "closeButton" | "dropDownBox" | "editBox" | "upDownEditBox" | "skin"
167        | "link" | "menuLink" | "sep" | "checkBox" | "radio" | "hScrollBar" | "vScrollBar" => {
168            Frame::DialogWidgets(DialogWidgets {
169                id: id(),
170                dialog: dialog.map(str::to_owned),
171                kind: name.to_owned(),
172                widgets: vec![attrs()],
173            })
174        }
175        // Containers, inventory and the remaining singletons live in
176        // `thin_frame_rest` so neither function exceeds the 100-line clippy
177        // ceiling; the split is alphabetically arbitrary but the boundary is
178        // stable, and both halves are one flat match.
179        _ => thin_frame_rest(name, tag),
180    }
181}

183/// The second half of [`thin_frame`]: containers, inventory, and the rest.
184fn thin_frame_rest(name: &str, tag: &str) -> Frame {
185    let id = || text::attribute(tag, "id").unwrap_or_default();
186    match name {
187        "container" => Frame::Container {
188            id: id(),
189            title: text::attribute(tag, "title"),
190            target: text::attribute(tag, "target"),
191            attrs: text::attributes(tag),
192        },
193        "clearContainer" => Frame::ClearContainer { id: id() },
194        // The dictionary version the server just stated. Typed rather than
195        // bagged because a client compares it against what it holds to know
196        // whether it is current.
197        "cmdtimestamp" => Frame::CmdTimestamp {
198            version: text::attribute(tag, "data").unwrap_or_default(),
199        },
200        // An inventory response's token is carried in `id=`, not in an
201        // attribute called `token`: reading `token` would leave the field
202        // always empty, so no response could be matched to its request.
203        // `inventoryManager` and `inventoryViewItem` are assembled rather
204        // than thinned -- one in `dispatch.rs` (its body is `<i>` children on
205        // one line), one in `view_item.rs` (its body spans lines). Neither
206        // reaches this function.
207        "launchURL" | "LaunchURL" => Frame::LaunchUrl {
208            url: text::attribute(tag, "src")
209                .or_else(|| text::attribute(tag, "url"))
210                .unwrap_or_default(),
211        },
212        // `mana` is the attribute's VALUE, not its presence. Presence carries
213        // no information at all: the wire always sends the attribute, and
214        // over 272 corpus files the split is almost exactly even -- 710
215        // `mana="1"` against 709 `mana="0"`. The two alternate, and `1` says
216        // the NEXT pulse brings mana: over 4,307 pulses in 250 Lich logs the
217        // mana bar rose with a pulse 708 times, every one of them `mana="0"`,
218        // the pulse after a `mana="1"`.
219        //
220        // The `min`/`max` defaults of 45/75 are load bearing: every `<pulse>`
221        // in that sample is bare (`<pulse mana="1"/>`), so the wire states
222        // the alternation and leaves the interval implied.
223        "pulse" => Frame::Pulse {
224            mana: text::attribute(tag, "mana").as_deref() == Some("1"),
225            min: text::attribute_u32(tag, "min").unwrap_or(PULSE_MIN_SECS),
226            max: text::attribute_u32(tag, "max").unwrap_or(PULSE_MAX_SECS),
227        },
228        "worldEvent" => Frame::WorldEvent {
229            realm: text::attribute(tag, "realm").unwrap_or_default(),
230            expires_min: text::attribute_u32(tag, "expires"),
231            text: inner_display_text(tag),
232        },
233        "PantheonStatus" => Frame::PantheonStatus {
234            value: text::attribute_u32(tag, "value").unwrap_or_default(),
235        },
236        // `dynaStream` and `clearDynaStream` are NOT window declarations --
237        // they feed and clear a streamBox control -- and are handled in
238        // `dispatch.rs` beside `<stream>`. `stream` itself is not in this arm
239        // for the same reason: every form of it is handled in `dispatch.rs`,
240        // and a name here would only invite the next caller of `thin_frame`
241        // to declare a window nobody sent.
242        "streamId" => Frame::StreamWindow {
243            id: id(),
244            title: text::attribute(tag, "title"),
245            subtitle: text::attribute(tag, "subtitle"),
246            attrs: text::attributes(tag),
247        },
248        // A known tag with no dedicated variant still becomes one: an effect
249        // row when it looks like one, otherwise a window hint carrying its
250        // attributes. Nothing is dropped.
251        _ if tags::is_known(name) => known_fallback(name, tag),
252        // Unmodelled markup, carried whole, never dropped.
253        _ => Frame::UnknownTag {
254            name: name.to_owned(),
255            raw: tag.to_owned(),
256        },
257    }
258}

260/// A known tag with no dedicated variant.
261fn known_fallback(name: &str, tag: &str) -> Frame {
262    let attrs = text::attributes(tag);
263    match name {
264        // Typed, not a bag: the session's readiness gate keys on it, and
265        // matching a `WindowHints` by its id string is re-reading markup.
266        "endSetup" => Frame::EndSetup,
267        // Typed for the same reason: the model's "is that link me?" keys on it.
268        "playerID" => Frame::PlayerId {
269            id: text::attribute(tag, "id").unwrap_or_default(),
270        },
271        // Typed for the same reason: a Lich started late is told its game
272        // by it.
273        "settingsInfo" => Frame::SettingsInfo {
274            instance: text::attribute(tag, "instance"),
275            attrs,
276        },
277        "settings" | "sentSettings" | "presets" | "palette" | "macros" | "mode" | "FEVersion"
278        | "LichWebUI" => Frame::WindowHints {
279            id: name.to_owned(),
280            attrs,
281        },
282        // `reward` is NOT here, though it looks like an effect row. The wire
283        // sends `<reward type='fame' amount='20000'/>` -- self-closing, with
284        // neither `id` nor `time` nor body text -- so every field this arm
285        // reads would be absent, and the result an empty husk with `type`
286        // and `amount` dropped on the floor. In a 1,547-file corpus sample
287        // every `<reward>` has that shape, `type` being `fame` or
288        // `experience`.
289        //
290        // Falling through to `WindowHints` below carries `attrs` whole, which
291        // keeps `type` and `amount` and anything Simutronics adds later, such
292        // as a `<reward type='custom' label='...'/>`.
293        "celebration" => Frame::ActiveEffect(ActiveEffect {
294            category: name.to_owned(),
295            id: text::attribute(tag, "id").unwrap_or_default(),
296            text: inner_display_text(tag),
297            time: text::attribute(tag, "time").and_then(|t| t.parse().ok()),
298        }),
299        _ => Frame::WindowHints {
300            id: text::attribute(tag, "id").unwrap_or_else(|| name.to_owned()),
301            attrs,
302        },
303    }
304}

306/// `<roommeta>`: eight environment codes, each `None` when unstated.
307///
308/// The codes are the game's own and are NOT decoded here; see
309/// [`crate::frame::RoomMeta`] for why, and for the census that says all
310/// eight arrive together.
311fn room_meta(tag: &str) -> crate::frame::RoomMeta {
312    let code = |name: &str| text::attribute(tag, name).and_then(|v| v.parse().ok());
313    crate::frame::RoomMeta {
314        attrs: text::attributes(tag),
315        weather: code("weather"),
316        bonfire: code("bonfire"),
317        inside: code("inside"),
318        water: code("water"),
319        sanctuary: code("sanctuary"),
320        realm: code("realm"),
321        climate: code("climate"),
322        terrain: code("terrain"),
323        locker: code("locker"),
324    }
325}

327/// An `<objective>` that arrived outside any `<objectives>` envelope.
328///
329/// Treated as a patch of one: the wire is not known to send this, and
330/// assuming a refresh would let one stray row erase the list.
331fn stray_objective(tag: &str) -> Frame {
332    Frame::ObjectivesUpdate {
333        action: crate::frame::ObjectivesAction::Patch,
334        entries: vec![crate::parser::dispatch::objective(tag)],
335    }
336}

338/// A script's timer, `<vellumTimer id= value=>`: a name, and the second it
339/// ends. One with no name, or no number for its end, cannot be read.
340fn script_timer(tag: &str) -> Frame {
341    match (
342        text::attribute(tag, "id").filter(|id| !id.is_empty()),
343        text::attribute_u32(tag, "value"),
344    ) {
345        (Some(id), Some(value)) => Frame::Timer {
346            id,
347            value,
348            category: text::attribute(tag, "category").filter(|list| !list.is_empty()),
349        },
350        _ => Frame::MalformedTag {
351            raw: tag.to_owned(),
352        },
353    }

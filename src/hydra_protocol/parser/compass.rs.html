1//! The `<dir value=>` tokens inside a `<compass>`.
2//!
3//! [`directions`] is apart from dispatch because it is not dispatch: it takes
4//! no `Parser` and emits no frame, and it has tests of its own.
5
6use crate::text;
7
8/// The `<dir value=>` tokens inside a `<compass>`.
9pub(super) fn directions(tag: &str) -> Vec<String> {
10    let mut out = Vec::new();
11    let mut rest = tag;
12    while let Some(at) = rest.find("<dir ") {
13        rest = &rest[at..];
14        let Some(end) = rest.find('>') else { break };
15        if let Some(value) = text::attribute(&rest[..=end], "value") {
16            out.push(value);
17        }
18        rest = &rest[end + 1..];
19    }
20    out
21}

23#[cfg(test)]
24mod tests {
25    use super::*;
26    use proptest::prelude::*;
27
28    /// A compass with an arbitrary run of `<dir>` tokens, well-formed or not.
29    ///
30    /// `tests/parser_never_panics.rs` cannot reach this function's loop: it
31    /// never writes `<dir`. `"compass"` is in its `tagish()` name list, but a
32    /// bare `<compass/>` walks this loop zero times, so neither the
33    /// `find('>')`-missing break nor the `attribute`-returns-`None` arm would
34    /// run. A test that passes over code it never runs proves nothing.
35    fn compass() -> impl Strategy<Value = String> {
36        let token = prop::sample::select(vec![
37            "<dir value=\"n\"/>",
38            "<dir value='sw'/>",
39            "<dir value=\"\"/>",
40            // No `value` attribute, but WITH the trailing space the scan
41            // requires -- `<dir/>` alone never matches `"<dir "` and so never
42            // reaches the `attribute` call at all.
43            "<dir foo=\"x\"/>",
44            "<dir />",
45            "<dir/>",           // unmatched by the scan; kept as a control
46            "<dir value=\"n\"", // never closed: the `break` arm
47            "<dir >",
48            "<notdir value=\"n\"/>",
49            "text between",
50        ]);
51        prop::collection::vec(token, 0..8)
52            .prop_map(|parts| format!("<compass>{}</compass>", parts.concat()))
53    }
54
55    proptest! {
56        #![proptest_config(ProptestConfig::with_cases(1024))]
57
58        /// **A direction is never invented.**
59        ///
60        /// The count can only fall below the number of `<dir ` tokens -- one
61        /// without a `value`, or one that never closes, yields nothing. It can
62        /// never rise. A compass that reported an exit the game did not send
63        /// would route a walk into a wall.
64        #[test]
65        fn directions_never_exceed_the_dir_tokens_present(tag in compass()) {
66            let found = directions(&tag);
67            let tokens = tag.matches("<dir ").count();
68            prop_assert!(
69                found.len() <= tokens,
70                "{} directions from {} tokens in {tag:?}",
71                found.len(),
72                tokens
73            );
74        }
75
76        /// Every direction returned was written in the input.
77        ///
78        /// Paired with the count bound above, this is the conservation claim:
79        /// nothing invented, nothing transformed. Together they say the
80        /// function only ever selects.
81        #[test]
82        fn every_direction_came_from_the_markup(tag in compass()) {
83            for value in directions(&tag) {
84                prop_assert!(
85                    tag.contains(&value) || value.is_empty(),
86                    "direction {value:?} appears nowhere in {tag:?}"
87                );
88            }
89        }
90
91        /// Arbitrary bytes never panic and never wedge the scan.
92        ///
93        /// The loop advances `rest` by hand, so a slice that failed to move
94        /// would hang rather than crash -- which a panic-only test would not
95        /// notice. Proptest's own timeout is what catches that here.
96        #[test]
97        fn arbitrary_input_terminates(tag in ".*") {
98            let _ = directions(&tag);
99        }
100    }
101
102    #[test]
103    fn a_real_compass_reads_every_exit() {
104        // Real wire shape, so the properties above are anchored to a case with
105        // a known answer. A property that holds vacuously over inputs yielding
106        // nothing would satisfy both bounds trivially.
107        let tag = concat!(
108            "<compass><dir value=\"n\"/><dir value=\"e\"/>",
109            "<dir value=\"sw\"/><dir value=\"out\"/></compass>"
110        );
111        assert_eq!(directions(tag), ["n", "e", "sw", "out"]);
112    }
113
114    #[test]
115    fn an_unterminated_dir_stops_the_scan_without_hanging() {
116        // The `break` arm.
117        assert_eq!(
118            directions("<compass><dir value=\"n\"/><dir value=\"e\""),
119            ["n"]
120        );
121    }
122
123    #[test]
124    fn a_dir_without_a_value_yields_nothing_rather_than_an_empty_exit() {
125        // The `attribute` -> None arm. An empty string pushed here would be a
126        // direction nothing could walk.
127        //
128        // **`<dir/>` DOES NOT TEST THIS**: the scan looks for `"<dir "` WITH a
129        // trailing space, so `<dir/>` is never matched and the loop runs zero
130        // times. Asserting on that input alone would pass a mutant pushing
131        // `unwrap_or_default()`: the assertion right, and the input never
132        // reaching the branch.
133        //
134        // `<dir foo="x"/>` is the form that reaches the `attribute` call and
135        // gets `None` back.
136        assert_eq!(
137            directions("<compass><dir foo=\"x\"/></compass>"),
138            Vec::<String>::new()
139        );
140        // The unmatched form, recorded so the distinction is not re-lost.
141        assert_eq!(
142            directions("<compass><dir/></compass>"),
143            Vec::<String>::new()
144        );
145    }
146}

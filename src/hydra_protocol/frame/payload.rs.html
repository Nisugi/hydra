6/// A run of text plus the markup that was open around it.
7#[derive(Clone, Debug, PartialEq, Eq)]
8pub struct TextFrame {
9    /// Display text, entity-decoded, control characters stripped.
10    pub content: String,
11    /// The stream this text belongs to (`""` is the main window).
12    pub stream: String,
13    /// Structural markup, not resolved colour. See the module docs.
14    pub style: Style,
15    /// The link this text sits inside, if any. The **outermost** one open.
16    pub link: Option<Link>,
17    /// The innermost open `<a exist=>`, when it is not already [`Self::link`].
18    ///
19    /// See [`Run::inner_link`](crate::runs::Run::inner_link) for why: one
20    /// clickable region can both NAME an object and SEND a command --
21    /// clicking the katar in `ready list` sends `store WEAPON clear` -- and
22    /// without this field only the command survived.
23    pub inner_link: Option<Link>,
24    /// Whether this run ended a **physical wire line**.
25    ///
26    /// # Why a frame has to say this at all
27    ///
28    /// `Parser::push_bytes` splits on newlines and the newline never reaches a
29    /// frame, so a consumer could not tell a *markup* boundary from a *line*
30    /// boundary -- and the parser emits one run per markup boundary. A terminal
31    /// printer that wrote a line per frame produced
32    ///
33    /// ```text
34    /// [inv]   a
35    /// [inv] pebbled grey leather doublet
36    /// ```
37    ///
38    /// because `<a exist=...>` sits between those two pieces; and one that
39    /// accumulated until a newline appeared joined every line in the room
40    /// together, because no newline ever arrives.
41    ///
42    /// A parser fed one line at a time leaves its caller knowing every line
43    /// boundary. This one splits *inside* `push_bytes` -- which is what lets
44    /// it handle a chunk that ends mid-line -- and this field is how the
45    /// boundary reaches the consumer.
46    ///
47    /// **True on the last run of a line only.** A line yielding three runs
48    /// carries `false, false, true`.
49    pub ends_line: bool,
50}
51
52impl TextFrame {
53    /// This run as a [`Run`](crate::runs::Run), for assembling into a line.
54    ///
55    /// A `TextFrame` is a `Run` plus the two facts a frame carries and a run does
56    /// not: which stream it went to, and whether it ended a line. Both are
57    /// consumed by whoever is doing the assembling, so what is left is exactly a
58    /// `Run` -- and this lives here, beside both types, rather than being an
59    /// open-coded struct literal in every consumer that buffers text.
60    ///
61    /// Clones rather than consuming: `apply` takes `&Frame`, because a frame is
62    /// broadcast to every subscriber as well as folded into state.
63    #[must_use]
64    pub fn as_run(&self) -> crate::runs::Run {
65        crate::runs::Run {
66            text: self.content.clone(),
67            style: self.style.clone(),
68            link: self.link.clone(),
69            inner_link: self.inner_link.clone(),
70        }
71    }
72
73    /// The game object this text refers to, whether or not it is nested.
74    #[must_use]
75    pub fn object(&self) -> Option<&Link> {
76        for link in [self.inner_link.as_ref(), self.link.as_ref()]
77            .into_iter()
78            .flatten()
79        {
80            if matches!(link.kind, LinkKind::Exist { .. }) {
81                return Some(link);
82            }
83        }
84        None
85    }
86}

104/// What a link does when clicked.
105#[derive(Clone, Debug, PartialEq, Eq)]
106pub enum LinkKind {
107    /// `<a exist= noun=>` -- a game object with an id.
108    Exist {
109        /// The object's wire id.
110        id: String,
111        /// The noun a command would target it by.
112        noun: String,
113    },
114    /// `<d cmd=>` -- send this command.
115    Direct {
116        /// The command to send.
117        cmd: String,
118    },
119    /// A bare `<d>` with no `cmd=`: **the link text is the command.**
120    ///
121    /// This is the dominant form on the wire, not an edge case. Measured over
122    /// 40 stratified corpus files: 27,527 bare `<d>` against 4,890 `<d cmd=>`,
123    /// so 85% of direct links carry their command as body text. Every room
124    /// exit is one -- `<d>east</d>`, `<d>out</d>`, `<d>northwest</d>`.
125    ///
126    /// It is a **separate variant rather than a `Direct` with the text copied
127    /// into `cmd`** so the distinction survives to the consumer: copying would
128    /// erase which of the two the wire sent. [`Link::command`] is the single
129    /// place that resolves both, so a consumer asks one question and cannot
130    /// accidentally read an empty `cmd` as a real command -- the failure this
131    /// variant exists to make unrepresentable.
132    DirectText,
133    /// A URL. Whether it is safe to open is the UI layer's policy.
134    Url {
135        /// The target, exactly as the wire spelled it.
136        href: String,
137    },
138    /// An `<a>` carrying none of `href`, `exist` or `cmd`: **not actionable.**
139    ///
140    /// The protocol uses `<a char= game=>` for player references, and any
141    /// attribute Simutronics adds later lands here too. Such a tag marks its
142    /// text as *significant* without saying what a click would do.
143    ///
144    /// It is a variant rather than `Option::None` because the anchor still
145    /// exists on the wire and the consumer should know it was there. It is
146    /// not [`LinkKind::DirectText`] -- "send the link text as a command" --
147    /// because then `<a char='Someone'>Someone</a>` would be a link that sends
148    /// a player's NAME to the game. That is invention rather than omission,
149    /// which is the failure a consumer is least able to detect, so
150    /// [`Link::command`] returns `None` for it.
151    NotActionable,
152}

154/// A clickable region of text.
155#[derive(Clone, Debug, PartialEq, Eq)]
156pub struct Link {
157    /// What clicking it does.
158    pub kind: LinkKind,
159    /// Display text inside the link.
160    pub text: String,
161    /// `coord=` for movement links, e.g. `"2524,1864"`.
162    pub coord: Option<String>,
163}
164
165impl Link {
166    /// The command this link sends, if it sends one.
167    ///
168    /// The one place the two spellings of a direct link are resolved:
169    /// [`LinkKind::Direct`] carries its command in `cmd=`, and
170    /// [`LinkKind::DirectText`] carries it as the link's own display text.
171    /// A consumer that wants "what would clicking this send" calls this and
172    /// does not need to know which spelling arrived.
173    ///
174    /// `None` for [`LinkKind::Exist`] and [`LinkKind::Url`]: an object link is
175    /// targeted by its noun and a URL is not a game command at all, so
176    /// neither has a command to send.
177    #[must_use]
178    pub fn command(&self) -> Option<&str> {
179        match &self.kind {
180            LinkKind::Direct { cmd } => Some(cmd),
181            LinkKind::DirectText => Some(&self.text),
182            LinkKind::Exist { .. } | LinkKind::Url { .. } | LinkKind::NotActionable => None,
183        }
184    }
185}

187/// `<progressBar>` -- a vitals bar, an experience bar, or a labelled gauge.
188#[derive(Clone, Debug, PartialEq, Eq)]
189pub struct ProgressBar {
190    /// `health`, `mana`, `stamina`, `spirit`, `mindState`, ...
191    pub id: String,
192    /// The dialog it arrived in, e.g. `minivitals`; `None` for a bare bar.
193    pub dialog: Option<String>,
194    /// `value=` verbatim. **This is a percentage, not the current value.**
195    pub percent: u32,
196    /// `text=` verbatim, e.g. `"health 213/223"` or `"numbed"`.
197    pub text: String,
198    /// `current/max` parsed out of `text`, when it genuinely says so.
199    ///
200    /// `None` for a label-only bar such as `text='numbed'`. Answering
201    /// `(percentage, 100)` there would fabricate a maximum the wire never
202    /// sent; `Option` is the honest shape.
203    pub amount: Option<Amount>,
204    /// `time=` as **whole seconds remaining**, for the bars that carry it.
205    ///
206    /// Buffs, debuffs and cooldowns arrive as progress bars with a countdown:
207    /// `<progressBar id='110572' value='100' text="Multi-Strike"
208    /// time='00:00:37'/>`. Without this a consumer could not tell when a
209    /// cooldown expires.
210    ///
211    /// Parsed to seconds rather than kept as the wire's `"00:00:37"` string,
212    /// because parsing belongs in this crate and a consumer wants to compare
213    /// durations, not strings. The wire form is uniform: every value seen in
214    /// recorded sessions matches `HH:MM:SS`. `None` when the attribute is
215    /// absent or does not have that shape.
216    pub time_remaining_secs: Option<u32>,
217    /// Every attribute, in wire order -- the modelled ones included.
218    ///
219    /// The fields above are the ones most consumers read; the wire sends
220    /// more, and none of it is dropped. `vitals_secondary.xml` carries
221    /// `top`/`left`/`width`/`height`/`align` on every bar and `tooltip` on
222    /// the stance bar -- the whole layout a frontend needs to draw the
223    /// dialog. The same bag `DialogOpen` and `StreamWindow` carry.
224    pub attrs: super::Attrs,
225}

240/// One `<i>` row from an `<inventoryManager>` snapshot: a single item.
241///
242/// The rows are assembled into
243/// [`Frame::InventoryManager`](crate::Frame::InventoryManager) with their
244/// envelope; apart, the inventory tree the response exists to deliver would
245/// never be assembled.
246///
247/// # `<i>` is an item, not italics
248///
249/// On HTML instinct `<i>` would be styling beside `<b>`, and every row here
250/// would degrade to `Frame::Structural` with its attributes trapped in a raw
251/// string. In recorded sessions every `<i>` is inside an
252/// `<inventoryManager>`, and none is italics (see `parser::markup::is_markup`).
253///
254/// # What the wire states
255///
256/// In recorded sessions (~149 items a snapshot):
257///
258/// ```sh
259/// grep -ohE '<inventoryManager [^>]*>.*' *.xml ///   | grep -oE '<i [^>]*>' | grep -oE '[a-z_]+=' | sort | uniq -c
260/// #  5364 id=   5364 loc=   5364 name=   5364 weight=
261/// #  1044 long=  468 in_max=   72 on_max=    36 flags=
262/// ```
263///
264/// So `id`, `loc`, `name` and `weight` are on every row and the rest are not.
265/// `encum`, `in_encum`, `in_selector`, `locker` and `familyvault` have not
266/// been seen but are read too, because the same feed can carry them and an
267/// attribute we do not read is one we silently drop.
268#[derive(Clone, Debug, Default, PartialEq, Eq)]
269pub struct InventoryItem {
270    /// `id=`: the item's exist id.
271    pub id: String,
272    /// `loc=`, split: where the item sits relative to [`Self::parent`].
273    ///
274    /// `worn,player` -> `("worn", "player")`; `in,309585704` -> `("in", that
275    /// container)`; the bare `room` -> `("room", "room")`.
276    ///
277    /// In recorded sessions about four in five are `in,*`; the rest are
278    /// `worn,*`.
279    pub relation: String,
280    /// What [`Self::relation`] is relative to: `player`, `room`, or a
281    /// container's exist id.
282    pub parent: String,
283    /// `name=`, rejoined for display: `"a scorched glowbark long bow"`.
284    pub name: String,
285    /// `name=`'s first comma field. May be empty.
286    pub article: String,
287    /// `name=`'s second comma field. May be empty.
288    pub adjective: String,
289    /// `name=`'s third comma field, or the whole value when it does not split
290    /// into three -- losing the item is worse than an odd noun.
291    pub noun: String,
292    /// `long=` with its `$_` emphasis markers stripped, when stated.
293    pub long: Option<String>,
294    /// `weight=` in pounds. `None` when unstated.
295    ///
296    /// **Signed, and `-1` is a sentinel** meaning the item cannot be picked
297    /// up -- room furniture and fixtures. See [`Self::can_pick_up`].
298    pub weight: Option<i32>,
299    /// `encum=`: an encumbrance override, `-1` for a fixed item.
300    pub encum: Option<i32>,
301    /// `in_max=`: packed contents capacity. Decode with [`Self::in_capacity`].
302    pub in_max: Option<u32>,
303    /// `on_max=`: packed surface capacity, same encoding.
304    pub on_max: Option<u32>,
305    /// `in_encum=`: pounds currently inside, when the server reports it.
306    pub in_encum: Option<u32>,
307    /// `in_selector=`: a noun phrase to address the container by instead of
308    /// its id (lockers and similar).
309    pub in_selector: Option<String>,
310    /// `locker="1"`.
311    pub locker: bool,
312    /// `familyvault="1"`.
313    pub familyvault: bool,
314    /// `flags=`, comma-split. Observed: `closed` (36 rows, all on a purse).
315    pub flags: Vec<String>,
316    /// Every attribute the wire sent that no field above reads, as `(name,
317    /// value)` in wire order: one the wire grows is kept, not dropped.
318    pub other: Vec<(String, String)>,
319}

321/// A container's capacity, decoded from a packed `in_max` / `on_max`.
322#[derive(Clone, Copy, Debug, PartialEq, Eq)]
323pub struct Capacity {
324    /// Weight capacity in pounds.
325    pub pounds: u32,
326    /// Maximum item count; `None` means unlimited.
327    pub max_items: Option<u32>,
328}
329
330impl InventoryItem {
331    /// Decode a packed capacity: `v / 10` pounds, `v % 10` item count with
332    /// 0 meaning unlimited.
333    const fn decode_capacity(packed: u32) -> Capacity {
334        Capacity {
335            pounds: packed / 10,
336            max_items: match packed % 10 {
337                0 => None,
338                n => Some(n),
339            },
340        }
341    }
342
343    /// Contents capacity, when this is a container (`in_max` nonzero).
344    #[must_use]
345    pub fn in_capacity(&self) -> Option<Capacity> {
346        self.in_max.filter(|v| *v > 0).map(Self::decode_capacity)
347    }
348
349    /// Surface capacity, when things rest on this (`on_max` nonzero).
350    #[must_use]
351    pub fn on_capacity(&self) -> Option<Capacity> {
352        self.on_max.filter(|v| *v > 0).map(Self::decode_capacity)
353    }
354
355    /// Whether the item holds things in either orientation.
356    #[must_use]
357    pub fn is_container(&self) -> bool {
358        self.in_capacity().is_some() || self.on_capacity().is_some()
359    }
360
361    /// Whether the item can be picked up.
362    ///
363    /// `encum == -1`, or an unstated `encum` with `weight == -1`, marks a
364    /// fixture. An item that states neither is assumed portable, which is the
365    /// common case.
366    #[must_use]
367    pub fn can_pick_up(&self) -> bool {
368        match self.encum {
369            Some(e) => e != -1,
370            None => self.weight != Some(-1),
371        }
372    }
373
374    /// Whether the wire flagged the item closed.
375    #[must_use]
376    pub fn is_closed(&self) -> bool {
377        self.flags.iter().any(|f| f == "closed")
378    }
379
380    /// Whether the wire flagged the item locked.
381    #[must_use]
382    pub fn is_locked(&self) -> bool {
383        self.flags.iter().any(|f| f == "locked")
384    }
385}

467/// `<roommeta>`: the room's environment, as the game's own codes.
468///
469/// Eight attributes, and the wire sends **all eight every time**, as
470/// recorded sessions show:
471///
472/// ```sh
473/// grep -oh '<roommeta [^>]*>' *.xml | grep -oE '[a-z]+=' | sort | uniq -c
474/// ```
475///
476/// # The codes are NOT decoded here
477///
478/// `terrain="13"` and `climate="2"` are the game's own integers, and nothing
479/// documents what they mean. Inventing a mapping would be fabricating a fact.
480///
481/// The protocol has been documented as sending `terrain='forest'`, and
482/// today's wire sends `terrain="1"`, so the attribute has changed shape at
483/// least once. That is the other reason to keep the code: it is what the
484/// server said.
485///
486/// `Option` on every field: a tag that omits one has not said, and an
487/// unknown is not a fabricated zero.
488///
489/// `attrs` is the whole tag as the wire had it, every attribute in order
490/// with its value as a string: an attribute the typed fields do not name,
491/// or a value that is not a number, is kept there and nowhere else.
492#[derive(Clone, Debug, Default, PartialEq, Eq)]
493pub struct RoomMeta {
494    /// Every attribute, in wire order, raw.
495    pub attrs: super::Attrs,
496    /// `weather=`. Observed: 0, 2.
497    pub weather: Option<u32>,
498    /// `bonfire=`. Observed: 0, 1 -- a flag in practice.
499    pub bonfire: Option<u32>,
500    /// `inside=`. Observed: 0, 1.
501    pub inside: Option<u32>,
502    /// `water=`. Observed: 0.
503    pub water: Option<u32>,
504    /// `sanctuary=`. Observed: 0, 1.
505    ///
506    /// The one a script asks about first: no combat in a sanctuary.
507    pub sanctuary: Option<u32>,
508    /// `realm=`. Observed: 15, 19.
509    pub realm: Option<u32>,
510    /// `climate=`. Observed: 1..=10.
511    pub climate: Option<u32>,
512    /// `terrain=`. Observed: 0..=14.
513    pub terrain: Option<u32>,
514    /// `locker=`. Observed once, `1`, in a room with a locker.
515    pub locker: Option<u32>,
516}

518/// What an `<objectives>` update does to the list.
519#[derive(Clone, Copy, Debug, PartialEq, Eq)]
520pub enum ObjectivesAction {
521    /// `full-refresh`: this IS the list now.
522    FullRefresh,
523    /// `patch-objective`: add or update these.
524    Patch,
525    /// `delete-objective`: remove these.
526    Delete,
527    /// `delete-group`: remove the rows its `<group>`s name -- a finished
528    /// bounty, whose group's id is its row's.
529    DeleteGroup,
530    /// An action this parser has not seen. Kept rather than dropped, because
531    /// a new verb is a protocol change and silently ignoring it would apply
532    /// the wrong operation to the list.
533    Other(&'static str),
534}
535
536impl ObjectivesAction {
537    /// The wire's word.
538    #[must_use]
539    pub fn parse(text: &str) -> Self {
540        match text {
541            "full-refresh" => Self::FullRefresh,
542            "patch-objective" => Self::Patch,
543            "delete-objective" => Self::Delete,
544            "delete-group" => Self::DeleteGroup,
545            // A leak-free `Other`: the three above are every action seen in
546            // recorded sessions, so an unknown one is a change worth seeing
547            // rather than a string worth keeping.
548            _ => Self::Other("unrecognised"),
549        }
550    }
551}

569    /// `location=`, e.g. `"The Rift"`.
570    pub location: Option<String>,
571    /// `cadence=`. Observed: `weekly`, `monthly`.
572    pub cadence: Option<String>,

631/// `<cmdlist>`: the server teaching the client new context-menu commands.
632///
633/// # This is how the dictionary stays current
634///
635/// A `<menu>` carries coordinates and no labels, so a client needs a
636/// dictionary to render one. That dictionary is NOT static data a client is
637/// expected to find: the server pushes changes to it and stamps a version.
638///
639/// As seen live, in a Wrayth-banner session:
640///
641/// ```text
642/// <cmdlist><cli coord="2524,12785" menu="sense @" command="sense #"
643///               menu_cat="5_roleplay"/>
644///          <cli coord="2524,12784" menu="whisper @ about %"
645///               command="whisper # about %" menu_cat="9_questions"/>
646/// </cmdlist><cmdtimestamp data='1788300900.1.1.1'/>
647/// ```
648///
649/// That timestamp is **exactly** the one in the Wrayth client's own
650/// `cmdlist1.xml` header, which is what identifies the file as a cache of
651/// these pushes rather than shipped data.
652///
653/// # A delta, not a full dump
654///
655/// The dictionary is over a thousand rows and the observed push carried
656/// **2**: pushing every command each time would be a lot to send. Only one
657/// push has been captured, so the merge is **additive**, the safe direction
658/// to be wrong in either way.
659///
660/// # It arrives mid-session
661///
662/// Well after `<endSetup/>`, not in the login burst.
663#[derive(Clone, Debug, Default, PartialEq, Eq)]
664pub struct CmdListUpdate {
665    /// The rows the server sent, in wire order.
666    pub entries: Vec<CmdListEntry>,
667}

669/// A context menu the game returned for one object.
670///
671/// # What a menu is, and why it carries no labels
672///
673/// Right-click menus are **coordinate lookups**. The client asks with
674/// `_menu #<exist id>`; the game answers with this, where every item is a
675/// key into a command dictionary (`cmdlist1.xml`, 588 entries) and nothing
676/// else. The dictionary maps each coordinate to a label and a command
677/// template, in which `@` substitutes the object's noun and `#` its exist
678/// id.
679///
680/// **The dictionary does not arrive at login.** Recorded logins carry
681/// menus and no `<cmdlist>`, though the protocol has been documented as
682/// sending it then -- so it is data a client ships, kept current by the
683/// occasional push [`CmdListUpdate`] carries, and only the response below
684/// is live.
685#[derive(Clone, Debug, Default, PartialEq, Eq)]
686pub struct Menu {
687    /// `id=`: which request this answers.
688    pub id: String,
689    /// `path=`, e.g. `" in #103330"` -- where the object is.
690    pub path: Option<String>,
691    /// `cat_list=`: the order to present categories in, e.g. `"1 2 3 6"`.
692    pub categories: Vec<String>,
693    /// The items, in wire order.
694    pub items: Vec<MenuItem>,
695}

697/// One `<mi>`: a key into the command dictionary.
698#[derive(Clone, Debug, Default, PartialEq, Eq)]
699pub struct MenuItem {
700    /// `coord=`, e.g. `"2524,1703"`. The dictionary key.
701    pub coord: Option<String>,
702    /// `noun=`, which **disambiguates a repeated coordinate**.
703    ///
704    /// Not decoration: one real menu carries nine items all on coordinate
705    /// `2524,1906`, separated only by this -- `amplify`, `codex`, `dust`,
706    /// `instill`, `reshape`, `shatter`, `unbind`, `unlock`, `writing`.
707    /// Dropping it collapses nine commands into one.
708    pub noun: Option<String>,
709    /// `menu_cat=`, which **overrides the dictionary's own category**.
710    ///
711    /// The dictionary gives each coordinate a category, and the wire may
712    /// disagree: `2524,1639` is `5_roleplay` in `cmdlist1.xml` and arrives
713    /// as `5_Survivalist's_Kit`. The game is grouping that item under the
714    /// container it came from, so the wire's word wins.
715    ///
716    /// In recorded sessions `coord` is on nearly every item, `noun` and
717    /// `menu_cat` on a few, and nothing else:
718    ///
719    /// ```sh
720    /// grep -o '<mi [^>]*>' *.xml | grep -oE '[a-z_]+=' | sort | uniq -c
721    /// ```
722    pub menu_cat: Option<String>,
723    /// Every attribute the wire sent that no field above reads, as `(name,
724    /// value)` in wire order: one the wire grows is kept, not dropped.
725    pub other: Vec<(String, String)>,
726}

1//! What must be known before a trip is priced: **a cost never acts**.
2//! A cost only reads a fact, and never sends a command while the pathfinder
3//! runs, so whatever a fact needs asked is asked once, before the first plan
4//! ([`super::drive`]'s pre-flight). **Pure**: the map and the game's
5//! answers in, facts out.
6
7use cena_map::{Map, RoomId};
8
9/// The silver a walk asks for: each room's `silver-cost:<to>:<silver>` tag
10/// for the room that follows it. `path` begins where the walker stands. A
11/// cost that does not parse costs nothing here rather than a guess
12/// (MEASURED: all 88 of the map's parse).
13pub(super) fn silver_for(map: &Map, path: &[RoomId]) -> u64 {
14    path.windows(2)
15        .filter_map(|pair| {
16            let (from, to) = (map.room(*pair.first()?)?, *pair.get(1)?);
17            from.tags.iter().find_map(|tag| {
18                let (room, silver) = tag.strip_prefix("silver-cost:")?.split_once(':')?;
19                (room.parse() == Ok(to.0)).then_some(())?;
20                silver.parse::<u64>().ok()
21            })
22        })
23        .sum()
24}
25
26/// What to say to a banker for this much: Pinefar's has no `withdraw`, and
27/// hands out no less than twenty.
28pub(super) fn withdraw_command(room_title: Option<&str>, silver: u64) -> String {
29    if room_title.is_some_and(|title| title.contains("Pinefar, Depository")) {
30        format!("ask banker for {} silvers", silver.max(20))
31    } else {
32        format!("withdraw {silver} silvers")
33    }
34}

36#[cfg(test)]
37mod tests {
38    use cena_map::Room;
39
40    use super::*;
41
42    #[test]
43    fn a_walk_costs_what_its_own_exits_ask_and_no_others() {
44        let rooms: Vec<Room> = serde_json::from_str(
45            r#"[{"id":1,"tags":["bank","silver-cost:2:50","silver-cost:9:7000"]},
46                {"id":2,"tags":["silver-cost:3:25"]},{"id":3},{"id":9}]"#,
47        )
48        .unwrap();
49        let map = Map::from_rooms(rooms).unwrap();
50        let path = [RoomId(1), RoomId(2), RoomId(3)];
51        assert_eq!(silver_for(&map, &path), 75);
52        assert_eq!(silver_for(&map, &path[1..]), 25);
53        assert_eq!(silver_for(&map, &[RoomId(3)]), 0);
54    }
55
56    #[test]
57    fn pinefars_banker_is_asked_and_for_no_less_than_twenty() {
58        assert_eq!(
59            withdraw_command(Some("[Bank, Teller]"), 5),
60            "withdraw 5 silvers"
61        );
62        assert_eq!(
63            withdraw_command(Some("[Pinefar, Depository]"), 5),
64            "ask banker for 20 silvers"
65        );
66        assert_eq!(withdraw_command(None, 300), "withdraw 300 silvers");
67    }

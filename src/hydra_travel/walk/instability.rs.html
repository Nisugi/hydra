1//! The way into the Elemental Confluence: the map has no exit into the
2//! plane, only the instabilities that open on it, which wander.
3//! **Pure**: the game's answers and the map in, rooms and facts out; the
4//! searching is the driver's (`drive::confluence`).
5//!
6//! In order: the walker's element (`attune`); an instability of that
7//! element seen in the last six hours within ten seconds' walk; else `attune sense`, which
8//! says where the fluctuations are -- [`Sensed`] -- and so which rooms may
9//! hold one ([`narrow`]); then each of those, nearest first, looked at until
10//! one of the walker's element is found. Pushed with a soulstone it opens
11//! on the plane, and **leaving the plane puts the walker back there**: so
12//! the way out to each town costs the walk from that instability
13//! ([`instability_table`]), which is the map's `instability` table.
14

33/// An instability seen: where, of what element, and when (seconds since
34/// 1970).
35#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
36pub struct Seen {
37    /// The map's room.
38    pub room: u32,
39    /// `air`, `lightning`, `fire`, `earth` or `water`.
40    pub element: String,
41    /// When it was first seen.
42    pub seen: u64,
43}

45/// What `attune` says of the walker's element: `Some(Ok(element))`, or
46/// `Some(Err(()))` when the walker is attuned to none.
47#[must_use]
48pub fn element_in(lines: &[String]) -> Option<Result<String, ()>> {
49    lines.iter().find_map(|line| {
50        if let Some(rest) = line
51            .trim()
52            .strip_prefix("You are attuned to the Element of ")
53        {
54            let element = rest.trim_end_matches('.').trim().to_lowercase();
55            return (!element.is_empty()).then_some(Ok(element));
56        }
57        line.contains("ATTUNE SET").then_some(Err(()))
58    })
59}

61/// The element an instability gives off, from `look instability`.
62#[must_use]
63pub fn element_given_off(lines: &[String]) -> Option<&'static str> {
64    const GIVES: [(&str, &str); 5] = [
65        ("gust of wind", "air"),
66        ("burst of sparks", "lightning"),
67        ("waft of heat", "fire"),
68        ("puff of rock dust", "earth"),
69        ("puff of mist", "water"),
70    ];
71    lines
72        .iter()
73        .filter(|line| line.contains("The air is oddly warped and distorted here"))
74        .find_map(|line| {
75            GIVES
76                .iter()
77                .find(|(given, _)| line.contains(given))
78                .map(|(_, element)| *element)
79        })
80}

82/// What one `attune sense` said.
83#[derive(Debug, Clone, PartialEq, Eq)]
84pub enum Sensed {
85    /// `You sense nothing unusual.`
86    Nothing,
87    /// A fluctuation somewhere else: the place's name.
88    From(String),
89    /// Pulled towards a strong one in this place: the three lines after,
90    /// the room's title, description and exits.
91    Pulled {
92        /// The room's title, as the first line after the pull.
93        title: String,
94        /// The room's description, as the second.
95        description: String,
96        /// The room's exits, as the third: `Obvious paths: ...`.
97        paths: String,
98    },
99    /// The walker stands in it.
100    Here,
101}
102
103/// Every sense in the answer to `attune sense`, in order. A pull takes the
104/// three lines after it as the room it is towards.
105#[must_use]
106pub fn senses(lines: &[String]) -> Vec<Sensed> {
107    let mut said = Vec::new();
108    let mut at = 0;
109    while let Some(line) = lines.get(at) {
110        let line = line.trim();
111        if line.starts_with("You sense nothing unusual.") {
112            said.push(Sensed::Nothing);
113        } else if let Some(rest) =
114            line.strip_prefix("You sense an unusual fluctuation emanating from ")
115        {
116            said.push(Sensed::From(rest.trim_end_matches('.').to_owned()));
117        } else if line.starts_with("You feel your senses being pulled towards a strong fluctuation")
118        {
119            let next = |n: usize| lines.get(at + n).map(|line| line.trim().to_owned());
120            if let (Some(title), Some(description), Some(paths)) = (next(1), next(2), next(3)) {
121                said.push(Sensed::Pulled {
122                    title,
123                    description,
124                    paths,
125                });
126                at += 3;
127            }
128        } else if line.starts_with("You feel a strong sense of instability surround you!") {
129            said.push(Sensed::Here);
130        }
131        at += 1;
132    }
133    said
134}

136/// Which rooms may hold an instability, and which do not, from what was
137/// sensed standing in `here`: the rooms that may (first) and those that do
138/// not (second). `None` for `Nothing`: there is no instability to be found.
139#[must_use]
140pub fn narrow(map: &Map, here: RoomId, sensed: &[Sensed]) -> Option<(Vec<RoomId>, Vec<RoomId>)> {
141    let in_place = |location: &str| -> Vec<RoomId> {
142        map.rooms()
143            .iter()
144            .filter(|room| room.location.as_deref() == Some(location))
145            .map(|room| room.id)
146            .collect()
147    };
148    let here_at = map.room(here).and_then(|room| room.location.clone());
149    let (mut good, mut bad) = (Vec::new(), Vec::new());
150    let add = |list: &mut Vec<RoomId>, room: RoomId| {
151        if !list.contains(&room) {
152            list.push(room);
153        }
154    };
155    for sense in sensed {
156        match sense {
157            Sensed::Nothing => return None,
158            Sensed::From(location) => {
159                for room in in_place(location) {
160                    if !good.contains(&room) {
161                        add(&mut bad, room);
162                    }
163                }
164            }
165            Sensed::Pulled {
166                title,
167                description,
168                paths,
169            } => {
170                let place = here_at.as_deref().unwrap_or("");
171                let fits = |room: &Room| {
172                    room.location.as_deref() == Some(place)
173                        && room.title.iter().any(|is| is.contains(title.as_str()))
174                        && room.paths.iter().any(|is| is.contains(paths.as_str()))
175                };
176                let mut found: Vec<RoomId> = map
177                    .rooms()
178                    .iter()
179                    .filter(|room| {
180                        fits(room)
181                            && room
182                                .description
183                                .iter()
184                                .any(|is| is.contains(description.as_str()))
185                    })
186                    .map(|room| room.id)
187                    .collect();
188                if found.is_empty() {
189                    // None for the whole description: try any one sentence of it.
190                    let pieces: Vec<&str> = description
191                        .split('.')
192                        .map(str::trim)
193                        .filter(|piece| !piece.is_empty())
194                        .collect();
195                    found = map
196                        .rooms()
197                        .iter()
198                        .filter(|room| {
199                            fits(room)
200                                && room
201                                    .description
202                                    .iter()
203                                    .any(|is| pieces.iter().any(|piece| is.contains(piece)))
204                        })
205                        .map(|room| room.id)
206                        .collect();
207                }
208                for room in found {
209                    add(&mut good, room);
210                }
211                for room in in_place(place) {
212                    if !good.contains(&room) {
213                        add(&mut bad, room);
214                    }
215                }
216            }
217            Sensed::Here => {
218                add(&mut good, here);
219                for room in here_at.as_deref().map(in_place).unwrap_or_default() {
220                    if room != here {
221                        add(&mut bad, room);
222                    }
223                }
224            }
225        }
226    }
227    bad.retain(|room| !good.contains(room));
228    Some((good, bad))
229}

231/// The map's `instability` table, for a walker that entered the plane by
232/// the instability in `entered`: the seconds from there to each of the
233/// [`TOWNS`] it can reach, priced by `price`. One search.
234#[must_use]
235pub fn instability_table(
236    map: &Map,
237    entered: RoomId,
238    price: impl Fn(&Room, &cena_map::Exit) -> Option<f64>,
239) -> HashMap<u32, f64> {
240    let routes = map.routes(entered, Target::Everything, price);
241    TOWNS
242        .iter()
243        .filter_map(|town| Some((*town, routes.seconds_to(RoomId(*town))?)))
244        .collect()
245}

247/// What a trip is asked for, of the plane.
248#[derive(Debug, Clone, Copy, PartialEq, Eq)]
249pub enum Into {
250    /// `confluence`: into the plane, wherever the instability opens.
251    Plane,
252    /// `confluence-hot`: into the plane, and on to its hot half.
253    Hot,
254    /// `confluence-cold`: into the plane, and on to its cold half.
255    Cold,
256    /// `instability`: to an instability of the walker's element, and no
257    /// further.
258    Instability,
259    /// A room of the plane, asked from outside it.
260    Room(RoomId),
261}
262
263/// What `what` asks of the plane, standing in `here`: one of the plane's
264/// names above, or a room of the plane from outside it. `None`: nothing of
265/// the plane.
266#[must_use]
267pub fn asked(map: &Map, here: RoomId, what: &str, goal: Option<RoomId>) -> Option<Into> {
268    let inside = |room: RoomId| {
269        map.room(room)
270            .is_some_and(|room| room.title.first().is_some_and(|title| title == PLANE))
271    };
272    match what.trim().to_lowercase().as_str() {
273        "confluence" => Some(Into::Plane),
274        "confluence-hot" => Some(Into::Hot),
275        "confluence-cold" => Some(Into::Cold),
276        "instability" => Some(Into::Instability),
277        _ => goal
278            .filter(|goal| inside(*goal) && !inside(here))
279            .map(Into::Room),
280    }
281}

283/// The tags of the rooms each half of the plane is reached by: its huge
284/// elementals.
285#[must_use]
286pub fn half_tag(hot: bool, tag: &str) -> bool {
287    let wanted: &[&str] = if hot {
288        &["fire", "lava", "steam", "lightning"]
289    } else {
290        &["air", "earth", "ice", "water"]
291    };
292    wanted
293        .iter()
294        .any(|element| tag == format!("huge {element} elemental"))
295}

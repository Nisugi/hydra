1//! `Puzzle::FamiliarDoors`: the stone doors a familiar opens, room 6486.
2//!
3//! `go stone doors`, and if that lands, done. Otherwise, for a walker who
4//! knows Call Familiar (920):
5//!
6//! 1. A familiar already out is told to `return`, and waited on for up to
7//!    fifteen seconds to show among the room's creatures; else 920 is cast.
8//! 2. `tell familiar to watch`. If it was not watching already, it is told to
9//!    stop at the end.
10//! 3. Four laps: in (`doors,w,s,arch,ne,e,privy,hole,w` the first time,
11//!    `nw,w,w,n` after), on through the four ring doors, `get rocks`, back
12//!    out, `drop basket`. Each `go` the familiar cannot make sense of marks
13//!    the run failed, and **the list it is in is still finished** before the
14//!    laps stop. A door it cannot
15//!    enter has it `get` that door's ring -- bronze and copper are crossed --
16//!    and go again.
17//! 4. With the `mapdb_hate_familiar` setting on, the familiar is sent home
18//!    to stay.
19//! 5. `go doors`, whatever happened. A walker without 920 does only this.
20//!
21//! # Details
22//!
23//! - An answer to `go` is looked for in what came back, and only then waited
24//!   for, up to fifteen seconds.
25//! - The familiar is looked for every half second.
26//! - A `return` that is not understood, or a door with no ring in the table,
27//!   is not waited on, and the routine goes on.
28//! - The wait for mana is bounded (`casting`).
29

37const FAMILIAR: u16 = 920;
38const IN_FIRST: &str = "doors,w,s,arch,ne,e,privy,hole,w";
39const IN_AGAIN: &str = "nw,w,w,n";
40const TO_ROCKS: &str = "w,w,sw,iron door,steel door,bronze door,copper door";
41const TO_BASKET: &str = "copper door,bronze door,steel door,iron door,ne,e,e,s,e,e,se";
42const LAPS: usize = 4;
43const ANSWERS: [&str; 4] = [
44    "You sense",
45    "just squeezed between the stone doors",
46    "Obvious exits",
47    "Obvious paths",
48];
49const ANSWER_MS: u64 = 15_000;
50const LOOK_MS: u64 = 500;
51const LOOKS: u32 = 30;

53/// The ring that opens each door.
54fn ring_for(door: &str) -> Option<&'static str> {
55    Some(match door {
56        "iron" => "iron",
57        "steel" => "steel",
58        "bronze" => "copper",
59        "copper" => "bronze",
60        _ => return None,
61    })
62}
63
64enum Op {
65    Go(&'static str),
66    Say(&'static str),
67    /// `break if fail`
68    Check,
69}
70
71/// Where one `tell familiar to go` stands.
72enum Going {
73    Sent,
74    Waited,
75    /// The ring was sent for.
76    Ring,
77    /// Gone again, whose answer is not read.
78    Again,
79}
80
81#[derive(Default)]
82enum At {
83    #[default]
84    Start,
85    Tried,
86    Mana,
87    Recalled,
88    /// Looking for the familiar by this name, this many times so far.
89    Coming(String, u32),
90    /// `watch` is next.
91    Watch,
92    /// `watch` was sent.
93    Watching,
94    Working,
95    /// What is left to ask for, the next first.
96    Ending(VecDeque<Next>),
97}
98
99#[derive(Default)]
100pub(super) struct FamiliarDoors {
101    at: At,
102    waits: u32,
103    ops: VecDeque<Op>,
104    going: Option<(&'static str, Going)>,
105    failed: bool,
106    stop_watching: bool,
107}
108
109fn laps() -> VecDeque<Op> {
110    let mut ops = VecDeque::new();
111    let walk = |ops: &mut VecDeque<Op>, list: &'static str| {
112        ops.extend(list.split(',').map(Op::Go));
113        ops.push_back(Op::Check);
114    };
115    for lap in 0..LAPS {
116        walk(&mut ops, if lap == 0 { IN_FIRST } else { IN_AGAIN });
117        walk(&mut ops, TO_ROCKS);
118        ops.push_back(Op::Say("tell familiar to get rocks"));
119        walk(&mut ops, TO_BASKET);
120        ops.push_back(Op::Say("tell familiar to drop basket"));
121    }
122    ops
123}

125/// The name in `You sense understanding from your <name>.`
126fn understood_by(seen: &Seen<'_>) -> Option<String> {
127    seen.answer.iter().find_map(|line| {
128        let text = line.text();
129        let name = text.strip_prefix("You sense understanding from your ")?;
130        Some(name[..name.find('.')?].to_owned())
131    })
132}
133
134/// The door in `You sense that your … is not able to enter the <door> door`.
135fn barred_door(seen: &Seen<'_>) -> Option<String> {
136    seen.answer.iter().find_map(|line| {
137        let text = line.text();
138        if !text.starts_with("You sense that your ") {
139            return None;
140        }
141        let door = text.split_once(" is not able to enter the ")?.1;
142        Some(door[..door.find(" door")?].to_owned())
143    })
144}

146impl FamiliarDoors {
147    fn watch(&mut self) -> Next {
148        self.at = At::Watching;
149        Next::Put("tell familiar to watch".into())
150    }
151
152    fn ending(&mut self, seen: &Seen<'_>) -> Next {
153        let mut left = VecDeque::new();
154        if self.stop_watching {
155            left.push_back(Next::Put("tell familiar to stop watching".into()));
156        }
157        let hates = seen
158            .walker
159            .settings
160            .get("mapdb_hate_familiar")
161            .is_some_and(|value| !matches!(value.trim(), "" | "false" | "nil"));
162        if hates {
163            left.push_back(Next::Pause(300));
164            left.push_back(Next::Put("tell familiar to return".into()));
165            left.push_back(Next::Put("tell familiar to stay".into()));
166        }
167        left.push_back(Next::Go("go doors".into()));
168        self.at = At::Ending(left);
169        self.next(seen)
170    }
171
172    /// The answer to a `go` is in: what it means, and what is next if
173    /// anything.
174    fn judge(&mut self, way: &'static str, seen: &Seen<'_>) -> Option<Next> {
175        let sensed = ANSWERS.iter().any(|line| seen.answered(line));
176        if !sensed || seen.answered("You sense confusion") {
177            self.failed = true;
178            return None;
179        }
180        let ring = ring_for(&barred_door(seen)?)?;
181        self.going = Some((way, Going::Ring));
182        Some(Next::Put(format!("tell familiar to get {ring} ring")))
183    }
184
185    fn work(&mut self, seen: &Seen<'_>) -> Next {
186        if let Some((way, going)) = self.going.take() {
187            let sensed = ANSWERS.iter().any(|line| seen.answered(line));
188            match going {
189                Going::Sent if !sensed => {
190                    self.going = Some((way, Going::Waited));
191                    return Next::Await(ANSWERS.map(str::to_owned).into(), ANSWER_MS);
192                }
193                Going::Sent | Going::Waited => {
194                    if let Some(next) = self.judge(way, seen) {
195                        return next;
196                    }
197                }
198                Going::Ring => {
199                    self.going = Some((way, Going::Again));
200                    return Next::Put(format!("tell familiar to go {way}"));
201                }
202                Going::Again => {}
203            }
204        }
205        while let Some(op) = self.ops.pop_front() {
206            match op {
207                Op::Go(way) => {
208                    self.going = Some((way, Going::Sent));
209                    return Next::Put(format!("tell familiar to go {way}"));
210                }
211                Op::Say(command) => return Next::Put(command.to_owned()),
212                Op::Check if self.failed => self.ops.clear(),
213                Op::Check => {}
214            }
215        }
216        self.ending(seen)
217    }
218}

220impl Solver for FamiliarDoors {
221    fn next(&mut self, seen: &Seen<'_>) -> Next {
222        match &mut self.at {
223            At::Start => {
224                self.at = At::Tried;
225                Next::Put("go stone doors".into())
226            }
227            At::Tried => {
228                if seen.here == Some(seen.goal) || seen.answered("Obvious") {
229                    return Next::Done;
230                }
231                if !casting::knows(seen, FAMILIAR) {
232                    return self.ending(seen);
233                }
234                if casting::active(seen, FAMILIAR) {
235                    self.at = At::Recalled;
236                    return Next::Put("tell familiar to return".into());
237                }
238                self.at = At::Mana;
239                self.next(seen)
240            }
241            At::Mana => {
242                let Some(name) = casting::name_of(FAMILIAR) else {
243                    return Next::Failed;
244                };
245                if !casting::affords(seen, FAMILIAR) {
246                    self.waits += 1;
247                    if self.waits > MANA_WAITS {
248                        return Next::Stop(
249                            "Waited ten minutes for the mana to call a familiar.".to_owned(),
250                        );
251                    }
252                    return Next::Pause(MANA_WAIT_MS);
253                }
254                self.at = At::Watch;
255                Next::Steps(vec![Step {
256                    action: Action::Cast(name.to_owned()),
257                    when: None,
258                }])
259            }
260            At::Recalled => match understood_by(seen) {
261                Some(name) => {
262                    self.at = At::Coming(name, 0);
263                    self.next(seen)
264                }
265                None => self.watch(),
266            },
267            At::Coming(name, looks) => {
268                let here = seen
269                    .state
270                    .room
271                    .creatures
272                    .iter()
273                    .any(|creature| creature.text.contains(name.as_str()));
274                if here || *looks >= LOOKS {
275                    return self.watch();
276                }
277                *looks += 1;
278                Next::Pause(LOOK_MS)
279            }
280            At::Watch => self.watch(),
281            At::Watching => {
282                self.stop_watching = seen.answered("You sense understanding");
283                self.ops = laps();
284                self.at = At::Working;
285                self.work(seen)
286            }
287            At::Working => self.work(seen),
288            At::Ending(left) => left.pop_front().unwrap_or(Next::Done),
289        }
290    }
291}

317    /// Through the cast and `watch`, to the first `go`. `watching`: the
318    /// familiar already was.
319    fn at_work(watching: bool) -> FamiliarDoors {
320        let mut doors = FamiliarDoors::default();
321        assert_eq!(knows().ask(&mut doors), put("go stone doors"));
322        let spell = casting::name_of(920).unwrap().to_owned();
323        assert_eq!(
324            knows().answered(&["You cannot do that."]).ask(&mut doors),
325            Next::Steps(vec![Step {
326                action: Action::Cast(spell),
327                when: None
328            }])
329        );
330        assert_eq!(knows().ask(&mut doors), put("tell familiar to watch"));
331        let answer = if watching {
332            knows().answered(&["Your cat is already watching."])
333        } else {
334            sensed()
335        };
336        assert_eq!(answer.ask(&mut doors), go("doors"));
337        doors
338    }

340    /// Answer each `go` well: the first of `ways` is already asked for.
341    fn walk(doors: &mut FamiliarDoors, ways: &str) {
342        for way in ways.split(',').skip(1) {
343            assert_eq!(sensed().ask(doors), go(way));
344        }
345    }

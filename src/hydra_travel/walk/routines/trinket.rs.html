1//! `Routine::Trinket`: the Mist Harbor trinket (31 exits).
2//!
3//! The trinket is the thing the profile names as `fwi_trinket`. If the model
4//! cannot already see it -- worn, or in a hand -- then free a hand when both
5//! are full, `get my <trinket>`, and note the container the game says it
6//! came out of; no trinket is the end of it. Write down the room being left
7//! as `fwi_return_room` -- or clear it, when leaving Mist Harbor -- then
8//! `turn #<id>`. A trinket that was taken out goes back where it came from,
9//! the hands are filled again, and the routine is done: arrival in Mist
10//! Harbor is a random room, so the trip looks at where it landed.
11//!
12//! Taking out, putting back and the hands are the trip's own steps
13//! (`Action::TakeOut`, `Action::PutBack`, `Action::EmptyHands`,
14//! `Action::FillHands`), so what is owed back is known to the trip that owes
15//! it, and is paid even if this routine is stopped half-way.
16//!
17//! # How it behaves, and why
18//!
19//! - **Which way the trinket is going is told by the goal**, not by where the
20//!   walker stands: a solver is not shown the map, so it cannot ask the
21//!   location of its own room. A goal in Mist Harbor is a town being left,
22//!   and any other goal is Mist Harbor being left. The two agree wherever no
23//!   trinket exit runs from the island to the island.
24//! - **Both hands are emptied**, not one: the trip's step for it has no
25//!   one-handed form.
26//! - **`turn my <trinket>`** when the model holds no id for it.
27
28use cena_map::{Action, Step};
29
30use super::{Next, Seen, Solver};
31
32/// The profile setting that names the trinket.
33const SETTING: &str = crate::settings::FWI_TRINKET;
34/// Where the way back from Mist Harbor leads.
35const MEMORY: &str = "fwi_return_room";
36
37pub(super) struct Trinket {
38    /// The goal is in Mist Harbor: a town is being left, and is remembered.
39    to_harbor: bool,
40    /// The trinket was taken out of a container, and goes back.
41    taken: bool,
42    /// Both hands were full, and are filled again.
43    emptied: bool,
44    at: At,
45}
46
47enum At {
48    Start,
49    HandsFreed,
50    TakenOut,
51    Noted,
52    Turned,
53    Finished,
54}
55
56impl Trinket {
57    pub fn new(to_harbor: bool) -> Self {
58        Trinket {
59            to_harbor,
60            taken: false,
61            emptied: false,
62            at: At::Start,
63        }
64    }
65
66    fn start(&mut self, seen: &Seen<'_>) -> Next {
67        let Some(name) = seen.walker.settings.get(SETTING) else {
68            return Next::Failed;
69        };
70        if at_hand(seen, name).is_some() || wearing(seen, name) {
71            return self.note(seen);
72        }
73        let hands = &seen.state;
74        if hands.left_hand.is_holding() && hands.right_hand.is_holding() {
75            self.emptied = true;
76            self.at = At::HandsFreed;
77            return steps(Action::EmptyHands);
78        }
79        self.take_out(name)
80    }
81
82    fn take_out(&mut self, name: &str) -> Next {
83        self.taken = true;
84        self.at = At::TakenOut;
85        steps(Action::TakeOut(name.to_owned()))
86    }
87
88    /// Remember the room being left as `fwi_return_room`, or forget it, before
89    /// the turn.
90    fn note(&mut self, seen: &Seen<'_>) -> Next {
91        self.at = At::Noted;
92        match seen.here {
93            Some(here) if self.to_harbor => {
94                steps(Action::Remember(MEMORY.to_owned(), here.0.to_string()))
95            }
96            _ => steps(Action::Forget(MEMORY.to_owned())),
97        }
98    }
99
100    fn turn(&mut self, seen: &Seen<'_>) -> Next {
101        self.at = At::Turned;
102        let name = seen.walker.settings.get(SETTING);
103        let id = name.and_then(|name| at_hand(seen, name).or_else(|| worn_id(seen, name)));
104        Next::Put(match (id, name) {
105            (Some(id), _) => format!("turn #{id}"),
106            (None, Some(name)) => format!("turn my {name}"),
107            (None, None) => return Next::Failed,
108        })
109    }
110
111    fn tidy(&mut self) -> Next {
112        self.at = At::Finished;
113        let mut owed = Vec::new();
114        if self.taken {
115            owed.push(step(Action::PutBack));
116        }
117        if self.emptied {
118            owed.push(step(Action::FillHands));
119        }
120        if owed.is_empty() {
121            Next::Done
122        } else {
123            Next::Steps(owed)
124        }
125    }
126}
127
128fn step(action: Action) -> Step {
129    Step { action, when: None }
130}
131
132fn steps(action: Action) -> Next {
133    Next::Steps(vec![step(action)])
134}
135
136/// Whether a thing called this is what the profile names: the whole name, its
137/// end (`a gold ring` for `gold ring`), or the noun alone.
138fn named(full: &str, noun: &str, wanted: &str) -> bool {
139    full == wanted || noun == wanted || full.ends_with(&format!(" {wanted}"))
140}
141
142/// The id of the trinket, if a hand holds it.
143fn at_hand(seen: &Seen<'_>, wanted: &str) -> Option<String> {
144    [&seen.state.right_hand, &seen.state.left_hand]
145        .into_iter()
146        .find(|hand| named(hand.name().unwrap_or(""), hand.noun().unwrap_or(""), wanted))
147        .and_then(|hand| hand.id().map(str::to_owned))
148}
149
150fn wearing(seen: &Seen<'_>, wanted: &str) -> bool {
151    let walker = seen.walker;
152    walker.worn.as_ref().is_some_and(|w| w.contains(wanted))
153        || walker
154            .worn_nouns
155            .as_ref()
156            .is_some_and(|w| w.contains(wanted))
157}
158
159/// The id of the trinket, if it is worn: from the inventory snapshot, or
160/// else the worn list the login sends.
161fn worn_id(seen: &Seen<'_>, wanted: &str) -> Option<String> {
162    seen.state
163        .inventory_snapshot
164        .on_person()
165        .find(|item| {
166            let full = format!("{} {}", item.adjective, item.noun);
167            named(full.trim(), &item.noun, wanted)
168        })
169        .map(|item| item.id.clone())
170        .or_else(|| {
171            seen.state
172                .worn
173                .items()?
174                .iter()
175                .find(|item| named(&item.text, &item.noun, wanted))
176                .map(|item| item.id.clone())
177        })
178}
179
180impl Solver for Trinket {
181    fn next(&mut self, seen: &Seen<'_>) -> Next {
182        match self.at {
183            At::Start => self.start(seen),
184            At::HandsFreed => match seen.walker.settings.get(SETTING) {
185                Some(name) => self.take_out(&name.clone()),
186                None => Next::Failed,
187            },
188            // No trinket to take out: the exit fails. The hands are the
189            // trip's to give back.
190            At::TakenOut if !seen.ok => Next::Failed,
191            At::TakenOut => self.note(seen),
192            At::Noted => self.turn(seen),
193            At::Turned => self.tidy(),
194            At::Finished => Next::Done,
195        }
196    }
197}

209    fn scene(here: u32) -> Scene {
210        let mut scene = Scene::at(here, 3668);

215    fn holding(id: &str, name: &str, noun: &str) -> Hand {
216        Hand::Holding {
217            id: Some(id.into()),
218            noun: Some(noun.into()),
219            name: name.into(),
220        }
221    }

223    /// The trinket worn, as both the walker's facts and the model have it.
224    fn wearing_it(mut scene: Scene) -> Scene {
225        scene.walker.worn = Some([NAME.to_owned()].into());
226        scene.walker.worn_nouns = Some(["locket".to_owned()].into());
227        let items = [
228            InventoryItem {
229                id: "11".into(),
230                relation: "worn".into(),
231                parent: "player".into(),
232                adjective: "dark".into(),
233                noun: "cloak".into(),
234                ..InventoryItem::default()
235            },
236            InventoryItem {
237                id: "77".into(),
238                relation: "worn".into(),
239                parent: "player".into(),
240                adjective: "silver".into(),
241                noun: "locket".into(),
242                ..InventoryItem::default()
243            },
244        ];
245        scene
246            .state
247            .inventory_snapshot
248            .apply_snapshot("7", &items, &[], None);
249        scene
250    }

252    fn remembers(room: &str) -> Next {
253        steps(Action::Remember(MEMORY.into(), room.into()))
254    }

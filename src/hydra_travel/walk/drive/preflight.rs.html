1//! Asking what a trip must know before it is priced, and fetching the
2//! silver it will be asked for. The reading is pure
3//! ([`super::super::preflight`], the day pass's readers); this sends,
4//! walks and stops.
5//!
6//! # The silver detour
7//!
8//! Add up what the route's exits charge -- and the way back, if the Travel
9//! page says so -- and compare it with `wealth`. Short, and allowed the bank
10//! (`get_silvers`): walk to the nearest bank the silver in hand can reach,
11//! price the walk again from there, withdraw the difference, and look again.
12//! Short and not allowed: say so, and go; the ferryman will say no soon
13//! enough, and the trip plans round him. **The walk halts** when the walker
14//! is too poor to reach a bank, or the bank has not enough -- both for the
15//! player to settle.
16//!
17//! **Silver nobody has stated is not no silver**: if `wealth quiet` leaves
18//! the model unable to say what is carried, this says so and goes, rather
19//! than walk a purse of platinum to the bank.

29/// The Long Snow's encampment and Cairnfang Manor's attic: the two ends of
30/// the long way to the Hinterwilds.
31const HINTERWILDS_WAYS: [u32; 2] = [29860, 22154];
32
33impl<F: FnMut(&[u32]), W: FnMut(&std::collections::BTreeMap<String, String>)> Driver<'_, F, W> {
34    /// Before the first plan. `goal` is where the trip is going.
35    pub(super) async fn preflight(
36        &mut self,
37        trip: &mut Trip,
38        map: &cena_map::Map,
39        setting: &Setting,
40    ) -> Result<(), Ended> {
41        let goal = trip.goal();
42        // Asked once, and only while nobody has said: the model reads the
43        // answer and keeps it, so a walk does not ask at every room.
44        let urchins = self
45            .state
46            .character
47            .standing
48            .urchins(self.state.game_time_now());
49        if setting.settings.on("use_urchins") && urchins.is_none() {
50            self.put(trip, "urchin status").await?;
51        }
52        self.day_passes(trip, setting).await?;
53        // Silver is priced from a room; an unplaced walker is the walk's to
54        // wait for, and the ferryman's to refuse.
55        let Some(here) = self.locate(map) else {
56            return Ok(());
57        };
58        self.by_gigas(trip, map, setting, here, goal).await?;
59        let Some(here) = self.locate(map) else {
60            return Ok(());
61        };
62        let needed = self.silver_needed(trip, map, setting, here, goal);
63        if needed == 0 {
64            return Ok(());
65        }
66        let Some(have) = self.silver(trip).await? else {
67            self.warn(&format!(
68                "this route asks for {needed} silver, and I could not tell how much you carry, \
69                 so I am going anyway."
70            ));
71            return Ok(());
72        };
73        if have >= needed {
74            return Ok(());
75        }
76        if !setting.settings.on("get_silvers") {
77            self.warn(&format!(
78                "this route asks for {needed} silver and you carry {have}. Fetch silver from \
79                 the bank is off, so I am going anyway."
80            ));
81            return Ok(());
82        }
83        self.by_way_of_the_bank(trip, map, setting, (here, goal), have)
84            .await
85    }

92    /// Which day passes the walker holds, asked once of a walker that uses
93    /// passes and names the sack they are kept in; each that still serves
94    /// becomes the flag the map asks for.
95    async fn day_passes(&mut self, trip: &mut Trip, setting: &Setting) -> Result<(), Ended> {
96        let sack = setting.settings.0.get("day_pass_sack");
97        let Some(sack) =
98            sack.filter(|sack| !sack.is_empty() && setting.settings.on("use_day_pass"))
99        else {
100            return Ok(());
101        };
102        let sack = day_pass::sack_in(&self.state, sack)
103            .map_or_else(|| format!("my {sack}"), |id| format!("#{id}"));
104        let mut inside = self.put(trip, &format!("look in {sack}")).await?;
105        let shut = inside.iter().any(|line| line.text().contains("closed"));
106        if shut {
107            self.put(trip, &format!("open {sack}")).await?;
108            inside = self.put(trip, &format!("look in {sack}")).await?;
109        }
110        let now = self.state.game_time().map(i64::from);
111        for id in day_pass::passes_in(&inside) {
112            let looked = self.put(trip, &format!("look #{id}")).await?;
113            let Some(pass) = day_pass::read_pass(&looked) else {
114                continue;
115            };
116            let flag = pass.towns.as_ref().and_then(|(one, other)| {
117                pass.serves(one, other, now)
118                    .then(|| day_pass::flag_for(one, other))?
119            });
120            if let Some(flag) = flag {
121                self.found.insert(flag, true);
122            }
123        }
124        if shut {
125            self.put(trip, &format!("close {sack}")).await?;
126        }
127        Ok(())
128    }

130    /// To or from the Hinterwilds by gigas fragments: when the route passes
131    /// the long way's ends, the Travel page says so, and `wealth gigas` counts at least the fragments kept
132    /// back, walk to the teleporter instead -- Ta'Illistim's if the route
133    /// passes Seethe Naedal, the Abbey's otherwise -- and `go sliver`; or,
134    /// leaving, to Sparkfinger's workroom and `order 3`. Which side was
135    /// entered by is remembered (`hinterwilds_location`) while the walker is
136    /// there. The trip then plans from wherever that landed.
137    async fn by_gigas(
138        &mut self,
139        trip: &mut Trip,
140        map: &cena_map::Map,
141        setting: &Setting,
142        here: RoomId,
143        goal: RoomId,
144    ) -> Result<(), Ended> {
145        const WORKROOM: i64 = 7_503_253;
146        if !setting.settings.on("use_gigas_hwtravel") {
147            return Ok(());
148        }
149        let walker = self.walker(setting);
150        let Some(path) = trip.path_from(map, &walker, here, goal) else {
151            return Ok(());
152        };
153        if !path.iter().any(|room| HINTERWILDS_WAYS.contains(&room.0)) {
154            return Ok(());
155        }
156        let wild = |room: RoomId| {
157            map.room(room).and_then(|room| room.location.as_deref()) == Some("the Hinterwilds")
158        };
159        let (leaving, arriving) = (wild(here), wild(goal));
160        if leaving == arriving {
161            return Ok(());
162        }
163        self.put(trip, "wealth gigas").await?;
164        let fragments = self
165            .state
166            .character
167            .currency
168            .gigas_artifact_fragments
169            .unwrap_or(0);
170        let least = setting
171            .settings
172            .0
173            .get("gigas_min_number")
174            .and_then(|least| least.parse().ok())
175            .unwrap_or(4);
176        if fragments < least {
177            return Ok(());
178        }
179        let elven = path.iter().any(|room| {
180            map.room(*room).is_some_and(|room| {
181                room.title
182                    .first()
183                    .is_some_and(|title| title.contains("Seethe Naedal"))
184            })
185        });
186        let (uid, side, commands): (i64, _, &[&str]) = match (arriving, elven) {
187            (true, true) => (13_205_202, Some("EN"), &["go sliver", "go sliver"]),
188            (true, false) => (4_132_054, Some("IM"), &["go sliver", "go sliver"]),
189            (false, _) => (WORKROOM, None, &["order 3", "order confirm"]),
190        };
191        let Some(teleporter) = map.ids_for_uid(Uid(uid)).first().copied() else {
192            return Ok(());
193        };
194        if !self.walk_to(trip, map, setting, teleporter).await? {
195            return Ok(());
196        }
197        for command in commands {
198            self.put(trip, command).await?;
199        }
200        // The side entered by is remembered only when the walker landed in
201        // the workroom; anywhere else it is forgotten.
202        let landed = self.locate(map);
203        let there = landed.is_some_and(|room| map.ids_for_uid(Uid(WORKROOM)).contains(&room));
204        match side.filter(|_| there) {
205            Some(side) => self
206                .memories
207                .insert("hinterwilds_location".to_owned(), side.to_owned()),
208            None => self.memories.remove("hinterwilds_location"),
209        };
210        (self.wrote)(&self.memories);
211        Ok(())
212    }

214    /// What the route there -- and back, if the Travel page asks -- charges.
215    fn silver_needed(
216        &self,
217        trip: &Trip,
218        map: &cena_map::Map,
219        setting: &Setting,
220        from: RoomId,
221        goal: RoomId,
222    ) -> u64 {
223        let walker = self.walker(setting);
224        let leg = |from, to| {
225            trip.path_from(map, &walker, from, to)
226                .map_or(0, |path| silver_for(map, &path))
227        };
228        let back = if setting.settings.on("get_return_trip_silvers") {
229            leg(goal, from)
230        } else {
231            0
232        };
233        leg(from, goal) + back
234    }

236    /// `wealth`, as the model read it. `None`: nobody has said, which is not
237    /// zero.
238    async fn silver(&mut self, trip: &mut Trip) -> Result<Option<u64>, Ended> {
239        self.put(trip, "wealth quiet").await?;
240        Ok(self.state.character.currency.silver)
241    }

243    async fn by_way_of_the_bank(
244        &mut self,
245        trip: &mut Trip,
246        map: &cena_map::Map,
247        setting: &Setting,
248        (here, goal): (RoomId, RoomId),
249        have: u64,
250    ) -> Result<(), Ended> {
251        let walker = self.walker(setting);
252        let mut banks: Vec<(f64, RoomId)> = map
253            .rooms()
254            .iter()
255            .filter(|room| room.tags.iter().any(|tag| tag == "bank"))
256            .filter_map(|room| {
257                let path = trip.path_from(map, &walker, here, room.id)?;
258                (silver_for(map, &path) <= have).then_some(())?;
259                let routes = map.routes(here, Target::Room(room.id), trip.pricing(&walker));
260                Some((routes.seconds_to(room.id)?, room.id))
261            })
262            .collect();
263        banks.sort_by(|(a, _), (b, _)| a.total_cmp(b));
264        let Some((_, bank)) = banks.first().copied() else {
265            return Err(halted("you are too poor to reach a bank."));
266        };
267        if !self.walk_to(trip, map, setting, bank).await? {
268            return Err(halted("I could not get to the bank."));
269        }
270        let needed = self.silver_needed(trip, map, setting, bank, goal);
271        // Known before the walk here, and the model keeps a stated amount;
272        // were it to forget, the amount known before withdraws the most.
273        let have = self.silver(trip).await?.unwrap_or(have);
274        if needed > have {
275            let walker = self.walker(setting);
276            let unseen = ["hidden", "invisible"]
277                .iter()
278                .any(|flag| walker.flags.get(*flag) == Some(&true));
279            if unseen {
280                self.put(trip, "unhide").await?;
281            }
282            let command = withdraw_command(self.state.room.title.as_deref(), needed - have);
283            self.put(trip, &command).await?;
284            // Only a shortfall the game stated halts the trip.
285            if self.silver(trip).await?.is_some_and(|have| have < needed) {
286                return Err(halted(
287                    "there is not enough silver in this bank for the trip.",
288                ));
289            }
290        }
291        Ok(())
292    }

1//! What a player types to travel: `.go2` and `.route2`, after Hydra's command
2//! symbol.
3//!
4//! ```text
5//! .go2 bank                 walk to the nearest bank
6//! .go2 228   .go2 u7120     ...to a room, by the map's number or the game's
7//! .go2 to Ashryn            ...to the room another character running here is in
8//! .go2 targets              the places there are to go
9//! .go2 list                 the names you have saved
10//! .go2 save den             this room is "den", for this character
11//! .go2 save den --global    ...for every character
12//! .go2 save den=228,current ...and these rooms; the nearest is meant
13//! .go2 delete den           forget it (--global: everyone's)
14//! .go2 stop                 stop walking
15//! .go2 coverage             how much of the map Hydra walks yet
16//! .go2 confluence           into the Elemental Confluence, by an instability
17//!                           of your element (-hot, -cold: on to that half;
18//!                           .go2 instability: to the instability only;
19//!                           --instability=<room>: by that one)
20//! .route2 bank              show the way, and send nothing; on the minimap too
21//! .route2 bank --brief      ...in a line: what a click on the minimap asks
22//! .route2 clear             take it off the minimap
23//! .sail kraken              make your ship ready and sail it to a port
24//! .sail route kraken        the course there on the chart, kept as you sail;
25//!                           sends nothing
26//! .sail help                the ports, and how far each is
27//! .go2 help                 these
28//! ```
29//!
30//! **A command may be shortened**: `.go` is `.go2`, `.ro` is `.route2`, and
31//! `.sa` is `.sail`
32//! -- any start of the word of two letters or more.
33//!
34//! A walk is stopped by `.go2 stop`, or by `.stop`, which stops everything.
35

61/// One thing asked of travel.
62#[derive(Debug, Clone, PartialEq, Eq)]
63pub enum Command {
64    /// The command list, [`HELP`].
65    Help,
66    /// Walk there. `to <character>` is where another character running
67    /// here stands.
68    Go(String),
69    /// Say the way there, and draw it on the minimap. Sends nothing.
70    Route {
71        /// Where, as [`Command::Go`] takes it.
72        what: String,
73        /// In a line, not the table: a click on the minimap.
74        brief: bool,
75    },
76    /// Take the way off the minimap.
77    ClearRoute,
78    /// List the places the map names, town by town.
79    Places,
80    /// List the names the player has saved.
81    List,
82    /// From now on `name` means these rooms: each a number, `u<uid>`, or
83    /// `current`. None given means `current`.
84    Save {
85        /// The name, as typed (it may hold spaces).
86        name: String,
87        /// The words after `=`, trimmed and lowercased; empty when none.
88        rooms: Vec<String>,
89        /// For every character, not just this one.
90        global: bool,
91    },
92    /// Forget a saved name.
93    Forget {
94        /// The name, as typed.
95        name: String,
96        /// Everyone's, not this character's.
97        global: bool,
98    },
99    /// Stop the walk under way.
100    Stop,
101    /// Take a way out of the room that is no compass direction: the one
102    /// named by number or by a word of it, or, with none named, the only
103    /// one; with several, list them.
104    Portal(Option<String>),
105    /// How much of the map the walker crosses yet.
106    Coverage,
107    /// Sail the character's ship to a port, or show the course there.
108    Sail {
109        /// The port, as typed; `help` for the list of them.
110        port: String,
111        /// Send nothing: keep the course on the chart while the player
112        /// sails.
113        route: bool,
114    },
115}

117/// Whether `typed` is `word`, or a start of it of two letters or more.
118fn is(typed: &str, word: &str) -> bool {
119    typed.len() >= 2 && word.starts_with(typed)
120}

122/// The travel command `line` is, **its symbol already gone**. `None`: not
123/// travel's. `Some(Err(_))`: travel's, said wrongly.
124#[must_use]
125pub fn parse(line: &str) -> Option<Result<Command, String>> {
126    let mut words = line.split_whitespace();
127    let script = words.next()?.to_lowercase();
128    let rest: Vec<&str> = words.collect();
129    let global = rest.contains(&"--global");
130    let brief = rest.contains(&"--brief");
131    let rest: Vec<&str> = rest
132        .into_iter()
133        .filter(|word| *word != "--global" && *word != "--brief")
134        .collect();
135    let said = rest.join(" ");
136    let first = rest.first().map(|word| word.to_lowercase());
137    if is(&script, "route2") {
138        return Some(match first.as_deref() {
139            Some("clear") if rest.len() == 1 => Ok(Command::ClearRoute),
140            Some(_) => Ok(Command::Route { what: said, brief }),
141            None => Err("route2 where? .route2 bank".to_owned()),
142        });
143    }
144    if is(&script, "sail") {
145        return Some(match first.as_deref() {
146            Some("route") if rest.len() > 1 => Ok(Command::Sail {
147                port: rest[1..].join(" "),
148                route: true,
149            }),
150            Some("route") => Err("the course where? .sail route kraken".to_owned()),
151            Some(_) => Ok(Command::Sail {
152                port: said,
153                route: false,
154            }),
155            None => Err("sail where? .sail kraken, or .sail help for the ports".to_owned()),
156        });
157    }
158    if is(&script, "portal") {
159        return Some(Ok(Command::Portal(
160            (!said.trim().is_empty()).then(|| said.trim().to_owned()),
161        )));
162    }
163    if !is(&script, "go2") {
164        return None;
165    }
166    let alone = rest.len() == 1;
167    Some(match first.as_deref() {
168        None => Err("go2 where? .go2 bank, or .go2 targets for a list".to_owned()),
169        Some("targets") if alone => Ok(Command::Places),
170        Some("help") if alone => Ok(Command::Help),
171        Some("list") if alone => Ok(Command::List),
172        Some("stop") if alone => Ok(Command::Stop),
173        Some("coverage") if alone => Ok(Command::Coverage),
174        Some("save") => named(&rest[1..]).and_then(|(name, rooms)| {
175            // `.go2 228` is room 228, and `u7120` a room by its number in the
176            // game: a name like that could never be gone to.
177            let digits = name.strip_prefix(['u', 'U']).unwrap_or(&name);
178            if !digits.is_empty() && digits.chars().all(|c| c.is_ascii_digit()) {
179                return Err(format!(
180                    "{name} is a room, not a name: .go2 {name} goes there. Give it a word:                      .go2 save den={name}"
181                ));
182            }
183            Ok(Command::Save {
184                name,
185                rooms,
186                global,
187            })
188        }),
189        Some("delete") => named(&rest[1..]).map(|(name, _)| Command::Forget { name, global }),
190        Some(_) => Ok(Command::Go(said)),
191    })
192}

194/// `den`, or `den=228,current`: the name, and the rooms if any were given.
195fn named(words: &[&str]) -> Result<(String, Vec<String>), String> {
196    let said = words.join(" ");
197    let (name, rooms) = said.split_once('=').unwrap_or((&said, ""));
198    let name = name.trim();
199    if name.is_empty() {
200        return Err("it needs a name: .go2 save den".to_owned());
201    }
202    let rooms = rooms
203        .split(',')
204        .map(str::trim)
205        .filter(|room| !room.is_empty())
206        .map(str::to_lowercase)
207        .collect();
208    Ok((name.to_owned(), rooms))
209}

211#[cfg(test)]

215    fn ok(line: &str) -> Command {
216        parse(line).unwrap().unwrap()
217    }
218
219    fn route(what: &str, brief: bool) -> Command {
220        Command::Route {
221            what: what.into(),
222            brief,
223        }
224    }

226    /// `.portal` alone takes the only way or lists them; a number or word
227    /// names one.
228    #[test]
229    fn portal_names_a_way_or_none() {
230        assert_eq!(ok("portal"), Command::Portal(None));
231        assert_eq!(ok("portal 2"), Command::Portal(Some("2".into())));
232        assert_eq!(
233            ok("portal iron gate"),
234            Command::Portal(Some("iron gate".into()))
235        );
236    }
237
238    #[test]
239    fn sail_names_a_port_and_route_only_shows_the_course() {
240        let sail = |port: &str, route: bool| Command::Sail {
241            port: port.into(),
242            route,
243        };
244        assert_eq!(ok("sail kraken"), sail("kraken", false));
245        assert_eq!(ok("sa Kraken's Fall"), sail("Kraken's Fall", false));
246        assert_eq!(ok("sail route kraken"), sail("kraken", true));
247        assert_eq!(ok("sail route Kraken's Fall"), sail("Kraken's Fall", true));
248        assert_eq!(ok("sail help"), sail("help", false));
249        assert!(parse("sail").unwrap().is_err());
250        assert!(parse("sail route").unwrap().is_err());
251        assert_eq!(parse("s kraken"), None);
252    }

254    /// The minimap's click asks for the way in a line; a companion is
255    /// named after `to`, and left for the binary, which knows who runs.
256    #[test]
257    fn brief_and_to_are_said_on() {
258        assert_eq!(ok("route2 228 --brief"), route("228", true));
259        assert_eq!(ok("go2 to Ashryn"), Command::Go("to Ashryn".into()));
260        assert_eq!(ok("route2 to Ashryn"), route("to Ashryn", false));
261    }

311    #[test]
312    fn a_name_is_saved_for_the_character_unless_told_global() {
313        let saved = |rooms: &[&str], global| Command::Save {
314            name: "my den".into(),
315            rooms: rooms.iter().map(|room| (*room).to_owned()).collect(),
316            global,
317        };
318        assert_eq!(ok("go2 save my den"), saved(&[], false));
319        assert_eq!(ok("go2 save my den --global"), saved(&[], true));
320        assert_eq!(ok("go2 save --global my den"), saved(&[], true));
321        assert_eq!(
322            ok("go2 save my den=228, Current,u7120"),
323            saved(&["228", "current", "u7120"], false)
324        );
325        assert_eq!(
326            ok("go2 delete my den --global"),
327            Command::Forget {
328                name: "my den".into(),
329                global: true
330            }
331        );
332    }

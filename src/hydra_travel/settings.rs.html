1//! The travel settings, kept in the character's settings file under
2//! [`SECTION`] and shown on its **Travel** page, in the settings menu like
3//! every other setting.
4//!
5//! # Every setting the map asks about, and the walker's own
6//!
7//! [`TABLE`] names each once. The map's are every name the built-in map asks
8//! (`Map::setting_names`, held to this table by a test); the walker's own are
9//! the ones Hydra's walk reads and the map never asks. The map spells two of
10//! its names two ways (`keysack`, `Mularos_Lover`): one row answers to both
11//! (`Key::also`).
12//!
13//! # Unset is not unknown
14//!
15//! A setting never set is simply not set: not `true`, and not a question
16//! without an answer. So a walker is given **every** setting
17//! ([`TravelSettings::for_walker`]), an unset one as its default -- empty for
18//! a switch that is off or words not given, `auto` for the ice -- and the
19//! map's questions about them always have an answer.
20
21use std::collections::{BTreeMap, HashMap};
22
23use serde::{Deserialize, Serialize};
24
25/// The section of the character's settings file these are kept in.
26pub const SECTION: &str = "travel";

28/// The settings as the file keeps them: each by name, as text. A switch is
29/// `true` when on and absent when off.
30#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
31#[serde(transparent)]
32pub struct TravelSettings(pub BTreeMap<String, String>);
33
34/// What sort of value a setting takes.
35#[derive(Debug, Clone, Copy, PartialEq)]
36pub enum KeyKind {
37    /// On or off.
38    Toggle,
39    /// Words, or nothing.
40    Text,
41    /// A whole number from `min` to `max`.
42    Whole {
43        /// The least.
44        min: u32,
45        /// The most.
46        max: u32,
47    },
48    /// A number of seconds from `min` to `max`.
49    Seconds {
50        /// The least.
51        min: f64,
52        /// The most.
53        max: f64,
54    },
55    /// One of these: the value, and what a player calls it.
56    Choice(&'static [(&'static str, &'static str)]),
57}
58
59/// One setting.
60#[derive(Debug, Clone, Copy, PartialEq)]
61pub struct Key {
62    /// Its name, as the map spells it.
63    pub name: &'static str,
64    /// Its other spellings in the map, answered by the same value.
65    pub also: &'static [&'static str],
66    /// What the page calls it.
67    pub label: &'static str,
68    /// What the page says of it.
69    pub help: &'static str,
70    /// What it takes.
71    pub kind: KeyKind,
72    /// Its value when unset: for a switch, the word the map asks for when
73    /// it is on (`true`, or `yes` where the map asks for that).
74    pub default: &'static str,
75}
76
77impl Key {
78    const fn toggle(name: &'static str, label: &'static str, help: &'static str) -> Key {
79        Key {
80            name,
81            also: &[],
82            label,
83            help,
84            kind: KeyKind::Toggle,
85            default: "true",
86        }
87    }
88
89    const fn text(name: &'static str, label: &'static str, help: &'static str) -> Key {
90        Key {
91            name,
92            also: &[],
93            label,
94            help,
95            kind: KeyKind::Text,
96            default: "",
97        }
98    }
99
100    const fn also(mut self, also: &'static [&'static str]) -> Key {
101        self.also = also;
102        self
103    }
104
105    const fn on_word(mut self, word: &'static str) -> Key {
106        self.default = word;
107        self
108    }
109}

301/// Whether the switch `name` is on among a walker's settings, as a routine
302/// reads them ([`TravelSettings::for_walker`]): `true`, `yes` or `on`.
303#[must_use]
304pub fn on<S: std::hash::BuildHasher>(settings: &HashMap<String, String, S>, name: &str) -> bool {
305    settings
306        .get(name)
307        .is_some_and(|is| on_word(is) == Some(true))
308}
309
310/// The setting called `name`, by any of its spellings, without regard to
311/// case.
312#[must_use]
313pub fn key(name: &str) -> Option<&'static Key> {
314    TABLE.iter().find(|key| {
315        key.name.eq_ignore_ascii_case(name)
316            || key.also.iter().any(|also| also.eq_ignore_ascii_case(name))
317    })
318}
319
320/// Whether `word` turns a switch on (`on`, `true`, `yes`) or off (`off`,
321/// `false`, `no`); anything else is neither.
322fn on_word(word: &str) -> Option<bool> {
323    match word.trim().to_ascii_lowercase().as_str() {
324        "on" | "true" | "yes" => Some(true),
325        "off" | "false" | "no" => Some(false),
326        _ => None,
327    }
328}

330impl TravelSettings {
331    /// Whether the switch `name` is on.
332    #[must_use]
333    pub fn on(&self, name: &str) -> bool {
334        self.0.get(name).is_some_and(|is| on_word(is) == Some(true))
335    }
336
337    /// The value of `key` as the page shows it: what is set, else its
338    /// default (a switch's default is off).
339    #[must_use]
340    pub fn shown(&self, key: &Key) -> String {
341        match (self.0.get(key.name), key.kind) {
342            (Some(set), _) => set.clone(),
343            (None, KeyKind::Toggle) => "false".to_owned(),
344            (None, _) => key.default.to_owned(),
345        }
346    }
347
348    /// Every setting as the map asks: each of [`TABLE`] by every spelling,
349    /// a switch that is on as the word the map asks for and one that is off
350    /// as nothing, the rest as set or by default. Unset is not unknown
351    /// (module docs).
352    #[must_use]
353    pub fn for_walker(&self) -> HashMap<String, String> {
354        let mut settings = HashMap::new();
355        for key in TABLE {
356            let value = match key.kind {
357                KeyKind::Toggle if self.on(key.name) => key.default.to_owned(),
358                KeyKind::Toggle => String::new(),
359                _ => self.shown(key),
360            };
361            for name in std::iter::once(key.name).chain(key.also.iter().copied()) {
362                settings.insert(name.to_owned(), value.clone());
363            }
364        }
365        settings
366    }
367
368    /// Set `name` to `to`, or back to its default (`None`). What was done,
369    /// in words; why not, when `to` is not a value it takes.
370    ///
371    /// # Errors
372    ///
373    /// No such setting, or a value it does not take.
374    pub fn set(&mut self, name: &str, to: Option<&str>) -> Result<String, String> {
375        let key = key(name).ok_or_else(|| format!("Travel has no setting {name}."))?;
376        let Some(to) = to.map(str::trim) else {
377            self.0.remove(key.name);
378            return Ok(format!("{} is back to {}.", key.label, self.said(key)));
379        };
380        let value = match key.kind {
381            KeyKind::Toggle => match on_word(to) {
382                Some(true) => "true".to_owned(),
383                Some(false) => {
384                    self.0.remove(key.name);
385                    return Ok(format!("{} is off.", key.label));
386                }
387                None => return Err(format!("{} is on or off.", key.label)),
388            },
389            KeyKind::Text => to.to_owned(),
390            KeyKind::Whole { min, max } => match to.parse::<u32>() {
391                Ok(number) if (min..=max).contains(&number) => number.to_string(),
392                _ => return Err(format!("{} is a whole number, {min} to {max}.", key.label)),
393            },
394            KeyKind::Seconds { min, max } => match to.parse::<f64>() {
395                Ok(seconds) if (min..=max).contains(&seconds) => seconds.to_string(),
396                _ => {
397                    return Err(format!(
398                        "{} is a number of seconds, {min} to {max}.",
399                        key.label
400                    ));
401                }
402            },
403            KeyKind::Choice(choices) => {
404                let Some((value, _)) = choices
405                    .iter()
406                    .find(|(value, _)| value.eq_ignore_ascii_case(to))
407                else {
408                    let values: Vec<&str> = choices.iter().map(|(value, _)| *value).collect();
409                    return Err(format!("{} is one of {}.", key.label, values.join(", ")));
410                };
411                (*value).to_owned()
412            }
413        };
414        if value.is_empty() {
415            self.0.remove(key.name);
416        } else {
417            self.0.insert(key.name.to_owned(), value);
418        }
419        Ok(format!("{} is {}.", key.label, self.said(key)))
420    }
421
422    /// `key`'s value now, as a sentence says it.
423    fn said(&self, key: &Key) -> String {
424        match key.kind {
425            KeyKind::Toggle if self.on(key.name) => "on".to_owned(),
426            KeyKind::Toggle => "off".to_owned(),
427            _ => match self.shown(key) {
428                shown if shown.is_empty() => "not set".to_owned(),
429                shown => shown,
430            },
431        }
432    }
433}

441    /// Every name the built-in map asks about is a setting here, so none is
442    /// left unknown to every walker; and each is named once.
443    #[test]
444    fn every_setting_the_map_asks_about_is_here() {
445        let map = cena_map::binary::decode(cena_gs_map::GS_MAP).expect("the built-in map");
446        let missing: Vec<String> = map
447            .setting_names()
448            .into_iter()
449            .filter(|name| {
450                !TABLE
451                    .iter()
452                    .any(|key| key.name == name || key.also.contains(&name.as_str()))
453            })
454            .collect();
455        assert_eq!(missing, Vec::<String>::new());
456        let mut names: Vec<&str> = TABLE
457            .iter()
458            .flat_map(|key| std::iter::once(key.name).chain(key.also.iter().copied()))
459            .collect();
460        let count = names.len();
461        names.sort_unstable();
462        names.dedup();
463        assert_eq!(names.len(), count, "a name twice");
464    }

466    #[test]
467    fn unset_is_not_set_and_not_unknown() {
468        let walker = cena_map::Walker {
469            settings: TravelSettings::default().for_walker(),
470            ..cena_map::Walker::default()
471        };
472        // An unset switch is not `true`, and an unset word is not set.
473        assert_eq!(
474            Cond::Setting("use_urchins".into(), "true".into()).ask(&walker),
475            Some(false)
476        );
477        assert_eq!(
478            Cond::SettingIsSet("rogue_password".into()).ask(&walker),
479            Some(false)
480        );
481        assert!(Cond::Setting("ice_mode".into(), "auto".into()).holds(&walker));
482    }
483
484    #[test]
485    fn a_switch_is_on_as_the_map_asks_and_by_every_spelling() {
486        let mut settings = TravelSettings::default();
487        assert_eq!(
488            settings.set("use_portmasters", Some("on")),
489            Ok("Use the portmasters is on.".into())
490        );
491        settings
492            .set("annoy_ylandra", Some("yes"))
493            .expect("a switch");
494        settings
495            .set("Cuddfan_Hollow", Some("true"))
496            .expect("a switch");
497        settings.set("keysack", Some("cloak")).expect("words");
498        let walker = cena_map::Walker {
499            settings: settings.for_walker(),
500            ..cena_map::Walker::default()
501        };
502        assert!(Cond::Setting("use_portmasters".into(), "true".into()).holds(&walker));
503        assert!(Cond::Setting("annoy_ylandra".into(), "yes".into()).holds(&walker));
504        assert!(Cond::Setting("Cuddfan_Hollow".into(), "true".into()).holds(&walker));
505        assert!(Cond::SettingIsSet("key_sack".into()).holds(&walker));
506        assert!(Cond::SettingIsSet("keysack".into()).holds(&walker));
507        assert_eq!(
508            settings.0.get("key_sack").map(String::as_str),
509            Some("cloak")
510        );
511    }

513    #[test]
514    fn off_and_unset_are_the_same_and_a_bad_value_is_refused() {
515        let mut settings = TravelSettings::default();
516        settings.set("premium", Some("on")).expect("a switch");
517        settings.set("premium", Some("off")).expect("a switch");
518        assert_eq!(settings, TravelSettings::default());
519        assert!(settings.set("premium", Some("maybe")).is_err());
520        assert!(settings.set("ice_mode", Some("skate")).is_err());
521        assert!(settings.set("typeahead", Some("9")).is_err());
522        assert!(settings.set("nonsense", Some("on")).is_err());
523        assert_eq!(
524            settings.set("ice_mode", Some("WAIT")),
525            Ok("Ice is wait.".into())
526        );
527        assert_eq!(
528            settings.set("ice_mode", None),
529            Ok("Ice is back to auto.".into())
530        );
531    }
